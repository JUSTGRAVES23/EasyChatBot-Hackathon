export async function getProjectControlSnapshot(
  projectId: string,
): Promise<ProjectControlSnapshot> {
  const encodedProjectId = encodeURIComponent(
    projectId,
  );

  const response = await fetch(
    `/project-control/projects/${encodedProjectId}`,
    {
      method: "GET",
      credentials: "include",
      headers: {
        Accept: "application/json",
      },
    },
  );

  if (!response.ok) {
    let detail = `Project Control API returned ${response.status}.`;

    try {
      const body = await response.json();

      if (
        body &&
        typeof body.detail === "string"
      ) {
        detail = body.detail;
      }
    } catch {
      // Preserve the safe generic API error.
    }

    throw new ProjectControlApiError(
      detail,
      response.status,
    );
  }

  return response.json() as Promise<ProjectControlSnapshot>;
}
