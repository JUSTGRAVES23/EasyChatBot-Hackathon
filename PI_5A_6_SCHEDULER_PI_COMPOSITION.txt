"""
Cybersecurity APScheduler Adapter.

Infrastructure boundary for scheduled workflows.

Responsibilities:
- register existing cybersecurity jobs;
- optionally register privacy governance reminders;
- optionally register Academy notification workflows;
- start and stop the single APScheduler instance.

Does NOT:
- own Academy eligibility;
- own Academy delivery state;
- send emails directly;
- send Telegram messages directly;
- make governance decisions;
- create a second scheduler.
"""

from __future__ import annotations

from cybersecurity.scheduler.scheduler_configuration import (
    DAILY_REPORT_HOUR,
    DAILY_REPORT_MINUTE,
    WEEKLY_REPORT_DAY,
    WEEKLY_REPORT_HOUR,
    WEEKLY_REPORT_MINUTE,
)


class CybersecurityAPSchedulerAdapter:
    """APScheduler implementation boundary.

    All scheduled application workflows share this single scheduler.

    Optional workflows remain dependency-injected so existing
    cybersecurity-only tests and deployments remain backwards compatible.
    """

    def __init__(
        self,
        scheduler,
        workflow_service,
        governance_reminder_workflow=None,
        governance_reminder_interval_minutes: int = 60,
        academy_one_week_workflow=None,
        academy_one_week_hour: int = 8,
        academy_one_week_minute: int = 15,
    ) -> None:
        if scheduler is None:
            raise ValueError("scheduler cannot be None.")

        if workflow_service is None:
            raise ValueError("workflow_service cannot be None.")

        if governance_reminder_interval_minutes <= 0:
            raise ValueError(
                "governance_reminder_interval_minutes must be > 0."
            )

        if not 0 <= academy_one_week_hour <= 23:
            raise ValueError(
                "academy_one_week_hour must be between 0 and 23."
            )

        if not 0 <= academy_one_week_minute <= 59:
            raise ValueError(
                "academy_one_week_minute must be between 0 and 59."
            )

        self.scheduler = scheduler
        self.workflow_service = workflow_service

        self.governance_reminder_workflow = (
            governance_reminder_workflow
        )
        self.governance_reminder_interval_minutes = (
            governance_reminder_interval_minutes
        )

        self.academy_one_week_workflow = (
            academy_one_week_workflow
        )
        self.academy_one_week_hour = academy_one_week_hour
        self.academy_one_week_minute = academy_one_week_minute

    def register_jobs(self) -> None:
        """Register all enabled workflows on the single scheduler."""

        self.scheduler.add_job(
            self.workflow_service.create_daily_report,
            trigger="cron",
            hour=DAILY_REPORT_HOUR,
            minute=DAILY_REPORT_MINUTE,
            id="cybersecurity_daily_report",
            replace_existing=True,
        )

        self.scheduler.add_job(
            self.workflow_service.create_weekly_report,
            trigger="cron",
            day_of_week=WEEKLY_REPORT_DAY,
            hour=WEEKLY_REPORT_HOUR,
            minute=WEEKLY_REPORT_MINUTE,
            id="cybersecurity_weekly_report",
            replace_existing=True,
        )

        if self.governance_reminder_workflow is not None:
            self.scheduler.add_job(
                self.governance_reminder_workflow.run,
                trigger="interval",
                minutes=self.governance_reminder_interval_minutes,
                id="privacy_governance_reminder",
                replace_existing=True,
            )

        if self.academy_one_week_workflow is not None:
            self.scheduler.add_job(
                self.academy_one_week_workflow.run,
                trigger="cron",
                hour=self.academy_one_week_hour,
                minute=self.academy_one_week_minute,
                id="academy_one_week_milestone",
                replace_existing=True,
            )

    def register_critical_handler(self, handler) -> None:
        self.critical_handler = handler

    async def trigger_critical_event(self, event: dict):
        if not hasattr(self, "critical_handler"):
            return None

        return await self.critical_handler(event)

    def start(self) -> None:
        if not self.scheduler.running:
            self.scheduler.start()

    def shutdown(self) -> None:
        if self.scheduler.running:
            self.scheduler.shutdown()


__all__ = [
    "CybersecurityAPSchedulerAdapter",
]

"""
Cybersecurity Scheduler Runtime.

Lifecycle manager for cybersecurity scheduling.

Responsibilities:
- start scheduler;
- register jobs;
- stop scheduler;
- expose runtime state.

Does NOT:
- execute reports;
- send emails;
- make security decisions;
- perform remediation.
"""

from __future__ import annotations


from cybersecurity.scheduler.cybersecurity_job_registry import (
    CybersecurityJobRegistry,
)



class CybersecuritySchedulerRuntime:
    """
    Scheduler lifecycle boundary.

    Supports:

    - registry based execution/testing;
    - production scheduler adapter.

    Architecture:

        FastAPI lifespan
              |
              v
    CybersecuritySchedulerRuntime
              |
              +----------------+
              |                |
              v                v

        Job Registry     Scheduler Adapter

          (tests)          (production)

    """



    def __init__(
        self,
        scheduler_adapter=None,
        job_registry: CybersecurityJobRegistry | None = None,
    ) -> None:
        """
        Initialize scheduler runtime.
        """


        self.scheduler_adapter = (
            scheduler_adapter
        )


        self.job_registry = (
            job_registry
            or CybersecurityJobRegistry()
        )


        self._started = False



    async def start(
        self,
    ) -> None:
        """
        Start scheduler runtime.

        Registers scheduled cybersecurity workflows.
        """


        if self.scheduler_adapter is not None:

            #
            # Production scheduler boundary.
            #
            # Example:
            # APScheduler adapter
            #

            self.scheduler_adapter.register_jobs()

            self.scheduler_adapter.start()


        else:

            #
            # Registry mode.
            #
            # Used for testing and controlled execution.
            #

            self.job_registry.register(

                name=(
                    "cybersecurity_daily_report"
                ),

                schedule=(
                    "daily_08:00"
                ),

            )


            self.job_registry.register(

                name=(
                    "cybersecurity_weekly_report"
                ),

                schedule=(
                    "weekly_monday_08:00"
                ),

            )


            self.job_registry.register(

                name=(
                    "cybersecurity_critical_alert"
                ),

                schedule=(
                    "event_immediate"
                ),

            )



        self._started = True



    async def stop(
        self,
    ) -> None:
        """
        Stop scheduler runtime.
        """


        if self.scheduler_adapter is not None:

            self.scheduler_adapter.shutdown()



        self._started = False



    def status(
        self,
    ) -> dict:
        """
        Return scheduler runtime status.

        Provides backward-compatible fields:

        - jobs
        - registered_jobs
        """


        jobs = (
            self.job_registry.list_jobs()
        )



        return {

            "started":
                self._started,


            "adapter_enabled":
                self.scheduler_adapter is not None,


            #
            # Existing test contract
            #
            "jobs":
                jobs,


            #
            # Extended API contract
            #
            "registered_jobs":
                jobs,

        }

from __future__ import annotations



from cybersecurity.control.cybersecurity_delivery_control import (
    CybersecurityDeliveryControl,
)



class CybersecuritySchedulerWorkflow:



    def __init__(
        self,
        report_service=None,
        email_service=None,
        delivery_control=None,
    ) -> None:


        self.report_service = (
            report_service
        )


        self.email_service = (
            email_service
        )


        self.delivery_control = (
            delivery_control
            or CybersecurityDeliveryControl()
        )



    async def create_daily_report(
        self,
    ) -> dict:


        if self.delivery_control.is_paused():

            return {

                "status":
                    "blocked",

                "reason":
                    "cybersecurity_delivery_paused",

            }



        report = {

            "type":
                "daily",

            "status":
                "generated",

        }



        return await self._deliver_email(
            "Daily Cybersecurity Report",
            report,
        )



    async def create_weekly_report(
        self,
    ) -> dict:


        if self.delivery_control.is_paused():

            return {

                "status":
                    "blocked",

                "reason":
                    "cybersecurity_delivery_paused",

            }



        report = {

            "type":
                "weekly",

            "status":
                "generated",

        }



        return await self._deliver_email(
            "Weekly Cybersecurity Report",
            report,
        )



    async def create_critical_alert(
        self,
        event: dict,
    ) -> dict:


        if self.delivery_control.is_paused():

            return {

                "status":
                    "blocked",

                "reason":
                    "cybersecurity_delivery_paused",

            }



        report = {

            "type":
                "critical",

            "event":
                event,

        }



        return await self._deliver_email(
            "Critical Cybersecurity Alert",
            report,
        )



    async def _deliver_email(
        self,
        subject: str,
        report: dict,
    ) -> dict:


        if self.email_service:

            email = await (
                self.email_service.send_report(
                    subject,
                    report,
                )
            )


            return {

                **report,

                "email":
                    email,

            }



        return {

            **report,

            "email":

                {

                    "status":
                        "not_configured"

                },

        }

@dataclass(frozen=True, slots=True)
class PlatformIntelligenceEscalationDeliveryAuditRecord:
    """Immutable append-only audit event for one PI delivery attempt."""

    audit_id: UUID
    notification_key: str
    recommendation_id: str
    risk_decision_id: str
    review_item_id: str
    correlation_id: str
    attempt: int
    status: PlatformIntelligenceEscalationDeliveryAuditStatus
    provider_message_id: str | None = None
    error_code: str | None = None
    occurred_at: datetime | None = None

    def __post_init__(self) -> None:
        for name in (
            "notification_key", "recommendation_id", "risk_decision_id",
            "review_item_id", "correlation_id",
        ):
            value = getattr(self, name)
            if not isinstance(value, str) or not value.strip():
                raise ValueError(f"{name} must be nonblank.")
        if self.attempt < 1:
            raise ValueError("attempt must be at least 1.")
        if self.occurred_at is not None and (

            raise ValueError("occurred_at must be timezone-aware.")
        if (
            self.status is PlatformIntelligenceEscalationDeliveryAuditStatus.SUCCEEDED
            and not self.provider_message_id
        ):
            raise ValueError("SUCCEEDED audit events require provider_message_id.")
        if (
            self.status is PlatformIntelligenceEscalationDeliveryAuditStatus.FAILED
            and not self.error_code
        ):
            raise ValueError("FAILED audit events require error_code.")
        if (
            self.status is PlatformIntelligenceEscalationDeliveryAuditStatus.SUCCEEDED
            and self.error_code is not None
        ):

from __future__ import annotations

import inspect
from dataclasses import dataclass
from enum import StrEnum
from typing import Any

from app.application.operations_center.platform_intelligence_escalation_delivery import (
    PlatformIntelligenceEscalationDeliveryStatus,

def _build_notification_key(recommendation_id: str) -> str:
    """Build the canonical notification key for one recommendation."""
    recommendation_id = recommendation_id.strip()

    if not recommendation_id:
        raise ValueError(
            "recommendation_id cannot be blank."
        )

    return (
        "platform_intelligence_escalation:"
        f"{recommendation_id}"
    )


class PlatformIntelligenceEscalationDispatchStatus(StrEnum):
    SENT = "SENT"
    IDEMPOTENT_REPLAY = "IDEMPOTENT_REPLAY"
    FAILED = "FAILED"


@dataclass(frozen=True, slots=True)
class PlatformIntelligenceEscalationDispatchRequest:
    """Authoritative state required before an escalation can be dispatched."""

    recommendation_id: str
    risk_decision_id: str
    review_item_id: str
    risk_level: str
    risk_score: float
    quality_gate_status: str
    review_status: str
    decision_status: str
    notification_key: str
    correlation_id: str
    route: str
    reasons: tuple[str, ...] = ()

    def __post_init__(self) -> None:
        required = (
            self.recommendation_id,
            self.risk_decision_id,
            self.review_item_id,
            self.notification_key,
            self.correlation_id,
            self.route,
            self.risk_level,
            self.quality_gate_status,
            self.review_status,
            self.decision_status,
        )

        if any(
            not isinstance(value, str) or not value.strip()
            for value in required
        ):
            raise ValueError(
                "Required Platform Intelligence dispatch fields "
                "cannot be blank."
            )

        if not 0.0 <= self.risk_score <= 1.0:
            raise ValueError(
                "risk_score must be between 0 and 1."
            )

        if any(
            not isinstance(reason, str) or not reason.strip()
            for reason in self.reasons
        ):
            raise ValueError(
                "Dispatch reasons cannot contain blank values."
            )

        expected_notification_key = _build_notification_key(
            self.recommendation_id
        )

        if self.notification_key != expected_notification_key:
            raise ValueError(
                "notification_key must match recommendation_id."
            )


@dataclass(frozen=True, slots=True)
class PlatformIntelligenceEscalationDispatchResult:
    status: PlatformIntelligenceEscalationDispatchStatus
    notification_key: str
    provider_message_id: str | None = None
    error_code: str | None = None
    attempt: int | None = None


class PlatformIntelligenceEscalationDispatchService:
    """Revalidate, deduplicate and dispatch one governed escalation."""

    def __init__(
        self,
        *,
        delivery_repository: Any,
        email_workflow: Any,
    ) -> None:
        if delivery_repository is None:
            raise ValueError(
                "delivery_repository cannot be None."
            )

        if email_workflow is None:
            raise ValueError(
                "email_workflow cannot be None."
            )

        self._repository = delivery_repository
        self._email_workflow = email_workflow

    @staticmethod
    def build_notification_key(
        recommendation_id: str,
    ) -> str:
        """Return the canonical recommendation-scoped notification key."""
        return _build_notification_key(recommendation_id)

    async def dispatch(
        self,
        *,
        request: PlatformIntelligenceEscalationDispatchRequest,
        email_request: Any,
    ) -> PlatformIntelligenceEscalationDispatchResult:
        self._revalidate(request)

        existing = await self._repository.get_by_notification_key(
            request.notification_key
        )

        if (
            existing is not None
            and existing.status
            is PlatformIntelligenceEscalationDeliveryStatus.SUCCEEDED
        ):
            return PlatformIntelligenceEscalationDispatchResult(
                status=(
                    PlatformIntelligenceEscalationDispatchStatus
                    .IDEMPOTENT_REPLAY
                ),
                notification_key=existing.notification_key,
                provider_message_id=(
                    existing.provider_message_id
                ),
                attempt=existing.attempt,
            )

        if existing is None:
            try:
                await self._repository.create_started(
                    notification_key=request.notification_key,
                    recommendation_id=request.recommendation_id,
                    risk_decision_id=request.risk_decision_id,
                    review_item_id=request.review_item_id,
                    correlation_id=request.correlation_id,
                )
            except Exception:
                # The durable repository owns the unique-key race.
                # If another worker created the record first, honour
                # the persisted state.
                existing = (
                    await self._repository
                    .get_by_notification_key(
                        request.notification_key
                    )
                )

                if existing is None:
                    raise

        elif (
            existing.status
            is PlatformIntelligenceEscalationDeliveryStatus.FAILED
        ):
            await self._repository.start_retry(
                notification_key=request.notification_key
            )

        elif (
            existing.status
            is PlatformIntelligenceEscalationDeliveryStatus.STARTED
        ):
            # An interrupted attempt is retryable, following the
            # established durable execution convention.
            pass

        try:
            workflow_result = self._email_workflow.send(
                email_request
            )

            if inspect.isawaitable(workflow_result):
                workflow_result = await workflow_result

            # The current PI email workflow may return its request
            # containing an awaitable `delivery` result.
            #
            # Resolve it before marking the durable delivery as
            # successful so that invocation of the workflow itself
            # cannot be mistaken for successful provider delivery.
            delivery = getattr(
                workflow_result,
                "delivery",
                None,
            )

            if inspect.isawaitable(delivery):

class PlatformIntelligenceEscalationRetryPolicyService:
    """Evaluate whether a durable PI delivery record may be retried.

    This service is side-effect free.

    It does not:
    - mutate delivery state;
    - call an email provider;
    - schedule work;
    - read the wall clock.

    The caller supplies the reference time used for the eligibility decision.
    The delivery record supplies the authoritative timestamp from which the
    retry backoff is calculated.
    """

    def __init__(
        self,
        *,
        policy: PlatformIntelligenceEscalationRetryPolicy,
    ) -> None:
        if policy is None:
            raise ValueError("policy cannot be None.")

        self._policy = policy

    @property
    def policy(self) -> PlatformIntelligenceEscalationRetryPolicy:
        return self._policy

    def evaluate(
        self,
        *,
        record: PlatformIntelligenceEscalationDeliveryRecord,
        reference_at: datetime,
    ) -> PlatformIntelligenceEscalationRetryResult:
        self._validate_reference_time(reference_at)

        if not record.notification_key.strip():
            raise ValueError(
                "notification_key cannot be blank."
            )

        if record.attempt < 1:
            raise ValueError(
                "delivery attempt must be at least 1."
            )

        if (
            self._policy.status
            is not PlatformIntelligenceEscalationRetryPolicyStatus.APPROVED
        ):
            return self._result(
                record=record,
                decision=(
                    PlatformIntelligenceEscalationRetryDecision
                    .NOT_ELIGIBLE
                ),
                reason="Retry policy is not APPROVED.",
            )

        if (
            record.status
            is PlatformIntelligenceEscalationDeliveryStatus.SUCCEEDED
        ):
            return self._result(
                record=record,
                decision=(
                    PlatformIntelligenceEscalationRetryDecision
                    .NOT_ELIGIBLE
                ),
                reason="Delivery already SUCCEEDED.",
            )

        if record.attempt >= self._policy.max_attempts:
            return self._result(
                record=record,
                decision=(
                    PlatformIntelligenceEscalationRetryDecision
                    .NOT_ELIGIBLE
                ),
                reason="Maximum delivery attempts reached.",
            )

        if (
            record.status
            is PlatformIntelligenceEscalationDeliveryStatus.FAILED
        ):
            if not record.error_code:
                return self._result(
                    record=record,
                    decision=(
                        PlatformIntelligenceEscalationRetryDecision
                        .NOT_ELIGIBLE
                    ),
                    reason=(
                        "FAILED delivery has no error code for retry "
                        "classification."
                    ),
                )

            if (
                record.error_code
                not in self._policy.retryable_error_codes
            ):
                return self._result(
                    record=record,
                    decision=(
                        PlatformIntelligenceEscalationRetryDecision
                        .NOT_ELIGIBLE
                    ),
                    reason=(
                        f"Error code {record.error_code!r} "
                        "is not retryable."
                    ),
                )

            return self._eligible_result(
                record=record,
                reference_at=reference_at,
            )

        if (
            record.status
            is PlatformIntelligenceEscalationDeliveryStatus.STARTED
        ):
            if not self._policy.retry_interrupted_started:
                return self._result(
                    record=record,
                    decision=(
                        PlatformIntelligenceEscalationRetryDecision
                        .NOT_ELIGIBLE
                    ),
                    reason=(
                        "Interrupted STARTED deliveries are not "
                        "retryable by policy."
                    ),
                )

            # STARTED represents recovery of an interrupted attempt.
            # Recovery does NOT increment the attempt, but it remains
            # subject to the same policy-controlled backoff.
            return self._eligible_result(
                record=record,
                reference_at=reference_at,
            )

        return self._result(
            record=record,
            decision=(
                PlatformIntelligenceEscalationRetryDecision
                .NOT_ELIGIBLE
            ),
            reason=(
                f"Delivery status {record.status.value!r} "
                "is not retryable."
            ),
        )

    def backoff_seconds(self, *, attempt: int) -> int:
        """Return deterministic exponential backoff for the retry.

        Attempt 1:
            base_backoff_seconds

        Attempt 2:
            base_backoff_seconds * 2

        Attempt 3:
            base_backoff_seconds * 4

        The result is capped at max_backoff_seconds.
        """
        if attempt < 1:
            raise ValueError(
                "attempt must be at least 1."
            )

        multiplier = 2 ** (attempt - 1)

        return min(
            self._policy.base_backoff_seconds * multiplier,
            self._policy.max_backoff_seconds,
        )

    def _eligible_result(
        self,
        *,
        record: PlatformIntelligenceEscalationDeliveryRecord,
        reference_at: datetime,
    ) -> PlatformIntelligenceEscalationRetryResult:
        delay = self.backoff_seconds(
            attempt=record.attempt
        )

        # The backoff is calculated from the persisted delivery timestamp,
        # not from the caller's current reference time.
        #
        # This is essential for deterministic due/not-due evaluation.
        if record.updated_at is None:
            return self._result(
                record=record,
                decision=(
                    PlatformIntelligenceEscalationRetryDecision
                    .NOT_ELIGIBLE
                ),
                reason=(
                    "Delivery record has no updated_at timestamp "
                    "from which retry timing can be established."
                ),
            )

        self._validate_timestamp(
            record.updated_at,
            field_name="record.updated_at",
        )

        next_eligible_at = (
            record.updated_at
            + timedelta(seconds=delay)
        )

        if reference_at < next_eligible_at:
            return self._result(
                record=record,
                decision=(
                    PlatformIntelligenceEscalationRetryDecision
                    .ELIGIBLE
                ),
                reason=(
                    "Delivery is eligible but the policy-derived "
                    "backoff window has not elapsed."
                ),
                next_eligible_at=next_eligible_at,
                backoff_seconds=delay,
            )

        return self._result(
            record=record,
            decision=(
                PlatformIntelligenceEscalationRetryDecision
                .ELIGIBLE
            ),
            reason=(
                "Delivery is eligible for a policy-controlled retry."
            ),
            next_eligible_at=next_eligible_at,
            backoff_seconds=delay,
        )

    def _result(
        self,
        *,
        record: PlatformIntelligenceEscalationDeliveryRecord,
        decision: PlatformIntelligenceEscalationRetryDecision,
        reason: str,
        next_eligible_at: datetime | None = None,
        backoff_seconds: int | None = None,
    ) -> PlatformIntelligenceEscalationRetryResult:
        return PlatformIntelligenceEscalationRetryResult(
            decision=decision,
            notification_key=record.notification_key,
            attempt=record.attempt,
            policy_id=self._policy.policy_id,
            policy_version=self._policy.policy_version,
            reason=reason,
            next_eligible_at=next_eligible_at,
            backoff_seconds=backoff_seconds,
        )

    @staticmethod
    def _validate_reference_time(
        reference_at: datetime,
    ) -> None:
        if (
            reference_at.tzinfo is None
            or reference_at.utcoffset() is None
        ):
            raise ValueError(
                "reference_at must be timezone-aware."
            )

    @staticmethod
    def _validate_timestamp(
        value: datetime,
        *,
        field_name: str,
    ) -> None:
        if value.tzinfo is None or value.utcoffset() is None:
            raise ValueError(
                f"{field_name} must be timezone-aware."
            )

class PlatformIntelligenceEscalationRetryExecutionService:
    """Execute one policy-authorized retry without owning scheduling.

    This service:
    - retrieves the authoritative durable delivery record;
    - evaluates retry policy;
    - enforces the policy-derived due time;
    - increments FAILED attempts only through the repository;
    - preserves the attempt number for STARTED recovery;
    - delegates actual delivery to the existing dispatch service.

    It does not:
    - schedule future work;
    - sleep;
    - call an email provider directly;
    - mutate delivery state directly.
    """

    def __init__(
        self,
        *,
        delivery_repository: object,
        retry_policy: PlatformIntelligenceEscalationRetryPolicyService,
        dispatch_service: PlatformIntelligenceEscalationDispatchService,
    ) -> None:
        if delivery_repository is None:
            raise ValueError(
                "delivery_repository cannot be None."
            )

        if retry_policy is None:
            raise ValueError(
                "retry_policy cannot be None."
            )

        if dispatch_service is None:
            raise ValueError(
                "dispatch_service cannot be None."
            )

        self._repository = delivery_repository
        self._retry_policy = retry_policy
        self._dispatch_service = dispatch_service

    async def retry(
        self,
        *,
        request: PlatformIntelligenceEscalationDispatchRequest,
        email_request: object,
        reference_at: datetime,
    ) -> PlatformIntelligenceEscalationRetryExecutionResult:
        record = await self._repository.get_by_notification_key(
            request.notification_key
        )

        if record is None:
            raise ValueError(
                "Retry requires an existing durable delivery record."
            )

        if record.notification_key != request.notification_key:
            raise ValueError(
                "Delivery record notification key does not match request."
            )

        policy_result = self._retry_policy.evaluate(
            record=record,
            reference_at=reference_at,
        )

        if (
            policy_result.decision
            is PlatformIntelligenceEscalationRetryDecision.NOT_ELIGIBLE
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_ELIGIBLE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=policy_result.reason,
            )

        if (
            policy_result.next_eligible_at is not None
            and reference_at < policy_result.next_eligible_at
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_DUE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=(
                    "Retry is eligible but the policy-derived "
                    "backoff window has not elapsed."
                ),
            )

        if (
            record.status
            is PlatformIntelligenceEscalationDeliveryStatus.FAILED
        ):
            await self._repository.start_retry(
                notification_key=record.notification_key,
            )

        # STARTED recovery deliberately does not call start_retry().
        # The durable attempt remains the same because this is recovery
        # of the interrupted attempt rather than a new FAILED retry.
        dispatch_result = await self._dispatch_service.dispatch(
            request=request,
            email_request=email_request,
        )

        return PlatformIntelligenceEscalationRetryExecutionResult(
            status=(
                PlatformIntelligenceEscalationRetryExecutionStatus
                .RETRIED
            ),
            notification_key=record.notification_key,
            attempt=(
                dispatch_result.attempt
                if dispatch_result.attempt is not None
                else record.attempt
            ),
            reason=policy_result.reason,
            dispatch_result=dispatch_result,
        )

    """Coordinates policy, durable claim, state revalidation and retry.

    This service deliberately does not discover candidates, implement retry
    policy, send email directly, or mutate human-review/risk state.

    The canonical retry execution service remains the owner of retry-policy
    enforcement and existing dispatch integration. The orchestrator only
    supplies the authoritative dispatch context after the durable claim and
    final state revalidation.
    """

    def __init__(
        self,
        *,
        state_reader: PlatformIntelligenceRetryStateReader,
        policy_evaluator: PlatformIntelligenceRetryPolicyEvaluator,
        claim_service: PlatformIntelligenceEscalationRetryClaimService,
        retry_execution: PlatformIntelligenceRetryExecutor,
    ) -> None:
        self._state_reader = state_reader
        self._policy_evaluator = policy_evaluator
        self._claim_service = claim_service
        self._retry_execution = retry_execution

    async def run_candidate(
        self,
        *,
        candidate: PlatformIntelligenceEscalationRetryCandidate,
        reference_at: datetime,
    ) -> PlatformIntelligenceRetryRunResult:
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("reference_at must be timezone-aware.")

        eligible = await self._policy_evaluator.evaluate(
            candidate=candidate,
            reference_at=reference_at,
        )
        if not eligible:
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason="RETRY_POLICY_NOT_ELIGIBLE",
                attempt=candidate.attempt,
            )

        claim = await self._claim_service.claim(
            notification_key=candidate.notification_key,
            claimed_at=reference_at,
        )
        if claim is None:
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason="RETRY_CANDIDATE_ALREADY_CLAIMED",
                attempt=candidate.attempt,
            )

            notification_key=candidate.notification_key,
            recommendation_id=candidate.recommendation_id,
        )
        if state is None:
            return self._skipped(candidate, "AUTHORITATIVE_STATE_NOT_FOUND")
        if state.notification_key != candidate.notification_key:
            return self._skipped(candidate, "NOTIFICATION_KEY_MISMATCH")
        if state.recommendation_id != candidate.recommendation_id:
            return self._skipped(candidate, "RECOMMENDATION_ID_MISMATCH")
        if state.recommendation_status != "RECOMMENDED":
            return self._skipped(

            )

            email_request=state.email_request,

class PlatformIntelligenceRetryWorker:
    """Process one bounded candidate batch through the governed retry chain.

    This class deliberately does not:
    - schedule itself;
    - sleep;
    - create background tasks;
    - access providers;
    - implement retry policy;
    - claim candidates directly;
    - mutate risk or human-review state.

    The caller owns invocation timing and supplies the reference timestamp.
    """

    def __init__(
        self,
        *,
        discovery_service: PlatformIntelligenceEscalationRetryCandidateDiscoveryService,
        orchestrator: PlatformIntelligenceEscalationRetryOrchestrator,
    ) -> None:
        if discovery_service is None:
            raise ValueError("discovery_service cannot be None.")

        if orchestrator is None:
            raise ValueError("orchestrator cannot be None.")

        self._discovery_service = discovery_service
        self._orchestrator = orchestrator

    async def run(
        self,
        *,
        request: PlatformIntelligenceRetryWorkerRequest,
        reference_at: datetime,
    ) -> PlatformIntelligenceRetryWorkerResult:
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("reference_at must be timezone-aware.")

        batch = await self._discovery_service.discover(
            limit=request.limit,
        )

        results: list[PlatformIntelligenceRetryWorkerItemResult] = []

        for candidate in batch.candidates:
            try:
                execution = await self._orchestrator.run_candidate(
                    candidate=candidate,
                    reference_at=reference_at,
                )

            except Exception as exc:
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=candidate.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .FAILED
                        ),
                        reason="CANDIDATE_EXECUTION_ERROR",
                        attempt=candidate.attempt,
                        error_type=type(exc).__name__,
                    )
                )
                continue

            if (
                execution.status
                is PlatformIntelligenceRetryRunStatus.SUCCEEDED
            ):
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .SUCCEEDED
                        ),
                        reason=execution.reason,
                        attempt=execution.attempt,
                        provider_message_id=(
                            execution.provider_message_id
                        ),
                    )
                )

            elif (
                execution.status
                is PlatformIntelligenceRetryRunStatus.SKIPPED
            ):
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(

from cybersecurity.email.resend_email_provider import ResendEmailProvider


class ResendPlatformIntelligenceEscalationEmailSender:
    """Translate governed PI email requests into the shared email provider."""

    def __init__(
        self,
        *,
        provider_manager: EmailProviderManager | None = None,
        recipients: dict[str, str] | None = None,
    ) -> None:
        self._recipients = dict(
            recipients
            if recipients is not None
            else PLATFORM_INTELLIGENCE_ESCALATION_RECIPIENTS
        )

        self._provider_manager = (
            provider_manager
            if provider_manager is not None
            else self._build_provider_manager()
        )

    @staticmethod
    def _build_provider_manager() -> EmailProviderManager:
        sender = PLATFORM_INTELLIGENCE_EMAIL_FROM.strip()

        if not sender:
            raise RuntimeError(
                "PLATFORM_INTELLIGENCE_EMAIL_FROM is not configured."
            )

        return EmailProviderManager(
            provider=ResendEmailProvider(
                sender=sender,
            ),
        )

    def send(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
        *,
        subject: str,
        body: str,
        html_body: str,
    ) -> object:
        if request is None:
            raise ValueError("request cannot be None.")

        recipient = self._recipients.get(request.route)

        if not recipient or not recipient.strip():
            raise RuntimeError(
                "No configured recipient for escalation route: "
                f"{request.route}"
            )

        if not subject.strip():
            raise ValueError("Email subject cannot be empty.")

        if not body.strip():
            raise ValueError("Email body cannot be empty.")

        if not html_body.strip():
            raise ValueError("HTML email body cannot be empty.")

        content: dict[str, Any] = {
            "recipient": recipient.strip(),
            "body": body,
            "html": html_body,
            "email_type": "platform_intelligence_escalation",
            "recommendation_id": request.recommendation_id,
            "risk_decision_id": request.risk_decision_id,
            "review_item_id": request.review_item_id,
            "escalation_route": request.route,
        }

        return self._provider_manager.send(
            subject.strip(),
            content,
        )
