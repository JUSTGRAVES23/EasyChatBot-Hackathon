def _create_announcement_components():
    telegram_client = TelegramAnnouncementClient()
    announcement_publisher = TelegramAnnouncementPublisher(
        telegram_client=telegram_client,
    )
    return telegram_client, announcement_publisher


@asynccontextmanager
async def lifespan(application: FastAPI):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print("Cybersecurity scheduler startup failed:", exc)

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print("Cybersecurity scheduler shutdown failed:", exc)
    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )
        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(application.state, "auth_engine", None)
        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )
        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )


def create_application() -> FastAPI:
    academy_config = AcademyActivationConfig.from_environment()

    auth_engine = create_async_database_engine()
    auth_session_factory = create_async_session_factory(auth_engine)
    auth_repository = PostgreSQLAuthenticationRepository(
        auth_session_factory,
    )

    # Provider-usage repository is synchronous by contract. It therefore
    # receives a synchronous engine/session factory. It must not receive
    # the async authentication session factory.
    provider_usage_engine = create_database_engine()
    provider_usage_session_factory = create_session_factory(
        provider_usage_engine,
    )
    provider_usage_repository = PostgreSQLProviderUsageRepository(
        provider_usage_session_factory,
    )
    provider_usage_dashboard_service = ProviderUsageDashboardService(
        repository=provider_usage_repository,
    )

    functionality_control_repository = PostgreSQLFunctionalityControlRepository(
        provider_usage_session_factory,
    )
    functionality_control_authorizer = AuthorizationFunctionalityControlAuthorizer()
    functionality_control_service = FunctionalityOperationalControlService(
        authorizer=functionality_control_authorizer,
        repository=functionality_control_repository,
    )

    academy_one_week_workflow = None
    announcement_telegram_client = None

    if academy_config.activation_allowed:
        academy_session_factory = auth_session_factory
        (
            announcement_telegram_client,
            _announcement_publisher,
        ) = _create_announcement_components()

        academy_one_week_workflow = create_academy_one_week_workflow(
            session_factory=academy_session_factory,
            telegram_bot=announcement_telegram_client,
        )

    scheduler_runtime = create_cybersecurity_scheduler_runtime(
        academy_one_week_workflow=academy_one_week_workflow,
    )

    application = FastAPI(
        title="EasyChatBot API",
        description="AI character learning platform API",
        version="0.1.0",
        lifespan=lifespan,
    )

    application.state.scheduler_runtime = scheduler_runtime
    application.state.auth_repository = auth_repository
    application.state.auth_engine = auth_engine
    application.state.provider_usage_engine = provider_usage_engine
    application.state.provider_usage_dashboard_service = (
        provider_usage_dashboard_service
    )
    application.state.functionality_control_service = (
        functionality_control_service
    )
    application.state.announcement_telegram_client = (
        announcement_telegram_client
    )

    application.add_middleware(
        CORSMiddleware,
        allow_origins=[
            "http://localhost:3000",
            "http://127.0.0.1:3000",
            "http://localhost:5173",
            "http://127.0.0.1:5173",
        ],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    application.include_router(authentication_router)
    application.include_router(voice_calls_router)
    application.include_router(voice_reconnect_router)
    application.include_router(voice_call_trigger_router)
    application.include_router(voice_notification_acceptance_router)

    operations_center_composition = create_operations_center_composition(
        provider_usage_dashboard_service=provider_usage_dashboard_service,
        functionality_control_service=functionality_control_service,
    )
    application.include_router(
        create_operations_center_router(
            composition=operations_center_composition,
        ),
    )

    application.include_router(create_email_workflows_router())
    application.include_router(privacy_governance_router)

    processing_registry = create_beta1_processing_registry()
    screening_service = ScreeningService()
    screening_adapter = ProcessingActivityScreeningAdapter(screening_service)
    application.include_router(
        create_privacy_screening_router(
            processing_registry=processing_registry,
            screening_adapter=screening_adapter,
        ),
    )

    application.include_router(provider_management_controls_router)
    application.include_router(governance_audit_router)
    application.include_router(governance_operations_router)
    application.include_router(governance_evidence_router)
    application.include_router(governance_change_lifecycle_router)
    application.include_router(cybersecurity_intelligence_router)
    application.include_router(cybersecurity_decision_support_router)
    application.include_router(cybersecurity_reports_router)
    application.include_router(cybersecurity_live_reports_router)

    return application


app = create_application()

class FunctionalityStopRequest(BaseModel):
    environment: OperationsEnvironment
    reason: str = Field(min_length=1, max_length=2000)


class FunctionalityResumeRequest(BaseModel):
    environment: OperationsEnvironment


def _control_unavailable() -> HTTPException:
    return HTTPException(
        status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
        detail="Functionality operational control is not configured.",
    )


def _result_as_dict(result: Any) -> dict[str, Any]:
    return {
        "functionality_id": result.functionality_id,
        "environment": result.environment,
        "availability": result.availability,
        "action": result.action,
        "reason": result.reason,
        "changed_at": result.changed_at.isoformat(),
        "changed_by": result.changed_by,
    }


def create_operations_center_router(
    *,
    composition: OperationsCenterComposition | None = None,
) -> APIRouter:
    resolved_composition = composition or create_operations_center_composition()

    dashboard_api = resolved_composition.dashboard_api
    chat_operations_api = resolved_composition.chat_operations_api
    functionality_health_service = resolved_composition.functionality_health_service
    functionality_control_service = (
        resolved_composition.functionality_control_service
    )
    event_service = resolved_composition.event_service
    provider_usage_dashboard_service = (
        resolved_composition.provider_usage_dashboard_service
    )

    router = APIRouter(
        prefix="/api/operations-center",
        tags=["operations-center"],
    )

    @router.get("/dashboard")
    async def get_dashboard() -> dict[str, Any]:
        return await dashboard_api.get_dashboard()

    @router.get("/chat-operations")
    async def get_chat_operations_dashboard() -> dict[str, Any]:
        return await chat_operations_api.get_dashboard()

    @router.get("/chat-operations/private-classroom")
    async def get_private_classroom_operations() -> dict[str, Any]:
        return await chat_operations_api.get_private_classroom_details()

    @router.get("/chat-operations/announcement-group")
    async def get_announcement_group_operations() -> dict[str, Any]:
        return await chat_operations_api.get_announcement_group_details()

    @router.get("/functionality-health")
    async def get_functionality_health(
        environment: OperationsEnvironment | None = None,
    ) -> dict[str, Any]:
        return await functionality_health_service.get_health(
            environment=environment
        )

    @router.post(
        "/functionality/{functionality_id}/stop",
        dependencies=[Depends(require_permission(Permission.FUNCTIONALITY_CONTROL))],
    )
    def stop_functionality(
        functionality_id: str,
        request: FunctionalityStopRequest,
        principal: Principal = Depends(
            require_permission(Permission.FUNCTIONALITY_CONTROL)
        ),
    ) -> dict[str, Any]:
        if functionality_control_service is None:
            raise _control_unavailable()

        try:
            result = functionality_control_service.stop(
                actor=principal,
                actor_name=principal.user_id,
                environment=request.environment.value,
                functionality_id=functionality_id,
                reason=request.reason,
            )
        except ValueError as exc:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=str(exc),
            ) from exc
        except PermissionError:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Insufficient permission.",
            ) from None

        return _result_as_dict(result)

    @router.post(
        "/functionality/{functionality_id}/resume",
        dependencies=[Depends(require_permission(Permission.FUNCTIONALITY_CONTROL))],
    )
    def resume_functionality(
        functionality_id: str,
        request: FunctionalityResumeRequest,
        principal: Principal = Depends(
            require_permission(Permission.FUNCTIONALITY_CONTROL)
        ),
    ) -> dict[str, Any]:
        if functionality_control_service is None:
            raise _control_unavailable()

        try:
            result = functionality_control_service.resume(
                actor=principal,
                actor_name=principal.user_id,
                environment=request.environment.value,
                functionality_id=functionality_id,
            )
        except ValueError as exc:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=str(exc),
            ) from exc
        except PermissionError:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Insufficient permission.",
            ) from None

        return _result_as_dict(result)

    @router.get("/operational-events")
    def get_operational_events() -> list[dict[str, Any]]:
        incidents = event_service.list_incidents()
        return [
            event.as_dict()
            for incident in incidents
            for event in incident.events
        ]

    @router.get("/provider-usage")
    def get_provider_usage() -> dict[str, Any]:
        if provider_usage_dashboard_service is None:
            return {
                "status": "not_configured",
                "reason": "provider_usage_dashboard_service_not_configured",
            }
        return provider_usage_dashboard_service.get_snapshot()

    return router


router = create_operations_center_router()

def _require_decision_role(
    role: str,
) -> str:
    """Validate the explicitly submitted governance role."""

    normalized = role.strip().upper()

    if normalized not in {
        "DPO",
        "GRC",
    }:
        raise HTTPException(
            status_code=400,
            detail="role must be DPO or GRC.",
        )

    return normalized


def _decision_response(
    *,
    case_id: str,
    record: object,
) -> dict[str, object]:
    """Return decision evidence, refreshed case state, and Excel sync status."""

    dashboard = _dashboard_api()

    try:
        case = dashboard.get_case_detail(
            case_id,
        )

    except KeyError as exc:
        raise HTTPException(
            status_code=404,
            detail=str(exc),
        ) from exc

    evidence = _evidence_sync().sync_decision(
        case_id,
    )

    return {
        "case": case,
        "approval_record": jsonable_encoder(
            record,
        ),
        "excel_sync": {
            "status": evidence.status,
            "workbook_path": evidence.workbook_path,
            "error": evidence.error,
        },
    }

@router.post(
    "/demo/create-case",
)
def create_demo_case(
    request: PrivacyGovernanceDemoCaseRequest | None = None,
) -> dict[str, object]:
    """Create a controlled Beta demonstration governance case.

    Development/demo only.

    Production case creation must use the governed application workflow.
    """

    composition = _composition()

    demo_request = (
        request
        if request is not None
        else PrivacyGovernanceDemoCaseRequest()
    )

    case_id = demo_request.case_id.strip()
    change_id = demo_request.change_id.strip()

    if not case_id:
        raise HTTPException(
            status_code=400,
            detail="case_id cannot be blank.",
        )

    if not change_id:
        raise HTTPException(
            status_code=400,
            detail="change_id cannot be blank.",
        )

    change = PrivacyChange(
        change_id=change_id,
        title="New learner data processing",
        description=(
            "Beta demonstration of a new privacy-impacting "
            "learner processing activity."
        ),
        risk=PrivacyChangeRisk.MEDIUM,
        dpo_required=True,
        grc_required=True,
    )

    try:
        result = composition.flow_service.process_new_processing(
            change,
            automated_validation_passed=True,
            case_id=case_id,
            trigger_source="Processing Registry",
        )

    except KeyError as exc:
        raise HTTPException(
            status_code=404,
            detail=str(exc),
        ) from exc

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    evidence = _evidence_sync().sync_case(
        result.case.case_id,
    )

    return {

===== PCD COMPOSITION =====
"""
Project Control Center production composition.

This module assembles the Project Control Center application service
from authoritative application-layer dependencies.

Responsibilities:

- construct the approved PCD source adapters;
- inject authoritative application services/registries;
- construct ProjectControlService.

This module does NOT:

- own authoritative project state;
- duplicate Health state;
- duplicate Operations state;
- duplicate Privacy Governance state;
- call providers directly;
- mutate production systems;
- execute project-management decisions.

Architecture:

    Authoritative Application Services / Registries
                         |
                         v
                 PCD Source Adapters
                         |
                         v
                ProjectControlService
                         |
                         v
                ProjectControlSnapshot
"""

from __future__ import annotations

from app.application.health.operational_dashboard_service import (
    OperationalDashboardService,
)
from app.application.health.production_health_service import (
    ProductionHealthService,
)
from app.application.privacy.privacy_governance_case_registry import (
    PrivacyGovernanceCaseRegistry,

from ..application.service import ProjectControlService
from ..domain.ports import ProjectRegistry, SnapshotRepository
from .governance_risk_source import (
    GovernanceRiskProjectStatusSource,
)
from .health_source import (
    ProductionHealthProjectStatusSource,
)
from .operations_source import (
    OperationsProjectStatusSource,
)


def create_project_control_service(
    *,
    registry: ProjectRegistry,
    snapshots: SnapshotRepository,
    health_service: ProductionHealthService,
    dashboard_service: OperationalDashboardService,
    case_registry: PrivacyGovernanceCaseRegistry,
) -> ProjectControlService:
    """
    Create the production ProjectControlService.

    All dependencies are supplied explicitly.

    The function constructs only the PCD read-only adapters. The
    authoritative application services and registries remain owned
    by their respective bounded contexts.

    Args:
        registry:
            Authoritative PCD project registry.

        snapshots:
            PCD snapshot repository.

        health_service:
            Existing ProductionHealthService.

        dashboard_service:
            Existing OperationalDashboardService.

        case_registry:
            Existing PrivacyGovernanceCaseRegistry.

    Returns:
        Fully composed ProjectControlService.

    Raises:
        ValueError:
            If a required dependency is None.
    """

    if registry is None:
        raise ValueError(
            "registry cannot be None."
        )

    if snapshots is None:
        raise ValueError(
            "snapshots cannot be None."
        )

    if health_service is None:
        raise ValueError(
            "health_service cannot be None."
        )

    if dashboard_service is None:
        raise ValueError(
            "dashboard_service cannot be None."
        )

    if case_registry is None:
        raise ValueError(
            "case_registry cannot be None."
        )

    health_source = ProductionHealthProjectStatusSource(
        health_service=health_service,
    )

    operations_source = OperationsProjectStatusSource(
        dashboard_service=dashboard_service,
    )

    governance_source = GovernanceRiskProjectStatusSource(
        case_registry=case_registry,
    )

    return ProjectControlService(
        registry=registry,
        sources=(
            health_source,
            operations_source,
            governance_source,
        ),
        snapshots=snapshots,
    )
