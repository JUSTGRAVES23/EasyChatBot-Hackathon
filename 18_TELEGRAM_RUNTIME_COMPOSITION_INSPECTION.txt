from __future__ import annotations

import logging
from typing import Callable, Awaitable

from app.application.beta_access.beta_access_service import BetaAccessService

logger = logging.getLogger(__name__)


class BetaTelegramStartHandler:
    """Adapter for /start <token>. Only private chats may bind Beta users."""

    def __init__(
        self,
        *,
        beta_access_service: BetaAccessService,
        reply: Callable[[str, object], Awaitable[None]],
    ) -> None:
        self._access = beta_access_service
        self._reply = reply

    async def handle(
        self,
        *,
        token: str | None,
        telegram_user_id: int,
        chat_id: int,
        chat_type: str,
    ) -> None:
        if chat_type != "private":
            await self._reply(
                "Please open a private chat with the Academy bot to connect your classroom.",
                None,
            )
            return

        if not token:
            await self._reply(
                "Your classroom connection link is missing or invalid.",
                None,
            )
            return

        try:
            user = await self._access.validate_and_bind(
                token=token,
                telegram_id=telegram_user_id,
                chat_id=chat_id,
            )
        except (ValueError, PermissionError) as exc:
            logger.warning(
                "Telegram Beta binding rejected telegram_id=%s reason=%s",
                telegram_user_id,
                exc,
            )
            await self._reply(str(exc), None)
            return

        await self._reply(
            (
                "Welcome to EasyChatBot Academy. "
                f"Your classroom is connected to World '{user.active_world_id}'."
            ),
            user,
        )

from __future__ import annotations


class BetaTelegramAdapter:
    """
    Thin adapter for python-telegram-bot.

    The adapter deliberately keeps Telegram transport concerns outside
    BetaAccessService.
    """

    def __init__(self, *, start_handler, private_chat_gate):
        self._start_handler = start_handler
        self._private_chat_gate = private_chat_gate

    async def handle_start(self, update, context):
        telegram_user_id = str(update.effective_user.id)
        chat_id = str(update.effective_chat.id)
        args = context.args or []

        token = args[0] if args else ""

        result = self._start_handler.handle(
            token=token,
            telegram_id=telegram_user_id,
            chat_id=chat_id,
        )

        await update.effective_message.reply_text(result.message)

    def is_private_chat_allowed(self, *, user_id: str, chat_type: str) -> bool:
        return self._private_chat_gate.authorize(
            user_id=user_id,
            chat_type=chat_type,
        )

@dataclass(frozen=True)
class BetaConnectionTokenResult:
    """
    Result of creating a Beta Telegram connection token.

    Compatibility:
    - ``result.token`` provides the raw token.
    - ``result.connection`` provides the persisted token record.
    - Tuple unpacking remains supported::

          raw_token, connection = result

    The raw token is never persisted. Only its SHA-256 hash is stored.
    """

    token: str
    connection: BetaConnectionToken

    def __iter__(self) -> Iterator[object]:
        yield self.token
        yield self.connection


class BetaAccessService:
    """Application boundary for Beta eligibility and Telegram binding."""

    def __init__(
        self,
        *,
        user_repository: BetaUserRepository,
        token_repository: BetaTokenRepository,
        binding_repository: TelegramBindingRepository,
        expiration_minutes: int = 15,
    ) -> None:
        if expiration_minutes <= 0:
            raise ValueError("expiration_minutes must be positive.")

        self._users = user_repository
        self._tokens = token_repository
        self._bindings = binding_repository
        self._expiration_minutes = expiration_minutes

    async def approve_beta(self, user_id: str) -> BetaUserState:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before Beta approval."
            )

        return await self._users.save(
            replace(user, beta_approved=True)
        )

    async def assign_world(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> BetaUserState:
        if not world_id:
            raise ValueError("world_id is required.")

        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before World assignment."
            )

        if not user.beta_approved:
            raise ValueError("User is not Beta approved.")

        return await self._users.save(
            replace(
                user,
                active_world_id=world_id,
            )
        )

    async def create_connection_token(
        self,
        user_id: str,
    ) -> BetaConnectionTokenResult:
        """
        Create a short-lived one-time Telegram connection token.

        The raw token is returned to the application layer but is never
        persisted. Only its SHA-256 hash is stored.
        """
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError("Email is not verified.")

        if not user.beta_approved:
            raise ValueError("User is not Beta approved.")

        if not user.active_world_id:
            raise ValueError(
                "User must have a Learning World before Telegram access."
            )

        raw_token = secrets.token_urlsafe(32)

        token = BetaConnectionToken(
            token_id=secrets.token_hex(16),
            user_id=user.user_id,
            token_hash=self.hash_token(raw_token),
            expires_at=(
                datetime.now(timezone.utc)
                + timedelta(minutes=self._expiration_minutes)
            ),
        )

        await self._tokens.save(token)

        logger.info(
            "Beta Telegram token created user_id=%s",
            user.user_id,
        )

        return BetaConnectionTokenResult(
            token=raw_token,
            connection=token,
        )

    async def validate_and_bind(
        self,
        *,
        token: str,
        telegram_id: int,
        chat_id: int,
    ) -> BetaUserState:
        if not token:
            raise ValueError(
                "Connection token is required."
            )

        if telegram_id <= 0:
            raise ValueError(
                "Telegram ID must be positive."
            )

        if chat_id == 0:
            raise ValueError(
                "Chat ID is required."
            )

        record = await self._tokens.get_by_hash(
            self.hash_token(token)
        )

        if record is None:
            raise ValueError(
                "Invalid connection token."
            )

        if record.used:
            raise ValueError(
                "Connection token already used."
            )

        expires_at = record.expires_at

        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(
                tzinfo=timezone.utc
            )

        if expires_at <= datetime.now(timezone.utc):
            raise ValueError(
                "Connection token expired."
            )

        user = await self._require_user(
            record.user_id
        )

        if not user.eligible:
            raise ValueError(
                "User is not eligible for Beta Telegram access."
            )

        if not user.active_world_id:
            raise ValueError(
                "User has no assigned Learning World."
            )

        existing = await self._users.get_by_telegram_id(
            telegram_id
        )

        if (
            existing is not None
            and existing.user_id != user.user_id
        ):
            raise ValueError(
                "Telegram account is already connected "
                "to another user."
            )

        existing_binding = await self._bindings.get_by_user_id(
            user.user_id
        )

        if (
            existing_binding is not None
            and existing_binding.telegram_user_id
            != str(telegram_id)
        ):
            raise ValueError(
                "User already has another Telegram binding."
            )

        await self._bindings.save(
            TelegramBinding(
                binding_id=secrets.token_hex(16),
                user_id=user.user_id,
                telegram_user_id=str(telegram_id),
                chat_id=str(chat_id),
                connected_at=datetime.now(timezone.utc),
            )
        )

        await self._tokens.mark_used(
            replace(
                record,
                used_at=datetime.now(timezone.utc),
            )
        )

        return await self._users.save(
            replace(
                user,
                telegram_id=telegram_id,
            )
        )

    async def authorize_private_chat(
        self,
        telegram_id: int,
    ) -> BetaUserState:
        if telegram_id <= 0:
            raise ValueError(
                "Telegram ID must be positive."
            )

        user = await self._users.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            raise PermissionError(
                "Telegram account is not connected."
            )

        if not user.active:
            raise PermissionError(
                "User is connected but is not ACTIVE "
                "for a Learning World."
            )

        return user

    @staticmethod
    def hash_token(token: str) -> str:
        return hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

    async def _require_user(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._users.get(user_id)

        if user is None:
            raise ValueError(
                "Academy user does not exist."
            )

        return user

from __future__ import annotations

import hashlib
import secrets


class BetaConnectionTokenService:
    TOKEN_BYTES = 32

    @classmethod
    def generate(cls) -> tuple[str, str]:
        raw = secrets.token_urlsafe(cls.TOKEN_BYTES)
        return raw, cls.hash_token(raw)

    @staticmethod
    def hash_token(raw_token: str) -> str:
        if not isinstance(raw_token, str) or not raw_token:
            raise ValueError("raw_token must be a non-empty string.")
        return hashlib.sha256(raw_token.encode("utf-8")).hexdigest()

class BetaTelegramInvitationService:
    """Creates the one-time Telegram deep link and sends the invitation email."""

    def __init__(
        self,
        *,
        beta_access_service: BetaAccessService,
        email_sender: BetaEmailSender,
        telegram_bot_username: str,
    ) -> None:
        if not telegram_bot_username:
            raise ValueError("telegram_bot_username is required.")
        self._access = beta_access_service
        self._email_sender = email_sender
        self._bot_username = telegram_bot_username.lstrip("@")

    async def send_invitation(self, *, user_id: str) -> str:
        raw_token, _ = await self._access.create_connection_token(user_id)
        classroom_url = (
            f"https://t.me/{self._bot_username}?start={raw_token}"
        )

        user = await self._access._require_user(user_id)
        await self._email_sender.send_beta_invitation(
            email=user.email,
            classroom_url=classroom_url,
        )
        return classroom_url

"""
OpenAI LLM Provider.

Infrastructure implementation of LLM provider.
"""


from __future__ import annotations


import os


from openai import AsyncOpenAI


from app.providers.llm.llm_provider import (
    LLMProvider,
)



class OpenAILLMProvider(
    LLMProvider
):


    def __init__(
        self,
    ):
        api_key = os.getenv(
            "OPENAI_API_KEY"
        )


        if not api_key:
            raise RuntimeError(
                "OPENAI_API_KEY missing"
            )


        self.client = AsyncOpenAI(
            api_key=api_key
        )



    async def generate(
        self,
        *,
        system_prompt: str,
        user_message: str,
    ) -> str:


        response = await self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {
                    "role": "system",
                    "content": system_prompt,
                },
                {
                    "role": "user",
                    "content": user_message,
                },
            ],
        )


        return (
            response
            .choices[0]
            .message
            .content
            or ""
        )

class ConversationAgent(BaseAgent):
    """
    Character-aware conversation agent.

    The agent consumes the provider-independent
    ExecutionContext.

    Memory and Live Intelligence remain application/runtime
    concerns. The agent only receives their prepared context.
    """

    LIVE_INTELLIGENCE_INSTRUCTION = (
        "When responding using current live intelligence, "
        "create a concise learning brief for the learner. "
        "Do not reproduce articles or provide a long news "
        "summary. Focus only on the most important information. "
        "Keep the main summary between approximately 60 and "
        "100 words. Present no more than three key ideas. "
        "Then briefly explain why the information matters "
        "to the learner. Adapt the language and complexity "
        "to the learner's level and the character's learning "
        "style. Use only information supported by the supplied "
        "live intelligence. Do not invent facts, sources, "
        "statistics, quotations, or URLs. Do not list every "
        "retrieved source in the response. Source metadata is "
        "retained separately by the application."
    )

    def __init__(
        self,
        character: CharacterProfile,
        llm_service: LLMService | None = None,
        llm_provider: LLMProvider | None = None,
    ) -> None:

        super().__init__(
            AgentManifest(
                agent_id="conversation_agent",
                name="Conversation Agent",
                description=(
                    "Provides character-aware "
                    "conversation responses."
                ),
            )
        )

        if (
            llm_service is not None
            and llm_provider is not None
        ):
            raise ValueError(
                "Provide either llm_service or "
                "llm_provider, not both."
            )

        self.character = character

        if llm_service is not None:
            self.llm_service = llm_service

        elif llm_provider is not None:
            self.llm_service = LLMService(
                llm_provider
            )

        else:
            self.llm_service = None

    def build_request(
        self,
        context: ExecutionContext,
    ) -> ConversationRequest:
        """
        Build a provider-independent conversation request.

        Memory and intelligence are explicitly represented
        as contextual information.

        Neither is treated as an instruction.
        """

        system_instruction = (
            f"You are {self.character.name}, "
            f"a character in EasyChatBot. "
            f"Your personality is: "
            f"{self.character.personality} "
            f"Your learning style is: "
            f"{self.character.learning_style} "
            f"Communicate in a way that is consistent "
            f"with this character."
        )

        context_sections: list[str] = []

        # -------------------------------------------------
        # Learner memory
        # -------------------------------------------------

        if context.has_memory_context:
            memory_text = (
                context.memory_context.text
            )

            if memory_text.strip():
                context_sections.append(
                    "Learner memory context:\n"
                    + memory_text
                )

        # -------------------------------------------------
        # Live intelligence
        # -------------------------------------------------

        has_live_intelligence = (
            context.has_intelligence_context
        )

        if has_live_intelligence:
            intelligence_text = (
                context.intelligence_context
                .to_context_text()
            )

            if intelligence_text.strip():
                context_sections.append(
                    "Current live intelligence:\n"
                    + intelligence_text
                )

                context_sections.append(
                    "Live intelligence response rules:\n"
                    + self.LIVE_INTELLIGENCE_INSTRUCTION
                )

        # -------------------------------------------------
        # Combined contextual information
        # -------------------------------------------------

        conversation_context = None

        if context_sections:
            conversation_context = (
                "\n\n".join(
                    context_sections
                )
            )

        # -------------------------------------------------
        # User prompt
        # -------------------------------------------------

        user_prompt = context.request

        if has_live_intelligence:
            user_prompt = (
                "Answer the learner's request using "
                "the supplied current live intelligence. "
                "Prioritize a concise learning experience "
                "over information volume.\n\n"
                "Learner request:\n"
                + context.request
            )

        return ConversationRequest(
            system_instruction=system_instruction,
            user_prompt=user_prompt,
            conversation_context=(
                conversation_context
            ),
            character_id=(
                self.character.character_id
            ),
            character_name=(
                self.character.name
            ),
        )

    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute the conversation agent.
        """

        conversation_request = (
            self.build_request(context)
        )

        provider_result = None

        if self.llm_service is not None:
            provider_result = (
                await self.llm_service.generate(
                    conversation_request
                )
            )

            if not provider_result.success:
                return AgentResult(
                    success=False,
                    agent_id=(
                        self.manifest.agent_id
                    ),
                    output={
                        "character": (
                            self.character.name
                        ),
                        "character_profile": {
                            "character_id": (
                                self.character.character_id
                            ),
                            "name": (
                                self.character.name
                            ),
                            "description": (
                                self.character.description
                            ),
                            "personality": (
                                self.character.personality
                            ),
                            "learning_style": (
                                self.character.learning_style
                            ),
                        },
                        "personality": (
                            self.character.personality
                        ),
                        "learning_style": (
                            self.character.learning_style
                        ),
                        "message": (
                            "The conversation provider "
                            "could not generate a response."
                        ),
                        "intelligence_context": (
                            conversation_request
                            .conversation_context
                        ),
                        "conversation_request": (
                            conversation_request
                        ),
                        "provider_id": (
                            provider_result.provider_id
                        ),
                        "provider_error": (
                            provider_result.error
                        ),
                    },
                )

        if provider_result is not None:
            message = provider_result.output

        else:
            message = (
                f"{self.character.name}: "
                f"Let's explore this together. "
                f"You asked: "
                f"{context.request}"
            )

        output = {
            "character": (
                self.character.name
            ),
            "character_profile": {
                "character_id": (
                    self.character.character_id
                ),
                "name": (
                    self.character.name
                ),
                "description": (
                    self.character.description
                ),
                "personality": (
                    self.character.personality
                ),
                "learning_style": (
                    self.character.learning_style
                ),
            },
            "personality": (
                self.character.personality
            ),
            "learning_style": (
                self.character.learning_style
            ),
            "message": message,
            "intelligence_context": (
                conversation_request
                .conversation_context
            ),
            "conversation_request": (
                conversation_request
            ),
        }

        if provider_result is not None:
            output["provider_id"] = (
                provider_result.provider_id
            )

        return AgentResult(
            success=True,
            agent_id=(
                self.manifest.agent_id
            ),
            output=output,
        )

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class ConversationRequest:
    """
    Immutable provider-independent conversation request.

    Attributes
    ----------
    system_instruction:
        Stable instruction defining how the character
        should behave.

    user_prompt:
        User-facing request to be answered.

    conversation_context:
        Optional contextual information, such as live
        intelligence retrieved by the Research Agent.

    character_id:
        Identifier of the character producing the response.

    character_name:
        Display name of the character.
    """

    system_instruction: str

    user_prompt: str

    conversation_context: str | None = None

    character_id: str = ""

    character_name: str = ""

    def __post_init__(self) -> None:
        """
        Validate request invariants.
        """

        if not isinstance(
            self.system_instruction,
            str,
        ):
            raise ValueError(
                "system_instruction must be a string."
            )

        if not isinstance(
            self.user_prompt,
            str,
        ):
            raise ValueError(
                "user_prompt must be a string."
            )

        system_instruction = (
            self.system_instruction.strip()
        )

        user_prompt = (
            self.user_prompt.strip()
        )

        if not system_instruction:
            raise ValueError(
                "system_instruction cannot be empty."
            )

        if not user_prompt:
            raise ValueError(
                "user_prompt cannot be empty."
            )

        object.__setattr__(
            self,
            "system_instruction",
            system_instruction,
        )

        object.__setattr__(
            self,
            "user_prompt",
            user_prompt,
        )

        if self.conversation_context is not None:
            context = (
                self.conversation_context.strip()
            )

            if not context:
                raise ValueError(
                    "conversation_context cannot be blank."
                )

            object.__setattr__(
                self,
                "conversation_context",
                context,
            )

        if not isinstance(
            self.character_id,
            str,
        ):
            raise ValueError(
                "character_id must be a string."
            )

        if not isinstance(
            self.character_name,
            str,
        ):
            raise ValueError(
                "character_name must be a string."
            )

        object.__setattr__(
            self,
            "character_id",
            self.character_id.strip(),
        )

        object.__setattr__(
            self,
            "character_name",
            self.character_name.strip(),
        )

    @property
    def has_conversation_context(self) -> bool:
        """
        Return whether contextual information exists.
        """

        return bool(
            self.conversation_context
        )

    @property
    def context_length(self) -> int:
        """
        Return the length of the conversation context.

        Useful for diagnostics and future token estimation.
        """

        if not self.conversation_context:
            return 0

        return len(
            self.conversation_context
        )

    def summary(self) -> dict[str, str]:
        """
        Return a safe diagnostic summary.

        Prompt and context content are deliberately
        excluded from the summary.
        """

        return {
            "character_id": self.character_id,
            "character_name": self.character_name,
            "has_context": str(
                self.has_conversation_context
            ),
            "system_instruction_length": str(
                len(self.system_instruction)
            ),
            "user_prompt_length": str(
                len(self.user_prompt)
            ),
            "context_length": str(
                self.context_length
            ),
        }

"""
LLM Service.

Application-level service responsible for executing
provider-independent ConversationRequest objects.

Architecture:

ConversationAgent
|
v
LLMService
|
v
LLMProvider
|
v
Concrete LLM Provider
|
v
ProviderResult

Responsibilities:

- receive a ConversationRequest
- delegate execution to an injected LLMProvider
- return the provider-independent ProviderResult
- keep provider execution behind an application service boundary

Does not:

- build prompts
- manage characters
- perform research
- select providers
- create providers
- call provider SDKs directly
- manage memory
- manage conversations
- modify application state
"""

from __future__ import annotations

import logging

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.providers.provider_result import (
    ProviderResult,
)


logger = logging.getLogger(__name__)


class LLMService:
    """
    Application service for LLM execution.

    The service depends only on the provider-independent
    LLMProvider contract.

    Provider construction and provider selection remain
    outside this service.
    """

    def __init__(
        self,
        llm_provider: LLMProvider,
    ) -> None:
        """
        Initialise the LLM service.

        Parameters
        ----------
        llm_provider:
            Provider implementing the LLMProvider contract.
        """

        if llm_provider is None:
            raise ValueError(
                "LLM provider cannot be None."
            )

        self.llm_provider = llm_provider

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate an LLM response.

        The service delegates execution to the injected
        provider without knowing the provider's concrete
        implementation.

        Parameters
        ----------
        request:
            Provider-independent conversation request.

        Returns
        -------
        ProviderResult
            Standard provider execution result.

        Raises
        ------
        ValueError
            If the conversation request is None.

        RuntimeError
            If the provider returns no result.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        logger.info(
            "LLM generation started."
        )

        result = await self.llm_provider.generate(
            request
        )

        if result is None:
            raise RuntimeError(
                "LLM provider returned no result."
            )

        logger.info(
            (
                "LLM generation completed. "
                "success=%s provider=%s"
            ),
            result.success,
            result.provider_id,
        )

        return result

from __future__ import annotations

from src.ai_runtime.capabilities.models import (
    CapabilityRequest,
    CapabilityType,
    CapabilityResult,
)

from src.ai_runtime.capabilities.gateway import (
    CapabilityGateway,

class LearningService:
    """
    Application service for AI learning workflows.
    """

    def __init__(
        self,
        *,
        capability_gateway: CapabilityGateway,
    ) -> None:
        """
        Initialise learning service.
        """

        if capability_gateway is None:
            raise ValueError(
                "Capability gateway cannot be None."
            )

        self._gateway = capability_gateway


    async def generate_flashcards(
        self,
        *,
        request_id: str,
        topic: str,
        learner_context: str | None = None,
    ) -> CapabilityResult:
        """
        Generate personalised flashcards.

        The service only orchestrates.

        Flashcard creation remains inside:
            FlashcardGenerationExecutor

            raise ValueError(
                "Topic cannot be empty."
            )

            capability=(
                CapabilityType.FLASHCARD_GENERATION
            ),
            request_id=request_id,
            payload={
                "topic": topic,
                "learner_context": learner_context,
            },
        )

        return await self._gateway.execute(
            request,
        )

"""
Companion Service.

Application orchestration layer for EasyChatBot.

Coordinates:

Character
Conversation Agent
Research Agent
Quality Agent
Memory Service
Memory Context Builder

Live Intelligence + Memory flow:

CompanionService
|
+--> MemoryService
|        |
|        v
|   MemoryRecord[]
|        |
|        v
|   MemoryContextBuilder
|        |
|        v
|   MemoryContext
|
+--> ResearchAgent
|        |
|        v
|   IntelligenceContext
|
+--> ConversationAgent
|        |
|        v
|   LLMService
|        |
|        v
|   LLMProvider
|
+--> QualityAgent
|
v
Interaction Result

Responsibilities:

- orchestrate companion interactions
- retrieve memories
- build bounded learner memory context
- optionally execute research
- enrich the execution context with memory
- enrich the execution context with live intelligence
- execute the ConversationAgent
- execute the QualityAgent
- persist the interaction
- return the combined result

Does not:

- create LLM providers
- select LLM providers
- call provider SDKs
- call OpenAI directly
- call Ollama directly
- manage provider routing
- build provider-specific prompts
- persist memory directly
"""

from __future__ import annotations

from dataclasses import replace
from typing import Any

from backend.agents.conversation_agent import (
    ConversationAgent,
)

from backend.agents.research_agent import (
    ResearchAgent,
)

from backend.agents.quality_agent import (
    QualityAgent,
)

from backend.characters.character_profile import (
    CharacterProfile,
)

from backend.memory.memory_service import (
    MemoryService,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)

from src.ai_runtime.memory.context_builder import (
    MemoryContextBuilder,
)


class CompanionService:
    """
    Coordinates AI companion experiences.

    The service orchestrates agents but does not contain
    provider-specific logic.

    Providers are injected into agents by the composition
    layer.

    Memory retrieval remains owned by MemoryService.

    MemoryContextBuilder converts already-retrieved memory
    into bounded runtime context.

    ExecutionContext remains immutable. Enrichment is
    performed using dataclasses.replace().
    """

    def __init__(
        self,
        character: CharacterProfile,
        conversation_agent: ConversationAgent | None = None,
        research_agent: ResearchAgent | None = None,
        quality_agent: QualityAgent | None = None,
        memory_service: MemoryService | None = None,
        memory_context_builder: (MemoryContextBuilder | None) = None,
    ) -> None:
        """
        Initialise the CompanionService.

        Dependencies are injectable so the service can be
        tested without invoking real infrastructure.
        """

        self.character = character

        self.conversation_agent = (
            conversation_agent
            if conversation_agent is not None
            else ConversationAgent(character)
        )

        self.research_agent = (
            research_agent if research_agent is not None else ResearchAgent()
        )

        self.quality_agent = (
            quality_agent if quality_agent is not None else QualityAgent()
        )

        self.memory_service = (
            memory_service if memory_service is not None else MemoryService()
        )

        self.memory_context_builder = (
            memory_context_builder
            if memory_context_builder is not None
            else MemoryContextBuilder()
        )

    async def interact(
        self,
        context: ExecutionContext,
        use_research: bool = False,
    ) -> dict[str, Any]:
        """
        Execute a complete companion interaction.

        Flow:

        MemoryService
            â†“
        MemoryContextBuilder
            â†“
        optional memory-enriched ExecutionContext
            â†“
        optional ResearchAgent
            â†“
        optional intelligence-enriched ExecutionContext
            â†“
        ConversationAgent
            â†“
        QualityAgent
            â†“
        MemoryService.remember()

        The original ExecutionContext is never mutated.

        Parameters
        ----------
        context:
            Runtime execution context.

        use_research:
            Whether live research should be performed
            before the conversation response.

        Returns
        -------
        dict[str, Any]
            Combined companion interaction result.

        Raises
        ------
        ValueError
            If the execution context is missing.
        """

        if context is None:
            raise ValueError("Execution context cannot be None.")

        # -----------------------------------------------------
        # Step 1 â€” Retrieve existing learner memory
        # -----------------------------------------------------

        memories = self.memory_service.recall(context.user_id)

        # -----------------------------------------------------
        # Step 2 â€” Build bounded learner memory context
        # -----------------------------------------------------

        memory_context = self.memory_context_builder.build(
            user_id=context.user_id,
            request=context.request,
            memories=memories,
        )

        # -----------------------------------------------------
        # Step 3 â€” Attach memory only when relevant memory
        # exists.
        #
        # If there is no relevant memory, preserve the
        # original ExecutionContext object.
        # -----------------------------------------------------

        if memory_context.is_empty:
            conversation_context = context
        else:
            conversation_context = replace(
                context,
                memory_context=memory_context,
            )

        # -----------------------------------------------------
        # Step 4 â€” Optional Live Intelligence
        #
        # Research receives the current runtime context but
        # does not receive or manage memory itself.
        #
        # If research succeeds, create another immutable
        # context preserving the existing memory_context.
        # -----------------------------------------------------

        research_result = None

        if use_research:
            research_result = await self.research_agent.execute(conversation_context)

            if research_result.success and research_result.output is not None:
                conversation_context = replace(
                    conversation_context,
                    intelligence_context=(research_result.output),
                )

        # -----------------------------------------------------
        # Step 5 â€” Conversation Agent
        #
        # ConversationAgent can now receive:
        #
        #   memory_context
        #   intelligence_context
        #
        # independently or together.
        # -----------------------------------------------------

        conversation_result = await self.conversation_agent.execute(
            conversation_context
        )

        # -----------------------------------------------------
        # Step 6 â€” Quality Agent
        # -----------------------------------------------------

        quality_result = await self.quality_agent.execute(conversation_context)

        # -----------------------------------------------------
        # Step 7 â€” Persist interaction
        #
        # Memory persistence remains owned by MemoryService.
        # -----------------------------------------------------

        self.memory_service.remember(
            user_id=context.user_id,
            category="conversation",
            content=context.request,
        )

        # -----------------------------------------------------
        # Step 8 â€” Return combined result
        # -----------------------------------------------------

        return {
            "character": self.character.name,
            "conversation": (conversation_result.output),
            "research": (
                research_result.output
                if (research_result is not None and research_result.success)
                else None
            ),
            "quality": (quality_result.output),
            "memory_count": len(memories),
            "memory_context_count": (memory_context.memory_count),
        }

"""
Runtime Application Service.

Application boundary for External Runtime execution.

Responsibilities:

- translate application requests into Runtime contracts;
- create CapabilityRequest objects;
- generate request identifiers;
- delegate execution to ExternalRuntimeService;
- return Runtime capability results.

This service does not:

- expose HTTP concerns;
- raise FastAPI HTTP exceptions;
- create providers;
- create agents;
- implement capabilities;
- implement governance;
- bypass Runtime controls.

Architecture:

API
 |
 v
RuntimeApplicationService
 |
 v
ExternalRuntimeService
 |
 v
CapabilityGateway
 |
 v
CapabilityExecutor
"""

from __future__ import annotations

from collections.abc import Mapping
from uuid import uuid4

from backend.services.external_runtime_service import (
    ExternalRuntimeService,
)

from src.ai_runtime.capabilities.models import (
    CapabilityName,
    CapabilityRequest,
    CapabilityResult,
)


class RuntimeApplicationService:
    """
    Application service for External Runtime execution.

    Keeps API routes thin by moving Runtime
    application orchestration outside HTTP layer.
    """

    def __init__(
        self,
        runtime_service: ExternalRuntimeService,
    ) -> None:
        """
        Initialise RuntimeApplicationService.

        Args:
            runtime_service:
                Configured ExternalRuntimeService.

        Raises:
            ValueError:
                If runtime service is missing.
        """

        if runtime_service is None:
            raise ValueError(
                "ExternalRuntimeService cannot be None."
            )

        self._runtime_service = runtime_service

    async def execute(
        self,
        *,
        capability: str,
        payload: Mapping[str, object],
    ) -> CapabilityResult:
        """
        Execute an External Runtime capability.

        The application layer translates external
        application requests into Runtime contracts.

        Args:

            capability:
                Requested Runtime capability.

            payload:
                Capability execution payload.

        Returns:

            CapabilityResult produced by Runtime.

        Raises:

            ValueError:
                If input is invalid.
        """

        if not isinstance(
            capability,
            str,
        ):
            raise ValueError(
                "Capability must be a string."
            )

        if not capability.strip():
            raise ValueError(
                "Capability cannot be empty."
            )

        if payload is None:
            raise ValueError(
                "Payload cannot be None."
            )

        try:
            capability_name = CapabilityName(
                capability,
            )

        except ValueError as exc:
            raise ValueError(
                f"Unsupported capability: {capability}"
            ) from exc

        execution_payload = {
            "user_id": "demo-user",
            **dict(payload),
        }

        request = CapabilityRequest(
            request_id=str(
                uuid4(),
            ),
            capability=capability_name,
            payload=execution_payload,
        )

        return await self._runtime_service.execute(
            request,
        )

    tokens = InMemoryBetaTokenRepository()
    bindings = InMemoryTelegramBindingRepository()

    service = BetaAccessService(
        user_repository=users,
        token_repository=tokens,
        binding_repository=bindings,
    )

    for user_id in ("a", "b"):
        await users.save(
            BetaUserState(
                user_id=user_id,
                email=f"{user_id}@example.com",
                email_verified=True,
                beta_approved=True,
                active_world_id="world-fr",
            )
        )

    first = await service.create_connection_token("a")
    await service.validate_and_bind(
        token=first.token,
        telegram_id=111,
        chat_id=111,
    )

    second = await service.create_connection_token("b")

    with pytest.raises(ValueError, match="already connected"):
        await service.validate_and_bind(
            token=second.token,
            telegram_id=111,
            chat_id=111,
        )

class Users:
    def __init__(self):
        self.items = {}
    async def get(self, user_id):
        return self.items.get(user_id)
    async def get_by_telegram_id(self, telegram_id):
        return next((u for u in self.items.values() if u.telegram_id == telegram_id), None)
    async def save(self, user):
        self.items[user.user_id] = user
        return user


class Tokens:
    def __init__(self):
        self.items = {}
    async def get_by_hash(self, token_hash):
        return next((t for t in self.items.values() if t.token_hash == token_hash), None)
    async def save(self, token):
        self.items[token.token_id] = token
        return token
    async def mark_used(self, token):
        self.items[token.token_id] = token
        return token


class Bindings:
    def __init__(self):
        self.items = {}
    async def get_by_telegram_user_id(self, telegram_user_id):
        return next((b for b in self.items.values() if b.telegram_user_id == telegram_user_id), None)
    async def get_by_user_id(self, user_id):
        return next((b for b in self.items.values() if b.user_id == user_id), None)
    async def save(self, binding):
        self.items[binding.binding_id] = binding
        return binding
