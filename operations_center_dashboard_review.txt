"""
EasyChatBot AI Operations Center dashboard service.

Provides a read-only, user-friendly operational view
for the AI Operations Center.

Architecture:

Provider Intelligence
        |
        v
Provider Health / Escalations / Workflow
        |
        v
OperationsCenterDashboardService
        |
        v
Dashboard Read Model
        |
        v
Frontend / API


Responsibilities:

- aggregate provider operational status;
- expose active providers;
- expose monitoring status;
- expose escalation counts;
- expose pending human approvals;
- calculate an operational confidence score;
- provide a stable presentation-independent read model.

Does NOT:

- execute providers;
- change Runtime routing;
- approve recommendations;
- send emails;
- modify escalation state;
- modify provider configuration.

This is a read-only application service.
"""

from __future__ import annotations

from dataclasses import dataclass
from enum import Enum
from typing import TYPE_CHECKING

from app.application.provider_management.escalation_history_service import (
    EscalationHistoryService,
)

from app.application.provider_management.escalation_workflow_service import (
    EscalationWorkflowStatus,
)

if TYPE_CHECKING:
    from app.application.provider_management.operations_center_provider_service import (
        OperationsCenterProviderService,
    )


class MonitoringStatus(str, Enum):
    """
    Overall monitoring state shown by the Operations Center.
    """

    HEALTHY = "healthy"

    ATTENTION_REQUIRED = (
        "attention_required"
    )

    INCIDENT = "incident"


@dataclass(frozen=True)
class ActiveProvider:
    """
    Provider information displayed by the Operations Center.
    """

    provider: str

    model: str

    capabilities: tuple[str, ...]

    active: bool = True


@dataclass(frozen=True)
class OperationsCenterDashboard:
    """
    User-friendly Operations Center dashboard read model.
    """

    system_confidence: float

    monitoring_status: MonitoringStatus

    active_providers: tuple[
        ActiveProvider,
        ...
    ]

    active_escalations: int

    pending_approvals: int

    approved_escalations: int

    rejected_escalations: int

    pending_notifications: int

    total_escalations: int


# Sentinel used to distinguish:
#
#     build_dashboard()
#
# from:
#
#     build_dashboard(active_providers=None)
#
# The former means "obtain providers from the configured
# provider service".
#
# The latter is invalid input and must remain rejected.
_UNSET = object()


class OperationsCenterDashboardService:
    """
    Builds the read-only Operations Center dashboard.

    Provider information can be explicitly supplied for
    deterministic tests and controlled application use.

    When active providers are omitted entirely, the configured
    OperationsCenterProviderService is used.

    Explicit None remains invalid.
    """

    def __init__(
        self,
        history_service: EscalationHistoryService,
        provider_service: (
            OperationsCenterProviderService
            | None
        ) = None,
    ) -> None:
        """
        Initialise the dashboard service.
        """

        if history_service is None:
            raise ValueError(
                "Escalation history service "
                "cannot be None."
            )

        self._history_service = (
            history_service
        )

        self._provider_service = (
            provider_service
        )

    def build_dashboard(
        self,
        *,
        active_providers: (
            tuple[
                ActiveProvider,
                ...
            ]
            | None
            | object
        ) = _UNSET,
    ) -> OperationsCenterDashboard:
        """
        Build the current Operations Center dashboard.

        Provider resolution rules:

        1. Explicit tuple:
               use the supplied providers.

        2. Explicit None:
               reject the invalid input.

        3. Argument omitted:
               use the configured provider service.

        This distinction preserves the original dashboard
        contract while allowing the production composition
        root to obtain real provider intelligence.
        """

        if active_providers is None:
            raise ValueError(
                "Active providers cannot be None."
            )

        if active_providers is _UNSET:
            if self._provider_service is None:
                raise ValueError(
                    "Active providers were not supplied "
                    "and no provider service is configured."
                )

            resolved_active_providers = (
                self._provider_service
                .get_active_providers()
            )
        else:
            resolved_active_providers = (
                active_providers
            )

        counts = (
            self._history_service
            .count_by_status()
        )

        pending_approvals = counts[
            EscalationWorkflowStatus
            .AWAITING_APPROVAL
        ]

        approved = counts[
            EscalationWorkflowStatus
            .APPROVED
        ]

        rejected = counts[
            EscalationWorkflowStatus
            .REJECTED
        ]

        pending_notifications = counts[
            EscalationWorkflowStatus
            .PENDING_NOTIFICATION
        ]

        total_escalations = sum(
            counts.values()
        )

        active_escalations = (
            pending_approvals
            + pending_notifications
        )

        monitoring_status = (
            self._determine_monitoring_status(
                active_escalations=(
                    active_escalations
                ),
                pending_approvals=(
                    pending_approvals
                ),
            )
        )

        confidence = (
            self._calculate_confidence(
                active_escalations=(
                    active_escalations
                ),
                pending_approvals=(
                    pending_approvals
                ),
                total_escalations=(
                    total_escalations
                ),
            )
        )

        return OperationsCenterDashboard(
            system_confidence=confidence,
            monitoring_status=monitoring_status,
            active_providers=(
                resolved_active_providers
            ),
            active_escalations=(
                active_escalations
            ),
            pending_approvals=(
                pending_approvals
            ),
            approved_escalations=approved,
            rejected_escalations=rejected,
            pending_notifications=(
                pending_notifications
            ),
            total_escalations=(
                total_escalations
            ),
        )

    @staticmethod
    def _determine_monitoring_status(
        *,
        active_escalations: int,
        pending_approvals: int,
    ) -> MonitoringStatus:
        """
        Determine the dashboard monitoring status.

        Critical operational state is represented by the
        existence of active escalations.

        Pending human approval is explicitly surfaced.
        """

        if active_escalations > 0:
            return MonitoringStatus.INCIDENT

        if pending_approvals > 0:
            return (
                MonitoringStatus
                .ATTENTION_REQUIRED
            )

        return MonitoringStatus.HEALTHY

    @staticmethod
    def _calculate_confidence(
        *,
        active_escalations: int,
        pending_approvals: int,
        total_escalations: int,
    ) -> float:
        """
        Calculate a deterministic dashboard confidence score.

        This is deliberately a simple operational indicator
        for the Hackathon.

        It is NOT an AI model confidence score.

        Rules:

        - 100 when there are no escalations;
        - subtract 15 per active escalation;
        - pending approval receives no additional penalty;
        - never below 0.

        Pending approvals are already represented in
        active_escalations, so applying another penalty would
        double-count the same operational condition.
        """

        if total_escalations == 0:
            return 100.0

        score = 100.0

        score -= (
            active_escalations * 15.0
        )

        return max(
            0.0,
            min(100.0, score),
        )

"""
EasyChatBot AI Operations Center dashboard facade.

Provides a stable application-facing contract for the
Operations Center presentation/API layer.

Architecture:

Provider Health
      |
Escalations
      |
Workflow
      |
History Service
      |
Dashboard Service
      |
      v
OperationsCenterDashboardFacade
      |
      v
API / Frontend


Responsibilities:

- expose dashboard information to presentation layers;
- convert domain/application read models into
  presentation-safe dictionaries;
- keep frontend/API code independent from internal
  workflow objects;
- provide stable field names for the Operations Center UI;
- preserve compatibility with explicit provider input.

Does NOT:

- execute providers;
- change Runtime routing;
- approve escalations;
- reject escalations;
- send emails;
- modify persistence;
- perform autonomous operational actions.

This is a read-only presentation boundary.
"""

from __future__ import annotations

from typing import Any

from app.application.provider_management.operations_center_dashboard_service import (
    ActiveProvider,
    OperationsCenterDashboard,
    OperationsCenterDashboardService,
)


_UNSET = object()


class OperationsCenterDashboardFacade:
    """
    Presentation-facing facade for the AI Operations Center.

    Provider resolution normally remains below this boundary.

    For compatibility and deterministic testing, callers may
    explicitly supply active providers.

    Calling get_dashboard() without provider input delegates
    provider resolution to OperationsCenterDashboardService.
    """

    def __init__(
        self,
        dashboard_service: OperationsCenterDashboardService,
    ) -> None:
        """
        Initialise the dashboard facade.
        """

        if dashboard_service is None:
            raise ValueError(
                "Operations Center dashboard service "
                "cannot be None."
            )

        self._dashboard_service = (
            dashboard_service
        )

    def get_dashboard(
        self,
        *,
        active_providers: (
            tuple[
                ActiveProvider,
                ...
            ]
            | None
            | object
        ) = _UNSET,
    ) -> dict[str, Any]:
        """
        Return a frontend-safe dashboard representation.

        Resolution rules:

        - omitted:
            use provider resolution configured in the
            dashboard service;

        - explicit tuple:
            use the supplied providers;

        - explicit None:
            reject the invalid input.

        This preserves the existing application contract
        while supporting the new production composition.
        """

        if active_providers is None:
            raise ValueError(
                "Active providers cannot be None."
            )

        if active_providers is _UNSET:
            dashboard = (
                self._dashboard_service
                .build_dashboard()
            )
        else:
            dashboard = (
                self._dashboard_service
                .build_dashboard(
                    active_providers=(
                        active_providers
                    )
                )
            )

        return self._serialize_dashboard(
            dashboard
        )

    @staticmethod
    def _serialize_dashboard(
        dashboard: OperationsCenterDashboard,
    ) -> dict[str, Any]:
        """
        Convert the dashboard read model into a stable
        presentation contract.
        """

        return {
            "system_confidence": (
                dashboard.system_confidence
            ),
            "monitoring_status": (
                dashboard.monitoring_status.value
            ),
            "active_providers": [
                OperationsCenterDashboardFacade
                ._serialize_provider(provider)
                for provider
                in dashboard.active_providers
            ],
            "active_escalations": (
                dashboard.active_escalations
            ),
            "pending_approvals": (
                dashboard.pending_approvals
            ),
            "approved_escalations": (
                dashboard.approved_escalations
            ),
            "rejected_escalations": (
                dashboard.rejected_escalations
            ),
            "pending_notifications": (
                dashboard.pending_notifications
            ),
            "total_escalations": (
                dashboard.total_escalations
            ),
        }

    @staticmethod
    def _serialize_provider(
        provider: ActiveProvider,
    ) -> dict[str, Any]:
        """
        Convert an active provider into a frontend-safe
        representation.
        """

        return {
            "provider": provider.provider,
            "model": provider.model,
            "capabilities": list(
                provider.capabilities
            ),
            "active": provider.active,
        }

"""
EasyChatBot AI Operations Center dashboard API.

Provides the application-facing API boundary for the
Operations Center dashboard.

Architecture:

Provider Health / Intelligence
        |
        v
OperationsCenterDashboardService
        |
        v
OperationsCenterDashboardFacade
        |
        v
OperationsCenterDashboardAPI
        |
        v
HTTP Router / Frontend


Responsibilities:

- expose the dashboard through a stable application API;
- delegate dashboard generation to the facade;
- preserve the existing explicit-provider contract;
- allow production composition to resolve providers below
  the API boundary;
- keep HTTP concerns outside the application layer.

Does NOT:

- access repositories directly;
- execute providers;
- change Runtime routing;
- approve escalations;
- reject escalations;
- send emails;
- modify provider configuration;
- perform autonomous operational actions.

The API is read-only.
"""

from __future__ import annotations

from typing import Any

from app.application.provider_management.operations_center_dashboard_facade import (
    OperationsCenterDashboardFacade,
)

from app.application.provider_management.operations_center_dashboard_service import (
    ActiveProvider,
)


_UNSET = object()


class OperationsCenterDashboardAPI:
    """
    Application-facing API for the Operations Center
    dashboard.

    Provider information may be supplied explicitly for
    deterministic callers/tests.

    In production, the composed facade/dashboard service can
    resolve provider information when the argument is omitted.
    """

    def __init__(
        self,
        facade: OperationsCenterDashboardFacade,
    ) -> None:
        """
        Initialise the dashboard API.
        """

        if facade is None:
            raise ValueError(
                "Operations Center dashboard facade "
                "cannot be None."
            )

        self._facade = facade

    def get_dashboard(
        self,
        *,
        active_providers: (
            tuple[
                ActiveProvider,
                ...
            ]
            | None
            | object
        ) = _UNSET,
    ) -> dict[str, Any]:
        """
        Return the current Operations Center dashboard.

        Resolution rules:

        - omitted:
            provider information is resolved by the composed
            application stack;

        - explicit tuple:
            the supplied provider information is passed through;

        - explicit None:
            rejected as invalid input.

        The API itself does not discover providers.
        """

        if active_providers is None:
            raise ValueError(
                "Active providers cannot be None."
            )

        if active_providers is _UNSET:
            return self._facade.get_dashboard()

        return self._facade.get_dashboard(
            active_providers=(
                active_providers
            )
        )

- pending approval is informational and is not
  penalised a second time;
- confidence cannot fall below 0.
"""

from datetime import datetime, timezone

def create_workflow(
    *,
    escalation_id: str,
    status: EscalationWorkflowStatus,
    minute: int,
) -> EscalationWorkflow:
    """
    Create deterministic workflow data.
    """

    return EscalationWorkflow(
        escalation_id=escalation_id,
        status=status,
        created_at=datetime(
            2026,
            8,
            14,
            14,
            minute,
            tzinfo=timezone.utc,
        ),
        notification_recipient=(
            "operations@example.com"
            if status
            != EscalationWorkflowStatus
            .PENDING_NOTIFICATION
            else None
        ),
        notification_subject=(
            "EasyChatBot AI Operations Escalation"
            if status
            != EscalationWorkflowStatus
            .PENDING_NOTIFICATION
            else None
        ),
        approved_at=(
            datetime(
                2026,
                8,
                14,
                15,
                minute,
                tzinfo=timezone.utc,
            )
            if status
            == EscalationWorkflowStatus.APPROVED
            else None
        ),
        rejected_at=(
            datetime(
                2026,
                8,
                14,
                15,
                minute,
                tzinfo=timezone.utc,
            )
            if status
            == EscalationWorkflowStatus.REJECTED
            else None
        ),
        decision_reason=(
            "Operations decision recorded."
            if status
            in (
                EscalationWorkflowStatus.APPROVED,
                EscalationWorkflowStatus.REJECTED,
            )
            else None
        ),
    )


def create_service() -> (
    tuple[
        OperationsCenterDashboardService,
        InMemoryEscalationWorkflowRepository,
    ]
):
    """
    Create a dashboard service and repository.
    """

    repository = (
        InMemoryEscalationWorkflowRepository()
    )

    history_service = (
        EscalationHistoryService(
            repository=repository
        )
    )

    dashboard_service = (
        OperationsCenterDashboardService(
            history_service=history_service
        )
    )

    return dashboard_service, repository

def test_facade_result_is_json_safe() -> None:
    """
    Validate that the complete dashboard result consists
    only of JSON-compatible primitive/container values.

    This deliberately avoids importing a JSON framework so
    the application layer remains infrastructure-independent.
    """

    facade, repository = create_service()

    repository.save(
        create_workflow(
            escalation_id="esc-001",
            status=(
                EscalationWorkflowStatus
                .AWAITING_APPROVAL
            ),
            minute=30,
        )
    )

    result = facade.get_dashboard(
        active_providers=(
            create_active_providers()
        )
    )

    def assert_json_safe(
        value: object,
    ) -> None:
        if value is None:
            return

        if isinstance(
            value,
            (
                str,
                int,
                float,
                bool,
            ),
        ):
            return

        if isinstance(value, list):
            for item in value:
                assert_json_safe(item)
            return

        if isinstance(value, dict):
            for key, item in value.items():
                assert isinstance(
                    key,
                    str,
                )
                assert_json_safe(item)
            return

        raise AssertionError(
            "Dashboard contains a non-JSON-safe "
            f"value: {type(value)!r}"
        )

    assert_json_safe(result)
