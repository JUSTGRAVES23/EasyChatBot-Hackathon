"""
================================================================================
File: runtime/policies/engine.py

Purpose:
Coordinates the evaluation of Runtime Policies.

The PolicyEngine delegates decision making to registered Policy
implementations while remaining independent of execution, providers,
models and business logic.

Author:
EasyChatBot AI Runtime
================================================================================

from __future__ import annotations

from runtime.policies.policy import (
    PolicyContext,
    PolicyDecision,
)
from runtime.policies.registry import PolicyRegistry


class PolicyEngine:
    """
    Central runtime policy evaluator.

    The PolicyEngine is responsible for resolving a registered
    policy and returning its PolicyDecision.

    It deliberately contains no business rules itself.
    """

    def __init__(
        self,
        registry: PolicyRegistry,
    ) -> None:
        """
        Initialize the PolicyEngine.

        Parameters
        ----------
        registry:
            Registry containing all available runtime policies.
        """
        self._registry = registry

    def evaluate(
        self,
        policy_name: str,
        context: PolicyContext,
    ) -> PolicyDecision:
        """
        Evaluate a registered policy.

        Parameters
        ----------
        policy_name:
            Name of the registered policy.

        context:
            Immutable policy evaluation context.

        Returns
        -------
        PolicyDecision
            Decision produced by the selected policy.

        Raises
        ------
        KeyError
            If the requested policy is not registered.
        """

        policy = self._registry.get(policy_name)

        return policy.evaluate(context)

    def exists(
        self,
        policy_name: str,
    ) -> bool:
        """
        Determine whether a policy is registered.
        """

        return self._registry.exists(policy_name)

    def list(self) -> tuple[str, ...]:
        """
        Return all registered policies.
        """

        return self._registry.list()

    def __repr__(self) -> str:
        return (
            f"{self.__class__.__name__}"
            f"(registered={len(self._registry)})"
        )

"""

File: runtime/policies/policy.py

Purpose:
Defines the common policy contracts used throughout the AI Runtime.

Policies evaluate a PolicyContext and return a PolicyDecision.

The policy subsystem centralizes runtime decisions such as:

- Provider selection
- Model selection
- Retry behaviour
- Approval decisions
- Cost governance
- Fallback behaviour

Policies contain decision logic only.
They never execute providers, modify runtime state or perform business logic.

EasyChatBot AI Runtime

from __future__ import annotations

from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any


@dataclass(slots=True, frozen=True)
class PolicyContext:
    """
    Immutable context supplied to policy evaluation.

    The context contains the information required for making
    runtime decisions. Additional fields may be introduced
    over time without affecting the Policy interface.
    """

    agent_id: str
    capability: str

    provider_id: str | None = None
    model_id: str | None = None

    metadata: dict[str, Any] = field(default_factory=dict)


@dataclass(slots=True, frozen=True)
class PolicyDecision:
    """
    Result returned by policy evaluation.

    The decision represents recommendations only.
    It never performs execution.
    """

    approved: bool = True

    provider_id: str | None = None
    model_id: str | None = None

    retry: bool = False
    max_retries: int = 0

    fallback_provider: str | None = None

    reason: str | None = None

    metadata: dict[str, Any] = field(default_factory=dict)


class Policy(ABC):
    """
    Base class for all runtime policies.

    Each policy evaluates a PolicyContext and returns
    a standardized PolicyDecision.
    """

    @property
    @abstractmethod
    def name(self) -> str:
        """
        Human-readable policy name.
        """
        raise NotImplementedError

    @abstractmethod
    def evaluate(
        self,
        context: PolicyContext,
    ) -> PolicyDecision:
        """
        Evaluate the supplied policy context.

        Parameters
        ----------
        context:
            Immutable policy evaluation context.

        Returns
        -------
        PolicyDecision
            Decision produced by the policy.
        """
        raise NotImplementedError

"""
================================================================================
File: runtime/policies/registry.py

Purpose:
Maintains the collection of available Runtime Policies.

Policies register themselves once during application startup and can
later be retrieved by name by the PolicyEngine.

The registry is intentionally lightweight and contains no policy
evaluation logic.

Author:
EasyChatBot AI Runtime
================================================================================
"""

from __future__ import annotations

from runtime.policies.policy import Policy


class PolicyRegistry:
    """
    Stores all available Runtime Policies.

    The registry is responsible for registering, retrieving,
    listing and validating policies.
    """

    def __init__(self) -> None:
        self._policies: dict[str, Policy] = {}

    def register(
        self,
        policy: Policy,
    ) -> None:
        """
        Register a policy.

        Raises
        ------
        ValueError
            If a policy with the same name is already registered.
        """

        if policy.name in self._policies:
            raise ValueError(
                f"Policy '{policy.name}' is already registered."
            )

        self._policies[policy.name] = policy

    def get(
        self,
        name: str,
    ) -> Policy:
        """
        Retrieve a registered policy.

        Raises
        ------
        KeyError
            If the policy does not exist.
        """

        try:
            return self._policies[name]

        except KeyError as exc:
            raise KeyError(
                f"Policy '{name}' is not registered."
            ) from exc

    def exists(
        self,
        name: str,
    ) -> bool:
        """
        Determine whether a policy exists.
        """

        return name in self._policies

    def unregister(
        self,
        name: str,
    ) -> None:
        """
        Remove a policy.

        Raises
        ------
        KeyError
            If the policy is not registered.
        """

        if name not in self._policies:
            raise KeyError(
                f"Policy '{name}' is not registered."
            )

        del self._policies[name]

    def list(self) -> tuple[str, ...]:
        """
        Return all registered policy names.
        """

        return tuple(sorted(self._policies.keys()))

    def clear(self) -> None:
        """
        Remove all registered policies.
        """

        self._policies.clear()

    def __contains__(
        self,
        name: str,
    ) -> bool:
        return name in self._policies

    def __len__(self) -> int:
        return len(self._policies)

    def __repr__(self) -> str:
        return (
            f"{self.__class__.__name__}"
            f"(registered={len(self)})"
        )

"""
AI Governance Policy Engine.

Applies explicit governance policies to AI capabilities.

Architecture:

Capability Request
        |
        v
Governance Policy Engine
        |
        v
Governance Result

Decision model:
- ALLOW   â†’ capability is permitted
- REVIEW  â†’ human review is required
- BLOCK   â†’ capability is prohibited

This module does not:
- execute agents
- call LLMs
- call external providers
- modify application state
- perform external side effects
"""

from src.governance.models import (
    GovernanceDecision,
    GovernancePolicy,
    GovernanceResult,
    RiskLevel,
)


class GovernancePolicyEngine:
    """
    Deterministic governance decision engine.
    """

    def evaluate(
        self,
        policy: GovernancePolicy,
    ) -> GovernanceResult:
        """
        Evaluate a governance policy.

        Decision precedence:

        1. Side effects are blocked by default.
        2. Critical risk requires blocking.
        3. Explicit human review requires REVIEW.
        4. Otherwise the capability is ALLOWED.
        """

        reasons: list[str] = []

        # ---------------------------------------------------------
        # Side effects
        # ---------------------------------------------------------

        if policy.allows_side_effects:
            return GovernanceResult(
                capability=policy.capability,
                decision=GovernanceDecision.BLOCK,
                risk_level=policy.risk_level,
                reasons=(
                    "Uncontrolled side effects are prohibited.",
                ),
            )

        # ---------------------------------------------------------
        # Critical risk
        # ---------------------------------------------------------

        if policy.risk_level == RiskLevel.CRITICAL:
            return GovernanceResult(
                capability=policy.capability,
                decision=GovernanceDecision.BLOCK,
                risk_level=policy.risk_level,
                reasons=(
                    "Critical-risk capabilities require "
                    "stronger controls than this policy permits.",
                ),
            )

        # ---------------------------------------------------------
        # High-risk operations
        # ---------------------------------------------------------

        if policy.risk_level == RiskLevel.HIGH:
            reasons.append(
                "High-risk capability requires human review."
            )

            return GovernanceResult(
                capability=policy.capability,
                decision=GovernanceDecision.REVIEW,
                risk_level=policy.risk_level,
                reasons=tuple(reasons),
            )

        # ---------------------------------------------------------
        # Explicit human review
        # ---------------------------------------------------------

        if policy.requires_human_review:
            reasons.append(
                "Policy explicitly requires human review."
            )

            return GovernanceResult(
                capability=policy.capability,
                decision=GovernanceDecision.REVIEW,
                risk_level=policy.risk_level,
                reasons=tuple(reasons),
            )

        # ---------------------------------------------------------
        # External calls
        # ---------------------------------------------------------

        if policy.allows_external_calls:
            reasons.append(
                "External calls are explicitly permitted."
            )
        else:
            reasons.append(
                "No external calls are required."
            )

        # ---------------------------------------------------------
        # Default allow
        # ---------------------------------------------------------

        reasons.append(
            "No blocking governance condition was detected."
        )

        return GovernanceResult(
            capability=policy.capability,
            decision=GovernanceDecision.ALLOW,
            risk_level=policy.risk_level,
            reasons=tuple(reasons),
        )

        try:
            governance_result = (
                self._governance.evaluate(request)
            )
        except Exception as exc:
            raise CapabilityGovernanceError(
                "Governance evaluation failed."
            ) from exc

        if not governance_result.allowed:
            return CapabilityResult(
                request_id=request.request_id,
                capability=request.capability,
                status=CapabilityExecutionStatus.REJECTED,
                error=(
                    "Capability execution was rejected "
                    "by governance."
                ),
                metadata={
                    "governance_decision": "rejected",
                },
            )

        executor = self._registry.resolve(
            request.capability
        )

        try:
            return await executor.execute(request)
        except Exception as exc:
            raise CapabilityExecutionError(
                "Capability executor failed: "
                f"{request.capability.value}"
            ) from exc

"""
AI Runtime capability contracts.

This module defines the provider-independent contract used by
the AI Runtime to request execution of an approved capability.

Architecture:

Application
    |
    v
AI Runtime
    |
    v
Capability Request
    |
    v
Governance
    |
    v
Capability Execution

This module contains contracts only.

It does not:

- execute agents
- call providers
- evaluate governance policies
- perform external side effects
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import StrEnum
from typing import Any, Mapping


class CapabilityType(StrEnum):
    """
    Supported AI Runtime capability categories.

    Capabilities are explicit Runtime contracts. Adding a new
    capability requires an explicit enum member so that the
    capability can be registered, governed, resolved, executed,
    and tested through the Runtime.

    The capability itself does not imply autonomous execution.
    In particular, PODCAST_LISTENING_PROPOSAL represents a
    passive proposal to the learner.
    """

    LIVE_INTELLIGENCE = "live_intelligence"

    PODCAST_DISCOVERY = "podcast_discovery"

    PODCAST_LISTENING_PROPOSAL = (
        "podcast_listening_proposal"
    )

    LEARNING_GENERATION = "learning_generation"

    EVALUATION = "evaluation"


class CapabilityExecutionStatus(StrEnum):
    """
    Runtime-level execution status.
    """

    ACCEPTED = "accepted"

    COMPLETED = "completed"

    REJECTED = "rejected"

    FAILED = "failed"


@dataclass(frozen=True, slots=True)
class CapabilityRequest:
    """
    Provider-independent request submitted to the AI Runtime.

    Attributes:
        capability:
            Logical capability requested by the application.

        request_id:
            Correlation identifier for the execution.

        payload:
            Explicit capability input.

        metadata:
            Non-business execution metadata.

        requested_by:
            Component or application requesting execution.
    """

    capability: CapabilityType

    request_id: str

    payload: Mapping[str, Any]

    metadata: Mapping[str, Any] = field(
        default_factory=dict
    )

    requested_by: str = "application"

    def __post_init__(self) -> None:
        if not self.request_id.strip():
            raise ValueError(
                "Capability request_id cannot be empty."
            )

        if not self.requested_by.strip():
            raise ValueError(
                "Capability requested_by cannot be empty."
            )


@dataclass(frozen=True, slots=True)
class CapabilityResult:
    """
    Provider-independent result returned by the AI Runtime.

    The result contains execution information but does not expose
    provider-specific response objects.
    """

    request_id: str

    capability: CapabilityType

    status: CapabilityExecutionStatus

    output: Any = None

    error: str | None = None

    metadata: Mapping[str, Any] = field(
        default_factory=dict
    )

    @property
    def succeeded(self) -> bool:
        """
        Return True when execution completed successfully.
        """

        return (
            self.status
            == CapabilityExecutionStatus.COMPLETED
        )

    @property
    def failed(self) -> bool:
        """
        Return True when execution failed.
        """

        return (
            self.status
            == CapabilityExecutionStatus.FAILED
        )

    @property
    def rejected(self) -> bool:
        """
        Return True when execution was rejected before
        capability execution.
        """

        return (
            self.status
            == CapabilityExecutionStatus.REJECTED
        )
