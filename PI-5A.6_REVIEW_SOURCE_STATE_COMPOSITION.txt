"""Application port for the canonical Platform Intelligence review projection."""
from __future__ import annotations

from abc import ABC, abstractmethod
from collections.abc import Sequence
from typing import Any


class PlatformIntelligenceReviewSource(ABC):
    """Read-only application port for PI human-review projections.

    The backend deliberately depends on this port rather than importing or
    constructing Platform Intelligence runtime services. A production adapter
    can bind this port to the canonical ``ai_agents`` runtime integration.
    """

    @abstractmethod
    def list_reviews(self) -> Sequence[Any]:
        """Return current governed review projections."""
        raise NotImplementedError

    @abstractmethod
    def get_review(self, review_item_id: str) -> Any | None:
        """Return one governed review projection, if available."""
        raise NotImplementedError

    @abstractmethod
    def list_escalations(self) -> Sequence[Any]:
        """Return governed review projections with escalation recommendations."""
        raise NotImplementedError


class UnconfiguredPlatformIntelligenceReviewSource(PlatformIntelligenceReviewSource):
    """Safe inert source object when the canonical PI integration is absent.

    The application API intentionally uses ``source=None`` to distinguish
    an unconfigured integration from a valid empty result set. This class is
    retained as an explicit port implementation for composition/tests and is
    not used as the default API source.
    """

    def list_reviews(self) -> Sequence[Any]:
        return ()

    def get_review(self, review_item_id: str) -> Any | None:
        return None

    def list_escalations(self) -> Sequence[Any]:
        return ()

class PlatformIntelligenceReviewNotConfigured(RuntimeError):
    """Raised when the canonical PI review source is not configured."""


@dataclass(frozen=True, slots=True)
class PlatformIntelligenceReviewAPI:
    """Read-only application boundary for PI review projections."""

    source: PlatformIntelligenceReviewSource | None = None

    @property
    def configured(self) -> bool:
        return self.source is not None

    def _require_source(self) -> PlatformIntelligenceReviewSource:
        if self.source is None:
            raise PlatformIntelligenceReviewNotConfigured(
                "Platform Intelligence review source is not configured."
            )

        return self.source

    def list_reviews(self) -> dict[str, Any]:
        source = self._require_source()

        items = [
            self._serialize_projection(item)
            for item in source.list_reviews()
        ]

        return {
            "status": "ready",
            "count": len(items),
            "items": items,
        }

    def get_review(self, review_item_id: str) -> dict[str, Any] | None:
        if not review_item_id or not review_item_id.strip():
            raise ValueError("review_item_id must not be blank.")

        source = self._require_source()
        item = source.get_review(review_item_id)

        if item is None:
            return None

        return self._serialize_projection(item)

    def list_escalations(self) -> dict[str, Any]:
        source = self._require_source()

        items = [
            self._serialize_projection(item)
            for item in source.list_escalations()
        ]

        recommended = [
            item
            for item in items
            if item.get("recommendation_status") == "RECOMMENDED"
        ]

        return {
            "status": "ready",
            "count": len(recommended),
            "items": recommended,
        }

    @staticmethod
    def _serialize_projection(item: Any) -> dict[str, Any]:
        """Expose only the approved PI review projection fields."""

        def value(attribute: str) -> Any:
            result = getattr(item, attribute, None)

            if hasattr(result, "value"):
                return result.value

            return result

        created_at = getattr(item, "created_at", None)
        assigned_at = getattr(item, "assigned_at", None)
        review_due_at = getattr(item, "review_due_at", None)

        return {
            "review_item_id": value("review_item_id"),
            "risk_decision_id": value("risk_decision_id"),
            "risk_assessment_id": value("risk_assessment_id"),
            "scope": value("scope"),
            "risk_level": value("risk_level"),
            "risk_score": value("risk_score"),
            "priority": value("priority"),
            "review_status": value("review_status"),
            "decision_status": value("decision_status"),
            "quality_gate_status": value("quality_gate_status"),
            "evidence_references": tuple(
                value("evidence_references") or ()
            ),
            "historical_references": tuple(
                value("historical_references") or ()
            ),
            "policy_id": value("policy_id"),
            "policy_version": value("policy_version"),
            "created_at": (
                created_at.isoformat()
                if created_at is not None
                else None
            ),
            "assigned_reviewer": value("assigned_reviewer"),
            "assigned_at": (
                assigned_at.isoformat()
                if assigned_at is not None
                else None
            ),
            "review_due_at": (
                review_due_at.isoformat()
                if review_due_at is not None
                else None
            ),
            "rationale": value("rationale"),
            "recommendation_status": value("recommendation_status"),
            "recommendation_route": value("recommendation_route"),
            "recommendation_reasons": tuple(
                value("recommendation_reasons") or ()
            ),
            "recommendation_id": value("recommendation_id"),
        }

    def __all_projection_fields(self) -> tuple[str, ...]:
        """Private marker preventing accidental metadata expansion."""
        return ()

class PlatformIntelligenceRetryRunStatus(StrEnum):
    SUCCEEDED = "SUCCEEDED"
    SKIPPED = "SKIPPED"
    FAILED = "FAILED"


@dataclass(frozen=True, slots=True)
class PlatformIntelligenceRetryState:
    """Authoritative state required immediately before retry dispatch."""

    notification_key: str
    recommendation_id: str
    recommendation_status: str
    review_status: str
    decision_status: str
    quality_gate_status: str
    dispatch_request: PlatformIntelligenceEscalationDispatchRequest
    email_request: Any

    def __post_init__(self) -> None:
        for field_name in (
            "notification_key",
            "recommendation_id",
            "recommendation_status",
            "review_status",
            "decision_status",
            "quality_gate_status",
        ):
            value = getattr(self, field_name)
            if not isinstance(value, str) or not value.strip():
                raise ValueError(f"{field_name} must be nonblank.")

        if self.dispatch_request.notification_key != self.notification_key:
            raise ValueError(
                "dispatch_request.notification_key must match notification_key."
            )
        if self.dispatch_request.recommendation_id != self.recommendation_id:
            raise ValueError(
                "dispatch_request.recommendation_id must match recommendation_id."
            )


@dataclass(frozen=True, slots=True)
class PlatformIntelligenceRetryRunResult:
    notification_key: str
    status: PlatformIntelligenceRetryRunStatus
    reason: str
    attempt: int | None = None
    provider_message_id: str | None = None

    def __post_init__(self) -> None:
        if not self.notification_key.strip():
            raise ValueError("notification_key must be nonblank.")
        if not self.reason.strip():
            raise ValueError("reason must be nonblank.")
        if self.attempt is not None and self.attempt < 1:
            raise ValueError("attempt must be at least 1.")

class PlatformIntelligenceEscalationRetryOrchestrator:
    """Coordinates policy, durable claim, state revalidation and retry.

    This service deliberately does not discover candidates, implement retry
    policy, send email directly, or mutate human-review/risk state.

    The canonical retry execution service remains the owner of retry-policy
    enforcement and existing dispatch integration. The orchestrator only
    supplies the authoritative dispatch context after the durable claim and
    final state revalidation.
    """

    def __init__(
        self,
        *,
        state_reader: PlatformIntelligenceRetryStateReader,
        policy_evaluator: PlatformIntelligenceRetryPolicyEvaluator,
        claim_service: PlatformIntelligenceEscalationRetryClaimService,
        retry_execution: PlatformIntelligenceRetryExecutor,
    ) -> None:
        self._state_reader = state_reader
        self._policy_evaluator = policy_evaluator
        self._claim_service = claim_service
        self._retry_execution = retry_execution

    async def run_candidate(
        self,
        *,
        candidate: PlatformIntelligenceEscalationRetryCandidate,
        reference_at: datetime,
    ) -> PlatformIntelligenceRetryRunResult:
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("reference_at must be timezone-aware.")

        eligible = await self._policy_evaluator.evaluate(
            candidate=candidate,
            reference_at=reference_at,
        )
        if not eligible:
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason="RETRY_POLICY_NOT_ELIGIBLE",
                attempt=candidate.attempt,
            )

        claim = await self._claim_service.claim(
            notification_key=candidate.notification_key,
            claimed_at=reference_at,
        )
        if claim is None:
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason="RETRY_CANDIDATE_ALREADY_CLAIMED",
                attempt=candidate.attempt,
            )

        state = await self._state_reader.get_retry_state(
            notification_key=candidate.notification_key,
            recommendation_id=candidate.recommendation_id,
        )
        if state is None:
            return self._skipped(candidate, "AUTHORITATIVE_STATE_NOT_FOUND")
        if state.notification_key != candidate.notification_key:
            return self._skipped(candidate, "NOTIFICATION_KEY_MISMATCH")
        if state.recommendation_id != candidate.recommendation_id:
            return self._skipped(candidate, "RECOMMENDATION_ID_MISMATCH")
        if state.recommendation_status != "RECOMMENDED":
            return self._skipped(
                candidate, "RECOMMENDATION_NO_LONGER_RECOMMENDED"
            )
        if state.review_status != "IN_REVIEW":
            return self._skipped(candidate, "REVIEW_NOT_IN_REVIEW")
        if state.decision_status != "PENDING_HUMAN_REVIEW":
            return self._skipped(
                candidate, "DECISION_NO_LONGER_PENDING_HUMAN_REVIEW"
            )
        if state.quality_gate_status != "PASSED":
            return self._skipped(
                candidate, "QUALITY_GATE_NO_LONGER_PASSED"
            )

        execution = await self._retry_execution.retry(
            request=state.dispatch_request,
            email_request=state.email_request,
            reference_at=reference_at,
        )

        if (
            execution.status
            is PlatformIntelligenceEscalationRetryExecutionStatus.RETRIED
        ):
            dispatch_result = execution.dispatch_result
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SUCCEEDED,
                reason=execution.reason,
                attempt=execution.attempt,
                provider_message_id=(
                    dispatch_result.provider_message_id
                    if dispatch_result is not None
                    else None
                ),
            )

        if execution.status in (
            PlatformIntelligenceEscalationRetryExecutionStatus.NOT_ELIGIBLE,
            PlatformIntelligenceEscalationRetryExecutionStatus.NOT_DUE,
        ):
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason=execution.reason,
                attempt=execution.attempt,
            )

        return PlatformIntelligenceRetryRunResult(
            notification_key=candidate.notification_key,
            status=PlatformIntelligenceRetryRunStatus.FAILED,
            reason=execution.reason,
            attempt=execution.attempt,
        )

    @staticmethod
    def _skipped(
        candidate: PlatformIntelligenceEscalationRetryCandidate,
        reason: str,
    ) -> PlatformIntelligenceRetryRunResult:
        return PlatformIntelligenceRetryRunResult(
            notification_key=candidate.notification_key,
            status=PlatformIntelligenceRetryRunStatus.SKIPPED,
            reason=reason,
            attempt=candidate.attempt,
        )

"""Governed Platform Intelligence escalation email workflow.

This application service prepares and dispatches an escalation notification only
when a governed PI escalation recommendation is present.

It does not create, accept, reject, or otherwise change the underlying
risk decision.
"""

from __future__ import annotations

import inspect
from dataclasses import dataclass, replace
from typing import Any, Protocol


@dataclass(frozen=True, slots=True)
class PlatformIntelligenceEscalationEmailRequest:
    """Immutable, traceable email request for a PI escalation recommendation."""

    recommendation_id: str
    risk_decision_id: str
    review_item_id: str
    scope: str
    risk_level: str
    risk_score: float
    quality_gate_status: str
    review_status: str
    decision_status: str
    route: str
    reasons: tuple[str, ...]
    evidence_references: tuple[str, ...]
    historical_references: tuple[str, ...]
    policy_id: str
    policy_version: str
    rationale: str
    generated_at: str
    review_url: str | None = None
    notification_key: str = ""
    delivery: Any | None = None

    def __post_init__(self) -> None:
        required = {
            "recommendation_id": self.recommendation_id,
            "risk_decision_id": self.risk_decision_id,
            "review_item_id": self.review_item_id,
            "scope": self.scope,
            "risk_level": self.risk_level,
            "quality_gate_status": self.quality_gate_status,
            "review_status": self.review_status,
            "decision_status": self.decision_status,
            "route": self.route,
            "policy_id": self.policy_id,
            "policy_version": self.policy_version,
            "rationale": self.rationale,
            "generated_at": self.generated_at,
        }

        for name, value in required.items():
            if not isinstance(value, str) or not value.strip():
                raise ValueError(f"{name} is required.")

        if not 0.0 <= self.risk_score <= 1.0:
            raise ValueError("risk_score must be between 0 and 1.")

        if self.quality_gate_status != "PASSED":
            raise ValueError(
                "Escalation email requires a passed quality gate."
            )

        if not self.notification_key.strip():
            raise ValueError("notification_key is required.")

        if self.decision_status != "PENDING_HUMAN_REVIEW":
            raise ValueError(
                "Escalation email requires a pending human decision."
            )

        if self.review_status != "IN_REVIEW":
            raise ValueError(
                "Escalation email requires an IN_REVIEW item."
            )

        if any(not value.strip() for value in self.reasons):
            raise ValueError("reasons cannot contain blanks.")

        if any(
            not value.strip()
            for value in self.evidence_references
        ):
            raise ValueError(
                "evidence_references cannot contain blanks."
            )

        if any(
            not value.strip()
            for value in self.historical_references
        ):
            raise ValueError(
                "historical_references cannot contain blanks."
            )


class PlatformIntelligenceEscalationEmailSender(Protocol):
    """Provider-independent delivery boundary."""

    def send(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
        *,
        subject: str,
        body: str,
        html_body: str,
    ) -> object:
        ...


class PlatformIntelligenceEscalationEmailService:
    """Prepare and dispatch governed PI escalation notifications.

    The service requires the caller to supply the authoritative PI review
    projection. It does not query the runtime, risk engine, or repositories.
    """

    workflow_id = "platform_intelligence_escalation"
    template_id = "platform_intelligence_escalation"

    def __init__(
        self,
        *,
        sender: PlatformIntelligenceEscalationEmailSender,
        workflow_active: bool = True,
        frontend_base_url: str | None = None,
    ) -> None:
        if sender is None:
            raise ValueError("sender is required.")

        self._sender = sender
        self._workflow_active = workflow_active
        self._frontend_base_url = (
            (frontend_base_url or "").strip().rstrip("/")
        )

    def prepare(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
    ) -> PlatformIntelligenceEscalationEmailRequest:
        if request is None:
            raise ValueError("request is required.")

        if not self._workflow_active:
            raise ValueError(
                "Platform Intelligence escalation email workflow "
                "is not active."
            )

        if not request.route.strip():
            raise ValueError("Escalation route is required.")

        return request

    def build_subject(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
    ) -> str:
        self.prepare(request)

        return (
            "[EasyWorld Platform Intelligence] "
            f"{request.risk_level} escalation review required "
            f"— {request.scope}"
        )

    def build_body(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
    ) -> str:
        self.prepare(request)

        lines = [
            "EASYWORLD · PLATFORM INTELLIGENCE",
            "",
            "HUMAN ESCALATION REVIEW REQUIRED",
            "",
            (
                "A governed Platform Intelligence escalation "
                "recommendation has been generated."
            ),
            (
                "This notification is a recommendation for human "
                "review only. It does not"
            ),
            (
                "constitute approval, rejection, remediation "
                "authorization, or production action."
            ),
            "",
            "RISK",
            f"Scope: {request.scope}",
            f"Risk level: {request.risk_level}",
            f"Risk score: {request.risk_score:.4f}",
            f"Escalation route: {request.route}",
            "",
            "TRACEABILITY",
            f"Recommendation ID: {request.recommendation_id}",
            f"Risk decision ID: {request.risk_decision_id}",
            f"Review item ID: {request.review_item_id}",
            f"Generated at: {request.generated_at}",
            "",
            "GOVERNANCE",
            f"Quality gate: {request.quality_gate_status}",
            f"Review status: {request.review_status}",
            f"Decision status: {request.decision_status}",
            f"Policy: {request.policy_id} / {request.policy_version}",
            "",
            "ESCALATION REASONS",
            *[
                f"- {reason}"
                for reason in request.reasons
            ],
            "",
            "EVIDENCE REFERENCES",
            *[
                f"- {reference}"
                for reference in request.evidence_references
            ],
            "",
            "HISTORICAL REFERENCES",
            *[
                f"- {reference}"
                for reference in request.historical_references
            ],
            "",
            "ASSESSMENT RATIONALE",
            request.rationale,
            "",
            "ACTION REQUIRED",
            (
                "Open the authoritative Platform Intelligence review "
                "record and complete the human review in the controlled "
                "system. Do not approve, reject, or authorize remediation "
                "from the email itself. Any SLA or response target is "
                "governed by the authoritative review record and "
                "escalation policy."
            ),
            "",
            "DECISION CONTROL",
            (
                "Email reply does not constitute acceptance, rejection, "
                "or authorization."
            ),
        ]

        if request.review_url:
            lines.extend(
                [
                    "",
                    "REVIEW",
                    request.review_url,
                ]
            )

        return "\n".join(lines)

    def build_html(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
    ) -> str:
        self.prepare(request)
        return _build_html(
            request,
            self._frontend_base_url,
        )

    def send(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
    ) -> PlatformIntelligenceEscalationEmailRequest:
        self.prepare(request)

        subject = self.build_subject(request)
        body = self.build_body(request)
        html_body = self.build_html(request)

        delivery = self._sender.send(
            request,
            subject=subject,
            body=body,
            html_body=html_body,
        )

        if inspect.isawaitable(delivery):
            return replace(
                request,
                delivery=delivery,
            )

        return request
