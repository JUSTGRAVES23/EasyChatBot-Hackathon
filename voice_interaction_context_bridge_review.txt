"""
Interaction Context Builder.

Application service responsible for composing
the complete UAI interaction context.

Responsibilities:
- Combine domain contexts.
- Coordinate runtime preparation.
- Create InteractionContext.

Does NOT:
- Call AI providers.
- Create LiveKit sessions.
- Persist memory.
- Execute agents.
"""

from __future__ import annotations

import logging

from app.domain.uai.context.interaction_context import (
    InteractionContext,
)

from app.domain.uai.context.learner_context import (
    LearnerContext,
)

from app.domain.uai.context.voice_context import (
    VoiceContext,
)

from app.domain.uai.context.world_context import (
    WorldContext,
)


from app.application.uai.character.character_resolver import (
    CharacterResolver,
)

from app.application.uai.world.world_resolver import (
    WorldResolver,
)


logger = logging.getLogger(__name__)


class InteractionContextBuilder:
    """
    Builds the complete UAI interaction context.
    """

    def __init__(
        self,
        *,
        world_resolver: WorldResolver,
        character_resolver: CharacterResolver,
    ) -> None:

        self._world_resolver = (
            world_resolver
        )

        self._character_resolver = (
            character_resolver
        )


    def build(
        self,
        *,
        learner: LearnerContext,
        world_id: str,
        character_id: str,
        voice: VoiceContext,
    ) -> InteractionContext:
        """
        Build interaction context.
        """

        logger.info(
            "Building interaction context",
            extra={
                "world_id": world_id,
                "character_id": character_id,
            },
        )


        world = (
            self._world_resolver
            .resolve(world_id)
        )


        character = (
            self._character_resolver
            .resolve(character_id)
        )


        context = InteractionContext(
            learner=learner,
            world=world,
            character=character,
            voice=voice,
        )


        logger.info(
            "Interaction context created",
        )


        return context
"""
Learner Context.

Represents learner information required
for AI interaction.

This object does not handle:
- authentication
- user accounts
- payments
- persistence

It only provides learning context.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(
    frozen=True,
    slots=True,
)
class LearnerContext:
    """
    Immutable learner context.
    """

    learner_id: str

    level: str | None = None

    native_language: str | None = None

    learning_goals: tuple[str, ...] = ()

    def __post_init__(self) -> None:
        """
        Validate learner context.
        """

        if not self.learner_id.strip():
            raise ValueError(
                "learner_id cannot be empty."
            )

        if self.level is not None:
            if not self.level.strip():
                raise ValueError(
                    "level cannot be blank."
                )

        if self.native_language is not None:
            if not self.native_language.strip():
                raise ValueError(
                    "native_language cannot be blank."
                )
"""
World Context.

Represents the active learning world
for an AI interaction.

Examples:

- French World
- Spanish World
- Business French World

The world determines:
- language
- learning environment
- available experience rules
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(
    frozen=True,
    slots=True,
)
class WorldContext:
    """
    Immutable learning world context.
    """

    world_id: str

    name: str

    language_code: str

    available_characters: tuple[str, ...] = ()

    def __post_init__(self) -> None:
        """
        Validate world context.
        """

        if not self.world_id.strip():
            raise ValueError(
                "world_id cannot be empty."
            )

        if not self.name.strip():
            raise ValueError(
                "world name cannot be empty."
            )

        if not self.language_code.strip():
            raise ValueError(
                "language_code cannot be empty."
            )
"""
Character Context.

Represents the selected AI character
for one interaction.

The character remains separate from:
- AI execution
- providers
- infrastructure
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(
    frozen=True,
    slots=True,
)
class CharacterContext:
    """
    Immutable character interaction context.
    """

    character_id: str

    name: str

    personality_reference: str

    teaching_style: str

    premium_required: bool = False

    def __post_init__(self) -> None:
        """
        Validate character context.
        """

        if not self.character_id.strip():
            raise ValueError(
                "character_id cannot be empty."
            )

        if not self.name.strip():
            raise ValueError(
                "character name cannot be empty."
            )

        if not self.personality_reference.strip():
            raise ValueError(
                "personality_reference cannot be empty."
            )
