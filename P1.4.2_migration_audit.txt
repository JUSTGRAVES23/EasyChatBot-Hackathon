from app.infrastructure.persistence import beta_access_models  # noqa: F401


config = context.config

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def get_database_url() -> str:
    """Return the configured database URL."""
    return database_url()


def run_migrations_offline() -> None:
    """Run migrations without creating a database connection."""
    url = get_database_url()

    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        compare_type=True,
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations using a live database connection."""
    configuration = config.get_section(config.config_ini_section) or {}
    configuration["sqlalchemy.url"] = get_database_url()

    connectable = engine_from_config(
        configuration,
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata,
            compare_type=True,
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

from __future__ import annotations

import os

from sqlalchemy import create_engine
from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker


class Base(DeclarativeBase):
    """Canonical SQLAlchemy metadata registry for the platform."""

    pass


def database_url() -> str:
    """Return the canonical PostgreSQL connection URL."""

    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for PostgreSQL persistence."
        )

    return value


def create_database_engine():
    """Create the existing synchronous PostgreSQL engine.

    This API is preserved for the existing Beta 1 persistence layer.
    """

    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(
    engine,
) -> sessionmaker[Session]:
    """Create the existing synchronous session factory.

    Preserved for existing Beta 1 repositories.
    """

    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )


def async_database_url() -> str:
    """Return DATABASE_URL adapted for SQLAlchemy async PostgreSQL.

    The environment remains the single source of truth. We only adapt the
    driver at the infrastructure boundary when the configured URL uses the
    standard PostgreSQL SQLAlchemy scheme.
    """

    value = database_url()

    if value.startswith("postgresql+asyncpg://"):
        return value

    if value.startswith("postgresql://"):
        return value.replace(
            "postgresql://",
            "postgresql+asyncpg://",
            1,
        )

    raise RuntimeError(
        "DATABASE_URL must use a PostgreSQL URL compatible with "
        "the synchronous or async PostgreSQL persistence boundary."
    )


def create_async_database_engine() -> AsyncEngine:
    """Create the canonical asynchronous PostgreSQL engine.

    This is used by durable WS-22 persistence while sharing the same
    DATABASE_URL and canonical Base as the existing persistence layer.
    """

    return create_async_engine(
        async_database_url(),
        pool_pre_ping=True,
    )


def create_async_session_factory(
    engine: AsyncEngine,
) -> async_sessionmaker[AsyncSession]:
    """Create the canonical asynchronous PostgreSQL session factory."""

    return async_sessionmaker(
        bind=engine,
        class_=AsyncSession,
        expire_on_commit=False,
    )

class PostgreSQLProviderUsageRepository(ProviderUsageRepository):
    """PostgreSQL implementation of the provider usage repository.

    Uses the existing synchronous SQLAlchemy session boundary because the
    ProviderUsageRepository contract is synchronous.

    def __init__(self, session_factory) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")
        self._session_factory = session_factory

    def save(self, event: ProviderUsageEvent) -> None:
        if event is None:
            raise ValueError("Provider usage event cannot be None.")

        with self._session_factory() as session:
            session.add(
                ProviderUsageRecord(
                    provider=event.provider,
                    model=event.model,
                    capability=event.capability,
                    functionality=event.functionality,
                    agent=event.agent,
                    input_tokens=event.input_tokens,
                    output_tokens=event.output_tokens,
                    latency_ms=event.latency_ms,
                    estimated_cost=event.estimated_cost,
                    quality_score=event.quality_score,
                    success=event.success,
                    created_at=event.created_at,
                )
            )
            session.commit()

    def get_all(self) -> list[ProviderUsageEvent]:
        with self._session_factory() as session:
            rows = (
                session.query(ProviderUsageRecord)
                .order_by(ProviderUsageRecord.created_at.asc(), ProviderUsageRecord.id.asc())
                .all()
            )

        return [_event_from_record(row) for row in rows]

    def clear(self) -> None:
        with self._session_factory() as session:
            session.query(ProviderUsageRecord).delete()
            session.commit()

"""
AI Runtime usage intelligence domain models.

Represents AI Runtime execution consumption.

This module does not:

- call providers;
- calculate external billing;
- access databases;
- execute agents.

It only defines usage intelligence contracts.

Architecture:

Capability Execution
        |
        v
AI Runtime Usage Event
        |
        +--> Cost Intelligence
        |
        +--> Quality Intelligence
        |
        +--> Provider Analytics
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from decimal import Decimal


@dataclass(frozen=True, slots=True)
class ProviderUsageEvent:
    """
    Immutable record of one AI Runtime execution.

    Created after a Runtime capability execution.

    Example:

        Capability
             |
             v
        Agent
             |
             v
        Provider
             |
             v
        Usage Event
    """

    provider: str

    model: str

    capability: str

    functionality: str

    agent: str

    input_tokens: int

    output_tokens: int

    latency_ms: int

    estimated_cost: Decimal

    quality_score: float

    success: bool

    created_at: datetime = field(
        default_factory=lambda:
            datetime.now(timezone.utc)
    )

    def __post_init__(self) -> None:

        required_fields = {
            "provider": self.provider,
            "model": self.model,
            "capability": self.capability,
            "functionality": self.functionality,
            "agent": self.agent,
        }

        for name, value in required_fields.items():

            if not value.strip():
                raise ValueError(
                    f"{name} cannot be empty."
                )

        if self.input_tokens < 0:
            raise ValueError(
                "Input tokens cannot be negative."
            )

        if self.output_tokens < 0:
            raise ValueError(
                "Output tokens cannot be negative."
            )

        if self.latency_ms < 0:
            raise ValueError(
                "Latency cannot be negative."
            )

        if self.estimated_cost < 0:
            raise ValueError(
                "Estimated cost cannot be negative."
            )

        if not 0 <= self.quality_score <= 100:
            raise ValueError(
                "Quality score must be between 0 and 100."
            )


@dataclass(frozen=True, slots=True)
class ProviderUsageSummary:
    """
    Aggregated AI Runtime usage intelligence.

    Used later by:

    - Operations Centre;
    - governance dashboards;
    - provider optimisation;
    - cost monitoring.
    """

    total_requests: int

    total_input_tokens: int

    total_output_tokens: int

    successful_requests: int

    failed_requests: int

    providers: tuple[str, ...]

    total_cost: Decimal = Decimal("0")

    @property
    def total_tokens(self) -> int:
        return (
            self.total_input_tokens
            + self.total_output_tokens
        )

    @property
    def success_rate(self) -> float:

        if self.total_requests == 0:
            return 0.0

        return (
            self.successful_requests
            / self.total_requests
        ) * 100

AI Runtime provider usage repository.

Defines the persistence boundary for AI Runtime

This module does not:

Responsibilities:

- store usage events;

ProviderUsageRepository

from __future__ import annotations

from abc import ABC, abstractmethod

from src.ai_runtime.providers.usage.models import (
    ProviderUsageEvent,
)


class ProviderUsageRepository(ABC):
    """
    Persistence contract for provider usage events.

    Application services depend on this abstraction,
    not directly on storage technology.
    """

    @abstractmethod
    def save(
        self,
        event: ProviderUsageEvent,
    ) -> None:
        """
        Store one provider usage event.
        """

        raise NotImplementedError(
            "Repository save must be implemented."
        )

    @abstractmethod
    def get_all(
        self,
    ) -> list[ProviderUsageEvent]:
        """
        Retrieve stored provider usage events.
        """

        raise NotImplementedError(
            "Repository get_all must be implemented."
        )

    @abstractmethod
    def clear(
        self,
    ) -> None:
        """
        Remove stored usage events.

        Mainly used for testing.
        """

        raise NotImplementedError(
            "Repository clear must be implemented."
        )


class InMemoryProviderUsageRepository(
    ProviderUsageRepository,
):
    """
    In-memory repository implementation.

    Used for:

    - development;
    - testing;
    - hackathon validation.

    This implementation can later be replaced
    by a database-backed repository.
    """

    def __init__(
        self,
    ) -> None:

        self._events: list[
            ProviderUsageEvent
        ] = []

    def save(
        self,
        event: ProviderUsageEvent,
    ) -> None:
        """
        Save a usage event.
        """

        if event is None:
            raise ValueError(
                "Provider usage event cannot be None."
            )

        self._events.append(
            event
        )

    def get_all(
        self,
    ) -> list[ProviderUsageEvent]:
        """
        Return all stored usage events.

        A copy is returned to prevent
        external mutation.
        """

        return list(
            self._events
        )

    def clear(
        self,
    ) -> None:
        """
        Remove all stored usage events.
        """

        self._events.clear()
"""
AI Runtime provider usage service.

Application service responsible for collecting
and aggregating AI Runtime usage events.

Responsibilities:

- record usage events;
- retrieve usage events through repository;
- generate usage summaries.

Non-responsibilities:

- provider execution;
- API calls;
- billing APIs;
- database implementation;
- dashboard rendering;
- agent execution.

Persistence is handled through the repository
boundary.
"""

from __future__ import annotations

from decimal import Decimal

from src.ai_runtime.providers.usage.models import (
    ProviderUsageEvent,
    ProviderUsageSummary,
)

from src.ai_runtime.providers.usage.repository import (
    ProviderUsageRepository,
)


class ProviderUsageService:
    """
    Application service for AI Runtime usage intelligence.

    The service depends on the repository abstraction.

    It does not know whether storage is:

    - memory;
    - SQLite;
    - PostgreSQL;
    - analytics warehouse.
    """

    def __init__(
        self,
        repository: ProviderUsageRepository,
    ) -> None:

        if repository is None:
            raise ValueError(
                "ProviderUsageRepository cannot be None."
            )

        self._repository = repository

    def record(
        self,
        event: ProviderUsageEvent,
    ) -> None:
        """
        Record one Runtime usage event.

        The event is delegated to the repository.

        The service does not:

        - call providers;
        - enrich events;
        - calculate external billing.
        """

        if event is None:
            raise ValueError(
                "Provider usage event cannot be None."
            )

        self._repository.save(
            event
        )

    def summary(
        self,
    ) -> ProviderUsageSummary:
        """
        Generate aggregated usage information.
        """

        events = (
            self._repository.get_all()
        )

        total_requests = len(
            events
        )

        total_input_tokens = sum(
            event.input_tokens
            for event in events
        )

        total_output_tokens = sum(
            event.output_tokens
            for event in events
        )

        successful_requests = sum(
            1
            for event in events
            if event.success
        )

        failed_requests = (
            total_requests
            - successful_requests
        )

        providers = tuple(
            sorted(
                {
                    event.provider
                    for event in events
                }
            )
        )

        total_cost = sum(
            (
                event.estimated_cost
                for event in events
            ),
            Decimal("0"),
        )

        return ProviderUsageSummary(
            total_requests=(
                total_requests
            ),
            total_input_tokens=(
                total_input_tokens
            ),
            total_output_tokens=(
                total_output_tokens
            ),
            successful_requests=(
                successful_requests
            ),
            failed_requests=(
                failed_requests
            ),
            providers=providers,
            total_cost=total_cost,
        )

    def clear(
        self,
    ) -> None:
        """
        Clear repository data.

        Mainly useful for:

        - tests;
        - demonstrations.

        Production systems should not rely
        on clearing runtime usage history.
        """

        self._repository.clear()
