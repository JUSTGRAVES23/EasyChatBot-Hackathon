export type GovernanceControl = {
  component: string;
  status: string;
  details: {
    paused?: boolean;
    reason?: string | null;
    changed_at?: string | null;
    [key: string]: unknown;
  };
};

export type GovernanceRecentEvent = {
  event: string;
  actor: string;
  reason: string;
  component: string;
  timestamp: string;
  severity?: "info" | "warning" | "critical" | string;
  evidence_available?: boolean;
};

export type GovernanceStatus = {
  yeti: {
    message: string;
  };
  controls: GovernanceControl[];
  recent_events: GovernanceRecentEvent[];
};

async function readJson<T>(response: Response, endpoint: string): Promise<T> {
  if (!response.ok) {
    throw new Error(
      `Governance API request failed (${response.status}) at ${endpoint}`,
    );
  }

  const contentType = response.headers.get("content-type") ?? "";

  if (!contentType.toLowerCase().includes("application/json")) {
    throw new Error(
      `Governance API returned non-JSON content at ${endpoint}. ` +
      `Check that the Vite proxy is running and FastAPI is available on port 8000.`,
    );
  }

  return response.json() as Promise<T>;
}

export async function getGovernanceStatus(): Promise<GovernanceStatus> {
  const endpoint = "/operations/governance/status";
  const response = await fetch(endpoint, {
    headers: { Accept: "application/json" },
  });

  return readJson<GovernanceStatus>(response, endpoint);
}

function formatDate(
  value?: string | null,
): string {
  if (!value) {
    return "—";
  }

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return new Intl.DateTimeFormat(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(date);
}

import { useMemo, useState } from "react";
import {
  submitGovernanceDecision,
  type GovernanceDecision,
  type GovernanceDecisionRequest,
  type GovernanceRole,
} from "../../lib/privacy-governance/decision-api";
import type {
  PrivacyGovernanceCaseDetail,
} from "../../lib/privacy-governance/types";
import styles from "./PrivacyGovernanceDecisionActions.module.css";

interface Props {
  caseData: PrivacyGovernanceCaseDetail;
  onDecisionComplete: (updatedCase: PrivacyGovernanceCaseDetail) => void;
}

function createApprovalId(
  caseId: string,
  role: GovernanceRole,
): string {
  const suffix =
    typeof crypto !== "undefined" && "randomUUID" in crypto
      ? crypto.randomUUID()
      : `${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;

  return `APR-UI-${role}-${caseId}-${suffix}`;
}

function statusLabel(value?: string | null): string {
  if (!value) return "Unknown";
  return value.replaceAll("_", " ");
}

function decisionLabel(decision: GovernanceDecision): string {
  switch (decision) {
    case "approve":
      return "Approve";
    case "reject":
      return "Reject";
    case "request-changes":
      return "Request changes";
  }
}

export function PrivacyGovernanceDecisionActions({
  caseData,
  onDecisionComplete,
}: Props) {
  const [role, setRole] = useState<GovernanceRole>("DPO");
  const [reviewer, setReviewer] = useState("");
  const [comment, setComment] = useState("");

  const [pendingDecision, setPendingDecision] =
    useState<GovernanceDecision | null>(null);

  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const isPending =
    caseData.governance_status === "PENDING_DPO_GRC";

  const approvalId = useMemo(
    () => createApprovalId(caseData.case_id, role),
    [caseData.case_id, role],
  );

  if (!isPending) {
    return (
      <section className={styles.section}>
        <div className={styles.completedCard}>
          <div>
            <span className={styles.eyebrow}>
              HUMAN GOVERNANCE DECISION
            </span>

            <strong>
              Governance decision recorded
            </strong>

            <p>
              This case is currently{" "}
              <b>{statusLabel(caseData.governance_status)}</b>.
              No additional decision is available while the
              case is outside the pending governance state.
            </p>
          </div>
        </div>
      </section>
    );
  }

  const canSubmit =
    reviewer.trim().length > 0 &&
    comment.trim().length > 0 &&
    Boolean(caseData.change_id) &&
    !submitting;

  async function executeDecision(
    decision: GovernanceDecision,
  ): Promise<void> {
    if (!caseData.change_id) {
      setError(
        "This governance case has no linked privacy change. "
        + "The decision cannot be submitted.",
      );
      return;
    }

    if (!reviewer.trim()) {
      setError("Reviewer identity is required.");
      return;
    }

    if (!comment.trim()) {
      setError("A decision comment is required.");
      return;
    }

    setSubmitting(true);
    setError(null);

    const request: GovernanceDecisionRequest = {
      change_id: caseData.change_id,
      approval_id: approvalId,
      reviewer: reviewer.trim(),
      role,
      comment: comment.trim(),
    };

    try {
      const result = await submitGovernanceDecision(
        caseData.case_id,
        decision,
        request,
      );

      onDecisionComplete(result.case);

      setPendingDecision(null);
    } catch (cause) {
      setError(
        cause instanceof Error
          ? cause.message
          : "Governance decision failed.",
      );
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <section className={styles.section}>
      <div className={styles.heading}>
        <div>
          <span className={styles.eyebrow}>
            HUMAN GOVERNANCE DECISION
          </span>

          <h3>Decision required</h3>

          <p>
            This case passed automated validation and is awaiting
            explicit human governance approval.
          </p>
        </div>

        <span className={styles.pendingBadge}>
          PENDING DPO/GRC
        </span>
      </div>

      <div className={styles.card}>
        <div className={styles.field}>
          <label htmlFor="governance-role">
            Decision role
          </label>

          <select
            id="governance-role"
            value={role}
            onChange={(event) =>
              setRole(event.target.value as GovernanceRole)
            }
            disabled={submitting}
          >
            <option value="DPO">
              DPO
            </option>

            <option value="GRC">
              GRC
            </option>
          </select>
        </div>

        <div className={styles.field}>
          <label htmlFor="governance-reviewer">
            Reviewer
          </label>

          <input
            id="governance-reviewer"
            type="text"
            value={reviewer}
            onChange={(event) =>
              setReviewer(event.target.value)
            }
            placeholder="Enter human reviewer name"
            disabled={submitting}
            autoComplete="off"
          />
        </div>

        <div className={styles.field}>
          <label htmlFor="governance-comment">
            Decision rationale
          </label>

          <textarea
            id="governance-comment"
            value={comment}
            onChange={(event) =>
              setComment(event.target.value)
            }
            placeholder="Record the reason for the governance decision."
            rows={4}
            disabled={submitting}
          />
        </div>

        {!caseData.change_id && (
          <div
            className={styles.warning}
            role="alert"
          >
            This case is missing its Change ID relationship.
            A governance decision is therefore blocked.
          </div>
        )}

        {error && (
          <div
            className={styles.error}
            role="alert"
          >
            {error}
          </div>
        )}

        <div className={styles.actions}>
          <button
            type="button"
            className={`${styles.actionButton} ${styles.approve}`}
            disabled={!canSubmit}
            onClick={() =>
              setPendingDecision("approve")
            }
          >
            Approve as {role}
          </button>

          <button
            type="button"
            className={`${styles.actionButton} ${styles.reject}`}
            disabled={!canSubmit}
            onClick={() =>
              setPendingDecision("reject")
            }
          >
            Reject
          </button>

          <button
            type="button"
            className={`${styles.actionButton} ${styles.changes}`}
            disabled={!canSubmit}
            onClick={() =>
              setPendingDecision("request-changes")
            }
          >
            Request changes
          </button>
        </div>
      </div>

      {pendingDecision && (
        <div
          className={styles.confirmBackdrop}
          role="presentation"
        >
          <div
            className={styles.confirmDialog}
            role="dialog"
            aria-modal="true"
            aria-labelledby="governance-confirm-title"
          >
            <span className={styles.eyebrow}>
              CONFIRM GOVERNANCE DECISION
            </span>

            <h3 id="governance-confirm-title">
              {decisionLabel(pendingDecision)}
            </h3>

            <p>
              You are about to record a formal human governance
              decision for:
            </p>

            <dl className={styles.confirmDetails}>
              <div>
                <dt>Case</dt>
                <dd>{caseData.case_id}</dd>
              </div>

              <div>
                <dt>Change</dt>
                <dd>{caseData.change_id}</dd>
              </div>

              <div>

                <dd>{role}</dd>
              </div>

              <div>
                <dt>Reviewer</dt>
                <dd>{reviewer}</dd>

              This action will be recorded as governance evidence
              and may change the governance state of the case.
            </p>

            <div className={styles.confirmActions}>

                type="button"

function normalizeStatus(status: string): GovernanceFilter {
  const value = status.toUpperCase();

  if (value.includes("PENDING")) {

  if (value.includes("APPROVED")) {
    return "APPROVED";

    return "BLOCKED";
  }

  return "ALL";
}

function statusLabel(value?: string | null): string {
  if (!value) {
    return "Unknown";
  }

  return value.replaceAll("_", " ");
}

function formatDate(value?: string | null): string {
  if (!value) {
    return "—";
  }

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return new Intl.DateTimeFormat(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(date);
}

export function PrivacyGovernanceDashboard({
  initialLoad = true,
  initialCaseId,
}: Props) {
  const [summary, setSummary] =
    useState<PrivacyGovernanceSummary>(
      EMPTY_SUMMARY,
    );

  const [statusChart, setStatusChart] =
    useState<PrivacyGovernanceStatusItem[]>([]);

  const [cases, setCases] =
    useState<PrivacyGovernanceCase[]>([]);

  const [filter, setFilter] =
    useState<GovernanceFilter>("ALL");

  const [loading, setLoading] =
    useState(initialLoad);

  const [refreshing, setRefreshing] =
    useState(false);

  const [error, setError] =
    useState<string | null>(null);

  const [selectedCaseId, setSelectedCaseId] =
    useState<string | null>(null);

  const [selectedCase, setSelectedCase] =
    useState<PrivacyGovernanceCase | null>(null);

  const [detailLoading, setDetailLoading] =
    useState(false);

  const [detailError, setDetailError] =
    useState<string | null>(null);

  /*
   * ---------------------------------------------------------
   * Dashboard data
   * ---------------------------------------------------------
   */

  const loadDashboard = useCallback(
    async (
      mode: "initial" | "refresh" = "refresh",
    ) => {
      if (mode === "initial") {
        setLoading(true);
      } else {
        setRefreshing(true);
      }

      setError(null);

      try {
        const result =
          await loadPrivacyGovernanceDashboard();

        setSummary(result.summary);
        setStatusChart(result.statusChart);
        setCases(result.cases);
      } catch (cause) {
        setError(
          cause instanceof Error
            ? cause.message
            : "Unable to load privacy governance data.",
        );
      } finally {
        setLoading(false);
        setRefreshing(false);
      }
    },
    [],
  );

  useEffect(() => {
    if (initialLoad) {
      void loadDashboard("initial");
    }
  }, [initialLoad, loadDashboard]);

  /*
   * ---------------------------------------------------------
   * Case filtering
   * ---------------------------------------------------------
   */

  const visibleCases = useMemo(() => {
    if (filter === "ALL") {
      return cases;
    }

    return cases.filter(
      (item) =>
        normalizeStatus(
          item.governance_status,
        ) === filter,
    );
  }, [cases, filter]);

  /*
   * ---------------------------------------------------------
   * Case detail
   * ---------------------------------------------------------
   */

  const openCase = useCallback(
    async (caseId: string) => {
      const normalizedCaseId = caseId.trim();

      if (!normalizedCaseId) {
        setSelectedCaseId(null);
        setSelectedCase(null);
        setDetailError(
          "A valid governance case ID is required.",
        );
        return;
      }

      setSelectedCaseId(normalizedCaseId);
      setSelectedCase(null);
      setDetailError(null);
      setDetailLoading(true);

      try {
        const detail =
          await getPrivacyGovernanceCase(
            normalizedCaseId,
          );

        setSelectedCase(detail);
      } catch (cause) {
        setDetailError(
          cause instanceof Error
            ? cause.message
            : "Unable to load this governance case.",
        );
      } finally {
        setDetailLoading(false);
      }
    },
    [],
  );

  /*
   * ---------------------------------------------------------
   * Privacy Governance Email Deep Link
   *
   * /privacy-governance/<actual-case-id>
   *
   * No case ID is hard-coded.
   * ---------------------------------------------------------
   */

  useEffect(() => {
    const caseId = initialCaseId?.trim();

    if (!caseId) {
      return;
    }

    void openCase(caseId);
  }, [initialCaseId, openCase]);

  const retryCase = useCallback(() => {
    if (selectedCaseId) {
      void openCase(selectedCaseId);
    }
  }, [openCase, selectedCaseId]);

  /*
   * ---------------------------------------------------------
   * Decision completed
   * ---------------------------------------------------------
   */

  const handleDecisionComplete = useCallback(
    (updatedCase: PrivacyGovernanceCase) => {
      setSelectedCase(updatedCase);

      setCases((current) =>
        current.map((item) =>
          item.case_id === updatedCase.case_id
            ? updatedCase
            : item,
        ),
      );

      void loadDashboard("refresh");
    },
    [loadDashboard],
  );

  /*
   * ---------------------------------------------------------
   * Status filtering
   * ---------------------------------------------------------
   */

  const filterFromStatus = useCallback(
    (status: string) => {
      setFilter(normalizeStatus(status));
    },
    [],
  );

  /*
   * ---------------------------------------------------------
   * KPI configuration
   * ---------------------------------------------------------
   */

  const kpis: Array<{
    label: string;
    value: number;
    filter: GovernanceFilter;
  }> = [
    {
      label: "Total cases",
      value: summary.total_cases,
      filter: "ALL",
    },
    {
      label: "Pending DPO/GRC",
      value: summary.pending,
      filter: "PENDING_DPO_GRC",
    },
    {
      label: "Approved",
      value: summary.approved,
      filter: "APPROVED",
    },
    {
      label: "Rejected / blocked",
      value: summary.blocked,
      filter: "BLOCKED",
    },
  ];

  /*
   * ---------------------------------------------------------
   * Render
   * ---------------------------------------------------------
   */

  return (
    <main className={styles.page}>
      <header className={styles.header}>
        <div>
          <p className={styles.brand}>
            EASYCHATBOT · GOVERNANCE
          </p>

          <h1>
            Privacy Governance Center
          </h1>

          <p className={styles.subtitle}>
            Read-only oversight of privacy-impacting
            changes, validation and human approval
            evidence.
          </p>
        </div>

        <button
          type="button"
          className={styles.refresh}
          onClick={() =>
            void loadDashboard("refresh")
          }
          disabled={refreshing}
        >
          {refreshing
            ? "Refreshing…"
            : "Refresh"}
        </button>
      </header>

      {error && (
        <section
          className={styles.errorBanner}
          role="alert"
        >
          <div>
            <strong>
              Unable to load governance data.
            </strong>

            <span>{error}</span>
          </div>

          <button
            type="button"
            onClick={() =>
              void loadDashboard("refresh")
            }
          >
            Retry
          </button>
        </section>
      )}

      <section
        className={styles.kpis}
        aria-label="Governance KPIs"
      >
        {kpis.map((kpi) => (
          <button
            type="button"
            key={kpi.label}
            className={`${styles.kpi} ${
              filter === kpi.filter
                ? styles.kpiActive
                : ""
            }`}
            onClick={() =>
              setFilter(kpi.filter)
            }
          >
            <span>{kpi.label}</span>

            <strong>
              {loading
                ? "…"
                : kpi.value}
            </strong>
          </button>
        ))}
      </section>

      <section className={styles.panel}>
        <div className={styles.panelHeader}>
          <div>
            <p className={styles.eyebrow}>
              OVERSIGHT
            </p>

            <h2>
              Governance status
            </h2>
          </div>

          <span>
            {loading
              ? "Loading…"
              : `${summary.approval_records} approval records`}
          </span>
        </div>

        <div className={styles.statusList}>
          {statusChart.length ? (
            statusChart.map((item) => {
              const maximum = Math.max(
                1,
                ...statusChart.map(
                  (entry) => entry.count,
                ),
              );

              return (
                <button
                  type="button"
                  className={`${styles.statusRow} ${
                    filter ===
                    normalizeStatus(
                      item.status,
                    )
                      ? styles.statusActive
                      : ""
                  }`}
                  key={item.status}
                  onClick={() =>
                    filterFromStatus(
                      item.status,
                    )
                  }
                >
                  <span>
                    {statusLabel(
                      item.status,
                    )}
                  </span>

                  <span
                    className={
                      styles.barTrack
                    }
                  >
                    <span
                      className={styles.bar}
                      style={{
                        width: `${
                          (item.count /
                            maximum) *
                          100
                        }%`,
                      }}
                    />
                  </span>

                  <strong>
                    {item.count}
                  </strong>
                </button>
              );
            })
          ) : (
            <div
              className={
                styles.emptyState
              }
            >
              No governance status data.
            </div>
          )}
        </div>
      </section>

      <section className={styles.panel}>
        <div className={styles.panelHeader}>
          <div>
            <p className={styles.eyebrow}>
              CASE REGISTER
            </p>

            <h2>
              Privacy governance cases
            </h2>
          </div>

          <div
            className={
              styles.caseMeta
            }
          >
            {filter !== "ALL" && (
              <button
                type="button"
                className={
                  styles.clearFilter
                }
                onClick={() =>
                  setFilter("ALL")
                }
              >
                {statusLabel(filter)} ×
              </button>
            )}

            <span>
              {loading
                ? "Loading…"
                : `${visibleCases.length} cases`}
            </span>
          </div>
        </div>

        <div
          className={styles.tableWrap}
        >
          <table>
            <thead>
              <tr>
                <th>CASE</th>
                <th>TRIGGER</th>
                <th>TRIGGERED</th>
                <th>RISK</th>
                <th>VALIDATION</th>
                <th>STATUS</th>
              </tr>
            </thead>

            <tbody>
              {!loading &&
                visibleCases.map(
                  (item) => (
                    <tr
                      key={item.case_id}
                    >
                      <td>
                        <button
                          type="button"
                          className={
                            styles.caseLink
                          }
                          onClick={() =>
                            void openCase(
                              item.case_id,
                            )
                          }
                          aria-label={`Open case ${item.case_id}`}
                        >
                          {item.case_id}
                        </button>
                      </td>

                      <td>
                        {statusLabel(
                          item.trigger?.type,
                        )}
                      </td>

                      <td>
                        {formatDate(
                          item.trigger?.at,
                        )}
                      </td>

                      <td>
                        {item.risk ?? "—"}
                      </td>

                      <td>
                        {statusLabel(
                          item.validation_status,
                        )}
                      </td>

                      <td>
                        <span
                          className={
                            styles.statusPill
                          }
                        >
                          {statusLabel(
                            item.governance_status,
                          )}
                        </span>
                      </td>
                    </tr>
                  ),
                )}
            </tbody>
          </table>

          {loading && (
            <div
              className={
                styles.emptyState
              }
              role="status"
            >
              Loading governance cases…
            </div>
          )}

          {!loading &&
            !visibleCases.length && (
              <div
                className={
                  styles.emptyState
                }
              >
                {filter === "ALL"
                  ? "No governance cases are currently registered."
                  : "No cases match the selected filter."}
              </div>
            )}
        </div>
      </section>

      <PrivacyGovernanceCaseDetail
        open={Boolean(
          selectedCaseId,
        )}
        loading={detailLoading}
        error={detailError}
        caseData={selectedCase}
        onClose={() => {
          setSelectedCaseId(null);
          setSelectedCase(null);
          setDetailError(null);
        }}
        onRetry={retryCase}
        onDecisionComplete={
          handleDecisionComplete
        }
      />
    </main>
  );
}

export default PrivacyGovernanceDashboard;

  title: string;

  summary: string;

   * triggered_at properties.

  trigger: PrivacyGovernanceCaseTrigger;

  /**
   * Governance risk classification.
   */
  risk: string;

  /**
   * Authoritative processing identifiers associated
   * with the governance case.
   */
  processing_ids: string[];

  /**
   * Authoritative data-mapping identifiers associated
   * with the governance case.
   */
  data_map_ids: string[];

  /**
   * Result of automated privacy validation.
   */
  validation_status: string;

  /**
   * Email delivery status.

  email_status: string;

  /**
