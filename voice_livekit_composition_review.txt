"""
Voice Call Context Factory.

Application factory responsible for
creating voice execution contexts.

Responsibilities:
- Convert voice call requests into
  domain execution context.
- Create voice sessions.
- Create room participants.
- Build validated execution boundary.

Does NOT:
- Handle HTTP.
- Execute runtime.
- Connect LiveKit.
- Select providers.
- Persist data.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.domain.uai.voice.voice_call_execution_context import (
    VoiceCallExecutionContext,
)


from app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)


from app.domain.uai.voice.voice_call_state import (
    VoiceCallState,
)


from app.domain.uai.voice.voice_call_room import (
    VoiceCallRoom,
)


from app.domain.uai.voice.voice_call_participant import (
    VoiceCallParticipant,
    VoiceParticipantRole,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallContextFactory:
    """
    Creates voice execution contexts.
    """



    def create(
        self,
        *,
        request: VoiceCallRequest,
    ) -> VoiceCallExecutionContext:
        """
        Create execution context.

        Lifecycle starts:

            READY
              |
              â†“
          Runtime ACTIVE
        """


        session = VoiceCallSession(
            session_id=(
                f"{request.character_id}-"
                f"{request.learner_id}"
            ),
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
            state=VoiceCallState.READY,
        )


        room = VoiceCallRoom(
            room_id=session.session_id,
            participants=(
                VoiceCallParticipant(
                    participant_id=request.learner_id,
                    role=(
                        VoiceParticipantRole.LEARNER
                    ),
                    display_name="Learner",
                ),
                VoiceCallParticipant(
                    participant_id=request.character_id,
                    role=(
                        VoiceParticipantRole.CHARACTER
                    ),
                    display_name=(
                        request.character_id
                    ),
                ),
            ),
        )


        return VoiceCallExecutionContext(
            session=session,
            room=room,
        )



__all__ = [
    "VoiceCallContextFactory",
]

from __future__ import annotations


from app.application.uai.voice.voice_character_session_service import (
    VoiceCharacterSessionService,
)


from app.application.uai.voice.voice_call_execution_service import (
    VoiceCallExecutionService,
)


from app.ai_runtime.voice.voice_speech_runtime import (
    VoiceSpeechRuntime,
)


from app.ai_runtime.voice.voice_call_execution_runtime import (
    VoiceCallExecutionRuntime,
)


from app.providers.speech.speech_provider_factory import (
    SpeechProviderFactory,
)


from app.configuration.providers.speech_provider_config import (
    SpeechProviderConfig,
)


from app.infrastructure.voice.livekit_adapter import (
    LiveKitAdapter,
)



def create_voice_character_session_service(
) -> VoiceCharacterSessionService:
    """
    Build voice session application service.
    """


    provider_config = SpeechProviderConfig(
        provider_name="openai",
        model_name="gpt-4o-mini-tts",
        default_language="fr",
    )


    speech_provider = (
        SpeechProviderFactory
        .create(
            provider_config
        )
    )


    speech_runtime = VoiceSpeechRuntime(
        speech_provider=speech_provider,
    )


    livekit_adapter = LiveKitAdapter()


    return VoiceCharacterSessionService(
        livekit_adapter=livekit_adapter,
        speech_runtime=speech_runtime,
    )



def create_voice_call_execution_service(
) -> VoiceCallExecutionService:
    """
    Build voice call execution service.
    """


    execution_runtime = (
        VoiceCallExecutionRuntime()
    )


    return VoiceCallExecutionService(
        execution_runtime=execution_runtime,
    )



__all__ = [
    "create_voice_character_session_service",
    "create_voice_call_execution_service",
]

"""
Voice Execution Composition.

Application composition root
for voice call execution services.

Responsibilities:
- Build execution dependencies.
- Wire application services.

Does NOT:
- Handle HTTP.
- Execute calls.
- Persist data.
- Manage providers.
"""

from __future__ import annotations


from app.application.uai.voice.voice_call_execution_service import (
    VoiceCallExecutionService,
)


from app.ai_runtime.voice.voice_call_execution_runtime import (
    VoiceCallExecutionRuntime,
)



def create_voice_call_execution_service(
) -> VoiceCallExecutionService:
    """
    Create configured voice execution service.
    """


    runtime = (
        VoiceCallExecutionRuntime()
    )


    return VoiceCallExecutionService(
        execution_runtime=runtime,
    )



def build_voice_call_execution_service(
) -> VoiceCallExecutionService:
    """
    Dependency builder.

    Used by application composition
    and adapters.
    """


    return (
        create_voice_call_execution_service()
    )
