from abc import ABC, abstractmethod

    MetricSnapshot,

class ProjectRegistry(ABC):
    @abstractmethod
    def get(self, project_id: str) -> ProjectDefinition | None: ...

    @abstractmethod
    def list_active(self) -> list[ProjectDefinition]: ...

    @abstractmethod
    def save(self, project: ProjectDefinition) -> ProjectDefinition: ...


class ProjectStatusSource(ABC):
    """Read-only adapter over an existing authoritative EasyChatBot source."""

    @property
    @abstractmethod
    def source_id(self) -> str: ...

    @property
    @abstractmethod
    def source_name(self) -> str: ...

    @abstractmethod
    def fetch_workstreams(self, project_id: str) -> list[WorkstreamSnapshot]: ...

    @abstractmethod
    def fetch_metrics(self, project_id: str) -> list[MetricSnapshot]: ...

    @abstractmethod
    def fetch_issues(self, project_id: str) -> list[IssueSnapshot]: ...

    @abstractmethod
    def fetch_risks(self, project_id: str) -> list[RiskSnapshot]: ...

    @abstractmethod
    def fetch_actions(self, project_id: str) -> list[ActionSnapshot]: ...

    @abstractmethod
    def fetch_milestones(self, project_id: str) -> list[MilestoneSnapshot]: ...

    @abstractmethod
    def health(self) -> SourceSnapshot: ...


class SnapshotRepository(ABC):

from __future__ import annotations

from datetime import datetime, timezone
from uuid import uuid4

from ..domain.models import ProjectEvent, ProjectEventType
from ..domain.ports import ProjectEventRepository


class ProjectEventService:
    """Records project-control events; it does not mutate source systems."""

    def __init__(self, repository: ProjectEventRepository) -> None:
        self._repository = repository

    def record(
        self,
        *,
        project_id: str,
        event_type: ProjectEventType,
        summary: str,
        source: str,
        metadata: dict[str, str] | None = None,
    ) -> ProjectEvent:
        event = ProjectEvent(
            event_id=f"pev-{uuid4()}",
            project_id=project_id,
            event_type=event_type,
            occurred_at=datetime.now(timezone.utc),
            summary=summary,
            source=source,
            metadata=tuple(sorted((metadata or {}).items())),
        )
        return self._repository.save(event)

from __future__ import annotations

from dataclasses import dataclass, field

from ..domain.models import ProjectControlSnapshot, ProjectDefinition, ProjectEvent
from ..domain.ports import (
    ProjectEventRepository,
    ProjectRegistry,
    SnapshotRepository,
)


@dataclass
class InMemoryProjectRegistry(ProjectRegistry):
    projects: dict[str, ProjectDefinition] = field(default_factory=dict)

    def get(self, project_id: str) -> ProjectDefinition | None:
        return self.projects.get(project_id)

    def list_active(self) -> list[ProjectDefinition]:
        return [p for p in self.projects.values() if p.active]

    def save(self, project: ProjectDefinition) -> ProjectDefinition:
        self.projects[project.project_id] = project
        return project


@dataclass
class InMemorySnapshotRepository(SnapshotRepository):
    data: dict[str, list[ProjectControlSnapshot]] = field(default_factory=dict)

    def save(self, snapshot: ProjectControlSnapshot) -> ProjectControlSnapshot:
        self.data.setdefault(snapshot.project_id, []).insert(0, snapshot)
        return snapshot

    def list_for_project(
        self, project_id: str, limit: int = 30
    ) -> list[ProjectControlSnapshot]:
        return self.data.get(project_id, [])[:limit]


@dataclass
class InMemoryProjectEventRepository(ProjectEventRepository):
    data: dict[str, list[ProjectEvent]] = field(default_factory=dict)

    def save(self, event: ProjectEvent) -> ProjectEvent:
        self.data.setdefault(event.project_id, []).insert(0, event)
        return event

    def list_for_project(
        self, project_id: str, limit: int = 100
    ) -> list[ProjectEvent]:
        return self.data.get(project_id, [])[:limit]
