"""
Memory Service.

Application-layer memory service.

This service adapts the legacy backend MemoryRecord contract
to the hackathon AI Runtime MemoryEntry / MemoryStore
persistence contract.

Architecture:

Application
    |
    v
MemoryService
    |
    v
MemoryEntry / MemoryType
    |
    v
MemoryStore
    |
    v
SQLite

Responsibilities:
- save application memories
- retrieve application memories
- translate application memory categories
- isolate persistence details from callers

Does not:
- perform semantic retrieval
- call an LLM
- execute agents
- make autonomous memory decisions
"""

from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from backend.memory.memory_record import (
    MemoryRecord,
)

from backend.memory.memory_store import (
    MemoryStore,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryService:
    """
    Controls application memory operations.

    The service keeps the existing MemoryRecord interface
    for the backend application while delegating persistence
    to the current AI Runtime MemoryStore.
    """

    DEFAULT_DATABASE_PATH = (
        Path("data") / "memory.db"
    )

    def __init__(
        self,
        store: MemoryStore | None = None,
        database_path: str | Path | None = None,
    ) -> None:
        """
        Initialise the memory service.

        Args:
            store:
                Optional injected MemoryStore.

                Primarily useful for tests and controlled
                application configuration.

            database_path:
                Optional SQLite database path.

                Ignored when an explicit store is supplied.
        """

        if store is not None:
            self.store = store
            return

        path = (
            Path(database_path)
            if database_path is not None
            else self.DEFAULT_DATABASE_PATH
        )

        path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        self.store = MemoryStore(
            database_path=path
        )

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> MemoryRecord:
        """
        Save an application memory.

        The legacy category is preserved in metadata while
        the persistence layer receives a supported MemoryType.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        if not category.strip():
            raise ValueError(
                "Memory category cannot be empty."
            )

        if not content.strip():
            raise ValueError(
                "Memory content cannot be empty."
            )

        created_at = datetime.now(
            timezone.utc
        )

        record = MemoryRecord(
            user_id=user_id,
            category=category,
            content=content,
            created_at=created_at,
        )

        memory_entry = MemoryEntry(
            user_id=user_id,
            memory_type=self._to_memory_type(
                category
            ),
            content=content,
            created_at=created_at,
            updated_at=created_at,
            metadata={
                "category": category,
            },
        )

        self.store.save(
            memory_entry
        )

        return record

    def recall(
        self,
        user_id: str,
    ) -> list[MemoryRecord]:
        """
        Retrieve memories for a user.

        Persistence-specific MemoryEntry objects are
        translated back into the application-facing
        MemoryRecord contract.
        """

        entries = self.store.list_for_user(
            user_id
        )

        return [
            self._to_memory_record(entry)
            for entry in entries
        ]

    @staticmethod
    def _to_memory_type(
        category: str,
    ) -> MemoryType:
        """
        Translate application categories into the
        supported AI Runtime memory types.

        The original category remains available in
        MemoryEntry.metadata.
        """

        normalized = category.strip().lower()

        if normalized == "saved_item":
            return MemoryType.SAVED_ITEM

        if normalized == "discovery_context":
            return MemoryType.DISCOVERY_CONTEXT

        return MemoryType.PREFERENCE

    @staticmethod
    def _to_memory_record(
        entry: MemoryEntry,
    ) -> MemoryRecord:
        """
        Translate a persistent MemoryEntry into the
        application-facing MemoryRecord.
        """

        category = entry.metadata.get(
            "category",
            entry.memory_type.value,
        )

        return MemoryRecord(
            user_id=entry.user_id,
            category=category,
            content=entry.content,
            created_at=entry.created_at,
        )

Converts a user-facing FlashDealCard into explicit commerce

Architecture:

FlashDealCard
      |

FlashDealCommerce
      |
      +----> CommerceAction
      |
      +----> PaymentPass
                    |
                    v
             CommerceService
                    |
                    v
             Vendor / Checkout

This module does not:
- process payments
- store payment information
- call vendor APIs
- execute autonomous purchases
- decide whether a product should be recommended

The discovery layer remains responsible for discovery.
The commerce layer remains responsible for commerce.
"""

from __future__ import annotations

    CommerceAction,
    CommerceActionType,
)

    PaymentPass,
)

    FlashDealCard,
)


class FlashDealCommerce:
    """
    Application boundary between a FlashDealCard and
    the commerce subsystem.
    """

    def create_action(
        self,
        *,
        user_id: str,
        card: FlashDealCard,
    ) -> CommerceAction:
        """
        Create an explicit purchase action from a Flash Deal.

        No purchase occurs here.
        """

        if not user_id.strip():
            raise ValueError(
                "Commerce user_id cannot be empty."
            )

        return CommerceAction(
            user_id=user_id,
            deal_id=card.deal_id,
            action_type=(
                CommerceActionType.INITIATE_PURCHASE
            ),
            destination_url=card.get_deal_url,
            metadata={
                "source": "flash_deal_card",
                "character_id": card.character_id,
                "vendor": card.vendor_name,
            },
        )

    def create_payment_pass(
        self,
        *,
        user_id: str,
        card: FlashDealCard,
        payment_pass_id: str | None = None,
    ) -> PaymentPass:
        """
        Create a demo-only payment handoff from a Flash Deal.

        No payment occurs here.

        The resulting PaymentPass is always:
        - READY
        - DEMO
        - not executed
        """

        if not user_id.strip():
            raise ValueError(
                "Commerce user_id cannot be empty."
            )

        return PaymentPass(
            payment_pass_id=(
                payment_pass_id
                if payment_pass_id is not None
                else f"payment-pass-{card.deal_id}"
            ),
            deal_id=card.deal_id,
            user_id=user_id,
            product_name=card.product_name,
            vendor_name=card.vendor_name,
            amount=card.current_price,
            currency=card.currency,
            checkout_url=card.get_deal_url,
        )

"""
Hackathon commerce contracts.

These contracts define the boundary between EasyChatBot's
discovery experience and an eventual commerce/payment
integration.

EasyChatBot does not implement a marketplace here.

The external commerce provider is responsible for the
actual payment and vendor transaction.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum
from uuid import uuid4


class CommerceActionType(str, Enum):
    """
    Supported commerce actions.
    """

    VIEW_PRODUCT = "view_product"

    INITIATE_PURCHASE = "initiate_purchase"


class CommerceActionStatus(str, Enum):
    """
    Result status of a commerce action.
    """

    READY = "ready"

    REDIRECT = "redirect"

    REJECTED = "rejected"


@dataclass(frozen=True)
class CommerceAction:
    """
    Explicit request to perform a commerce action.

    This is an application-level contract.

    It is intentionally independent from Stripe, Amazon,
    OLX, or any other vendor.
    """

    user_id: str

    deal_id: str

    action_type: CommerceActionType

    destination_url: str

    action_id: str = field(
        default_factory=lambda: str(uuid4())
    )

    metadata: dict[str, str] = field(
        default_factory=dict
    )

    def __post_init__(self) -> None:
        if not self.user_id.strip():
            raise ValueError(
                "Commerce action user_id cannot be empty."
            )

        if not self.deal_id.strip():
            raise ValueError(
                "Commerce action deal_id cannot be empty."
            )

        if not self.destination_url.strip():
            raise ValueError(
                "Commerce action destination URL "
                "cannot be empty."
            )

        if not (
            self.destination_url.startswith(
                "https://"
            )
            or self.destination_url.startswith(
                "http://"
            )
        ):
            raise ValueError(
                "Commerce action destination URL "
                "must use HTTP or HTTPS."
            )


@dataclass(frozen=True)
class CommerceActionResult:
    """
    Provider-independent result of a commerce action.
    """

    action_id: str

    deal_id: str

    status: CommerceActionStatus

    next_url: str | None = None

    message: str | None = None

    metadata: dict[str, str] = field(
        default_factory=dict
    )

"""
Hackathon Flash Deal Card presentation contract.

This module defines the user-facing data contract for the
EasyChatBot Flash Deal Card shown by a character.

Architecture:

Parallel / Discovery
        |
        v
     DealCard
        |
        +---- Memory Context
        |
        +---- Character
        |
        v
  FlashDealCard
        |
        +---- GET DEAL
        |
        +---- MORE INFORMATION
        |
        v
 External Vendor / Future Commerce Flow

The model is presentation-oriented.

It does not:
- call providers
- access SQLite
- perform payments
- execute agents
- scrape vendor pages
- invent product information
"""

from __future__ import annotations

from dataclasses import dataclass, field

from src.ai_runtime.discovery.models import DealCard


@dataclass(frozen=True)
class FlashDealCard:
    """
    User-facing Flash Deal Card.

    This contract maps directly to the hackathon UX:

    - character discovery
    - product information
    - pricing
    - discount
    - personal relevance
    - vendor
    - GET DEAL action
    - MORE INFORMATION action

    The card deliberately contains no payment implementation.
    """

    deal_id: str
    character_id: str
    character_name: str
    character_message: str

    product_name: str
    product_image_url: str
    tagline: str
    highlights: tuple[str, ...]

    current_price: float
    currency: str
    previous_price: float | None
    discount_percentage: float | None
    savings: float | None

    relevance_reason: str

    vendor_name: str
    more_information_url: str
    get_deal_url: str

    badge: str = "NEW FIND!"
    card_id: str | None = None
    metadata: dict[str, str] = field(
        default_factory=dict
    )

    def __post_init__(self) -> None:
        """
        Validate the presentation contract.
        """

        required_text_fields = {
            "deal_id": self.deal_id,
            "character_id": self.character_id,
            "character_name": self.character_name,
            "character_message": self.character_message,
            "product_name": self.product_name,
            "product_image_url": self.product_image_url,
            "tagline": self.tagline,
            "currency": self.currency,
            "relevance_reason": self.relevance_reason,
            "vendor_name": self.vendor_name,
            "more_information_url": self.more_information_url,
            "get_deal_url": self.get_deal_url,
        }

        for field_name, value in required_text_fields.items():
            if not value.strip():
                raise ValueError(
                    f"FlashDealCard {field_name} cannot be empty."
                )

        if not self.highlights:
            raise ValueError(
                "FlashDealCard must contain at least one highlight."
            )

        if any(
            not highlight.strip()
            for highlight in self.highlights
        ):
            raise ValueError(
                "FlashDealCard highlights cannot contain empty values."
            )

        if self.current_price < 0:
            raise ValueError(
                "FlashDealCard current_price cannot be negative."
            )

        if (
            self.previous_price is not None
            and self.previous_price < 0
        ):
            raise ValueError(
                "FlashDealCard previous_price cannot be negative."
            )

        if (
            self.discount_percentage is not None
            and not 0 <= self.discount_percentage <= 100
        ):
            raise ValueError(
                "FlashDealCard discount_percentage must be "
                "between 0 and 100."
            )

        if self.savings is not None and self.savings < 0:
            raise ValueError(
                "FlashDealCard savings cannot be negative."
            )

        self._validate_url(
            self.product_image_url,
            "product_image_url",
        )
        self._validate_url(
            self.more_information_url,
            "more_information_url",
        )
        self._validate_url(
            self.get_deal_url,
            "get_deal_url",
        )

    @staticmethod
    def _validate_url(
        url: str,
        field_name: str,
    ) -> None:
        """
        Validate a user-facing URL.
        """

        if not (
            url.startswith("https://")
            or url.startswith("http://")
        ):
            raise ValueError(
                f"FlashDealCard {field_name} must use "
                "HTTP or HTTPS."
            )

    @classmethod
    def from_deal(
        cls,
        *,
        deal: DealCard,
        character_id: str,
        character_name: str,
        character_message: str,
        relevance_reason: str,
        product_image_url: str,
        tagline: str,
        highlights: list[str] | tuple[str, ...],
        badge: str = "NEW FIND!",
        metadata: dict[str, str] | None = None,
    ) -> "FlashDealCard":
        """
        Build a presentation card from a domain DealCard.

        Product, pricing and vendor information comes from the
        DealCard. Character and personalisation information is
        supplied explicitly by the application layer.

        No product or pricing information is invented here.
        """

        savings: float | None = None

        if (
            deal.previous_price is not None
            and deal.previous_price > deal.price
        ):
            savings = round(
                deal.previous_price - deal.price,
                2,
            )

        return cls(
            deal_id=deal.deal_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            product_name=deal.product_name,
            product_image_url=product_image_url,
            tagline=tagline,
            highlights=tuple(highlights),
            current_price=deal.price,
            currency=deal.currency,
            previous_price=deal.previous_price,
            discount_percentage=deal.discount_percentage,
            savings=savings,
            relevance_reason=relevance_reason,
            vendor_name=deal.vendor,
            more_information_url=(
                deal.commerce_link.destination_url
            ),
            get_deal_url=(
                deal.commerce_link.click_url
            ),
            badge=badge,
            metadata=(
                dict(metadata)
                if metadata is not None
                else {}
            ),
        )

    @property
    def has_discount(self) -> bool:
        """Return whether a verified discount exists."""

        return (
            self.discount_percentage is not None
            and self.discount_percentage > 0
        )

    @property
    def has_previous_price(self) -> bool:
        """Return whether a previous price is available."""

        return self.previous_price is not None

    @property
    def display_discount(self) -> str | None:
        """Return a UI-friendly discount label."""

        if self.discount_percentage is None:
            return None

        return f"{self.discount_percentage:.0f}% OFF"

    @property
    def display_savings(self) -> str | None:
        """Return a UI-friendly savings label."""

        if self.savings is None:
            return None

        return f"Save {self.currency} {self.savings:.2f}"

"""
Flash Deal Card presenter.

Pure presentation layer for the EasyChatBot hackathon.

Architecture:

FlashDealCard
      |
      v
FlashDealPresenter
      |
      v
Markdown / terminal output

Responsibilities:
- render an existing FlashDealCard
- present verified product and pricing information
- present character context
- present personal relevance
- expose the existing vendor actions

Non-responsibilities:
- discovery
- memory access

- product data generation

The presenter must never mutate the FlashDealCard.
"""

from __future__ import annotations

from src.ai_runtime.discovery.flash_deal_card import (
    FlashDealCard,
)


class FlashDealPresenter:
    """
    Pure renderer for the user-facing Flash Deal Card.

    The presenter consumes the existing FlashDealCard contract
    and produces deterministic Markdown suitable for the
    hackathon terminal demonstration or a future chat UI.
    """

    def render(
        self,
        card: FlashDealCard,
    ) -> str:
        """
        Render a FlashDealCard as user-facing Markdown.

        No external calls or business decisions are performed.
        """

        sections: list[str] = []

        sections.append(
            f"## {card.character_name}"
        )

        sections.append(
            f"> {card.character_message}"
        )

        sections.append(
            f"### {card.badge}"
        )

"""
EasyChatBot — Memory Context Builder.

H3-01 — Memory-Aware Live Intelligence.

The builder operates on the APPLICATION memory contract:

    MemoryService
        ↓
    MemoryRecord
        ↓
    MemoryContextBuilder
        ↓
    MemoryContext

It deliberately does NOT consume MemoryEntry directly.

MemoryEntry belongs to the persistence/runtime storage boundary.
MemoryRecord is the application-facing memory contract.

Responsibilities:
- validate already-retrieved memory
- enforce user isolation
- determine deterministic relevance
- bound memory count
- bound context size
- produce provider-independent context

Does not:
- retrieve memory
- persist memory
- call an LLM
- call a provider
- execute an agent
- mutate ExecutionContext
- perform semantic/vector search
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
import re

from backend.memory.memory_record import (
    MemoryRecord,
)


DEFAULT_MAX_MEMORY_ITEMS = 5
DEFAULT_MAX_CONTEXT_CHARACTERS = 4000

_WORD_PATTERN = re.compile(
    r"[\wÀ-ÿ'-]+",
    re.UNICODE,
)


# Application-level memory categories.
#
# These are deliberately strings because MemoryService exposes
# MemoryRecord.category rather than MemoryEntry.memory_type.
_MEMORY_CATEGORY_PRIORITY: dict[str, int] = {
    "preference": 30,
    "learning_goal": 30,
    "discovery_context": 20,
    "saved_item": 10,
    "conversation": 10,
}


@dataclass(frozen=True)
class MemoryContext:
    """
    Bounded, provider-independent learner context.
    """

    entries: tuple[MemoryRecord, ...]
    text: str

    @property
    def is_empty(self) -> bool:
        """Return True when no memory was selected."""
        return not self.entries

    @property
    def memory_count(self) -> int:
        """Return the number of selected memories."""
        return len(self.entries)


class MemoryContextBuilder:
    """
    Build bounded learner context from MemoryRecord objects.

    The builder is stateless and deterministic.
    """

    def __init__(
        self,
        *,
        max_memory_items: int = DEFAULT_MAX_MEMORY_ITEMS,
        max_context_characters: int = (
            DEFAULT_MAX_CONTEXT_CHARACTERS
        ),
    ) -> None:
        if max_memory_items < 1:
            raise ValueError(
                "max_memory_items must be greater than zero."
            )

        if max_context_characters < 1:
            raise ValueError(
                "max_context_characters must be greater than zero."
            )

        self._max_memory_items = max_memory_items
        self._max_context_characters = (
            max_context_characters
        )

    def build(
        self,
        *,
        user_id: str,
        request: str,
        memories: list[MemoryRecord]
        | tuple[MemoryRecord, ...],
    ) -> MemoryContext:
        """
        Build bounded learner context.

        Memory is supplied by the caller.

        The builder does not retrieve or persist memory.
        """

        if not user_id.strip():
            raise ValueError(
                "user_id cannot be empty."
            )

        if not request.strip():
            raise ValueError(
                "request cannot be empty."
            )

        if not memories:
            return MemoryContext(
                entries=(),
                text="",
            )

        request_tokens = self._tokens(
            request
        )

        candidates: list[
            tuple[
                int,
                datetime,
                str,
                MemoryRecord,
            ]
        ] = []

        for memory in memories:

            if not isinstance(
                memory,
                MemoryRecord,
            ):
                continue

            # Defence in depth.
            if memory.user_id != user_id:
                continue

            if not memory.category.strip():
                continue

            if not memory.content.strip():
                continue

            score = self._score(
                memory=memory,
                request_tokens=request_tokens,
            )

            # A memory category alone does NOT make the memory
            # relevant to the current request.
            if score <= 0:
                continue

            candidates.append(
                (
                    score,
                    memory.created_at,
                    memory.content,
                    memory,
                )
            )

        # Deterministic ordering:
        #
        # 1. relevance descending
        # 2. newest memory first
        # 3. content as stable tie-breaker
        candidates.sort(
            key=lambda item: (
                -item[0],
                -item[1].timestamp(),
                item[2].casefold(),
            )
        )

        selected: list[MemoryRecord] = []
        rendered_items: list[str] = []

        current_length = len(
            "Learner context:\n"
        )

        for (
            _score,
            _created_at,
            _content,
            memory,
        ) in candidates:

            if (
                len(selected)
                >= self._max_memory_items
            ):
                break

            line = self._render_memory(
                memory
            )

            separator_length = (
                1
                if rendered_items
                else 0
            )

            projected_length = (
                current_length
                + separator_length
                + len(line)
            )

            if (
                projected_length
                > self._max_context_characters
            ):
                # Do not partially truncate an individual
                # memory because that could alter meaning.
                continue

            selected.append(memory)
            rendered_items.append(line)

            current_length = projected_length

        if not selected:
            return MemoryContext(
                entries=(),
                text="",
            )

        text = (
            "Learner context:\n"
            + "\n".join(
                f"- {item}"
                for item in rendered_items
            )
        )

        if (
            len(text)
            > self._max_context_characters
        ):
            raise AssertionError(
                "MemoryContextBuilder produced "
                "an oversized context."
            )

        return MemoryContext(
            entries=tuple(selected),
            text=text,
        )

    @staticmethod
    def _tokens(
        value: str,
    ) -> set[str]:
        """
        Return normalised tokens for deterministic matching.
        """

        return {
            token.casefold()
            for token in _WORD_PATTERN.findall(
                value
            )
            if len(token) > 1
        }

    @classmethod
    def _score(
        cls,
        *,
        memory: MemoryRecord,
        request_tokens: set[str],
    ) -> int:
        """
        Calculate deterministic relevance.

        Relevance requires actual token overlap.

        This prevents an unrelated preference such as
        "Prefers hiking" from being injected into a request
        about "European AI regulation" simply because it is
        a preference.
        """

        content_tokens = cls._tokens(
            memory.content
        )

        category_tokens = cls._tokens(
            memory.category
        )

        overlap = request_tokens & (
            content_tokens
            | category_tokens
        )

        if not overlap:
            return 0

        category = (
            memory.category.strip().casefold()
        )

        category_priority = (
            _MEMORY_CATEGORY_PRIORITY.get(
                category,
                10,
            )
        )

        return (
            category_priority
            + len(overlap)
        )

    @staticmethod
    def _render_memory(
        memory: MemoryRecord,
    ) -> str:
        """
        Render memory explicitly as learner context.

        Memory content is treated as data, not instructions.
        """

        category = (
            memory.category.strip()
        )

        content = " ".join(
            memory.content.split()
        )

        return (
            f"[{category}] {content}"
        )

"""
EasyChatBot Hackathon — Podcast Discovery Agent.

The Agent converts explicit learner context and preference memory into
provider-neutral search intent.

It does not call Apple/Spotify, write memory, select a provider, or invent
podcast metadata.
"""

from __future__ import annotations

import re

from src.ai_runtime.memory.models import MemoryEntry, MemoryType
from src.ai_runtime.podcast.models import (
    LearningContext,
    PodcastSearchIntent,
)


class PodcastDiscoveryAgent:
    """Build provider-neutral podcast discovery intent."""

    _STOP_WORDS = {
        "a", "an", "and", "are", "be", "for", "from", "i", "in", "is",
        "like", "of", "on", "podcasts", "prefers", "prefer", "the", "to",
        "with", "you", "your",
    }

    def build_search_intent(
        self,
        *,
        user_id: str,
        learning_context: LearningContext,
        memories: list[MemoryEntry],
        max_results: int = 10,
    ) -> PodcastSearchIntent:
        """
        Convert explicit preferences + learning-world context into intent.

        Only PREFERENCE memories are used, matching the existing
        EasyChatBot discovery contract.
        """

        if not user_id.strip():
            raise ValueError("Podcast Agent user_id cannot be empty.")

        preferences = [
            memory
            for memory in memories
            if memory.user_id == user_id
            and memory.memory_type == MemoryType.PREFERENCE
        ]

        topics: list[str] = []
        keywords: list[str] = []

        for memory in preferences:
            topics.extend(
                self._metadata_terms(
                    memory.metadata,
                    "topics",
                    "interests",
                    "preferred_topics",
                )
            )
            keywords.extend(
                self._metadata_terms(
                    memory.metadata,
                    "keywords",
                )
            )

            if not topics and not keywords:
                keywords.extend(
                    self._content_keywords(memory.content)
                )

        # The learning world is always part of the search context.
        topics.append(learning_context.world_id)

        # Target language is deliberately a hard eligibility constraint,
        # not just another search keyword.
        return PodcastSearchIntent(
            user_id=user_id,
            learning_context=learning_context,
            topics=tuple(self._dedupe(topics)),
            keywords=tuple(self._dedupe(keywords)),
            language_required=True,
            max_results=max_results,
        )

    @staticmethod
    def _metadata_terms(
        metadata: dict[str, str],
        *keys: str,
    ) -> list[str]:
        terms: list[str] = []
        for key in keys:
            raw = metadata.get(key, "")
            if raw:
                terms.extend(
                    item.strip()
                    for item in re.split(r"[,;|]", raw)
                    if item.strip()
                )
        return terms

    @classmethod
    def _content_keywords(cls, content: str) -> list[str]:
        words = re.findall(
            r"[A-Za-zÀ-ÿ0-9][A-Za-zÀ-ÿ0-9'-]{2,}",
            content.lower(),
        )
        return [
            word for word in words
            if word not in cls._STOP_WORDS
        ][:8]

    @staticmethod
    def _dedupe(values: list[str]) -> list[str]:
        seen: set[str] = set()
        result: list[str] = []
        for value in values:
            key = value.strip().lower()
            if key and key not in seen:
                seen.add(key)
                result.append(value.strip())
        return result

"""
EasyChatBot Hackathon — Apple Podcasts catalogue provider.

Uses Apple's iTunes Search API for read-only podcast catalogue discovery.
"""

from __future__ import annotations

import json
from urllib.parse import urlencode
from urllib.request import Request, urlopen

from src.ai_runtime.podcast.models import (
    PodcastProviderName,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)


class ApplePodcastProvider:
    """Read-only Apple podcast catalogue provider."""

    BASE_URL = "https://itunes.apple.com/search"

    def __init__(self, *, timeout_seconds: float = 10.0) -> None:
        self._timeout_seconds = timeout_seconds

    def search(
        self,
        intent: PodcastSearchIntent,
    ) -> list[PodcastSearchResult]:
        query_terms = list(intent.topics) + list(intent.keywords)
        params = {
            "term": " ".join(query_terms[:6]),
            "country": intent.learning_context.country_code.upper(),
            "media": "podcast",
            "entity": "podcast",
            "limit": str(intent.max_results),
            "explicit": "No",
        }

        request = Request(
            f"{self.BASE_URL}?{urlencode(params)}",
            headers={
                "Accept": "application/json",
                "User-Agent": "EasyChatBot-Hackathon/1.0",
            },
            method="GET",
        )

        with urlopen(
            request,
            timeout=self._timeout_seconds,
        ) as response:
            payload = json.loads(
                response.read().decode("utf-8")
            )

        return [
            self._normalise(item)
            for item in payload.get("results", [])
            if self._is_podcast_result(item)
        ]

    @staticmethod
    def _is_podcast_result(item: dict[str, object]) -> bool:
        return (
            str(item.get("kind", "")).lower() == "podcast"
            or str(item.get("wrapperType", "")).lower() == "track"
        )

    @staticmethod
    def _normalise(
        item: dict[str, object],
    ) -> PodcastSearchResult:
        language = normalise_language(
            str(item.get("language") or "unknown")
        )

        return PodcastSearchResult(
            provider=PodcastProviderName.APPLE,
            external_id=str(
                item.get("collectionId")
                or item.get("trackId")
                or ""
            ),
            title=str(
                item.get("collectionName")
                or item.get("trackName")
                or ""
            ),
            description=str(
                item.get("description")
                or item.get("longDescription")
                or ""
            ),
            publisher=str(
                item.get("artistName")
                or "Unknown publisher"
            ),
            languages=(language,),
            url=str(
                item.get("collectionViewUrl")
                or item.get("trackViewUrl")
                or ""
            ),
            image_url=str(
                item.get("artworkUrl600")
                or item.get("artworkUrl100")
                or ""
            ) or None,
            published_at=str(
                item.get("releaseDate") or ""
            ) or None,
            explicit=(
                str(
                    item.get("trackExplicitness", "")
                ).lower() == "explicit"
            ),
            metadata={"provider": "apple_podcasts"},
        )

@dataclass(frozen=True)
class VocabularyItem:
    """A vocabulary item presented to the learner."""

    term: str
    meaning: str
    example: str

    def __post_init__(self) -> None:
        for name, value in {
            "term": self.term,
            "meaning": self.meaning,
            "example": self.example,
        }.items():
            if not value.strip():
                raise ValueError(
                    f"VocabularyItem {name} cannot be empty."
                )


@dataclass(frozen=True)
class ComprehensionQuestion:
    """A multiple-choice comprehension question."""

    question: str
    options: tuple[str, ...]
    correct_option: int

    def __post_init__(self) -> None:
        if not self.question.strip():
            raise ValueError(
                "ComprehensionQuestion question cannot be empty."
            )

        if len(self.options) < 2:
            raise ValueError(
                "ComprehensionQuestion requires at least two options."
            )

        if any(not option.strip() for option in self.options):
            raise ValueError(
                "ComprehensionQuestion options cannot be empty."
            )

        if not 0 <= self.correct_option < len(self.options):
            raise ValueError(
                "ComprehensionQuestion correct_option is out of range."
            )

            if not value.strip():

        if not self.vocabulary:

                "Review the vocabulary and try again."

            challenge_id=challenge.challenge_id,
            user_id=challenge.user_id,
            score=score,
            total_questions=total,
            passed=passed,
            vocabulary_review_required=vocabulary_review_required,
            message=message,
        )

class PodcastProviderName(str, Enum):
    APPLE = "apple_podcasts"
    SPOTIFY = "spotify"


@dataclass(frozen=True)
class LearningContext:
    """Explicit learning-world context for podcast discovery."""

    world_id: str
    target_language: str
    learner_level: str
    learning_goal: str = ""
    preferred_languages: tuple[str, ...] = ()
    country_code: str = "LU"

    def __post_init__(self) -> None:
        if not self.world_id.strip():
            raise ValueError("LearningContext world_id cannot be empty.")
        if not self.target_language.strip():
            raise ValueError(
                "LearningContext target_language cannot be empty."
            )
        if not self.learner_level.strip():
            raise ValueError(
                "LearningContext learner_level cannot be empty."
            )
        if len(self.country_code.strip()) != 2:
            raise ValueError(
                "LearningContext country_code must be a two-letter code."
            )


@dataclass(frozen=True)
class PodcastSearchIntent:
    """
    Provider-neutral search intent produced by the Podcast Agent.

    The Agent does not know whether Apple or Spotify will execute it.
    """

    user_id: str
    learning_context: LearningContext
    topics: tuple[str, ...]
    keywords: tuple[str, ...] = ()
    language_required: bool = True
    max_results: int = 10
    intent_id: str = field(default_factory=lambda: str(uuid4()))

    def __post_init__(self) -> None:
        if not self.user_id.strip():
            raise ValueError("PodcastSearchIntent user_id cannot be empty.")
        if not self.topics and not self.keywords:
            raise ValueError(
                "PodcastSearchIntent requires at least one topic or keyword."
            )
        if self.max_results < 1:
            raise ValueError(
                "PodcastSearchIntent max_results must be at least 1."
            )


@dataclass(frozen=True)
class PodcastSearchResult:
    """Normalised podcast/episode candidate from any provider."""

    provider: PodcastProviderName
    external_id: str
    title: str
    description: str
    publisher: str
    languages: tuple[str, ...]
    url: str
    image_url: str | None = None
    published_at: str | None = None
    duration_seconds: int | None = None
    explicit: bool = False
    metadata: dict[str, str] = field(default_factory=dict)

    def __post_init__(self) -> None:
        for name, value in {
            "external_id": self.external_id,
            "title": self.title,
            "publisher": self.publisher,
            "url": self.url,
        }.items():
            if not value.strip():
                raise ValueError(
                    f"PodcastSearchResult {name} cannot be empty."
                )

        if not (
            self.url.startswith("https://")
            or self.url.startswith("http://")
        ):
            raise ValueError(
                "PodcastSearchResult url must use HTTP or HTTPS."
            )

        if not self.languages:
            raise ValueError(
                "PodcastSearchResult requires at least one language."
            )


@dataclass(frozen=True)
class PodcastRecommendation:
    """Character-facing recommendation selected from verified candidates."""

    recommendation_id: str
    user_id: str
    character_id: str
    character_name: str
    character_message: str
    result: PodcastSearchResult
    relevance_reason: str
    match_score: float

    def __post_init__(self) -> None:
        for name, value in {
            "recommendation_id": self.recommendation_id,
            "user_id": self.user_id,
            "character_id": self.character_id,
            "character_name": self.character_name,
            "character_message": self.character_message,
            "relevance_reason": self.relevance_reason,
        }.items():
            if not value.strip():
                raise ValueError(
                    f"PodcastRecommendation {name} cannot be empty."
                )

        if not 0 <= self.match_score <= 100:
            raise ValueError(
                "PodcastRecommendation match_score must be between 0 and 100."
            )


LANGUAGE_ALIASES: Final[dict[str, str]] = {
    "fr": "fr", "fra": "fr", "fre": "fr",
    "french": "fr", "français": "fr", "francais": "fr",
    "en": "en", "eng": "en", "english": "en",
    "es": "es", "spa": "es", "spanish": "es", "español": "es",
    "de": "de", "deu": "de", "ger": "de", "german": "de",
    "it": "it", "ita": "it", "italian": "it",
    "ru": "ru", "rus": "ru", "russian": "ru",
    "pt": "pt", "por": "pt", "portuguese": "pt",
}


def normalise_language(value: str) -> str:
    """Return a stable ISO-639-1-style code when recognised."""
    return LANGUAGE_ALIASES.get(
        value.strip().lower(),
        value.strip().lower(),
    )

"""
EasyChatBot Hackathon — personalised podcast recommendation service.

This is the application-level selection boundary.

Language is a HARD filter because EasyChatBot is a language-learning
platform. Preference/topic relevance is evaluated only after a candidate
is language-eligible.
"""

from __future__ import annotations

import re
from uuid import uuid4

from src.ai_runtime.podcast.models import (
    PodcastRecommendation,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)
from src.ai_runtime.podcast.providers import PodcastProvider


class PodcastRecommendationService:
    """Search providers and select a personalised candidate."""

    def __init__(self, providers: list[PodcastProvider]) -> None:
        if not providers:
            raise ValueError(
                "PodcastRecommendationService requires at least one provider."
            )
        self._providers = list(providers)

    def recommend(
        self,
        *,
        intent: PodcastSearchIntent,
        character_id: str,
        character_name: str,
        character_message: str,
    ) -> PodcastRecommendation:
        candidates: list[PodcastSearchResult] = []

        for provider in self._providers:
            candidates.extend(provider.search(intent))

        ranked = self.rank_candidates(
            candidates=candidates,
            intent=intent,
        )

        if not ranked:
            raise LookupError(
                "No podcast matched the learner's language and preferences."
            )

        result, score, reason = ranked[0]

        return PodcastRecommendation(
            recommendation_id=str(uuid4()),
            user_id=intent.user_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            result=result,
            relevance_reason=reason,
            match_score=score,
        )

    @classmethod
    def rank_candidates(
        cls,
        *,
        candidates: list[PodcastSearchResult],
        intent: PodcastSearchIntent,
    ) -> list[tuple[PodcastSearchResult, float, str]]:
        ranked: list[tuple[PodcastSearchResult, float, str]] = []

        target_language = normalise_language(
            intent.learning_context.target_language
        )

        preference_terms = [
            term.lower()
            for term in list(intent.topics) + list(intent.keywords)
            if term.strip()
        ]

        for candidate in candidates:
            candidate_languages = {
                normalise_language(language)
                for language in candidate.languages
            }

            # Critical learning-platform rule:
            # unknown/wrong-language content is not eligible.
            if (
                intent.language_required
                and target_language not in candidate_languages
            ):
                continue

            haystack = " ".join(
                [
                    candidate.title,
                    candidate.description,
                    candidate.publisher,
                ]
            ).lower()

            matched_terms = [
                term
                for term in preference_terms
                if cls._term_matches(term, haystack)
            ]

            score = 50.0

            if matched_terms:
                score += min(30.0, len(matched_terms) * 10.0)

            if intent.learning_context.world_id.lower() in haystack:
                score += 10.0

            level = intent.learning_context.learner_level.lower()
            if level and level in haystack:
                score += 10.0

            reason = cls._build_reason(
                language=intent.learning_context.target_language,
                matched_terms=matched_terms,
                world_id=intent.learning_context.world_id,
            )

            ranked.append(
                (candidate, min(score, 100.0), reason)
            )

        ranked.sort(
            key=lambda item: (
                item[1],
                item[0].published_at or "",
            ),
            reverse=True,
        )

        return ranked[: intent.max_results]

    @staticmethod
    def _term_matches(term: str, haystack: str) -> bool:
        term = term.strip().lower()
        if not term:
            return False
        if term in haystack:
            return True

        words = [
            word
            for word in re.findall(r"[a-z0-9À-ÿ'-]+", term)
            if len(word) > 2
        ]
        return bool(words) and any(
            word in haystack
            for word in words
        )

    @staticmethod
    def _build_reason(
        *,
        language: str,
        matched_terms: list[str],
        world_id: str,
    ) -> str:
        reasons = [f"It is available in {language}."]

        if matched_terms:
            reasons.append(
                "It matches your interests in "
                + ", ".join(matched_terms[:3])
                + "."
            )

        reasons.append(
            f"It fits your {world_id} learning world."
        )

        return " ".join(reasons)

"""
EasyChatBot Hackathon — Spotify podcast catalogue provider.

The MVP searches episodes, because episode metadata exposes a `languages`
field. This allows EasyChatBot's language-learning rule to remain strict.

Credentials are injected as an access token; this module does not manage
OAuth state or user accounts.
"""

from __future__ import annotations

import json
from urllib.parse import urlencode
from urllib.request import Request, urlopen

from src.ai_runtime.podcast.models import (
    PodcastProviderName,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)


class SpotifyPodcastProvider:
    """Read-only Spotify catalogue provider."""

    BASE_URL = "https://api.spotify.com/v1/search"

    def __init__(
        self,
        *,
        access_token: str,
        timeout_seconds: float = 10.0,
        market: str | None = None,
    ) -> None:
        if not access_token.strip():
            raise ValueError(
                "SpotifyPodcastProvider access_token cannot be empty."
            )
        self._access_token = access_token
        self._timeout_seconds = timeout_seconds
        self._market = market

    def search(
        self,
        intent: PodcastSearchIntent,
    ) -> list[PodcastSearchResult]:
        query_terms = list(intent.topics) + list(intent.keywords)
        params = {
            "q": " ".join(query_terms[:6]),
            "type": "episode",
            "limit": str(intent.max_results),
        }

        market = self._market or intent.learning_context.country_code
        if market:
            params["market"] = market.upper()

        request = Request(
            f"{self.BASE_URL}?{urlencode(params)}",
            headers={
                "Accept": "application/json",
                "Authorization": f"Bearer {self._access_token}",
                "User-Agent": "EasyChatBot-Hackathon/1.0",
            },
            method="GET",
        )

        with urlopen(
            request,
            timeout=self._timeout_seconds,
        ) as response:
            payload = json.loads(
                response.read().decode("utf-8")
            )

        episodes = payload.get("episodes", {})
        if not isinstance(episodes, dict):
            return []

        items = episodes.get("items", [])
        if not isinstance(items, list):
            return []

        results: list[PodcastSearchResult] = []

        for item in items:
            if not isinstance(item, dict):
                continue

            result = self._normalise_episode(item)
            if result is not None:
                results.append(result)

        return results

    @staticmethod
    def _normalise_episode(
        item: dict[str, object],
    ) -> PodcastSearchResult | None:
        external_id = str(item.get("id") or "")
        title = str(item.get("name") or "")

        external_urls = item.get("external_urls")
        url = ""
        if isinstance(external_urls, dict):
            url = str(
                external_urls.get("spotify") or ""
            )

        if not external_id or not title or not url:
            return None

        raw_languages = item.get("languages", [])
        if not isinstance(raw_languages, list):
            return None

        languages = tuple(
            normalise_language(str(language))
            for language in raw_languages
            if str(language).strip()
        )

        # Unknown language is not invented. The recommendation service
        # will reject a candidate if language_required=True.
        if not languages:
            return None

        images = item.get("images", [])
        image_url = None
        if isinstance(images, list) and images:
            first = images[0]
            if isinstance(first, dict):
                image_url = str(first.get("url") or "") or None

        duration_ms = item.get("duration_ms")
        duration_seconds = (
            int(duration_ms) // 1000
            if isinstance(duration_ms, (int, float))
            else None
        )

        return PodcastSearchResult(
            provider=PodcastProviderName.SPOTIFY,
            external_id=external_id,
            title=title,
            description=str(item.get("description") or ""),
            publisher="Spotify Podcast",
            languages=languages,
            url=url,
            image_url=image_url,
            published_at=str(
                item.get("release_date") or ""
            ) or None,
            duration_seconds=duration_seconds,
            explicit=bool(item.get("explicit", False)),
            metadata={
                "provider": "spotify",
                "content_type": "episode",
            },
        )

The builder consumes the application-facing
MemoryRecord contract returned by MemoryService.

No LLM, provider, database, or external service is used.
"""

from backend.memory.memory_record import (
    MemoryRecord,
)

from src.ai_runtime.memory.context_builder import (
    DEFAULT_MAX_CONTEXT_CHARACTERS,
    DEFAULT_MAX_MEMORY_ITEMS,
    MemoryContextBuilder,
)


def make_memory(
    *,
    user_id: str = "user-001",
    category: str = "preference",
    content: str = (
        "Prefers technology topics."
    ),
) -> MemoryRecord:
    return MemoryRecord(
        user_id=user_id,
        category=category,
        content=content,
        created_at=__import__(
            "datetime"
        ).datetime.now(
            __import__(
                "datetime"
            ).timezone.utc
        ),
    )


def test_empty_memory_returns_empty_context() -> None:
    builder = MemoryContextBuilder()

    result = builder.build(
        user_id="user-001",
        request="European AI regulation",
        memories=[],
    )

    assert result.is_empty is True
    assert result.entries == ()
    assert result.text == ""


def test_relevant_preference_is_included() -> None:
    builder = MemoryContextBuilder()

    memory = make_memory(
        content=(
            "Prefers European AI regulation topics."
        ),
    )

    result = builder.build(
        user_id="user-001",
        request=(
            "latest European AI regulation"
        ),
        memories=[memory],
    )

    assert result.memory_count == 1
    assert result.entries == (memory,)
    assert (
        "European AI regulation"
        in result.text
    )


def test_irrelevant_memory_is_excluded() -> None:
    builder = MemoryContextBuilder()

def test_ordering_is_deterministic() -> None:
    builder = MemoryContextBuilder()

    first_memory = make_memory(
        content=(
            "Prefers European AI regulation."
        ),
    )

    second_memory = make_memory(
        content=(
            "Interested in European AI policy."
        ),
    )

    memories = [
        first_memory,
        second_memory,
    ]

    first = builder.build(
        user_id="user-001",
        request="European AI regulation",
        memories=memories,
    )

    second = builder.build(
        user_id="user-001",
        request="European AI regulation",
        memories=memories,
    )

    assert first.entries == second.entries
    assert first.text == second.text

from datetime import datetime

import pytest

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


def test_memory_entry_creates_valid_preference():

    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.PREFERENCE,
        content="Prefers lightweight laptops.",
    )

    assert memory.user_id == "user-001"

    assert (
        memory.memory_type
        == MemoryType.PREFERENCE
    )

    assert (
        memory.content
        == "Prefers lightweight laptops."
    )

    assert memory.memory_id

    assert isinstance(
        memory.created_at,
        datetime,
    )


def test_memory_entry_supports_discovery_context():

    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.DISCOVERY_CONTEXT,
        content="Looking for a laptop under EUR 1000.",
    )

    assert (
        memory.memory_type
        == MemoryType.DISCOVERY_CONTEXT
    )


def test_memory_entry_supports_saved_items():

    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.SAVED_ITEM,
        content="MacBook Air M4",
        metadata={
            "category": "laptop",
        },
    )

    assert (
        memory.memory_type
        == MemoryType.SAVED_ITEM
    )

    assert (
        memory.metadata["category"]
        == "laptop"
    )


def test_memory_requires_user_id():

    with pytest.raises(
        ValueError,
        match="user_id cannot be empty",
    ):
        MemoryEntry(
            user_id="",
            memory_type=MemoryType.PREFERENCE,
            content="Prefers lightweight products.",
        )


def test_memory_requires_content():

    with pytest.raises(
        ValueError,
        match="content cannot be empty",
    ):
        MemoryEntry(
            user_id="user-001",
            memory_type=MemoryType.PREFERENCE,
            content="",
        )


def test_memory_to_dict_is_persistence_safe():

    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.PREFERENCE,
        content="Prefers lightweight laptops.",
    )

    data = memory.to_dict()

    assert data["memory_id"] == memory.memory_id

    assert data["user_id"] == "user-001"

    assert (
        data["memory_type"]
        == "preference"
    )

    assert (
        data["content"]
        == "Prefers lightweight laptops."
    )

    assert isinstance(
        data["created_at"],
        str,
    )

    assert isinstance(
        data["updated_at"],
        str,
    )

def context() -> LearningContext:
    return LearningContext(
        world_id="French World",
        target_language="fr",
        learner_level="B2",

def test_agent_is_provider_agnostic_and_uses_preferences() -> None:

        user_id="user-001",

        content="Prefers technology podcasts.",
        metadata={
            "topics": "artificial intelligence, European regulation",
        },

        user_id="user-001",

        memories=[memory],
    )

    assert intent.learning_context.target_language == "fr"
    assert intent.learning_context.world_id == "French World"
    assert "artificial intelligence" in intent.topics
    assert "European regulation" in intent.topics
    assert intent.language_required is True

        user_id="user-001",

        content="Discussed artificial intelligence.",
    )

    intent = PodcastDiscoveryAgent().build_search_intent(
        user_id="user-001",

import pytest

from httpx import ASGITransport, AsyncClient

from backend.api.main import app
from backend.api import routes


class FakeCompanionService:
    """
    Test double for the application service.

    This test deliberately does not invoke Parallel.
    """

    def __init__(
        self,
        character,
    ):
        self.character = character
        self.received_context = None
        self.received_use_research = None

    async def interact(
        self,
        context,
        use_research=False,
    ):
        self.received_context = context
        self.received_use_research = (
            use_research
        )

        return {
            "character": self.character.name,
            "conversation": {
                "character": self.character.name,
                "message": (
                    f"{self.character.name}: "
                    f"Test response"
                ),
                "intelligence_context": (
                    "TEST INTELLIGENCE"
                    if use_research
                    else None
                ),
            },
            "research": (
                {
                    "query": context.request,
                    "test": True,
                }
                if use_research
                else None
            ),
            "quality": {
                "quality_score": 1.0,
                "passed": True,
                "findings": [],
                "recommendations": [],
            },
            "memory_count": 0,
        }


@pytest.mark.asyncio
async def test_chat_passes_research_false(
    monkeypatch,
):
    """
    Verify /chat passes research=False
    to CompanionService.
    """

    captured = {}

    def fake_factory(
        character,
    ):
        service = FakeCompanionService(
            character
        )

        captured["service"] = service

        return service

    monkeypatch.setattr(
        routes,
        "create_companion_service",
        fake_factory,
    )

    transport = ASGITransport(
        app=app
    )

    async with AsyncClient(
        transport=transport,
        base_url="http://test",
    ) as client:

        response = await client.post(
            "/chat",
            json={
                "user_id": "api-test-user",
                "character": "yeti",
                "message": (
                    "Explain artificial intelligence"
                ),
                "research": False,
            },
        )

    assert response.status_code == 200

    data = response.json()

    service = captured["service"]

    assert (
        service.received_use_research
        is False
    )

    assert (

@pytest.mark.asyncio
async def test_companion_passes_research_to_conversation(
    tmp_path,
):

    conversation_agent = (
        FakeConversationAgent()
    )

    service = CompanionService(
        character=create_character(),
        conversation_agent=(
            conversation_agent
        ),
        research_agent=(
            FakeResearchAgent()
        ),
        quality_agent=(
            FakeQualityAgent()
        ),
        memory_service=(
            create_memory_service(
                tmp_path
            )
        ),
    )

    context = ExecutionContext(
        user_id="demo-user",
        request=(
            "Explain the latest "
            "European AI regulation"
        ),
    )

    result = await service.interact(
        context,
        use_research=True,
    )

    assert result["character"] == (
        "Yeti"
    )

    assert (
        result["research"]
        is not None
    )

    assert isinstance(
        result["research"],
        IntelligenceContext,
    )

    received_context = (
        conversation_agent.received_context
    )

    assert (
        received_context is not None
    )

    assert (
        received_context
        is not context
    )

    assert (
        received_context.intelligence_context
        is not None
    )

    assert (
        received_context.intelligence_context.query
        == context.request
    )

    assert (
        received_context.intelligence_context.source_count
        == 1
    )

    assert (
        result["conversation"][
            "intelligence_context"
        ]
        is received_context.intelligence_context
    )

    assert (
        context.intelligence_context
        is None
    )
