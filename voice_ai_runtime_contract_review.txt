"""
Speech Provider Contract.

Defines the abstraction for speech services.

Supported capabilities:
- Speech to Text (STT)
- Text to Speech (TTS)

The AI Runtime depends on this contract,
not on a specific vendor.

Possible implementations:
- OpenAI Speech
- ElevenLabs
- Azure Speech
- Local models

Does NOT:
- Manage conversations.
- Manage characters.
- Manage LiveKit rooms.
- Store user data.
"""

from __future__ import annotations

from abc import (
    ABC,
    abstractmethod,
)



class SpeechProvider(
    ABC
):
    """
    Abstract speech provider.

    Provider implementations must
    implement this interface.
    """


    @abstractmethod
    def transcribe(
        self,
        audio: bytes,
        language_code: str,
    ) -> str:
        """
        Convert audio into text.

        Parameters
        ----------
        audio:
            Raw audio bytes.

        language_code:
            Expected language.

        Returns
        -------
        str
            Transcribed text.
        """

        raise NotImplementedError



    @abstractmethod
    def synthesize(
        self,
        text: str,
        voice_profile: str,
        language_code: str,
    ) -> bytes:
        """
        Convert text into audio.

        Parameters
        ----------
        text:
            Text to speak.

        voice_profile:
            Character voice profile.

        language_code:
            Output language.

        Returns
        -------
        bytes
            Generated audio.
        """

        raise NotImplementedError

"""
Speech Provider Factory.

Creates speech provider implementations
from configuration.

Responsibilities:
- Convert configuration into providers.
- Centralize provider selection.

Does NOT:
- Execute speech operations.
- Store credentials.
- Manage runtime sessions.
"""

from __future__ import annotations


from app.configuration.providers.speech_provider_config import (
    SpeechProviderConfig,
)


from app.providers.speech.speech_provider import (
    SpeechProvider,
)


from app.providers.speech.openai_speech_provider import (
    OpenAISpeechProvider,
)



class SpeechProviderFactory:
    """
    Factory for speech providers.
    """

    @staticmethod
    def create(
        config: SpeechProviderConfig,
    ) -> SpeechProvider:
        """
        Create speech provider from configuration.

        Parameters
        ----------
        config:
            Provider configuration.

        Returns
        -------
        SpeechProvider
            Configured speech provider.

        Raises
        ------
        ValueError
            If provider is unsupported.
        """

        provider_name = (
            config.provider_name
            .lower()
            .strip()
        )


        if provider_name == "openai":

            return OpenAISpeechProvider()


        raise ValueError(
            f"Unsupported speech provider: {provider_name}"
        )
