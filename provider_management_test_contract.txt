def test_recommendation_is_generated_from_comparison_evidence() -> None:
    """
    Validate that provider comparison evidence can
    produce an optimisation recommendation.
    """

    comparison_service = (
        ProviderComparisonService()
    )

    optimisation_service = (
        CapabilityOptimisationService()
    )

    events = [
        create_event(
            provider="openai",
            model="gpt-5-mini",
            capability="conversation",
            latency_ms=800,
            cost="0.010",
            success=True,
        ),
        create_event(
            provider="openai",
            model="gpt-5-mini",
            capability="conversation",
            latency_ms=850,
            cost="0.010",
            success=True,
        ),
        create_event(
            provider="gemini",
            model="gemini-2.5-flash",
            capability="conversation",
            latency_ms=1400,
            cost="0.015",
            success=False,
        ),
        create_event(
            provider="gemini",
            model="gemini-2.5-flash",
            capability="conversation",
            latency_ms=1300,
            cost="0.015",
            success=True,
        ),
    ]

    comparisons = comparison_service.compare(
        events=events
    )

    recommendation = (
        optimisation_service
        .recommend_from_comparisons(
            capability="conversation",
            current_provider="gemini",
            current_model="gemini-2.5-flash",
            comparisons=comparisons,
        )
    )

    assert (
        recommendation.capability
        == "conversation"
    )

    assert (
        recommendation.current_provider
        == "gemini"
    )

    assert (
        recommendation.current_model
        == "gemini-2.5-flash"
    )

    assert (
        recommendation.recommended_provider
        == "openai"
    )

    assert (
        recommendation.recommended_model
        == "gpt-5-mini"
    )

    assert recommendation.confidence > 0.0

    assert recommendation.confidence <= 99.0

    assert (
        "strongest observed"
        in recommendation.reason
    )

    capabilities.

    Summarisation evidence makes Gemini stronger.

    The optimiser must evaluate each capability
    independently.

    comparison_service = (
        ProviderComparisonService()
    )

    optimisation_service = (
        CapabilityOptimisationService()
    )

    events = [
        create_event(
            provider="openai",
            model="gpt-5-mini",
            capability="conversation",
            latency_ms=800,
            cost="0.010",
            success=True,
        ),

            provider="gemini",
            model="gemini-2.5-flash",
            capability="conversation",
            latency_ms=1400,
            cost="0.015",
            success=True,
        ),
        create_event(
            provider="gemini",
            model="gemini-2.5-flash",
            capability="summarisation",
            latency_ms=600,
            cost="0.006",
            success=True,
        ),
        create_event(
            provider="ollama",

        == "conversation"

- status counts;
- pending approval detection.

The service must not mutate workflow state.
"""

from datetime import datetime, timezone

import pytest

from app.application.provider_management.escalation_history_service import (
    EscalationHistoryService,
)

from app.application.provider_management.escalation_workflow_repository import (
    InMemoryEscalationWorkflowRepository,
)

from app.application.provider_management.escalation_workflow_service import (
    EscalationWorkflow,
    EscalationWorkflowStatus,
)


def create_workflow(
    *,
    escalation_id: str,
    status: EscalationWorkflowStatus,
    minute: int,
) -> EscalationWorkflow:
    """
    Create deterministic workflow history data.
    """

    return EscalationWorkflow(
        escalation_id=escalation_id,
        status=status,
        created_at=datetime(
            2026,
            8,
            14,
            14,
            minute,
            tzinfo=timezone.utc,
        ),
        notification_recipient=(
            "operations@example.com"
            if status
            != EscalationWorkflowStatus
            .PENDING_NOTIFICATION
            else None
        ),
        notification_subject=(
            "EasyChatBot AI Operations Escalation"
            if status
            != EscalationWorkflowStatus
            .PENDING_NOTIFICATION
            else None
        ),
        approved_at=(
            datetime(
                2026,
                8,
                14,
                15,
                minute,
                tzinfo=timezone.utc,
            )
            if status
            == EscalationWorkflowStatus
            .APPROVED
            else None
        ),
        rejected_at=(
            datetime(
                2026,
                8,
                14,
                15,
                minute,
                tzinfo=timezone.utc,
            )
            if status
            == EscalationWorkflowStatus
            .REJECTED
            else None
        ),
        decision_reason=(
            "Operations decision recorded."
            if status
            in (
                EscalationWorkflowStatus.APPROVED,
                EscalationWorkflowStatus.REJECTED,
            )
            else None
        ),
    )


def create_service() -> (
    tuple[
        EscalationHistoryService,
        InMemoryEscalationWorkflowRepository,
    ]
):
    """
    Create a history service and repository.
    """

    repository = (
        InMemoryEscalationWorkflowRepository()
    )

    service = EscalationHistoryService(
        repository=repository
    )

    return service, repository


def populate_repository(
    repository: InMemoryEscalationWorkflowRepository,
) -> None:
    """
    Populate deterministic escalation history.
    """

    repository.save(
        create_workflow(
            escalation_id="esc-001",
            status=(
                EscalationWorkflowStatus
                .PENDING_NOTIFICATION
            ),
            minute=30,
        )
    )

    repository.save(
        create_workflow(
            escalation_id="esc-002",
            status=(
                EscalationWorkflowStatus
                .AWAITING_APPROVAL
            ),
            minute=31,
        )
    )

    repository.save(
        create_workflow(
            escalation_id="esc-003",
            status=(
                EscalationWorkflowStatus
                .APPROVED
            ),
            minute=32,
        )
    )

    repository.save(
        create_workflow(
            escalation_id="esc-004",
            status=(
                EscalationWorkflowStatus
                .REJECTED
            ),
            minute=33,
        )
    )

    repository.save(
        create_workflow(
            escalation_id="esc-005",
            status=(
                EscalationWorkflowStatus
                .AWAITING_APPROVAL
            ),
            minute=34,
        )
    )


def test_list_history_returns_all_workflows() -> None:
    """
    Validate complete history retrieval.

    service, repository = create_service()

    populate_repository(repository)

    history = service.list_history()

    assert len(history) == 5

    assert [

    ] == [

        "esc-003",

    ]

def test_facade_result_is_json_safe() -> None:
    """
    Validate that the complete dashboard result consists
    only of JSON-compatible primitive/container values.

    This deliberately avoids importing a JSON framework so
    the application layer remains infrastructure-independent.
    """

    facade, repository = create_service()

    repository.save(
        create_workflow(
            escalation_id="esc-001",
            status=(
                EscalationWorkflowStatus
                .AWAITING_APPROVAL
            ),
            minute=30,
        )
    )

    result = facade.get_dashboard(
        active_providers=(
            create_active_providers()
        )
    )

    def assert_json_safe(
        value: object,
    ) -> None:
        if value is None:
            return

        if isinstance(
            value,
            (
                str,
                int,
                float,
                bool,
            ),
        ):
            return

        if isinstance(value, list):
            for item in value:
                assert_json_safe(item)
            return

        if isinstance(value, dict):
            for key, item in value.items():
                assert isinstance(
                    key,
                    str,
                )
                assert_json_safe(item)
            return

        raise AssertionError(
            "Dashboard contains a non-JSON-safe "
            f"value: {type(value)!r}"
        )

    assert_json_safe(result)

    """Validate that the response contains JSON-safe values."""

    event = create_event(
        event_type=OperationalEventType.AGENT_EXECUTION_FAILED,
        severity=OperationalEventSeverity.ERROR,
    )
    service = create_service_mock((event,))

    api = OperationsCenterOperationalEventsAPI(
        operational_event_service=service,
    )

    result = api.list_events()

    def assert_json_safe(value: object) -> None:
        if value is None:
            return

        if isinstance(value, (str, int, float, bool)):
            return

        if isinstance(value, list):
            for item in value:
                assert_json_safe(item)
            return

        if isinstance(value, dict):
            for key, item in value.items():
                assert isinstance(key, str)
                assert_json_safe(item)
            return

        raise AssertionError(
            "Response contains a non-JSON-safe "
            f"value: {type(value)!r}"
        )

    assert_json_safe(result)

            10,
