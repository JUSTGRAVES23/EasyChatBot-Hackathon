"""
EasyChatBot Character Registry.

The registry provides the approved characters available
to the application.

Characters define identity and behaviour.

Agents remain responsible for capabilities and execution.

Character selection readiness is determined by artwork
availability.

A character without approved EasyChatBot artwork is not
selectable from the character selection experience.
"""

from characters.character_profile import (
    CharacterProfile,
)


class CharacterRegistry:
    """
    Registry of approved EasyChatBot AI companions.

    Character identity and behaviour belong here.

    Artwork readiness also belongs to the character profile
    because artwork availability determines whether the
    character may be offered for selection.
    """

    def __init__(self) -> None:

        self._characters = {

            "yeti": CharacterProfile(
                character_id="yeti",
                name="Yeti",
                description=(
                    "A friendly AI companion who helps "
                    "you discover knowledge and practice languages."
                ),
                personality=(
                    "Curious, encouraging and supportive."
                ),
                learning_style=(
                    "Conversation based learning."
                ),
                artwork_available=True,
                artwork_status="approved",
                selectable=True,
            ),

            "clara": CharacterProfile(
                character_id="clara",
                name="Clara",
                description=(
                    "A professional learning companion "
                    "focused on technology and career topics."
                ),
                personality=(
                    "Analytical, structured and supportive."
                ),
                learning_style=(
                    "Structured and practical learning."
                ),
                artwork_available=True,
                artwork_status="approved",
                selectable=True,
            ),

            "luminate": CharacterProfile(
                character_id="luminate",
                name="Luminate",
                description=(
                    "A creative companion for culture, "
                    "ideas and exploration."
                ),
                personality=(
                    "Creative, inspiring and curious."
                ),
                learning_style=(
                    "Exploration and discussion."
                ),
                artwork_available=True,
                artwork_status="approved",
                selectable=True,
            ),

        }

    def get(
        self,
        character_id: str,
    ) -> CharacterProfile:
        """
        Return an approved character.

        Raises:
            ValueError:
                If the character does not exist.
        """

        normalized_id = (
            character_id.strip().lower()
        )

        try:

            return self._characters[
                normalized_id
            ]

        except KeyError as exc:

            raise ValueError(
                f"Unknown character: {character_id}"
            ) from exc

    def list_characters(
        self,
    ) -> list[CharacterProfile]:
        """
        Return all approved characters.
        """

        return list(
            self._characters.values()
        )

    def list_selectable_characters(
        self,
    ) -> list[CharacterProfile]:
        """
        Return characters that are currently available
        for learner selection.

        Selection requires:

        - approved artwork;
        - artwork_available=True;
        - selectable=True.
        """

        return [
            character
            for character in self._characters.values()
            if (
                character.artwork_available
                and character.artwork_status == "approved"
                and character.selectable
            )
        ]

"""
Character Resolver.

Application service responsible for converting
character profile configuration into runtime
character context.

Responsibilities:
- Resolve character profiles.
- Extract interaction-level character data.
- Create CharacterContext.

Does not:
- Load files directly.
- Manage AI providers.
- Create voice sessions.
- Manage subscriptions.
"""

from __future__ import annotations

import logging

from app.domain.uai.context.character_context import (
    CharacterContext,
)


logger = logging.getLogger(__name__)


class CharacterResolver:
    """
    Resolves character runtime context.
    """

    def __init__(
        self,
        *,
        character_repository,
    ) -> None:
        """
        Initialise resolver.

        Parameters
        ----------
        character_repository:
            Repository providing character profiles.
        """

        self._character_repository = (
            character_repository
        )


    def resolve(
        self,
        character_id: str,
    ) -> CharacterContext:
        """
        Resolve character configuration.

        Parameters
        ----------
        character_id:
            Character identifier.

        Returns
        -------
        CharacterContext

        Raises
        ------
        ValueError
            If character does not exist.
        """

        logger.info(
            "Resolving character context",
            extra={
                "character_id": character_id,
            },
        )


        character = (
            self._character_repository
            .get_by_id(character_id)
        )


        if character is None:
            raise ValueError(
                f"Character not found: {character_id}"
            )


        identity = (
            character.get(
                "identity",
                {},
            )
        )


        behaviour = (
            character.get(
                "behaviour",
                {},
            )
        )


        context = CharacterContext(
            character_id=(
                character["id"]
            ),

            name=(
                character["name"]
            ),

            personality_reference=(
                identity.get(
                    "role",
                    "default",
                )
            ),

            teaching_style=(
                identity.get(
                    "communication_style",
                    "default",
                )
            ),

            premium_required=(
                character.get(
                    "premium_required",
                    False,
                )
            ),
        )


        logger.info(
            "Character context resolved",
            extra={
                "character_id": character_id,
            },
        )


        return context

"""
YAML Character Repository.

Infrastructure adapter responsible for loading
existing Character Profile YAML files.

Source of truth:

FrenchTeacherAI/docs/specifications/characters/profiles

This repository does not contain character
business rules.

Responsibilities:
- Locate character profile files.
- Load YAML.
- Return raw character configuration.

Business decisions belong to:
- Character Resolver
- Entitlement services
- UAI Runtime
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

import yaml


class YamlCharacterRepository:
    """
    Loads Character Profiles from YAML files.
    """

    def __init__(
        self,
        *,
        profiles_path: Path,
    ) -> None:
        """
        Initialise repository.

        Parameters
        ----------
        profiles_path:
            Location containing character YAML profiles.
        """

        self._profiles_path = profiles_path


    def get_by_id(
        self,
        character_id: str,
    ) -> dict[str, Any] | None:
        """
        Load a character profile.

        Parameters
        ----------
        character_id:
            Character identifier.

        Returns
        -------
        dict | None
            Character configuration if found.
        """

        file_path = (
            self._profiles_path
            / f"{character_id}.yaml"
        )

        if not file_path.exists():
            return None


        with file_path.open(
            "r",
            encoding="utf-8",
        ) as file:

            data = yaml.safe_load(file)


        return data

"""
Companion Service.

Application orchestration layer for EasyChatBot.

Coordinates:

Character
Conversation Agent
Research Agent
Quality Agent
Memory Service
Memory Context Builder

Live Intelligence + Memory flow:

CompanionService
|
+--> MemoryService
|        |
|        v
|   MemoryRecord[]
|        |
|        v
|   MemoryContextBuilder
|        |
|        v
|   MemoryContext
|
+--> ResearchAgent
|        |
|        v
|   IntelligenceContext
|
+--> ConversationAgent
|        |
|        v
|   LLMService
|        |
|        v
|   LLMProvider
|
+--> QualityAgent
|
v
Interaction Result

Responsibilities:

- orchestrate companion interactions
- retrieve memories
- build bounded learner memory context
- optionally execute research
- enrich the execution context with memory
- enrich the execution context with live intelligence
- execute the ConversationAgent
- execute the QualityAgent
- persist the interaction
- return the combined result

Does not:

- create LLM providers
- select LLM providers
- call provider SDKs
- call OpenAI directly
- call Ollama directly
- manage provider routing
- build provider-specific prompts
- persist memory directly
"""

from __future__ import annotations

from dataclasses import replace
from typing import Any

from agents.conversation_agent import (
    ConversationAgent,
)

from agents.research_agent import (
    ResearchAgent,
)

from agents.quality_agent import (
    QualityAgent,
)

from characters.character_profile import (
    CharacterProfile,
)

from memory.memory_service import (
    MemoryService,
)

from runtime.context.execution_context import (
    ExecutionContext,
)

from src.ai_runtime.memory.context_builder import (
    MemoryContextBuilder,
)


class CompanionService:
    """
    Coordinates AI companion experiences.

    The service orchestrates agents but does not contain
    provider-specific logic.

    Providers are injected into agents by the composition
    layer.

    Memory retrieval remains owned by MemoryService.

    MemoryContextBuilder converts already-retrieved memory
    into bounded runtime context.

    ExecutionContext remains immutable. Enrichment is
    performed using dataclasses.replace().
    """

    def __init__(
        self,
        character: CharacterProfile,
        conversation_agent: ConversationAgent | None = None,
        research_agent: ResearchAgent | None = None,
        quality_agent: QualityAgent | None = None,
        memory_service: MemoryService | None = None,
        memory_context_builder: (MemoryContextBuilder | None) = None,
    ) -> None:
        """
        Initialise the CompanionService.

        Dependencies are injectable so the service can be
        tested without invoking real infrastructure.
        """

        self.character = character

        self.conversation_agent = (
            conversation_agent
            if conversation_agent is not None
            else ConversationAgent(character)
        )

        self.research_agent = (
            research_agent if research_agent is not None else ResearchAgent()
        )

        self.quality_agent = (
            quality_agent if quality_agent is not None else QualityAgent()
        )

        self.memory_service = (
            memory_service if memory_service is not None else MemoryService()
        )

        self.memory_context_builder = (
            memory_context_builder
            if memory_context_builder is not None
            else MemoryContextBuilder()
        )

    async def interact(
        self,
        context: ExecutionContext,
        use_research: bool = False,
    ) -> dict[str, Any]:
        """
        Execute a complete companion interaction.

        Flow:

        MemoryService
            â†“
        MemoryContextBuilder
            â†“
        optional memory-enriched ExecutionContext
            â†“
        optional ResearchAgent
            â†“
        optional intelligence-enriched ExecutionContext
            â†“
        ConversationAgent
            â†“
        QualityAgent
            â†“
        MemoryService.remember()

        The original ExecutionContext is never mutated.

        Parameters
        ----------
        context:
            Runtime execution context.

        use_research:
            Whether live research should be performed
            before the conversation response.

        Returns
        -------
        dict[str, Any]
            Combined companion interaction result.

        Raises
        ------
        ValueError
            If the execution context is missing.
        """

        if context is None:
            raise ValueError("Execution context cannot be None.")

        # -----------------------------------------------------
        # Step 1 â€” Retrieve existing learner memory
        # -----------------------------------------------------

        memories = self.memory_service.recall(context.user_id)

        # -----------------------------------------------------
        # Step 2 â€” Build bounded learner memory context
        # -----------------------------------------------------

        memory_context = self.memory_context_builder.build(
            user_id=context.user_id,
            request=context.request,
            memories=memories,
        )

        # -----------------------------------------------------
        # Step 3 â€” Attach memory only when relevant memory
        # exists.
        #
        # If there is no relevant memory, preserve the
        # original ExecutionContext object.
        # -----------------------------------------------------

        if memory_context.is_empty:
            conversation_context = context
        else:
            conversation_context = replace(
                context,
                memory_context=memory_context,
            )

        # -----------------------------------------------------
        # Step 4 â€” Optional Live Intelligence
        #
        # Research receives the current runtime context but
        # does not receive or manage memory itself.
        #
        # If research succeeds, create another immutable
        # context preserving the existing memory_context.
        # -----------------------------------------------------

        research_result = None

        if use_research:
            research_result = await self.research_agent.execute(conversation_context)

            if research_result.success and research_result.output is not None:
                conversation_context = replace(
                    conversation_context,
                    intelligence_context=(research_result.output),
                )

        # -----------------------------------------------------
        # Step 5 â€” Conversation Agent
        #
        # ConversationAgent can now receive:
        #
        #   memory_context
        #   intelligence_context
        #
        # independently or together.
        # -----------------------------------------------------

        conversation_result = await self.conversation_agent.execute(
            conversation_context
        )

        # -----------------------------------------------------
        # Step 6 â€” Quality Agent
        # -----------------------------------------------------

        quality_result = await self.quality_agent.execute(conversation_context)

        # -----------------------------------------------------
        # Step 7 â€” Persist interaction
        #
        # Memory persistence remains owned by MemoryService.
        # -----------------------------------------------------

        self.memory_service.remember(
            user_id=context.user_id,
            category="conversation",
            content=context.request,
        )

        # -----------------------------------------------------
        # Step 8 â€” Return combined result
        # -----------------------------------------------------

        return {
            "character": self.character.name,
            "conversation": (conversation_result.output),
            "research": (
                research_result.output
                if (research_result is not None and research_result.success)
                else None
            ),
            "quality": (quality_result.output),
            "memory_count": len(memories),
            "memory_context_count": (memory_context.memory_count),
        }

"""
EasyChatBot Character Theme API.

Bridges the existing Character Registry with the
Character Theme System.

This module does not own learner state.

It only resolves the visual presentation for an
already-selected character.
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

from app.character_theme_system.application.theme_resolver import (
    CharacterThemeResolver,
)
from app.character_theme_system.configuration.theme_loader import (
    CharacterThemeLoader,
)


class CharacterThemeService:
    """
    Resolve EasyChatBot dashboard themes.

    Character identity remains owned by the backend
    CharacterRegistry.

    Visual presentation remains owned by the
    Character Theme System.
    """

    def __init__(
        self,
        *,
        themes_directory: Path,
    ) -> None:
        self._loader = CharacterThemeLoader.from_directory(
            themes_directory=themes_directory,
        )

        self._resolver = CharacterThemeResolver(
            loader=self._loader,
        )

    def resolve(
        self,
        *,
        character_id: str,
        world_id: str,
        primary_language: str,
        secondary_language: str | None = None,
    ) -> dict[str, Any]:
        """
        Resolve a dashboard theme.

        The learner's selected character/world/languages
        are inputs only. This service does not persist
        them.
        """

        result = self._resolver.resolve(
            character_id=character_id,
            world_id=world_id,
            primary_language=primary_language,
            secondary_language=secondary_language,
        )

        return {
            "character_id": result.character_id,
            "world_id": result.world_id,
            "languages": {
                "primary": result.context.primary_language,
                "secondary": (
                    result.context.secondary_language
                ),
            },
            "theme": result.theme,
        }

# =========================================================
# Character Theme Service
# =========================================================

# The API package lives under:
#
#     backend/api/
#
# Therefore parents[1] resolves to:
#
#     backend/
#
# The active theme configuration is owned by the backend:
#
#     backend/app/configuration/character_themes/themes/
#
# Do not resolve this from the repository-root app/ directory.
_theme_directory = (
    Path(__file__).resolve().parents[1]
    / "app"
    / "configuration"
    / "character_themes"
    / "themes"
)

    themes_directory=_theme_directory,
)


# =========================================================
# Character Theme Endpoint
# =========================================================


@router.get("")
def get_character_theme(
    character_id: str,
    world_id: str,
    primary_language: str,
    secondary_language: str | None = None,
):
    """
    Resolve the visual theme for a character/world/language
    context.

    This endpoint does not store learner state.

    It only resolves presentation from the supplied context.
    """

    if not character_id.strip():
        raise HTTPException(
            status_code=400,
            detail="character_id is required",
        )

    if not world_id.strip():
        raise HTTPException(
            status_code=400,
            detail="world_id is required",
        )

    if not primary_language.strip():
        raise HTTPException(
            status_code=400,
            detail="primary_language is required",
        )

    try:
        return character_theme_service.resolve(
            character_id=character_id,
            world_id=world_id,
            primary_language=primary_language,
            secondary_language=secondary_language,
        )

    except FileNotFoundError as exc:
        raise HTTPException(
            status_code=404,
            detail=str(exc),
        ) from exc

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail="Unable to resolve character theme.",
        ) from exc

/*
 * =========================================================
 * Character Artwork Lookup
 * =========================================================
 */

function getCharacterArtwork(
    characterId
) {

    if (
        !characterId
    ) {

        return null;

    }


    return (
        characterArtwork[
            characterId
        ] || null
    );

}


/*
 * =========================================================
 * Artwork Availability
 * =========================================================
 */

function hasCharacterArtwork(
    characterId
) {

    const artwork =
        getCharacterArtwork(
            characterId
        );


    if (!artwork) {

        return false;

    }


    return (
        artwork.artworkAvailable === true
    );

}


/*
 * =========================================================
 * Selection Availability
 * =========================================================
 *
 * A character may exist in the catalogue but remain
 * unavailable for learner selection until its EasyChatBot
 * artwork has been approved.
 */

function isCharacterSelectable(
    characterId
) {

    const artwork =
        getCharacterArtwork(
            characterId
        );


    if (!artwork) {

        return false;

    }


    return (
        artwork.artworkAvailable === true &&
        artwork.artworkStatus === "approved" &&
        artwork.selectable === true
    );

}


/*
 * =========================================================
 * Front Card URL
 * =========================================================
 */

function getCharacterFrontCard(
    characterId
) {

    const artwork =
        getCharacterArtwork(
            characterId
        );


    if (
        !artwork ||
        !artwork.frontCard
    ) {

        return null;

    }


    return artwork.frontCard;

}

export default function WorldPrivateChatPage() {
  const { worldSlug } = useParams<{
    worldSlug: string;
  }>();

  const world = useMemo<WorldCatalogueEntry | undefined>(
    () => getWorldBySlug(worldSlug ?? ""),
    [worldSlug],
  );

  const presentation = useMemo(
    () => getWorldPresentation(worldSlug ?? ""),
    [worldSlug],
  );

  const [conversationId, setConversationId] =
    useState<string | null>(null);

  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [draft, setDraft] = useState("");
  const [loadingConversation, setLoadingConversation] = useState(true);
  const [sending, setSending] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let active = true;

    async function initialiseConversation() {
      if (!world) {
        setLoadingConversation(false);
        return;
      }

      setLoadingConversation(true);
      setError(null);
      setConversationId(null);
      setMessages([]);

      try {
        const conversation =
          await createPrivateChatConversation(world.worldId);

        if (!active) return;

        setConversationId(conversation.conversation_id);
      } catch (cause) {
        if (!active) return;

        setError(
          cause instanceof Error
            ? cause.message
            : "Unable to create the private conversation.",
        );
      } finally {
        if (active) {
          setLoadingConversation(false);
        }
      }
    }

    void initialiseConversation();

    return () => {
      active = false;
    };
  }, [world]);

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    const message = draft.trim();

    if (!message || !conversationId || sending) {
      return;
    }

    setSending(true);
    setError(null);

    const userMessage: ChatMessage = {
      id: `user-${Date.now()}`,
      role: "user",
      content: message,
    };

    setMessages((current) => [...current, userMessage]);
    setDraft("");

    try {
      const result =
        await sendPrivateChatMessage(
          conversationId,
          message,
        );

      setMessages((current) => [
        ...current,
        {
          id: `assistant-${Date.now()}`,
          role: "assistant",
          content: result.response,
        },
      ]);
    } catch (cause) {
      setError(
        cause instanceof Error
          ? cause.message
          : "Unable to send the message.",
      );
    } finally {
      setSending(false);
    }
  }

  function usePrompt(prompt: string) {
    setDraft(prompt);
  }

  if (!world) {
    return (

            </div>
          </section>
        </div>
      </main>
    );
  }
