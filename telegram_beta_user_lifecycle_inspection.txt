"""
YAML Character Repository.

Infrastructure adapter responsible for loading
existing Character Profile YAML files.

Source of truth:

FrenchTeacherAI/docs/specifications/characters/profiles

This repository does not contain character
business rules.

Responsibilities:
- Locate character profile files.
- Load YAML.
- Return raw character configuration.

Business decisions belong to:
- Character Resolver
- Entitlement services
- UAI Runtime
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

import yaml


class YamlCharacterRepository:
    """
    Loads Character Profiles from YAML files.
    """

    def __init__(
        self,
        *,
        profiles_path: Path,
    ) -> None:
        """
        Initialise repository.

        Parameters
        ----------
        profiles_path:
            Location containing character YAML profiles.
        """

        self._profiles_path = profiles_path


    def get_by_id(
        self,
        character_id: str,
    ) -> dict[str, Any] | None:
        """
        Load a character profile.

        Parameters
        ----------
        character_id:
            Character identifier.

        Returns
        -------
        dict | None
            Character configuration if found.
        """

        file_path = (
            self._profiles_path
            / f"{character_id}.yaml"
        )

        if not file_path.exists():
            return None


        with file_path.open(
            "r",
            encoding="utf-8",
        ) as file:

            data = yaml.safe_load(file)


        return data

    VoiceCallEventRepository,

        self,

            event
            for event in self._events
            if event.voice_call_id

        ]

        self,
    ) -> list[VoiceCallEvent]:
        """
        Return all stored events.

        Useful for Beta diagnostics

"""
LiveKit Agent Dispatcher.

Infrastructure adapter responsible for dispatching
an approved EasyChatBot voice call to the LiveKit
realtime agent worker.

Responsibilities:
- Create a LiveKit agent dispatch.
- Target the EasyChatBot voice worker.
- Pass approved voice-call metadata to the worker.
- Keep LiveKit API details inside infrastructure.

Does NOT:
- Decide whether a call is authorized.
- Create voice-call business state.
- Manage application lifecycle.
- Resolve worlds or characters.
- Execute AI agents.
- Persist conversations.
- Generate participant tokens.
"""

from __future__ import annotations


import json
import logging
import os


from dataclasses import dataclass


from livekit import api


logger = logging.getLogger(
    __name__
)


@dataclass(
    frozen=True,
    slots=True,
)
class LiveKitAgentDispatcher:
    """
    Infrastructure adapter for LiveKit Agent dispatch.

    The application layer supplies an already-approved
    voice-call context. This adapter translates that
    context into a LiveKit AgentDispatch request.

    LiveKit credentials are read from the environment
    and remain inside the infrastructure boundary.
    """

    agent_name: str = "easychatbot-voice"


    def dispatch(
        self,
        *,
        room_name: str,
        learner_id: str,
        world_id: str,
        character_id: str,
        language_code: str,
    ) -> str:
        """
        Dispatch the EasyChatBot voice agent to a room.

        Parameters
        ----------
        room_name:
            LiveKit room that the agent must join.

        learner_id:
            Approved learner identifier.

        world_id:
            Active learning world.

        character_id:
            Approved AI character.

        language_code:
            Active interaction language.

        Returns
        -------
        str
            LiveKit dispatch identifier.

        Raises
        ------
        ValueError
            If required values are empty or credentials
            are not configured.
        """

        self._validate_required_value(
            room_name,
            "room_name",
        )

        self._validate_required_value(
            learner_id,
            "learner_id",
        )

        self._validate_required_value(
            world_id,
            "world_id",
        )

        self._validate_required_value(
            character_id,
            "character_id",
        )

        self._validate_required_value(
            language_code,
            "language_code",
        )


        api_key = os.getenv(
            "LIVEKIT_API_KEY",
        )

        api_secret = os.getenv(
            "LIVEKIT_API_SECRET",
        )

        livekit_url = os.getenv(
            "LIVEKIT_URL",
        )


        if not api_key:
            raise ValueError(
                "LIVEKIT_API_KEY is not configured."
            )


        if not api_secret:
            raise ValueError(
                "LIVEKIT_API_SECRET is not configured."
            )


        if not livekit_url:
            raise ValueError(
                "LIVEKIT_URL is not configured."
            )


        metadata = {
            "learner_id": learner_id.strip(),
            "world_id": world_id.strip(),
            "character_id": character_id.strip(),
            "language_code": language_code.strip(),
        }


        metadata_json = json.dumps(
            metadata,
            separators=(
                ",",
                ":",
            ),
        )


        logger.info(
            "Dispatching EasyChatBot LiveKit agent",
            extra={
                "room_name": room_name,
                "agent_name": self.agent_name,
                "learner_id": learner_id,
                "world_id": world_id,
                "character_id": character_id,
                "language_code": language_code,
            },
        )


        livekit_api = api.LiveKitAPI(
            url=livekit_url,
            api_key=api_key,
            api_secret=api_secret,
        )


        try:
            dispatch = (
                livekit_api
                .agent_dispatch
                .create_dispatch(
                    api.CreateAgentDispatchRequest(
                        agent_name=(
                            self.agent_name
                        ),
                        room=room_name.strip(),
                        metadata=metadata_json,
                    )
                )
            )


            logger.info(
                "LiveKit agent dispatch created",
                extra={
                    "dispatch_id": (
                        dispatch.id
                    ),
                    "room_name": room_name,
                    "agent_name": (
                        self.agent_name
                    ),
                },
            )


            return dispatch.id

        finally:
            livekit_api.aclose()


    @staticmethod
    def _validate_required_value(
        value: str,
        field_name: str,
    ) -> None:
        """
        Validate a required string field.
        """

        if not isinstance(
            value,
            str,
        ):
            raise ValueError(
                f"{field_name} must be a string."
            )


        if not value.strip():
            raise ValueError(
                f"{field_name} cannot be empty."
            )


__all__ = [
    "LiveKitAgentDispatcher",
]

"""
LiveKit Conversation LLM Adapter.

Infrastructure adapter between LiveKit Agents
and EasyChatBot AI Runtime.

Responsibilities:
- Implement LiveKit LLM boundary.
- Translate LiveKit chat requests.
- Forward requests to ConversationRuntime.
- Convert runtime responses back to LiveKit format.

Does NOT:
- Call providers directly.
- Store conversations.
- Manage users.
- Manage LiveKit lifecycle.
- Own business logic.
"""

from __future__ import annotations


from dataclasses import dataclass


from livekit.agents import llm


from app.ai_runtime.conversation.conversation_runtime import (
    ConversationRuntime,
)



@dataclass(
    frozen=True,
    slots=True,
)
class LiveKitConversationLLM(
    llm.LLM,
):
    """
    Adapter exposing EasyChatBot ConversationRuntime
    as a LiveKit Agents LLM implementation.
    """


    conversation_runtime: ConversationRuntime

    character_instruction: str



    async def chat(
        self,
        *,
        chat_ctx: llm.ChatContext,
        tools: list[llm.Tool] | None = None,
        conn_options: llm.LLMOptions | None = None,
    ) -> llm.LLMStream:
        """
        Execute a conversation request.

        LiveKit sends the conversation context.
        EasyChatBot Runtime generates the response.

        The generated response is returned through
        a LiveKit compatible stream.
        """


        messages = []


        for item in chat_ctx.items:

            if hasattr(
                item,
                "role",
            ):

                messages.append(
                    {
                        "role": item.role,
                        "content": (
                            item.content
                            if isinstance(
                                item.content,
                                str,
                            )
                            else ""
                        ),
                    }
                )



        learner_message = ""


        for message in reversed(messages):

            if (
                message["role"]
                ==
                "user"
            ):

                learner_message = (
                    message["content"]
                )

                break



        response = await (
            self.conversation_runtime.respond(
                character_instruction=(
                    self.character_instruction
                ),
                learner_message=(
                    learner_message
                ),
            )
        )


        return (
            self._create_stream(
                response,
            )
        )



    def _create_stream(
        self,
        response: str,
    ) -> llm.LLMStream:
        """
        Convert EasyChatBot response into
        a LiveKit LLMStream.
        """


        stream = (
            llm.LLMStream(
                self,
                chat_ctx=llm.ChatContext(),
            )
        )


        stream._event_queue.put_nowait(
            llm.ChatChunk(
                delta=llm.ChoiceDelta(
                    content=response,
                )
            )
        )


        stream._event_queue.put_nowait(
            None
        )


        return stream



__all__ = [
    "LiveKitConversationLLM",
]

"""
LiveKit Token Service.

Infrastructure service responsible for generating
real LiveKit participant access tokens.

Responsibilities:
- Generate signed LiveKit JWT access tokens.
- Grant participant access to a specific room.
- Keep LiveKit credentials inside infrastructure.
- Return provider-independent VoiceConnectionToken.

Does NOT:
- Handle HTTP.
- Manage application workflows.
- Decide business authorization.
- Persist tokens.
- Execute AI agents.
"""

from __future__ import annotations


from datetime import (
    datetime,
    timedelta,
    timezone,
)


import os
import uuid


from dotenv import load_dotenv


from livekit import api


from app.domain.uai.voice.voice_connection_token import (
    VoiceConnectionToken,
)


load_dotenv()



class LiveKitTokenService:
    """
    Real LiveKit access-token infrastructure adapter.
    """



    def create_token(
        self,
        *,
        room_name: str,
        identity: str | None = None,
        expires_minutes: int = 30,
    ) -> VoiceConnectionToken:
        """
        Create a signed LiveKit participant token.

        Parameters
        ----------
        room_name:
            LiveKit room the participant may join.

        identity:
            Unique LiveKit participant identity.

        expires_minutes:
            Token validity period.
        """



        if not room_name.strip():

            raise ValueError(
                "room_name cannot be empty."
            )



        if expires_minutes <= 0:

            raise ValueError(
                "expires_minutes must be positive."
            )



        participant_identity = (
            identity.strip()
            if identity is not None
            else f"learner-{uuid.uuid4()}"
        )



        if not participant_identity:

            raise ValueError(
                "identity cannot be empty."
            )



        api_key = os.getenv(
            "LIVEKIT_API_KEY"
        )



        api_secret = os.getenv(
            "LIVEKIT_API_SECRET"
        )



        if not api_key:

            raise RuntimeError(
                "LIVEKIT_API_KEY is not configured."
            )



        if not api_secret:

            raise RuntimeError(
                "LIVEKIT_API_SECRET is not configured."
            )



        expires_at = (
            datetime.now(
                timezone.utc
            )
            + timedelta(
                minutes=expires_minutes,
            )
        )



        access_token = (
            api.AccessToken(
                api_key=api_key,
                api_secret=api_secret,
            )
            .with_identity(
                participant_identity,
            )
            .with_name(
                participant_identity,
            )
            .with_ttl(
                timedelta(
                    minutes=expires_minutes,
                )
            )
            .with_grants(
                api.VideoGrants(
                    room_join=True,
                    room=room_name,
                    can_publish=True,
                    can_subscribe=True,
                    can_publish_data=True,
                )
            )
        )



        token = (
            access_token.to_jwt()
        )



        return VoiceConnectionToken(
            token=token,
            room_name=room_name,
            provider="livekit",
            expires_at=expires_at,
        )



__all__ = [
    "LiveKitTokenService",

Infrastructure adapter between LiveKit realtime events
and the EasyChatBot AI Runtime voice layer.

Responsibilities:
- Bridge LiveKit audio events to VoiceSpeechRuntime.
- Keep LiveKit-specific handling inside infrastructure.
- Delegate speech operations to AI Runtime.
- Preserve provider independence.

Does NOT:
- Own AI conversation logic.
- Select speech providers.
- Store conversations.
- Manage learner authorization.
- Manage LiveKit rooms.
- Bypass EasyChatBot Runtime.

from dataclasses import dataclass

    VoiceSpeechRuntime,
)


logger = logging.getLogger(
    __name__

@dataclass(
    frozen=True,
    slots=True,
)
class LiveKitVoiceRuntimeAdapter:
    """
    Adapter between LiveKit and EasyChatBot voice runtime.

    LiveKit remains an infrastructure transport.
    VoiceSpeechRuntime remains the application/runtime
    abstraction for speech operations.
    """

    speech_runtime: VoiceSpeechRuntime


    async def transcribe_audio(
        self,
        *,
        audio: bytes,
        language_code: str,
    ) -> str:
        """

        Delegates processing to the configured
        SpeechProvider through VoiceSpeechRuntime.

            "Transcribing LiveKit audio",
            extra={
                "language_code": language_code,
                "audio_size": len(audio),
            },
        )


        return self.speech_runtime.transcribe(
            audio=audio,
            language_code=language_code,
        )


    async def synthesize_text(
        self,
        *,
        text: str,
        voice_profile: str,
        language_code: str,
    ) -> bytes:
        """
        Convert generated text into audio.

        Delegates processing to the configured
        SpeechProvider through VoiceSpeechRuntime.

            "Synthesizing LiveKit speech",
            extra={
                "language_code": language_code,
                "voice_profile": voice_profile,

class LiveKitVoiceWorker:

    Dependencies are injected through composition.
    """

    context_factory: VoiceCallContextFactory


    interaction_context_service: (
        VoiceInteractionContextService
    )


    conversation_runtime: ConversationRuntime



    async def run(
        self,
        *,
        ctx: JobContext,
    ) -> None:
        """
        Connect to LiveKit and start
        EasyChatBot realtime character session.
        """


        metadata = (
            self._read_job_metadata(
                ctx,
            )
        )


        request = (
            self._create_voice_call_request(
                metadata,
            )
        )



        execution_context = (
            self.context_factory.create(
                request=request,
            )
        )



        interaction_context = (
            self.interaction_context_service.build(
                context=execution_context,
            )
        )



        character = (
            interaction_context.character
        )



        instructions = (
            "You are "
            f"{character.name}. "
            "You are a French learning character. "
            "Your personality reference is "
            f"{character.personality_reference}. "
            "Your teaching style is "
            f"{character.teaching_style}. "
            "Speak naturally and help the learner "
            "practice the target language."
        )



        realtime_agent = LiveKitRealtimeAgent(
            context=interaction_context,
            instructions=instructions,
            conversation_runtime=(
                self.conversation_runtime
            ),
        )



        ctx.log_context_fields = {
            "room_name": ctx.room.name,
            "learner_id": request.learner_id,
            "world_id": request.world_id,
            "character_id": request.character_id,
            "language_code": request.language_code,
        }



        logger.info(
            "Starting EasyChatBot LiveKit voice agent",
            extra={
                "room_name": ctx.room.name,
                "learner_id": request.learner_id,
                "world_id": request.world_id,
                "character_id": request.character_id,
                "language_code": request.language_code,
            },
        )



        await ctx.connect()



        agent = (
            realtime_agent.create_agent()
        )



        session = (
            realtime_agent.create_session()
        )



        await session.start(
            agent=agent,
            room=ctx.room,
        )



    @staticmethod
    def _read_job_metadata(
        ctx: JobContext,
    ) -> dict[str, object]:
        """
        Read and validate LiveKit job metadata.
        """


        raw_metadata = (
            ctx.job.metadata
        )



        if not raw_metadata:
            raise ValueError(
                "LiveKit job metadata is required."
            )



        try:

            metadata = json.loads(
                raw_metadata,
            )

        except json.JSONDecodeError as exc:

            raise ValueError(
                "LiveKit job metadata must be valid JSON."
            ) from exc



        if not isinstance(
            metadata,
            dict,
        ):

            raise ValueError(
                "LiveKit job metadata must be a JSON object."
            )



        return metadata



    @staticmethod
    def _create_voice_call_request(
        metadata: dict[str, object],
    ) -> VoiceCallRequest:
        """
        Convert LiveKit metadata into
        application request contract.
        """


        required_fields = (
            "learner_id",
            "world_id",
            "character_id",
            "language_code",
        )



        values: dict[str, str] = {}



        for field in required_fields:

            value = metadata.get(
                field,
            )



            if not isinstance(
                value,
                str,
            ):

                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' must be a string."
                )



            if not value.strip():

                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' cannot be empty."
                )



            values[field] = value.strip()



        return VoiceCallRequest(
            learner_id=values["learner_id"],
            world_id=values["world_id"],
            character_id=values["character_id"],
            language_code=values["language_code"],
        )

Infrastructure implementation for loading
World configurations from YAML files.

Implements:

WorldRepository
"""

from __future__ import annotations

from pathlib import Path

import yaml


class WorldConfiguration:
    """
    Infrastructure representation of a world.
    """

    def __init__(
        self,
        data: dict,
    ) -> None:

        self.id = data["id"]

        self.name = data["name"]

        self.language_code = (
            data["language_code"]
        )

        self.available_characters = (
            data.get(
                "available_characters",
                [],
            )
        )


class YamlWorldRepository:
    """
    Loads worlds from YAML files.
    """

    def __init__(
        self,
        *,
        worlds_path: Path,
    ) -> None:

        self._worlds_path = worlds_path


    def get_by_id(
        self,
        world_id: str,
    ) -> WorldConfiguration | None:

        file_path = (
    self._worlds_path.resolve()
    / f"{world_id}.yaml"
)

        if not file_path.exists():
            return None


        with file_path.open(
            "r",
            encoding="utf-8",
        ) as file:

            data = yaml.safe_load(file)


        return WorldConfiguration(data)

    ],

    "/call/{voice_call_id}/reconnect",
    response_model=VoiceReconnectResponseSchema,
)
def reconnect_voice_call(
    voice_call_id: str,
) -> VoiceReconnectResponseSchema:
    """
    Reconnect an existing voice call.

    The application lifecycle service owns
    reconnect validation, session creation,
    and lifecycle event handling.
    """


    service = (
        build_voice_call_lifecycle_service()
    )


    if service.voice_call_management_service is None:

        raise RuntimeError(
            "Voice call management service "
            "not configured."
        )


    if service.voice_call_session_service is None:

        raise RuntimeError(
            "Voice call session service "
            "not configured."
        )


    voice_call = (
        service.voice_call_management_service
        .get_call(
            voice_call_id=voice_call_id,
        )
    )


    if voice_call is None:

        raise HTTPException(
            status_code=404,
            detail=(
                "Voice call not found."
            ),
        )


    try:

        session = (
            service.reconnect_call(
                voice_call=voice_call,
            )
        )


    except ValueError as exc:

        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc


    return VoiceReconnectResponseSchema(

        voice_call_id=(
            voice_call.id
        ),

        character_id=(
            voice_call.character_id
        ),

        language_code=(
            voice_call.language_code
        ),

        session_id=(
            session.session_id
        ),

        state=(
            session.state.value
        ),
    )
