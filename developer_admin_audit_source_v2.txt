from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from typing import Any


class OperationalSurface(str, Enum):
    """
    Operational surface being monitored.

    Private classroom and announcement-group operations
    remain explicitly separated.
    """

    PRIVATE_CLASSROOM = "private_classroom"
    ANNOUNCEMENT_GROUP = "announcement_group"


class OperationalEventType(str, Enum):
    """Supported operational event categories."""

    MESSAGE_FAILURE = "message_failure"
    RATE_LIMIT = "rate_limit"
    ABUSE = "abuse"
    PROVIDER_FAILURE = "provider_failure"
    DATABASE_FAILURE = "database_failure"
    TELEGRAM_FAILURE = "telegram_failure"
    NOTIFICATION_FAILURE = "notification_failure"


class OperationalSeverity(str, Enum):
    """Operational incident severity."""

    CRITICAL = "critical"
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"
    INFO = "info"


@dataclass(frozen=True)
class OperationalEvent:
    """
    Sanitized operational event.

    The event intentionally contains operational metadata only.

    It must never contain:
        - learner message content
        - prompts
        - model responses
        - passwords
        - API keys
        - database URLs
        - Telegram bot tokens
    """

    event_type: OperationalEventType
    surface: OperationalSurface
    correlation_id: str

    severity: OperationalSeverity = (
        OperationalSeverity.MEDIUM
    )

    occurred_at: datetime = field(
        default_factory=lambda: datetime.now(
            timezone.utc
        )
    )

    user_reference: str | None = None
    world_id: str | None = None

    metadata: dict[str, Any] = field(
        default_factory=dict
    )

    def __post_init__(self) -> None:
        correlation_id = (
            self.correlation_id
            .replace("\r", "")
            .replace("\n", "")
            .strip()
        )

        if not correlation_id:
            raise ValueError(
                "correlation_id cannot be empty."
            )

        object.__setattr__(
            self,
            "correlation_id",
            correlation_id,
        )

        if self.user_reference is not None:
            object.__setattr__(
                self,
                "user_reference",
                self._clean_identifier(
                    self.user_reference
                ),
            )

        if self.world_id is not None:
            object.__setattr__(
                self,
                "world_id",
                self._clean_identifier(
                    self.world_id
                ),
            )

    @staticmethod
    def _clean_identifier(
        value: str,
    ) -> str:
        return (
            str(value)
            .replace("\r", "")
            .replace("\n", "")
            .strip()
        )

    def as_dict(self) -> dict[str, Any]:
        """
        Return an operationally safe representation.

        Learner content is not part of the event contract.
        """

        return {
            "event_type": self.event_type.value,
            "surface": self.surface.value,
            "correlation_id": self.correlation_id,
            "severity": self.severity.value,
            "occurred_at": (
                self.occurred_at.isoformat()
            ),
            "user_reference": self.user_reference,
            "world_id": self.world_id,
            "metadata": dict(self.metadata),
        }


__all__ = [
    "OperationalEvent",
    "OperationalEventType",
    "OperationalSeverity",
    "OperationalSurface",
]

from __future__ import annotations

from collections.abc import Iterable
from uuid import uuid4

from .operational_event import (
    OperationalEvent,
)
from .operational_incident import (
    IncidentStatus,
    OperationalIncident,
)


class OperationalEventService:
    """
    Application service for operational event correlation
    and incident lifecycle management.

    This service intentionally uses in-memory state for WS-21.9.

    Persistence will be introduced separately once the event
    semantics are proven.

    Responsibilities:
        - accept sanitized operational events
        - correlate repeated events
        - create incidents
        - update occurrence counts
        - maintain incident lifecycle
        - expose dashboard-safe snapshots

    Does NOT:
        - perform remediation
        - send Telegram messages
        - modify production systems
        - execute AI
        - access learner conversation content
    """

    def __init__(
        self,
        *,
        initial_incidents: Iterable[
            OperationalIncident
        ] = (),
    ) -> None:
        self._incidents: dict[
            str,
            OperationalIncident,
        ] = {
            incident.incident_id: incident
            for incident in initial_incidents
        }

        self._correlation_index: dict[
            str,
            str,
        ] = {}

        for incident in self._incidents.values():
            self._correlation_index[
                incident.correlation_id
            ] = incident.incident_id

    def record_event(
        self,
        event: OperationalEvent,
    ) -> OperationalIncident:
        """
        Record an event.

        If an open incident already exists for the
        correlation ID, the event is aggregated into it.

        Otherwise a new incident is created.
        """

        existing_id = (
            self._correlation_index.get(
                event.correlation_id
            )
        )

        if existing_id is not None:
            incident = self._incidents[
                existing_id
            ]

            if not incident.is_open:
                incident = self._create_incident(
                    event
                )
            else:
                incident.add_event(event)

            return incident

        return self._create_incident(event)

    def get_incident(
        self,
        incident_id: str,
    ) -> OperationalIncident | None:
        return self._incidents.get(
            incident_id
        )

    def get_by_correlation_id(
        self,
        correlation_id: str,
    ) -> OperationalIncident | None:
        incident_id = (
            self._correlation_index.get(
                correlation_id
            )
        )

        if incident_id is None:
            return None

        return self._incidents.get(
            incident_id
        )

    def list_incidents(
        self,
        *,
        surface=None,
        status=None,
    ) -> tuple[
        OperationalIncident,
        ...
    ]:
        """
        Return incidents with optional operational filters.
        """

        incidents = tuple(
            self._incidents.values()
        )

        if surface is not None:
            incidents = tuple(
                incident
                for incident in incidents
                if incident.surface is surface
            )

        if status is not None:
            incidents = tuple(
                incident
                for incident in incidents
                if incident.status is status
            )

        return incidents

    def transition_incident(
        self,
        *,
        incident_id: str,
        status: IncidentStatus,
    ) -> OperationalIncident:
        """
        Explicitly transition an incident.
        """

        incident = self._incidents.get(
            incident_id
        )

        if incident is None:
            raise KeyError(
                f"Unknown incident: {incident_id}"
            )

        incident.transition(status)

        return incident

    def count_open_incidents(self) -> int:
        return sum(
            incident.is_open
            for incident
            in self._incidents.values()
        )

    def dashboard_snapshot(self) -> dict:
        """
        Return a dashboard-safe operational snapshot.
        """

        incidents = tuple(
            self._incidents.values()
        )

        return {
            "total_incidents": len(
                incidents
            ),
            "open_incidents": sum(
                incident.is_open
                for incident in incidents
            ),
            "critical_incidents": sum(
                incident.severity.value
                == "critical"
                and incident.is_open
                for incident in incidents
            ),
            "private_classroom_incidents": sum(
                incident.surface.value
                == "private_classroom"
                and incident.is_open
                for incident in incidents
            ),
            "announcement_group_incidents": sum(
                incident.surface.value
                == "announcement_group"
                and incident.is_open
                for incident in incidents
            ),
            "incidents": [
                incident.as_dict()
                for incident in incidents
            ],
        }

    def _create_incident(
        self,
        event: OperationalEvent,
    ) -> OperationalIncident:
        incident = OperationalIncident(
            incident_id=(
                f"incident-{uuid4().hex}"
            ),
            correlation_id=(
                event.correlation_id
            ),
            event_type=event.event_type,
            surface=event.surface,
            severity=event.severity,
            user_reference=(
                event.user_reference
            ),
            world_id=event.world_id,
            events=[event],
        )

        self._incidents[
            incident.incident_id
        ] = incident

        self._correlation_index[
            incident.correlation_id
        ] = incident.incident_id

        return incident


__all__ = [
    "OperationalEventService",
]

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone
from enum import Enum


class AnnouncementAuditEventType(str, Enum):
    CREATED = "created"
    APPROVAL_REQUESTED = "approval_requested"
    APPROVED = "approved"
    PUBLISH_STARTED = "publish_started"
    PUBLISHED = "published"
    PUBLISH_FAILED = "publish_failed"


@dataclass(frozen=True)
class AnnouncementAuditEvent:
    publication_id: str
    event_type: AnnouncementAuditEventType
    occurred_at: datetime
    details: str | None = None


class InMemoryAnnouncementAuditSink:
    def __init__(self) -> None:
        self._events: list[AnnouncementAuditEvent] = []

    def record(
        self,
        publication_id: str,
        event_type: AnnouncementAuditEventType,
        details: str | None = None,
    ) -> AnnouncementAuditEvent:
        event = AnnouncementAuditEvent(
            publication_id=publication_id,
            event_type=event_type,
            occurred_at=datetime.now(timezone.utc),
            details=details,
        )
        self._events.append(event)
        return event

    def list_for(self, publication_id: str) -> tuple[AnnouncementAuditEvent, ...]:
        return tuple(e for e in self._events if e.publication_id == publication_id)

"""WS-22.3.4 execution audit adapters.

WS-22.3.4.1:
    PublicationExecutionAuditAdapter
    Bridges the execution audit contract to the existing in-memory
    announcement audit vocabulary.

WS-22.3.4.2:
    DurablePublicationExecutionAudit
    Bridges the same execution contract to the durable audit repository.

Both implementations intentionally share the same execution-facing
contract. They differ only in persistence destination.
"""

from __future__ import annotations

from uuid import UUID

from app.application.announcements.audit import (
    AnnouncementAuditEventType,
    InMemoryAnnouncementAuditSink,
)


class PublicationExecutionAuditAdapter:
    """WS-22.3.4.1 adapter for the existing announcement audit sink."""

    def __init__(
        self,
        *,
        sink: InMemoryAnnouncementAuditSink,
    ) -> None:
        if sink is None:
            raise ValueError("sink cannot be None.")

        self._sink = sink

    async def record_success(
        self,
        *,
        publication_id: UUID,
        actor: str,
        correlation_id: str,
        provider_message_id: str | None,
    ) -> None:
        self._sink.record(
            publication_id=str(publication_id),
            event_type=AnnouncementAuditEventType.PUBLISHED,
            details=(
                f"actor={actor};"
                f"correlation_id={correlation_id};"
                f"provider_message_id={provider_message_id}"
            ),
        )

    async def record_failure(
        self,
        *,
        publication_id: UUID,
        actor: str,
        correlation_id: str,
        error_code: str,
    ) -> None:
        self._sink.record(
            publication_id=str(publication_id),
            event_type=AnnouncementAuditEventType.PUBLISH_FAILED,
            details=(
                f"actor={actor};"
                f"correlation_id={correlation_id};"
                f"error_code={error_code}"
            ),
        )


class DurablePublicationExecutionAudit:
    """WS-22.3.4.2 adapter for the durable audit repository."""

    def __init__(self, *, repository) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")

        self._repository = repository

    async def record_success(
        self,
        *,
        publication_id: UUID,
        actor: str,
        correlation_id: str,
        provider_message_id: str | None,
    ) -> None:
        await self._repository.record_success(
            publication_id=publication_id,
            actor=actor,
            correlation_id=correlation_id,
            provider_message_id=provider_message_id,
        )

    async def record_failure(
        self,
        *,
        publication_id: UUID,
        actor: str,
        correlation_id: str,
        error_code: str,
    ) -> None:
        await self._repository.record_failure(
            publication_id=publication_id,
            actor=actor,
            correlation_id=correlation_id,
            error_code=error_code,
        )


__all__ = [

        self,
        *,

        return self._repository.save(event)

from __future__ import annotations

from dataclasses import dataclass
from uuid import UUID

from app.domain.authorization.permissions import Role
from app.infrastructure.auth.postgresql_auth_repository import (
    AuthUserRecord,
    PostgreSQLAuthenticationRepository,
)


class DeveloperAdminError(Exception):
    """Base error for Developer Administration operations."""


class DeveloperAdminUserNotFoundError(DeveloperAdminError):
    """Requested target user does not exist."""


class DeveloperAdminUserInactiveError(DeveloperAdminError):
    """Requested target user cannot receive Developer Admin privileges."""


class DeveloperAdminLastAdminError(DeveloperAdminError):
    """The final Developer Administrator cannot be removed."""


@dataclass(frozen=True, slots=True)
class DeveloperAdminRecord:
    user_id: UUID
    email: str
    username: str
    status: str
    email_verified: bool


class DeveloperAdminService:
    """Application boundary for controlled Developer Administrator management."""

    ROLE = Role.DEVELOPER_ADMIN.value

    def __init__(
        self,
        repository: PostgreSQLAuthenticationRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository is required.")

        self._repository = repository

    async def list_developer_administrators(
        self,
    ) -> tuple[DeveloperAdminRecord, ...]:
        users = await self._repository.get_users_with_role(
            role=self.ROLE,
        )

        return tuple(
            self._to_record(user)
            for user in users
        )

    async def grant_developer_admin(
        self,
        *,
        user_id: UUID,
    ) -> DeveloperAdminRecord:
        user = await self._repository.get_user_for_identity(user_id)

        if user is None:
            raise DeveloperAdminUserNotFoundError(
                "Target user does not exist."
            )

        if user.status != "active" or not user.email_verified:
            raise DeveloperAdminUserInactiveError(
                "Developer Administrator privileges require an active, "
                "verified user."
            )

        await self._repository.add_user_role(
            user_id=user_id,
            role=self.ROLE,
        )

        return self._to_record(user)

    async def revoke_developer_admin(
        self,
        *,
        user_id: UUID,
    ) -> None:
        user = await self._repository.get_user_for_identity(user_id)

        if user is None:
            raise DeveloperAdminUserNotFoundError(
                "Target user does not exist."
            )

        current_count = await self._repository.count_users_with_role(
            role=self.ROLE,
        )

        roles = await self._repository.get_user_roles(user_id)

        if self.ROLE not in roles:
            return

        if current_count <= 1:
            raise DeveloperAdminLastAdminError(
                "The last Developer Administrator cannot be removed."
            )

        await self._repository.remove_user_role(
            user_id=user_id,
            role=self.ROLE,
        )

    @staticmethod
    def _to_record(
        user: AuthUserRecord,
    ) -> DeveloperAdminRecord:
        return DeveloperAdminRecord(
            user_id=user.id,
            email=user.email,
            username=user.username,
            status=user.status,
            email_verified=user.email_verified,
        )

class FakeDeveloperAdminRepository:
    def __init__(
        self,
        *,
        users: tuple[AuthUserRecord, ...] = (),
        roles: dict | None = None,
    ) -> None:
        self.users = {
            user.id: user
            for user in users
        }

        self.roles = {
            user_id: set(user_roles)
            for user_id, user_roles in (roles or {}).items()
        }

    async def get_users_with_role(
        self,
        *,
        role: str,
    ) -> tuple[AuthUserRecord, ...]:
        return tuple(
            user
            for user in self.users.values()
            if role in self.roles.get(user.id, set())
        )

    async def get_user_for_identity(
        self,
        user_id,
    ) -> AuthUserRecord | None:
        return self.users.get(user_id)

    async def add_user_role(
        self,
        *,
        user_id,
        role: str,
    ) -> bool:
        user_roles = self.roles.setdefault(
            user_id,
            set(),
        )

        if role in user_roles:
            return False

        user_roles.add(role)
        return True

    async def remove_user_role(
        self,
        *,
        user_id,
        role: str,
    ) -> bool:
        user_roles = self.roles.setdefault(
            user_id,
            set(),
        )

        if role not in user_roles:
            return False

        user_roles.remove(role)
        return True

    async def count_users_with_role(
        self,
        *,
        role: str,
    ) -> int:
        return sum(
            role in user_roles
            for user_roles in self.roles.values()
        )

    async def get_user_roles(
        self,
        user_id,
    ) -> frozenset[str]:
        return frozenset(
            self.roles.get(
                user_id,
                set(),
            )
        )

class FakeDeveloperAdminApiService:
    def __init__(self) -> None:
        self.administrators = [
            {
                "user_id": uuid4(),
                "email": "admin@example.com",
                "username": "admin",
                "status": "active",
                "email_verified": True,
            }
        ]
        self.grant_calls: list[UUID] = []
        self.revoke_calls: list[UUID] = []
        self.grant_error: Exception | None = None
        self.revoke_error: Exception | None = None

    async def list_developer_administrators(self):
        return [
            _record_from_dict(item)
            for item in self.administrators
        ]

    async def grant_developer_admin(self, *, user_id: UUID):
        self.grant_calls.append(user_id)

        if self.grant_error is not None:
            raise self.grant_error

        for item in self.administrators:
            if item["user_id"] == user_id:
                return _record_from_dict(item)

        record = {
            "user_id": user_id,
            "email": "target@example.com",
            "username": "target",
            "status": "active",
            "email_verified": True,
        }
        self.administrators.append(record)
        return _record_from_dict(record)

    async def revoke_developer_admin(self, *, user_id: UUID):
        self.revoke_calls.append(user_id)

        if self.revoke_error is not None:
            raise self.revoke_error


def _record_from_dict(item: dict):
