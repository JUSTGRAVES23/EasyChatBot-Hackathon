# WS-21.2 Secret Safety Audit

# 1. Check whether .env is tracked by Git
git ls-files --error-unmatch ".env"

# 2. Check .gitignore for .env protection
if (Test-Path ".gitignore") {
    Select-String -Path ".gitignore" -Pattern '^\s*\.env($|\.)'
} else {
    Write-Host ".gitignore NOT FOUND"
}

# 3. Search backend source/config files for hard-coded secrets
Get-ChildItem ".\backend" -Recurse -File -Include *.py,*.yaml,*.yml,*.json,*.toml,*.ini |
Select-String -Pattern 'TELEGRAM_BOT_TOKEN|DATABASE_URL|SECRET_KEY|API_KEY|PASSWORD' |
Select-Object Path,LineNumber,Line

# 4. Verify required environment variables without exposing values
python -c "from dotenv import load_dotenv; load_dotenv('.env'); import os; names=['DATABASE_URL','TELEGRAM_BOT_TOKEN','SECRET_KEY','API_KEY','PASSWORD']; [print(f'{n}: '+('SET' if os.getenv(n) else 'NOT SET')) for n in names]"
