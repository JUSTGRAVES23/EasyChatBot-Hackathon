from __future__ import annotations

from datetime import datetime
from typing import Any
from uuid import UUID

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker


class PostgreSQLAcademyUserQuery:
    """PostgreSQL adapter for authoritative Academy registration data."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        self._session_factory = session_factory

    async def list_one_week_candidates(
        self,
        *,
        window_start: datetime,
        window_end: datetime,
    ):
        from backend.app.application.academy.academy_notification_delivery import (
            AcademyEligibleUser,
        )

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT id, email, username, created_at
                    FROM users
                    WHERE created_at >= :window_start
                      AND created_at < :window_end
                      AND status = 'active'
                    ORDER BY created_at ASC
                    """
                ),
                {
                    "window_start": window_start,
                    "window_end": window_end,
                },
            )

            return [
                AcademyEligibleUser(
                    user_id=row.id,
                    email=row.email,
                    username=row.username,
                    created_at=row.created_at,
                )
                for row in result
            ]


class PostgreSQLAcademyDeliveryRepository:
    """Durable Academy delivery/idempotency adapter."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        self._session_factory = session_factory

    async def claim(
        self,
        *,
        user_id: UUID,
        notification_type: str,
        now: datetime,
    ):
        from backend.app.application.academy.academy_notification_delivery import (
            AcademyDeliveryRecord,
            AcademyDeliveryStatus,
        )

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    INSERT INTO academy_notification_deliveries (
                        id,
                        user_id,
                        notification_type,
                        status,
                        attempt_count,
                        created_at,
                        updated_at
                    )
                    VALUES (
                        gen_random_uuid(),
                        :user_id,
                        :notification_type,
                        'SENDING',
                        1,
                        :now,
                        :now
                    )
                    ON CONFLICT (user_id, notification_type)
                    DO UPDATE SET
                        status = 'SENDING',
                        attempt_count =
                            academy_notification_deliveries.attempt_count + 1,
                        updated_at = EXCLUDED.updated_at
                    WHERE academy_notification_deliveries.status = 'FAILED'
                    RETURNING
                        user_id,
                        notification_type,
                        status,
                        attempt_count,
                        provider_message_id,
                        last_error
                    """
                ),
                {
                    "user_id": user_id,
                    "notification_type": notification_type,
                    "now": now,
                },
            )

            row = result.first()

            if row is None:
                await session.rollback()
                return None

            record = AcademyDeliveryRecord(
                user_id=row.user_id,
                notification_type=row.notification_type,
                status=AcademyDeliveryStatus(row.status),
                attempt_count=row.attempt_count,
                provider_message_id=row.provider_message_id,
                last_error=row.last_error,
            )

            await session.commit()

            return record

    async def mark_succeeded(
        self,
        *,
        user_id: UUID,
        notification_type: str,
        provider_message_id: str | None,
        now: datetime,
    ) -> None:
        async with self._session_factory() as session:
            await session.execute(
                text(
                    """
                    UPDATE academy_notification_deliveries
                    SET status = 'SUCCEEDED',
                        provider_message_id = :provider_message_id,
                        last_error = NULL,
                        sent_at = :now,
                        updated_at = :now
                    WHERE user_id = :user_id
                      AND notification_type = :notification_type
                    """
                ),
                {
                    "user_id": user_id,
                    "notification_type": notification_type,
                    "provider_message_id": provider_message_id,
                    "now": now,
                },
            )

            await session.commit()

    async def mark_failed(
        self,
        *,
        user_id: UUID,
        notification_type: str,
        error: str,
        now: datetime,
    ) -> None:
        async with self._session_factory() as session:
            await session.execute(
                text(
                    """
                    UPDATE academy_notification_deliveries
                    SET status = 'FAILED',
                        last_error = :error,
                        updated_at = :now
                    WHERE user_id = :user_id
                      AND notification_type = :notification_type
                    """
                ),
                {
                    "user_id": user_id,
                    "notification_type": notification_type,
                    "error": error[:2000],
                    "now": now,
                },
            )

            await session.commit()

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from enum import StrEnum
from typing import Protocol
from uuid import UUID


class AcademyDeliveryStatus(StrEnum):
    PENDING = "PENDING"
    SENDING = "SENDING"
    SUCCEEDED = "SUCCEEDED"
    FAILED = "FAILED"


@dataclass(frozen=True, slots=True)
class AcademyEligibleUser:
    user_id: UUID
    email: str
    username: str
    created_at: datetime


@dataclass(frozen=True, slots=True)
class AcademyDeliveryRecord:
    user_id: UUID
    notification_type: str
    status: AcademyDeliveryStatus
    attempt_count: int
    provider_message_id: str | None = None
    last_error: str | None = None


class AcademyUserQuery(Protocol):
    async def list_one_week_candidates(
        self,
        *,
        window_start: datetime,
        window_end: datetime,
    ) -> list[AcademyEligibleUser]:
        ...


class AcademyDeliveryRepository(Protocol):
    async def claim(
        self,
        *,
        user_id: UUID,
        notification_type: str,
        now: datetime,
    ) -> AcademyDeliveryRecord | None:
        ...

        self,
        *,
        user_id: UUID,
        notification_type: str,
        provider_message_id: str | None,
        now: datetime,

        ...

    async def mark_failed(
        self,
        *,
        user_id: UUID,

    ) -> None:

from __future__ import annotations

from datetime import datetime, timedelta, timezone

from backend.app.application.academy_notification_service import (
    OneWeekNotificationOrchestrator,
)
from backend.app.application.academy.academy_notification_delivery import (
    AcademyDeliveryRepository,
    AcademyUserQuery,
)


class AcademyOneWeekWorkflow:
    """Execute the Academy 1-week workflow without owning scheduler state."""

    NOTIFICATION_TYPE = "1_WEEK_MILESTONE"

    def __init__(
        self,
        *,
        users: AcademyUserQuery,
        deliveries: AcademyDeliveryRepository,
        notifications: OneWeekNotificationOrchestrator,
        clock=None,
    ) -> None:
        self._users = users
        self._deliveries = deliveries
        self._notifications = notifications
        self._clock = clock or (lambda: datetime.now(timezone.utc))

    async def run(self) -> dict[str, int]:
        now = self._clock()
        if now.tzinfo is None:
            raise ValueError("Workflow clock must return a timezone-aware datetime.")

        window_start = now - timedelta(days=8)
        window_end = now - timedelta(days=7)

        candidates = await self._users.list_one_week_candidates(
            window_start=window_start,
            window_end=window_end,
        )

        attempted = 0
        succeeded = 0
        skipped = 0
        failed = 0

        for user in candidates:
            # The query is deliberately bounded by the same eligibility window,
            # while the domain rule remains the final guard.
            if not self._notifications.is_one_week_eligible(
                registered_at=user.created_at,
                now=now,
            ):
                skipped += 1
                continue

            claim = await self._deliveries.claim(
                user_id=user.user_id,
                notification_type=self.NOTIFICATION_TYPE,
                now=now,
            )

            if claim is None:
                skipped += 1
                continue

            attempted += 1

            try:
                result = await self._notifications.send_one_week(
                    email=user.email,
                    username=user.username,
                    telegram_chat_id=None,
                )
                email_result = result.get("email") or {}
                provider_message_id = email_result.get("message_id")

                await self._deliveries.mark_succeeded(
                    user_id=user.user_id,
                    notification_type=self.NOTIFICATION_TYPE,
                    provider_message_id=(
                        str(provider_message_id)
                        if provider_message_id is not None
                        else None
                    ),
                    now=now,
                )
                succeeded += 1
            except Exception as exc:
                await self._deliveries.mark_failed(
                    user_id=user.user_id,
                    notification_type=self.NOTIFICATION_TYPE,
                    error=str(exc)[:2000],
                    now=now,
                )
                failed += 1

        return {
            "candidates": len(candidates),
            "attempted": attempted,
            "succeeded": succeeded,
            "failed": failed,
            "skipped": skipped,
        }
