@dataclass(frozen=True)
class PublicationPreview:
    publication_type: str
    template_id: str
    template_version: int
    content: str


class AnnouncementPublicationService:
    def __init__(
        self,
        *,
        templates: AnnouncementTemplateRegistry,
        publications: PublicationRepository,
    ) -> None:
        if templates is None or publications is None:
            raise ValueError("templates and publications are required.")
        self._templates = templates
        self._publications = publications

    def preview(
        self,
        *,
        template_id: str,
        values: Mapping[str, str],
        version: int | None = None,
    ) -> PublicationPreview:
        template = self._templates.get_version(template_id, version)
        content = self._templates.preview(
            template_id=template_id,
            values=dict(values),
            version=template.version,
        )
        return PublicationPreview(
            publication_type=template.publication_type.value,
            template_id=template.template_id,
            template_version=template.version,
            content=content,
        )

    def create(
        self,
        *,
        publication_id: str,
        template_id: str,
        title: str,
        values: Mapping[str, str],
        version: int | None = None,
    ) -> Publication:
        template = self._templates.get_version(template_id, version)
        content = self._templates.preview(
            template_id=template_id,
            values=dict(values),
            version=template.version,
        )
        publication = Publication(
            publication_id=publication_id,
            template_id=template.template_id,
            template_version=template.version,
            publication_type=template.publication_type,
            title=title,
            content=content,
            status=PublicationStatus.DRAFT,
        )
        self._publications.save(publication)
        return publication

    def _get(self, publication_id: str) -> Publication:
        publication = self._publications.get(publication_id)
        if publication is None:
            raise LookupError(f"Unknown publication '{publication_id}'.")
        return publication

    def request_approval(self, publication_id: str) -> Publication:
        publication = self._get(publication_id)
        publication.request_approval()
        self._publications.save(publication)
        return publication

    def approve(self, publication_id: str) -> Publication:
        publication = self._get(publication_id)
        publication.approve()
        self._publications.save(publication)
        return publication

    def begin_publish(self, publication_id: str) -> Publication:
        publication = self._get(publication_id)
        publication.begin_publish()
        self._publications.save(publication)
        return publication

    def mark_publish_failure(self, publication_id: str, reason: str) -> Publication:
        publication = self._get(publication_id)
        publication.mark_failure(reason)
        self._publications.save(publication)
        return publication

    def mark_publish_success(self, publication_id: str) -> Publication:
        publication = self._get(publication_id)
        publication.publish_success()
        self._publications.save(publication)
        return publication

    publication: Publication
    actor: str
    approved_at: datetime


class AsyncAnnouncementPublicationApprovalService:
    """Async PostgreSQL-backed approval application boundary.

    The domain model remains the authority for lifecycle transitions.
    This service never invokes WS-22 execution or Telegram.
    """

    def __init__(
        self,
        *,
        publications: AsyncPublicationStore,
        audit: PublicationApprovalAudit,
    ) -> None:
        if publications is None or audit is None:
            raise ValueError("publications and audit are required.")
        self._publications = publications
        self._audit = audit

    async def approve(
        self,
        *,
        publication_id: str,
        actor: str,
    ) -> AsyncPublicationApprovalResult:
        if not actor.strip():
            raise ValueError("actor cannot be empty.")

        publication = await self._publications.get_async(publication_id)
        if publication is None:
            raise LookupError(f"Unknown publication '{publication_id}'.")

        if publication.status is not PublicationStatus.PENDING_APPROVAL:
            # Use the domain transition so the canonical lifecycle error remains
            # authoritative and consistent with the existing domain model.
            publication.approve()

        else:
            publication.approve()

        approved_at = publication.approved_at or datetime.now(timezone.utc)

        await self._publications.save_async(publication)
        await self._audit.record_approval(
            publication=publication,
            actor=actor.strip(),
            occurred_at=approved_at,
        )

        return AsyncPublicationApprovalResult(
            publication=publication,
            actor=actor.strip(),
            approved_at=approved_at,
        )

    durable_execution: DurablePublicationExecutionService

    def __post_init__(self) -> None:
        if self.durable_execution is None:
            raise ValueError("durable_execution cannot be None.")

    async def execute(
        self,
        *,
        publication: Publication,
        actor: str,
        idempotency_key: str,
        correlation_id: str,
    ) -> PublicationExecutionResult:
        """Delegate an approved publication to the existing WS-22 service.

        WS-22 remains responsible for:
        - durable execution;
        - idempotency;
        - execution persistence;
        - execution audit;
        - retry semantics;
        - controlled provider invocation.
        """

        if publication is None:
            raise ValueError("publication cannot be None.")

        if not actor.strip():
            raise ValueError("actor cannot be empty.")

        if not idempotency_key.strip():
            raise ValueError("idempotency_key cannot be empty.")

        if not correlation_id.strip():
            raise ValueError("correlation_id cannot be empty.")

        try:
            publication_uuid = UUID(publication.publication_id)
        except (ValueError, AttributeError) as exc:
            raise ValueError(
                "Approved publication ID must be a UUID for WS-22 durable execution."
            ) from exc

        request = PublicationExecutionRequest(
            publication_id=publication_uuid,
            idempotency_key=idempotency_key.strip(),
            correlation_id=correlation_id.strip(),
            actor=actor.strip(),

            publication=_PublicationExecutionAdapter(publication),
            request=request,

"""WS-22.3.2 durable execution orchestration."""

from __future__ import annotations

from app.application.announcements.execution.contracts import (
    AnnouncementPublisher,
    PublicationExecutionRequest,
    PublicationExecutionResult,
    PublicationExecutionStatus,
)
from app.application.announcements.execution.durable_contracts import (
    DurableExecutionStatus,
    PublicationExecutionRepository,
)


class DurablePublicationExecutionService:
    """Retry-safe execution service backed by durable idempotency state.

    A completed idempotency key is never sent again.

    A STARTED record is treated as an interrupted attempt and is retryable.

    A FAILED record is also retryable. Before a retry is executed, the
    repository is responsible for incrementing the durable attempt counter
    and transitioning the record back to STARTED.

    The repository implementation must make creation atomic against a
    unique idempotency key.
    """

    def __init__(
        self,
        *,
        publisher: AnnouncementPublisher,
        execution_repository: PublicationExecutionRepository,
        audit,
    ) -> None:
        if publisher is None:
            raise ValueError("publisher cannot be None.")

        if execution_repository is None:
            raise ValueError("execution_repository cannot be None.")

        if audit is None:
            raise ValueError("audit cannot be None.")

        self._publisher = publisher
        self._repository = execution_repository
        self._audit = audit

    async def execute(
        self,
        *,
        publication,
        request: PublicationExecutionRequest,
    ) -> PublicationExecutionResult:
        existing = await self._repository.get_by_idempotency_key(
            request.idempotency_key
        )

        # A successful execution is terminal for the idempotency key.
        # Never call the external provider again.
        if (
            existing is not None
            and existing.status is DurableExecutionStatus.SUCCEEDED
        ):
            return PublicationExecutionResult(
                publication_id=existing.publication_id,
                status=PublicationExecutionStatus.IDEMPOTENT_REPLAY,
                idempotency_key=existing.idempotency_key,
                correlation_id=request.correlation_id,
                provider_message_id=existing.provider_message_id,
            )

        self._assert_approved(publication)

        # The repository owns the atomic unique-key operation. If another
        # worker already created the same key, re-read and honour its state.
        if existing is None:
            try:
                record = await self._repository.create_started(
                    publication_id=publication.id,
                    idempotency_key=request.idempotency_key,
                    correlation_id=request.correlation_id,
                )
            except Exception:
                record = await self._repository.get_by_idempotency_key(
                    request.idempotency_key
                )

                if record is None:
                    raise

        elif existing.status is DurableExecutionStatus.FAILED:
            # A FAILED execution is retryable. Reuse the same durable
            # idempotency record and increment its attempt number rather
            # than creating a second row.
            record = await self._repository.start_retry(
                idempotency_key=request.idempotency_key,
            )

        else:
            # STARTED represents an interrupted/incomplete attempt.
            # Preserve the existing durable attempt number and retry it.
            record = existing

        publication.begin_publish()

        try:
            provider_message_id = await self._publisher.publish(
                publication_id=publication.id,
                title=publication.title,
                content=publication.content,
                correlation_id=request.correlation_id,
                idempotency_key=request.idempotency_key,
            )
        except Exception as exc:
            error_code = self._error_code(exc)

            publication.mark_failure(error_code)

            await self._repository.mark_failed(
                idempotency_key=request.idempotency_key,
                error_code=error_code,
            )

            await self._audit.record_failure(
                publication_id=publication.id,
                actor=request.actor,
                correlation_id=request.correlation_id,
                error_code=error_code,
            )

            return PublicationExecutionResult(
                publication_id=publication.id,
                status=PublicationExecutionStatus.FAILED,
                idempotency_key=request.idempotency_key,
                correlation_id=request.correlation_id,
                error_code=error_code,
            )

        publication.publish_success(provider_message_id)

        await self._repository.mark_succeeded(
            idempotency_key=request.idempotency_key,
            provider_message_id=provider_message_id,
        )

        await self._audit.record_success(
            publication_id=publication.id,
            actor=request.actor,
            correlation_id=request.correlation_id,
            provider_message_id=provider_message_id,
        )

        return PublicationExecutionResult(
            publication_id=publication.id,
            status=PublicationExecutionStatus.SUCCEEDED,
            idempotency_key=request.idempotency_key,
            correlation_id=request.correlation_id,
            provider_message_id=provider_message_id,
        )

    @staticmethod
    def _assert_approved(publication) -> None:
        value = getattr(publication.status, "value", publication.status)

        if value != "approved":
            raise ValueError(
                "Only approved publications may be executed."
            )

    @staticmethod
    def _error_code(exc: Exception) -> str:
        explicit = getattr(exc, "error_code", None)

        return (
            explicit.strip()[:100]
            if isinstance(explicit, str) and explicit.strip()
            else type(exc).__name__.upper()[:100]
        )

This module is the composition boundary for the WS-22 durable execution
vertical slice.

It deliberately does not create a new persistence architecture. The
canonical PostgreSQL engine and async session factory are supplied by the

    PostgreSQL

                  |

Provider selection remains outside this module.

def create_durable_publication_execution_composition(
    *,
    publisher,
) -> DurablePublicationExecutionComposition:

    The publisher is deliberately injected because provider selection belongs
    outside the durable execution composition.

    This function creates exactly one PostgreSQL engine and exactly one async

        session_factory

        repository=audit_repository,
    )

    service = DurablePublicationExecutionService(
        publisher=publisher,
        execution_repository=execution_repository,
        audit=audit,
    )

    return DurablePublicationExecutionComposition(
        engine=engine,
        session_factory=session_factory,
        execution_repository=execution_repository,
        audit_repository=audit_repository,
        audit=audit,
        service=service,

class ExecutablePublication(Protocol):
    """Minimum aggregate surface required by the execution service."""

    id: UUID
    title: str
    content: str
    status: object

    def begin_publish(self) -> None: ...

    def publish_success(self, provider_message_id: str | None = None) -> None: ...

    def mark_failure(self, reason: str) -> None: ...


class PublicationExecutionAudit(Protocol):
    """Minimal audit boundary used by execution.

    The concrete publication-event repository remains an existing
    application concern; this protocol prevents execution from coupling to
    a particular persistence implementation.
    """

    async def record_success(
        self,
        *,
        publication_id: UUID,
        actor: str,
        correlation_id: str,
        provider_message_id: str | None,
    ) -> None: ...

    async def record_failure(
        self,
        *,
        publication_id: UUID,
        actor: str,
        correlation_id: str,
        error_code: str,
    ) -> None: ...


class PublicationExecutionService:
    """Execute approved publications through a provider-independent port.

    The service owns orchestration and idempotency. It does not know which
    channel/provider performs the external publication.
    """

    def __init__(
        self,
        *,
        publisher: AnnouncementPublisher,
        audit: PublicationExecutionAudit,
    ) -> None:
        if publisher is None:
            raise ValueError("publisher cannot be None.")
        if audit is None:
            raise ValueError("audit cannot be None.")
        self._publisher = publisher
        self._audit = audit
        self._completed: dict[str, PublicationExecutionResult] = {}

    async def execute(
        self,
        *,
        publication: ExecutablePublication,
        request: PublicationExecutionRequest,
    ) -> PublicationExecutionResult:
        if publication is None:
            raise ValueError("publication cannot be None.")

        existing = self._completed.get(request.idempotency_key)
        if existing is not None:
            return PublicationExecutionResult(
                publication_id=existing.publication_id,
                status=PublicationExecutionStatus.IDEMPOTENT_REPLAY,
                idempotency_key=existing.idempotency_key,
                correlation_id=request.correlation_id,
                provider_message_id=existing.provider_message_id,
                error_code=existing.error_code,
                error_message=existing.error_message,
            )

        self._assert_approved(publication)
        publication.begin_publish()

        try:
            provider_message_id = await self._publisher.publish(
                publication_id=publication.id,
                title=publication.title,
                content=publication.content,
                correlation_id=request.correlation_id,
                idempotency_key=request.idempotency_key,
            )
        except Exception as exc:
            error_code = self._error_code(exc)
            publication.mark_failure(error_code)
            await self._audit.record_failure(
                publication_id=publication.id,
                actor=request.actor,
                correlation_id=request.correlation_id,
                error_code=error_code,
            )
            result = PublicationExecutionResult(
                publication_id=publication.id,
                status=PublicationExecutionStatus.FAILED,
                idempotency_key=request.idempotency_key,
                correlation_id=request.correlation_id,
                error_code=error_code,
            )
            return result

        publication.publish_success(provider_message_id)
        await self._audit.record_success(
            publication_id=publication.id,
            actor=request.actor,
            correlation_id=request.correlation_id,
            provider_message_id=provider_message_id,
        )

        result = PublicationExecutionResult(
            publication_id=publication.id,
            status=PublicationExecutionStatus.SUCCEEDED,
            idempotency_key=request.idempotency_key,
            correlation_id=request.correlation_id,
            provider_message_id=provider_message_id,
        )
        self._completed[request.idempotency_key] = result
        return result

    @staticmethod
    def _assert_approved(publication: ExecutablePublication) -> None:
        status_value = getattr(publication.status, "value", publication.status)
        if status_value != "approved":
            raise ValueError(
                "Only approved publications may be executed."
            )

    @staticmethod
    def _error_code(exc: Exception) -> str:
        """Return a stable, non-sensitive operational error code."""
        explicit = getattr(exc, "error_code", None)
        if isinstance(explicit, str) and explicit.strip():
            return explicit.strip()[:100]
        return type(exc).__name__.upper()[:100]

from __future__ import annotations

from app.application.announcements.repositories import PublicationRepository
from app.domain.announcements.publication import Publication
from app.infrastructure.announcements.sqlalchemy_publication_repository import (
    SqlAlchemyPublicationRepository,

class AsyncPublicationRepositoryAdapter(PublicationRepository):
    """Synchronous application contract adapter for the async PostgreSQL adapter.

    This adapter is intentionally not used from an async event loop. It exists
    for compatibility with the current synchronous AnnouncementPublicationService
    contract. Production composition should prefer an async application boundary
    when the application service is migrated.

    Calling these methods from an active async event loop would require blocking
    the loop, so this adapter fails closed rather than performing unsafe
    synchronous database I/O.
    """

    def __init__(self, repository: SqlAlchemyPublicationRepository) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")
        self._repository = repository

    def _reject_sync_boundary(self) -> None:
        raise RuntimeError(
            "AsyncPublicationRepositoryAdapter cannot be used through the "
            "synchronous PublicationRepository contract. Use the async "
            "approval composition."
        )

    def save(self, publication: Publication) -> None:
        self._reject_sync_boundary()

    def get(self, publication_id: str) -> Publication | None:

            template_version=model.template_version,
            publication_type=PublicationType(model.publication_type),

            created_at=model.created_at,
            approved_at=model.approved_at,
            published_at=model.published_at,
            failure_reason=model.failure_reason,
        )

    def _from_domain(self, publication: Publication) -> PublicationModel:
        return PublicationModel(
            publication_id=publication.publication_id,
            template_id=publication.template_id,
            template_version=publication.template_version,
            publication_type=publication.publication_type.value,

            published_at=publication.published_at,

        async with self._session_factory() as session:
            model = await session.get(
                PublicationModel,
                publication.publication_id,
            )
            if model is None:
                session.add(self._from_domain(publication))
            else:

                model.publication_type = publication.publication_type.value

                model.created_at = publication.created_at or model.created_at
                model.approved_at = publication.approved_at
                model.published_at = publication.published_at
                model.failure_reason = publication.failure_reason
            await session.commit()

        async with self._session_factory() as session:
            model = await session.get(PublicationModel, publication_id)
            return self._to_domain(model) if model else None

class SqlAlchemyPublicationExecutionRepository(
    PublicationExecutionRepository
):
    """Durable repository using the existing injected async session factory."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    async def get_by_idempotency_key(
        self,
        idempotency_key: str,
    ) -> PublicationExecutionRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )
            model = result.scalar_one_or_none()

            return self._to_domain(model) if model else None

    async def create_started(
        self,
        *,
        publication_id,
        idempotency_key,
        correlation_id,
    ) -> PublicationExecutionRecord:
        """Create the first durable execution attempt."""

        now = datetime.now(timezone.utc)

        model = PublicationExecutionRecordModel(
            publication_id=str(publication_id),
            idempotency_key=idempotency_key,
            correlation_id=correlation_id,
            attempt=1,
            status=DurableExecutionStatus.STARTED.value,
            created_at=now,
            updated_at=now,
        )

        async with self._session_factory() as session:
            session.add(model)
            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def mark_succeeded(
        self,
        *,
        idempotency_key,
        provider_message_id,
    ) -> PublicationExecutionRecord:
        """Persist successful completion without changing the attempt count."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.SUCCEEDED.value,
            provider_message_id=provider_message_id,
            error_code=None,
        )

    async def mark_failed(
        self,
        *,
        idempotency_key,
        error_code,
    ) -> PublicationExecutionRecord:
        """Persist failure while retaining the current attempt number."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.FAILED.value,
            error_code=error_code,
        )

    async def start_retry(
        self,
        *,
        idempotency_key: str,
    ) -> PublicationExecutionRecord:
        """Increment the durable attempt counter for a retry.

        A retry is valid only for an existing FAILED execution. The same
        idempotency record is retained; no second database row is created.
        """

        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            if model.status != DurableExecutionStatus.FAILED.value:
                raise ValueError(
                    "Only failed executions can be retried."
                )

            model.attempt += 1
            model.status = DurableExecutionStatus.STARTED.value
            model.error_code = None
            model.provider_message_id = None
            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def _update(
        self,
        idempotency_key,
        **values,
    ) -> PublicationExecutionRecord:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            for key, value in values.items():
                setattr(model, key, value)

            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    @staticmethod
    def _to_domain(
        model: PublicationExecutionRecordModel,
    ) -> PublicationExecutionRecord:
        return PublicationExecutionRecord(
            publication_id=UUID(model.publication_id),
            idempotency_key=model.idempotency_key,
            correlation_id=model.correlation_id,
            attempt=model.attempt,
            status=DurableExecutionStatus(model.status),
            provider_message_id=model.provider_message_id,
            error_code=model.error_code,
            created_at=model.created_at,
            updated_at=model.updated_at,
        )

Architecture:

from uuid import UUID

class TelegramPublicationPublisherAdapter(AnnouncementPublisher):
    """Bridge the canonical WS-22 publisher contract to Telegram."""

    def __init__(
        self,
        *,
        publisher: ControlledAnnouncementPublisher,
        channel_id: str | None = None,
        announcement_type: str = "beta_update",
    ) -> None:
        if publisher is None:
            raise ValueError("publisher cannot be None.")

        resolved_channel_id = channel_id or getenv(
            "BETA1_ANNOUNCEMENT_CHANNEL_ID"
        )

        if not resolved_channel_id or not resolved_channel_id.strip():
            raise ValueError(
                "BETA1_ANNOUNCEMENT_CHANNEL_ID is not configured."
            )

        if not announcement_type.strip():
            raise ValueError(
                "announcement_type cannot be empty."
            )

        self._publisher = publisher
        self._channel_id = resolved_channel_id.strip()
        self._announcement_type = announcement_type.strip()

    async def publish(
        self,
        *,
        publication_id: UUID,
        title: str,
        content: str,
        correlation_id: str,
        idempotency_key: str,
    ) -> str | None:
        """
        Publish through the existing controlled Telegram boundary.

        WS-22 remains responsible for:
        - idempotency;
        - execution persistence;
        - execution audit;
        - retry semantics.

        This adapter only translates the publication request.
        """

        if not isinstance(publication_id, UUID):
            raise ValueError(
                "publication_id must be a UUID."
            )

        if not title.strip():
            raise ValueError(
                "title cannot be empty."
            )

        if not content.strip():
            raise ValueError(
                "content cannot be empty."
            )

        if not correlation_id.strip():
            raise ValueError(
                "correlation_id cannot be empty."
            )

        if not idempotency_key.strip():
            raise ValueError(
                "idempotency_key cannot be empty."
            )

        text = (
            f"*{title.strip()}*\n\n"
            f"{content.strip()}"
        )

        record = await self._publisher.publish(
            channel_id=self._channel_id,
            text=text,
            announcement_type=self._announcement_type,
        )

        return record.message_id

from app.domain.identity.principal import Principal


class AuthenticationRequiredError(PermissionError):
    """No valid authenticated principal exists."""


class PermissionDeniedError(PermissionError):
    """Authenticated principal lacks a permission."""


class AuthorizationService:
    """Provider-neutral authorization boundary."""

    def require_authenticated(
        self,
        principal: Principal | None,
    ) -> Principal:
        if principal is None or not principal.is_active:
            raise AuthenticationRequiredError(
                "Authenticated application identity is required."
            )
        return principal

    def require_permission(
        self,
        *,
        principal: Principal | None,
        permission: Permission,
    ) -> Principal:
        authenticated = self.require_authenticated(principal)

        if not has_permission(
            roles=authenticated.roles,
            permission=permission,
        ):
            raise PermissionDeniedError(
                f"Permission required: {permission.value}."
            )

        return authenticated

from __future__ import annotations

from enum import StrEnum


class Permission(StrEnum):
    ANNOUNCEMENT_VIEW = "announcement.view"
    ANNOUNCEMENT_PUBLISH = "announcement.publish"
    FUNCTIONALITY_CONTROL = "functionality.control"
    INTERNAL_DATA_READ = "internal_data.read"


class Role(StrEnum):
    BETA_USER = "beta_user"
    OPERATOR = "operator"
    ADMIN = "admin"


ROLE_PERMISSIONS: dict[Role, frozenset[Permission]] = {
    Role.BETA_USER: frozenset(),
    Role.OPERATOR: frozenset(
        {
            Permission.ANNOUNCEMENT_VIEW,
            Permission.ANNOUNCEMENT_PUBLISH,
            Permission.FUNCTIONALITY_CONTROL,
            Permission.INTERNAL_DATA_READ,
        }
    ),
    Role.ADMIN: frozenset(
        {
            Permission.ANNOUNCEMENT_VIEW,
            Permission.ANNOUNCEMENT_PUBLISH,
            Permission.FUNCTIONALITY_CONTROL,
            Permission.INTERNAL_DATA_READ,
        }
    ),
}


def has_permission(
    *,
    roles: frozenset[str],
    permission: Permission,
) -> bool:
    return any(
        permission in ROLE_PERMISSIONS.get(
            Role(raw_role),
            frozenset(),
        )
        for raw_role in roles
        if raw_role in {role.value for role in Role}
    )
