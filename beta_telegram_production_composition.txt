@dataclass(frozen=True)
class BetaTelegramComposition:
    """
    Fully composed Beta Telegram classroom application graph.

    This object contains application/infrastructure components only.
    It does not start Telegram polling.
    """

    beta_access_service: BetaAccessService
    conversation_runtime: ConversationRuntime
    conversation_service: TelegramConversationService
    conversation_boundary: BetaTelegramConversationBoundary
    private_chat_gate: BetaPrivateChatGate
    start_handler: BetaTelegramStartHandler
    adapter: BetaTelegramAdapter
    runtime: TelegramBotRuntime


def create_beta_telegram_composition(
    *,
    user_repository: Any,
    token_repository: Any,
    binding_repository: Any,
    conversation_event_repository: Any,
    privacy_identifier_service: Any,
    conversation_event_validator: Any,
    expiration_minutes: int = 1440,
) -> BetaTelegramComposition:
    """
    Compose the production Beta Telegram classroom graph.

    Dependency direction:

        Infrastructure
              â†“
        Application services
              â†“
        AI Runtime
              â†“
        Provider abstraction

    No Telegram component creates an LLM provider directly.
    """

    beta_access_service = BetaAccessService(
        user_repository=user_repository,
        token_repository=token_repository,
        binding_repository=binding_repository,
        expiration_minutes=expiration_minutes,
    )

    conversation_runtime = ConversationRuntime(
        llm_provider=LLMProviderFactory.create(),
    )

    conversation_service = TelegramConversationService(
        conversation_runtime=conversation_runtime,
        conversation_event_repository=(
            conversation_event_repository
        ),
        privacy_identifier_service=(
            privacy_identifier_service
        ),
        conversation_event_validator=(
            conversation_event_validator
        ),
    )

    conversation_boundary = (
        BetaTelegramConversationBoundary(
            conversation_service=conversation_service,
            beta_access_service=beta_access_service,
        )
    )

    private_chat_gate = BetaPrivateChatGate(
        beta_access_service=beta_access_service,
    )

    start_handler = BetaTelegramStartHandler(
        beta_access_service=beta_access_service,
    )

    adapter = BetaTelegramAdapter(
        start_handler=start_handler,
        private_chat_gate=private_chat_gate,
        conversation_boundary=conversation_boundary,
    )

    runtime = TelegramBotRuntime(
        adapter=adapter,
        conversation_boundary=conversation_boundary,
    )

    return BetaTelegramComposition(
        beta_access_service=beta_access_service,
        conversation_runtime=conversation_runtime,
        conversation_service=conversation_service,
        conversation_boundary=conversation_boundary,
        private_chat_gate=private_chat_gate,
        start_handler=start_handler,
        adapter=adapter,
        runtime=runtime,
    )


__all__ = [
    "BetaTelegramComposition",
    "create_beta_telegram_composition",
]

from __future__ import annotations

import os

from sqlalchemy import create_engine
from sqlalchemy.orm import DeclarativeBase, sessionmaker, Session


class Base(DeclarativeBase):
    pass


def database_url() -> str:
    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for Beta Telegram persistence."
        )
    return value


def create_database_engine():
    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(engine):
    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )

def _utcnow() -> datetime:
    return datetime.now(timezone.utc)


class PostgreSQLBetaUserRepository(BetaUserRepository):

    def __init__(
        self,
        session_factory,
    ):
        self._session_factory = session_factory


    async def get(
        self,
        user_id: str,
    ) -> BetaUserState | None:

        with self._session_factory() as session:

            row = session.get(
                BetaUserRecord,
                user_id,
            )

            return (
                None
                if row is None
                else _user_from_record(row)
            )


    async def get_by_telegram_id(
        self,
        telegram_id: int,
    ) -> BetaUserState | None:
        """
        Resolve Telegram identity.

        Source of truth:

            telegram_bindings
                    |
                    v
                beta_users
        """

        with self._session_factory() as session:

            binding = session.scalar(
                select(TelegramBindingRecord).where(
                    TelegramBindingRecord.telegram_user_id
                    == str(telegram_id)
                )
            )


            if binding is None:
                return None


            row = session.get(
                BetaUserRecord,
                binding.user_id,
            )


            if row is None:
                return None


            user = _user_from_record(row)


            #
            # Telegram binding is authoritative.
            #
            # Do not depend on beta_users.telegram_id.
            #
            return BetaUserState(
                user_id=user.user_id,
                email=user.email,
                email_verified=user.email_verified,
                beta_approved=user.beta_approved,
                telegram_id=int(
                    binding.telegram_user_id
                ),
                active_world_id=user.active_world_id,
            )


    async def save(
        self,
        user: BetaUserState,
    ) -> BetaUserState:

        now = _utcnow()

        with self._session_factory() as session:

            row = session.get(
                BetaUserRecord,
                user.user_id,
            )


            if row is None:

                row = BetaUserRecord(
                    id=user.user_id,
                    email=user.email,
                    email_verified=user.email_verified,
                    approved=user.beta_approved,
                    active_world_id=user.active_world_id,
                    telegram_id=(
                        None
                        if user.telegram_id is None
                        else str(user.telegram_id)
                    ),
                    created_at=now,
                    updated_at=now,
                )

                session.add(row)


            else:

                row.email = user.email
                row.email_verified = user.email_verified
                row.approved = user.beta_approved
                row.active_world_id = user.active_world_id

                row.telegram_id = (
                    None
                    if user.telegram_id is None
                    else str(user.telegram_id)
                )

                row.updated_at = now


            session.commit()


        return user



class PostgreSQLBetaTokenRepository(
    BetaTokenRepository
):

    def __init__(
        self,
        session_factory,
    ):
        self._session_factory = session_factory


    async def get_by_hash(
        self,
        token_hash: str,
    ) -> BetaConnectionToken | None:

        with self._session_factory() as session:

            row = session.scalar(
                select(BetaConnectionTokenRecord).where(
                    BetaConnectionTokenRecord.token_hash
                    == token_hash
                )
            )

            return (
                None
                if row is None
                else _token_from_record(row)
            )


    async def save(
        self,
        token: BetaConnectionToken,
    ) -> BetaConnectionToken:

        now = _utcnow()

        with self._session_factory() as session:

            row = session.get(
                BetaConnectionTokenRecord,
                token.token_id,
            )


            if row is None:

                session.add(
                    BetaConnectionTokenRecord(
                        id=token.token_id,
                        token_hash=token.token_hash,
                        user_id=token.user_id,
                        expires_at=token.expires_at,
                        used_at=token.used_at,
                        created_at=now,
                    )
                )

            else:

                row.used_at = token.used_at


            session.commit()


        return token


    async def mark_used(
        self,
        token: BetaConnectionToken,
    ) -> BetaConnectionToken:

        return await self.save(token)

from __future__ import annotations

from datetime import datetime

from sqlalchemy import select, update
from sqlalchemy.orm import Session

from app.domain.beta_access.beta_access_repository import BetaConnectionTokenRepository
from app.domain.beta_access.beta_connection_token import BetaConnectionToken
from app.infrastructure.persistence.beta_connection_token_models import BetaConnectionTokenModel


class PostgreSQLBetaConnectionTokenRepository(BetaConnectionTokenRepository):
    def __init__(self, session: Session) -> None:
        self._session = session

    def add(self, token: BetaConnectionToken) -> None:
        self._session.add(BetaConnectionTokenModel(
            id=token.id, user_id=token.user_id, token_hash=token.token_hash,
            expires_at=token.expires_at, used_at=token.used_at, created_at=token.created_at,
        ))
        self._session.flush()

    def get_by_hash(self, token_hash: str) -> BetaConnectionToken | None:
        row = self._session.scalar(select(BetaConnectionTokenModel).where(
            BetaConnectionTokenModel.token_hash == token_hash
        ))
        if row is None:
            return None
        return BetaConnectionToken(
            id=row.id, user_id=row.user_id, token_hash=row.token_hash,
            expires_at=row.expires_at, created_at=row.created_at, used_at=row.used_at,
        )

    def mark_used(self, token_id: str, used_at: datetime) -> None:
        self._session.execute(
            update(BetaConnectionTokenModel)
            .where(BetaConnectionTokenModel.id == token_id)
            .where(BetaConnectionTokenModel.used_at.is_(None))
            .values(used_at=used_at)
        )
        self._session.flush()

class PostgreSQLConversationEventRepository(
    ConversationEventRepository
):
    """
    PostgreSQL implementation of conversation event storage.


    Privacy isolation:

        telegram_user_hash
                +
          conversation_id
                +
             world_id


    Responsibility:

    - persist conversation events
    - retrieve conversation history


    Does NOT:

    - validate business rules
    - create memories
    - analyse learning behaviour
    - create RAG knowledge
    """



    def __init__(
        self,
        session_factory,
    ) -> None:


        if session_factory is None:

            raise ValueError(
                "session_factory cannot be None."
            )


        self._session_factory = session_factory






    async def save(
        self,
        event: ConversationEvent,
    ) -> ConversationEvent:


        with self._session_factory() as session:


            record = ConversationEventRecord(


                id=event.event_id,


                conversation_id=(
                    event.conversation_id
                ),


                telegram_user_hash=(
                    event.telegram_user_hash
                ),


                world_id=(
                    event.world_id
                ),


                channel=(
                    event.channel
                ),


                role=(
                    event.role
                ),


                content=(
                    event.content
                ),


                metadata_json=(
                    event.metadata
                ),


                created_at=(
                    event.created_at
                ),
            )


            session.add(record)

            session.commit()



        return event







    async def get_recent(
        self,
        *,
        telegram_user_hash: str,
        world_id: str,
        limit: int = 20,
    ) -> list[ConversationEvent]:


        with self._session_factory() as session:


            rows = session.scalars(


                select(
                    ConversationEventRecord
                )


                .where(

                    ConversationEventRecord.telegram_user_hash
                    ==
                    telegram_user_hash

                )


                .where(

                    ConversationEventRecord.world_id
                    ==
                    world_id

                )


                .order_by(

                    ConversationEventRecord.created_at.desc()

                )


                .limit(limit)


            ).all()



        return [

            _event_from_record(row)

            for row in reversed(rows)

        ]








def _event_from_record(
    row: ConversationEventRecord,
) -> ConversationEvent:


    return ConversationEvent(

        event_id=row.id,


        conversation_id=(
            row.conversation_id
        ),


        telegram_user_hash=(
            row.telegram_user_hash
        ),


        world_id=(
            row.world_id
        ),


        channel=(
            row.channel
        ),


        role=(
            row.role
        ),


        content=(
            row.content
        ),


        metadata=(
            row.metadata_json
        ),


        created_at=(
            row.created_at
        ),
    )
