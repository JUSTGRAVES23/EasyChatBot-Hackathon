import { StrictMode } from "react";
import { createRoot } from "react-dom/client";

async function start() {
  const path = window.location.pathname.replace(/\/+$/, "") || "/";

  if (path === "/operations-center/governance") {
    const { default: GovernanceOperationsCenter } =
      await import("./pages/operations-center/GovernanceOperationsCenter");

    document.body.innerHTML = '<div id="root"></div>';

    const rootElement = document.getElementById("root");

    if (!rootElement) {
      throw new Error("Governance root element could not be created.");
    }

    createRoot(rootElement).render(
      <StrictMode>
        <GovernanceOperationsCenter />
      </StrictMode>,
    );

    return;
  }

  // The existing learner Home remains controlled by index.html/app.js.
}

void start();
