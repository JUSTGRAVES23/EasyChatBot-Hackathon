const EMPTY_SUMMARY: PrivacyGovernanceSummary = {
  total_cases: 0,
  pending: 0,
  approved: 0,
  blocked: 0,
  approval_records: 0,
};

interface Props {
  initialLoad?: boolean;

  /**
   * Optional case ID supplied by the privacy-governance
   * deep-link route:
   *
   * /privacy-governance/:caseId
   *
   * This is intentionally dynamic. No case ID is hard-coded.
   */
  initialCaseId?: string;
}

function normalizeStatus(status: string): GovernanceFilter {
  const value = status.toUpperCase();

  if (value.includes("PENDING")) {
    return "PENDING_DPO_GRC";
  }

  if (value.includes("APPROVED")) {
    return "APPROVED";
  }

  if (
    value.includes("BLOCKED") ||
    value.includes("REJECTED")
  ) {
    return "BLOCKED";
  }

  return "ALL";
}

function statusLabel(value: string): string {
  return value.replaceAll("_", " ");
}

function formatDate(value: string): string {
  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return new Intl.DateTimeFormat(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(date);
}

  /*
   * ---------------------------------------------------------
   * Dashboard data
   * ---------------------------------------------------------
   */

  const loadDashboard = useCallback(
    async (
      mode: "initial" | "refresh" = "refresh",
    ) => {
      if (mode === "initial") {
        setLoading(true);
      } else {
        setRefreshing(true);
      }

      setError(null);

      try {
        const result =
          await loadPrivacyGovernanceDashboard();

        setSummary(result.summary);
        setStatusChart(result.statusChart);
        setCases(result.cases);
      } catch (cause) {
        setError(
          cause instanceof Error
            ? cause.message
            : "Unable to load privacy governance data.",
        );

        setLoading(false);
        setRefreshing(false);
      }

  useEffect(() => {
    if (initialLoad) {
      void loadDashboard("initial");
    }
  }, [initialLoad, loadDashboard]);

  /*
   * ---------------------------------------------------------
   * Case filtering
   * ---------------------------------------------------------
   */

  const visibleCases = useMemo(() => {
    if (filter === "ALL") {
      return cases;
    }

      (item) =>
        normalizeStatus(
          item.governance_status,
        ) === filter,
    );
  }, [cases, filter]);

  /*
   * ---------------------------------------------------------
   * Case detail
   * ---------------------------------------------------------
   */

  const openCase = useCallback(
    async (caseId: string) => {
      const normalizedCaseId = caseId.trim();

      if (!normalizedCaseId) {
        setSelectedCaseId(null);
        setSelectedCase(null);
        setDetailError(
          "A valid governance case ID is required.",
        );
        return;
      }

      setSelectedCaseId(normalizedCaseId);
      setSelectedCase(null);
      setDetailError(null);
      setDetailLoading(true);

      try {
        const detail =
          await getPrivacyGovernanceCase(
            normalizedCaseId,
          );

        setSelectedCase(detail);
      } catch (cause) {
        setDetailError(
          cause instanceof Error
            ? cause.message
            : "Unable to load this governance case.",
        );
      } finally {
        setDetailLoading(false);
      }
    },
    [],

  /*

   * Reminder email:

  useEffect(() => {

      return;
    }

    void openCase(caseId);
  }, [initialCaseId, openCase]);

  const retryCase = useCallback(() => {

    }

      setCases((current) =>
        current.map((item) =>
          item.case_id === updatedCase.case_id

      );

function formatDate(value?: string | null): string {
  if (!value) {
    return "â€”";
