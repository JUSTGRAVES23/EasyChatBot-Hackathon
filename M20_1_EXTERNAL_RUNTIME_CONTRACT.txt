"""
Tests External Runtime Container.

Validates:

Application Bootstrap
        |
        v
External Runtime Container
        |
        v
ExternalRuntimeService

The test uses controlled dependencies.

No external providers are called.
"""

from __future__ import annotations

import pytest

from backend.runtime.external_runtime_container import (
    configure_external_runtime_service,
    get_external_runtime_service,
)

from backend.services.external_runtime_service import (
    ExternalRuntimeService,
)


class FakeGateway:
    """
    Controlled Runtime gateway.
    """

    async def execute(
        self,
        request,
    ):
        return None


def test_container_registers_runtime_service():
    """
    Configured service should be retrievable.
    """

    service = ExternalRuntimeService(
        gateway=FakeGateway(),
    )

    configure_external_runtime_service(
        service,
    )

    result = get_external_runtime_service()

    assert result is service


def test_container_rejects_missing_service():
    """
    Container should reject invalid configuration.
    """

    with pytest.raises(
        ValueError,
        match="ExternalRuntimeService cannot be None",
    ):
        configure_external_runtime_service(
            None,
        )

"""
Tests for the External Runtime application service.

The production service does not exist yet.

These tests define the required application-layer contract before
implementation.
"""

from __future__ import annotations

from unittest.mock import AsyncMock, Mock

import pytest

from src.ai_runtime.capabilities.gateway import (
    CapabilityGateway,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)

from backend.services.external_runtime_service import (
    ExternalRuntimeService,
)


def make_request() -> CapabilityRequest:
    """
    Create a minimal external Runtime capability request.
    """

    return CapabilityRequest(
        capability=CapabilityType.PODCAST_DISCOVERY,
        request_id="application-runtime-test-001",
        payload={
            "query": "French learning podcast",
        },
        requested_by="learning_application",
    )


@pytest.mark.asyncio
async def test_external_runtime_service_delegates_to_gateway():
    """
    Verify that the application service delegates execution to
    the injected CapabilityGateway.
    """

    gateway = Mock(
        spec=CapabilityGateway,
    )

    expected_result = CapabilityResult(
        request_id="application-runtime-test-001",
        capability=CapabilityType.PODCAST_DISCOVERY,
        status=CapabilityExecutionStatus.COMPLETED,
        output={
            "ok": True,
        },
    )

    gateway.execute = AsyncMock(
        return_value=expected_result,
    )

    service = ExternalRuntimeService(
        gateway=gateway,
    )

    request = make_request()

    result = await service.execute(
        request,
    )

    assert result is expected_result

    gateway.execute.assert_awaited_once_with(
        request,
    )


def test_external_runtime_service_requires_gateway():
    """
    Verify that the application service rejects a missing gateway.
    """

    with pytest.raises(
        ValueError,
        match="CapabilityGateway cannot be None",
    ):
        ExternalRuntimeService(
            gateway=None,  # type: ignore[arg-type]
        )


@pytest.mark.asyncio
async def test_external_runtime_service_rejects_missing_request():
    """
    Verify that the application service rejects a missing request
    before attempting Runtime execution.
    """

    gateway = Mock(
        spec=CapabilityGateway,
    )

    gateway.execute = AsyncMock()

    service = ExternalRuntimeService(
        gateway=gateway,
    )

    with pytest.raises(
        ValueError,
        match="CapabilityRequest cannot be None",
    ):
        await service.execute(
            None,  # type: ignore[arg-type]
        )

    gateway.execute.assert_not_awaited()

===== backend/application/bootstrap.py =====
"""
EasyChatBot Application Bootstrap.

Responsible for application startup composition.

Responsibilities:

- create application-level services;
- compose approved Runtime dependencies;
- register configured services.

This module does not:

- execute agents;
- execute capabilities;
- contain business logic;
- expose runtime state;
- serve API requests.

Runtime construction remains delegated to:

    src.ai_runtime.composition

Live Intelligence Agent construction remains delegated to:

    src.ai_agents.live_intelligence_agent.composition

Internal Documentation Agent construction remains delegated to:

    backend.application.internal_documentation_composition
"""

from __future__ import annotations

from collections.abc import Sequence

from backend.application.factory import (
    ApplicationFactory,
)

from backend.application.internal_documentation_composition import (
    create_internal_documentation_agent_from_repository,
)

from backend.runtime.external_runtime_container import (
    configure_external_runtime_service,
    get_external_runtime_service as _get_external_runtime_service,
)

from backend.services.external_runtime_service import (
    ExternalRuntimeService,
)

from src.ai_runtime.composition import (
    create_external_runtime_gateway,
)

from src.ai_agents.live_intelligence_agent.composition import (
    create_live_intelligence_agent,
)

from src.ai_runtime.podcast.parallel import (
    ParallelPodcastProvider,
)


_bootstrapped = False


def bootstrap_application() -> None:
    """
    Bootstrap application dependencies.

    Creates the ExternalRuntimeService and registers it
    inside the application runtime container.
    """

    global _bootstrapped

    if _bootstrapped:
        return

    factory = ApplicationFactory()

    gateway = create_external_runtime_gateway(
        live_intelligence_agent=(
            create_live_intelligence_agent()
        ),
        podcast_discovery_agent=(
            create_podcast_discovery_agent()
        ),
        podcast_providers=(
            create_podcast_providers()
        ),
        governance=(
            create_governance()
        ),
        internal_documentation_agent=(
            create_internal_documentation_agent_from_repository()
        ),
    )

    runtime_service = (
        factory.create_external_runtime_service(
            gateway=gateway,
        )
    )

    configure_external_runtime_service(
        runtime_service,
    )

    _bootstrapped = True


def get_external_runtime_service() -> ExternalRuntimeService:
    """
    Backward-compatible Runtime accessor.
    """

    return _get_external_runtime_service()


def reset_bootstrap() -> None:
    """
    Test-only reset.
    """

    global _bootstrapped

    _bootstrapped = False
