"""
Voice Call Notification Service.

Application service responsible for
creating and managing notifications
for AI character voice calls.

Responsibilities:
- Create notifications.
- Link notifications to invitations.
- Manage notification lifecycle.
- Coordinate notification repository.

Does NOT:
- Send emails.
- Send push notifications.
- Manage inbox UI.
- Execute voice calls.
- Manage LiveKit.
"""

from __future__ import annotations


from dataclasses import dataclass


from datetime import (
    datetime,
    timezone,
)


import uuid


from app.application.uai.notifications.voice_call_notification_repository import (
    VoiceCallNotificationRepository,
)


from app.domain.uai.notifications.voice_call_notification import (
    VoiceCallNotification,
    VoiceCallNotificationChannel,
    VoiceCallNotificationStatus,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallNotificationService:
    """
    Application service for voice call
    notification lifecycle.
    """

    repository: VoiceCallNotificationRepository



    def create_notification(
        self,
        *,
        invitation_id: str,
        voice_call_id: str,
        learner_id: str,
        character_id: str,
        channel: VoiceCallNotificationChannel,
    ) -> VoiceCallNotification:
        """
        Create notification linked
        to a voice call invitation.
        """


        now = datetime.now(
            timezone.utc
        )


        notification = VoiceCallNotification(
            id=(
                f"notification-{uuid.uuid4()}"
            ),
            invitation_id=invitation_id,
            voice_call_id=voice_call_id,
            learner_id=learner_id,
            character_id=character_id,
            channel=channel,
            status=(
                VoiceCallNotificationStatus.CREATED
            ),
            created_at=now,
        )


        return (
            self.repository
            .save(
                notification=notification,
            )
        )



    def get_notification(
        self,
        *,
        notification_id: str,
    ) -> VoiceCallNotification | None:
        """
        Retrieve notification.
        """


        return (
            self.repository
            .get_by_id(
                notification_id=notification_id,
            )
        )



    def mark_pending(
        self,
        *,
        notification: VoiceCallNotification,
    ) -> VoiceCallNotification:
        """
        Move notification to pending.
        """


        updated = (
            notification
            .mark_pending()
        )


        return (
            self.repository
            .save(
                notification=updated,
            )
        )



    def mark_sent(
        self,
        *,
        notification: VoiceCallNotification,
    ) -> VoiceCallNotification:
        """
        Mark notification delivered.
        """


        updated = (
            notification
            .mark_sent()
        )


        return (
            self.repository
            .save(
                notification=updated,
            )
        )



    def mark_read(
        self,
        *,
        notification: VoiceCallNotification,
    ) -> VoiceCallNotification:
        """
        Mark notification read.
        """


        updated = (
            notification
            .mark_read()
        )


        return (
            self.repository
            .save(
                notification=updated,
            )
        )



    def action_notification(
        self,
        *,
        notification: VoiceCallNotification,
    ) -> VoiceCallNotification:
        """
        Mark learner actioned
        the voice call notification.
        """


        updated = (
            notification
            .mark_actioned()
        )


        return (
            self.repository
            .save(
                notification=updated,
            )
        )



    def fail_notification(
        self,
        *,
        notification: VoiceCallNotification,
    ) -> VoiceCallNotification:
        """
        Mark notification failed.
        """


        updated = (
            notification
            .mark_failed()
        )


        return (
            self.repository
            .save(
                notification=updated,
            )
        )



__all__ = [
    "VoiceCallNotificationService",
]

from __future__ import annotations

from dataclasses import dataclass

from app.application.uai.voice.voice_call_context_factory import (
    VoiceCallContextFactory,
)
from app.application.uai.voice.voice_call_execution_coordinator import (
    VoiceCallExecutionCoordinator,
)
from app.application.uai.voice.voice_call_event_service import (
    VoiceCallEventService,
)
from app.application.uai.voice.voice_call_invitation_service import (
    VoiceCallInvitationService,
)
from app.application.uai.voice.voice_call_management_service import (
    VoiceCallManagementService,
)
from app.application.uai.notifications.voice_call_notification_service import (
    VoiceCallNotificationService,
)
from app.application.uai.voice.voice_call_session_service import (
    VoiceCallSessionService,
)
from app.domain.uai.voice.voice_call import VoiceCall
from app.domain.uai.voice.voice_call_event import VoiceCallEventType
from app.domain.uai.notifications.voice_call_notification import (
    VoiceCallNotification,
)


@dataclass
class VoiceCallAcceptanceService:
    """
    Application service responsible for accepting an incoming voice-call
    notification and converging the acceptance workflow into the canonical
    voice execution path.

    Responsibilities:
    - Resolve the actionable notification.
    - Resolve and validate the associated invitation.
    - Resolve the associated VoiceCall.
    - Create or reconnect the authoritative VoiceCallSession.
    - Build the execution context from that existing session.
    - Delegate execution to VoiceCallExecutionCoordinator.
    - Record the ACCEPTED lifecycle event.

    This service does not:
    - Access LiveKit directly.
    - Construct infrastructure providers.
    - Persist directly.
    - Implement voice execution itself.
    """

    voice_call_management_service: VoiceCallManagementService
    voice_call_invitation_service: VoiceCallInvitationService
    voice_call_notification_service: VoiceCallNotificationService
    voice_call_session_service: VoiceCallSessionService
    voice_call_event_service: VoiceCallEventService
    voice_call_context_factory: VoiceCallContextFactory
    voice_call_execution_coordinator: VoiceCallExecutionCoordinator

    def accept_voice_call(
        self,
        notification: VoiceCallNotification,
    ) -> dict[str, object]:
        """
        Accept a voice-call notification and start the canonical execution
        workflow using the authoritative VoiceCallSession.
        """

        actionable_notification = (
            self.voice_call_notification_service.action_notification(
                notification
            )
        )

        invitation = self.voice_call_invitation_service.get_invitation(
            actionable_notification.invitation_id
        )

        voice_call = self.voice_call_management_service.get_voice_call(
            invitation.voice_call_id
        )

        if voice_call is None:
            raise ValueError(
                f"Voice call not found: {invitation.voice_call_id}"
            )

        if voice_call.state == "CREATED":
            session = self.voice_call_session_service.create_session(
                voice_call
            )
        elif self.voice_call_session_service.can_reconnect(voice_call):
            session = self.voice_call_session_service.reconnect_session(
                voice_call
            )
        else:
            raise ValueError(
                f"Voice call cannot be accepted in state: {voice_call.state}"
            )

        context = self.voice_call_context_factory.create_from_session(
            session=session,
            voice_call=voice_call,
        )

        self.voice_call_execution_coordinator.start(
            context=context,
        )

        self.voice_call_event_service.record_event(
            voice_call_id=voice_call.id,
            event_type=VoiceCallEventType.ACCEPTED,
        )

        return {
            "notification": actionable_notification,
            "voice_call": voice_call,
            "session": session,
        }


__all__ = ["VoiceCallAcceptanceService"]

"""
Voice Call Notification Domain Model.

Represents a notification requesting
the learner to join or resume an AI
character voice call.

Responsibilities:
- Store notification identity.
- Link notification to voice invitation.
- Track notification lifecycle.
- Represent delivery intent.

Does NOT:
- Send emails.
- Send push notifications.
- Manage inbox storage.
- Execute voice calls.
- Manage LiveKit.
"""

from __future__ import annotations


from dataclasses import (
    dataclass,
    replace,
)


from datetime import (
    datetime,
    timezone,
)


from enum import Enum



class VoiceCallNotificationStatus(
    str,
    Enum,
):
    """
    Notification lifecycle states.
    """

    CREATED = "created"

    PENDING = "pending"

    SENT = "sent"

    READ = "read"

    ACTIONED = "actioned"

    EXPIRED = "expired"

    FAILED = "failed"



class VoiceCallNotificationChannel(
    str,
    Enum,
):
    """
    Supported notification channels.

    Delivery implementations remain
    outside this domain.
    """

    INBOX = "inbox"

    EMAIL = "email"



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallNotification:
    """
    Domain entity representing a
    voice call notification.

    Relationship:

        Notification
              |
              ↓
        VoiceCallInvitation
              |
              ↓
        VoiceCall
    """

    id: str

    invitation_id: str

    voice_call_id: str

    learner_id: str

    character_id: str

    channel: VoiceCallNotificationChannel

    status: VoiceCallNotificationStatus

    created_at: datetime



    def __post_init__(
        self,
    ) -> None:
        """
        Validate domain invariants.
        """


        if not self.id:

            raise ValueError(
                "Notification id is required."
            )


        if not self.invitation_id:

            raise ValueError(
                "Invitation id is required."
            )


        if not self.voice_call_id:

            raise ValueError(
                "Voice call id is required."
            )


        if not self.learner_id:

            raise ValueError(
                "Learner id is required."
            )


        if not self.character_id:

            raise ValueError(
                "Character id is required."
            )



    def mark_pending(
        self,
    ) -> "VoiceCallNotification":
        """
        Mark notification ready
        for delivery.
        """


        return replace(
            self,
            status=(
                VoiceCallNotificationStatus.PENDING
            ),
        )



    def mark_sent(
        self,
    ) -> "VoiceCallNotification":
        """
        Mark notification delivered.
        """


        return replace(
            self,
            status=(
                VoiceCallNotificationStatus.SENT
            ),
        )



    def mark_read(
        self,
    ) -> "VoiceCallNotification":
        """
        Mark notification opened
        by learner.
        """


        return replace(
            self,
            status=(
                VoiceCallNotificationStatus.READ
            ),
        )



    def mark_actioned(
        self,
    ) -> "VoiceCallNotification":
        """
        Mark learner accepted
        the voice call.
        """


        return replace(
            self,
            status=(
                VoiceCallNotificationStatus.ACTIONED
            ),
        )



    def mark_failed(
        self,
    ) -> "VoiceCallNotification":
        """
        Mark delivery failure.
        """


        return replace(
            self,
            status=(
                VoiceCallNotificationStatus.FAILED
            ),
        )



    def is_recent(
        self,
        *,
        now: datetime,
        max_age_minutes: int = 60,
    ) -> bool:
        """
        Check whether notification
        remains relevant.
        """


        if now.tzinfo is None:

            now = now.replace(
                tzinfo=timezone.utc
            )


        created_at = self.created_at


        if created_at.tzinfo is None:

            created_at = created_at.replace(
                tzinfo=timezone.utc
            )


        age_seconds = (
            now - created_at
        ).total_seconds()


        return (
            age_seconds
            <=
            max_age_minutes * 60
        )



__all__ = [
    "VoiceCallNotification",
    "VoiceCallNotificationStatus",
    "VoiceCallNotificationChannel",
]
