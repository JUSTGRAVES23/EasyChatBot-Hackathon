Application HTTP boundary.

Responsibilities:

API

Application Services

CapabilityGateway

# =========================================================
# Runtime Service Configuration
# =========================================================

_external_runtime_service_override: (
    ExternalRuntimeService | None
) = None


def configure_external_runtime_service(
    service: ExternalRuntimeService,
) -> None:
    """
    Configure an explicit Runtime service override.

    Primarily used by application tests and controlled
    application composition.

    Production Runtime ownership remains with:

        backend.runtime.external_runtime_container
    """

    global _external_runtime_service_override

    if service is None:
        raise ValueError(
            "ExternalRuntimeService cannot be None."
        )

    _external_runtime_service_override = service


def create_external_runtime_service() -> ExternalRuntimeService:
    """
    Retrieve the configured Runtime service.

    Resolution order:

    1. Explicit test/application override.
    2. Runtime container.
    3. Controlled in-memory Runtime fallback.

    The fallback exists only so API tests can execute
    without requiring full application bootstrap.
    """

    if (
        _external_runtime_service_override
        is not None
    ):
        return _external_runtime_service_override

    try:
        return get_external_runtime_service()

    except RuntimeError:
        gateway = create_external_runtime_gateway(
            live_intelligence_agent=(
                TestLiveIntelligenceAgent()
            ),
            podcast_discovery_agent=(
                _create_test_podcast_agent()
            ),
            podcast_providers=[
                _create_test_podcast_provider()
            ],
            governance=(
                _create_test_governance()
            ),
        )

        return (
            application_factory
            .create_external_runtime_service(
                gateway=gateway,
            )
        )


def create_runtime_application_service(
) -> RuntimeApplicationService:
    """
    Create the application-layer Runtime service.

    API routes remain unaware of Runtime composition.
    """

    return (
        application_factory
        .create_runtime_application_service(
            runtime_service=(
                create_external_runtime_service()
            ),
        )
    )

def create_companion_service(
    character,
) -> CompanionService:
    """
    Create CompanionService for a selected character.
    """

    llm_service = create_llm_service()

    agent = ConversationAgent(
        character=character,
        llm_service=llm_service,
    )

    return CompanionService(
        character=character,
        conversation_agent=agent,
    )

# =========================================================
# Chat Endpoint
# =========================================================


@router.post(
    "/chat",
    response_model=ChatResponse,
)
async def chat(
    payload: ChatRequest,
) -> ChatResponse:
    """
    Execute a Companion conversation.
    """

    if (
        payload.message is None
        or not payload.message.strip()
    ):
        raise HTTPException(
            status_code=400,
            detail="message is required",
        )

    try:
        character = character_registry.get(
            payload.character,
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    service = create_companion_service(
        character,
    )

    context = ExecutionContext(
        user_id=payload.user_id,
        request=payload.message,
    )

    result = await service.interact(
        context,
        use_research=payload.research,
    )

    return ChatResponse(
        character=result["character"],
        conversation=result["conversation"],
        research=result["research"],
        quality=result["quality"],
        memory_count=result["memory_count"],
        memory_context_count=result.get(
            "memory_context_count",
            0,
        ),
    )


# =========================================================
# Runtime Execute Endpoint
# =========================================================


@router.post(
    "/runtime/execute",
    response_model=RuntimeExecuteResponse,
)
async def execute_runtime(
    payload: RuntimeExecuteRequest,
) -> RuntimeExecuteResponse:
    """
    Execute an approved Runtime capability.

    HTTP concerns remain here.

    Runtime orchestration remains in:

        RuntimeApplicationService
    """

    service = create_runtime_application_service()

    try:
        result = await service.execute(
            capability=payload.capability,
            payload=payload.payload,
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail=str(exc),
        ) from exc

    if isinstance(result, dict):
        return RuntimeExecuteResponse(
            request_id=result["request_id"],
            capability=result["capability"],
            status=result["status"],
            output=result.get("output"),
            error=result.get("error"),
        )

    return RuntimeExecuteResponse(
        request_id=result.request_id,
        capability=result.capability,
        status=result.status,
        output=result.output,
        error=getattr(
            result,
            "error",
            None,
        ),
    )

# Podcast Discovery Endpoint
# =========================================================


@router.post(
    "/capabilities/podcast/discover",
)
async def discover_podcast(
    payload: dict,
):
    """
    Execute Podcast Discovery through the Runtime.

    The endpoint only prepares transport-level defaults.
    Discovery remains owned by the Runtime capability.
    """

    if not payload.get("query"):
        raise HTTPException(
            status_code=400,
            detail="query is required",
        )

    service = create_runtime_application_service()

    enriched_payload = {
        "user_id": payload.get(
            "user_id",
            "demo-user",
        ),
        "character_id": payload.get(
            "character_id",
            "yeti",
        ),
        "character_name": payload.get(
            "character_name",
            "Yeti",
        ),
        "character_message": payload.get(
            "character_message",
            (
                "Help me discover useful "
                "learning content."
            ),
        ),
        **payload,
    }

    try:
        result = await service.execute(
            capability="podcast_discovery",
            payload=enriched_payload,
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail=str(exc),
        ) from exc

    if isinstance(result, dict):
        return {
            "request_id": result["request_id"],
            "capability": result["capability"],
            "status": result["status"],
            "output": result.get("output"),
        }

    return {
        "request_id": result.request_id,
        "capability": result.capability,
        "status": result.status,
        "output": result.output,
    }

@router.post(
    "/hackathon/demo",
    response_model=HackathonDemoResponse,
)
async def hackathon_demo(
    payload: HackathonDemoRequest,
) -> HackathonDemoResponse:
    """
    Hackathon Runtime vertical slice.

    API
     |
     v
    RuntimeApplicationService
     |
     v
    ExternalRuntimeService
     |
     v
    CapabilityGateway
     |
     v
    Live Intelligence Executor
     |
     v
    LiveIntelligenceAgent
    """

    service = create_runtime_application_service()

    try:
        result = await service.execute(
            capability="live_intelligence",
            payload={
                "user_id": payload.user_id,
                "topic": payload.message,
                "language": "French",
                "learner_level": "B2",
            },
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail=str(exc),
        ) from exc

    if isinstance(result, dict):
        return HackathonDemoResponse(
            status=result["status"],
            capability=result["capability"],
            output=result.get("output"),
            request_id=result["request_id"],
        )

    return HackathonDemoResponse(
        status=result.status,
        capability=result.capability,
        output=result.output,
        request_id=result.request_id,
    )

===== API SCHEMAS =====

"""
EasyChatBot API Schemas.

Defines the HTTP request and response contracts
for the EasyChatBot API.

The API schema layer is responsible only for
transport validation and serialisation.

It does not:

- execute agents;
- select providers;
- build prompts;
- manage memory;
- perform research;
- call provider SDKs;
- execute Runtime capabilities.
"""

from __future__ import annotations

from typing import Any

from pydantic import BaseModel
from pydantic import Field


class ChatRequest(BaseModel):
    """
    Request payload for a companion interaction.

    The message field remains optional at the Pydantic
    transport layer so the API route can preserve the
    existing EasyChatBot HTTP contract:

        missing message -> HTTP 400

    This maintains backward compatibility with the
    existing API behaviour and tests.
    """

    user_id: str = Field(
        default="demo-user",
        min_length=1,
        description="Learner/user identifier.",
    )

    character: str = Field(
        default="yeti",
        min_length=1,
        description="Selected EasyChatBot character identifier.",
    )

    message: str | None = Field(
        default=None,
        description="Learner message.",
    )

    research: bool = Field(
        default=False,
        description="Whether live research should be performed.",
    )


class ChatResponse(BaseModel):
    """
    Response payload for a companion interaction.

    Agent output remains flexible during Beta 1 because
    the underlying result contracts are intentionally
    extensible.
    """

    character: str

    conversation: Any

    research: Any = None

    quality: Any

    memory_count: int

    memory_context_count: int = 0


class PodcastDiscoveryRequest(BaseModel):
    """
    HTTP request contract for Podcast Discovery.

    This schema is responsible only for transport
    validation.

    It does not:

    - execute discovery;
    - select providers;
    - apply governance;
    - execute agents.
    """

    query: str = Field(
        min_length=1,
        description="Podcast discovery query.",
    )

    target_language: str = Field(
        min_length=1,
        description="Target language for the learner.",
    )

    learner_level: str = Field(
        min_length=1,
        description="Learner proficiency level.",
    )


class PodcastDiscoveryResponse(BaseModel):
    """
    HTTP response contract for Podcast Discovery.

    The Runtime remains the source of the capability
    result. The API schema only serialises the result
    for HTTP transport.
    """

    request_id: str

    capability: Any

    status: Any

    output: Any = None


class RuntimeExecuteRequest(BaseModel):
    """
    HTTP request contract for External Runtime execution.

    The API schema layer only validates transport data.

    It does not:

    - execute capabilities;
    - call providers;
    - execute agents;
    - apply governance decisions.
    """

    capability: str = Field(
        min_length=1,
        description="Requested Runtime capability.",
    )

    payload: dict[str, Any] = Field(
        default_factory=dict,
        description="Capability execution payload.",
    )

    user_id: str = Field(
        default="demo-user",
        min_length=1,
        description="Learner/user identifier.",
    )


class RuntimeExecuteResponse(BaseModel):
    """
    HTTP response contract for External Runtime execution.

    The Runtime remains responsible for execution.
    This model only serialises the result.
    """

    request_id: str | None = None

    capability: str | None = None

    status: str

    output: Any = None

    error: str | None = None

class HackathonDemoRequest(BaseModel):
    """
    Request contract for the hackathon demonstration flow.

    This endpoint is intentionally separated from the
    production ChatRequest contract.

    Purpose:

    - allow hackathon iteration;
    - protect existing chat functionality;
    - provide a stable demo entry point.

    This schema does not:

    - execute agents;
    - call providers;
    - bypass Runtime controls.
    """

    user_id: str = Field(
        default="demo-user",
        min_length=1,
        description="Demo learner identifier.",
    )

    message: str = Field(
        min_length=1,
        description="Hackathon demo instruction.",
    )


class HackathonDemoResponse(BaseModel):
    """
    Response contract for the hackathon vertical slice.

    The response represents the application result
    returned after Runtime execution.
    """

    status: str

    capability: str

    output: Any = None

    request_id: str | None = None
