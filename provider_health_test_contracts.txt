===== PROVIDER HEALTH SERVICE =====
"""
EasyChatBot AI Operations Center provider health tests.

Validates:

Provider Health Observation
        |
        v
ProviderHealthService
        |
        +----------------------+
        |                      |
        v                      v
Provider Health State   Health Transition
        |                      |
        +----------+-----------+
                   |
                   v
          Provider Incident
"""

from datetime import datetime, timezone

import pytest

from app.application.provider_management.provider_health_models import (
    ProviderHealthObservation,
    ProviderHealthStatus,
    ProviderHealthTransition,
)

from app.application.provider_management.provider_health_service import (
    ProviderHealthService,
)


def create_observation(
    *,
    provider: str = "openai",
    status: ProviderHealthStatus = (
        ProviderHealthStatus.HEALTHY
    ),
    observed_at: datetime | None = None,
    capability: str | None = "conversation",
    consecutive_failures: int = 0,
    consecutive_successes: int = 1,
) -> ProviderHealthObservation:
    """
    Create deterministic provider health evidence.
    """

    if observed_at is None:
        observed_at = datetime(
            2026,
            8,
            14,
            14,
            32,
            tzinfo=timezone.utc,
        )

    return ProviderHealthObservation(
        provider=provider,
        model="gpt-5-mini",
        capability=capability,
        status=status,
        observed_at=observed_at,
        latency_ms=800,
        consecutive_failures=(
            consecutive_failures
        ),
        consecutive_successes=(
            consecutive_successes
        ),
    )


def create_service() -> ProviderHealthService:
    """
    Create a fresh health service.
    """

    return ProviderHealthService()


def test_first_failure_creates_degraded_state() -> None:
    """
    Validate that the first failure produces a degraded
    provider state rather than an immediate outage.
    """

    service = create_service()

    event = service.observe(
        create_observation(
            status=ProviderHealthStatus.DEGRADED,
            consecutive_failures=1,
            consecutive_successes=0,
        )
    )

    state = service.get_state("openai")

    assert state is not None

    assert (
        state.status
        == ProviderHealthStatus.DEGRADED
    )

    assert (
        event.transition
        == ProviderHealthTransition.DEGRADED
    )

    assert (
        event.current_status
        == ProviderHealthStatus.DEGRADED
    )

    assert (
        event.consecutive_failures
        == 1
    )


def test_three_consecutive_failures_create_unavailable_state() -> None:
    """
    Validate the outage threshold.

    Three consecutive failures must move the provider
    from degraded to unavailable.
    """

    service = create_service()

    first = service.observe(
        create_observation(
            status=ProviderHealthStatus.DEGRADED,
            consecutive_failures=1,
            consecutive_successes=0,
        )
    )

    second = service.observe(
        create_observation(
            status=ProviderHealthStatus.DEGRADED,
            consecutive_failures=2,
            consecutive_successes=0,
            observed_at=datetime(
                2026,
                8,
                14,
                14,
                33,
                tzinfo=timezone.utc,
            ),
        )
    )

    third = service.observe(
        create_observation(
            status=ProviderHealthStatus.DEGRADED,
            consecutive_failures=3,
            consecutive_successes=0,
            observed_at=datetime(
                2026,
                8,
                14,
                14,
                34,
                tzinfo=timezone.utc,
            ),
        )
    )

    state = service.get_state("openai")

    assert state is not None

    assert (
        first.transition
        == ProviderHealthTransition.DEGRADED
    )

    assert (
        second.transition
        == ProviderHealthTransition.DEGRADED
    )

    assert (
        third.transition
        == ProviderHealthTransition.UNAVAILABLE
    )

    assert (
        state.status
        == ProviderHealthStatus.UNAVAILABLE
    )

    assert (
        state.consecutive_failures
        == 3
    )

    assert (
        state.unavailable_since
        == datetime(
            2026,
            8,
            14,
            14,
            34,
            tzinfo=timezone.utc,
        )
    )


def test_successful_observation_recovers_provider() -> None:
    """
    Validate provider recovery.

    A successful observation after an outage returns
    the provider to healthy state and generates a
    recovered transition.
    """

    service = create_service()

    service.observe(
        create_observation(
            status=ProviderHealthStatus.DEGRADED,
            consecutive_failures=1,
            consecutive_successes=0,
        )
    )

    service.observe(
        create_observation(
            status=ProviderHealthStatus.DEGRADED,
            consecutive_failures=2,
            consecutive_successes=0,
        )
    )

    service.observe(
        create_observation(
            status=ProviderHealthStatus.DEGRADED,
            consecutive_failures=3,
            consecutive_successes=0,
        )
    )

    recovery = service.observe(
        create_observation(
            status=ProviderHealthStatus.HEALTHY,
            consecutive_failures=0,
            consecutive_successes=1,
            observed_at=datetime(
                2026,
                8,
                14,
                14,
                35,
                tzinfo=timezone.utc,
            ),
        )
    )

    state = service.get_state("openai")

    assert state is not None

    assert (
        state.status
        == ProviderHealthStatus.HEALTHY
    )

    assert (
        recovery.transition
        == ProviderHealthTransition.RECOVERED
    )

    assert (
        recovery.previous_status
        == ProviderHealthStatus.UNAVAILABLE
    )

    assert (
        recovery.current_status
        == ProviderHealthStatus.HEALTHY
    )

    assert (
        state.last_success_at
        == datetime(
            2026,
            8,
            14,
            14,
            35,
            tzinfo=timezone.utc,
        )
    )

    assert (
        state.unavailable_since
        is None
    )

def test_recovery_updates_provider_health() -> None:
    """
    Validate successful recovery updates the read model.

    Provider outage:

        10:32

    Provider recovery:

        10:39

    The current health contract treats the first healthy
    observation as recovery.
    """

    service = ProviderHealthMonitorService()

    recovery_time = datetime(
        2026,
        8,
        15,
        10,
        39,
        tzinfo=timezone.utc,
    )

    failure_timestamps = (
        datetime(
            2026,
            8,
            15,
            10,
            30,
            tzinfo=timezone.utc,
        ),
        datetime(
            2026,
            8,
            15,
            10,
            31,
            tzinfo=timezone.utc,
        ),
        datetime(
            2026,
            8,
            15,
            10,
            32,
            tzinfo=timezone.utc,
        ),
    )

    failure_transitions = []

    for index, timestamp in enumerate(
        failure_timestamps,
        start=1,
    ):
        failure_transitions.append(
            service.observe(
                create_observation(
                    provider="openai",
                    model="gpt-5-mini",
                    capability="conversation",
                    status=ProviderHealthStatus.DEGRADED,
                    observed_at=timestamp,
                    consecutive_failures=index,
                )
            )
        )

    assert (
        failure_transitions[-1].transition
        == ProviderHealthTransition.UNAVAILABLE
    )

    recovery_transition = service.observe(
        create_observation(
            provider="openai",
            model="gpt-5-mini",
            capability="conversation",
            status=ProviderHealthStatus.HEALTHY,
            observed_at=recovery_time,
            consecutive_successes=1,
        )
    )

    assert (
        recovery_transition.transition
        == ProviderHealthTransition.RECOVERED
    )

    result = service.get_provider_health(
        "openai"
    )

    assert result is not None

    assert (
        result.status
        == "healthy"
    )

    assert (
        result.consecutive_failures
        == 0
    )

    assert (
        result.consecutive_successes
        == 1
    )

    assert (
        result.last_success_at
        == recovery_time
    )

    assert (
        result.unavailable_since
        is None
    )

    assert (
        result.degraded_since
        is None
    )

- reject mismatched provider state;
- remain free of health calculation logic.

def create_state() -> ProviderHealthState:
    """
    Create deterministic calculated provider state.
    """

    checked_at = datetime(
        2026,
        8,
        15,
        10,
        30,
        tzinfo=timezone.utc,
    )

    degraded_since = datetime(
        2026,
        8,
        15,
        10,
        29,
        tzinfo=timezone.utc,
    )

    last_failure_at = datetime(
        2026,
        8,
        15,
        10,
        30,
        tzinfo=timezone.utc,
    )

    return ProviderHealthState(
        provider="openai",
        status=ProviderHealthStatus.DEGRADED,
        checked_at=checked_at,
        consecutive_failures=2,
        consecutive_successes=0,
