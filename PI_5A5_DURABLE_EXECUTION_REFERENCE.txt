"""WS-22.3.2 durable execution contracts.

The application layer depends on these contracts, not on SQLAlchemy or a
specific database. Infrastructure supplies the durable repository.
"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from enum import Enum
from uuid import UUID


class DurableExecutionStatus(str, Enum):
    STARTED = "started"
    SUCCEEDED = "succeeded"
    FAILED = "failed"


@dataclass(frozen=True, slots=True)
class PublicationExecutionRecord:
    publication_id: UUID
    idempotency_key: str
    correlation_id: str
    attempt: int
    status: DurableExecutionStatus
    provider_message_id: str | None = None
    error_code: str | None = None
    created_at: datetime | None = None
    updated_at: datetime | None = None


class PublicationExecutionRepository:
    """Port for durable execution state.

    Implementations must persist records transactionally using the
    application's existing persistence/session infrastructure.
    """

    async def get_by_idempotency_key(
        self, idempotency_key: str
    ) -> PublicationExecutionRecord | None:
        raise NotImplementedError

    async def create_started(
        self,
        *,
        publication_id: UUID,
        idempotency_key: str,
        correlation_id: str,
    ) -> PublicationExecutionRecord:
        raise NotImplementedError

    async def mark_succeeded(
        self,
        *,
        idempotency_key: str,
        provider_message_id: str | None,
    ) -> PublicationExecutionRecord:
        raise NotImplementedError

    async def mark_failed(
        self,
        *,
        idempotency_key: str,
        error_code: str,
    ) -> PublicationExecutionRecord:
        raise NotImplementedError

"""WS-22.3.2 durable execution orchestration."""

from __future__ import annotations

from app.application.announcements.execution.contracts import (
    AnnouncementPublisher,
    PublicationExecutionRequest,
    PublicationExecutionResult,
    PublicationExecutionStatus,
)
from app.application.announcements.execution.durable_contracts import (
    DurableExecutionStatus,
    PublicationExecutionRepository,
)


class DurablePublicationExecutionService:
    """Retry-safe execution service backed by durable idempotency state.

    A completed idempotency key is never sent again.

    A STARTED record is treated as an interrupted attempt and is retryable.

    A FAILED record is also retryable. Before a retry is executed, the
    repository is responsible for incrementing the durable attempt counter
    and transitioning the record back to STARTED.

    The repository implementation must make creation atomic against a
    unique idempotency key.
    """

    def __init__(
        self,
        *,
        publisher: AnnouncementPublisher,
        execution_repository: PublicationExecutionRepository,
        audit,
    ) -> None:
        if publisher is None:
            raise ValueError("publisher cannot be None.")

        if execution_repository is None:
            raise ValueError("execution_repository cannot be None.")

        if audit is None:
            raise ValueError("audit cannot be None.")

        self._publisher = publisher
        self._repository = execution_repository
        self._audit = audit

    async def execute(
        self,
        *,
        publication,
        request: PublicationExecutionRequest,
    ) -> PublicationExecutionResult:
        existing = await self._repository.get_by_idempotency_key(
            request.idempotency_key
        )

        # A successful execution is terminal for the idempotency key.
        # Never call the external provider again.
        if (
            existing is not None
            and existing.status is DurableExecutionStatus.SUCCEEDED
        ):
            return PublicationExecutionResult(
                publication_id=existing.publication_id,
                status=PublicationExecutionStatus.IDEMPOTENT_REPLAY,
                idempotency_key=existing.idempotency_key,
                correlation_id=request.correlation_id,
                provider_message_id=existing.provider_message_id,
            )

        self._assert_approved(publication)

        # The repository owns the atomic unique-key operation. If another
        # worker already created the same key, re-read and honour its state.
        if existing is None:
            try:
                record = await self._repository.create_started(
                    publication_id=publication.id,
                    idempotency_key=request.idempotency_key,
                    correlation_id=request.correlation_id,
                )
            except Exception:
                record = await self._repository.get_by_idempotency_key(
                    request.idempotency_key
                )

                if record is None:
                    raise

        elif existing.status is DurableExecutionStatus.FAILED:
            # A FAILED execution is retryable. Reuse the same durable
            # idempotency record and increment its attempt number rather
            # than creating a second row.
            record = await self._repository.start_retry(
                idempotency_key=request.idempotency_key,
            )

        else:
            # STARTED represents an interrupted/incomplete attempt.
            # Preserve the existing durable attempt number and retry it.
            record = existing

        publication.begin_publish()

        try:
            provider_message_id = await self._publisher.publish(
                publication_id=publication.id,
                title=publication.title,
                content=publication.content,
                correlation_id=request.correlation_id,
                idempotency_key=request.idempotency_key,
            )
        except Exception as exc:
            error_code = self._error_code(exc)

            publication.mark_failure(error_code)

            await self._repository.mark_failed(
                idempotency_key=request.idempotency_key,
                error_code=error_code,
            )

            await self._audit.record_failure(
                publication_id=publication.id,
                actor=request.actor,
                correlation_id=request.correlation_id,
                error_code=error_code,
            )

            return PublicationExecutionResult(
                publication_id=publication.id,
                status=PublicationExecutionStatus.FAILED,
                idempotency_key=request.idempotency_key,
                correlation_id=request.correlation_id,
                error_code=error_code,
            )

        publication.publish_success(provider_message_id)

        await self._repository.mark_succeeded(
            idempotency_key=request.idempotency_key,
            provider_message_id=provider_message_id,
        )

        await self._audit.record_success(
            publication_id=publication.id,
            actor=request.actor,
            correlation_id=request.correlation_id,
            provider_message_id=provider_message_id,
        )

        return PublicationExecutionResult(
            publication_id=publication.id,
            status=PublicationExecutionStatus.SUCCEEDED,
            idempotency_key=request.idempotency_key,
            correlation_id=request.correlation_id,
            provider_message_id=provider_message_id,
        )

    @staticmethod
    def _assert_approved(publication) -> None:
        value = getattr(publication.status, "value", publication.status)

        if value != "approved":
            raise ValueError(
                "Only approved publications may be executed."
            )

    @staticmethod
    def _error_code(exc: Exception) -> str:
        explicit = getattr(exc, "error_code", None)

        return (
            explicit.strip()[:100]
            if isinstance(explicit, str) and explicit.strip()
            else type(exc).__name__.upper()[:100]
        )

"""SQLAlchemy durable execution repository for WS-22.

Uses the canonical EasyChatBot SQLAlchemy Base. This module deliberately does
not create a second DeclarativeBase/metadata registry.
"""

from __future__ import annotations

from datetime import datetime, timezone
from uuid import UUID

from sqlalchemy import DateTime, Integer, String, select
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker
from sqlalchemy.orm import Mapped, mapped_column

from app.application.announcements.execution.durable_contracts import (
    DurableExecutionStatus,
    PublicationExecutionRecord,
    PublicationExecutionRepository,
)
from app.infrastructure.persistence.database import Base


class PublicationExecutionRecordModel(Base):
    """Durable operational state for a WS-22 publication execution."""

    __tablename__ = "announcement_publication_executions"

    publication_id: Mapped[str] = mapped_column(String(36), nullable=False)
    idempotency_key: Mapped[str] = mapped_column(
        String(255), primary_key=True
    )
    correlation_id: Mapped[str] = mapped_column(
        String(255),
        nullable=False,
    )
    attempt: Mapped[int] = mapped_column(
        Integer,
        nullable=False,
        default=1,
    )
    status: Mapped[str] = mapped_column(
        String(32),
        nullable=False,
    )
    provider_message_id: Mapped[str | None] = mapped_column(
        String(255),
        nullable=True,
    )
    error_code: Mapped[str | None] = mapped_column(
        String(100),
        nullable=True,
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
    )


class SqlAlchemyPublicationExecutionRepository(
    PublicationExecutionRepository
):
    """Durable repository using the existing injected async session factory."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    async def get_by_idempotency_key(
        self,
        idempotency_key: str,
    ) -> PublicationExecutionRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )
            model = result.scalar_one_or_none()

            return self._to_domain(model) if model else None

    async def create_started(
        self,
        *,
        publication_id,
        idempotency_key,
        correlation_id,
    ) -> PublicationExecutionRecord:
        """Create the first durable execution attempt."""

        now = datetime.now(timezone.utc)

        model = PublicationExecutionRecordModel(
            publication_id=str(publication_id),
            idempotency_key=idempotency_key,
            correlation_id=correlation_id,
            attempt=1,
            status=DurableExecutionStatus.STARTED.value,
            created_at=now,
            updated_at=now,
        )

        async with self._session_factory() as session:
            session.add(model)
            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def mark_succeeded(
        self,
        *,
        idempotency_key,
        provider_message_id,
    ) -> PublicationExecutionRecord:
        """Persist successful completion without changing the attempt count."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.SUCCEEDED.value,
            provider_message_id=provider_message_id,
            error_code=None,
        )

    async def mark_failed(
        self,
        *,
        idempotency_key,
        error_code,
    ) -> PublicationExecutionRecord:
        """Persist failure while retaining the current attempt number."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.FAILED.value,
            error_code=error_code,
        )

    async def start_retry(
        self,
        *,
        idempotency_key: str,
    ) -> PublicationExecutionRecord:
        """Increment the durable attempt counter for a retry.

        A retry is valid only for an existing FAILED execution. The same
        idempotency record is retained; no second database row is created.
        """

        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            if model.status != DurableExecutionStatus.FAILED.value:
                raise ValueError(
                    "Only failed executions can be retried."
                )

            model.attempt += 1
            model.status = DurableExecutionStatus.STARTED.value
            model.error_code = None
            model.provider_message_id = None
            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def _update(
        self,
        idempotency_key,
        **values,
    ) -> PublicationExecutionRecord:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            for key, value in values.items():
                setattr(model, key, value)

            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    @staticmethod
    def _to_domain(
        model: PublicationExecutionRecordModel,
    ) -> PublicationExecutionRecord:
        return PublicationExecutionRecord(
            publication_id=UUID(model.publication_id),
            idempotency_key=model.idempotency_key,
            correlation_id=model.correlation_id,
            attempt=model.attempt,
            status=DurableExecutionStatus(model.status),
            provider_message_id=model.provider_message_id,
            error_code=model.error_code,
            created_at=model.created_at,
            updated_at=model.updated_at,
        )

def _require_database() -> None:
    load_dotenv()

    if not os.getenv("DATABASE_URL"):
        pytest.skip(
            "DATABASE_URL is not configured for PostgreSQL integration tests."
        )


class RecordingPublisher:
    def __init__(self, result: str) -> None:
        self.result = result
        self.calls: list[dict] = []

    async def publish(self, **kwargs) -> str:
        self.calls.append(kwargs)
        return self.result


@dataclass
class ApprovedPublication:
    id: UUID
    title: str = "WS-22.3.5.6 test"
    content: str = "Terminal idempotency test."
    status: str = "approved"

    def begin_publish(self) -> None:
        pass

    def publish_success(
        self,
        provider_message_id: str | None = None,
    ) -> None:
        self.status = "published"

    def mark_failure(self, error_code: str) -> None:
        self.status = "failed"


async def _cleanup(
    composition,
    *,
    idempotency_key: str,
    publication_id: UUID,
) -> None:
    async with composition.session_factory() as session:
        await session.execute(
            delete(PublicationAuditEventModel).where(
                PublicationAuditEventModel.publication_id
                == str(publication_id)
            )
        )

        await session.execute(
            delete(PublicationExecutionRecordModel).where(
                PublicationExecutionRecordModel.idempotency_key
                == idempotency_key
            )
        )

        await session.commit()


@pytest.mark.asyncio
async def test_success_becomes_terminal_idempotent_replay() -> None:
    _require_database()

    publication_id = uuid4()
    idempotency_key = f"ws22-3-5-6-{uuid4()}"
    correlation_id = f"corr-{uuid4()}"

    request = PublicationExecutionRequest(
        publication_id=publication_id,
        idempotency_key=idempotency_key,
        actor="ws22-3-5-6-test",
        correlation_id=correlation_id,
    )

    publisher_1 = RecordingPublisher(
        result="provider-terminal-message",
    )

    composition_1 = create_durable_publication_execution_composition(
        publisher=publisher_1,
    )

    try:
        first = await composition_1.service.execute(
            publication=ApprovedPublication(id=publication_id),
            request=request,
        )

        assert first.status is PublicationExecutionStatus.SUCCEEDED
        assert first.provider_message_id == "provider-terminal-message"
        assert len(publisher_1.calls) == 1

        async with composition_1.session_factory() as session:
            execution = await session.scalar(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            assert execution is not None
            assert execution.status == "succeeded"
            assert execution.attempt == 1
            assert execution.provider_message_id == (
                "provider-terminal-message"
            )
            assert execution.error_code is None

    finally:
        await composition_1.dispose()

    # Simulate a real application restart.
    publisher_2 = RecordingPublisher(
        result="MUST-NOT-BE-SENT",
    )

    composition_2 = create_durable_publication_execution_composition(
        publisher=publisher_2,
    )

    try:
        replay = await composition_2.service.execute(
            publication=ApprovedPublication(id=publication_id),
            request=request,
        )

        assert replay.status is PublicationExecutionStatus.IDEMPOTENT_REPLAY
        assert replay.provider_message_id == "provider-terminal-message"

        # Terminal success means the external provider must never be called
        # again for the same idempotency key.
        assert publisher_2.calls == []

        async with composition_2.session_factory() as session:
            execution = await session.scalar(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            assert execution is not None
            assert execution.status == "succeeded"
            assert execution.attempt == 1
            assert execution.provider_message_id == (
                "provider-terminal-message"
            )

            audit_events = (
                await session.scalars(
                    select(PublicationAuditEventModel)
                    .where(
                        PublicationAuditEventModel.publication_id
                        == str(publication_id)
                    )
                    .order_by(PublicationAuditEventModel.id)
                )
            ).all()

            # Replay must not create another publication audit event.
            assert len(audit_events) == 1
            assert audit_events[0].event_type == "published"

    finally:
        await _cleanup(
            composition_2,
            idempotency_key=idempotency_key,
            publication_id=publication_id,
        )
        await composition_2.dispose()
