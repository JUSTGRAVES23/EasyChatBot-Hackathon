"""
===============================================================================
AI ENGINEERING PLATFORM
===============================================================================

Folder:
runtime/contracts

File:
capability_definition.py

Layer:
Domain / Runtime Contract Layer

Purpose:
Defines the canonical contract describing an AI Runtime capability.

A CapabilityDefinition represents a reusable AI capability exposed by the
Runtime workforce.

Architecture:

Agent

    |

    v

CapabilityDefinition

    |

    v

CapabilityRegistry

    |

    v

Runtime Execution


Responsibilities:

- Define capability identity.
- Describe capability purpose.
- Define ownership metadata.
- Provide governance information.
- Support capability discovery.


This module does NOT:

- Execute capabilities.
- Resolve agents.
- Generate prompts.
- Call providers.
- Manage memory.
- Contain business logic.


Design Principles:

- Immutable contracts.
- Runtime ownership.
- Provider independence.
- Explicit boundaries.
- Testability.
- Governance readiness.


Author:
EasyChatBot AI Runtime
===============================================================================
"""

from __future__ import annotations


from dataclasses import dataclass, field


from typing import Any



@dataclass(
    frozen=True,
    slots=True,
)
class CapabilityDefinition:
    """
    Canonical definition of an AI Runtime capability.

    A capability represents a specific AI workforce function.

    Examples:

    - project_management
    - architecture_review
    - security_analysis
    - language_tutoring


    The Runtime uses this contract to understand:

    - What capability exists.
    - Who owns it.
    - What purpose it serves.
    - Whether it is enabled.
    """



    # ==========================================================================
    # Identity
    # ==========================================================================


    capability_id: str
    """
    Unique Runtime capability identifier.

    Example:

        project_management
    """


    name: str
    """
    Human-readable capability name.

    Example:

        Project Management Analysis
    """


    description: str
    """
    Capability purpose description.
    """



    # ==========================================================================
    # Ownership / Governance
    # ==========================================================================


    owner: str = "Engineering Council"
    """
    Capability owner.

    Used for governance and audit purposes.
    """


    version: str = "1.0"
    """
    Capability definition version.
    """


    enabled: bool = True
    """
    Indicates whether Runtime may use this capability.
    """



    # ==========================================================================
    # Runtime Mapping
    # ==========================================================================


    agent_id: str = ""
    """
    Associated Runtime agent identifier.

    Example:

        project_manager
    """


    tags: tuple[str, ...] = field(
        default_factory=tuple,
    )
    """
    Capability classification tags.
    """


    metadata: dict[str, Any] = field(
        default_factory=dict,
    )
    """
    Additional governance metadata.
    """



    # ==========================================================================
    # Validation
    # ==========================================================================


    def validate(
        self,
    ) -> None:
        """
        Validate capability definition.

        Raises
        ------

        ValueError:
            If capability definition is invalid.
        """

        if not self.capability_id.strip():

            raise ValueError(
                "CapabilityDefinition requires capability_id."
            )


        if not self.name.strip():

            raise ValueError(
                "CapabilityDefinition requires name."
            )


        if not self.description.strip():

            raise ValueError(
                "CapabilityDefinition requires description."
            )


        if not self.version.strip():

            raise ValueError(
                "CapabilityDefinition requires version."
            )


        if self.agent_id and not self.agent_id.strip():

            raise ValueError(
                "agent_id cannot contain only whitespace."
            )



    # ==========================================================================
    # Helpers
    # ==========================================================================


    def is_available(
        self,
    ) -> bool:
        """
        Determine whether capability can be used by Runtime.
        """

        return self.enabled



    def has_agent_binding(
        self,
    ) -> bool:
        """
        Check whether capability is linked to an agent.
        """

        return bool(
            self.agent_id
        )



    def has_tag(
        self,
        tag: str,
    ) -> bool:
        """
        Check whether capability contains a tag.
        """

        if not isinstance(
            tag,
            str,
        ):

            return False


        return tag in self.tags

"""
===============================================================================
AI ENGINEERING PLATFORM
===============================================================================

Folder:
runtime/contracts

File:
capability_execution_request.py

Layer:
Capability Runtime Contract Layer


Purpose:

Defines the canonical request entering the Capability Runtime.

CapabilityExecutionRequest represents the intention to execute
a Runtime capability.

It is created after transport mapping and before capability
resolution begins.


Architecture Boundary:


External Request

        |

        v

DTO

        |

        v

CapabilityExecutionRequest

        |

        v

CapabilityResolver

        |

        v

Capability Execution Runtime


Design Separation:


CapabilityExecutionRequest owns:

- Capability intention.
- Capability identifier.
- User objective.
- Capability parameters.
- Execution preferences.
- Request metadata.


RuntimeContext owns:

- Execution lifecycle.
- Pipeline state.
- Memory.
- Provider state.
- Runtime events.


CapabilityExecutionRequest intentionally contains no:

- Provider implementations.
- Prompt generation.
- Agent execution logic.
- Infrastructure dependencies.


Design Principles:

- Framework independence.
- Provider independence.
- Immutable contract.
- Capability-first architecture.
- Testability.
- Long-term maintainability.


Author:
EasyChatBot AI Runtime
===============================================================================
"""

from __future__ import annotations


from dataclasses import dataclass, field


from datetime import datetime, timezone


from typing import Any


from uuid import uuid4





@dataclass(
    frozen=True,
    slots=True,
)
class CapabilityExecutionRequest:
    """
    Canonical request entering the Capability Runtime.

    Represents an intention to execute
    a registered Runtime capability.


    Responsibilities:

    - Identify capability.
    - Carry user objective.
    - Carry capability parameters.
    - Carry execution preferences.
    - Carry request metadata.


    Does not contain:

    - Runtime lifecycle state.
    - Memory.
    - Provider logic.
    - Agent implementation.
    """



    # ==========================================================================
    # Identity
    # ==========================================================================


    execution_id: str = field(
        default_factory=lambda: str(uuid4())
    )


    created_at: datetime = field(
        default_factory=lambda: datetime.now(
            timezone.utc
        )
    )



    # ==========================================================================
    # Capability Target
    # ==========================================================================


    capability_id: str = ""



    # ==========================================================================
    # Optional Agent Binding
    # ==========================================================================
    #
    # Normally resolved later.
    #
    # Capability should not require an agent.
    #
    # Example:
    #
    # project_management
    #
    # may resolve to:
    #
    # project_manager agent
    #
    # ==========================================================================


    agent_id: str | None = None



    # ==========================================================================
    # User Intent
    # ==========================================================================


    user_request: str = ""



    # ==========================================================================
    # Capability Parameters
    # ==========================================================================


    parameters: dict[str, Any] = field(
        default_factory=dict
    )



    # ==========================================================================
    # Request Metadata
    # ==========================================================================


    metadata: dict[str, Any] = field(
        default_factory=dict
    )



    # ==========================================================================
    # Runtime Preferences
    # ==========================================================================


    priority: str = "normal"



    # ==========================================================================
    # Validation
    # ==========================================================================


    def validate(
        self,
    ) -> None:
        """
        Validate capability execution request.

        Raises
        ------
        ValueError
            Invalid request state.
        """


        if not self.capability_id.strip():

            raise ValueError(
                "CapabilityExecutionRequest requires capability_id."
            )



        if not self.user_request.strip():

            raise ValueError(
                "CapabilityExecutionRequest requires user_request."
            )



        allowed_priorities = {

            "low",

            "normal",

            "high",

            "critical",

        }



        if self.priority not in allowed_priorities:

            raise ValueError(
                (
                    f"Invalid priority '{self.priority}'. "
                    f"Allowed values: {allowed_priorities}"
                )
            )



    # ==========================================================================
    # Helpers
    # ==========================================================================


    def with_metadata(
        self,
        key: str,
        value: Any,
    ) -> "CapabilityExecutionRequest":
        """
        Return a new request with metadata added.

        The contract is immutable.
        """


        updated_metadata = dict(
            self.metadata
        )


        updated_metadata[key] = value



        return CapabilityExecutionRequest(

            execution_id=self.execution_id,

            created_at=self.created_at,

            capability_id=self.capability_id,

            agent_id=self.agent_id,

            user_request=self.user_request,

            parameters=dict(
                self.parameters
            ),

            metadata=updated_metadata,

            priority=self.priority,

        )



    def get_parameter(
        self,
        key: str,
        default: Any = None,
    ) -> Any:
        """
        Retrieve capability parameter.
        """


        return self.parameters.get(
            key,
            default,
        )



    def has_agent_binding(
        self,
    ) -> bool:
        """
        Indicates whether an agent has already been assigned.
        """


        return bool(
            self.agent_id
        )

"""
===============================================================================
AI ENGINEERING PLATFORM
===============================================================================

Folder:
runtime/capabilities/services

File:
capability_executor.py

Layer:
Runtime Capability Service Layer


Purpose:

Provides the controlled Runtime boundary for executing
registered AI capabilities.


Architecture:


CapabilityExecutionRequest

        |

        v

CapabilityExecutor

        |

        +----------------------+

        |                      |

        v                      v

CapabilityResolver     AgentExecutionService


        |

        v


AgentExecutionResponse



Responsibilities:

- Validate capability execution requests.
- Resolve capability definitions.
- Coordinate capability execution.
- Delegate reasoning execution.
- Return normalized execution responses.


This service does NOT:

- Execute providers.
- Build prompts.
- Select models.
- Manage memory.
- Create agents.
- Contain capability business logic.


Design Principles:

- Clean Architecture.
- Dependency Injection.
- Capability-first design.
- Runtime ownership.
- Explicit boundaries.
- Provider independence.


Author:
EasyChatBot AI Runtime
===============================================================================
"""

from __future__ import annotations


import logging


from runtime.capabilities.services.capability_resolver import (
    CapabilityResolver,
)


from runtime.contracts.capability_execution_request import (
    CapabilityExecutionRequest,
)


from runtime.core.runtime_context import (
    RuntimeContext,
)


from runtime.agents.services.agent_execution_service import (
    AgentExecutionService,
)


from runtime.contracts.agent_execution_response import (
    AgentExecutionResponse,
)



logger = logging.getLogger(__name__)





class CapabilityExecutor:
    """
    Runtime service responsible for capability execution coordination.

    This class provides the application boundary between:

        Capability Runtime

    and

        Agent Runtime


    It owns:

    - Capability validation.
    - Capability resolution.
    - Execution delegation.


    It does not own:

    - AI reasoning.
    - Providers.
    - Prompts.
    - Memory.
    """



    def __init__(
        self,
        *,
        resolver: CapabilityResolver,
        agent_execution_service: AgentExecutionService,
    ) -> None:
        """
        Initialise CapabilityExecutor.

        Parameters
        ----------
        resolver:
            Capability resolution service.

        agent_execution_service:
            Runtime agent execution boundary.
        """


        if resolver is None:

            raise ValueError(
                "CapabilityResolver cannot be None."
            )



        if agent_execution_service is None:

            raise ValueError(
                "AgentExecutionService cannot be None."
            )



        self._resolver = resolver

        self._agent_execution_service = (
            agent_execution_service
        )



        logger.info(
            "CapabilityExecutor initialised."
        )



    async def execute(
        self,
        *,
        request: CapabilityExecutionRequest,
        context: RuntimeContext,
    ) -> AgentExecutionResponse:
        """
        Execute Runtime capability.

        Parameters
        ----------
        request:
            Capability execution request.

        context:
            Runtime execution context.


        Returns
        -------
        AgentExecutionResponse
            Normalized capability execution response.
        """


        if request is None:

            raise ValueError(
                "CapabilityExecutionRequest cannot be None."
            )



        if context is None:

            raise ValueError(
                "RuntimeContext cannot be None."
            )



        request.validate()



        logger.info(
            (
                "Executing capability. "
                "capability=%s execution_id=%s"
            ),
            request.capability_id,
            context.execution_id,
        )



        capability = (
            self._resolver.resolve(
                request.capability_id,
            )
        )



        if not capability.enabled:

            raise RuntimeError(
                (
                    f"Capability '{request.capability_id}' "
                    "is disabled."
                )
            )



        agent_id = (
            request.agent_id
            or capability.agent_id
        )



        if not agent_id:

            raise RuntimeError(
                (
                    f"Capability '{capability.capability_id}' "
                    "does not define an execution agent."
                )
            )



        logger.info(
            (
                "Capability resolved. "
                "capability=%s agent=%s"
            ),
            capability.capability_id,
            agent_id,
        )



        response = await (
            self._agent_execution_service.execute(
                request=self._create_agent_request(
                    request=request,
                    capability=capability,
                    agent_id=agent_id,
                ),
                context=context,
            )
        )



        logger.info(
            (
                "Capability execution completed. "
                "capability=%s execution_id=%s"
            ),
            capability.capability_id,
            context.execution_id,
        )



        return response



    def _create_agent_request(
        self,
        *,
        request: CapabilityExecutionRequest,
        capability,
        agent_id: str,
    ):
        """
        Create AgentExecutionRequest from capability request.

        Keeps capability contracts separate from agent contracts.
        """


        from runtime.contracts.agent_execution_request import (
            AgentExecutionRequest,
        )



        return AgentExecutionRequest(

            agent_id=agent_id,

            capability=capability.capability_id,

            parameters=request.parameters,

            metadata={

                "capability_id": (
                    capability.capability_id
                ),

                "capability_name": (
                    capability.name
                ),

            },

        )

"""
================================================================================
File: runtime/capabilities/capability_resolver.py

Purpose:
Resolves the most appropriate AI Agent for a requested capability.

The CapabilityResolver encapsulates the runtime selection strategy,
separating capability storage (CapabilityRegistry) from agent selection.

The resolver is responsible only for determining which agent should
execute a request. It performs no execution, routing or provider
selection.

Author:
EasyChatBot AI Runtime
================================================================================
"""

from __future__ import annotations

import logging

from runtime.capabilities.capability_registry import CapabilityRegistry
from runtime.contracts.agent import Agent
from runtime.contracts.execution_context import ExecutionContext

logger = logging.getLogger(__name__)


class CapabilityResolver:
    """
    Resolves the most appropriate agent for a requested capability.
    """

    def __init__(
        self,
        registry: CapabilityRegistry,
    ) -> None:
        """
        Initialise the capability resolver.

        Parameters
        ----------
        registry:
            Registry containing all registered runtime agents.
        """

        self._registry = registry

    def resolve(
        self,
        context: ExecutionContext,
    ) -> Agent:
        """
        Resolve the appropriate agent for the execution request.

        Parameters
        ----------
        context:
            Immutable execution context.

        Returns
        -------
        Agent
            Selected runtime agent.

        Raises
        ------
        ValueError
            If no capability has been specified.

        LookupError
            If no agent is registered for the requested capability.
        """

        capability = context.capability.strip()

        if not capability:
            raise ValueError(
                "ExecutionContext.capability cannot be empty."
            )

        agent = self._registry.get_agent(capability)

        if agent is None:
            raise LookupError(
                f"No agent registered for capability '{capability}'."
            )

        logger.debug(
            (
                "Resolved capability '%s' to agent '%s' "
                "(version=%s)."
            ),
            capability,
            agent.descriptor.id,
            agent.descriptor.version,
        )

        return agent
