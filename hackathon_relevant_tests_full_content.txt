@pytest.mark.asyncio
async def test_companion_passes_research_to_conversation(
    tmp_path,
):

    conversation_agent = (
        FakeConversationAgent()
    )

    service = CompanionService(
        character=create_character(),
        conversation_agent=(
            conversation_agent
        ),
        research_agent=(
            FakeResearchAgent()
        ),
        quality_agent=(
            FakeQualityAgent()
        ),
        memory_service=(
            create_memory_service(
                tmp_path
            )
        ),
    )

    context = ExecutionContext(
        user_id="demo-user",
        request=(
            "Explain the latest "
            "European AI regulation"
        ),
    )

    result = await service.interact(
        context,
        use_research=True,
    )

    assert result["character"] == (
        "Yeti"
    )

    assert (
        result["research"]
        is not None
    )

    assert isinstance(
        result["research"],
        IntelligenceContext,
    )

    received_context = (
        conversation_agent.received_context
    )

    assert (
        received_context is not None
    )

    assert (
        received_context
        is not context
    )

    assert (
        received_context.intelligence_context
        is not None
    )

    assert (
        received_context.intelligence_context.query
        == context.request
    )

    assert (
        received_context.intelligence_context.source_count
        == 1
    )

    assert (
        result["conversation"][
            "intelligence_context"
        ]
        is received_context.intelligence_context
    )

    assert (
        context.intelligence_context
        is None
    )

def test_memory_service_deletes_memory(
    tmp_path,
):

    service = MemoryService(
        store=MemoryStore(
            tmp_path / "memory.db"
        )
    )

    memory = service.remember(
        user_id="user-001",
        memory_type=MemoryType.SAVED_ITEM,
        content="Travel backpack",
    )

    assert service.delete(
        memory.memory_id
    ) is True

    assert (
        service.get(memory.memory_id)
        is None
    )
