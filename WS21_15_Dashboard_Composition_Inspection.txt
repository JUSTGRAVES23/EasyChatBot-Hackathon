"""
EasyChatBot AI Operations Center dashboard API.

Provides the application-facing read-only boundary for
the Operations Center dashboard.

Architecture:

Operations Center Dashboard Service
                |
                v
      OperationsCenterDashboardFacade
                |
                v
        OperationsCenterDashboardAPI
                |
                v
          HTTP Router

Responsibilities:

- expose the dashboard application contract;
- delegate dashboard generation to the facade;
- keep HTTP concerns outside the application layer;
- preserve compatibility with explicit provider input;
- provide a stable boundary for future Operations Center
  integrations.

Does NOT:

- execute providers;
- execute agents;
- modify Runtime routing;
- access repositories directly;
- approve escalations;
- send notifications;
- perform autonomous remediation.

The dashboard is read-only.
"""

from __future__ import annotations

from typing import Any

from app.application.provider_management.operations_center_dashboard_facade import (
    OperationsCenterDashboardFacade,
)

from app.application.provider_management.operations_center_dashboard_service import (
    ActiveProvider,
)


_UNSET = object()


class OperationsCenterDashboardAPI:
    """
    Application-facing API for the Operations Center dashboard.

    The API deliberately contains no business logic.
    It delegates dashboard generation to the facade.

    For compatibility and deterministic testing, callers may
    explicitly supply active providers.

    Calling get_dashboard() without provider input preserves
    the facade's normal provider-resolution behaviour.
    """

    def __init__(
        self,
        *,
        facade: OperationsCenterDashboardFacade,
    ) -> None:
        """
        Initialise the dashboard API.
        """

        if facade is None:
            raise ValueError(
                "Operations Center dashboard facade "
                "cannot be None."
            )

        self._facade = facade

    def get_dashboard(
        self,
        *,
        active_providers: (
            tuple[
                ActiveProvider,
                ...
            ]
            | None
            | object
        ) = _UNSET,
    ) -> dict[str, Any]:
        """
        Return the current Operations Center dashboard.

        Resolution rules:

        - omitted:
            delegate to the facade without explicit provider input;

        - explicit tuple:
            delegate using the supplied providers;

        - explicit None:
            reject invalid provider input.

        The API performs no dashboard business logic.
        """

        if active_providers is None:
            raise ValueError(
                "Active providers cannot be None."
            )

        if active_providers is _UNSET:
            return self._facade.get_dashboard()

        return self._facade.get_dashboard(
            active_providers=active_providers,
        )

"""EasyChatBot AI Operations Center composition root.

The same ProviderHealthHistoryRepository instance is shared by:
- provider-health observation ingestion;
- provider intelligence analytics.

This keeps dashboard intelligence aligned with the observed
provider-health lifecycle.
"""

from __future__ import annotations

from dataclasses import dataclass

@dataclass(frozen=True)
class OperationsCenterComposition:
    """Composed Operations Center application graph."""

    dashboard_api: OperationsCenterDashboardAPI
    health_application_service: ProviderHealthApplicationService
    operational_event_service: OperationalEventService
    operational_events_api: OperationsCenterOperationalEventsAPI
    health_refresh_service: ProviderHealthRefreshService | None


def create_operations_center_composition() -> OperationsCenterComposition:
    """Construct the complete Operations Center application graph."""

    health_monitor = ProviderHealthMonitorService()

    # One repository is intentionally shared across health ingestion
    # and provider intelligence.
    provider_health_history_repository = ProviderHealthHistoryRepository()

    provider_intelligence_service = (
        OperationsCenterProviderIntelligenceService(
            repository=provider_health_history_repository,
        )
    )

    operational_event_repository = InMemoryOperationalEventRepository()

    operational_event_service = OperationalEventService(
        repository=operational_event_repository,
    )

    provider_operational_event_service = ProviderOperationalEventService(
        operational_event_service=operational_event_service,
    )

    operational_events_api = OperationsCenterOperationalEventsAPI(
        operational_event_service=operational_event_service,
    )

    incident_repository = InMemoryProviderIncidentRepository()

    incident_service = ProviderIncidentService(
        repository=incident_repository,
        operational_event_service=provider_operational_event_service,
    )

    # IMPORTANT: use the SAME history repository instance here.
    health_application_service = ProviderHealthApplicationService(
        health_monitor=health_monitor,
        incident_service=incident_service,
        history_repository=provider_health_history_repository,
    )

    health_refresh_service = None

    if os.getenv("OLLAMA_API_KEY"):
        ollama_probe = OllamaCloudHealthProbe()

        probe_runner = ProviderHealthProbeRunner(
            health_application_service=health_application_service,
            probes=[ollama_probe],
        )

        health_refresh_service = ProviderHealthRefreshService(
            probe_runner=probe_runner,
        )

    workflow_repository = InMemoryEscalationWorkflowRepository()

    history_service = EscalationHistoryService(
        repository=workflow_repository,
    )

    provider_service = OperationsCenterProviderService(
        health_monitor=health_monitor,
    )

    dashboard_service = OperationsCenterDashboardService(
        history_service=history_service,
        provider_service=provider_service,
        provider_intelligence_service=provider_intelligence_service,
    )

    facade = OperationsCenterDashboardFacade(
        dashboard_service=dashboard_service,
    )

    dashboard_api = OperationsCenterDashboardAPI(
        facade=facade,
    )

    return OperationsCenterComposition(
        dashboard_api=dashboard_api,
        health_application_service=health_application_service,
        operational_event_service=operational_event_service,
        operational_events_api=operational_events_api,
        health_refresh_service=health_refresh_service,
    )


def create_operations_center_dashboard_api() -> OperationsCenterDashboardAPI:
    """Backward-compatible dashboard API composition entry point."""
    return create_operations_center_composition().dashboard_api


__all__ = [
    "OperationsCenterComposition",
    "create_operations_center_composition",
    "create_operations_center_dashboard_api",
]

"""
EasyChatBot AI Operations Center dashboard service.

Builds the read-only Operations Center dashboard.

Architecture:

Provider Health
        |
Provider Service
        |
Provider Intelligence Service
        |
OperationsCenterDashboardService
        |
OperationsCenterDashboard
        |
Dashboard API


Responsibilities:

- compose dashboard information;
- aggregate escalation state;
- calculate operational confidence;
- create dashboard read models;
- include provider intelligence when available.

Does NOT:

- execute providers;
- modify Runtime routing;
- approve escalations;
- send notifications;
- perform remediation.
"""

from __future__ import annotations


from app.application.provider_management.escalation_history_service import (
    EscalationHistoryService,
)


from app.application.provider_management.operations_center_dashboard_models import (
    ActiveProvider,
    EscalationSummary,
    MonitoringStatus,
    MonitoringSummary,
    OperationsCenterDashboard,
    OperationsMetrics,
)


_UNSET = object()



class OperationsCenterDashboardService:
    """
    Application service responsible for dashboard composition.
    """

    def __init__(
        self,
        *,
        history_service: EscalationHistoryService,
        provider_service=None,
        provider_intelligence_service=None,
    ) -> None:

        if history_service is None:
            raise ValueError(
                "Escalation history service cannot be None."
            )


        self._history_service = (
            history_service
        )


        self._provider_service = (
            provider_service
        )


        self._provider_intelligence_service = (
            provider_intelligence_service
        )



    def build_dashboard(
        self,
        *,
        active_providers=_UNSET,
    ) -> OperationsCenterDashboard:
        """
        Build Operations Center dashboard.
        """


        #
        # Provider state
        #

        if active_providers is _UNSET:

            if self._provider_service is None:

                providers = ()

            else:

                providers = (
                    self._provider_service
                    .get_active_providers()
                )

        else:

            if active_providers is None:
                raise ValueError(
                    "Active providers cannot be None."
                )

            providers = active_providers



        #
        # Provider intelligence
        #

        provider_intelligence = ()


        if self._provider_intelligence_service is not None:

            provider_intelligence = (
                self
                ._provider_intelligence_service
                .get_all_provider_intelligence()
            )



        #
        # Escalation aggregation
        #

        pending_notifications = len(
            self._history_service
            .list_pending_notification()
        )


        pending_approvals = len(
            self._history_service
            .list_pending_approval()
        )


        approved = len(
            self._history_service
            .list_approved()
        )


        rejected = len(
            self._history_service
            .list_rejected()
        )


        total = (
            pending_notifications
            +
            pending_approvals
            +
            approved
            +
            rejected
        )


        active_escalations = (
            pending_notifications
            +
            pending_approvals
        )


        escalation_summary = EscalationSummary(
            active=active_escalations,

            pending=pending_approvals,

            pending_notifications=(
                pending_notifications
            ),

            approved=approved,

            rejected=rejected,

            total=total,
        )



        #
        # Monitoring state
        #

        if active_escalations > 0:

            monitoring = MonitoringSummary(
                status=MonitoringStatus.INCIDENT,
                message=(
                    "Operational attention required"
                ),
            )

        else:

            monitoring = MonitoringSummary(
                status=MonitoringStatus.HEALTHY,
                message=(
                    "Monitoring is healthy"
                ),
            )



        #
        # Confidence calculation
        #

        confidence = (
            100.0
            -
            (
                active_escalations
                *
                15.0
            )
        )



        return OperationsCenterDashboard(

            system_confidence=max(
                0.0,
                confidence,
            ),

            metrics=OperationsMetrics(),

            providers=tuple(
                providers
            ),

            escalations=escalation_summary,

            monitoring=monitoring,

            provider_intelligence=tuple(
                provider_intelligence
            ),
        )



#
# Backward compatibility exports
#

__all__ = [
    "ActiveProvider",
    "OperationsCenterDashboard",
    "OperationsCenterDashboardService",
]

"""
EasyChatBot AI Operations Center dashboard facade.

Provides a stable application-facing contract for the
Operations Center presentation/API layer.

Architecture:

Provider Health
      |
Escalations
      |
Workflow
      |
History Service
      |
Dashboard Service
      |
      v
OperationsCenterDashboardFacade
      |
      +----------------------+
      |                      |
      v                      v
Provider Mapper       Dashboard Contract
      |
      v
API / Frontend


Responsibilities:

- expose dashboard information to presentation layers;
- convert dashboard read models into stable API output;
- keep frontend/API independent from application internals;
- preserve compatibility with existing consumers.

Does NOT:

- execute providers;
- calculate intelligence;
- query repositories;
- change Runtime routing;
- approve escalations;
- reject escalations;
- send notifications;
- perform autonomous operational actions.

This is a read-only presentation boundary.
"""

from __future__ import annotations


from typing import Any


from app.application.provider_management.operations_center_dashboard_service import (
    ActiveProvider,
    OperationsCenterDashboard,
    OperationsCenterDashboardService,
)


from app.application.provider_management.operations_center_provider_intelligence_dashboard_mapper import (
    OperationsCenterProviderIntelligenceDashboardMapper,
)



_UNSET = object()



class OperationsCenterDashboardFacade:
    """
    Presentation-facing facade for the AI Operations Center.
    """


    def __init__(
        self,
        dashboard_service: OperationsCenterDashboardService,
    ) -> None:
        """
        Initialise the dashboard facade.
        """

        if dashboard_service is None:
            raise ValueError(
                "Operations Center dashboard service "
                "cannot be None."
            )


        self._dashboard_service = (
            dashboard_service
        )



    def get_dashboard(
        self,
        *,
        active_providers=(
            _UNSET
        ),
    ) -> dict[str, Any]:
        """
        Return a frontend-safe dashboard representation.
        """

        if active_providers is None:
            raise ValueError(
                "Active providers cannot be None."
            )


        if active_providers is _UNSET:

            dashboard = (
                self._dashboard_service
                .build_dashboard()
            )

        else:

            dashboard = (
                self._dashboard_service
                .build_dashboard(
                    active_providers=(
                        active_providers
                    )
                )
            )


        return self._serialize_dashboard(
            dashboard
        )



    @staticmethod
    def _serialize_dashboard(
        dashboard: OperationsCenterDashboard,
    ) -> dict[str, Any]:
        """
        Convert dashboard read model into
        stable API contract.
        """

        return {

            "system_confidence": (
                dashboard.system_confidence
            ),


            "monitoring_status": (
                dashboard.monitoring_status.value
            ),


            "active_providers": [

                OperationsCenterDashboardFacade
                ._serialize_provider(
                    provider
                )

                for provider

                in dashboard.active_providers
            ],


            "provider_intelligence": (

                OperationsCenterProviderIntelligenceDashboardMapper
                .to_dashboard_list(
                    dashboard.provider_intelligence
                )

            ),


            "active_escalations": (
                dashboard.active_escalations
            ),


            "pending_approvals": (
                dashboard.pending_approvals
            ),


            "approved_escalations": (
                dashboard.approved_escalations
            ),


            "rejected_escalations": (
                dashboard.rejected_escalations
            ),


            "pending_notifications": (
                dashboard.pending_notifications
            ),


            "total_escalations": (
                dashboard.total_escalations
            ),
        }



    @staticmethod
    def _serialize_provider(
        provider: ActiveProvider,
    ) -> dict[str, Any]:
        """
        Convert provider into existing frontend contract.

        Intentionally minimal to preserve
        backward compatibility.
        """

        return {

            "provider": (
                provider.provider
            ),

            "model": (
                provider.model
            ),

            "capabilities": list(
                provider.capabilities
            ),

            "active": (
                provider.active
            ),
        }



__all__ = [
    "OperationsCenterDashboardFacade",
]
