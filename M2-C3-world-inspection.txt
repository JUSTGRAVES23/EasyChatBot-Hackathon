"""
Tests for YAML World Repository.
"""

from pathlib import Path

import pytest

from app.infrastructure.worlds.yaml_world_repository import (
    WorldConfiguration,
    YamlWorldRepository,
)


BACKEND_APP_PATH = (
    Path(__file__)
    .resolve()
    .parents[3]
    / "app"
)


WORLDS_PATH = (
    BACKEND_APP_PATH
    / "configuration"
    / "worlds"
)


def test_load_french_world():

    repository = YamlWorldRepository(
        worlds_path=WORLDS_PATH
    )

    world = repository.get_by_id(
        "french_world"
    )

    assert world is not None

    assert world.id == "french_world"

    assert world.language_code == "fr"

    assert "yeti" in (
        world.available_characters
    )

    assert "alix" in (
        world.available_characters
    )


def test_unknown_world_returns_none():

    repository = YamlWorldRepository(
        worlds_path=WORLDS_PATH
    )

    world = repository.get_by_id(
        "unknown_world"
    )

    assert world is None

"""
Tests for World Resolver.
"""

from pathlib import Path

from app.application.uai.world.world_resolver import (
    WorldResolver,
)

from app.infrastructure.worlds.yaml_world_repository import (
    YamlWorldRepository,
)


BACKEND_PATH = (
    Path(__file__)
    .resolve()
    .parents[4]
)


WORLDS_PATH = (
    BACKEND_PATH
    / "app"
    / "configuration"
    / "worlds"
)


def test_resolve_french_world():

    repository = YamlWorldRepository(
        worlds_path=WORLDS_PATH
    )

    resolver = WorldResolver(
        world_repository=repository
    )

    context = resolver.resolve(
        "french_world"
    )

    assert context.world_id == (
        "french_world"
    )

    assert context.language_code == "fr"

    assert "yeti" in (
        context.available_characters
    )

    assert "alix" in (
        context.available_characters
    )
