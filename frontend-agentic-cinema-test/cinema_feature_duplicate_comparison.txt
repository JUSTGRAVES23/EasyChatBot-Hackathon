  getSpotifySearchUrl,

  | "profile";

const PROMPTS = [
  "What are the most important developments in the film industry this week?",
  "What streaming trends should filmmakers watch this month?",

export default function AgenticCinemaPage() {
  const [view, setView] = useState<View>("home");
  const [message, setMessage] = useState("");
  const [result, setResult] = useState<CinemaResult | null>(null);
  const [loading, setLoading] = useState(false);
  const [saved, setSaved] = useState(false);
  const [actionMessage, setActionMessage] = useState("");
  const [challengeAnswer, setChallengeAnswer] = useState("");
  const [challengeFeedback, setChallengeFeedback] = useState("");
  const [challengeLoading, setChallengeLoading] = useState(false);

  const intelligenceText = useMemo(() => {
    if (!result) {
      return "";
    }

    return (
      result.intelligence ||
      result.summary ||
      result.result ||
      ""
    );
  }, [result]);

  async function executeCinema(question: string) {
    const clean = question.trim();

    if (!clean || loading) {
      return;
    }

    setMessage(clean);
    setLoading(true);
    setResult(null);
    setActionMessage("");
    setView("results");

    try {
      const data = await runCinema(clean);
      setResult(data);
    } catch (error) {
      setResult({
        success: false,
        error:
          error instanceof Error
            ? error.message
            : "Unable to reach the cinema service.",
      });
    } finally {
      setLoading(false);
    }
  }

  async function saveToMemory() {
    if (!result || saved) {
      return;
    }

    setActionMessage("");

    try {
      const response = await runCinemaAction("save", {
        user_id: "hackathon-judge",
        question: message,
        result,
      });

      if (response.success) {
        setSaved(true);
        setActionMessage("Saved to Academy Memory.");
      } else {
        setActionMessage(
          response.error ||
            "Memory service is not configured.",
        );
      }
    } catch (error) {
      setActionMessage(
        error instanceof Error
          ? error.message
          : "Memory service could not be reached.",
      );
    }
  }

  async function evaluateChallenge() {
    if (!result?.learning_challenge) {
      return;
    }

    setChallengeLoading(true);
    setChallengeFeedback("");

    try {
      const response = await runCinemaAction(
        "challenge",
        {
          user_id: "hackathon-judge",
          answer: challengeAnswer,
          challenge: result.learning_challenge,
        },
      );

      setChallengeFeedback(
        response.feedback ||
          response.error ||
          "Challenge evaluation completed.",
      );
    } catch (error) {
      setChallengeFeedback(
        error instanceof Error
          ? error.message
          : "Evaluation service could not be reached.",
      );
    } finally {
      setChallengeLoading(false);
    }
  }

  function openSpotify() {
    const query =
      message ||
      "cinematic film soundtrack French cinema";

    window.open(
      getSpotifySearchUrl(query),
      "_blank",
      "noopener,noreferrer",
    );
  }

  return (
    <div className={styles.shell}>
      <aside className={styles.sidebar}>
        <div className={styles.brand}>
          <span className={styles.brandMark}>âœ¦</span>

          <div>

        <nav

        >
          <NavigationButton

            active={view === "home"}
            onClick={() => setView("home")}
          />

          <NavigationButton
            icon="â—‹"
            label="Chat"
            active={view === "chat"}
            onClick={() => setView("chat")}
          />

          <NavigationButton
            icon="âœ¦"
            label="Agentic Cinema"
            active
            accent
            onClick={() => setView("home")}
          />

          <NavigationButton
            icon="â™œ"
            label="Academy"
            active={view === "academy"}
            onClick={() => setView("academy")}
          />

          <NavigationButton

            onClick={() => setView("academy")}
          />

          />

          <NavigationButton
            icon="â™™"
            label="Profile"
            active={view === "profile"}
            onClick={() => setView("profile")}
          />
        </nav>

        <button
          className={styles.luminateMini}
          onClick={() => setView("luminate")}
        >

              className={styles.academyButton}

            >

        </header>

        {view === "home" && (
          <Home

            onPrompt={executeCinema}
            onRecommendations={() =>
              setView("recommendations")
            }
            onLuminate={() => setView("luminate")}

        )}

        {view === "chat" && (

            loading={loading}
            onSubmit={() => executeCinema(message)}
            onBack={() => setView("home")}
          />
        )}

        {view === "results" && (
          <Results
            question={message}
            result={result}
            loading={loading}
            intelligenceText={intelligenceText}
            saved={saved}
            actionMessage={actionMessage}
            onBack={() => setView("chat")}
            onSave={saveToMemory}
            onChallenge={() => setView("challenge")}

              setView("recommendations")
            }

          />
        )}

        {view === "challenge" && (
          <Challenge
            result={result}
            answer={challengeAnswer}
            setAnswer={setChallengeAnswer}
            feedback={challengeFeedback}
            loading={challengeLoading}
            onEvaluate={evaluateChallenge}
            onBack={() => setView("results")}
          />
        )}

        {view === "recommendations" && (
          <Recommendations

            onSpotify={openSpotify}
            onBack={() => setView("home")}
          />
        )}

        {view === "academy" && (
          <Academy

            onChallenge={() => setView("challenge")}
            onBack={() => setView("home")}
          />
        )}

        {view === "memory" && (
          <Memory

            onBack={() => setView("home")}
          />
        )}

        {view === "luminate" && (
          <Luminate
            onStart={() => setView("chat")}

              setView("recommendations")
            }
            onBack={() => setView("home")}

        {view === "profile" && (
          <Profile
            onBack={() => setView("home")}

        {actionMessage && (
          <div
            className={styles.toast}
            role="status"
          >
            <span>{actionMessage}</span>

            <button
              onClick={() => setActionMessage("")}
            >

export async function runCinema(
  message: string,
): Promise<CinemaResult> {
  const response = await fetch("/api/cinema", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      message,
      userId: USER_ID,
      language: "French",
      learnerLevel: "B2",
    }),
  });

  if (!response.ok) {
    throw new Error(
      `Cinema service returned HTTP ${response.status}.`,
    );
  }

  return (await response.json()) as CinemaResult;
}

export async function runCinemaAction(
  action: "save" | "challenge",

      payload,
    }),

    );

function normaliseDeal(raw: Deal) {
  return {
    id: raw.deal_id ?? "demo-deal",
    character: raw.character?.name ?? "Yeti",
    message: raw.character?.message ?? "Look what I found! This could be perfect for you.",
    product: raw.product?.name ?? raw.product_name ?? "Recommendation",
    tagline: raw.product?.tagline ?? "A personalised find.",
    image: raw.product?.image_url ?? raw.product_image_url ?? "",
    highlights: raw.product?.highlights ?? [],
    current: raw.pricing?.current ?? raw.current_price ?? null,
    previous: raw.pricing?.previous ?? raw.previous_price ?? null,
    discount: raw.pricing?.discount_percentage ?? raw.discount_percentage ?? null,
    currency: raw.pricing?.currency ?? raw.currency ?? "EUR",
    vendor: raw.vendor?.name ?? raw.vendor_name ?? "Demo Vendor",
    vendorUrl: raw.vendor?.url ?? raw.actions?.get_deal_url ?? "#",
    infoUrl: raw.actions?.more_information_url ?? raw.vendor?.url ?? "#",
    reason: raw.relevance_reason ?? raw.relevanceReason ?? "Recommended from your current preferences.",
    demo: raw.demo === true || raw.mode === "demo",
  };
}

export default function ShopPage() {
  const [deal, setDeal] = useState<ReturnType<typeof normaliseDeal> | null>(null);
  const [status, setStatus] = useState<"loading" | "ready" | "error">("loading");
  const [error, setError] = useState("");

  useEffect(() => {
    const controller = new AbortController();
    fetch("/api/hackathon/flash-deal", { signal: controller.signal })
      .then(async (response) => {
        const body = await response.json().catch(() => ({}));
        if (!response.ok) throw new Error(body.detail ?? `Recommendation request failed (${response.status})`);
        return body;
      })
      .then((body) => {
        setDeal(normaliseDeal(body));
        setStatus("ready");
      })
      .catch((cause) => {
        if (cause?.name === "AbortError") return;
        setError(cause instanceof Error ? cause.message : "Unable to load the recommendation.");
        setStatus("error");
      });
    return () => controller.abort();
  }, []);

  vendor?: { name?: string; url?: string };
  actions?: { get_deal_url?: string; more_information_url?: string };
  relevance_reason?: string;
  // Compatibility with an older flat demo response.
  product_name?: string;
  vendor_name?: string;
  current_price?: number;
  previous_price?: number;
  discount_percentage?: number;
  currency?: string;
  product_image_url?: string;
  relevanceReason?: string;
};

function normaliseDeal(raw: Deal) {
  return {
    id: raw.deal_id ?? "demo-deal",
    character: raw.character?.name ?? "Yeti",
    message: raw.character?.message ?? "Look what I found! This could be perfect for you.",
    product: raw.product?.name ?? raw.product_name ?? "Recommendation",
    tagline: raw.product?.tagline ?? "A personalised find.",
    image: raw.product?.image_url ?? raw.product_image_url ?? "",
    highlights: raw.product?.highlights ?? [],
    current: raw.pricing?.current ?? raw.current_price ?? null,
    previous: raw.pricing?.previous ?? raw.previous_price ?? null,
    discount: raw.pricing?.discount_percentage ?? raw.discount_percentage ?? null,
    currency: raw.pricing?.currency ?? raw.currency ?? "EUR",
    vendor: raw.vendor?.name ?? raw.vendor_name ?? "Demo Vendor",
    vendorUrl: raw.vendor?.url ?? raw.actions?.get_deal_url ?? "#",
    infoUrl: raw.actions?.more_information_url ?? raw.vendor?.url ?? "#",
    reason: raw.relevance_reason ?? raw.relevanceReason ?? "Recommended from your current preferences.",
    demo: raw.demo === true || raw.mode === "demo",
  };
}

export default function ShopPage() {

        setDeal(normaliseDeal(body));
        setStatus("ready");

        if (cause?.name === "AbortError") return;

        setStatus("error");
      });

  }, []);

  const price = useMemo(() => {

  }, [deal]);

  return (
