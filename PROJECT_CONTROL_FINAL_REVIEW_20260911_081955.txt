from __future__ import annotations

from fastapi import APIRouter, Depends, HTTPException
from fastapi.encoders import jsonable_encoder

from app.api.dependencies.authorization import require_permission
from app.domain.authorization.permissions import Permission

from ..application.service import ProjectControlService


def create_project_control_router(
    project_control: ProjectControlService,
) -> APIRouter:
    """Create the read-only Project Control Center API router.

    Authorization is enforced at the HTTP boundary before the
    ProjectControlService is executed.

    The router:
    - requires FUNCTIONALITY_CONTROL permission;
    - exposes read-only project-control snapshots;
    - does not execute the Project Manager Agent;
    - does not call LLM/provider infrastructure;
    - does not mutate project state.
    """
    if project_control is None:
        raise ValueError(
            "project_control cannot be None."
        )

    router = APIRouter(
        prefix="/project-control",
        tags=["Project Control Center"],
    )

    permission_dependency = require_permission(
        Permission.FUNCTIONALITY_CONTROL,
    )

    @router.get(
        "/projects/{project_id}",
        summary="Get project control snapshot",
        description=(
            "Return the unified read-only Project Control "
            "Center snapshot for a project."
        ),
        dependencies=[
            Depends(permission_dependency),
        ],
    )
    async def get_project_control_snapshot(
        project_id: str,
    ) -> dict[str, object]:
        try:
            snapshot = (
                await project_control.get_snapshot_async(
                    project_id,
                    persist=False,
                )
            )
        except ValueError as exc:
            raise HTTPException(
                status_code=404,
                detail=str(exc),
            ) from exc

        return jsonable_encoder(snapshot)

    return router


__all__ = [
    "create_project_control_router",
]

    project_id: str
    snapshots: tuple[ProjectControlSnapshot, ...]
    progress_delta: float | None
    blocker_delta: int | None
    risk_delta: int | None


class ProjectHistoryService:
    def __init__(self, repository: SnapshotRepository) -> None:
        self._repository = repository

    def trend(self, project_id: str, limit: int = 30) -> ProjectTrend:
        snapshots = tuple(self._repository.list_for_project(project_id, limit))
        if len(snapshots) < 2:
            return ProjectTrend(project_id, snapshots, None, None, None)

        latest, previous = snapshots[0], snapshots[1]

        progress_delta = None
        if (
            latest.overall_progress_percent is not None
            and previous.overall_progress_percent is not None
        ):
            progress_delta = round(
                latest.overall_progress_percent
                - previous.overall_progress_percent,
                1,
            )

        return ProjectTrend(
            project_id=project_id,
            snapshots=snapshots,
            progress_delta=progress_delta,
            blocker_delta=latest.open_blockers - previous.open_blockers,
            risk_delta=latest.open_risks - previous.open_risks,
        )

"""
Project Control Center application service.

Composes read-only project status sources into a unified
ProjectControlSnapshot.

The service does not own authoritative project state.
Authoritative state remains in the source systems exposed
through ProjectStatusSource adapters.

Both synchronous and asynchronous composition paths are
supported:

- get_snapshot() for synchronous sources;
- get_snapshot_async() for sources that expose asynchronous
  application services.

No event-loop bridging is performed.
"""

from __future__ import annotations

import inspect
from dataclasses import dataclass
from typing import Any

from ..domain.models import (
    ProjectControlSnapshot,
    ProjectDefinition,
    ProjectStatus,
    WorkstreamStatus,

from ..domain.ports import (

async def _maybe_await(value: Any) -> Any:
    """
    Resolve an awaitable value while preserving synchronous values.

    This helper allows the asynchronous composition path to work
    with both synchronous and asynchronous source implementations.

    It deliberately does not create or manage an event loop.
    """

    if inspect.isawaitable(value):
        return await value

    return value


@dataclass(frozen=True, slots=True)
class ProjectControlService:
    """
    Application service responsible for project-control composition.

    The service is read-oriented. It retrieves project data from
    registered ProjectStatusSource adapters and builds a unified
    ProjectControlSnapshot.

    It does not modify authoritative source systems.
    """

    registry: ProjectRegistry
    sources: tuple[ProjectStatusSource, ...]
    snapshots: SnapshotRepository

    def get_snapshot(
        self,
        project_id: str,
        *,
        persist: bool = True,
    ) -> ProjectControlSnapshot:
        """
        Build a project-control snapshot from synchronous sources.

        This method preserves the original synchronous PCD contract.

        Asynchronous sources must be consumed through
        get_snapshot_async().
        """

        project = self.registry.get(project_id)

        if project is None:
            raise ValueError(
                f"Unknown project: {project_id}"
            )

        workstreams = []
        metrics = []
        issues = []
        risks = []
        actions = []
        milestones = []
        source_health = []

        for source in self.sources:
            source_health.append(source.health())

            workstreams.extend(
                source.fetch_workstreams(project_id)
            )
            metrics.extend(
                source.fetch_metrics(project_id)
            )
            issues.extend(
                source.fetch_issues(project_id)
            )
            risks.extend(
                source.fetch_risks(project_id)
            )
            actions.extend(
                source.fetch_actions(project_id)
            )
            milestones.extend(
                source.fetch_milestones(project_id)
            )

        snapshot = self._build_snapshot(
            project=project,
            workstreams=workstreams,
            metrics=metrics,
            issues=issues,
            risks=risks,
            actions=actions,
            milestones=milestones,
            source_health=source_health,
        )

        if persist:
            self.snapshots.save(snapshot)

        return snapshot

    async def get_snapshot_async(
        self,
        project_id: str,
        *,
        persist: bool = True,
    ) -> ProjectControlSnapshot:
        """
        Build a project-control snapshot from sync/async sources.

        This is the production composition path for PCD adapters
        backed by asynchronous application services.

        The method awaits asynchronous source operations directly.
        It does not call asyncio.run(), create a nested event loop,
        or otherwise bridge asynchronous execution into synchronous
        code.
        """

        project = self.registry.get(project_id)

        if project is None:
            raise ValueError(
                f"Unknown project: {project_id}"
            )

        workstreams = []
        metrics = []
        issues = []
        risks = []
        actions = []
        milestones = []
        source_health = []

        for source in self.sources:
            source_health.append(source.health())

            workstreams.extend(
                await _maybe_await(
                    source.fetch_workstreams(project_id)
                )
            )
            metrics.extend(
                await _maybe_await(
                    source.fetch_metrics(project_id)
                )
            )
            issues.extend(
                await _maybe_await(
                    source.fetch_issues(project_id)
                )
            )
            risks.extend(
                await _maybe_await(
                    source.fetch_risks(project_id)
                )
            )
            actions.extend(
                await _maybe_await(
                    source.fetch_actions(project_id)
                )
            )
            milestones.extend(
                await _maybe_await(
                    source.fetch_milestones(project_id)
                )
            )

        snapshot = self._build_snapshot(
            project=project,
            workstreams=workstreams,
            metrics=metrics,
            issues=issues,
            risks=risks,
            actions=actions,
            milestones=milestones,
            source_health=source_health,
        )

        if persist:
            self.snapshots.save(snapshot)

        return snapshot

    @staticmethod
    def _build_snapshot(
        *,
        project: ProjectDefinition,
        workstreams: list,
        metrics: list,
        issues: list,
        risks: list,
        actions: list,
        milestones: list,
        source_health: list,
    ) -> ProjectControlSnapshot:
        """
        Deterministically compose the unified project snapshot.

        No source is mutated and no new authoritative state is
        created here.
        """

        progress_values = [
            workstream.completion_percent
            for workstream in workstreams
            if workstream.completion_percent is not None
        ]

        progress = (
            round(
                sum(progress_values) / len(progress_values),
                1,
            )
            if progress_values
            else None
        )

        blockers = (
            sum(
                1
                for workstream in workstreams
                if workstream.status is WorkstreamStatus.BLOCKED
            )
            + sum(
                1
                for issue in issues
                if issue.status.value in {
                    "open",
                    "in_progress",
                }
                and issue.severity.value in {
                    "high",
                    "critical",
                }
            )
        )

        open_risks = sum(
            1
            for risk in risks
            if risk.status.lower()
            not in {
                "closed",
                "resolved",
            }
        )

        warnings = sum(
            1
            for workstream in workstreams
            if workstream.status is WorkstreamStatus.WARNING
        )

        if blockers:
            status = ProjectStatus.BLOCKED
        elif progress is not None and progress >= 100:
            status = ProjectStatus.COMPLETE
        elif warnings or open_risks:
            status = ProjectStatus.AT_RISK
        elif workstreams:
            status = ProjectStatus.ON_TRACK
        else:
            status = ProjectStatus.UNKNOWN

        if blockers:
            readiness = "NOT READY"
        elif progress is not None and progress >= 80:
            readiness = "BETA READY"
        elif progress is not None:
            readiness = "IN PROGRESS"
        else:
            readiness = "UNKNOWN"

        current_milestone = next(
            (
                milestone.name
                for milestone in milestones
                if milestone.status.lower()
                in {
                    "active",
                    "in_progress",
                }
            ),
            None,
        )

        current_workstream = next(
            (
                workstream.name
                for workstream in workstreams
                if workstream.status
                in {
                    WorkstreamStatus.WARNING,
                    WorkstreamStatus.BLOCKED,
                }
            ),
            None,
        )

        return ProjectControlSnapshot(
            project_id=project.project_id,
            project_name=project.name,
            project_status=status,
            overall_progress_percent=progress,
            current_milestone=current_milestone,
            current_workstream=current_workstream,
            release_readiness=readiness,
            open_blockers=blockers,
            open_risks=open_risks,
            workstreams=tuple(workstreams),
            metrics=tuple(metrics),
            issues=tuple(issues),
            risks=tuple(risks),
            actions=tuple(actions),
            milestones=tuple(milestones),
            sources=tuple(source_health),
        )

    def list_projects(self) -> list[ProjectDefinition]:
        """Return active projects from the authoritative registry."""

        return self.registry.list_active()

"""
Project Control Center production composition.

This module assembles the Project Control Center application service
from authoritative application-layer dependencies.

Responsibilities:

- construct the approved PCD source adapters;
- inject authoritative application services/registries;
- construct ProjectControlService.

This module does NOT:

- own authoritative project state;
- duplicate Health state;
- duplicate Operations state;
- duplicate Privacy Governance state;
- call providers directly;
- mutate production systems;
- execute project-management decisions.

Architecture:

    Authoritative Application Services / Registries
                         |
                         v
                 PCD Source Adapters
                         |
                         v
                ProjectControlService
                         |
                         v
                ProjectControlSnapshot
"""

from __future__ import annotations

from app.application.health.operational_dashboard_service import (
    OperationalDashboardService,
)
from app.application.health.production_health_service import (

)
from app.application.privacy.privacy_governance_case_registry import (

from ..application.service import ProjectControlService
from ..domain.ports import ProjectRegistry, SnapshotRepository
from .governance_risk_source import (
    GovernanceRiskProjectStatusSource,
)
from .health_source import (

)
from .operations_source import (

def create_project_control_service(
    *,
    registry: ProjectRegistry,
    snapshots: SnapshotRepository,
    health_service: ProductionHealthService,
    dashboard_service: OperationalDashboardService,
    case_registry: PrivacyGovernanceCaseRegistry,
) -> ProjectControlService:
    """
    Create the production ProjectControlService.

    All dependencies are supplied explicitly.

    The function constructs only the PCD read-only adapters. The
    authoritative application services and registries remain owned
    by their respective bounded contexts.

    Args:
        registry:
            Authoritative PCD project registry.

        snapshots:
            PCD snapshot repository.

        health_service:
            Existing ProductionHealthService.

        dashboard_service:
            Existing OperationalDashboardService.

        case_registry:
            Existing PrivacyGovernanceCaseRegistry.

    Returns:
        Fully composed ProjectControlService.

    Raises:
        ValueError:
            If a required dependency is None.
    """

    if registry is None:
        raise ValueError(
            "registry cannot be None."
        )

    if snapshots is None:
        raise ValueError(
            "snapshots cannot be None."
        )

    if health_service is None:
        raise ValueError(
            "health_service cannot be None."
        )

    if dashboard_service is None:
        raise ValueError(
            "dashboard_service cannot be None."
        )

    if case_registry is None:
        raise ValueError(
            "case_registry cannot be None."
        )

    health_source = ProductionHealthProjectStatusSource(
        health_service=health_service,
    )

    operations_source = OperationsProjectStatusSource(
        dashboard_service=dashboard_service,
    )

    governance_source = GovernanceRiskProjectStatusSource(
        case_registry=case_registry,
    )

    return ProjectControlService(
        registry=registry,
        sources=(
            health_source,
            operations_source,
            governance_source,
        ),
        snapshots=snapshots,
    )

                    "source": w.source,

def make_identity_override(
    principal: Principal | None,
):
    """
    Override only the external identity/session resolution.

    This deliberately does NOT override authorization.

    Consequently:

        test Principal
            ↓
        real authorization dependency
            ↓
        real AuthorizationService
            ↓
        FUNCTIONALITY_CONTROL
    """

    async def override_get_current_principal() -> Principal:
        if principal is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Authentication required.",
            )

        return principal

    return override_get_current_principal

class FakeProductionHealthService:
    def __init__(
        self,
        *,
        result: ProductionReadinessResult | None = None,
        error: Exception | None = None,
    ) -> None:
        self.result = result
        self.error = error
        self.calls = 0

    async def check_production_health(
        self,
    ) -> ProductionReadinessResult:
        self.calls += 1

        if self.error is not None:
            raise self.error

        assert self.result is not None
        return self.result


def make_result(
    status: HealthStatus,
) -> ProductionReadinessResult:
    return ProductionReadinessResult(
        status=status,
        checks=(
            HealthCheckResult(
                name="postgresql",
                status=status,
                details=(
                    {"connectivity": "ok"}
                    if status is HealthStatus.HEALTHY
                    else {"reason": "database_unavailable"}
                ),
            ),
        ),
    )


@pytest.mark.asyncio

    service = FakeProductionHealthService(
        result=make_result(HealthStatus.HEALTHY),

@dataclass
class FakeDashboardService:
    snapshot: OperationalDashboardSnapshot
    calls: int = 0

    async def create_dashboard_snapshot(
        self,
    ) -> OperationalDashboardSnapshot:
        self.calls += 1
        return self.snapshot


def make_dashboard(
    *,
    incidents: tuple[DashboardIncident, ...] = (),
) -> OperationalDashboardSnapshot:
    generated_at = datetime.now(timezone.utc)

    empty_surface = DashboardSurfaceStatus(
        surface=OperationalSurface.PRIVATE_CLASSROOM,
        status=HealthStatus.HEALTHY,
        incidents=DashboardIncidentSummary(),
        metrics=DashboardOperationalMetrics(),
    )

    empty_announcement = DashboardSurfaceStatus(
        surface=OperationalSurface.ANNOUNCEMENT_GROUP,
        status=HealthStatus.HEALTHY,
        incidents=DashboardIncidentSummary(),
        metrics=DashboardOperationalMetrics(),
    )

    return OperationalDashboardSnapshot(
        generated_at=generated_at,
        overall_status=HealthStatus.HEALTHY,
        health_score=100,
        private_classroom=empty_surface,
        announcement_group=empty_announcement,
        incidents=incidents,
    )


def make_incident(
    *,
    incident_id: str = "incident-001",
    severity: str = "high",
    status: str = "open",
) -> DashboardIncident:
    now = datetime.now(timezone.utc)

    return DashboardIncident(
        incident_id=incident_id,
        correlation_id=f"corr-{incident_id}",
        surface=OperationalSurface.PRIVATE_CLASSROOM,
        severity=severity,
        status=status,
        first_seen=now,
        last_seen=now,
        occurrence_count=1,
    )
