"""
Speech Provider Configuration.

Defines provider settings required
for speech integrations.

Configuration only.

Does NOT:
- call providers
- store secrets directly
- execute speech operations
"""

from __future__ import annotations

from dataclasses import dataclass



@dataclass(
    frozen=True,
    slots=True,
)
class SpeechProviderConfig:
    """
    Immutable speech provider configuration.
    """

    provider_name: str

    model_name: str

    default_language: str

    enabled: bool = True


    def __post_init__(self) -> None:
        """
        Validate configuration.
        """

        if not self.provider_name.strip():
            raise ValueError(
                "provider_name cannot be empty."
            )

        if not self.model_name.strip():
            raise ValueError(
                "model_name cannot be empty."
            )

        if not self.default_language.strip():
            raise ValueError(
                "default_language cannot be empty."
            )
"""
Tests for OpenAI Speech Provider adapter.
"""

import pytest


from app.providers.speech.speech_provider import (
    SpeechProvider,
)

from app.providers.speech.openai_speech_provider import (
    OpenAISpeechProvider,
)



def test_openai_provider_implements_contract():

    provider = OpenAISpeechProvider()


    assert isinstance(
        provider,
        SpeechProvider,
    )



def test_openai_provider_transcription_not_ready():

    provider = OpenAISpeechProvider()


    with pytest.raises(
        NotImplementedError
    ):

        provider.transcribe(
            audio=b"test-audio",
            language_code="fr",
        )



def test_openai_provider_synthesis_not_ready():

    provider = OpenAISpeechProvider()


    with pytest.raises(
        NotImplementedError
    ):

        provider.synthesize(
            text="Bonjour",
            voice_profile="yeti",
            language_code="fr",
        )
"""
Tests for Speech Provider Factory.
"""

import pytest


from app.configuration.providers.speech_provider_config import (
    SpeechProviderConfig,
)


from app.providers.speech.speech_provider_factory import (
    SpeechProviderFactory,
)


from app.providers.speech.openai_speech_provider import (
    OpenAISpeechProvider,
)



def test_factory_creates_openai_provider():

    config = SpeechProviderConfig(
        provider_name="openai",
        model_name="gpt-4o-mini-tts",
        default_language="fr",
    )


    provider = (
        SpeechProviderFactory.create(
            config
        )
    )


    assert isinstance(
        provider,
        OpenAISpeechProvider,
    )



def test_factory_rejects_unknown_provider():

    config = SpeechProviderConfig(
        provider_name="unknown",
        model_name="unknown-model",
        default_language="fr",
    )


    with pytest.raises(
        ValueError,
        match="Unsupported speech provider",
    ):

        SpeechProviderFactory.create(
            config
        )
