_telegram_runtime: TelegramBotRuntime | None = None


def create_telegram_runtime() -> TelegramBotRuntime:
    """
    Compose Telegram Beta runtime.

    Composition layer only.

    Flow:

        Database
            |
            v
        Repositories
            |
            v
        Application Services
            |
            v
        Runtime
    """

    global _telegram_runtime

    if _telegram_runtime is not None:
        return _telegram_runtime

    # =================================================
    # Database
    # =================================================

    engine = create_database_engine()

    session_factory = create_session_factory(
        engine,
    )

    # =================================================
    # Beta access
    # =================================================

    beta_access_service = BetaAccessService(
        user_repository=(
            PostgreSQLBetaUserRepository(
                session_factory,
            )
        ),
        token_repository=(
            PostgreSQLBetaTokenRepository(
                session_factory,
            )
        ),
        binding_repository=(
            PostgreSQLTelegramBindingRepository(
                session_factory,
            )
        ),
    )

    # =================================================
    # Conversation quality governance
    #
    # Runs before persistence.
    #
    # Checks:
    #
    # - no raw identity
    # - no personal context leakage
    # - metadata completeness
    # - retention classification
    #
    # =================================================

    conversation_event_validator = (
        ConversationEventValidator()
    )

    # =================================================
    # Conversation event repository
    #
    # Persistence ONLY.
    #
    # No validation logic here.
    #
    # =================================================

    conversation_event_repository = (
        PostgreSQLConversationEventRepository(
            session_factory,
        )
    )

    # =================================================
    # Privacy identifier
    #
    # Telegram ID
    #       |
    #       v
    # pseudonymous hash
    #
    # =================================================

    privacy_identifier_service = (
        PrivacyIdentifierService()
    )

    # =================================================
    # Private Telegram access gate
    # =================================================

    private_chat_gate = BetaPrivateChatGate(
        beta_access_service=beta_access_service,
    )

    # =================================================
    # Telegram start handler
    #
    # The application handler does not own Telegram
    # transport replies. Reply delivery remains the
    # responsibility of the Telegram adapter/runtime.
    # =================================================

    start_handler = BetaTelegramStartHandler(
        beta_access_service=beta_access_service,
    )

    # =================================================
    # AI runtime
    # =================================================

    conversation_runtime = (
        create_conversation_runtime()
    )

    # =================================================
    # Conversation application service
    #
    # Validator injected here.
    #
    # =================================================

    conversation_service = TelegramConversationService(
        conversation_runtime=conversation_runtime,
        conversation_event_repository=(
            conversation_event_repository
        ),
        privacy_identifier_service=(
            privacy_identifier_service
        ),
        conversation_event_validator=(
            conversation_event_validator
        ),
    )

    # =================================================
    # Boundary
    # =================================================

    conversation_boundary = (
        BetaTelegramConversationBoundary(
            conversation_service=conversation_service,
            beta_access_service=beta_access_service,
        )
    )

    # =================================================
    # Telegram adapter
    # =================================================

    adapter = BetaTelegramAdapter(
        start_handler=start_handler,
        private_chat_gate=private_chat_gate,
    )

    # =================================================
    # Runtime
    # =================================================

    _telegram_runtime = TelegramBotRuntime(
        adapter=adapter,
        conversation_boundary=conversation_boundary,
    )

    logger.info(
        "Telegram runtime composed with PostgreSQL persistence, "
        "privacy layer, conversation validation and governance."
    )

    return _telegram_runtime


async def start_telegram_runtime() -> None:
    runtime = create_telegram_runtime()

    await runtime.start()


async def stop_telegram_runtime() -> None:
    if _telegram_runtime is None:
        return

    await _telegram_runtime.stop()


def reset_telegram_runtime() -> None:
    global _telegram_runtime

    _telegram_runtime = None

_bootstrapped = False


def bootstrap_application() -> None:
    """
    Bootstrap application dependencies.

    Creates the ExternalRuntimeService and registers it
    inside the application runtime container.
    """

    global _bootstrapped

    if _bootstrapped:
        return

    factory = ApplicationFactory()

    gateway = create_external_runtime_gateway(
        live_intelligence_agent=(
            create_live_intelligence_agent()
        ),
        podcast_discovery_agent=(
            create_podcast_discovery_agent()
        ),
        podcast_providers=(
            create_podcast_providers()
        ),
        governance=(
            create_governance()
        ),
    )

    runtime_service = (
        factory.create_external_runtime_service(
            gateway=gateway,
        )
    )

    configure_external_runtime_service(
        runtime_service,
    )

    _bootstrapped = True


def get_external_runtime_service() -> ExternalRuntimeService:
    """
    Backward-compatible Runtime accessor.
    """

    return _get_external_runtime_service()


def reset_bootstrap() -> None:
    """
    Test-only reset.
    """

    global _bootstrapped

    _bootstrapped = False

from __future__ import annotations

import logging
import os
from typing import Any

from telegram import Update
from telegram.ext import (
    Application,
    CommandHandler,
    ContextTypes,
    MessageHandler,
    filters,
)

from backend.app.interfaces.telegram.beta_telegram_adapter import (
    BetaTelegramAdapter,
)


logger = logging.getLogger(__name__)


class TelegramBotRuntime:
    """
    Infrastructure adapter for python-telegram-bot.

    Responsibilities:
    - Telegram application lifecycle;
    - Telegram handler registration;
    - delegation to BetaTelegramAdapter.

    The runtime MUST NOT:
    - perform Beta authorization;
    - execute conversation logic;
    - deliver classroom responses returned by the adapter.

    IMPORTANT RESPONSE OWNERSHIP:

        Telegram update
              |
              v
        TelegramBotRuntime
              |
              v
        BetaTelegramAdapter
              |
              +----> reply_text()
              |
              v
          response

    The adapter owns Telegram classroom response delivery.

    Therefore the runtime MUST NOT call reply_text() with the
    response returned by the adapter.

    This prevents duplicate Telegram messages.
    """

    def __init__(
        self,
        *,
        adapter: BetaTelegramAdapter,
        conversation_boundary: Any = None,
    ) -> None:
        if adapter is None:
            raise ValueError(
                "adapter cannot be None."
            )

        self._adapter = adapter

        # Retained for constructor compatibility with existing
        # composition. Runtime does not execute the boundary.
        self._conversation_boundary = conversation_boundary

        self._application: Application | None = None
        self._running = False

    # =========================================================
    # Application construction
    # =========================================================

    def build(self) -> Application:
        token = os.getenv(
            "TELEGRAM_BOT_TOKEN"
        )

        if not token:
            raise RuntimeError(
                "TELEGRAM_BOT_TOKEN is missing."
            )

        application = (
            Application
            .builder()
            .token(token)
            .build()
        )

        application.add_handler(
            CommandHandler(
                "start",
                self._handle_start,
            )
        )

        application.add_handler(
            MessageHandler(
                filters.ChatType.PRIVATE
                & filters.TEXT
                & ~filters.COMMAND,
                self._handle_private_message,
            )
        )

        self._application = application

        logger.info(
            "Telegram bot runtime created."
        )

        return application

    # =========================================================
    # Lifecycle
    # =========================================================

    async def start(self) -> None:
        if self._running:
            logger.warning(
                "Telegram runtime already running."
            )
            return

        if self._application is None:
            self.build()

        assert self._application is not None

        await self._application.initialize()
        await self._application.start()

        if self._application.updater is not None:
            await self._application.updater.start_polling(
                drop_pending_updates=False,
            )

        self._running = True

        logger.info(
            "Telegram bot polling started."
        )

    async def stop(self) -> None:
        if self._application is None:
            return

        if (
            self._application.updater is not None
            and self._running
        ):
            await self._application.updater.stop()

        await self._application.stop()
        await self._application.shutdown()

        self._running = False

        logger.info(
            "Telegram bot runtime stopped."
        )

    # =========================================================
    # Telegram handlers
    # =========================================================

    async def _handle_start(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        await self._adapter.handle_start(
            update,
            context,
        )

    async def _handle_private_message(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        if (
            update.effective_user is None
            or update.effective_chat is None
            or update.effective_message is None
        ):
            return

        try:
            # =================================================
            # SINGLE DELEGATION POINT
            # =================================================
            #
            # BetaTelegramAdapter:
            #
            #   1. validates transport
            #   2. authorizes access
            #   3. invokes conversation boundary
            #   4. sends Telegram response
            #   5. returns response for observation/testing
            #
            # The returned value MUST NOT be sent again.
            #

            await self._adapter.handle_message(
                update,
                context,
            )

        except Exception:
            logger.exception(
                "Unexpected Telegram adapter failure."
            )

            # This is ONLY an infrastructure-level fallback for
            # failures escaping the adapter itself.
            #
            # Normal conversation failures are already contained
            # by BetaTelegramAdapter.
            await update.effective_message.reply_text(
                "EasyChatBot is temporarily unavailable. "
                "Please try again."
            )
