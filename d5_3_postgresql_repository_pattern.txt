from __future__ import annotations

import os

from sqlalchemy import create_engine
from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker


class Base(DeclarativeBase):
    """Canonical SQLAlchemy metadata registry for the platform."""

    pass


def database_url() -> str:
    """Return the canonical PostgreSQL connection URL."""

    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for PostgreSQL persistence."
        )

    return value


def create_database_engine():
    """Create the existing synchronous PostgreSQL engine.

    This API is preserved for the existing Beta 1 persistence layer.
    """

    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(
    engine,
) -> sessionmaker[Session]:
    """Create the existing synchronous session factory.

    Preserved for existing Beta 1 repositories.
    """

    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )


def async_database_url() -> str:
    """Return DATABASE_URL adapted for SQLAlchemy async PostgreSQL.

    The environment remains the single source of truth. We only adapt the
    driver at the infrastructure boundary when the configured URL uses the
    standard PostgreSQL SQLAlchemy scheme.
    """

    value = database_url()

    if value.startswith("postgresql+asyncpg://"):
        return value

    if value.startswith("postgresql://"):
        return value.replace(
            "postgresql://",
            "postgresql+asyncpg://",
            1,
        )

    raise RuntimeError(
        "DATABASE_URL must use a PostgreSQL URL compatible with "
        "the synchronous or async PostgreSQL persistence boundary."
    )


def create_async_database_engine() -> AsyncEngine:
    """Create the canonical asynchronous PostgreSQL engine.

    This is used by durable WS-22 persistence while sharing the same
    DATABASE_URL and canonical Base as the existing persistence layer.
    """

    return create_async_engine(
        async_database_url(),
        pool_pre_ping=True,
    )


def create_async_session_factory(
    engine: AsyncEngine,
) -> async_sessionmaker[AsyncSession]:
    """Create the canonical asynchronous PostgreSQL session factory."""

    return async_sessionmaker(
        bind=engine,
        class_=AsyncSession,
        expire_on_commit=False,
    )

class PostgreSQLFunctionalityControlRepository:
    """Durable functionality control state and audit repository."""

    def __init__(self, session_factory) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")

        self._session_factory = session_factory

    def get_availability(
        self,
        *,
        environment: str,
        functionality_id: str,
    ) -> str:
        with self._session_factory() as session:
            record = session.scalar(
                select(FunctionalityControlStateRecord).where(
                    FunctionalityControlStateRecord.environment
                    == environment,
                    FunctionalityControlStateRecord.functionality_id
                    == functionality_id,
                )
            )

        if record is None:
            return "enabled"

        return record.availability

    def get_last_action(
        self,
        *,
        environment: str,
        functionality_id: str,
    ) -> OperationalControlResult | None:
        with self._session_factory() as session:
            record = session.scalar(
                select(FunctionalityControlStateRecord).where(
                    FunctionalityControlStateRecord.environment
                    == environment,
                    FunctionalityControlStateRecord.functionality_id
                    == functionality_id,
                )
            )

        if record is None or record.changed_at is None:
            return None

        return OperationalControlResult(
            functionality_id=record.functionality_id,
            environment=record.environment,
            availability=record.availability,
            action=record.action or "unknown",
            reason=record.reason,
            changed_at=record.changed_at,
            changed_by=record.changed_by or "unknown",
        )

    def record_transition(
        self,
        *,
        functionality_id: str,
        environment: str,
        availability: str,
        action: str,
        reason: str | None,
        changed_at: datetime,
        changed_by: str,
    ) -> OperationalControlResult:
        with self._session_factory() as session:
            state = session.scalar(
                select(FunctionalityControlStateRecord).where(
                    FunctionalityControlStateRecord.environment
                    == environment,
                    FunctionalityControlStateRecord.functionality_id
                    == functionality_id,
                )
            )

            if state is None:
                state = FunctionalityControlStateRecord(
                    functionality_id=functionality_id,
                    environment=environment,
                )
                session.add(state)

            state.availability = availability
            state.action = action
            state.reason = reason
            state.changed_at = changed_at
            state.changed_by = changed_by

            session.add(
                FunctionalityControlAuditRecord(
                    functionality_id=functionality_id,
                    environment=environment,
                    action=action,
                    availability=availability,
                    changed_by=changed_by,
                    reason=reason,
                    changed_at=changed_at,
                )
            )

            session.commit()

        return OperationalControlResult(
            functionality_id=functionality_id,
            environment=environment,
            availability=availability,
            action=action,
            reason=reason,
            changed_at=changed_at,
            changed_by=changed_by,
        )

from src.ai_runtime.providers.usage.models import ProviderUsageEvent
from src.ai_runtime.providers.usage.repository import ProviderUsageRepository

from .provider_usage_records import ProviderUsageRecord


class PostgreSQLProviderUsageRepository(ProviderUsageRepository):
    def __init__(self, session_factory) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")

        self._session_factory = session_factory

    def save(self, event: ProviderUsageEvent) -> None:
        if event is None:
            raise ValueError("Provider usage event cannot be None.")

        with self._session_factory() as session:
            session.add(
                ProviderUsageRecord(
                    provider=event.provider,
                    model=event.model,
                    capability=event.capability,
                    functionality=event.functionality,
                    agent=event.agent,
                    input_tokens=event.input_tokens,
                    output_tokens=event.output_tokens,
                    total_tokens=event.total_tokens,
                    latency_ms=event.latency_ms,
                    estimated_cost=event.estimated_cost,
                    quality_score=event.quality_score,
                    success=event.success,
                    provider_request_id=event.provider_request_id,
                    created_at=event.created_at,
                )
            )
            session.commit()

    def get_all(self) -> list[ProviderUsageEvent]:
        with self._session_factory() as session:
            rows = (
                session.query(ProviderUsageRecord)
                .order_by(
                    ProviderUsageRecord.created_at.asc(),
                    ProviderUsageRecord.id.asc(),
                )
                .all()
            )

        return [_event_from_record(row) for row in rows]

    def clear(self) -> None:
        with self._session_factory() as session:
            session.query(ProviderUsageRecord).delete()
            session.commit()


def _event_from_record(row: ProviderUsageRecord) -> ProviderUsageEvent:
    return ProviderUsageEvent(

        provider_request_id=row.provider_request_id,
        created_at=row.created_at,
    )

class PostgreSQLAuthenticationRepository:
    """PostgreSQL adapter for the existing EasyChatBot auth tables.

    The adapter intentionally uses SQLAlchemy Core statements rather than
    importing speculative ORM models. This binds the application contract to
    the inspected database schema while keeping persistence isolated.
    """

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")

        self._session_factory = session_factory

    async def email_exists(self, email: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    "SELECT 1 FROM users "
                    "WHERE email = :email "
                    "LIMIT 1"
                ),
                {"email": email},
            )

            return result.first() is not None

    async def username_exists(self, username: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    "SELECT 1 FROM users "
                    "WHERE username = :username "
                    "LIMIT 1"
                ),
                {"username": username},
            )

            return result.first() is not None

    async def create_user(
        self,
        *,
        user_id: UUID,
        email: str,
        username: str,
    ) -> None:
        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO users
                            (
                                id,
                                email,
                                username,
                                status,
                                trial_status,
                                created_at,
                                email_verified
                            )
                        VALUES
                            (
                                :id,
                                :email,
                                :username,
                                'active',
                                'not_started',
                                :created_at,
                                FALSE
                            )
                        """
                    ),
                    {
                        "id": user_id,
                        "email": email,
                        "username": username,
                        "created_at": now,
                    },
                )

                await session.execute(
                    text(
                        """
                        INSERT INTO auth_user_roles (user_id, role)
                        VALUES (:user_id, 'beta_user')
                        ON CONFLICT DO NOTHING
                        """
                    ),
                    {"user_id": user_id},
                )

    async def create_password_credential(

                    ),
                    {
                        "id": uuid4(),
                        "user_id": user_id,
                        "password_hash": password_hash,
                        "created_at": now,
                        "updated_at": now,
                    },
                )

    async def create_email_verification(
        self,
        *,
        user_id: UUID,
        email: str,
    ) -> str:
        """Create a one-time verification token.

        The raw token is returned to the application boundary. Only its
        SHA-256 digest is persisted. Delivery is deliberately outside this
        repository.
        """
        raw_token = secrets.token_urlsafe(32)
        token_hash = hashlib.sha256(
            raw_token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)
        verification_id = uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_email_verification_tokens
                            (
                                id,
                                user_id,
                                email,
                                token_hash,
                                expires_at,
                                created_at
                            )
                        VALUES
                            (
                                :id,
                                :user_id,
                                :email,
                                :token_hash,
                                :expires_at,
                                :created_at
                            )
                        """
                    ),
                    {
                        "id": verification_id,
                        "user_id": user_id,
                        "email": email,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=24),
                        "created_at": now,
                    },
                )

        return raw_token

    async def create_password_reset_token(
        self,
        *,
        user_id: UUID,
    ) -> str:
        """Create a secure one-time password-reset token.

        Only the SHA-256 digest is persisted. The raw token is returned
        exclusively to the application layer for email delivery.
        """
        raw_token = secrets.token_urlsafe(32)

        token_hash = hashlib.sha256(
            raw_token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)
        reset_id = uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                # A newly issued reset link supersedes older unused links.
                await session.execute(
                    text(
                        """
                        UPDATE auth_password_reset_tokens
                        SET used_at = :now
                        WHERE user_id = :user_id
                          AND used_at IS NULL
                        """
                    ),
                    {
                        "now": now,
                        "user_id": user_id,
                    },
                )

                await session.execute(
                    text(
                        """
                        INSERT INTO auth_password_reset_tokens
                            (
                                id,
                                user_id,
                                token_hash,
                                expires_at,
                                created_at
                            )
                        VALUES
                            (
                                :id,
                                :user_id,
                                :token_hash,
                                :expires_at,
                                :created_at
                            )
                        """
                    ),
                    {
                        "id": reset_id,
                        "user_id": user_id,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=1),
                        "created_at": now,
                    },
                )

        return raw_token

    async def count_recent_password_reset_tokens(
        self,
        *,
        user_id: UUID,
        since: datetime,
    ) -> int:
        """Count password-reset requests in the supplied time window."""
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT COUNT(*)
                    FROM auth_password_reset_tokens
                    WHERE user_id = :user_id
                      AND created_at >= :since
                    """
                ),
                {
                    "user_id": user_id,
                    "since": since,
                },
            )

            return int(result.scalar_one())

    async def get_password_reset_context(
        self,
        *,
        token: str,
    ) -> PasswordResetContextRecord | None:
        """Resolve a valid reset token to non-sensitive account context.

        The raw token is never persisted or returned. Only a valid, unused,
        unexpired token can resolve its account email and current credential
        hash. The credential hash is consumed only by the application service
        for the previous-password check and is never exposed by the HTTP API.
        """
        token_hash = hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT
                        t.user_id,
                        u.email,
                        c.password_hash
                    FROM auth_password_reset_tokens AS t
                    INNER JOIN users AS u
                        ON u.id = t.user_id
                    INNER JOIN user_credentials AS c
                        ON c.user_id = t.user_id
                       AND c.credential_type = 'PASSWORD'
                    WHERE t.token_hash = :token_hash
                      AND t.used_at IS NULL
                      AND t.expires_at > :now
                    LIMIT 1
                    """
                ),
                {
                    "token_hash": token_hash,
                    "now": now,
                },
            )

            row = result.mappings().first()

            if row is None:
                return None

            return PasswordResetContextRecord(
                user_id=row["user_id"],
                email=row["email"],
                password_hash=row["password_hash"],
            )

    async def get_password_reset_user(
        self,
        token: str,
    ):
        """Return the bound reset user for a valid, unused, unexpired token."""
        context = await self.get_password_reset_context(
            token=token,
        )

        if context is None:
            return None

        return context

    async def reset_password(
        self,
        *,
        token: str,
        password_hash: str,
        expected_password_hash: str | None = None,
    ) -> bool:
        """Atomically consume a reset token, update the password,
        and revoke existing sessions.
        """
        token_hash = hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    text(
                        """
                        SELECT id, user_id
                        FROM auth_password_reset_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now
                        FOR UPDATE
                        """
                    ),
                    {
                        "token_hash": token_hash,
                        "now": now,
                    },
                )

                row = result.mappings().first()

                if row is None:
                    return False

                user_id = row["user_id"]

                credential_result = await session.execute(
                    text(
                        """
                        SELECT id
                        FROM user_credentials
                        WHERE user_id = :user_id
                          AND credential_type = 'PASSWORD'
                        LIMIT 1
                        """
                    ),
                    {
                        "user_id": user_id,
                    },
                )

                credential = credential_result.mappings().first()

                if credential is None:
                    return False

                update_result = await session.execute(
                    text(
                        """
                        UPDATE user_credentials
                        SET password_hash = :password_hash,
                            updated_at = :updated_at
                        WHERE id = :credential_id
                          AND (
                              CAST(:expected_password_hash AS TEXT) IS NULL
                              OR password_hash = CAST(:expected_password_hash AS TEXT)
                          )
                        """
                    ),
                    {
                        "password_hash": password_hash,
                        "updated_at": now,
                        "credential_id": credential["id"],
                        "expected_password_hash": expected_password_hash,
                    },
                )

                if update_result.rowcount != 1:
                    return False

                await session.execute(
                    text(
                        """
                        UPDATE auth_password_reset_tokens
                        SET used_at = :used_at
                        WHERE id = :id
                        """
                    ),
                    {
                        "used_at": now,
                        "id": row["id"],
                    },
                )

                await session.execute(
                    text(
                        """
                        UPDATE user_sessions
                        SET is_revoked = TRUE,
                            status = 'revoked'
                        WHERE user_id = :user_id
                          AND is_revoked = FALSE
                        """
                    ),
                    {
                        "user_id": user_id,
                    },
                )

                return True
