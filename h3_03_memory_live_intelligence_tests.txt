from src.challenge_engine.models import (

    DeterministicEvaluator,

from src.evaluation.engine import (

def _create_h1_learning_challenge() -> LearningChallenge:
    """
    Create a representative H1 learning challenge.

    This mirrors the structure produced by the H1
    Challenge Engine without calling external services.
    """

    return LearningChallenge(
        title="AI Act Learning Challenge",
        topic="AI Act",
        language="French",
        learner_level="B2",
        summary=(
            "The European Union AI Act establishes "
            "a risk-based framework for artificial "
            "intelligence."
        ),
        vocabulary=[
            VocabularyItem(
                term="gouvernance",
                definition=(
                    "La manière dont une organisation "
                    "est dirigée et contrôlée."
                ),
                example=(
                    "La gouvernance de l'IA devient "
                    "essentielle."
                ),
            ),
            VocabularyItem(
                term="réglementation",
                definition=(
                    "Ensemble des règles qui encadrent "
                    "une activité."
                ),
                example=(
                    "La réglementation européenne évolue."
                ),
            ),
        ],
        discussion_questions=[
            DiscussionQuestion(
                question=(
                    "Pourquoi la réglementation de "
                    "l'IA est-elle importante ?"
                ),
                difficulty="B2",
            ),
            DiscussionQuestion(
                question=(
                    "Quels sont les avantages d'une "
                    "réglementation européenne de l'IA ?"
                ),
                difficulty="B2",
            ),
        ],
    )


def test_h1_learning_challenge_can_be_evaluated():
    """
    Validate the complete H1 → H2 evaluation boundary.
    """

    challenge = _create_h1_learning_challenge()

    evaluator = DeterministicEvaluator()

    scores = evaluator.evaluate(
        challenge
    )

    assert len(scores) == 4

    score_names = {
        score.criterion
        for score in scores
    }

    assert score_names == {
        "completeness",
        "language_level",
        "educational_value",
        "source_coverage",
    }


def test_h1_learning_challenge_produces_quality_result():
    """
    Evaluate the H1 challenge through the H2 scoring engine.
    """

    challenge = _create_h1_learning_challenge()

    evaluator = DeterministicEvaluator()

    scores = evaluator.evaluate(
        challenge
    )

    engine = EvaluationEngine(
        criteria=[
            COMPLETENESS,
            LANGUAGE_LEVEL,
            EDUCATIONAL_VALUE,
            SOURCE_COVERAGE,
        ],
        passing_score=0.70,
    )

    result = engine.evaluate(
        scores
    )

    assert result.overall_score == 1.0

    assert result.passed is True

    assert len(result.scores) == 4

    assert result.evaluator == "deterministic"


def test_h1_learning_challenge_contains_high_quality_components():
    """
    Verify the H1 challenge satisfies the basic quality
    dimensions currently supported by deterministic evaluation.
    """

    challenge = _create_h1_learning_challenge()

    evaluator = DeterministicEvaluator()

    scores = evaluator.evaluate(
        challenge
    )

    for score in scores:
        assert 0.0 <= score.score <= 1.0
        assert score.reasoning.strip()

    assert all(
        score.score == 1.0
        for score in scores
    )

"""
Integration test for H3 governance of the H1 Live Intelligence capability.

Flow:

H1 Live Intelligence Capability
        |
        v
Governance Policy
        |
        v
Governance Policy Engine
        |
        v
Governance Decision

This test validates the governance boundary only.

It does not:
- execute the Conversation Agent
- execute the Parallel API
- modify H1 implementation
- modify H2 evaluation
- perform external side effects
"""

from src.governance.engine import (
    GovernancePolicyEngine,
)

from src.governance.models import (
    GovernanceDecision,
    GovernancePolicy,
    RiskLevel,
)


def create_live_intelligence_policy() -> GovernancePolicy:
    """
    Define the governance policy for H1 Live Intelligence.

    Live research:
    - may access external information
    - must not create external side effects
    - is considered medium risk
    - does not require human approval for passive research
    """

    return GovernancePolicy(
        capability="live_intelligence_research",
        risk_level=RiskLevel.MEDIUM,
        requires_human_review=False,
        allows_external_calls=True,
        allows_side_effects=False,
    )


def test_live_intelligence_capability_is_governed():

    engine = GovernancePolicyEngine()

    policy = create_live_intelligence_policy()

    result = engine.evaluate(
        policy
    )

    assert (
        result.capability
        == "live_intelligence_research"
    )

    assert (
        result.risk_level
        == RiskLevel.MEDIUM
    )

    assert (
        result.decision
        == GovernanceDecision.ALLOW
    )


def test_live_intelligence_allows_external_research():

    engine = GovernancePolicyEngine()

    result = engine.evaluate(
        create_live_intelligence_policy()
    )

    assert any(
        "External calls are explicitly permitted."
        in reason
        for reason in result.reasons
    )


def test_live_intelligence_does_not_allow_side_effects():

    policy = create_live_intelligence_policy()

    assert policy.allows_side_effects is False


def test_live_intelligence_with_side_effects_is_blocked():

    engine = GovernancePolicyEngine()

    unsafe_policy = GovernancePolicy(
        capability="live_intelligence_research",
        risk_level=RiskLevel.MEDIUM,
        requires_human_review=False,
        allows_external_calls=True,
        allows_side_effects=True,
    )

    result = engine.evaluate(
        unsafe_policy
    )

    assert (
        result.decision
        == GovernanceDecision.BLOCK
    )


def test_live_intelligence_high_risk_requires_review():

    engine = GovernancePolicyEngine()

    high_risk_policy = GovernancePolicy(
        capability="live_intelligence_research",
        risk_level=RiskLevel.HIGH,
        requires_human_review=False,
        allows_external_calls=True,
        allows_side_effects=False,
    )

    result = engine.evaluate(
        high_risk_policy

@dataclass
class FakeMemory:
    """
    Deterministic memory record for tests.
    """

    user_id: str
    category: str
    content: str


class FakeMemoryService:
    """
    Deterministic MemoryService test double.
    """

    def __init__(self) -> None:
        self.memories = [
            FakeMemory(
                user_id="demo-user",
                category="conversation",
                content="Previous conversation",
            ),
        ]

        self.recall_calls = []
        self.remember_calls = []

    def recall(
        self,
        user_id: str,
    ):
        self.recall_calls.append(
            user_id
        )

        return [
            memory
            for memory in self.memories
            if memory.user_id == user_id
        ]

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> None:
        self.remember_calls.append(
            {
                "user_id": user_id,
                "category": category,
                "content": content,
            }
        )
