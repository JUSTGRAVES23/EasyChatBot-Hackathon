===== VOICE CHARACTER RUNTIME =====
"""
Voice Character Runtime.

Coordinates character identity
with speech capabilities.

Responsibilities:
- Hold character runtime context.
- Generate character speech requests.
- Preserve personality boundaries.

Does NOT:
- Call providers directly.
- Manage LiveKit.
- Store conversations.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.ai_runtime.voice.voice_speech_runtime import (
    VoiceSpeechRuntime,
)


from app.domain.uai.context.interaction_context import (
    InteractionContext,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCharacterRuntime:
    """
    Runtime for speaking AI characters.
    """

    context: InteractionContext

    speech_runtime: VoiceSpeechRuntime



    def speak(
        self,
        text: str,
    ) -> bytes:
        """
        Generate character audio.

        Uses the selected character voice profile.
        """

        return self.speech_runtime.synthesize(
            text=text,
            voice_profile=(
                self.context
                .voice
                .voice_profile
            ),
            language_code=(
                self.context
                .voice
                .language_code
            ),
        )



    def character_id(
        self,
    ) -> str:
        """
        Return active character identity.
        """

        return (
            self.context
            .character
            .character_id
        )
===== VOICE SPEECH RUNTIME =====
"""
Voice Speech Runtime.

Runtime bridge between AI characters
and speech providers.

Responsibilities:
- Coordinate speech generation.
- Use provider abstraction.
- Keep AI runtime provider independent.

Does NOT:
- Call vendors directly.
- Manage LiveKit rooms.
- Store secrets.
- Manage conversations.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.providers.speech.speech_provider import (
    SpeechProvider,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceSpeechRuntime:
    """
    Runtime service for speech operations.

    Depends on the SpeechProvider abstraction.
    """

    speech_provider: SpeechProvider



    def transcribe(
        self,
        *,
        audio: bytes,
        language_code: str,
    ) -> str:
        """
        Convert audio input into text.
        """

        return self.speech_provider.transcribe(
            audio=audio,
            language_code=language_code,
        )



    def synthesize(
        self,
        *,
        text: str,
        voice_profile: str,
        language_code: str,
    ) -> bytes:
        """
        Convert text into generated audio.
        """

        return self.speech_provider.synthesize(
            text=text,
            voice_profile=voice_profile,
            language_code=language_code,
        )
===== VOICE CALL RUNTIME =====
"""
Voice Call Runtime.

Runtime orchestration layer for
AI character voice calls.

Responsibilities:
- Manage voice call lifecycle execution.
- Coordinate voice session activation.
- Apply runtime boundaries.
- Return runtime state changes.

Does NOT:
- Handle HTTP requests.
- Create API responses.
- Manage persistence.
- Select providers.
- Directly manage LiveKit.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)


from app.domain.uai.voice.voice_call_state import (
    VoiceCallState,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallRuntime:
    """
    Runtime coordinator for voice calls.
    """



    def prepare(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        Prepare a voice call for activation.

        Transition:

        CREATED
            |
            â†“
        PREPARING
            |
            â†“
        READY
        """


        if session.state not in {
            VoiceCallState.CREATED,
            VoiceCallState.PREPARING,
        }:
            raise ValueError(
                "Voice call cannot be prepared "
                f"from state {session.state.value}."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.READY,
        )



    def activate(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        Activate a prepared voice call.

        Transition:

        READY
          |
          â†“
        ACTIVE
        """


        if session.state != (
            VoiceCallState.READY
        ):
            raise ValueError(
                "Voice call must be ready "
                "before activation."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.ACTIVE,
        )



    def end(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        End an active voice call.

        Transition:

        ACTIVE
          |
          â†“
        ENDED
        """


        if session.state != (
            VoiceCallState.ACTIVE
        ):
            raise ValueError(
                "Only active calls can be ended."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.ENDED,
        )



    def fail(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        Mark voice call as failed.
        """


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.FAILED,
        )
===== VOICE CALL EXECUTION RUNTIME =====
"""
Voice Call Execution Runtime.

Runtime execution coordinator for
AI character voice calls.

Responsibilities:
- Execute prepared voice call sessions.
- Coordinate execution lifecycle.
- Validate execution context.
- Provide runtime boundaries.

Does NOT:
- Handle HTTP.
- Persist sessions.
- Manage authentication.
- Own LiveKit connections.
- Select providers.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.domain.uai.voice.voice_call_execution_context import (
    VoiceCallExecutionContext,
)


from app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)


from app.domain.uai.voice.voice_call_state import (
    VoiceCallState,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallExecutionRuntime:
    """
    Executes prepared voice calls.
    """



    def start(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> VoiceCallSession:
        """
        Start voice call execution.

        Expected lifecycle:

        READY
          |
          â†“
        ACTIVE
        """


        session = (
            context.session
        )


        if session.state != (
            VoiceCallState.READY
        ):
            raise ValueError(
                "Voice call must be READY "
                "before execution."
            )


        if not context.room.has_character():

            raise ValueError(
                "Voice call room requires "
                "a character participant."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.ACTIVE,
        )



    def stop(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> VoiceCallSession:
        """
        Stop voice call execution.

        Expected lifecycle:

        ACTIVE
          |
          â†“
        ENDED
        """


        session = (
            context.session
        )


        if session.state != (
            VoiceCallState.ACTIVE
        ):
            raise ValueError(
                "Only ACTIVE voice calls "
                "can be stopped."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.ENDED,
        )



    def fail(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> VoiceCallSession:
        """
        Mark execution as failed.
        """


        session = (
            context.session
        )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.FAILED,
        )

"""
Speech Provider Contract.

Defines the abstraction for speech services.

Supported capabilities:
- Speech to Text (STT)
- Text to Speech (TTS)

The AI Runtime depends on this contract,
not on a specific vendor.

Possible implementations:
- OpenAI Speech
- ElevenLabs
- Azure Speech
- Local models

Does NOT:
- Manage conversations.
- Manage characters.
- Manage LiveKit rooms.
- Store user data.
"""

from __future__ import annotations

from abc import (
    ABC,
    abstractmethod,
)



class SpeechProvider(
    ABC
):
    """
    Abstract speech provider.

    Provider implementations must
    implement this interface.
    """


    @abstractmethod
    def transcribe(
        self,
        audio: bytes,
        language_code: str,
    ) -> str:
        """
        Convert audio into text.

        Parameters
        ----------
        audio:
            Raw audio bytes.

        language_code:
            Expected language.

        Returns
        -------
        str
            Transcribed text.
        """

        raise NotImplementedError



    @abstractmethod
    def synthesize(
        self,
        text: str,
        voice_profile: str,
        language_code: str,
    ) -> bytes:
        """
        Convert text into audio.

        Parameters
        ----------
        text:
            Text to speak.

        voice_profile:
            Character voice profile.

        language_code:
            Output language.

        Returns
        -------
        bytes
            Generated audio.
        """

        raise NotImplementedError
"""
Speech Provider Factory.

Creates speech provider implementations
from configuration.

Responsibilities:
- Convert configuration into providers.
- Centralize provider selection.

Does NOT:
- Execute speech operations.
- Store credentials.
- Manage runtime sessions.
"""

from __future__ import annotations


from app.configuration.providers.speech_provider_config import (
    SpeechProviderConfig,
)


from app.providers.speech.speech_provider import (
    SpeechProvider,
)


from app.providers.speech.openai_speech_provider import (
    OpenAISpeechProvider,
)



class SpeechProviderFactory:
    """
    Factory for speech providers.
    """

    @staticmethod
    def create(
        config: SpeechProviderConfig,
    ) -> SpeechProvider:
        """
        Create speech provider from configuration.

        Parameters
        ----------
        config:
            Provider configuration.

        Returns
        -------
        SpeechProvider
            Configured speech provider.

        Raises
        ------
        ValueError
            If provider is unsupported.
        """

        provider_name = (
            config.provider_name
            .lower()
            .strip()
        )


        if provider_name == "openai":

            return OpenAISpeechProvider()


        raise ValueError(
            f"Unsupported speech provider: {provider_name}"
        )

"""
LiveKit Infrastructure Adapter.

Infrastructure boundary for LiveKit room configuration.

Responsibilities:
- Translate voice session information into
  LiveKit provider configuration.
- Keep LiveKit-specific details inside infrastructure.
- Prepare provider connection metadata.

Does NOT:
- Decide when a call starts.
- Manage learner business logic.
- Execute AI agents.
- Manage application lifecycle.
- Persist data.
- Generate participant tokens.
"""

from __future__ import annotations


import logging


from app.domain.uai.voice.voice_session import (
    VoiceSession,
)


logger = logging.getLogger(
    __name__
)



class LiveKitAdapter:
    """
    Infrastructure adapter for LiveKit.

    Converts the provider-independent voice session
    into LiveKit room configuration.

    Token generation is deliberately handled by
    LiveKitTokenService.
    """



    provider = "livekit"



    def create_room_configuration(
        self,
        *,
        session: VoiceSession,
    ) -> dict[str, str]:
        """
        Prepare LiveKit room configuration.

        Parameters
        ----------
        session:
            Provider-independent voice session.

        Returns
        -------
        dict[str, str]
            LiveKit room metadata.
        """



        if not session.room_name.strip():

            raise ValueError(
                "Voice room name cannot be empty."
            )



        if not session.character_id.strip():

            raise ValueError(
                "Voice session character id "
                "cannot be empty."
            )



        logger.info(
            "Preparing LiveKit room configuration",
            extra={
                "session_id": (
                    session.session_id
                ),
                "room_name": (
                    session.room_name
                ),
                "character_id": (
                    session.character_id
                ),
                "language_code": (
                    session.language_code
                ),
            },
        )



        return {
            "room_name": (
                session.room_name
            ),
            "character_id": (
                session.character_id
            ),
            "language_code": (
                session.language_code
            ),
            "provider": self.provider,
        }



__all__ = [
    "LiveKitAdapter",
]
"""
LiveKit Token Service.

Infrastructure service responsible for generating
real LiveKit participant access tokens.

Responsibilities:
- Generate signed LiveKit JWT access tokens.
- Grant participant access to a specific room.
- Keep LiveKit credentials inside infrastructure.
- Return provider-independent VoiceConnectionToken.

Does NOT:
- Handle HTTP.
- Manage application workflows.
- Decide business authorization.
- Persist tokens.
- Execute AI agents.
"""

from __future__ import annotations


from datetime import (
    datetime,
    timedelta,
    timezone,
)


import os
import uuid


from dotenv import load_dotenv


from livekit import api


from app.domain.uai.voice.voice_connection_token import (
    VoiceConnectionToken,
)


load_dotenv()



class LiveKitTokenService:
    """
    Real LiveKit access-token infrastructure adapter.
    """



    def create_token(
        self,
        *,
        room_name: str,
        identity: str | None = None,
        expires_minutes: int = 30,
    ) -> VoiceConnectionToken:
        """
        Create a signed LiveKit participant token.

        Parameters
        ----------
        room_name:
            LiveKit room the participant may join.

        identity:
            Unique LiveKit participant identity.

        expires_minutes:
            Token validity period.
        """



        if not room_name.strip():

            raise ValueError(
                "room_name cannot be empty."
            )



        if expires_minutes <= 0:

            raise ValueError(
                "expires_minutes must be positive."
            )



        participant_identity = (
            identity.strip()
            if identity is not None
            else f"learner-{uuid.uuid4()}"
        )



        if not participant_identity:

            raise ValueError(
                "identity cannot be empty."
            )



        api_key = os.getenv(
            "LIVEKIT_API_KEY"
        )



        api_secret = os.getenv(
            "LIVEKIT_API_SECRET"
        )



        if not api_key:

            raise RuntimeError(
                "LIVEKIT_API_KEY is not configured."
            )



        if not api_secret:

            raise RuntimeError(
                "LIVEKIT_API_SECRET is not configured."
            )



        expires_at = (
            datetime.now(
                timezone.utc
            )
            + timedelta(
                minutes=expires_minutes,
            )
        )



        access_token = (
            api.AccessToken(
                api_key=api_key,
                api_secret=api_secret,
            )
            .with_identity(
                participant_identity,
            )
            .with_name(
                participant_identity,
            )
            .with_ttl(
                timedelta(
                    minutes=expires_minutes,
                )
            )
            .with_grants(
                api.VideoGrants(
                    room_join=True,
                    room=room_name,
                    can_publish=True,
                    can_subscribe=True,
                    can_publish_data=True,
                )
            )
        )



        token = (
            access_token.to_jwt()
        )



        return VoiceConnectionToken(
            token=token,
            room_name=room_name,
            provider="livekit",
            expires_at=expires_at,
        )



__all__ = [
    "LiveKitTokenService",
]
