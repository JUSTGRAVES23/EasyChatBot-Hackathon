from __future__ import annotations

import os

from sqlalchemy import create_engine
from sqlalchemy.orm import DeclarativeBase, sessionmaker, Session


class Base(DeclarativeBase):
    pass


def database_url() -> str:
    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for Beta Telegram persistence."
        )
    return value


def create_database_engine():
    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(engine):
    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )

"""
LLM Provider Contract.

Defines abstraction for language model providers.

Supported providers:
- OpenAI
- Ollama
- Future enterprise providers

Does NOT:
- Manage conversations.
- Manage users.
- Manage memory.
- Manage LiveKit.
"""


from __future__ import annotations


from abc import ABC, abstractmethod


class LLMProvider(
    ABC
):
    """
    Abstract LLM provider.
    """


    @abstractmethod
    async def generate(
        self,
        *,
        system_prompt: str,
        user_message: str,
    ) -> str:
        """
        Generate model response.
        """

        raise NotImplementedError

"""
LLM Provider Factory.

Responsible only for selecting the configured LLM provider.

Supported providers:
- openai
- ollama

Configuration:

LLM_PROVIDER=openai

or

LLM_PROVIDER=ollama
"""

from __future__ import annotations

import os

from backend.app.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.app.providers.llm.openai_llm_provider import (
    OpenAILLMProvider,
)

from backend.app.providers.llm.ollama_llm_provider import (
    OllamaLLMProvider,
)


class LLMProviderFactory:
    """
    Factory for creating LLM providers.

    Provider selection is configuration driven.
    """

    @staticmethod
    def create() -> LLMProvider:

        provider = os.getenv(
            "LLM_PROVIDER",
            "ollama",
        ).lower()


        if provider == "openai":

            return OpenAILLMProvider()


        if provider == "ollama":

            return OllamaLLMProvider()


        raise ValueError(
            f"Unsupported LLM provider: {provider}"
        )

"""
Ollama LLM Provider.

Provides LLM execution through Ollama runtime.

Configuration:

OLLAMA_BASE_URL=http://localhost:11434

OLLAMA_MODEL=gpt-oss:120b-cloud
"""

from __future__ import annotations

import logging
import os
from typing import Any

import requests

from backend.app.providers.llm.llm_provider import (
    LLMProvider,
)


logger = logging.getLogger(__name__)


class OllamaLLMProvider(LLMProvider):
    """
    Ollama implementation of the LLM provider interface.

    Responsibilities:
    - Send prompts to Ollama
    - Return generated text
    - Keep provider details isolated

    Does NOT:
    - contain business logic
    - manage conversations
    - manage Telegram
    - store memory
    """

    def __init__(self) -> None:

        self.base_url = os.getenv(
            "OLLAMA_BASE_URL",
            "http://localhost:11434",
        )

        self.model = os.getenv(
            "OLLAMA_MODEL",
            "gpt-oss:120b-cloud",
        )

        logger.info(
            "Ollama provider initialized. Model=%s",
            self.model,
        )


    async def generate(
        self,
        *,
        system_prompt: str,
        user_message: str,
        **kwargs: Any,
    ) -> str:
        """
        Generate a response from Ollama.

        Compatible with ConversationRuntime contract:

        generate(
            system_prompt="...",
            user_message="..."
        )
        """

        if not user_message:

            raise ValueError(
                "User message cannot be empty."
            )


        prompt = f"""
System:
{system_prompt}

User:
{user_message}

Assistant:
"""


        payload = {
            "model": self.model,
            "prompt": prompt,
            "stream": False,
        }


        try:

            response = requests.post(
                f"{self.base_url}/api/generate",
                json=payload,
                timeout=300,
            )

            response.raise_for_status()


        except requests.RequestException as exc:

            logger.exception(
                "Ollama request failed."
            )

            raise RuntimeError(
                "Ollama provider unavailable."
            ) from exc


        data = response.json()


        result = data.get(
            "response"
        )


        if not result:

            raise RuntimeError(
                "Ollama returned empty response."
            )


        return result.strip()

"""
OpenAI LLM Provider.

Infrastructure implementation of LLM provider.
"""


from __future__ import annotations


import os


from openai import AsyncOpenAI


from app.providers.llm.llm_provider import (
    LLMProvider,
)



class OpenAILLMProvider(
    LLMProvider
):


    def __init__(
        self,
    ):
        api_key = os.getenv(
            "OPENAI_API_KEY"
        )


        if not api_key:
            raise RuntimeError(
                "OPENAI_API_KEY missing"
            )


        self.client = AsyncOpenAI(
            api_key=api_key
        )



    async def generate(
        self,
        *,
        system_prompt: str,
        user_message: str,
    ) -> str:


        response = await self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {
                    "role": "system",
                    "content": system_prompt,
                },
                {
                    "role": "user",
                    "content": user_message,
                },
            ],
        )


        return (
            response
            .choices[0]
            .message
            .content
            or ""
        )

from __future__ import annotations

import logging
import os
from typing import Any

from telegram import Update
from telegram.ext import (
    Application,
    CommandHandler,
    ContextTypes,
    MessageHandler,
    filters,
)

from backend.app.interfaces.telegram.beta_telegram_adapter import (
    BetaTelegramAdapter,
)


logger = logging.getLogger(__name__)


class TelegramBotRuntime:
    """
    Infrastructure adapter for python-telegram-bot.

    Responsibilities:
    - Telegram application lifecycle;
    - Telegram handler registration;
    - delegation to BetaTelegramAdapter.

    The runtime MUST NOT:
    - perform Beta authorization;
    - execute conversation logic;
    - deliver classroom responses returned by the adapter.

    IMPORTANT RESPONSE OWNERSHIP:

        Telegram update
              |
              v
        TelegramBotRuntime
              |
              v
        BetaTelegramAdapter
              |
              +----> reply_text()
              |
              v
          response

    The adapter owns Telegram classroom response delivery.

    Therefore the runtime MUST NOT call reply_text() with the
    response returned by the adapter.

    This prevents duplicate Telegram messages.
    """

    def __init__(
        self,
        *,
        adapter: BetaTelegramAdapter,
        conversation_boundary: Any = None,
    ) -> None:
        if adapter is None:
            raise ValueError(
                "adapter cannot be None."
            )

        self._adapter = adapter

        # Retained for constructor compatibility with existing
        # composition. Runtime does not execute the boundary.
        self._conversation_boundary = conversation_boundary

        self._application: Application | None = None
        self._running = False

    # =========================================================
    # Application construction
    # =========================================================

    def build(self) -> Application:
        token = os.getenv(
            "TELEGRAM_BOT_TOKEN"
        )

        if not token:
            raise RuntimeError(
                "TELEGRAM_BOT_TOKEN is missing."
            )

        application = (
            Application
            .builder()
            .token(token)
            .build()
        )

        application.add_handler(
            CommandHandler(
                "start",
                self._handle_start,
            )
        )

        application.add_handler(
            MessageHandler(
                filters.ChatType.PRIVATE
                & filters.TEXT
                & ~filters.COMMAND,
                self._handle_private_message,
            )
        )

        self._application = application

        logger.info(
            "Telegram bot runtime created."
        )

        return application

    # =========================================================
    # Lifecycle
    # =========================================================

    async def start(self) -> None:
        if self._running:
            logger.warning(
                "Telegram runtime already running."
            )
            return

        if self._application is None:
            self.build()

        assert self._application is not None

        await self._application.initialize()
        await self._application.start()

        if self._application.updater is not None:
            await self._application.updater.start_polling(
                drop_pending_updates=False,
            )

        self._running = True

        logger.info(
            "Telegram bot polling started."
        )

    async def stop(self) -> None:
        if self._application is None:
            return

        if (
            self._application.updater is not None
            and self._running
        ):
            await self._application.updater.stop()

        await self._application.stop()
        await self._application.shutdown()

        self._running = False

        logger.info(
            "Telegram bot runtime stopped."
        )

    # =========================================================
    # Telegram handlers
    # =========================================================

    async def _handle_start(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        await self._adapter.handle_start(
            update,
            context,
        )

    async def _handle_private_message(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        if (
            update.effective_user is None
            or update.effective_chat is None
            or update.effective_message is None
        ):
            return

        try:
            # =================================================
            # SINGLE DELEGATION POINT
            # =================================================
            #
            # BetaTelegramAdapter:
            #
            #   1. validates transport
            #   2. authorizes access
            #   3. invokes conversation boundary
            #   4. sends Telegram response
            #   5. returns response for observation/testing
            #
            # The returned value MUST NOT be sent again.
            #

            await self._adapter.handle_message(
                update,
                context,
            )

        except Exception:
            logger.exception(
                "Unexpected Telegram adapter failure."
            )

            # This is ONLY an infrastructure-level fallback for
            # failures escaping the adapter itself.
            #
            # Normal conversation failures are already contained
            # by BetaTelegramAdapter.
            await update.effective_message.reply_text(
                "EasyChatBot is temporarily unavailable. "
                "Please try again."
            )
