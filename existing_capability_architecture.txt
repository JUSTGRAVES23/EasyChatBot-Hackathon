"""
================================================================================
File: runtime/capabilities/capability_registry.py

Purpose:
Maintains the mapping between runtime capabilities and registered AI
Agents.

The CapabilityRegistry is responsible solely for storing and retrieving
agent registrations. It does not perform agent selection; that
responsibility belongs to the CapabilityResolver.

Author:
EasyChatBot AI Runtime
================================================================================
"""

from __future__ import annotations

import logging
from collections import defaultdict

from runtime.contracts.agent import Agent

logger = logging.getLogger(__name__)


class CapabilityRegistry:
    """
    Registry of runtime capabilities.
    """

    def __init__(self) -> None:
        self._capabilities: dict[str, list[Agent]] = defaultdict(list)

    def register(
        self,
        agent: Agent,
    ) -> None:
        """
        Register an agent for its declared capability.

        Parameters
        ----------
        agent:
            Agent implementing the runtime contract.

        Raises
        ------
        ValueError
            If the capability is empty.
        """

        capability = agent.descriptor.capability

        if not capability:
            raise ValueError(
                "Agent capability cannot be empty."
            )

        registered_agents = self._capabilities[capability]

        if any(
            existing.descriptor.id == agent.descriptor.id
            for existing in registered_agents
        ):
            logger.debug(
                "Agent '%s' already registered for capability '%s'.",
                agent.descriptor.id,
                capability,
            )
            return

        registered_agents.append(agent)

        logger.info(
            "Registered agent '%s' for capability '%s'.",
            agent.descriptor.id,
            capability,
        )

    def unregister(
        self,
        agent: Agent,
    ) -> None:
        """
        Remove an agent from the registry.
        """

        capability = agent.descriptor.capability

        if capability not in self._capabilities:
            return

        self._capabilities[capability] = [
            registered
            for registered in self._capabilities[capability]
            if registered.descriptor.id != agent.descriptor.id
        ]

        if not self._capabilities[capability]:
            del self._capabilities[capability]

        logger.info(
            "Unregistered agent '%s' from capability '%s'.",
            agent.descriptor.id,
            capability,
        )

    def get_agent(
        self,
        capability: str,
    ) -> Agent | None:
        """
        Return the first registered agent for a capability.

        This convenience method is intended for V1 where a capability
        normally maps to a single preferred agent.

        Future versions should prefer CapabilityResolver for advanced
        routing.
        """

        agents = self._capabilities.get(capability)

        if not agents:
            return None

        return agents[0]

    def get_agents(
        self,
        capability: str,
    ) -> list[Agent]:
        """
        Return all agents registered for a capability.
        """

        return list(
            self._capabilities.get(
                capability,
                [],
            )
        )

    def has_capability(
        self,
        capability: str,
    ) -> bool:
        """
        Determine whether a capability is registered.
        """

        return capability in self._capabilities

    def list_capabilities(
        self,
    ) -> list[str]:
        """
        Return all registered capabilities.
        """

        return sorted(self._capabilities.keys())

    def clear(
        self,
    ) -> None:
        """
        Remove all registrations.
        """

        self._capabilities.clear()

        logger.info(
            "Capability registry cleared."
        )

"""
================================================================================
File: runtime/capabilities/capability_resolver.py

Purpose:
Resolves the most appropriate AI Agent for a requested capability.

The CapabilityResolver encapsulates the runtime selection strategy,
separating capability storage (CapabilityRegistry) from agent selection.

The resolver is responsible only for determining which agent should
execute a request. It performs no execution, routing or provider
selection.

Author:
EasyChatBot AI Runtime
================================================================================
"""

from __future__ import annotations

import logging

from runtime.capabilities.capability_registry import CapabilityRegistry
from runtime.contracts.agent import Agent
from runtime.contracts.execution_context import ExecutionContext

logger = logging.getLogger(__name__)


class CapabilityResolver:
    """
    Resolves the most appropriate agent for a requested capability.
    """

    def __init__(
        self,
        registry: CapabilityRegistry,
    ) -> None:
        """
        Initialise the capability resolver.

        Parameters
        ----------
        registry:
            Registry containing all registered runtime agents.
        """

        self._registry = registry

    def resolve(
        self,
        context: ExecutionContext,
    ) -> Agent:
        """
        Resolve the appropriate agent for the execution request.

        Parameters
        ----------
        context:
            Immutable execution context.

        Returns
        -------
        Agent
            Selected runtime agent.

        Raises
        ------
        ValueError
            If no capability has been specified.

        LookupError
            If no agent is registered for the requested capability.
        """

        capability = context.capability.strip()

        if not capability:
            raise ValueError(
                "ExecutionContext.capability cannot be empty."
            )

        agent = self._registry.get_agent(capability)

        if agent is None:
            raise LookupError(
                f"No agent registered for capability '{capability}'."
            )

        logger.debug(
            (
                "Resolved capability '%s' to agent '%s' "
                "(version=%s)."
            ),
            capability,
            agent.descriptor.id,
            agent.descriptor.version,
        )

        return agent

from __future__ import annotations

import logging
from pathlib import Path

import yaml

from runtime.capabilities.agent_manifest import AgentManifest

logger = logging.getLogger(__name__)


class AgentProfileLoader:
    """
    Loads Agent profiles from the filesystem.
    """

    def __init__(
        self,
        agents_directory: Path,
    ) -> None:
        self._agents_directory = agents_directory

    def load(
        self,
        agent_id: str,
    ) -> AgentManifest:
        """
        Load an AgentManifest from disk.

        Parameters
        ----------
        agent_id:
            Unique identifier of the agent.

        Returns
        -------
        AgentManifest
            The validated Agent manifest.

        Raises
        ------
        FileNotFoundError
            If the profile does not exist.

        ValueError
            If the profile is empty.

        yaml.YAMLError
            If the YAML file is invalid.

        pydantic.ValidationError
            If the manifest fails schema validation.
        """

        profile_path = (
            self._agents_directory
            / agent_id
            / "profile.yaml"
        )

        if not profile_path.is_file():
            raise FileNotFoundError(
                f"Agent profile not found: {profile_path}"
            )

        logger.info(
            "Loading agent '%s' from '%s'.",
            agent_id,
            profile_path,
        )

        with profile_path.open(
            "r",
            encoding="utf-8",
        ) as stream:
            data = yaml.safe_load(stream)

        if data is None:
            raise ValueError(
                f"Agent profile is empty: {profile_path}"
            )

        return AgentManifest.model_validate(data)

        self,
        loader: AgentProfileLoader,
    ) -> None:
        """

        Parameters
        ----------

        """

        if loader is None:

            raise ValueError(
                "AgentProfileLoader cannot be None."
            )


        self._loader = loader

        self._cache: dict[str, AgentManifest] = {}


        logger.info(
            "AgentManifestRegistry initialised."
        )



    def get(
        self,
        agent_id: str,
    ) -> AgentManifest:
        """

        The manifest is loaded once and cached.

        Parameters
        ----------
        agent_id:

        AgentManifest

        if not isinstance(
            agent_id,

        ):

            raise TypeError(
                "agent_id must be a string."
            )


        if not agent_id.strip():

            raise ValueError(
                "agent_id cannot be empty."
            )



        manifest = self._cache.get(
            agent_id,
        )


        if manifest is not None:

            logger.debug(
                "Agent manifest loaded from cache. agent=%s",
                agent_id,
            )

            return manifest



        logger.info(
            "Loading agent manifest. agent=%s",
            agent_id,
        )


        manifest = self._loader.load(
            agent_id,
        )


        self._cache[agent_id] = manifest


        return manifest

        return (
            agent_id

        """

        logger.info(

        return len(

        return skill_id in self._cache

        logger.info(
            "Clearing Skill Registry cache."

        self._cache.clear()

    def count(
        self,
    ) -> int:
        """
        Return the number of cached Skill manifests.

        return len(self._cache)

class CapabilityExecutor:
    """
    Runtime service responsible for capability execution coordination.

    This class provides the application boundary between:

        Capability Runtime

    and

        Agent Runtime


    It owns:

    - Capability validation.
    - Capability resolution.
    - Execution delegation.


    It does not own:

    - AI reasoning.
    - Providers.
    - Prompts.
    - Memory.
    """



    def __init__(
        self,
        *,
        resolver: CapabilityResolver,
        agent_execution_service: AgentExecutionService,
    ) -> None:
        """
        Initialise CapabilityExecutor.

        Parameters
        ----------
        resolver:
            Capability resolution service.

        agent_execution_service:
            Runtime agent execution boundary.
        """


        if resolver is None:

            raise ValueError(
                "CapabilityResolver cannot be None."
            )



        if agent_execution_service is None:

            raise ValueError(
                "AgentExecutionService cannot be None."
            )



        self._resolver = resolver

        self._agent_execution_service = (
            agent_execution_service
        )



        logger.info(
            "CapabilityExecutor initialised."
        )



    async def execute(
        self,
        *,
        request: CapabilityExecutionRequest,
        context: RuntimeContext,
    ) -> AgentExecutionResponse:
        """
        Execute Runtime capability.

        Parameters
        ----------
        request:
            Capability execution request.

        context:
            Runtime execution context.


        Returns
        -------
        AgentExecutionResponse
            Normalized capability execution response.
        """


        if request is None:

            raise ValueError(
                "CapabilityExecutionRequest cannot be None."
            )



        if context is None:

            raise ValueError(
                "RuntimeContext cannot be None."
            )



        request.validate()



        logger.info(
            (
                "Executing capability. "
                "capability=%s execution_id=%s"
            ),
            request.capability_id,
            context.execution_id,
        )



        capability = (
            self._resolver.resolve(

                The capability's default configuration.

                The final configuration to be executed by the Runtime.

            return ExecutionConfiguration(
                capability_id=capability.id,
                provider=capability.default_provider,
                model=capability.default_model,
                temperature=capability.default_temperature,
                reasoning=capability.default_reasoning,
                memory_enabled=capability.default_memory_enabled,
            )

        return ExecutionConfiguration(
            capability_id=capability.id,
            provider=(
                user_settings.provider
                if user_settings.provider is not None
                else capability.default_provider
            ),
            model=(
                user_settings.model

                else capability.default_model

from collections.abc import Iterator


from runtime.contracts.capability_definition import (
    CapabilityDefinition,
)



logger = logging.getLogger(__name__)




class CapabilityRegistry:
    """
    Central catalogue of Runtime capabilities.

    Stores capability definitions only.

    It does not:

    - Instantiate agents.
    - Execute workflows.
    - Call providers.
    - Generate AI responses.
    """



    def __init__(
        self,
    ) -> None:
        """
        Initialise empty capability registry.
        """

        self._capabilities: dict[
            str,
            CapabilityDefinition,
        ] = {}


        logger.info(
            "CapabilityRegistry initialised."
        )



    # ==========================================================================
    # Registration
    # ==========================================================================


    def register(
        self,
        capability: CapabilityDefinition,
    ) -> None:
        """
        Register a capability definition.

        Parameters
        ----------
        capability:
            Capability definition.

        Raises
        ------
        TypeError:
            If invalid object type.

        ValueError:
            If duplicate capability exists.
        """

        if not isinstance(
            capability,
            CapabilityDefinition,
        ):

            raise TypeError(
                "capability must be CapabilityDefinition."
            )


        capability.validate()


        capability_id = (
            capability.capability_id.strip()
        )


        if capability_id in self._capabilities:

            raise ValueError(
                (
                    f"Capability '{capability_id}' "
                    "is already registered."
                )
            )


        self._capabilities[capability_id] = (
            capability
        )


        logger.info(
            "Capability registered. capability=%s",
            capability_id,
        )



    # ==========================================================================
    # Resolution
    # ==========================================================================


    def resolve(
        self,
        capability_id: str,
    ) -> CapabilityDefinition:
        """
        Resolve capability definition.

        Parameters
        ----------
        capability_id:
            Capability identifier.

        Returns
        -------
        CapabilityDefinition
            Registered capability.

        Raises
        ------
        KeyError:
            Capability does not exist.
        """

        if not isinstance(
            capability_id,
            str,
        ):

            raise TypeError(
                "capability_id must be string."
            )


        capability_id = (
            capability_id.strip()
        )


        if not capability_id:

            raise ValueError(
                "capability_id cannot be empty."
            )


        try:

            capability = (
                self._capabilities[
                    capability_id
                ]
            )


        except KeyError as exc:

            raise KeyError(
                (
                    f"Unknown capability "
                    f"'{capability_id}'."
                )
            ) from exc



        logger.debug(
            "Capability resolved. capability=%s",
            capability_id,
        )


        return capability



    # ==========================================================================
    # Queries
    # ==========================================================================


    def contains(
        self,
        capability_id: str,
    ) -> bool:
        """
        Check whether capability exists.
        """

        if not isinstance(
            capability_id,
            str,
        ):

            return False


        return (
            capability_id.strip()
            in
            self._capabilities
        )



    def all(
        self,
    ) -> tuple[CapabilityDefinition, ...]:
        """
        Return immutable capability snapshot.
        """

        return tuple(
            self._capabilities.values()
        )



    def count(
        self,
    ) -> int:
        """
        Return registered capability count.
        """

        return len(
            self._capabilities
        )



    # ==========================================================================
    # Lifecycle
    # ==========================================================================


    def remove(
        self,
        capability_id: str,
    ) -> None:
        """
        Remove capability definition.

        Parameters
        ----------
        capability_id:
            Capability identifier.
        """

        if not isinstance(
            capability_id,
            str,
        ):

            raise TypeError(
                "capability_id must be string."
            )


        capability_id = (
            capability_id.strip()
        )


        if capability_id not in self._capabilities:

            raise KeyError(
                (
                    f"Unknown capability "
                    f"'{capability_id}'."
                )
            )


        del self._capabilities[
            capability_id
        ]


        logger.info(
            "Capability removed. capability=%s",
            capability_id,
        )



    def clear(
        self,
    ) -> None:
        """
        Remove all capabilities.
        """

        self._capabilities.clear()


        logger.info(
            "Capability registry cleared."
        )



    # ==========================================================================
    # Python Protocols
    # ==========================================================================


    def __contains__(
        self,
        capability_id: str,
    ) -> bool:
        """
        Support membership testing.
        """

        return self.contains(
            capability_id,
        )



    def __len__(
        self,
    ) -> int:
        """
        Return capability count.
        """

        return self.count()



    def __iter__(
        self,
    ) -> Iterator[CapabilityDefinition]:
        """
        Iterate through registered capabilities.
        """

        return iter(
            self._capabilities.values()
        )



    def __repr__(
        self,
    ) -> str:
        """
        Developer representation.
        """

        return (
            f"{self.__class__.__name__}"
            f"(capabilities={len(self)})"
        )

    """

            capability_id,
            str,
