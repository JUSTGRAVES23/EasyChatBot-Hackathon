"""
Voice Calls API Router.

HTTP boundary for preparing
character voice calls.

Responsibilities:
- Receive API requests.
- Validate request payload.
- Delegate to application services.
- Return HTTP responses.

Does NOT:
- Execute AI agents.
- Connect to LiveKit directly.
- Manage conversations.
- Select providers.
- Create runtime dependencies.
- Build domain execution contexts.
"""

from __future__ import annotations


from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
)


from pydantic import BaseModel


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.voice.voice_call_lifecycle_service import (
    VoiceCallLifecycleService,
)


from app.application.uai.voice.voice_call_context_factory import (
    VoiceCallContextFactory,
)


from app.application.composition.voice_lifecycle_composition import (
    build_voice_call_lifecycle_service,
)



router = APIRouter(
    prefix="/voice",
    tags=[
        "voice",
    ],
)



class VoiceCallRequestModel(
    BaseModel
):
    """
    HTTP request model.
    """

    learner_id: str

    world_id: str

    character_id: str

    language_code: str



class VoiceCallResponseModel(
    BaseModel
):
    """
    HTTP response model.
    """

    room_name: str

    character_id: str

    language_code: str

    provider: str

    state: str

    token: str

    expires_at: str



def get_voice_call_lifecycle_service(
) -> VoiceCallLifecycleService:
    """
    Dependency provider.

    Composition decisions remain
    outside the HTTP adapter.
    """

    return (
        build_voice_call_lifecycle_service()
    )



def get_voice_call_context_factory(
) -> VoiceCallContextFactory:
    """
    Dependency provider for context creation.

    Domain creation remains inside
    application layer.
    """

    return VoiceCallContextFactory()



@router.post(
    "/call",
    response_model=VoiceCallResponseModel,
)
def prepare_voice_call(
    request: VoiceCallRequestModel,
    service: VoiceCallLifecycleService = Depends(
        get_voice_call_lifecycle_service
    ),
    context_factory: VoiceCallContextFactory = Depends(
        get_voice_call_context_factory
    ),
) -> VoiceCallResponseModel:
    """
    Prepare and start a character voice call.
    """


    try:

        call_request = VoiceCallRequest(
            learner_id=request.learner_id,
            world_id=request.world_id,
            character_id=request.character_id,
            language_code=request.language_code,
        )


        context = (
            context_factory.create(
                request=call_request,
            )
        )


        result = (
            service.start_call(
                context=context,
            )
        )


        room = result["room"]

        session = result["session"]

        connection = result["connection"]


        return VoiceCallResponseModel(
            room_name=room["room_name"],
            character_id=session.character_id,
            language_code=session.language_code,
            provider=room["provider"],
            state=session.state.value,
            token=connection.token,
            expires_at=(
                connection.expires_at
                .isoformat()
            ),
        )


    except ValueError as exc:

        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc



__all__ = [
    "router",
]
"""
Voice Call Trigger API.

HTTP boundary for starting
voice call invitation workflows.

Responsibilities:
- Receive trigger requests.
- Validate input.
- Delegate to application layer.
- Return response.

Does NOT:
- Execute voice runtime.
- Manage providers.
- Send notifications.
- Persist data.
"""

from __future__ import annotations


from fastapi import (
    APIRouter,
    Depends,
)


from pydantic import BaseModel


from app.application.uai.voice.voice_call_trigger_service import (
    VoiceCallTriggerService,
)


from app.application.composition.voice_call_trigger_composition import (
    build_voice_call_trigger_service,
)


from app.domain.uai.notifications.voice_call_notification import (
    VoiceCallNotificationChannel,
)



router = APIRouter(
    prefix="/voice",
    tags=[
        "voice-trigger",
    ],
)



class VoiceCallTriggerRequest(
    BaseModel,
):
    """
    Trigger request payload.
    """

    learner_id: str

    character_id: str

    world_id: str

    language_code: str

    channel: VoiceCallNotificationChannel = (
        VoiceCallNotificationChannel.INBOX
    )



class VoiceCallTriggerResponse(
    BaseModel,
):
    """
    Trigger response payload.
    """

    voice_call_id: str

    invitation_id: str

    notification_id: str



def get_voice_call_trigger_service(
) -> VoiceCallTriggerService:
    """
    Dependency provider.
    """

    return (
        build_voice_call_trigger_service()
    )



@router.post(
    "/trigger",
    response_model=VoiceCallTriggerResponse,
)
def trigger_voice_call(
    request: VoiceCallTriggerRequest,
    service: VoiceCallTriggerService = Depends(
        get_voice_call_trigger_service
    ),
) -> VoiceCallTriggerResponse:
    """
    Trigger a voice call workflow.
    """


    result = (
        service.trigger_voice_call(
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
            notification_channel=request.channel,
        )
    )


    return VoiceCallTriggerResponse(
        voice_call_id=(
            result["voice_call"].id
        ),
        invitation_id=(
            result["invitation"].id
        ),
        notification_id=(
            result["notification"].id
        ),
    )

"""
Voice Notification Acceptance API.

HTTP boundary for accepting
voice call notifications.

Responsibilities:
- Receive acceptance request.
- Delegate workflow.
- Return stable API contract.

Does NOT:
- Execute runtime.
- Manage providers.
- Persist data.
"""

from __future__ import annotations


from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
)


from app.application.composition.voice_call_acceptance_composition import (
    build_voice_call_acceptance_service,
)


from app.application.uai.voice.voice_call_acceptance_service import (
    VoiceCallAcceptanceService,
)


from app.api.schemas.voice_call import (
    VoiceCallAcceptanceResponseSchema,
)



router = APIRouter(
    prefix="/voice",
    tags=[
        "voice-notification",
    ],
)



def get_voice_call_acceptance_service(
) -> VoiceCallAcceptanceService:
    """
    Dependency provider.
    """

    return (
        build_voice_call_acceptance_service()
    )



@router.post(
    "/notification/{notification_id}/accept",
    response_model=VoiceCallAcceptanceResponseSchema,
)
def accept_voice_notification(
    notification_id: str,
    service: VoiceCallAcceptanceService = Depends(
        get_voice_call_acceptance_service,
    ),
) -> VoiceCallAcceptanceResponseSchema:
    """
    Accept voice notification.
    """


    notification_service = (
        service.voice_call_notification_service
    )


    notification = (
        notification_service
        .get_notification(
            notification_id=notification_id,
        )
    )


    if notification is None:

        raise HTTPException(
            status_code=404,
            detail=(
                "Voice notification not found."
            ),
        )


    try:

        result = (
            service.accept_voice_call(
                notification=notification,
            )
        )


    except ValueError as exc:

        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc



    voice_call = result[
        "voice_call"
    ]


    session = result[
        "session"
    ]



    return VoiceCallAcceptanceResponseSchema(
        voice_call_id=(
            voice_call.id
        ),

        learner_id=(
            voice_call.learner_id
        ),

        character_id=(
            voice_call.character_id
        ),

        session_id=(
            session.session_id
        ),

        state=(
            session.state.value
        ),
    )



__all__ = [
    "router",
]
"""
Voice Call Reconnect API.

HTTP boundary for reconnecting
existing voice calls.

Responsibilities:
- Receive reconnect request.
- Validate lifecycle.
- Delegate application service.
- Return stable response.

Does NOT:
- Manage LiveKit.
- Manage providers.
- Persist directly.
"""

from __future__ import annotations


from datetime import datetime, timezone


from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
)


from app.application.uai.voice.voice_call_management_service import (
    VoiceCallManagementService,
)


from app.application.uai.voice.voice_call_session_service import (
    VoiceCallSessionService,
)


from app.application.composition.voice_lifecycle_composition import (
    build_voice_call_lifecycle_service,
)


from app.api.schemas.voice_call import (
    VoiceReconnectResponseSchema,
)



router = APIRouter(
    prefix="/voice",
    tags=[
        "voice-reconnect",
    ],
)



@router.post(
    "/call/{voice_call_id}/reconnect",
    response_model=VoiceReconnectResponseSchema,
)
def reconnect_voice_call(
    voice_call_id: str,
) -> VoiceReconnectResponseSchema:
    """
    Reconnect existing voice call.
    """

    service = (
        build_voice_call_lifecycle_service()
    )


    voice_call = (
        service.voice_call_management_service
        .get_call(
            voice_call_id=voice_call_id,
        )
    )


    if voice_call is None:

        raise HTTPException(
            status_code=404,
            detail=(
                "Voice call not found."
            ),
        )


    if not service.voice_call_management_service.can_reconnect(
        voice_call_id=voice_call_id,
        now=datetime.now(
            timezone.utc
        ),
    ):

        raise HTTPException(
            status_code=400,
            detail=(
                "Voice call cannot reconnect."
            ),
        )



    session = (
        service.voice_call_session_service
        .reconnect_session(
            voice_call=voice_call,
        )
    )


    return VoiceReconnectResponseSchema(
    voice_call_id=(
        voice_call.id
    ),

    character_id=(
        voice_call.character_id
    ),

    language_code=(
        voice_call.language_code
    ),

    session_id=(
        session.session_id
    ),

    state=(
        session.state.value
    ),
)



__all__ = [
    "router",
]
