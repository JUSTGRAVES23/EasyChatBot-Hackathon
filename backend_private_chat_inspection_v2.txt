from __future__ import annotations

from fastapi import APIRouter, Depends, HTTPException, Request, status

from app.api.dependencies.identity import get_current_principal
from app.api.schemas.private_chat import (
    CreatePrivateChatRequest,
    CreatePrivateChatResponse,
    SendPrivateChatMessageRequest,
    SendPrivateChatMessageResponse,
)
from app.domain.identity.principal import Principal

router = APIRouter(
    prefix="/private-chat",
    tags=["private-chat"],
)


def _get_private_chat_service(request: Request):
    factory = getattr(
        request.app.state,
        "private_chat_service_factory",
        None,
    )
    if factory is None:
        raise RuntimeError("private_chat_service_factory is not configured")
    return factory()


@router.post(
    "/conversations",
    response_model=CreatePrivateChatResponse,
    status_code=status.HTTP_201_CREATED,
)
async def create_private_chat(
    payload: CreatePrivateChatRequest,
    request: Request,
    principal: Principal = Depends(get_current_principal),
) -> CreatePrivateChatResponse:
    service = _get_private_chat_service(request)
    try:
        conversation = service.create_conversation(
            user_id=str(principal.user_id),
            world_id=payload.world_id,
        )
    except ValueError as exc:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=str(exc),
        ) from None
    finally:
        service.close()

    return CreatePrivateChatResponse(
        conversation_id=conversation.conversation_id,
        world_id=conversation.world_id,
    )


@router.post(
    "/messages",
    response_model=SendPrivateChatMessageResponse,
    status_code=status.HTTP_200_OK,
)
async def send_private_chat_message(
    payload: SendPrivateChatMessageRequest,
    request: Request,
    principal: Principal = Depends(get_current_principal),
) -> SendPrivateChatMessageResponse:
    service = _get_private_chat_service(request)
    try:
        result = await service.send_message(
            user_id=str(principal.user_id),
            conversation_id=payload.conversation_id,
            message=payload.message,
        )
    except ValueError as exc:
        detail = str(exc)
        code = (
            status.HTTP_404_NOT_FOUND
            if detail in {"Conversation not found", "World not found"}
            else status.HTTP_400_BAD_REQUEST
        )
        raise HTTPException(
            status_code=code,
            detail=detail,
        ) from None
    finally:
        service.close()

    return SendPrivateChatMessageResponse(
        conversation_id=result.conversation.conversation_id,
        world_id=result.conversation.world_id,
        response=result.assistant_message.content,
        response_language=result.response_language,
    )

from __future__ import annotations

from collections.abc import Callable
from pathlib import Path

from sqlalchemy.orm import Session, sessionmaker

from app.ai_runtime.conversation.conversation_runtime import ConversationRuntime
from app.application.private_chat.service import PrivateChatService
from app.application.uai.language.language_detector import LanguageDetector
from app.application.uai.language.language_response_policy import (
    LanguageResponsePolicy,
)
from app.application.uai.world.world_resolver import WorldResolver
from app.infrastructure.persistence.postgresql_private_chat_repository import (
    PostgreSQLPrivateChatRepository,
)
from app.infrastructure.worlds.yaml_world_repository import YamlWorldRepository


def build_private_chat_service_factory(
    *,
    session_factory: sessionmaker[Session],
    conversation_runtime: ConversationRuntime,
    character_instruction: str,
    language_detector: LanguageDetector | None = None,
) -> Callable[[], PrivateChatService]:
    """Build a request-scoped private-chat service factory.

    Every invocation creates exactly one SQLAlchemy Session. The returned
    service closes that session through its repository when the HTTP request
    finishes. No database session is shared between requests.
    """

    if session_factory is None:
        raise ValueError("session_factory is required.")
    if conversation_runtime is None:
        raise ValueError("conversation_runtime is required.")
    if not character_instruction or not character_instruction.strip():
        raise ValueError("character_instruction is required.")

    app_root = Path(__file__).resolve().parents[2]
    world_repository = YamlWorldRepository(
        worlds_path=app_root / "configuration" / "worlds",
    )
    world_resolver = WorldResolver(
        world_repository=world_repository,
    )
    language_response_policy = LanguageResponsePolicy()

    def factory() -> PrivateChatService:
        session = session_factory()
        repository = PostgreSQLPrivateChatRepository(session)

        return PrivateChatService(
            repository=repository,
            world_resolver=world_resolver,
            language_response_policy=language_response_policy,
            language_detector=language_detector,
            conversation_runtime=conversation_runtime,
            character_instruction=character_instruction,
        )

    return factory

from app.application.private_chat.repository import PrivateChatRepository
from app.application.uai.language.language_detector import LanguageDetector
from app.application.uai.language.language_response_policy import (
    LanguageResponsePolicy,
)
from app.application.uai.world.world_resolver import WorldResolver


MAX_PRIVATE_CHAT_MESSAGE_LENGTH = 10_000


class PrivateChatService:
    """Application service for authenticated World private chat.

    The client supplies only the authenticated identity, conversation id,
    and learner message. The World is derived from the server-owned
    conversation record. Provider, model, character, and response language
    remain application-controlled.

    A transaction-scoped repository is supplied by the composition layer.
    The service commits successful persistence and rolls back failed
    persistence without knowing the underlying database implementation.
    """

    def __init__(
        self,
        *,
        repository: PrivateChatRepository,
        world_resolver: WorldResolver,
        language_response_policy: LanguageResponsePolicy,
        language_detector: LanguageDetector | None,
        conversation_runtime: ConversationRuntime,
        character_instruction: str,
    ) -> None:
        if not character_instruction or not character_instruction.strip():
            raise ValueError("character_instruction is required")

        self._repository = repository
        self._world_resolver = world_resolver
        self._language_response_policy = language_response_policy
        self._language_detector = language_detector
        self._conversation_runtime = conversation_runtime
        self._character_instruction = character_instruction.strip()

    def create_conversation(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> PrivateChatConversation:
        if not user_id or not user_id.strip():
            raise ValueError("user_id is required")
        if not world_id or not world_id.strip():
            raise ValueError("world_id is required")

        normalized_user_id = user_id.strip()
        normalized_world_id = world_id.strip()

        self._world_resolver.resolve(normalized_world_id)

        try:
            conversation = self._repository.create_conversation(
                user_id=normalized_user_id,
                world_id=normalized_world_id,
            )
            self._repository.commit()
            return conversation
        except Exception:
            self._repository.rollback()
            raise

    async def send_message(
        self,
        *,
        user_id: str,
        conversation_id: UUID,
        message: str,
    ) -> PrivateChatResponse:
        """Process one authenticated message against its owned World."""

        if not user_id or not user_id.strip():
            raise ValueError("user_id is required")
        if not isinstance(message, str) or not message.strip():
            raise ValueError("message is required")
        if len(message) > MAX_PRIVATE_CHAT_MESSAGE_LENGTH:
            raise ValueError(
                "message exceeds the maximum allowed length of "
                f"{MAX_PRIVATE_CHAT_MESSAGE_LENGTH} characters"
            )

        normalized_user_id = user_id.strip()
        normalized_message = message.strip()

        # World is deliberately derived from the persisted conversation.
        conversation = self._repository.get_conversation_for_user(
            conversation_id=conversation_id,
            user_id=normalized_user_id,
        )
        if conversation is None:
            raise ValueError("Conversation not found")

        world = self._world_resolver.resolve(conversation.world_id)

        response_language = (
            self._language_response_policy.resolve_response_language(
                world=world,
                user_message=normalized_message,
                language_detector=self._language_detector,
            )
        )

        # Runtime/provider execution happens before persistence.
        response = await self._conversation_runtime.respond(
            character_instruction=self._character_instruction,
            learner_message=normalized_message,
        )

        now = datetime.now(timezone.utc)

        user_message = PrivateChatMessage(
            conversation_id=conversation_id,
            role="user",
            content=normalized_message,
            created_at=now,
        )
        assistant_message = PrivateChatMessage(
            conversation_id=conversation_id,
            role="assistant",
            content=response,
            created_at=now,
        )

        try:
            self._repository.save_message(
                conversation_id=conversation_id,
                user_id=normalized_user_id,
                world_id=conversation.world_id,
                message=user_message,
            )
            self._repository.save_message(
                conversation_id=conversation_id,
                user_id=normalized_user_id,
                world_id=conversation.world_id,
                message=assistant_message,
            )
            self._repository.touch_conversation(
                conversation_id=conversation_id,
                user_id=normalized_user_id,
                world_id=conversation.world_id,
            )
            self._repository.commit()
        except Exception:
            self._repository.rollback()
            raise

        updated_conversation = PrivateChatConversation(
            conversation_id=conversation.conversation_id,
            user_id=conversation.user_id,
            world_id=conversation.world_id,
            created_at=conversation.created_at,
            updated_at=now,
        )

        return PrivateChatResponse(
            conversation=updated_conversation,
            user_message=user_message,
            assistant_message=assistant_message,
            response_language=response_language,
        )

    def close(self) -> None:
        """Release the request-scoped persistence resource."""
        self._repository.close()

from __future__ import annotations

from datetime import datetime, timezone
from uuid import UUID

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.application.private_chat.models import (
    PrivateChatConversation,
    PrivateChatMessage,
)
from app.application.private_chat.repository import PrivateChatRepository
from app.infrastructure.persistence.private_chat_models import (
    PrivateChatConversationRecord,
    PrivateChatMessageRecord,
)


class PostgreSQLPrivateChatRepository(PrivateChatRepository):
    """PostgreSQL adapter for authenticated private-chat persistence."""

    def __init__(self, session: Session) -> None:
        self._session = session

    def get_conversation(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
    ) -> PrivateChatConversation | None:
        """
        Return a conversation only when it belongs to the supplied user
        and selected World.

        Conversation ownership and World binding are therefore enforced
        by the persistence query rather than by caller-side filtering.
        """
        record = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
                PrivateChatConversationRecord.world_id == world_id,
            )
        )

        if record is None:
            return None

        return self._to_conversation(record)

    def get_conversation_for_user(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
    ) -> PrivateChatConversation | None:
        """
        Return a conversation owned by the supplied user.

        This method remains available for the existing application
        repository contract. It deliberately does not require a World
        because that is the historical repository-level lookup contract.
        """
        record = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
            )
        )

        if record is None:
            return None

        return self._to_conversation(record)

    def create_conversation(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> PrivateChatConversation:
        now = datetime.now(timezone.utc)

        record = PrivateChatConversationRecord(
            user_id=user_id,
            world_id=world_id,
            created_at=now,
            updated_at=now,
        )

        self._session.add(record)
        self._session.flush()

        return self._to_conversation(record)

    def save_message(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        message: PrivateChatMessage,
    ) -> PrivateChatMessage:
        record = PrivateChatMessageRecord(
            conversation_id=conversation_id,
            user_id=user_id,
            world_id=world_id,
            role=message.role,
            content=message.content,
            created_at=message.created_at,
        )

        self._session.add(record)
        self._session.flush()

        return self._to_message(record)

    def get_recent_messages(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        limit: int = 20,
    ) -> list[PrivateChatMessage]:
        if limit < 1:
            raise ValueError("limit must be greater than zero")

        records = self._session.scalars(
            select(PrivateChatMessageRecord)
            .where(
                PrivateChatMessageRecord.conversation_id
                == conversation_id,
                PrivateChatMessageRecord.user_id == user_id,
                PrivateChatMessageRecord.world_id == world_id,
            )
            .order_by(
                PrivateChatMessageRecord.created_at.desc()
            )
            .limit(limit)
        ).all()

        return [
            self._to_message(record)
            for record in reversed(records)
        ]

    def touch_conversation(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
    ) -> None:
        record = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
                PrivateChatConversationRecord.world_id == world_id,
            )
        )

        if record is None:
            raise ValueError(
                "Private chat conversation not found"
            )

        record.updated_at = datetime.now(timezone.utc)
        self._session.flush()

    def commit(self) -> None:
        self._session.commit()

    def rollback(self) -> None:
        self._session.rollback()

    def close(self) -> None:
        self._session.close()

    @staticmethod
    def _to_conversation(
        record: PrivateChatConversationRecord,
    ) -> PrivateChatConversation:
        return PrivateChatConversation(
            conversation_id=record.conversation_id,
            user_id=record.user_id,
            world_id=record.world_id,
            created_at=record.created_at,
            updated_at=record.updated_at,
        )

    @staticmethod
    def _to_message(
        record: PrivateChatMessageRecord,
    ) -> PrivateChatMessage:
        return PrivateChatMessage(
            conversation_id=record.conversation_id,
            role=record.role,
            content=record.content,
            created_at=record.created_at,
        )

from __future__ import annotations

from app.application.beta_access.beta_access_service import (
    BetaAccessService,
)


class BetaPrivateChatGate:
    """Authorization boundary for Beta classroom Telegram interactions.

    Telegram transport is allowed to continue only when:
    - the interaction is a private chat; and
    - BetaAccessService confirms the Telegram identity is authorized.

    World onboarding/selection is deliberately outside this gate.

    This is the canonical Beta 1 / Beta 2 Telegram private-chat
    authorization boundary.
    """

    def __init__(
        self,
        *,
        beta_access_service: BetaAccessService,
    ) -> None:
        if beta_access_service is None:
            raise ValueError(
                "beta_access_service cannot be None."
            )

        self._access = beta_access_service

    async def authorize(
        self,
        *,
        telegram_user_id: int,
        chat_type: str = "private",
    ) -> bool:
        """Return whether the Telegram user may use the private classroom."""

        if chat_type != "private":
            return False

        try:
            await self._access.authorize_private_chat(
                telegram_user_id,
            )
        except (ValueError, PermissionError):
            return False

        return True

    async def require(
        self,
        *,
        telegram_user_id: int,
        chat_type: str = "private",
    ):
        """Require private-chat authorization or raise PermissionError."""

        if chat_type != "private":
            raise PermissionError(
                "Beta classroom access is only available "
                "in a private Telegram chat."
            )

        return await self._access.authorize_private_chat(
            telegram_user_id,
        )


__all__ = [
    "BetaPrivateChatGate",
]

from __future__ import annotations

from sqlalchemy import select

from app.application.conversation.conversation_event_models import (
    ConversationEvent,

from app.application.conversation.conversation_event_repository import (
    ConversationEventRepository,

from .conversation_event_records import (

class PostgreSQLConversationEventRepository(
    ConversationEventRepository
):
    """
    PostgreSQL implementation of conversation event storage.

    Privacy isolation:

        telegram_user_hash
                +
          conversation_id
                +
             world_id

    Governance:

        privacy_hash_version
                +
        retention_policy_id

    Responsibility:

    - persist conversation events
    - retrieve conversation history

    Does NOT:

    - validate business rules
    - create memories
    - analyse learning behaviour
    - create RAG knowledge
    """

    def __init__(
        self,
        session_factory,
    ) -> None:
        if session_factory is None:
            raise ValueError(
                "session_factory cannot be None."
            )

        self._session_factory = session_factory

    async def save(
        self,
        event: ConversationEvent,
    ) -> ConversationEvent:
        """
        Persist an anonymised conversation event.

        Privacy and retention governance values are taken
        from the validated event metadata and persisted into
        their dedicated database columns.
        """

        metadata = event.metadata or {}

        privacy_hash_version = metadata.get(
            "privacy_hash_version",
        )

        retention_policy_id = metadata.get(
            "retention_policy_id",
        )

        if not privacy_hash_version:
            raise ValueError(
                "Conversation event is missing "
                "privacy_hash_version."
            )

        if not retention_policy_id:
            raise ValueError(
                "Conversation event is missing "
                "retention_policy_id."
            )

        with self._session_factory() as session:
            record = ConversationEventRecord(
                id=event.event_id,
                conversation_id=event.conversation_id,
                telegram_user_hash=(
                    event.telegram_user_hash
                ),

    ) -> list[ConversationEvent]:
        """

        pseudonymised Telegram identity, conversation,

class PostgreSQLPrivateChatRepository(PrivateChatRepository):
    """PostgreSQL adapter for authenticated private-chat persistence."""

    def __init__(self, session: Session) -> None:
        self._session = session

    def get_conversation(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
    ) -> PrivateChatConversation | None:
        """
        Return a conversation only when it belongs to the supplied user
        and selected World.

        Conversation ownership and World binding are therefore enforced
        by the persistence query rather than by caller-side filtering.
        """
        record = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
                PrivateChatConversationRecord.world_id == world_id,
            )
        )

        if record is None:
            return None

        return self._to_conversation(record)

    def get_conversation_for_user(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
    ) -> PrivateChatConversation | None:
        """
        Return a conversation owned by the supplied user.

        This method remains available for the existing application
        repository contract. It deliberately does not require a World
        because that is the historical repository-level lookup contract.
        """
        record = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
            )
        )

        if record is None:
            return None

        return self._to_conversation(record)

    def create_conversation(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> PrivateChatConversation:
        now = datetime.now(timezone.utc)

        record = PrivateChatConversationRecord(
            user_id=user_id,
            world_id=world_id,
            created_at=now,
            updated_at=now,
        )

        self._session.add(record)
        self._session.flush()

        return self._to_conversation(record)

    def save_message(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        message: PrivateChatMessage,
    ) -> PrivateChatMessage:
        record = PrivateChatMessageRecord(
            conversation_id=conversation_id,
            user_id=user_id,
            world_id=world_id,
            role=message.role,
            content=message.content,
            created_at=message.created_at,
        )

        self._session.add(record)
        self._session.flush()

        return self._to_message(record)

    def get_recent_messages(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        limit: int = 20,
    ) -> list[PrivateChatMessage]:
        if limit < 1:
            raise ValueError("limit must be greater than zero")

        records = self._session.scalars(
            select(PrivateChatMessageRecord)
            .where(
                PrivateChatMessageRecord.conversation_id
                == conversation_id,
                PrivateChatMessageRecord.user_id == user_id,
                PrivateChatMessageRecord.world_id == world_id,
            )
            .order_by(
                PrivateChatMessageRecord.created_at.desc()
            )
            .limit(limit)
        ).all()

        return [
            self._to_message(record)
            for record in reversed(records)
        ]

        self,
        *,

        user_id: str,
        world_id: str,

        )

        self,
        *,
        conversation_id: UUID,
        user_id: str,
    ) -> PrivateChatConversation | None:
        conversation = self.conversations.get(conversation_id)
        if conversation is None or conversation.user_id != user_id:
            return None
        return conversation

    def create_conversation(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> PrivateChatConversation:
        now = datetime.now(timezone.utc)
        conversation = PrivateChatConversation(
            conversation_id=uuid4(),
            user_id=user_id,
            world_id=world_id,
            created_at=now,
            updated_at=now,
        )
        self.conversations[conversation.conversation_id] = conversation
        return conversation

    def save_message(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        message: PrivateChatMessage,
    ) -> PrivateChatMessage:
        conversation = self.get_conversation_for_user(
            conversation_id=conversation_id,
            user_id=user_id,
        )
        if conversation is None or conversation.world_id != world_id:
            raise ValueError("Conversation not found")
        self.messages.append(message)
        return message

    def get_recent_messages(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        limit: int = 20,
    ) -> list[PrivateChatMessage]:
        if limit < 1:
            raise ValueError("limit must be greater than zero")
        conversation = self.get_conversation_for_user(
            conversation_id=conversation_id,
            user_id=user_id,
        )
        if conversation is None or conversation.world_id != world_id:
            return []
        return [
            message
            for message in self.messages
            if message.conversation_id == conversation_id
        ][-limit:]

        self,

        user_id: str,
        world_id: str,

            user_id=conversation.user_id,
            world_id=conversation.world_id,
            created_at=conversation.created_at,
            updated_at=datetime.now(timezone.utc),
        )

    def commit(self) -> None:
        self.commit_count += 1

    def rollback(self) -> None:
        self.rollback_count += 1

    def close(self) -> None:
        self.close_count += 1

class FakeConversationEventRepository:
    """
    In-memory conversation-event repository.

    Implements the production repository isolation contract:

        telegram_user_hash
                +
        conversation_id
                +
              world_id
    """

    def __init__(self) -> None:
        self.events: list[Any] = []
        self.get_recent_calls: list[dict[str, Any]] = []

    async def save(self, event):
        self.events.append(event)
        return event

    async def get_recent(
        self,
        *,
        telegram_user_hash: str,
        conversation_id: str,
        world_id: str,
        limit: int = 20,
    ):
        self.get_recent_calls.append(
            {
                "telegram_user_hash": telegram_user_hash,
                "conversation_id": conversation_id,
                "world_id": world_id,
                "limit": limit,
            }
        )

        if limit <= 0:
            return []

        events = [
            event
            for event in self.events
            if (
                event.telegram_user_hash
                == telegram_user_hash
                and event.conversation_id
                == conversation_id
                and event.world_id
                == world_id
            )
        ]

        return events[-limit:]


class FakePrivacyIdentifierService:

class FakeConversationEventValidator:
    """Minimal event governance validator."""

    def __init__(self) -> None:
        self.calls: list[dict[str, Any]] = []

    def validate(
        self,
        *,
        content: str,
        metadata: dict[str, Any],
        telegram_user_hash: str,
    ) -> dict[str, bool]:
        self.calls.append(
            {
                "content": content,
                "metadata": metadata,
                "telegram_user_hash": telegram_user_hash,
            }
        )

        if not content.strip():
            raise ValueError(
                "Conversation content cannot be empty."
            )

        if not telegram_user_hash:
            raise ValueError(
                "Conversation event requires pseudonymous identity."
            )

        if metadata is None:
            raise ValueError(
                "Conversation metadata missing."
            )

        return {"valid": True}

from app.interfaces.telegram.beta_private_chat_gate import (
    BetaPrivateChatGate,
)


class StubAccess:
    def __init__(
        self,
        *,
        active: bool,
    ) -> None:
        self.active = active
        self.calls: list[int] = []

    async def authorize_private_chat(
        self,
        telegram_id: int,
    ):
        self.calls.append(telegram_id)

        if not self.active:
            raise PermissionError(
                "User is not active."
            )

        return object()


def test_private_chat_gate_requires_service():
    with pytest.raises(
        ValueError,
        match="beta_access_service",
    ):
        BetaPrivateChatGate(
            beta_access_service=None,
        )


@pytest.mark.asyncio
async def test_private_chat_gate_rejects_group():
    access = StubAccess(active=True)

    gate = BetaPrivateChatGate(
        beta_access_service=access,
    )

    allowed = await gate.authorize(
        telegram_user_id=123,
        chat_type="group",
    )

    assert allowed is False
    assert access.calls == []


@pytest.mark.asyncio
async def test_private_chat_gate_accepts_active_private_user():
    access = StubAccess(active=True)

    gate = BetaPrivateChatGate(
        beta_access_service=access,
    )

    allowed = await gate.authorize(
        telegram_user_id=123,
        chat_type="private",
    )

    assert allowed is True
    assert access.calls == [123]


@pytest.mark.asyncio
async def test_private_chat_gate_rejects_inactive_private_user():
    access = StubAccess(active=False)

    gate = BetaPrivateChatGate(
        beta_access_service=access,
    )

    allowed = await gate.authorize(
        telegram_user_id=123,
        chat_type="private",
    )

    assert allowed is False
    assert access.calls == [123]
