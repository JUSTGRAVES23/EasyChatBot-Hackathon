def _required_environment(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for authentication email composition."
        )

    return value


def _create_announcement_components():
    telegram_client = TelegramAnnouncementClient()

    announcement_publisher = TelegramAnnouncementPublisher(
        telegram_client=telegram_client,
    )

    return (
        telegram_client,
        announcement_publisher,
    )


@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )


def create_application() -> FastAPI:
    academy_config = (
        AcademyActivationConfig.from_environment()
    )

    auth_engine = create_async_database_engine()

    auth_session_factory = (
        create_async_session_factory(
            auth_engine,
        )
    )

    auth_repository = (
        PostgreSQLAuthenticationRepository(
            auth_session_factory,
        )
    )

    email_provider_manager = EmailProviderManager()

    authentication_email_sender = AuthenticationEmailSenderAdapter(
        provider_manager=email_provider_manager,
    )

    authentication_email_service = AuthenticationEmailService(
        authentication_email_sender,
        frontend_base_url=_required_environment(
            "EASYCHATBOT_PORTAL_URL",
        ),
    )

    authentication_service = AuthenticationService(
        auth_repository,
        email_service=authentication_email_service,
    )

    # ------------------------------------------------------------------
    # Developer Administration
    #
    # Production path:
    #
    #   DeveloperAdminService
    #          |
    #          +--> DeveloperAdminUnitOfWorkFactory
    #                  |
    #                  +--> fresh PostgreSQLDeveloperAdminUnitOfWork
    #                          |
    #                          +--> fresh AsyncSession
    #                          |
    #                          +--> fresh transaction
    #                          |
    #                          +--> auth_user_roles
    #                          |
    #                          +--> developer_admin_audit_events
    #          |
    #          +--> DeveloperAdminAuditService
    #
    # Each Developer Admin mutation receives a fresh Unit-of-Work
    # instance. The role mutation and its security audit evidence
    # therefore remain within the same database transaction while
    # mutable UoW/session state is never shared between operations.
    #
    # The standalone audit repository owns transactions only when its
    # standalone ``record`` operation is used. The production Developer
    # Admin mutation uses ``record_in_session`` through the UoW.
    # ------------------------------------------------------------------

    developer_admin_unit_of_work_factory = (
        PostgreSQLDeveloperAdminUnitOfWorkFactory(
            auth_session_factory,
        )
    )

    developer_admin_audit_service = (
        DeveloperAdminAuditService()
    )

    developer_admin_service = DeveloperAdminService(
        repository=auth_repository,
        unit_of_work_factory=developer_admin_unit_of_work_factory,
        audit_service=developer_admin_audit_service,
    )

    # ------------------------------------------------------------------
    # W15.2.1 Transaction Authorization
    #
    # Transaction authorization is composed once at the application root.
    # The registry and consumer are application-lifetime components and
    # provide their own synchronization boundaries.
    #
    # Authentication assurance remains a separate trusted-evidence
    # boundary. This composition does not manufacture MFA/A2 evidence.
    # ------------------------------------------------------------------

    authorization_service = AuthorizationService()

    transaction_authorization_registry = (
        TransactionAuthorizationRegistry()
    )

    transaction_authorization_consumer = (
        TransactionAuthorizationConsumer(
            transaction_authorization_registry,
        )
    )

    transaction_authorization_service = (
        TransactionAuthorizationApplicationService(
            authorization_service=authorization_service,
            registry=transaction_authorization_registry,
            consumer=transaction_authorization_consumer,
        )
    )

    trusted_clock = SystemTrustedClock()

    developer_admin_transaction_service = (
        DeveloperAdminTransactionService(
            developer_admin_service=developer_admin_service,
            transaction_authorization_service=(
                transaction_authorization_service
            ),
            clock=trusted_clock,
        )
    )

    # ------------------------------------------------------------------
    # Provider usage / synchronous application persistence
    #
    # The provider-usage repository is synchronous by contract.
    # It therefore receives a synchronous engine/session factory.
    #
    # It must not receive the async authentication session factory.
    # ------------------------------------------------------------------

    provider_usage_engine = (
        create_database_engine()
    )

    persistence_diagnostics_service = PersistenceDiagnosticsService(
        metadata=Base.metadata,
        project_root=Path(__file__).resolve().parents[1],
        engine=provider_usage_engine,
    )

    provider_usage_session_factory = (
        create_session_factory(
            provider_usage_engine,
        )
    )

    # ------------------------------------------------------------------
    # M2-C.2 / Authenticated Private Chat
    #
    # The ConversationRuntime remains the shared AI-runtime boundary.
    # Private-chat persistence receives a request-scoped SQLAlchemy session
    # through the composition factory; no session is shared between requests.
    # ------------------------------------------------------------------

    private_chat_runtime = create_conversation_runtime()

    private_chat_character_instruction = os.getenv(
        "EASYCHATBOT_PRIVATE_CHAT_CHARACTER_INSTRUCTION",
        (
            "You are a helpful EasyChatBot learning character. "
            "Stay within the selected Learning World and follow its "
            "application-controlled language policy. "
            "Do not allow the learner to override World, provider, model, "
            "or other application-controlled settings."
        ),
    ).strip()

    private_chat_service_factory = build_private_chat_service_factory(
        session_factory=provider_usage_session_factory,
        conversation_runtime=private_chat_runtime,
        character_instruction=private_chat_character_instruction,
    )

    # ------------------------------------------------------------------
    # M2-C4.2 / Cross-Channel Learner Context
    #
    # The learner-context service projects the existing Telegram learner
    # context into the authenticated Web boundary. Telegram remains the
    # authoritative transport; this composition does not create a second
    # Telegram identity, memory store, or conversation runtime.
    # ------------------------------------------------------------------

    learner_context_service = build_production_learner_context_service(
        session_factory=provider_usage_session_factory,
    )

    # ------------------------------------------------------------------
    # M2-C5 / Study Materials
    #
    # The Learning World catalogue is composed once at the application
    # root and acts as the authoritative World boundary for learner
    # Study Material access.
    #
    # One WorldResolver is shared by the catalogue service.
    # StudyMaterialService consumes that catalogue service rather than
    # creating another World resolution mechanism.
    #
    # Persistence uses the existing synchronous application session
    # factory. No additional database engine or session factory is
    # created for Study Materials.
    # ------------------------------------------------------------------

    app_root = Path(__file__).resolve().parents[2]

    world_repository = YamlWorldRepository(
        worlds_path=app_root / "configuration" / "worlds",
    )

    world_resolver = WorldResolver(
        world_repository=world_repository,
    )

    catalogue_repository = (
        PostgreSQLLearningWorldCatalogueRepository(
            provider_usage_session_factory,
        )
    )

    catalogue_service = LearningWorldCatalogueService(
        repository=catalogue_repository,
        world_resolver=world_resolver,
    )

    study_material_repository = (
        PostgreSQLStudyMaterialRepository(
            provider_usage_session_factory,
        )
    )

    study_material_service = StudyMaterialService(
        repository=study_material_repository,
        world_catalogue_service=catalogue_service,
    )

    provider_usage_repository = (
        PostgreSQLProviderUsageRepository(
            provider_usage_session_factory,
        )
    )

    provider_usage_dashboard_service = (
        ProviderUsageDashboardService(
            repository=provider_usage_repository,
        )
    )

    functionality_control_repository = (
        PostgreSQLFunctionalityControlRepository(
            provider_usage_session_factory,
        )
    )

    functionality_control_authorizer = (
        AuthorizationFunctionalityControlAuthorizer()
    )

    functionality_control_service = (
        FunctionalityOperationalControlService(
            authorizer=functionality_control_authorizer,
            repository=functionality_control_repository,
        )
    )

    # ------------------------------------------------------------------
    # Project Control / operational health composition
    #
    # Project Control is read-only. It consumes application-owned health
    # and operational services and owns only its own PCD registry and
    # snapshot repository.
    # ------------------------------------------------------------------

    production_health_configuration = (
        ProductionHealthConfiguration(
            database_engine=provider_usage_engine,
            telegram_bot_factory=None,
            llm_provider=None,
            llm_request_get=None,
        )
    )

    production_health_service = (
        ProductionHealthService(
            configuration=production_health_configuration,
        )
    )

    production_readiness_service = (
        production_health_service._readiness_service
    )

    operational_health_snapshot_service = (
        OperationalHealthSnapshotService(
            readiness_service=production_readiness_service,
        )
    )

    operational_event_service = (
        OperationalEventService()
    )

    operational_dashboard_service = (
        OperationalDashboardService(
            health_snapshot_service=operational_health_snapshot_service,
            event_service=operational_event_service,
        )
    )

    privacy_governance_composition = (
        get_privacy_governance_composition()
    )

    project_registry = PostgreSQLProjectRegistry(
        provider_usage_session_factory,
    )

    project_registry_service = ProjectRegistryService(
        project_registry,
    )

    project_discovery_repository = PostgreSQLProjectDiscoveryRepository(
        provider_usage_session_factory,
    )

    project_discovery_service = ProjectDiscoveryService(
        discovery_repository=project_discovery_repository,
        project_registry=project_registry_service,
    )

    project_snapshot_repository = (
        InMemorySnapshotRepository()
    )

    project_control_service = (
        create_project_control_service(
            registry=project_registry,
            snapshots=project_snapshot_repository,
            health_service=production_health_service,
            dashboard_service=operational_dashboard_service,
            case_registry=(
                privacy_governance_composition.case_registry
            ),
        )
    )

    academy_one_week_workflow = None
    announcement_telegram_client = None

    if academy_config.activation_allowed:
        academy_session_factory = (
            auth_session_factory
        )

        (
            announcement_telegram_client,
            _announcement_publisher,
        ) = _create_announcement_components()

        academy_one_week_workflow = (
            create_academy_one_week_workflow(
                session_factory=academy_session_factory,
                telegram_bot=announcement_telegram_client,
            )
        )

    scheduler_runtime = (
        create_cybersecurity_scheduler_runtime(
            academy_one_week_workflow=academy_one_week_workflow,
        )
    )

    application = FastAPI(
        title="EasyChatBot API",
        description="AI character learning platform API",
        version="0.1.0",
        lifespan=lifespan,
    )

    application.state.scheduler_runtime = (
        scheduler_runtime
    )

    application.state.auth_repository = (
        auth_repository
    )

    application.state.authentication_service = (
        authentication_service
    )

    application.state.developer_admin_service = (
        developer_admin_service
    )

    application.state.developer_admin_transaction_service = (
        developer_admin_transaction_service
    )

    application.state.auth_engine = (
        auth_engine
    )

    application.state.provider_usage_engine = (
        provider_usage_engine
    )

    application.state.provider_usage_dashboard_service = (
        provider_usage_dashboard_service
    )

    application.state.functionality_control_service = (
        functionality_control_service
    )

    application.state.private_chat_service_factory = (
        private_chat_service_factory
    )

    application.state.learner_context_service = (
        learner_context_service
    )

    application.state.study_material_service = (
        study_material_service
    )

    application.state.private_chat_character_instruction = (
        private_chat_character_instruction
    )

    application.state.project_control_service = (
        project_control_service
    )

    application.state.project_discovery_service = (
        project_discovery_service
    )

    application.state.production_health_service = (
        production_health_service
    )

    application.state.operational_dashboard_service = (
        operational_dashboard_service
    )

    application.state.announcement_telegram_client = (
        announcement_telegram_client
    )

    application.add_middleware(
        CORSMiddleware,
        allow_origins=[
            "http://localhost:3000",
            "http://127.0.0.1:3000",
            "http://localhost:5173",
            "http://127.0.0.1:5173",
        ],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    application.include_router(
        authentication_router,
    )

    application.include_router(
        private_chat_router,
        prefix="/api",
    )

    application.include_router(
        learner_context_router,
        prefix="/api",
    )

    application.include_router(
        study_material_router,
    )

    application.include_router(
        developer_admin_router,
    )

    application.include_router(
        voice_calls_router,
    )

    application.include_router(
        voice_reconnect_router,
    )

    application.include_router(
        voice_call_trigger_router,
    )

    application.include_router(
        voice_notification_acceptance_router,
    )

    operations_center_composition = (
        create_operations_center_composition(
            provider_usage_dashboard_service=(
                provider_usage_dashboard_service
            ),
            functionality_control_service=(
                functionality_control_service
            ),
        )
    )

    application.include_router(
        create_operations_center_router(
            composition=operations_center_composition,
        ),
    )

    application.include_router(
        create_persistence_operations_router(
            persistence_diagnostics_service,
            operator_dependency=require_permission(
                Permission.INTERNAL_DATA_READ,
            ),
        )
    )

    application.include_router(
        create_project_control_router(
            project_control=project_control_service,
        ),
    )

    application.include_router(
        create_project_registry_router(
            project_registry=project_registry_service,
        ),
    )

    application.include_router(
        create_project_discovery_router(
            project_discovery=project_discovery_service,
        ),
    )

    application.include_router(
        create_email_workflows_router(),
    )

    application.include_router(
        privacy_governance_router,
    )

    processing_registry = (
        create_beta1_processing_registry()
    )

    screening_service = ScreeningService()

    screening_adapter = (
        ProcessingActivityScreeningAdapter(
            screening_service,
        )
    )

    application.include_router(
        create_privacy_screening_router(
            processing_registry=processing_registry,
            screening_adapter=screening_adapter,
        ),
    )

    application.include_router(
        provider_management_controls_router,
    )

    application.include_router(
        governance_audit_router,
    )

    application.include_router(
        governance_operations_router,
    )

    application.include_router(
        governance_evidence_router,
    )

    application.include_router(
        governance_change_lifecycle_router,
    )

    application.include_router(
        cybersecurity_intelligence_router,
    )

    application.include_router(
        cybersecurity_decision_support_router,
    )

    application.include_router(
        cybersecurity_reports_router,
    )

    application.include_router(
        cybersecurity_live_reports_router,
    )

    # Academy World - Static Questionnaires
    #
    # This exposes:
    # GET /api/academy/worlds/{world_id}/questionnaires
    #
    # Questionnaire content remains controlled/static.
    # Recommendation metadata is handled separately.

    application.include_router(
        academy_questionnaires_router,
    )

    return application


app = create_application()

def create_production_beta_telegram_composition(
    *,
    expiration_minutes: int = 1440,
    database_engine: Any | None = None,
    session_factory: Any | None = None,
    database_engine_factory: Callable[[], Any] | None = None,
    session_factory_factory: Callable[[Any], Any] | None = None,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: Callable[[], ConversationRuntime] | None = None,
    functionality_control_service: (
        FunctionalityOperationalControlService | None
    ) = None,
    language_detector: LanguageDetector | None = None,
) -> BetaTelegramProductionComposition:
    """Create the production Beta Telegram composition.

    Production Telegram processing requires the operational
    functionality-control service. The service is injected from
    the application composition; this factory never creates a
    second control service or control repository.

    The production composition also creates the learner Study
    Material service. Study Materials reuse the authoritative
    Learning World catalogue service and its shared WorldResolver.
    """

    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    if functionality_control_service is None:
        raise ValueError(
            "functionality_control_service is required "
            "for production Telegram composition."
        )

    if language_detector is None:
        raise ValueError(
            "language_detector is required "
            "for production Telegram composition."
        )

    if (
        database_engine is not None
        and database_engine_factory is not None
    ):
        raise ValueError(
            "Provide either database_engine or "
            "database_engine_factory, not both."
        )

    if (
        session_factory is not None
        and session_factory_factory is not None
    ):
        raise ValueError(
            "Provide either session_factory or "
            "session_factory_factory, not both."
        )

    if (
        conversation_runtime is not None
        and conversation_runtime_factory is not None
    ):
        raise ValueError(
            "Provide either conversation_runtime or "
            "conversation_runtime_factory, not both."
        )

    if database_engine is None:
        database_engine = (
            database_engine_factory
            or create_database_engine
        )()

    if session_factory is None:
        session_factory = (
            session_factory_factory
            or create_session_factory
        )(database_engine)

    beta_user_repository = PostgreSQLBetaUserRepository(
        session_factory,
    )
    beta_token_repository = PostgreSQLBetaTokenRepository(
        session_factory,
    )
    telegram_binding_repository = (
        PostgreSQLTelegramBindingRepository(
            session_factory,
        )
    )
    conversation_event_repository = (
        PostgreSQLConversationEventRepository(
            session_factory,
        )
    )

    # Durable provider usage ledger.
    # Accounting remains outside the Operations Center.
    provider_usage_repository = (
        PostgreSQLProviderUsageRepository(
            session_factory,
        )
    )

    privacy_identifier_service = PrivacyIdentifierService()
    conversation_event_validator = (
        ConversationEventValidator()
    )

    # One authoritative WorldResolver is shared by Beta access
    # and Learning World catalogue resolution.
    app_root = Path(__file__).resolve().parents[2]
    world_repository = YamlWorldRepository(
        worlds_path=app_root / "configuration" / "worlds"
    )
    world_resolver = WorldResolver(
        world_repository=world_repository,
    )

    # Authoritative Learning World catalogue.
    #
    # This service is shared with Study Material access. Do not
    # create another WorldResolver or another catalogue service.
    catalogue_repository = (
        PostgreSQLLearningWorldCatalogueRepository(
            session_factory
        )
    )
    catalogue_service = LearningWorldCatalogueService(
        repository=catalogue_repository,
        world_resolver=world_resolver,
    )

    # M2-C5 Study Materials.
    #
    # The repository owns PostgreSQL persistence while the
    # application service owns learner-facing catalogue access.
    #
    # Study Material World validation deliberately reuses the
    # authoritative LearningWorldCatalogueService above.
    study_material_repository = (
        PostgreSQLStudyMaterialRepository(
            session_factory,
        )
    )
    study_material_service = StudyMaterialService(
        repository=study_material_repository,
        world_catalogue_service=catalogue_service,
    )

    beta_telegram = create_beta_telegram_composition(
        user_repository=beta_user_repository,
        token_repository=beta_token_repository,
        binding_repository=telegram_binding_repository,
        conversation_event_repository=(
            conversation_event_repository
        ),
        privacy_identifier_service=(
            privacy_identifier_service
        ),
        conversation_event_validator=(
            conversation_event_validator
        ),
        expiration_minutes=expiration_minutes,
        conversation_runtime=conversation_runtime,
        conversation_runtime_factory=(
            conversation_runtime_factory
        ),
        provider_usage_repository=(
            provider_usage_repository
        ),
        functionality_control_service=(
            functionality_control_service
        ),
        world_resolver=world_resolver,
        language_response_policy=LanguageResponsePolicy(),
        language_detector=language_detector,
    )

    logger.info(
        "Production Beta Telegram composition created "
        "with durable usage ledger, functionality control, "
        "and Study Material service."
    )

    return BetaTelegramProductionComposition(
        beta_telegram=beta_telegram,
        database_engine=database_engine,
        session_factory=session_factory,
        catalogue_service=catalogue_service,
        study_material_service=study_material_service,
    )

def _env_positive_int(name: str, default: int) -> int:
    raw = os.getenv(name)

    if raw is None:
        return default

    try:
        value = int(raw)
    except ValueError as exc:
        raise ValueError(
            f"{name} must be a positive integer."
        ) from exc

    if value <= 0:
        raise ValueError(
            f"{name} must be a positive integer."
        )

    return value


def _env_positive_float(name: str, default: float) -> float:
    raw = os.getenv(name)

    if raw is None:
        return default

    try:
        value = float(raw)
    except ValueError as exc:
        raise ValueError(
            f"{name} must be a positive number."
        ) from exc

    if value <= 0:
        raise ValueError(
            f"{name} must be a positive number."
        )

    return value


def create_beta_telegram_composition(
    *,
    user_repository: Any,
    token_repository: Any,
    binding_repository: Any,
    conversation_event_repository: Any,
    privacy_identifier_service: Any,
    conversation_event_validator: Any,
    expiration_minutes: int = 1440,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: Callable[[], ConversationRuntime] | None = None,
    rate_limiter: Any | None = None,
    abuse_protector: Any | None = None,
    observability: Any | None = None,
    provider_usage_capture: ProviderUsageCapture | None = None,
    provider_usage_repository: ProviderUsageRepository | None = None,
    functionality_control_service: (
        FunctionalityOperationalControlService | None
    ) = None,
    world_resolver: WorldResolver | None = None,
    language_response_policy: LanguageResponsePolicy | None = None,
    language_detector: LanguageDetector | None = None,
) -> BetaTelegramComposition:
    """Create the Beta Telegram application composition.

    ``functionality_control_service`` is optional at this lower-level
    composition boundary so isolated application/unit compositions can
    continue to construct Telegram without requiring infrastructure.

    Production composition is responsible for requiring and supplying
    the authoritative functionality-control service.
    """

    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    if (
        conversation_runtime is not None
        and conversation_runtime_factory is not None
    ):
        raise ValueError(
            "Provide either conversation_runtime or "
            "conversation_runtime_factory, not both."
        )

    # Application configuration root:
    #
    # backend/app
    #
    app_root = (
        Path(__file__)
        .resolve()
        .parents[2]
    )

    world_repository = YamlWorldRepository(
        worlds_path=(
            app_root
            / "configuration"
            / "worlds"
        )
    )

    if world_resolver is None:
        world_resolver = WorldResolver(
            world_repository=world_repository,
        )

    beta_access_service = BetaAccessService(
        user_repository=user_repository,
        token_repository=token_repository,
        binding_repository=binding_repository,
        world_resolver=world_resolver,
        expiration_minutes=expiration_minutes,
    )

    resolved_usage_repository = provider_usage_repository

    if conversation_runtime is not None:
        resolved_conversation_runtime = conversation_runtime

    elif conversation_runtime_factory is not None:
        resolved_conversation_runtime = conversation_runtime_factory()

        # P1.6.1: when production supplies a durable usage repository,
        # attach usage capture to the factory-created ConversationRuntime.
        # Keep the existing factory API and explicit runtime injection seam.
        if (
            provider_usage_repository is not None
            and provider_usage_capture is None
            and isinstance(
                resolved_conversation_runtime,
                ConversationRuntime,
            )
            and resolved_conversation_runtime.usage_capture is None
        ):
            provider_usage_service = ProviderUsageService(
                provider_usage_repository
            )

            provider_usage_capture = ProviderUsageCapture(
                provider_usage_service
            )

            resolved_conversation_runtime = replace(
                resolved_conversation_runtime,
                usage_capture=provider_usage_capture,
                usage_context=ProviderUsageContext(
                    capability="conversation",
                    functionality="chat",
                    agent="conversation_agent",
                ),
            )

    else:
        if provider_usage_capture is None:
            if resolved_usage_repository is None:
                resolved_usage_repository = (
                    InMemoryProviderUsageRepository()
                )

            provider_usage_service = ProviderUsageService(
                resolved_usage_repository
            )

            provider_usage_capture = ProviderUsageCapture(
                provider_usage_service
            )

        resolved_conversation_runtime = ConversationRuntime(
            llm_provider=LLMProviderFactory.create(),
            usage_capture=provider_usage_capture,
            usage_context=ProviderUsageContext(
                capability="conversation",
                functionality="chat",
                agent="conversation_agent",
            ),
        )

    resolved_language_response_policy = (
        language_response_policy or LanguageResponsePolicy()
    )

    if language_detector is None:
        raise ValueError(
            "language_detector is required for Beta Telegram composition."
        )

    conversation_service = TelegramConversationService(
        conversation_runtime=resolved_conversation_runtime,
        conversation_event_repository=conversation_event_repository,
        privacy_identifier_service=privacy_identifier_service,
        conversation_event_validator=conversation_event_validator,
        language_response_policy=resolved_language_response_policy,
        language_detector=language_detector,
        world_resolver=world_resolver,
    )

    resolved_rate_limiter = (
        rate_limiter
        or InMemoryTelegramRateLimiter(
            max_messages=_env_positive_int(
                "TELEGRAM_RATE_LIMIT_MAX_MESSAGES",
                100,
            ),
            window_seconds=_env_positive_float(
                "TELEGRAM_RATE_LIMIT_WINDOW_SECONDS",
                86400.0,
            ),
        )
    )

    resolved_abuse_protector = (
        abuse_protector
        or InMemoryTelegramAbuseProtector(
            max_messages=_env_positive_int(
                "TELEGRAM_ABUSE_MAX_MESSAGES",
                10,
            ),
            window_seconds=_env_positive_float(
                "TELEGRAM_ABUSE_WINDOW_SECONDS",
                60.0,
            ),
        )
    )

    resolved_observability = (
        observability
        or LoggingTelegramObservability(
            privacy_identifier_service=privacy_identifier_service,
        )
    )

    conversation_boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation_service,
        beta_access_service=beta_access_service,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,
        observability=resolved_observability,
        functionality_control_service=functionality_control_service,
    )

    private_chat_gate = BetaPrivateChatGate(
        beta_access_service=beta_access_service
    )

    start_handler = BetaTelegramStartHandler(
        beta_access_service=beta_access_service
    )

    adapter = BetaTelegramAdapter(
        start_handler=start_handler,
        private_chat_gate=private_chat_gate,
        conversation_boundary=conversation_boundary,
    )

    runtime = TelegramBotRuntime(
        adapter=adapter,
        conversation_boundary=conversation_boundary,
    )

    return BetaTelegramComposition(
        beta_access_service=beta_access_service,
        conversation_runtime=resolved_conversation_runtime,
        conversation_service=conversation_service,
        conversation_boundary=conversation_boundary,
        private_chat_gate=private_chat_gate,
        start_handler=start_handler,
        adapter=adapter,
        runtime=runtime,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,
        observability=resolved_observability,
        provider_usage_repository=resolved_usage_repository,
    )

from __future__ import annotations

from uuid import UUID

from app.application.world_catalogue.service import (
    LearningWorldCatalogueService,
)
from app.domain.study_material.study_material import StudyMaterial

from .repositories import StudyMaterialRepository


class StudyMaterialService:
    """
    Application service for learner Study Material catalogue access.

    Responsibilities:
    - validate the requested Learning World identifier;
    - validate that the Learning World is available through the
      authoritative Learning World catalogue;
    - retrieve published Study Materials for that World;
    - return canonical immutable domain models.

    This service does not:
    - accept or resolve Telegram identity;
    - manage learner progress or mastery;
    - create conversations;
    - call an LLM;
    - perform RAG;
    - process uploads;
    - perform content authoring;
    - access PostgreSQL directly.
    """

    def __init__(
        self,
        *,
        repository: StudyMaterialRepository,
        world_catalogue_service: LearningWorldCatalogueService,
    ) -> None:
        if repository is None:
            raise ValueError("repository is required.")

        if world_catalogue_service is None:
            raise ValueError(
                "world_catalogue_service is required."
            )

        self._repository = repository
        self._world_catalogue_service = world_catalogue_service

    def list_published(
        self,
        *,
        world_id: str,
    ) -> list[StudyMaterial]:
        """
        Return published Study Materials for one Learning World.

        The supplied World identifier is the business-facing
        Learning World catalogue identifier.

        The authoritative catalogue service validates that the
        requested World exists, is enabled and has runtime
        configuration before materials are retrieved.
        """

        world_id = self._validate_world_id(world_id)

        self._world_catalogue_service.get_by_identifier(
            world_id,
        )

        materials = self._repository.list_published_by_world(
            world_id,
        )

        for material in materials:
            if material.world_id != world_id:
                raise ValueError(
                    "Study Material World mismatch."
                )

            if material.status.value != "PUBLISHED":
                raise ValueError(
                    "Study Material visibility violation."
                )

        return materials

    def get_published(
        self,
        *,
        material_id: UUID,
        world_id: str,
    ) -> StudyMaterial | None:
        """
        Return one published Study Material belonging to the
        supplied Learning World.

        The World is validated through the authoritative Learning
        World catalogue before the material repository is queried.
        """

        if not isinstance(material_id, UUID):
            raise ValueError(
                "material_id must be a UUID."
            )

        world_id = self._validate_world_id(world_id)

        self._world_catalogue_service.get_by_identifier(
            world_id,
        )

        material = self._repository.get_published(
            material_id,
            world_id,
        )

        if material is None:
            return None

        if material.world_id != world_id:
            raise ValueError(
                "Study Material World mismatch."
            )

        if material.status.value != "PUBLISHED":
            raise ValueError(
                "Study Material visibility violation."
            )

        return material

    @staticmethod
    def _validate_world_id(
        world_id: str,
    ) -> str:
        if not isinstance(world_id, str):
            raise TypeError(
                "world_id must be a string."
            )

        normalized = world_id.strip()

        if not normalized:
            raise ValueError(
                "world_id cannot be empty."
            )

        if len(normalized) > 128:
            raise ValueError(
                "world_id is too long. "
                "Maximum length is 128 characters."
            )

        return normalized

from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException, Request, status

router = APIRouter(

def _get_study_material_service(
    request: Request,
) -> StudyMaterialService:
    service = getattr(
        request.app.state,
        "study_material_service",
        None,
    )

    if service is None:
        raise RuntimeError(
            "Study Material service is not configured."
        )

    return service


def _to_response(
    material: StudyMaterial,
) -> StudyMaterialResponse:
    return StudyMaterialResponse(
        material_id=material.material_id,
        world_id=material.world_id,
        title=material.title,
        description=material.description,
        material_type=material.material_type,
        status=material.status,
        display_order=material.display_order,
        created_at=material.created_at,
        updated_at=material.updated_at,
    )


@router.get(
    "/worlds/{world_id}/materials",
    response_model=StudyMaterialListResponse,
)
def list_study_materials(
    world_id: str,
    _principal: Principal = Depends(get_current_principal),
    service: StudyMaterialService = Depends(
        _get_study_material_service,
    ),
) -> StudyMaterialListResponse:
    """
    Return published Study Materials for a Learning World.

    Learner identity is obtained from the authenticated principal.
    The client supplies only the business-facing Learning World
    identifier.
    """

    try:
        materials = service.list_published(
            world_id=world_id,
        )
    except (TypeError, ValueError) as exc:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=str(exc),
        ) from exc

    return StudyMaterialListResponse(
        world_id=world_id.strip(),
        materials=[
            _to_response(material)
            for material in materials
        ],
    )


@router.get(
    "/worlds/{world_id}/materials/{material_id}",
    response_model=StudyMaterialResponse,
)
def get_study_material(
    world_id: str,
    material_id: UUID,
    _principal: Principal = Depends(get_current_principal),
    service: StudyMaterialService = Depends(
        _get_study_material_service,
    ),
) -> StudyMaterialResponse:
    """
    Return one published Study Material within a Learning World.

    Learner identity is obtained from the authenticated principal.
    The material must belong to the requested Learning World.
    """

    try:
        material = service.get_published(
            material_id=material_id,
            world_id=world_id,
        )
    except (TypeError, ValueError) as exc:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=str(exc),
        ) from exc

    if material is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Study Material not found.",
        )

    return _to_response(material)

async def get_current_principal(request: Request) -> Principal:
    """Resolve identity from the HttpOnly application session cookie."""

    session_id = request.cookies.get("easychatbot_session")

    repository = request.app.state.auth_repository

    provider = SessionIdentityProvider(
        session_id=session_id,
        repository=repository,
    )

    principal = await provider.get_current_principal()

    if principal is None or not principal.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authentication required.",
        )

    return principal

    Learner-facing application services must expose only

    PUBLISHED = "PUBLISHED"
    ARCHIVED = "ARCHIVED"


@dataclass(frozen=True, slots=True)
class StudyMaterial:
    """
    Immutable learner-facing Study Material catalogue entity.

    StudyMaterial represents the metadata and catalogue identity
    of material belonging to a Learning World.

    It deliberately does NOT contain:

    - learner identity
    - Telegram identity
    - conversation identity
    - AI-generated content
    - RAG state
    - file-storage state
    - learner progress
    - mastery state

    Content/versioning will be introduced as a separate model when
    the content lifecycle is implemented.
    """

    material_id: UUID
    world_id: str
    title: str
    description: str | None
    material_type: StudyMaterialType
    status: StudyMaterialStatus
    display_order: int
    created_at: datetime
    updated_at: datetime

    def __post_init__(self) -> None:
        """
        Validate the Study Material domain invariants.
        """

        if not isinstance(self.material_id, UUID):
            raise ValueError(
                "material_id must be a UUID."
            )

        if not isinstance(self.world_id, str):
            raise ValueError(
                "world_id must be a string."
            )

        if not self.world_id.strip():
            raise ValueError(
                "world_id cannot be empty."
            )

        if not isinstance(self.title, str):
            raise ValueError(
                "title must be a string."
            )

        if not self.title.strip():
            raise ValueError(
                "title cannot be empty."
            )

        if self.description is not None:
            if not isinstance(self.description, str):
                raise ValueError(
                    "description must be a string or None."
                )

        if not isinstance(
            self.material_type,
            StudyMaterialType,
        ):
            raise ValueError(
                "material_type must be a StudyMaterialType."
            )

        if not isinstance(
            self.status,
            StudyMaterialStatus,
        ):
            raise ValueError(
                "status must be a StudyMaterialStatus."
            )

        if not isinstance(self.display_order, int):
            raise ValueError(
                "display_order must be an integer."
            )

        if isinstance(self.display_order, bool):
            raise ValueError(
                "display_order must be an integer."
            )

        if self.display_order < 0:
            raise ValueError(
                "display_order cannot be negative."
            )

        if not isinstance(self.created_at, datetime):
            raise ValueError(
                "created_at must be a datetime."
            )

        if not isinstance(self.updated_at, datetime):
            raise ValueError(
                "updated_at must be a datetime."
            )

        if self.updated_at < self.created_at:
            raise ValueError(
                "updated_at cannot be earlier than created_at."

class LearningWorldCatalogueService:
    """Apply catalogue policy and delegate runtime resolution to WorldResolver."""
    def __init__(self, *, repository: LearningWorldCatalogueRepository, world_resolver: WorldResolver) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")
        if world_resolver is None:
            raise ValueError("world_resolver cannot be None.")
        self._repository = repository
        self._world_resolver = world_resolver

    def get_by_identifier(self, identifier: str):
        if not identifier or not identifier.strip():
            raise ValueError("catalogue identifier is required.")
        return self._repository.get_by_identifier(identifier.strip())

    def resolve_runtime_world(self, catalogue_identifier: str):
        entry = self.get_by_identifier(catalogue_identifier)
        if entry is None:
            raise ValueError(f"Learning World not found: {catalogue_identifier.strip()}")
        if not entry.enabled:
            raise ValueError(f"Learning World is disabled: {entry.identifier}")
        if not entry.runtime_world_id:
            raise ValueError(f"Learning World has no runtime configuration: {entry.identifier}")
        return self._world_resolver.resolve(entry.runtime_world_id)

"""
World Resolver.

Application service responsible for resolving a learner's
active World into validated runtime context.

from __future__ import annotations

import logging
from typing import Protocol

from app.domain.uai.context.world_context import (
    WorldContext,
)


logger = logging.getLogger(__name__)


class WorldRepository(Protocol):
    """
    Repository contract for World retrieval.
    """

    def get_by_id(
        self,
        world_id: str,
    ):
        """
        Retrieve a World configuration.
        """
        ...


class WorldResolver:
    """
    Resolves a World configuration into runtime context.
    """

    def __init__(
        self,
        *,
        world_repository: WorldRepository,
    ) -> None:
        self._world_repository = world_repository

    def resolve(
        self,
        world_id: str,
    ) -> WorldContext:
        """
        Resolve World configuration.
        """

        logger.info(
            "Resolving world context",
            extra={
                "world_id": world_id,
            },
        )

        world = (
            self._world_repository
            .get_by_id(world_id)
        )

        if world is None:
            raise ValueError(
                f"World not found: {world_id}"
            )

        return WorldContext(
            world_id=world.id,
            name=world.name,
            language_code=world.language_code,
            world_type=world.world_type,
            learning_domain=world.learning_domain,
            language_policy=world.language_policy,
            available_characters=tuple(
                world.available_characters
            ),
        )

"""PostgreSQL adapter for the Study Material repository."""

from __future__ import annotations

from uuid import UUID

from sqlalchemy import select
from sqlalchemy.orm import Session, sessionmaker

from app.application.study_material.repositories import (
    StudyMaterialRepository,
)
from app.domain.study_material.study_material import (
    StudyMaterial,
    StudyMaterialStatus,
    StudyMaterialType,
)
from app.infrastructure.persistence.learning_world_models import (
    LearningWorldModel,
)
from app.infrastructure.persistence.study_material_models import (
    StudyMaterialRecord,
)


class PostgreSQLStudyMaterialRepository(
    StudyMaterialRepository,
):
    """
    PostgreSQL implementation of the Study Material repository.

    The application-facing World identifier is the authoritative
    Learning World `identifier` (for example, `french_world`).

    The Study Material persistence table stores the corresponding
    Learning World UUID in `study_materials.world_id`.

    This adapter therefore joins against the authoritative
    Learning World catalogue when reading materials.

    Learner identity, Telegram identifiers, conversation identifiers,
    progress, AI/RAG state and content ingestion are deliberately
    outside this repository.
    """

    def __init__(
        self,
        session_factory: sessionmaker[Session],
    ) -> None:
        if session_factory is None:
            raise ValueError(
                "session_factory cannot be None."
            )

        self._session_factory = session_factory

    def list_published_by_world(
        self,
        world_id: str,
    ) -> list[StudyMaterial]:
        """
        Return published Study Materials for one Learning World.

        Results are deterministically ordered by:

        1. display_order ascending
        2. material_id ascending
        """

        normalized_world_id = self._validate_world_id(
            world_id
        )

        with self._session_factory() as session:
            rows = session.scalars(
                select(StudyMaterialRecord)
                .join(
                    LearningWorldModel,
                    StudyMaterialRecord.world_id
                    == LearningWorldModel.id,
                )
                .where(
                    LearningWorldModel.identifier
                    == normalized_world_id,
                    StudyMaterialRecord.status
                    == StudyMaterialStatus.PUBLISHED.value,
                )
                .order_by(
                    StudyMaterialRecord.display_order.asc(),
                    StudyMaterialRecord.material_id.asc(),
                )
            ).all()

            return [
                self._to_domain(
                    row,
                    world_id=normalized_world_id,
                )
                for row in rows
            ]

    def get_published(
        self,
        material_id: UUID,
        world_id: str,
    ) -> StudyMaterial | None:
        """
        Return one published Study Material belonging to
        the supplied Learning World.

        The World identifier is part of the query boundary so a
        material belonging to another World cannot be returned
        merely because its UUID is known.
        """

        if not isinstance(material_id, UUID):
            raise ValueError(
                "material_id must be a UUID."
            )

        normalized_world_id = self._validate_world_id(
            world_id
        )

        with self._session_factory() as session:
            row = session.scalar(
                select(StudyMaterialRecord)
                .join(
                    LearningWorldModel,
                    StudyMaterialRecord.world_id
                    == LearningWorldModel.id,
                )
                .where(
                    StudyMaterialRecord.material_id
                    == material_id,
                    LearningWorldModel.identifier
                    == normalized_world_id,
                    StudyMaterialRecord.status
                    == StudyMaterialStatus.PUBLISHED.value,
                )
            )

            if row is None:
                return None

            return self._to_domain(
                row,
                world_id=normalized_world_id,
            )

    @staticmethod
    def _validate_world_id(
        world_id: str,
    ) -> str:
        if not isinstance(world_id, str):
            raise TypeError(
                "world_id must be a string."
            )

        normalized = world_id.strip()

        if not normalized:
            raise ValueError(
                "world_id cannot be empty."
            )

        return normalized

    @staticmethod
    def _to_domain(
        row: StudyMaterialRecord,
        *,
        world_id: str,
    ) -> StudyMaterial:
        try:
            material_type = StudyMaterialType(
                row.material_type
            )
        except ValueError as exc:
            raise ValueError(
                "Unsupported Study Material type: "
                f"{row.material_type!r}."
            ) from exc

        try:
            status = StudyMaterialStatus(
                row.status
            )
        except ValueError as exc:
            raise ValueError(
                "Unsupported Study Material status: "
                f"{row.status!r}."
            ) from exc

        return StudyMaterial(
            material_id=row.material_id,
            world_id=world_id,
            title=row.title,
            description=row.description,
            material_type=material_type,
            status=status,
            display_order=row.display_order,
            created_at=row.created_at,
            updated_at=row.updated_at,
        )


__all__ = [
    "PostgreSQLStudyMaterialRepository",
]

from __future__ import annotations

import os

from sqlalchemy import create_engine
from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker


class Base(DeclarativeBase):
    """Canonical SQLAlchemy metadata registry for the platform."""

    pass


def database_url() -> str:
    """Return the canonical PostgreSQL connection URL."""

    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for PostgreSQL persistence."
        )

    return value


def create_database_engine():
    """Create the existing synchronous PostgreSQL engine.

    This API is preserved for the existing Beta 1 persistence layer.
    """

    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(
    engine,
) -> sessionmaker[Session]:
    """Create the existing synchronous session factory.

    Preserved for existing Beta 1 repositories.
    """

    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )


def async_database_url() -> str:
    """Return DATABASE_URL adapted for SQLAlchemy async PostgreSQL.

    The environment remains the single source of truth. We only adapt the
    driver at the infrastructure boundary when the configured URL uses the
    standard PostgreSQL SQLAlchemy scheme.
    """

    value = database_url()

    if value.startswith("postgresql+asyncpg://"):
        return value

    if value.startswith("postgresql://"):
        return value.replace(
            "postgresql://",
            "postgresql+asyncpg://",
            1,
        )

    raise RuntimeError(
        "DATABASE_URL must use a PostgreSQL URL compatible with "
        "the synchronous or async PostgreSQL persistence boundary."
    )


def create_async_database_engine() -> AsyncEngine:
    """Create the canonical asynchronous PostgreSQL engine.

    This is used by durable WS-22 persistence while sharing the same
    DATABASE_URL and canonical Base as the existing persistence layer.
    """

    return create_async_engine(
        async_database_url(),
        pool_pre_ping=True,
    )


def create_async_session_factory(
    engine: AsyncEngine,
) -> async_sessionmaker[AsyncSession]:
    """Create the canonical asynchronous PostgreSQL session factory."""

    return async_sessionmaker(
        bind=engine,
        class_=AsyncSession,
        expire_on_commit=False,
    )

class FakeScalars:
    """Minimal SQLAlchemy scalars result double."""

    def __init__(self, records: list[object]) -> None:
        self._records = records

    def all(self) -> list[object]:
        return self._records


class FakeSession:
    """Minimal session double for repository unit tests."""

    def __init__(self) -> None:
        self.scalar_result: object | None = None
        self.scalars_result: list[object] = []
        self.statements: list[object] = []

    def __enter__(self) -> "FakeSession":
        return self

    def __exit__(
        self,
        exc_type: object,
        exc_value: object,
        traceback: object,
    ) -> None:
        return None

    def scalar(self, statement: object) -> object | None:
        self.statements.append(statement)
        return self.scalar_result

    def scalars(self, statement: object) -> FakeScalars:
        self.statements.append(statement)
        return FakeScalars(self.scalars_result)


class FakeSessionFactory:
    """Callable session factory matching the production boundary."""

    def __init__(self, session: FakeSession) -> None:
        self.session = session
        self.calls = 0

    def __call__(self) -> FakeSession:
        self.calls += 1
        return self.session
