"""
EasyChatBot AI Companion Demo.

Demonstrates:

User
 |
Character
 |
Conversation Agent
 |
Research Agent
 |
Parallel Search
"""


import asyncio


from backend.services.companion_service import (
    CompanionService,
)

from backend.characters.character_profile import (
    CharacterProfile,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


async def main():

    yeti = CharacterProfile(
        character_id="yeti",
        name="Yeti",
        description=(
            "Friendly AI learning companion"
        ),
        personality=(
            "Curious and encouraging"
        ),
        learning_style=(
            "Conversation based learning"
        ),
    )


    companion = CompanionService(
        character=yeti
    )


    context = ExecutionContext(
        user_id="demo-user",
        request=(
            "Tell me about today's "
            "artificial intelligence news"
        ),
    )


    result = await companion.interact(
        context,
        use_research=True,
    )


    print("=" * 60)
    print("EasyChatBot Companion Demo")
    print("=" * 60)

    print()

    print(
        "Character:",
        result["character"],
    )

    print()

    print(
        "Conversation:"
    )

    print(
        result["conversation"]
    )

    print()

    print(
        "Research:"
    )

    print(
        result["research"]
    )


if __name__ == "__main__":

    asyncio.run(main())

"""

class DemoSearchProvider:
    """
    Controlled search provider for the unified demo.

    The real Parallel provider is already validated separately
    by H1 integration tests.

    This keeps the unified demo deterministic and suitable for
    repeatable hackathon demonstrations.
    """

    async def search(
        self,
        query: str,
        max_results: int = 5,
    ) -> list[SearchResult]:

        return [
            SearchResult(
                title="European AI Act — Demo Source",
                url="https://example.com/ai-act",
                content=(
                    "The European Union AI Act establishes "
                    "a risk-based framework for artificial "
                    "intelligence."
                ),
                source="demo-provider",
            )
        ]


class DemoIntelligenceProcessor:
    """
    Controlled intelligence processor.

    The real provider-backed intelligence flow is validated
    separately in H1.
    """

    async def summarise(
        self,
        results: list[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, list[str]]:

        return (
            "The European Union AI Act establishes "
            "a risk-based framework for artificial "
            "intelligence.",
            [
                "AI governance",
                "risk management",
                "compliance",
            ],
        )


class DemoChallengeContentGenerator:

    async def generate_vocabulary(

                    "Ensemble des règles qui encadrent "

                ),

    async def generate_questions(
        self,
        summary: str,
        language: str,
        learner_level: str,

        return [
            DiscussionQuestion(
                question=(
                    "Pourquoi la réglementation de "

                difficulty=learner_level,
            ),
            DiscussionQuestion(
                question=(
                    "Quels sont les avantages d'une "
                    "réglementation européenne de l'IA ?"
                ),
                difficulty=learner_level,
            ),
        ]


def create_governance_policy() -> GovernancePolicy:
    """
    Governance policy for the live intelligence demo.

    Live research may access external information but
    must never create external side effects.
    """

    return GovernancePolicy(
        capability="live_intelligence_research",
        risk_level=RiskLevel.MEDIUM,
        requires_human_review=False,
        allows_external_calls=True,
        allows_side_effects=False,
    )


async def create_learning_challenge() -> LearningChallenge:
    """
    Execute the H1 learning challenge flow.
    """

    intelligence_agent = LiveIntelligenceAgent(
        search_provider=DemoSearchProvider(),
        processor=DemoIntelligenceProcessor(),
    )

    intelligence = await intelligence_agent.execute(
        LiveIntelligenceRequest(
            topic="AI Act",
            language="French",
            learner_level="B2",
        )
    )

    challenge_generator = ChallengeGenerator(
        content_generator=DemoChallengeContentGenerator()
    )

    return await challenge_generator.generate(
        intelligence=intelligence,
        language="French",
        learner_level="B2",
    )


async def run_demo() -> None:
    """
    Execute the complete hackathon demonstration.
    """

    print()
    print("=" * 72)
    print("EasyChatBot — HACKATHON DEMONSTRATION")
    print("=" * 72)
    print()

    # -----------------------------------------------------------------------
    # STEP 1 — Governance
    # -----------------------------------------------------------------------

    print("STEP 1 — GOVERNANCE")
    print("-" * 72)

    governance_engine = GovernancePolicyEngine()

    governance_result = governance_engine.evaluate(
        create_governance_policy()
    )

    print(
        f"Capability: {governance_result.capability}"
    )

    print(
        f"Risk level: "
        f"{governance_result.risk_level.value}"
    )

    print(
        f"Decision: "
        f"{governance_result.decision.value.upper()}"
    )

    if governance_result.decision != GovernanceDecision.ALLOW:
        raise RuntimeError(
            "Hackathon demo stopped because governance "
            "did not allow the capability."
        )

    print("Governance check: PASSED")
    print()

    # -----------------------------------------------------------------------
    # STEP 2 — Live Intelligence + Challenge
    # -----------------------------------------------------------------------

    print("STEP 2 — LIVE INTELLIGENCE")
    print("-" * 72)

    challenge = await create_learning_challenge()

    print(f"Topic: {challenge.topic}")
    print(f"Language: {challenge.language}")
    print(f"Learner level: {challenge.learner_level}")
    print()

    print("Summary:")
    print(challenge.summary)
    print()

    print("Vocabulary:")

    for item in challenge.vocabulary:
        print(f"  • {item.term}: {item.definition}")

    print()

    print("Discussion questions:")

    for index, question in enumerate(
        challenge.discussion_questions,
        start=1,
    ):
        print(
            f"  {index}. {question.question}"
        )

    print()

    # -----------------------------------------------------------------------
    # STEP 3 — Quality Evaluation
    # -----------------------------------------------------------------------

    print("STEP 3 — AI QUALITY EVALUATION")
    print("-" * 72)

    evaluator = DeterministicEvaluator()

    criterion_scores = evaluator.evaluate(
        challenge
    )

    evaluation_engine = EvaluationEngine(
        criteria=[
            COMPLETENESS,
            LANGUAGE_LEVEL,
            EDUCATIONAL_VALUE,
            SOURCE_COVERAGE,
        ],
        passing_score=0.70,
    )

    evaluation_result = evaluation_engine.evaluate(
        criterion_scores
    )

    report = EvaluationReport().generate(
        evaluation_result
    )

    print(report)

    # -----------------------------------------------------------------------
    # STEP 4 — PERSONALISED FLASH DEAL
    # -----------------------------------------------------------------------

    print("STEP 4 — PERSONALISED DISCOVERY")
    print("-" * 72)

    # Build the same domain inputs that a real discovery
    # application flow would receive.
    demo_deal = build_demo_deal()
    demo_memory = build_demo_memory()

    discovery_request = DiscoveryRequest(
        user_id=demo_memory.user_id,
        deal=demo_deal,
        character=CharacterContext(
            character_id="yeti",
            character_name="Yeti",
            message=(
                "Look what I found! "
                "This could be perfect for you."
            ),
        ),
        product_image_url=(
            "https://example.com/demo/iphone17-pro.jpg"
        ),
        tagline="Pro. Beautiful. Powerful.",
        highlights=(
            "Pro camera system",
            "High-performance chip",
            "All-day battery life",
            "Premium display",
        ),
    )

    flash_deal_card = DiscoveryService().create_flash_deal(
        discovery_request,
        memories=[demo_memory],
    )

    print(
        render_demo_flash_deal_card(
            flash_deal_card
        )
    )

    assert flash_deal_card.character_id == "yeti"
    assert flash_deal_card.deal_id == demo_deal.deal_id
    assert flash_deal_card.product_name
    assert demo_memory.user_id == "demo-user"
    assert flash_deal_card.relevance_reason
    assert flash_deal_card.get_deal_url
    assert flash_deal_card.more_information_url

    # -----------------------------------------------------------------------
    # STEP 5 — COMMERCE ACTION
    # -----------------------------------------------------------------------

    print("STEP 5 — COMMERCE ACTION")
    print("-" * 72)

    commerce_action = FlashDealCommerce().create_action(
        user_id="demo-user",
        card=flash_deal_card,
    )

    print(
        f"Product: {flash_deal_card.product_name}"
    )
    print(
        f"Vendor: {flash_deal_card.vendor_name}"
    )
    print(
        f"Commerce action: "
        f"{commerce_action.action_type.value.upper()}"
    )

    commerce_result = CommerceService().initiate(
        commerce_action
    )

    print(
        f"Status: "
        f"{commerce_result.status.value.upper()}"
    )

    print()
    print("GET DEAL")
    print("  -> Approved commerce boundary")
    print("  -> Vendor handoff")
    print()
    print(
        "Payment is not executed in the hackathon demo."
    )

    assert (
        commerce_result.status
        == CommerceActionStatus.REDIRECT
    )
    assert (
        commerce_result.deal_id
        == flash_deal_card.deal_id
    )
    assert (
        commerce_result.next_url
        == flash_deal_card.get_deal_url
    )

    # -----------------------------------------------------------------------
    # STEP 6 — Final result
    # -----------------------------------------------------------------------

    print()
    print("=" * 72)

    if evaluation_result.passed:
        print("HACKATHON DEMO: SUCCESS")
        print()
        # ASCII-safe output keeps the executable reliable when
        # stdout is captured by Windows pytest/subprocess.
        # Preserve the stable hackathon readiness marker.
        print(
            "Governance -> Intelligence -> Learning -> Evaluation"
        )
        print(
            "Governance -> Intelligence -> Learning -> "
            "Evaluation -> Personalised Discovery"
        )
        print(
            "All demonstrated capabilities completed successfully."
        )
    else:
        print("HACKATHON DEMO: QUALITY CHECK FAILED")

    print("=" * 72)
    print()

    assert governance_result.decision == GovernanceDecision.ALLOW
    assert challenge.topic == "AI Act"
    assert evaluation_result.passed is True


def main() -> None:
    """
    Application entry point.
    """

    asyncio.run(run_demo())


if __name__ == "__main__":
    main()

            )

        raw_results = provider_output.get(

            {},
        )

        if isinstance(

            )

class ConversationRequest:
    """
    Immutable provider-independent conversation request.

    Attributes
    ----------
    system_instruction:
        Stable instruction defining how the character
        should behave.

    user_prompt:
        User-facing request to be answered.

        Validate request invariants.
        """

            self.system_instruction,
            str,
        ):
            raise ValueError(
                "system_instruction must be a string."
            )

        if not isinstance(
            self.user_prompt,
            str,
        ):
            raise ValueError(
                "user_prompt must be a string."
            )

        system_instruction = (
            self.system_instruction.strip()
        )

        user_prompt = (
            self.user_prompt.strip()
        )

        if not system_instruction:
            raise ValueError(
                "system_instruction cannot be empty."
            )

        if not user_prompt:
            raise ValueError(
                "user_prompt cannot be empty."
            )

        object.__setattr__(
            self,
            "system_instruction",
            system_instruction,
        )

        object.__setattr__(
            self,
            "user_prompt",
            user_prompt,
        )

        if self.conversation_context is not None:
            context = (
                self.conversation_context.strip()
            )

            if not context:
                raise ValueError(
                    "conversation_context cannot be blank."
                )

            object.__setattr__(
                self,
                "conversation_context",
                context,
            )

        if not isinstance(
            self.character_id,
            str,
        ):
            raise ValueError(
                "character_id must be a string."
            )

        if not isinstance(
            self.character_name,
            str,
        ):
            raise ValueError(
                "character_name must be a string."
            )

        object.__setattr__(
            self,
            "character_id",
            self.character_id.strip(),
        )

        object.__setattr__(
            self,
            "character_name",
            self.character_name.strip(),
        )

    @property
    def has_conversation_context(self) -> bool:
        """
        Return whether contextual information exists.
        """

        return bool(

        )

    @property
    def context_length(self) -> int:

        """

class ParallelClientWrapper:
    """
    Wrapper around the official Parallel Search SDK.

    Responsibilities:
    - authenticate with Parallel
    - execute searches
    - normalize SDK responses

    Does not:
    - create learning content
    - manage agents
    - decide workflows
    """

    def __init__(
        self,
        api_key: [REDACTED]"""
        Initialise the Parallel SDK client.
        """

        self.api_key = [REDACTED]"PARALLEL_API_KEY")
        )

        if not self.api_key:
            [REDACTED]"PARALLEL_API_KEY environment variable is required"
            )

        self._client = self._create_client()


    def _create_client(self):
        """
        Create the official Parallel SDK client.

        SDK:
            parallel-web 1.1.0
        """

        from parallel import Parallel

        return Parallel(
            api_key=[REDACTED]"""
        Convert the Parallel SDK response into
        application-safe data.

        Pydantic SDK models are converted to dictionaries.
        Other JSON-compatible values are returned unchanged.
        """

        if hasattr(response, "model_dump"):
            return response.model_dump()

        if hasattr(response, "dict"):
            return response.dict()

        return response


    async def search(
        self,
        query: str,
    ) -> dict[str, Any]:
        """
        Execute a Parallel web search.

        Args:
            query:
                Natural-language research objective.

        Returns:
            Normalized application-level search result.
        """

        search_query = self._build_search_query(
            query
        )

        response = await asyncio.to_thread(
            self._client.search,
            search_queries=[
                search_query
            ],
            mode="advanced",
            objective=query,
        )

        return {
            "query": query,
            "search_queries": [
                search_query
            ],
            "results": self._normalize_response(
                response
            ),
        }


    @staticmethod
    def _build_search_query(
        query: str,
    ) -> str:
        """
        Build a concise search query from the
        user's research request.

        The full request remains the Parallel
        objective while the search query is kept
        concise.
        """

        words = query.strip().split()

        if len(words) <= 6:
            return query.strip()

        return " ".join(words[:6])

"""
Parallel Search Provider.

Provides live world information
through the Parallel Search API.
"""

from backend.providers.base_provider import (
    BaseProvider,
)

from backend.providers.provider_result import (
    ProviderResult,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class ParallelSearchProvider(BaseProvider):
    """
    Provider implementation using Parallel Search.

    The SDK client is created only when required.
    This allows tests to run without API credentials.
    """

    def __init__(
        self,
        client=None,
    ):
        super().__init__(
            provider_id="parallel_search"
        )

        self.client = client


    def _get_client(self):
        """
        Lazy-load the real Parallel client.

        Avoids requiring API keys during unit tests.
        """

        if self.client is None:

            from backend.providers.parallel.parallel_client import (
                ParallelClientWrapper,
            )

            self.client = ParallelClientWrapper()

        return self.client


    async def execute(
        self,
        context: ExecutionContext,
    ) -> ProviderResult:
        """
        Execute a live Parallel search.
        """

        try:

            client = self._get_client()

            result = await client.search(
                query=context.request
            )

            return ProviderResult(
                success=True,
                provider_id=self.provider_id,
                output=result,
            )

        except Exception as exc:

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error=str(exc),
            )

Live Intelligence Agent Handler.

Responsible for executing the Live Intelligence Agent workflow.

Architecture:

AI Runtime
    |
    v
Live Intelligence Agent Handler
    |
    +--> Search Provider Interface
    |
    +--> Knowledge Processing
    |
    v
Structured Intelligence Output

Responsibilities:
- Validate agent input
- Call approved provider interfaces
- Normalise research output
- Return explicit structured results

Non-responsibilities:
- Provider SDK management
- Direct external API handling
- Autonomous actions
- Configuration changes
"""

from dataclasses import dataclass
from typing import List, Protocol

from src.providers.search_provider.interface import (
    SearchProvider,
    SearchResult,
)


@dataclass(frozen=True)
class LiveIntelligenceRequest:
    """
    Input contract for Live Intelligence Agent.
    """

    topic: str
    language: str = "French"
    learner_level: str = "B2"


@dataclass(frozen=True)
class LiveIntelligenceResponse:
    """
    Output contract from Live Intelligence Agent.
    """

    topic: str
    summary: str
    sources: List[SearchResult]
    key_information: List[str]


class IntelligenceProcessor(Protocol):
    """
    Contract for future LLM-based processing.

    Keeps the agent independent from
    any specific model/provider.
    """

    async def summarise(
        self,
        results: List[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, List[str]]:
        ...


class LiveIntelligenceAgent:
    """
    Live Intelligence Agent implementation.
    """

    def __init__(
        self,
        search_provider: SearchProvider,
        processor: IntelligenceProcessor,
    ) -> None:
        self._search_provider = search_provider
        self._processor = processor

    async def execute(
        self,
        request: LiveIntelligenceRequest,
    ) -> LiveIntelligenceResponse:
        """
        Execute research workflow.

        Flow:

        Request
          |
          v
        Search Provider
          |
          v
        Intelligence Processor
          |
          v
        Structured Response
        """

        self._validate_request(request)

        search_results = await self._search_provider.search(
            query=request.topic,
            max_results=5,
        )

        summary, key_information = await self._processor.summarise(
            results=search_results,
            language=request.language,
            learner_level=request.learner_level,
        )

        return LiveIntelligenceResponse(
            topic=request.topic,
            summary=summary,
            sources=search_results,
            key_information=key_information,
        )

    @staticmethod
    def _validate_request(
        request: LiveIntelligenceRequest,
    ) -> None:
        """
        Validate required input.
        """

        if not request.topic.strip():
            raise ValueError(
                "Topic cannot be empty"
            )

"""
AI Runtime executor for Live Intelligence.

This adapter bridges the provider-independent AI Runtime
capability contract with the existing LiveIntelligenceAgent.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
LiveIntelligenceExecutor
    |
    v
LiveIntelligenceAgent
    |
    v
LiveIntelligenceResponse

The adapter:

- validates the Runtime capability
- translates Runtime payloads into LiveIntelligenceRequest
- invokes the existing Live Intelligence Agent
- translates the agent response into CapabilityResult

It does not:

- implement intelligence logic
- call Parallel directly
- modify the LiveIntelligenceAgent
- perform governance decisions
"""

from __future__ import annotations

from src.ai_agents.live_intelligence_agent.handler import (
    LiveIntelligenceAgent,
    LiveIntelligenceRequest,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)


class LiveIntelligenceExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for the existing LiveIntelligenceAgent.
    """

    def __init__(
        self,
        agent: LiveIntelligenceAgent,
    ) -> None:
        self._agent = agent

    @property
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """

        return CapabilityType.LIVE_INTELLIGENCE

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Translate and execute a Live Intelligence request.
        """

        if (
            request.capability
            != CapabilityType.LIVE_INTELLIGENCE
        ):
            raise ValueError(
                "LiveIntelligenceExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        agent_request = (
            self._build_agent_request(request)
        )

        response = await self._agent.execute(
            agent_request
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=response,
            metadata={
                "executor": (
                    "live_intelligence_executor"
                ),
            },
        )

    @staticmethod
    def _build_agent_request(
        request: CapabilityRequest,
    ) -> LiveIntelligenceRequest:
        """
        Translate Runtime payload into the existing
        LiveIntelligenceRequest contract.
        """

        payload = request.payload

        topic = payload.get("topic")

        if not isinstance(topic, str):
            raise ValueError(
                "Live Intelligence capability requires "
                "a string 'topic'."
            )

        language = payload.get(
            "language",
            "French",
        )

        learner_level = payload.get(
            "learner_level",
            "B2",
        )

        if not isinstance(language, str):
            raise ValueError(
                "Live Intelligence 'language' "
                "must be a string."
            )

        if not isinstance(
            learner_level,
            str,
        ):
            raise ValueError(
                "Live Intelligence 'learner_level' "
                "must be a string."
            )

        return LiveIntelligenceRequest(
            topic=topic,
            language=language,
            learner_level=learner_level,
        )

"""
Hackathon Flash Deal Card presentation contract.

This module defines the user-facing data contract for the
EasyChatBot Flash Deal Card shown by a character.

Architecture:

Parallel / Discovery
        |
        v
     DealCard
        |
        +---- Memory Context
        |
        +---- Character
        |
        v
  FlashDealCard
        |
        +---- GET DEAL
        |
        +---- MORE INFORMATION
        |
        v
 External Vendor / Future Commerce Flow

The model is presentation-oriented.

It does not:
- call providers
- access SQLite
- perform payments
- execute agents
- scrape vendor pages
- invent product information
"""

from __future__ import annotations

from dataclasses import dataclass, field

from src.ai_runtime.discovery.models import DealCard


@dataclass(frozen=True)
class FlashDealCard:
    """
    User-facing Flash Deal Card.

    This contract maps directly to the hackathon UX:

    - character discovery
    - product information
    - pricing
    - discount
    - personal relevance
    - vendor
    - GET DEAL action
    - MORE INFORMATION action

    The card deliberately contains no payment implementation.
    """

    deal_id: str
    character_id: str
    character_name: str
    character_message: str

    product_name: str
    product_image_url: str
    tagline: str
    highlights: tuple[str, ...]

    current_price: float
    currency: str
    previous_price: float | None
    discount_percentage: float | None
    savings: float | None

    relevance_reason: str

    vendor_name: str
    more_information_url: str
    get_deal_url: str

    badge: str = "NEW FIND!"
    card_id: str | None = None
    metadata: dict[str, str] = field(
        default_factory=dict
    )

    def __post_init__(self) -> None:
        """
        Validate the presentation contract.
        """

        required_text_fields = {
            "deal_id": self.deal_id,
            "character_id": self.character_id,
            "character_name": self.character_name,
            "character_message": self.character_message,
            "product_name": self.product_name,
            "product_image_url": self.product_image_url,
            "tagline": self.tagline,
            "currency": self.currency,
            "relevance_reason": self.relevance_reason,
            "vendor_name": self.vendor_name,
            "more_information_url": self.more_information_url,
            "get_deal_url": self.get_deal_url,
        }

        for field_name, value in required_text_fields.items():
            if not value.strip():
                raise ValueError(
                    f"FlashDealCard {field_name} cannot be empty."
                )

        if not self.highlights:
            raise ValueError(
                "FlashDealCard must contain at least one highlight."
            )

        if any(
            not highlight.strip()
            for highlight in self.highlights
        ):
            raise ValueError(
                "FlashDealCard highlights cannot contain empty values."
            )

        if self.current_price < 0:
            raise ValueError(
                "FlashDealCard current_price cannot be negative."
            )

        if (
            self.previous_price is not None
            and self.previous_price < 0
        ):
            raise ValueError(
                "FlashDealCard previous_price cannot be negative."
            )

        if (
            self.discount_percentage is not None
            and not 0 <= self.discount_percentage <= 100
        ):
            raise ValueError(
                "FlashDealCard discount_percentage must be "
                "between 0 and 100."
            )

        if self.savings is not None and self.savings < 0:
            raise ValueError(
                "FlashDealCard savings cannot be negative."
            )

        self._validate_url(
            self.product_image_url,
            "product_image_url",
        )
        self._validate_url(
            self.more_information_url,
            "more_information_url",
        )
        self._validate_url(
            self.get_deal_url,
            "get_deal_url",
        )

    @staticmethod
    def _validate_url(
        url: str,
        field_name: str,
    ) -> None:
        """
        Validate a user-facing URL.
        """

        if not (
            url.startswith("https://")
            or url.startswith("http://")
        ):
            raise ValueError(
                f"FlashDealCard {field_name} must use "
                "HTTP or HTTPS."
            )

    @classmethod
    def from_deal(
        cls,
        *,
        deal: DealCard,
        character_id: str,
        character_name: str,
        character_message: str,
        relevance_reason: str,
        product_image_url: str,
        tagline: str,
        highlights: list[str] | tuple[str, ...],
        badge: str = "NEW FIND!",
        metadata: dict[str, str] | None = None,
    ) -> "FlashDealCard":
        """
        Build a presentation card from a domain DealCard.

        Product, pricing and vendor information comes from the
        DealCard. Character and personalisation information is
        supplied explicitly by the application layer.

        No product or pricing information is invented here.
        """

        savings: float | None = None

        if (
            deal.previous_price is not None
            and deal.previous_price > deal.price
        ):
            savings = round(
                deal.previous_price - deal.price,
                2,
            )

        return cls(
            deal_id=deal.deal_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            product_name=deal.product_name,
            product_image_url=product_image_url,
            tagline=tagline,
            highlights=tuple(highlights),
            current_price=deal.price,
            currency=deal.currency,
            previous_price=deal.previous_price,
            discount_percentage=deal.discount_percentage,
            savings=savings,
            relevance_reason=relevance_reason,
            vendor_name=deal.vendor,
            more_information_url=(
                deal.commerce_link.destination_url
            ),
            get_deal_url=(
                deal.commerce_link.click_url
            ),
            badge=badge,
            metadata=(
                dict(metadata)
                if metadata is not None
                else {}
            ),
        )

    @property
    def has_discount(self) -> bool:
        """Return whether a verified discount exists."""

        return (
            self.discount_percentage is not None
            and self.discount_percentage > 0
        )

    @property
    def has_previous_price(self) -> bool:
        """Return whether a previous price is available."""

        return self.previous_price is not None

    @property
    def display_discount(self) -> str | None:
        """Return a UI-friendly discount label."""

        if self.discount_percentage is None:
            return None

        return f"{self.discount_percentage:.0f}% OFF"

    @property
    def display_savings(self) -> str | None:
        """Return a UI-friendly savings label."""

        if self.savings is None:
            return None

        return f"Save {self.currency} {self.savings:.2f}"

"""
EasyChatBot Hackathon — Apple Podcasts catalogue provider.

Uses Apple's iTunes Search API for read-only podcast catalogue discovery.
"""

from __future__ import annotations

import json
from urllib.parse import urlencode
from urllib.request import Request, urlopen

from src.ai_runtime.podcast.models import (
    PodcastProviderName,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)


class ApplePodcastProvider:
    """Read-only Apple podcast catalogue provider."""

    BASE_URL = "https://itunes.apple.com/search"

    def __init__(self, *, timeout_seconds: float = 10.0) -> None:
        self._timeout_seconds = timeout_seconds

    def search(
        self,
        intent: PodcastSearchIntent,
    ) -> list[PodcastSearchResult]:
        query_terms = list(intent.topics) + list(intent.keywords)
        params = {
            "term": " ".join(query_terms[:6]),
            "country": intent.learning_context.country_code.upper(),
            "media": "podcast",
            "entity": "podcast",
            "limit": str(intent.max_results),
            "explicit": "No",
        }

        request = Request(
            f"{self.BASE_URL}?{urlencode(params)}",
            headers={
                "Accept": "application/json",
                "User-Agent": "EasyChatBot-Hackathon/1.0",
            },
            method="GET",
        )

        with urlopen(
            request,
            timeout=self._timeout_seconds,
        ) as response:
            payload = json.loads(
                response.read().decode("utf-8")
            )

        return [
            self._normalise(item)
            for item in payload.get("results", [])
            if self._is_podcast_result(item)
        ]

    @staticmethod
    def _is_podcast_result(item: dict[str, object]) -> bool:
        return (
            str(item.get("kind", "")).lower() == "podcast"
            or str(item.get("wrapperType", "")).lower() == "track"
        )

    @staticmethod
    def _normalise(
        item: dict[str, object],
    ) -> PodcastSearchResult:
        language = normalise_language(
            str(item.get("language") or "unknown")
        )

        return PodcastSearchResult(
            provider=PodcastProviderName.APPLE,
            external_id=str(
                item.get("collectionId")
                or item.get("trackId")
                or ""
            ),
            title=str(
                item.get("collectionName")
                or item.get("trackName")
                or ""
            ),
            description=str(
                item.get("description")
                or item.get("longDescription")
                or ""
            ),
            publisher=str(
                item.get("artistName")
                or "Unknown publisher"
            ),
            languages=(language,),
            url=str(
                item.get("collectionViewUrl")
                or item.get("trackViewUrl")
                or ""
            ),
            image_url=str(
                item.get("artworkUrl600")
                or item.get("artworkUrl100")
                or ""
            ) or None,
            published_at=str(
                item.get("releaseDate") or ""
            ) or None,
            explicit=(
                str(
                    item.get("trackExplicitness", "")
                ).lower() == "explicit"
            ),
            metadata={"provider": "apple_podcasts"},
        )

"""
EasyChatBot Hackathon — personalised podcast recommendation service.

This is the application-level selection boundary.

Language is a HARD filter because EasyChatBot is a language-learning
platform. Preference/topic relevance is evaluated only after a candidate
is language-eligible.
"""

from __future__ import annotations

import re
from uuid import uuid4

from src.ai_runtime.podcast.models import (
    PodcastRecommendation,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)
from src.ai_runtime.podcast.providers import PodcastProvider


class PodcastRecommendationService:
    """Search providers and select a personalised candidate."""

    def __init__(self, providers: list[PodcastProvider]) -> None:
        if not providers:
            raise ValueError(
                "PodcastRecommendationService requires at least one provider."
            )
        self._providers = list(providers)

    def recommend(
        self,
        *,
        intent: PodcastSearchIntent,
        character_id: str,
        character_name: str,
        character_message: str,
    ) -> PodcastRecommendation:
        candidates: list[PodcastSearchResult] = []

        for provider in self._providers:
            candidates.extend(provider.search(intent))

        ranked = self.rank_candidates(
            candidates=candidates,
            intent=intent,
        )

        if not ranked:
            raise LookupError(
                "No podcast matched the learner's language and preferences."
            )

        result, score, reason = ranked[0]

        return PodcastRecommendation(
            recommendation_id=str(uuid4()),
            user_id=intent.user_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            result=result,
            relevance_reason=reason,
            match_score=score,
        )

    @classmethod
    def rank_candidates(
        cls,
        *,
        candidates: list[PodcastSearchResult],
        intent: PodcastSearchIntent,
    ) -> list[tuple[PodcastSearchResult, float, str]]:
        ranked: list[tuple[PodcastSearchResult, float, str]] = []

        target_language = normalise_language(
            intent.learning_context.target_language
        )

        preference_terms = [
            term.lower()
            for term in list(intent.topics) + list(intent.keywords)
            if term.strip()
        ]

        for candidate in candidates:
            candidate_languages = {
                normalise_language(language)
                for language in candidate.languages
            }

            # Critical learning-platform rule:
            # unknown/wrong-language content is not eligible.
            if (
                intent.language_required
                and target_language not in candidate_languages
            ):
                continue

            haystack = " ".join(
                [
                    candidate.title,
                    candidate.description,
                    candidate.publisher,
                ]
            ).lower()

            matched_terms = [
                term
                for term in preference_terms
                if cls._term_matches(term, haystack)
            ]

            score = 50.0

            if matched_terms:
                score += min(30.0, len(matched_terms) * 10.0)

            if intent.learning_context.world_id.lower() in haystack:
                score += 10.0

            level = intent.learning_context.learner_level.lower()
            if level and level in haystack:
                score += 10.0

            reason = cls._build_reason(
                language=intent.learning_context.target_language,
                matched_terms=matched_terms,
                world_id=intent.learning_context.world_id,
            )

            ranked.append(
                (candidate, min(score, 100.0), reason)
            )

        ranked.sort(
            key=lambda item: (
                item[1],
                item[0].published_at or "",
            ),
            reverse=True,
        )

        return ranked[: intent.max_results]

    @staticmethod
    def _term_matches(term: str, haystack: str) -> bool:
        term = term.strip().lower()
        if not term:
            return False
        if term in haystack:
            return True

        words = [
            word
            for word in re.findall(r"[a-z0-9À-ÿ'-]+", term)
            if len(word) > 2
        ]
        return bool(words) and any(
            word in haystack
            for word in words
        )

    @staticmethod
    def _build_reason(
        *,
        language: str,
        matched_terms: list[str],
        world_id: str,
    ) -> str:
        reasons = [f"It is available in {language}."]

        if matched_terms:
            reasons.append(
                "It matches your interests in "
                + ", ".join(matched_terms[:3])
                + "."
            )

        reasons.append(
            f"It fits your {world_id} learning world."
        )

        return " ".join(reasons)

        item: dict[str, object],
    ) -> PodcastSearchResult | None:

        title = str(item.get("name") or "")

        external_urls = item.get("external_urls")
        url = ""
        if isinstance(external_urls, dict):
            url = str(
                external_urls.get("spotify") or ""
            )

        if not external_id or not title or not url:
            return None

        raw_languages = item.get("languages", [])
        if not isinstance(raw_languages, list):
            return None

        languages = tuple(
            normalise_language(str(language))
            for language in raw_languages
            if str(language).strip()
        )

        # Unknown language is not invented. The recommendation service
        # will reject a candidate if language_required=True.
        if not languages:
            return None

        images = item.get("images", [])
        image_url = None
        if isinstance(images, list) and images:
            first = images[0]
            if isinstance(first, dict):
                image_url = str(first.get("url") or "") or None

        duration_ms = item.get("duration_ms")
        duration_seconds = (
            int(duration_ms) // 1000
            if isinstance(duration_ms, (int, float))
            else None
        )

        return PodcastSearchResult(
            provider=PodcastProviderName.SPOTIFY,
            external_id=external_id,
            title=title,
            description=str(item.get("description") or ""),
            publisher="Spotify Podcast",
            languages=languages,
            url=url,
            image_url=image_url,
            published_at=str(
                item.get("release_date") or ""
            ) or None,
            duration_seconds=duration_seconds,
            explicit=bool(item.get("explicit", False)),
            metadata={
                "provider": "spotify",
                "content_type": "episode",
            },

"""
Search Provider Interface

Defines the provider-independent contract for external search capabilities.

Architecture:
Agent -> Provider Interface -> Provider Adapter

The agent must never depend directly on a vendor SDK.
"""

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List


@dataclass(frozen=True)
class SearchResult:
    """
    Normalised search result returned by any provider.
    """

    title: str
    url: str
    content: str
    source: str


class SearchProvider(ABC):
    """
    Abstract search provider contract.

    Implementations:
    - ParallelSearchProvider
    - Future providers
    """

    @abstractmethod
    async def search(
        self,
        query: str,
        max_results: int = 5,
    ) -> List[SearchResult]:
        """
        Execute a search query.

        Args:
            query:
                Search query.

            max_results:
                Maximum number of results.

        Returns:
            Normalised search results.
        """
        raise NotImplementedError

"""
Parallel Search Provider Adapter.

Bridges the provider-independent SearchProvider contract
to the existing EasyChatBot ParallelClientWrapper.

Architecture:

Live Intelligence Agent
        |
        v
SearchProvider Interface
        |
        v
ParallelSearchProvider
        |
        v
Existing ParallelClientWrapper
        |
        v
Parallel Search API

The official Parallel SDK remains isolated inside:

    backend/providers/parallel/parallel_client.py

This adapter must not import or instantiate the Parallel SDK
directly.
"""

from typing import Any

from backend.providers.parallel.parallel_client import (
    ParallelClientWrapper,
)

from src.providers.search_provider.interface import (
    SearchProvider,
    SearchResult,
)


class ParallelSearchProvider(SearchProvider):
    """
    SearchProvider implementation backed by the existing
    ParallelClientWrapper.
    """

    def __init__(
        self,
        api_key: [REDACTED]"""
        Initialise the provider.

        Args:
            api_key:
                [REDACTED]"""

        self._client = client or ParallelClientWrapper(
            api_key=[REDACTED]"""
        Execute a Parallel search and normalise its results.

        Args:
            query:
                Natural-language research objective.

            max_results:
                Maximum number of results returned.

        Returns:
            Provider-independent SearchResult objects.
        """

        if not query.strip():
            raise ValueError(
                "Search query cannot be empty"
            )

        if max_results <= 0:
            raise ValueError(
                "max_results must be greater than zero"
            )

        response = await self._client.search(
            query=query
        )

        raw_results = self._extract_results(
            response
        )

        normalised_results: list[SearchResult] = []

        for raw_result in raw_results[:max_results]:
            result = self._normalise_result(
                raw_result
            )

            if result is not None:
                normalised_results.append(result)

        return normalised_results

    @staticmethod
    def _extract_results(
        response: Any,
    ) -> list[Any]:
        """
        Extract the actual result collection from the
        ParallelClientWrapper response.

        The existing wrapper returns:

            {
                "query": "...",
                "search_queries": [...],
                "results": <normalised SDK response>
            }

        The SDK response may itself contain:

            {
                "results": [...]
            }

        Therefore this method handles both levels of wrapping.
        """

        if not isinstance(response, dict):
            raise TypeError(
                "Parallel client response must be a dictionary"
            )

        raw_results = response.get(
            "results",
            []
        )

        # The SDK response itself may contain a nested
        # "results" collection.
        if isinstance(raw_results, dict):
            nested_results = raw_results.get(
                "results"
            )

            if isinstance(nested_results, list):
                return nested_results

            # Some SDK versions may expose the result
            # collection under another common field.
            nested_data = raw_results.get(
                "data"
            )

            if isinstance(nested_data, list):
                return nested_data

        if isinstance(raw_results, list):
            return raw_results

        # If the SDK returned an object that was not converted
        # to a dictionary, try to inspect its result attribute.
        nested_results = getattr(
            raw_results,
            "results",
            None,
        )

        if isinstance(nested_results, list):
            return nested_results

        raise TypeError(
            "Parallel search response does not contain "
            "a supported results collection"
        )

    @staticmethod
    def _normalise_result(
        raw_result: Any,
    ) -> SearchResult | None:
        """
        Convert one Parallel result into SearchResult.

        Parallel results may arrive either as dictionaries
        or SDK/Pydantic objects.
        """

        if isinstance(raw_result, dict):
            title = str(
                raw_result.get(
                    "title",
                    ""
                )
            ).strip()

            url = str(
                raw_result.get(
                    "url",
                    ""
                )
            ).strip()

            excerpts = raw_result.get(
                "excerpts",
                []
            )

            # Support providers/SDK versions that use
            # "snippet" or "content" instead.
            if not excerpts:
                excerpts = raw_result.get(
                    "snippet",
                    raw_result.get(
                        "content",
                        []
                    ),
                )

        else:
            title = str(
                getattr(
                    raw_result,
                    "title",
                    ""
                )
            ).strip()

            url = str(
                getattr(
                    raw_result,
                    "url",
                    ""
                )
            ).strip()

            excerpts = getattr(
                raw_result,
                "excerpts",
                None,
            )

            if not excerpts:
                excerpts = getattr(
                    raw_result,
                    "snippet",
                    getattr(
                        raw_result,
                        "content",
                        [],
                    ),
                )

        if not title or not url:
            return None

        content = ParallelSearchProvider._normalise_excerpts(
            excerpts
        )

        return SearchResult(
            title=title,
            url=url,
            content=content,
            source="parallel",
        )

    @staticmethod
    def _normalise_excerpts(
        excerpts: Any,
    ) -> str:
        """
        Convert Parallel excerpts/content into a single
        application-level content string.
        """

        if excerpts is None:
            return ""

        if isinstance(excerpts, str):
            return excerpts.strip()

        if isinstance(excerpts, list):
            return "\n\n".join(
                str(excerpt).strip()
                for excerpt in excerpts
                if str(excerpt).strip()
            )

        return str(excerpts).strip()

    """

    - is considered medium risk
    - does not require human approval for passive research

    return GovernancePolicy(

        requires_human_review=False,
        allows_external_calls=True,

        == RiskLevel.MEDIUM
    )

        == GovernanceDecision.ALLOW
    )

@dataclass
class FakeMemory:
    """
    Deterministic memory record for tests.
    """

    user_id: str
    category: str
    content: str


class FakeMemoryService:
    """
    Deterministic MemoryService test double.
    """

    def __init__(self) -> None:
        self.memories = [
            FakeMemory(
                user_id="demo-user",
                category="conversation",
                content="Previous conversation",
            ),
        ]

        self.recall_calls = []
        self.remember_calls = []

    def recall(
        self,
        user_id: str,
    ):
        self.recall_calls.append(
            user_id
        )

        return [
            memory
            for memory in self.memories
            if memory.user_id == user_id
        ]

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> None:
        self.remember_calls.append(
            {
                "user_id": user_id,
                "category": category,
                "content": content,
            }
        )
