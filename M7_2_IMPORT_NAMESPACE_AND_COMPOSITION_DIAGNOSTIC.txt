from __future__ import annotations

from datetime import datetime, timezone
from typing import Any

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel, Field

from app.application.privacy.processing_registry import ProcessingRegistry
from app.application.privacy.screening.adapter import ProcessingActivityScreeningAdapter


class ScreeningRequest(BaseModel):
    """Production screening command. Only processing_id is client-controlled."""

    processing_id: str = Field(min_length=1)


def create_privacy_screening_router(
    *,
    processing_registry: ProcessingRegistry,
    screening_adapter: ProcessingActivityScreeningAdapter,
) -> APIRouter:
    router = APIRouter(
        prefix="/api/privacy/screening",
        tags=["privacy-screening"],
    )

    @router.get("/processing-activities")
    def list_processing_activities() -> list[dict[str, Any]]:
        return [
            {
                "processing_id": activity.processing_id,
                "name": activity.name,
                "purpose": activity.purpose,
                "data_subjects": list(activity.data_subjects),
                "data_categories": list(activity.data_categories),
                "special_category_data": bool(
                    activity.metadata.get("special_category_data", False)
                ),
                "processor_ids": list(activity.processor_ids),
                "transfer_status": activity.metadata.get("transfer_status"),
                "last_material_change_at": activity.metadata.get(
                    "last_material_change_at"
                ),
            }
            for activity in processing_registry.all()
        ]

    @router.post("/screen")
    def screen(request: ScreeningRequest) -> dict[str, Any]:
        try:
            activity = processing_registry.get(request.processing_id)
        except KeyError as exc:
            raise HTTPException(
                status_code=404,
                detail="Processing activity not found.",
            ) from exc

        result = screening_adapter.screen(activity)

        def criterion_payload(criterion: Any) -> dict[str, Any]:
            return {
                "code": criterion.code,
                "label": criterion.label,
                "satisfied": criterion.satisfied,
                "source": criterion.source,
                "source_version": criterion.source_version,
                "rationale": criterion.rationale,
            }

        return {
            "screening_id": result.screening_id,
            "processing_id": result.processing_id,
            "status": result.status,
            "outcomes": [
                {
                    "assessment": outcome["assessment"],
                    "outcome": str(outcome["outcome"]),
                    "matched_criteria": [
                        criterion_payload(c)
                        for c in outcome.get("matched_criteria", [])
                    ],
                    "rationale": outcome["rationale"],
                }
                for outcome in result.outcomes
            ],
            "reassessment_required": result.reassessment_required,
            "provenance": [
                criterion_payload(c) for c in result.provenance
            ],
            "rationale": result.rationale,
            "created_at": datetime.now(timezone.utc).isoformat(),
        }

    return router


__all__ = ["ScreeningRequest", "create_privacy_screening_router"]

from __future__ import annotations

"""Adapter between authoritative ProcessingActivity data and M3 screening."""

from ..processing_activity import ProcessingActivity
from .models import ScreeningCriterion, ScreeningResult
from .service import ScreeningService


class ProcessingActivityScreeningAdapter:
    """Derive governed screening input and invoke the existing M3 seam.

    The browser never supplies screening criteria or legal-materiality
    assertions. Missing legally material information fails closed.
    """

    def __init__(self, screening_service: ScreeningService | None = None) -> None:
        self._screening_service = screening_service or ScreeningService()

    def screen(self, activity: ProcessingActivity) -> ScreeningResult:
        metadata = activity.metadata

        criteria = (
            ScreeningCriterion(
                code="SPECIAL_CATEGORY_DATA",
                label="Special-category data is explicitly recorded",
                satisfied=bool(metadata.get("special_category_data", False)),
                source="ProcessingActivity.metadata",
                source_version="beta1-processing-catalog-v1",
                rationale="Derived from the controlled Processing Activity record.",
            ),
            ScreeningCriterion(
                code="EXTERNAL_DESTINATION",
                label="External destination is recorded",
                satisfied=bool(activity.destinations),
                source="ProcessingActivity",
                source_version="beta1-processing-catalog-v1",
                rationale="Derived from authoritative processing destinations.",
            ),
            ScreeningCriterion(
                code="LEGAL_BASIS_PENDING",
                label="Lawful basis is not yet finally established",
                satisfied=activity.legal_basis.startswith("PENDING_"),
                source="ProcessingActivity",
                source_version="beta1-processing-catalog-v1",
                rationale="A pending lawful-basis value prevents final applicability.",
            ),
        )

        return self._screening_service.screen(
            processing_id=activity.processing_id,
            criteria=criteria,
            legal_materiality_established=bool(
                metadata.get("legal_materiality_established", False)
            ),
            material_change_detected=bool(
                metadata.get("material_change_detected", False)
            ),
        )


__all__ = ["ProcessingActivityScreeningAdapter"]

from uuid import uuid4

from .models import AssessmentOutcome, ScreeningCriterion, ScreeningResult


class ScreeningService:
    """M3 screening/applicability execution seam.

    Final DPO/legal decisions remain outside this service. Unknown legally
    material inputs fail closed.
    """

    def screen(
        self,
        *,
        processing_id: str,
        criteria: tuple[ScreeningCriterion, ...],
        legal_materiality_established: bool,
        material_change_detected: bool,
    ) -> ScreeningResult:
        screening_id = f"SCR-{uuid4().hex[:12].upper()}"

        if not legal_materiality_established:
            return ScreeningResult(
                screening_id=screening_id,
                processing_id=processing_id,
                status="LEGAL_REVIEW_REQUIRED",
                outcomes=(
                    {
                        "assessment": "DPIA",
                        "outcome": AssessmentOutcome.LEGAL_REVIEW_REQUIRED,
                        "matched_criteria": [],
                        "rationale": "Legally material screening input cannot be established.",
                    },
                    {
                        "assessment": "FRIA",
                        "outcome": AssessmentOutcome.LEGAL_REVIEW_REQUIRED,
                        "matched_criteria": [],
                        "rationale": "Fundamental-rights applicability requires human/legal determination.",
                    },
                    {
                        "assessment": "TIA",
                        "outcome": AssessmentOutcome.LEGAL_REVIEW_REQUIRED,
                        "matched_criteria": [],
                        "rationale": "Transfer applicability requires approved methodology and transfer context.",
                    },
                ),
                reassessment_required=material_change_detected,
                provenance=criteria,
                rationale="Human DPO/legal review is required before applicability can be concluded.",
            )

        matched = tuple(item for item in criteria if item.satisfied)
        return ScreeningResult(
            screening_id=screening_id,
            processing_id=processing_id,
            status="COMPLETED",
            outcomes=(
                {
                    "assessment": "DPIA",
                    "outcome": AssessmentOutcome.REQUIRED if matched else AssessmentOutcome.NOT_REQUIRED,
                    "matched_criteria": list(matched),
                    "rationale": (
                        "At least one configured governed criterion is satisfied."
                        if matched
                        else "No configured governed criterion is satisfied."
                    ),
                },
            ),
            reassessment_required=material_change_detected,
            provenance=criteria,
            rationale="Screening completed from governed processing context.",
        )
