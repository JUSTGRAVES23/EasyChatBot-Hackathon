"""
Memory Service.

Application-layer memory service.

This service adapts the legacy backend MemoryRecord contract
to the hackathon AI Runtime MemoryEntry / MemoryStore
persistence contract.

Architecture:

Application
    |
    v
MemoryService
    |
    v
MemoryEntry / MemoryType
    |
    v
MemoryStore
    |
    v
SQLite

Responsibilities:
- save application memories
- retrieve application memories
- translate application memory categories
- isolate persistence details from callers

Does not:
- perform semantic retrieval
- call an LLM
- execute agents
- make autonomous memory decisions
"""

from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from backend.memory.memory_record import (
    MemoryRecord,
)

from backend.memory.memory_store import (
    MemoryStore,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryService:
    """
    Controls application memory operations.

    The service keeps the existing MemoryRecord interface
    for the backend application while delegating persistence
    to the current AI Runtime MemoryStore.
    """

    DEFAULT_DATABASE_PATH = (
        Path("data") / "memory.db"
    )

    def __init__(
        self,
        store: MemoryStore | None = None,
        database_path: str | Path | None = None,
    ) -> None:
        """
        Initialise the memory service.

        Args:
            store:
                Optional injected MemoryStore.

                Primarily useful for tests and controlled
                application configuration.

            database_path:
                Optional SQLite database path.

                Ignored when an explicit store is supplied.
        """

        if store is not None:
            self.store = store
            return

        path = (
            Path(database_path)
            if database_path is not None
            else self.DEFAULT_DATABASE_PATH
        )

        path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        self.store = MemoryStore(
            database_path=path
        )

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> MemoryRecord:
        """
        Save an application memory.

        The legacy category is preserved in metadata while
        the persistence layer receives a supported MemoryType.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        if not category.strip():
            raise ValueError(
                "Memory category cannot be empty."
            )

        if not content.strip():
            raise ValueError(
                "Memory content cannot be empty."
            )

        created_at = datetime.now(
            timezone.utc
        )

        record = MemoryRecord(
            user_id=user_id,
            category=category,
            content=content,
            created_at=created_at,
        )

        memory_entry = MemoryEntry(
            user_id=user_id,
            memory_type=self._to_memory_type(
                category
            ),
            content=content,
            created_at=created_at,
            updated_at=created_at,
            metadata={
                "category": category,
            },
        )

        self.store.save(
            memory_entry
        )

        return record

    def recall(
        self,
        user_id: str,
    ) -> list[MemoryRecord]:
        """
        Retrieve memories for a user.

        Persistence-specific MemoryEntry objects are
        translated back into the application-facing
        MemoryRecord contract.
        """

        entries = self.store.list_for_user(
            user_id
        )

        return [
            self._to_memory_record(entry)
            for entry in entries
        ]

    @staticmethod
    def _to_memory_type(
        category: str,
    ) -> MemoryType:
        """
        Translate application categories into the
        supported AI Runtime memory types.

        The original category remains available in
        MemoryEntry.metadata.
        """

        normalized = category.strip().lower()

        if normalized == "saved_item":
            return MemoryType.SAVED_ITEM

        if normalized == "discovery_context":
            return MemoryType.DISCOVERY_CONTEXT

        return MemoryType.PREFERENCE

    @staticmethod
    def _to_memory_record(
        entry: MemoryEntry,
    ) -> MemoryRecord:
        """
        Translate a persistent MemoryEntry into the
        application-facing MemoryRecord.
        """

        category = entry.metadata.get(
            "category",
            entry.memory_type.value,
        )

        return MemoryRecord(
            user_id=entry.user_id,
            category=category,
            content=entry.content,
            created_at=entry.created_at,
        )

"""
Hackathon Memory persistence.

This module provides the minimal persistent Memory Store
required by the hackathon.

Architecture:

MemoryService
    |
    v
MemoryStore
    |
    v
SQLite

The store is deliberately small.

It is responsible only for persistence and retrieval of
MemoryEntry objects.

It does not:

- decide what should be remembered
- call an LLM
- perform semantic retrieval
- execute agents
- perform governance decisions
- manage products
- manage vendor links
- process payments
"""

from __future__ import annotations

import json
import sqlite3
from datetime import datetime
from pathlib import Path

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryStore:
    """
    SQLite-backed persistence for hackathon memories.

    The database schema is created automatically when the
    store is initialised.
    """

    def __init__(
        self,
        database_path: str | Path,
    ) -> None:
        self._database_path = str(database_path)

        self._initialise_database()

    def _connect(self) -> sqlite3.Connection:
        """
        Create a SQLite connection.
        """

        connection = sqlite3.connect(
            self._database_path
        )

        connection.row_factory = sqlite3.Row

        return connection

    def _initialise_database(self) -> None:
        """
        Create the Memory table if it does not exist.
        """

        with self._connect() as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    memory_id TEXT PRIMARY KEY,
                    user_id TEXT NOT NULL,
                    memory_type TEXT NOT NULL,
                    content TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    metadata_json TEXT NOT NULL
                )
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_id
                ON memories(user_id)
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_type
                ON memories(user_id, memory_type)
                """
            )

    def save(
        self,
        memory: MemoryEntry,
    ) -> MemoryEntry:
        """
        Persist a MemoryEntry.

        Existing entries with the same memory_id are
        replaced.
        """

        with self._connect() as connection:
            connection.execute(
                """
                INSERT OR REPLACE INTO memories (
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                )
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.memory_id,
                    memory.user_id,
                    memory.memory_type.value,
                    memory.content,
                    memory.created_at.isoformat(),
                    memory.updated_at.isoformat(),
                    json.dumps(
                        memory.metadata,
                        ensure_ascii=False,
                    ),
                ),
            )

        return memory

    def get(
        self,
        memory_id: str,
    ) -> MemoryEntry | None:
        """
        Retrieve a single memory by identifier.
        """

        with self._connect() as connection:
            row = connection.execute(
                """
                SELECT
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            ).fetchone()

        if row is None:
            return None

        return self._row_to_memory(row)

    def list_for_user(
        self,
        user_id: str,
        memory_type: MemoryType | None = None,
    ) -> list[MemoryEntry]:
        """
        Retrieve memories belonging to a user.

        Results are returned newest first.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        with self._connect() as connection:
            if memory_type is None:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                    ORDER BY updated_at DESC
                    """,
                    (user_id,),
                ).fetchall()

            else:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                      AND memory_type = ?
                    ORDER BY updated_at DESC
                    """,
                    (
                        user_id,
                        memory_type.value,
                    ),
                ).fetchall()

        return [
            self._row_to_memory(row)
            for row in rows
        ]

    def delete(
        self,
        memory_id: str,
    ) -> bool:
        """
        Delete a memory.

        Returns True when a memory was deleted.
        """

        with self._connect() as connection:
            cursor = connection.execute(
                """
                DELETE FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            )

        return cursor.rowcount > 0

    @staticmethod
    def _row_to_memory(
        row: sqlite3.Row,
    ) -> MemoryEntry:
        """
        Convert a SQLite row into a MemoryEntry.
        """

        return MemoryEntry(
            memory_id=row["memory_id"],
            user_id=row["user_id"],
            memory_type=MemoryType(
                row["memory_type"]
            ),
            content=row["content"],
            created_at=datetime.fromisoformat(
                row["created_at"]
            ),
            updated_at=datetime.fromisoformat(
                row["updated_at"]
            ),
            metadata=json.loads(
                row["metadata_json"]
            ),
        )

"""
Runtime Execution Context.

Provides immutable runtime context shared with agents.

Responsibilities:
- identify an execution
- carry user request information
- provide runtime metadata
- optionally carry live intelligence context

Does not:
- execute providers
- manage memory
- modify systems
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any
from uuid import uuid4


@dataclass(frozen=True)
class ExecutionContext:
    """
    Runtime execution context shared with agents.

    Attributes
    ----------
    user_id:
        Identifier of the user initiating the execution.

    request:
        User request being processed.

    execution_id:
        Unique identifier for this execution.

    created_at:
        UTC timestamp when the context was created.

    metadata:
        Runtime metadata.

    intelligence_context:
        Optional provider-independent live intelligence
        generated by the Research Agent.

    Responsibilities:
    - identify an execution
    - carry user request information
    - provide runtime metadata
    - optionally carry live intelligence context

    Does not:
    - execute providers
    - manage memory
    - modify systems
    """

    user_id: str

    request: str

    execution_id: str = field(
        default_factory=lambda: str(uuid4())
    )

    created_at: datetime = field(
        default_factory=lambda: datetime.now(timezone.utc)
    )

    metadata: dict[str, Any] = field(
        default_factory=dict
    )

    intelligence_context: Any | None = None

    @property
    def has_intelligence_context(self) -> bool:
        """
        Return whether live intelligence is attached
        to this execution.
        """

        return self.intelligence_context is not None

class CompanionService:
    """
    Coordinates AI companion experiences.

    The service orchestrates agents but does not contain
    provider-specific logic.

    Providers are injected into agents by the composition
    layer.
    """

    def __init__(
        self,
        character: CharacterProfile,
        conversation_agent: ConversationAgent | None = None,
        research_agent: ResearchAgent | None = None,
        quality_agent: QualityAgent | None = None,
        memory_service: MemoryService | None = None,
    ) -> None:
        """
        Initialise the CompanionService.

        Dependencies are injectable so the service can be
        tested without invoking real infrastructure.
        """

        self.character = character

        self.conversation_agent = (
            conversation_agent
            if conversation_agent is not None
            else ConversationAgent(
                character
            )
        )

        self.research_agent = (
            research_agent
            if research_agent is not None
            else ResearchAgent()
        )

        self.quality_agent = (
            quality_agent
            if quality_agent is not None
            else QualityAgent()
        )

        self.memory_service = (
            memory_service
            if memory_service is not None
            else MemoryService()
        )

    async def interact(
        self,
        context: ExecutionContext,
        use_research: bool = False,
    ) -> dict[str, Any]:
        """
        Execute a complete companion interaction.

        When research is requested, the Research Agent
        produces an IntelligenceContext which is attached
        to a new immutable ExecutionContext before the
        Conversation Agent executes.

        The original ExecutionContext is never mutated.

        Parameters
        ----------
        context:
            Runtime execution context.

        use_research:
            Whether live research should be performed
            before the conversation response.
        """

        memories = self.memory_service.recall(
            context.user_id
        )

        research_result = None

        conversation_context = context

        if use_research:
            research_result = (
                await self.research_agent.execute(
                    context
                )
            )

            if (
                research_result.success
                and research_result.output is not None
            ):
                conversation_context = replace(
                    context,
                    intelligence_context=(
                        research_result.output
                    ),
                )

        conversation_result = (
            await self.conversation_agent.execute(
                conversation_context
            )
        )

        quality_result = (
            await self.quality_agent.execute(
                conversation_context
            )
        )

        self.memory_service.remember(
            user_id=context.user_id,
            category="conversation",
            content=context.request,
        )

        return {
            "character": self.character.name,

            "conversation": (
                conversation_result.output
            ),

            "research": (
                research_result.output
                if research_result is not None
                else None
            ),

            "quality": (
                quality_result.output
            ),

            "memory_count": len(memories),
        }

"""

Responsible for executing the Live Intelligence Agent workflow.

from dataclasses import dataclass
from typing import List, Protocol

from src.providers.search_provider.interface import (
    SearchProvider,
    SearchResult,
)


@dataclass(frozen=True)
class LiveIntelligenceRequest:
    """
    Input contract for Live Intelligence Agent.
    """

    topic: str
    language: str = "French"
    learner_level: str = "B2"


@dataclass(frozen=True)
class LiveIntelligenceResponse:
    """
    Output contract from Live Intelligence Agent.
    """

    topic: str
    summary: str
    sources: List[SearchResult]
    key_information: List[str]


class IntelligenceProcessor(Protocol):
    """
    Contract for future LLM-based processing.

    Keeps the agent independent from
    any specific model/provider.
    """

    async def summarise(
        self,
        results: List[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, List[str]]:
        ...


class LiveIntelligenceAgent:
    """
    Live Intelligence Agent implementation.
    """

    def __init__(
        self,
        search_provider: SearchProvider,
        processor: IntelligenceProcessor,
    ) -> None:
        self._search_provider = search_provider
        self._processor = processor

    async def execute(
        self,
        request: LiveIntelligenceRequest,
    ) -> LiveIntelligenceResponse:
        """
        Execute research workflow.

        Flow:

        Request
          |
          v
        Search Provider
          |
          v
        Intelligence Processor
          |
          v
        Structured Response
        """

        self._validate_request(request)

        search_results = await self._search_provider.search(
            query=request.topic,
            max_results=5,
        )

        summary, key_information = await self._processor.summarise(
            results=search_results,
            language=request.language,
            learner_level=request.learner_level,
        )

        return LiveIntelligenceResponse(
            topic=request.topic,
            summary=summary,
            sources=search_results,
            key_information=key_information,
        )

    @staticmethod
    def _validate_request(
        request: LiveIntelligenceRequest,
    ) -> None:
        """
        Validate required input.
        """

        if not request.topic.strip():
            raise ValueError(
                "Topic cannot be empty"
            )

"""
AI Runtime executor for Live Intelligence.

This adapter bridges the provider-independent AI Runtime
capability contract with the existing LiveIntelligenceAgent.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
LiveIntelligenceExecutor
    |
    v
LiveIntelligenceAgent
    |
    v
LiveIntelligenceResponse

The adapter:

- validates the Runtime capability
- translates Runtime payloads into LiveIntelligenceRequest
- invokes the existing Live Intelligence Agent
- translates the agent response into CapabilityResult

It does not:

- implement intelligence logic
- call Parallel directly
- modify the LiveIntelligenceAgent
- perform governance decisions
"""

from __future__ import annotations

from src.ai_agents.live_intelligence_agent.handler import (
    LiveIntelligenceAgent,
    LiveIntelligenceRequest,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)


class LiveIntelligenceExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for the existing LiveIntelligenceAgent.
    """

    def __init__(
        self,
        agent: LiveIntelligenceAgent,
    ) -> None:
        self._agent = agent

    @property
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """

        return CapabilityType.LIVE_INTELLIGENCE

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Translate and execute a Live Intelligence request.
        """

        if (
            request.capability
            != CapabilityType.LIVE_INTELLIGENCE
        ):
            raise ValueError(
                "LiveIntelligenceExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        agent_request = (
            self._build_agent_request(request)
        )

        response = await self._agent.execute(
            agent_request
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=response,
            metadata={
                "executor": (
                    "live_intelligence_executor"
                ),
            },
        )

    @staticmethod
    def _build_agent_request(
        request: CapabilityRequest,
    ) -> LiveIntelligenceRequest:
        """
        Translate Runtime payload into the existing
        LiveIntelligenceRequest contract.
        """

        payload = request.payload

        topic = payload.get("topic")

        if not isinstance(topic, str):
            raise ValueError(
                "Live Intelligence capability requires "
                "a string 'topic'."
            )

        language = payload.get(
            "language",
            "French",
        )

        learner_level = payload.get(
            "learner_level",
            "B2",
        )

        if not isinstance(language, str):
            raise ValueError(
                "Live Intelligence 'language' "
                "must be a string."
            )

        if not isinstance(
            learner_level,
            str,
        ):
            raise ValueError(
                "Live Intelligence 'learner_level' "
                "must be a string."
            )

        return LiveIntelligenceRequest(
            topic=topic,
            language=language,
            learner_level=learner_level,
        )

        memories: list[MemoryEntry] | None = None,
    ) -> FlashDealCard:
        """
        Create a personalised FlashDealCard.

        Memory is read-only input to this operation.

        Preference memory is used to generate the personal
        relevance explanation. Other memory types are not
        automatically treated as preferences.
        """

        memory_context = memories or []

        relevance_reason = self._build_relevance_reason(
            memory_context
        )

        return FlashDealCard.from_deal(
            deal=request.deal,
            character_id=request.character.character_id,
            character_name=request.character.character_name,
            character_message=request.character.message,
            relevance_reason=relevance_reason,
            product_image_url=request.product_image_url,
            tagline=request.tagline,
            highlights=request.highlights,
            metadata={
                "user_id": request.user_id,
                "discovery_service": "hackathon",
            },
        )

    @staticmethod
    def _build_relevance_reason(
        memories: list[MemoryEntry],
    ) -> str:
        """
        Build a deterministic relevance explanation.

        Only explicit preference memories are used.

        This is deliberately deterministic for the hackathon;
        an LLM-based personalisation layer can be introduced later
        behind an explicit interface.
        """

        preferences = [
            memory
            for memory in memories
            if memory.memory_type
            == MemoryType.PREFERENCE
        ]

        if not preferences:
            return (
                "This discovery may be relevant to your "
                "interests."
            )

        preference = preferences[0].content.strip()

        return (
            "This may be relevant because you "
            f"{preference.rstrip('.') }."
        )

"""
Hackathon Memory contracts.

This module defines the minimal persistent memory model
required by the hackathon discovery flow.

This is intentionally lightweight.

It is NOT the production EasyChatBot memory architecture.

Hackathon responsibilities:

- represent user memories
- distinguish memory categories
- provide stable identifiers
- preserve timestamps
- support future persistence

Out of scope:

- vector embeddings
- semantic retrieval
- LLM-generated memory extraction
- autonomous memory writes
- purchase transactions
- production learner-memory architecture
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from uuid import uuid4


class MemoryType(str, Enum):
    """
    Supported hackathon memory categories.
    """

    PREFERENCE = "preference"

    DISCOVERY_CONTEXT = "discovery_context"

    SAVED_ITEM = "saved_item"


@dataclass(frozen=True)
class MemoryEntry:
    """
    Persistent user memory entry.

    Memory is explicit application state.

    Agents must not directly mutate this object or
    the underlying persistence layer.
    """

    user_id: str

    memory_type: MemoryType

    content: str

    memory_id: str = field(
        default_factory=lambda: str(uuid4())
    )

    created_at: datetime = field(
        default_factory=lambda: datetime.now(
            timezone.utc
        )
    )

    updated_at: datetime = field(
        default_factory=lambda: datetime.now(
            timezone.utc
        )
    )

    metadata: dict[str, str] = field(
        default_factory=dict
    )

    def __post_init__(self) -> None:
        """
        Validate the memory contract.
        """

        if not self.user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        if not self.content.strip():
            raise ValueError(
                "Memory content cannot be empty."
            )

    def to_dict(self) -> dict[str, object]:
        """
        Convert the memory entry into a persistence-safe
        representation.
        """

        return {
            "memory_id": self.memory_id,
            "user_id": self.user_id,
            "memory_type": self.memory_type.value,
            "content": self.content,
            "created_at": self.created_at.isoformat(),
            "updated_at": self.updated_at.isoformat(),
            "metadata": dict(self.metadata),
        }

        self,

        self,
        *,

            user_id=user_id,
            memory_type=memory_type,

            metadata=(
                dict(metadata)
                if metadata is not None
                else {}
            ),
        )

        return self._store.save(
            memory
        )

    def get(
        self,
        memory_id: str,

        )

"""
Hackathon Memory persistence.

This module provides the minimal persistent Memory Store
required by the hackathon.

Architecture:

MemoryService
      |
      v
MemoryStore
      |
      v
SQLite

The store is deliberately small.

It is responsible only for persistence and retrieval of
MemoryEntry objects.

It does not:
- decide what should be remembered
- call an LLM
- perform semantic retrieval
- execute agents
- perform governance decisions
- manage purchases
"""

from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryStore:
    """
    SQLite-backed persistence for hackathon memories.

    The database schema is created automatically when the
    store is initialised.
    """

    def __init__(
        self,
        database_path: str | Path,
    ) -> None:
        self._database_path = str(database_path)

        self._initialise_database()

    def _connect(self) -> sqlite3.Connection:
        """
        Create a SQLite connection.
        """

        connection = sqlite3.connect(
            self._database_path
        )

        connection.row_factory = sqlite3.Row

        return connection

    def _initialise_database(self) -> None:
        """
        Create the Memory table if it does not exist.
        """

        with self._connect() as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    memory_id TEXT PRIMARY KEY,
                    user_id TEXT NOT NULL,
                    memory_type TEXT NOT NULL,
                    content TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    metadata_json TEXT NOT NULL
                )
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_id
                ON memories(user_id)
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_type
                ON memories(user_id, memory_type)
                """
            )

    def save(
        self,
        memory: MemoryEntry,
    ) -> MemoryEntry:
        """
        Persist a MemoryEntry.

        Existing entries with the same memory_id are
        replaced.
        """

        with self._connect() as connection:
            connection.execute(
                """
                INSERT OR REPLACE INTO memories (
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                )
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.memory_id,
                    memory.user_id,
                    memory.memory_type.value,
                    memory.content,
                    memory.created_at.isoformat(),
                    memory.updated_at.isoformat(),
                    json.dumps(
                        memory.metadata,
                        ensure_ascii=False,
                    ),
                ),
            )

        return memory

    def get(
        self,
        memory_id: str,
    ) -> MemoryEntry | None:
        """
        Retrieve a single memory by identifier.
        """

        with self._connect() as connection:
            row = connection.execute(
                """
                SELECT
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            ).fetchone()

        if row is None:
            return None

        return self._row_to_memory(row)

    def list_for_user(
        self,
        user_id: str,
        memory_type: MemoryType | None = None,
    ) -> list[MemoryEntry]:
        """
        Retrieve memories belonging to a user.

        Results are returned newest first.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        with self._connect() as connection:

            if memory_type is None:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                    ORDER BY updated_at DESC
                    """,
                    (user_id,),
                ).fetchall()

            else:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                      AND memory_type = ?
                    ORDER BY updated_at DESC
                    """,
                    (
                        user_id,
                        memory_type.value,
                    ),
                ).fetchall()

        return [
            self._row_to_memory(row)
            for row in rows
        ]

    def delete(
        self,
        memory_id: str,
    ) -> bool:
        """
        Delete a memory.

        Returns True when a memory was deleted.
        """

        with self._connect() as connection:
            cursor = connection.execute(
                """
                DELETE FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            )

        return cursor.rowcount > 0

    @staticmethod
    def _row_to_memory(
        row: sqlite3.Row,
    ) -> MemoryEntry:
        """
        Convert a SQLite row into a MemoryEntry.
        """

        return MemoryEntry(
            memory_id=row["memory_id"],
            user_id=row["user_id"],
            memory_type=MemoryType(
                row["memory_type"]
            ),
            content=row["content"],
            created_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["created_at"]
                )
            ),
            updated_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["updated_at"]
                )
            ),
            metadata=json.loads(
                row["metadata_json"]
            ),
        )

    """
    Deterministic MemoryService test double.
    """

    def __init__(self) -> None:
        self.memories = [
            FakeMemory(
                user_id="demo-user",

            ),
        ]

        self.recall_calls = []
        self.remember_calls = []

        self,

        self,

        category: str,
        content: str,
    ) -> None:

@pytest.mark.asyncio
async def test_companion_service_orchestrates_conversation():
    """
    CompanionService should execute the ConversationAgent
    and QualityAgent for a normal interaction.

    Research must not execute when use_research=False.
    """

    conversation_agent = FakeConversationAgent(
        AgentResult(
            success=True,
            agent_id="conversation_agent",
            output={
                "message": "Fake conversation response",
            },
        )
    )

    research_agent = FakeResearchAgent(
        AgentResult(
            success=True,
            agent_id="research_agent",
            output="Should not execute",
        )
    )

    quality_agent = FakeQualityAgent(
        AgentResult(
            success=True,
            agent_id="quality_agent",
            output={
                "quality_score": 1.0,
                "passed": True,
            },
        )
    )

    memory_service = FakeMemoryService()

    service = CompanionService(
        character=create_character(),
        conversation_agent=conversation_agent,
        research_agent=research_agent,
        quality_agent=quality_agent,
        memory_service=memory_service,
    )

    context = create_context()

    result = await service.interact(
        context,
        use_research=False,
    )

    assert result["character"] == "Yeti"

    assert result["conversation"] == {
        "message": "Fake conversation response",
    }

    assert result["research"] is None

    assert result["quality"] == {
        "quality_score": 1.0,
        "passed": True,
    }

    assert result["memory_count"] == 1

    assert (
        len(
            conversation_agent.received_contexts
        )
        == 1
    )

    assert (
        conversation_agent.received_contexts[0]
        is context
    )

    assert (
        len(
            research_agent.received_contexts
        )
        == 0
    )

    assert (
        len(
            quality_agent.received_contexts
        )
        == 1
    )

    assert (
        quality_agent.received_contexts[0]
        is context
    )

    assert memory_service.recall_calls == [
        "demo-user",
    ]

    assert memory_service.remember_calls == [
        {
            "user_id": "demo-user",
            "category": "conversation",
            "content": (
                "Explain artificial intelligence"
            ),
        }
    ]


@pytest.mark.asyncio
async def test_companion_service_executes_research_when_requested():
    """
    When research is requested, CompanionService should:

    1. execute ResearchAgent
    2. attach its IntelligenceContext to a new
       ExecutionContext
    3. execute ConversationAgent with the enriched context
    4. execute QualityAgent with the enriched context
    """

    intelligence_context = object()

    research_agent = FakeResearchAgent(
        AgentResult(
            success=True,
            agent_id="research_agent",
            output=intelligence_context,
        )
    )

    conversation_agent = FakeConversationAgent(
        AgentResult(
            success=True,
            agent_id="conversation_agent",
            output={
                "message": "Research-aware response",
            },
        )
    )

    quality_agent = FakeQualityAgent(
        AgentResult(
            success=True,
            agent_id="quality_agent",
            output={
                "quality_score": 0.95,
                "passed": True,
            },
        )
    )

    memory_service = FakeMemoryService()

    service = CompanionService(
        character=create_character(),
        conversation_agent=conversation_agent,
        research_agent=research_agent,
        quality_agent=quality_agent,
        memory_service=memory_service,
    )

    context = create_context(
        "What are the latest AI developments?"
    )

    result = await service.interact(
        context,
        use_research=True,
    )

    assert result["character"] == "Yeti"

    assert result["conversation"] == {
        "message": "Research-aware response",
    }

    assert (
        result["research"]
        is intelligence_context
    )

    assert result["quality"] == {
        "quality_score": 0.95,
        "passed": True,
    }

    assert (
        len(
            research_agent.received_contexts
        )
        == 1
    )

    assert (
        research_agent.received_contexts[0]
        is context
    )

    assert (
        len(
            conversation_agent.received_contexts
        )
        == 1
    )

    enriched_context = (
        conversation_agent.received_contexts[0]
    )

    assert enriched_context is not context

    assert (
        enriched_context.request
        == context.request
    )

    assert (
        enriched_context.user_id
        == context.user_id
    )

    assert (
        enriched_context.intelligence_context
        is intelligence_context
    )

    assert (
        quality_agent.received_contexts[0]
        is enriched_context
    )

    service = CompanionService(
        character=create_character(),
