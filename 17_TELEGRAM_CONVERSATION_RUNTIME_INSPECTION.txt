from __future__ import annotations

import logging
from typing import Callable, Awaitable

from app.application.beta_access.beta_access_service import BetaAccessService

logger = logging.getLogger(__name__)


class BetaTelegramStartHandler:
    """Adapter for /start <token>. Only private chats may bind Beta users."""

    def __init__(
        self,
        *,
        beta_access_service: BetaAccessService,
        reply: Callable[[str, object], Awaitable[None]],
    ) -> None:
        self._access = beta_access_service
        self._reply = reply

    async def handle(
        self,
        *,
        token: str | None,
        telegram_user_id: int,
        chat_id: int,
        chat_type: str,
    ) -> None:
        if chat_type != "private":
            await self._reply(
                "Please open a private chat with the Academy bot to connect your classroom.",
                None,
            )
            return

        if not token:
            await self._reply(
                "Your classroom connection link is missing or invalid.",
                None,
            )
            return

        try:
            user = await self._access.validate_and_bind(
                token=token,
                telegram_id=telegram_user_id,
                chat_id=chat_id,
            )
        except (ValueError, PermissionError) as exc:
            logger.warning(
                "Telegram Beta binding rejected telegram_id=%s reason=%s",
                telegram_user_id,
                exc,
            )
            await self._reply(str(exc), None)
            return

        await self._reply(
            (
                "Welcome to EasyChatBot Academy. "
                f"Your classroom is connected to World '{user.active_world_id}'."
            ),
            user,
        )

from __future__ import annotations


class BetaTelegramAdapter:
    """
    Thin adapter for python-telegram-bot.

    The adapter deliberately keeps Telegram transport concerns outside
    BetaAccessService.
    """

    def __init__(self, *, start_handler, private_chat_gate):
        self._start_handler = start_handler
        self._private_chat_gate = private_chat_gate

    async def handle_start(self, update, context):
        telegram_user_id = str(update.effective_user.id)
        chat_id = str(update.effective_chat.id)
        args = context.args or []

        token = args[0] if args else ""

        result = self._start_handler.handle(
            token=token,
            telegram_id=telegram_user_id,
            chat_id=chat_id,
        )

        await update.effective_message.reply_text(result.message)

    def is_private_chat_allowed(self, *, user_id: str, chat_type: str) -> bool:
        return self._private_chat_gate.authorize(
            user_id=user_id,
            chat_type=chat_type,
        )

from __future__ import annotations

import hashlib
import logging
import secrets
from dataclasses import dataclass, replace
from datetime import datetime, timedelta, timezone
from typing import Iterator

from .models import BetaConnectionToken, BetaUserState, TelegramBinding
from .repositories import (
    BetaTokenRepository,
    BetaUserRepository,
    TelegramBindingRepository,
)

logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class BetaConnectionTokenResult:
    """
    Result of creating a Beta Telegram connection token.

    Compatibility:
    - ``result.token`` provides the raw token.
    - ``result.connection`` provides the persisted token record.
    - Tuple unpacking remains supported::

          raw_token, connection = result

    The raw token is never persisted. Only its SHA-256 hash is stored.
    """

    token: str
    connection: BetaConnectionToken

    def __iter__(self) -> Iterator[object]:
        yield self.token
        yield self.connection


class BetaAccessService:
    """Application boundary for Beta eligibility and Telegram binding."""

    def __init__(
        self,
        *,
        user_repository: BetaUserRepository,
        token_repository: BetaTokenRepository,
        binding_repository: TelegramBindingRepository,
        expiration_minutes: int = 15,
    ) -> None:
        if expiration_minutes <= 0:
            raise ValueError("expiration_minutes must be positive.")

        self._users = user_repository
        self._tokens = token_repository
        self._bindings = binding_repository
        self._expiration_minutes = expiration_minutes

    async def approve_beta(self, user_id: str) -> BetaUserState:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before Beta approval."
            )

        return await self._users.save(
            replace(user, beta_approved=True)
        )

    async def assign_world(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> BetaUserState:
        if not world_id:
            raise ValueError("world_id is required.")

        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before World assignment."
            )

        if not user.beta_approved:
            raise ValueError("User is not Beta approved.")

        return await self._users.save(
            replace(
                user,
                active_world_id=world_id,
            )
        )

    async def create_connection_token(
        self,
        user_id: str,
    ) -> BetaConnectionTokenResult:
        """
        Create a short-lived one-time Telegram connection token.

        The raw token is returned to the application layer but is never
        persisted. Only its SHA-256 hash is stored.
        """
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError("Email is not verified.")

        if not user.beta_approved:
            raise ValueError("User is not Beta approved.")

        if not user.active_world_id:
            raise ValueError(
                "User must have a Learning World before Telegram access."
            )

        raw_token = secrets.token_urlsafe(32)

        token = BetaConnectionToken(
            token_id=secrets.token_hex(16),
            user_id=user.user_id,
            token_hash=self.hash_token(raw_token),
            expires_at=(
                datetime.now(timezone.utc)
                + timedelta(minutes=self._expiration_minutes)
            ),
        )

        await self._tokens.save(token)

        logger.info(
            "Beta Telegram token created user_id=%s",
            user.user_id,
        )

        return BetaConnectionTokenResult(
            token=raw_token,
            connection=token,
        )

    async def validate_and_bind(
        self,
        *,
        token: str,
        telegram_id: int,
        chat_id: int,
    ) -> BetaUserState:
        if not token:
            raise ValueError(
                "Connection token is required."
            )

        if telegram_id <= 0:
            raise ValueError(
                "Telegram ID must be positive."
            )

        if chat_id == 0:
            raise ValueError(
                "Chat ID is required."
            )

        record = await self._tokens.get_by_hash(
            self.hash_token(token)
        )

        if record is None:
            raise ValueError(
                "Invalid connection token."
            )

        if record.used:
            raise ValueError(
                "Connection token already used."
            )

        expires_at = record.expires_at

        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(
                tzinfo=timezone.utc
            )

        if expires_at <= datetime.now(timezone.utc):
            raise ValueError(
                "Connection token expired."
            )

        user = await self._require_user(
            record.user_id
        )

        if not user.eligible:
            raise ValueError(
                "User is not eligible for Beta Telegram access."
            )

        if not user.active_world_id:
            raise ValueError(
                "User has no assigned Learning World."
            )

        existing = await self._users.get_by_telegram_id(
            telegram_id
        )

        if (
            existing is not None
            and existing.user_id != user.user_id
        ):
            raise ValueError(
                "Telegram account is already connected "
                "to another user."
            )

        existing_binding = await self._bindings.get_by_user_id(
            user.user_id
        )

        if (
            existing_binding is not None
            and existing_binding.telegram_user_id
            != str(telegram_id)
        ):
            raise ValueError(
                "User already has another Telegram binding."
            )

        await self._bindings.save(
            TelegramBinding(
                binding_id=secrets.token_hex(16),
                user_id=user.user_id,
                telegram_user_id=str(telegram_id),
                chat_id=str(chat_id),
                connected_at=datetime.now(timezone.utc),
            )
        )

        await self._tokens.mark_used(
            replace(
                record,
                used_at=datetime.now(timezone.utc),
            )
        )

        return await self._users.save(
            replace(
                user,
                telegram_id=telegram_id,
            )
        )

    async def authorize_private_chat(
        self,
        telegram_id: int,
    ) -> BetaUserState:
        if telegram_id <= 0:
            raise ValueError(
                "Telegram ID must be positive."
            )

        user = await self._users.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            raise PermissionError(
                "Telegram account is not connected."
            )

        if not user.active:
            raise PermissionError(
                "User is connected but is not ACTIVE "
                "for a Learning World."
            )

        return user

    @staticmethod
    def hash_token(token: str) -> str:
        return hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

    async def _require_user(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._users.get(user_id)

        if user is None:
            raise ValueError(
                "Academy user does not exist."
            )

        return user

from __future__ import annotations

import hashlib
import secrets


class BetaConnectionTokenService:
    TOKEN_BYTES = 32

    @classmethod
    def generate(cls) -> tuple[str, str]:
        raw = secrets.token_urlsafe(cls.TOKEN_BYTES)
        return raw, cls.hash_token(raw)

    @staticmethod
    def hash_token(raw_token: str) -> str:
        if not isinstance(raw_token, str) or not raw_token:
            raise ValueError("raw_token must be a non-empty string.")
        return hashlib.sha256(raw_token.encode("utf-8")).hexdigest()

"""
OpenAI Speech Provider.

Concrete speech provider implementation for OpenAI.

Responsibilities:
- Transcribe audio using OpenAI speech-to-text.
- Synthesize speech using OpenAI text-to-speech.
- Hide OpenAI SDK details behind SpeechProvider.

Does NOT:
- Handle HTTP.
- Manage application workflows.
- Manage LiveKit.
- Persist conversations.
- Execute AI agents.
"""

from __future__ import annotations


import io


import os


from typing import Any


from openai import OpenAI


from app.providers.speech.speech_provider import (
    SpeechProvider,
)



class OpenAISpeechProvider(
    SpeechProvider,
):
    """
    OpenAI implementation of the speech provider contract.

    The OpenAI client is created lazily so that importing
    the provider and running unit tests does not require
    OPENAI_API_KEY to be present.

    A client can also be injected for testing.
    """


    DEFAULT_TRANSCRIPTION_MODEL = (
        "gpt-4o-mini-transcribe"
    )


    DEFAULT_TTS_MODEL = (
        "gpt-4o-mini-tts"
    )


    DEFAULT_VOICE = (
        "alloy"
    )


    def __init__(
        self,
        *,
        client: Any | None = None,
        transcription_model: str | None = None,
        tts_model: str | None = None,
    ) -> None:
        """
        Initialize the OpenAI speech provider.

        Parameters
        ----------
        client:
            Optional OpenAI client.

            Primarily intended for tests and dependency
            injection.

        transcription_model:
            Optional OpenAI transcription model.

        tts_model:
            Optional OpenAI text-to-speech model.
        """

        self._client = client

        self._transcription_model = (
            transcription_model
            or os.getenv(
                "OPENAI_TRANSCRIPTION_MODEL",
                self.DEFAULT_TRANSCRIPTION_MODEL,
            )
        )

        self._tts_model = (
            tts_model
            or os.getenv(
                "OPENAI_TTS_MODEL",
                self.DEFAULT_TTS_MODEL,
            )
        )


    @property
    def client(self) -> OpenAI:
        """
        Return the configured OpenAI client.

        The client is initialized lazily.
        """

        if self._client is None:

            api_key = os.getenv(
                "OPENAI_API_KEY"
            )

            if not api_key:

                raise RuntimeError(
                    "OPENAI_API_KEY is not configured."
                )

            self._client = OpenAI(
                api_key=api_key,
            )

        return self._client


    def transcribe(
        self,
        *,
        audio: bytes,
        language_code: str,
    ) -> str:
        """
        Transcribe audio into text.

        Parameters
        ----------
        audio:
            Raw audio bytes.

        language_code:
            ISO language code, for example ``fr``.

        Returns
        -------
        str
            Transcribed text.
        """

        if not audio:

            raise ValueError(
                "audio cannot be empty."
            )


        if not language_code.strip():

            raise ValueError(
                "language_code cannot be empty."
            )


        audio_file = io.BytesIO(
            audio
        )


        audio_file.name = (
            "voice_input.wav"
        )


        response = (
            self.client
            .audio
            .transcriptions
            .create(
                model=(
                    self._transcription_model
                ),
                file=audio_file,
                language=(
                    language_code
                ),
            )
        )


        text = getattr(
            response,
            "text",
            None,
        )


        if text is None:

            raise RuntimeError(
                "OpenAI transcription response "
                "did not contain text."
            )


        return str(text)



    def synthesize(
        self,
        *,
        text: str,
        voice_profile: str,
        language_code: str,
    ) -> bytes:
        """
        Generate speech audio.

        Parameters
        ----------
        text:
            Text to synthesize.

        voice_profile:
            Character voice profile.

        language_code:
            Language requested by the caller.

        Returns
        -------
        bytes
            Generated audio bytes.
        """

        if not text.strip():

            raise ValueError(
                "text cannot be empty."
            )


        if not voice_profile.strip():

            raise ValueError(
                "voice_profile cannot be empty."
            )


        if not language_code.strip():

            raise ValueError(
                "language_code cannot be empty."
            )


        # Character voice profiles are currently mapped
        # to provider-supported voices at the infrastructure
        # boundary. This prevents provider-specific voice
        # identifiers from leaking into the domain/runtime.
        voice = self._resolve_voice(
            voice_profile
        )


        response = (
            self.client
            .audio
            .speech
            .create(
                model=(
                    self._tts_model
                ),
                voice=voice,
                input=text,
            )
        )


        audio_content = getattr(
            response,
            "content",
            None,
        )


        if audio_content is not None:

            return bytes(
                audio_content
            )


        if hasattr(
            response,
            "read",
        ):

            return bytes(
                response.read()
            )


        raise RuntimeError(
            "OpenAI speech response "
            "did not contain audio content."
        )



    @classmethod
    def _resolve_voice(
        cls,
        voice_profile: str,
    ) -> str:
        """
        Map an application character voice profile
        to an OpenAI-supported voice.

        Character identifiers such as ``yeti`` must not
        be sent directly to the provider because they are
        application concepts rather than provider voice IDs.
        """

        normalized = (
            voice_profile
            .strip()
            .lower()
        )


        voice_mapping = {
            "yeti": "onyx",
        }


        return voice_mapping.get(
            normalized,
            normalized
            if normalized in {
                "alloy",
                "ash",
                "ballad",
                "coral",
                "echo",
                "fable",
                "nova",
                "onyx",
                "sage",
                "shimmer",
                "verse",
                "marin",
                "cedar",
            }
            else cls.DEFAULT_VOICE,
        )



__all__ = [
    "OpenAISpeechProvider",
]

def run_migrations_offline() -> None:
    """Run migrations without creating a database connection."""
    url = get_database_url()

    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        compare_type=True,
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations using a live database connection."""
    configuration = config.get_section(config.config_ini_section) or {}
    configuration["sqlalchemy.url"] = get_database_url()

    connectable = engine_from_config(
        configuration,
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata,
            compare_type=True,
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

from __future__ import annotations

from sqlalchemy.orm import DeclarativeBase, sessionmaker, Session


class Base(DeclarativeBase):
    pass


def database_url() -> str:
    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for Beta Telegram persistence."
        )
    return value


def create_database_engine():
    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )

    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )
