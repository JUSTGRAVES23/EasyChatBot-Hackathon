=== GEMINI PROVIDER ===
"""
Gemini LLM Provider.

Concrete Gemini implementation of the provider-independent
LLMProvider contract used by the EasyChatBot hackathon runtime.

Architecture:

    ConversationAgent
            |
            v
    ConversationRequest
            |
            v
       LLMProvider
            |
            v
      GeminiProvider
            |
            v
       Gemini API
            |
            v
      ProviderResult

Responsibilities:
- execute ConversationRequest through Gemini
- use the configured Gemini model
- translate Gemini responses into ProviderResult
- isolate the Gemini SDK from the application/runtime layers

Does not:
- build application workflows
- perform research
- manage memory
- select providers
- manage conversations
- modify application state

The provider uses Google's current `google-genai` SDK.
The SDK client is injectable so unit tests do not need a real
Gemini API call.
"""

from __future__ import annotations

import logging
import os
from typing import Any

from backend.conversation.conversation_request import (
    ConversationRequest,
)
from backend.providers.llm.llm_provider import (
    LLMProvider,
)
from backend.providers.provider_result import (
    ProviderResult,
)


logger = logging.getLogger(__name__)


class GeminiProvider(LLMProvider):
    """
    Gemini implementation of the LLMProvider contract.

    The Google Gemini SDK is deliberately isolated inside this
    infrastructure adapter.

    Args:
        model:
            Gemini model identifier.

        client:
            Optional injected `google.genai.Client`.

            Dependency injection allows unit tests to provide a
            fake client without making a real API call.
    """

    def __init__(
        self,
        model: str = "gemini-3.6-flash",
        client: Any | None = None,
    ) -> None:
        super().__init__(
            provider_id="gemini",
        )

        if not isinstance(model, str) or not model.strip():
            raise ValueError(
                "Gemini model cannot be empty."
            )

        self.model = model.strip()
        self._client = client

    def _get_client(self) -> Any:
        """
        Lazily create the Google GenAI client.

        The API key is read from GEMINI_API_KEY and is never
        hard-coded into the provider.
        """

        if self._client is not None:
            return self._client

        api_key = os.getenv("GEMINI_API_KEY")

        if not api_key:
            raise RuntimeError(
                "GEMINI_API_KEY environment variable is required."
            )

        try:
            from google import genai
        except ImportError as exc:
            raise RuntimeError(
                "The google-genai package is required for "
                "GeminiProvider. Install it with: "
                "pip install google-genai"
            ) from exc

        self._client = genai.Client(
            api_key=api_key,
        )

        return self._client

    @staticmethod
    def _build_contents(
        request: ConversationRequest,
    ) -> str:
        """
        Convert the provider-independent request into the
        user-facing content sent to Gemini.

        System behaviour remains separate and is supplied through
        Gemini's system_instruction configuration.

        Live intelligence, when present, is deliberately carried
        through the existing ConversationRequest context rather
        than retrieved by Gemini itself.
        """

        sections: list[str] = []

        if request.conversation_context:
            sections.extend(
                (
                    "CONTEXT:",
                    request.conversation_context.strip(),
                    "",
                )
            )

        sections.extend(
            (
                "USER REQUEST:",
                request.user_prompt.strip(),
            )
        )

        return "\n".join(sections)

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate a response using Gemini.

        Returns a provider-independent ProviderResult and never
        exposes Gemini SDK response objects to the application.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        try:
            client = self._get_client()

            from google.genai import types

            contents = self._build_contents(
                request,
            )

            config = types.GenerateContentConfig(
                system_instruction=(
                    request.system_instruction
                ),
            )

            response = await client.aio.models.generate_content(
                model=self.model,
                contents=contents,
                config=config,
            )

        except Exception as exc:
            logger.exception(
                "Gemini provider execution failed."
            )

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error=str(exc),
            )

        output_text = getattr(
            response,
            "text",
            None,
        )

        if not output_text or not output_text.strip():
            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error="Gemini returned an empty response.",
            )

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output=output_text.strip(),
        )

=== LLM CONFIG ===
"""
LLM Provider Configuration.

Defines approved provider configuration
for the EasyChatBot hackathon runtime.

Configuration is environment-driven so the
application does not hard-code deployment-specific
provider settings.

Local development supports a .env file located
at the EasyChatBot project root.

For the Hackathon runtime, the project .env file
is authoritative. This prevents stale local shell
environment variables from silently overriding the
configured development credentials.

Future production versions should use an external
secret/configuration management system.
"""

from __future__ import annotations

import os
from pathlib import Path

from dotenv import load_dotenv


# Resolve the EasyChatBot project root.
#
# config.py:
# backend/
#   providers/
#     llm/
#       config.py
#
# parents[3] -> EasyChatBot-Hackathon/
PROJECT_ROOT = Path(
    __file__
).resolve().parents[3]

ENV_FILE = PROJECT_ROOT / ".env"


# Load the Hackathon project's local configuration.
#
# override=True is intentional for the local Hackathon
# runtime because .env is the authoritative project
# configuration source.
#
# Production deployments must not rely on this mechanism.
load_dotenv(
    dotenv_path=ENV_FILE,
    override=True,
)


DEFAULT_PROVIDER = "openai"

DEFAULT_MODEL = "gpt-5-mini"

DEFAULT_GEMINI_MODEL = "gemini-3.6-flash"


def get_llm_provider_name() -> str:
    """
    Return the configured LLM provider name.

    Environment variable:

        EASYCHATBOT_LLM_PROVIDER

    Default:

        openai

    Returns
    -------
    str
        Normalised provider identifier.

    Raises
    ------
    ValueError
        If the configured provider name is empty.
    """

    provider = os.getenv(
        "EASYCHATBOT_LLM_PROVIDER",
        DEFAULT_PROVIDER,
    )

    provider = provider.strip().lower()

    if not provider:
        raise ValueError(
            "LLM provider name cannot be empty."
        )

    return provider


def get_llm_model() -> str:
    """
    Return the configured model for the
    default/current LLM path.

    Environment variable:

        EASYCHATBOT_LLM_MODEL

    Default:

        gpt-5-mini

    Returns
    -------
    str
        Configured model identifier.

    Raises
    ------
    ValueError
        If the configured model is empty.
    """

    model = os.getenv(
        "EASYCHATBOT_LLM_MODEL",
        DEFAULT_MODEL,
    )

    model = model.strip()

    if not model:
        raise ValueError(
            "LLM model cannot be empty."
        )

    return model


def get_gemini_model() -> str:
    """
    Return the configured Gemini model.

    Environment variable:

        EASYCHATBOT_GEMINI_MODEL

    Default:

        gemini-3.6-flash

    Returns
    -------
    str
        Configured Gemini model identifier.

    Raises
    ------
    ValueError
        If the configured Gemini model is empty.
    """

    model = os.getenv(
        "EASYCHATBOT_GEMINI_MODEL",
        DEFAULT_GEMINI_MODEL,
    )

    model = model.strip()

    if not model:
        raise ValueError(
            "Gemini model cannot be empty."
        )

    return model

=== PROVIDER RESULT ===
"""
Provider Result Contract.

Defines the standard response returned
by external providers.

All providers return this structure so
the Runtime can process results consistently.
"""

from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class ProviderResult:
    """
    Immutable provider execution result.

    Attributes:

        success:
            Indicates whether provider execution succeeded.

        provider_id:
            Identifier of the provider.

        output:
            Provider response payload.

        error:
            Error details if execution failed.
    """

    success: bool
    provider_id: str
    output: Any = None
    error: str | None = None

=== CONVERSATION REQUEST ===
"""
Provider-independent conversation request.

Represents the information required to generate a
character-aware conversational response.

This contract belongs to the application/runtime boundary.

It deliberately does not know about:

- Gemini
- OpenAI
- Ollama
- Parallel
- voice
- HTTP
- persistence
- provider SDKs

Architecture:

CharacterProfile
        +
ExecutionContext
        +
Optional IntelligenceContext
        |
        v
ConversationRequest
        |
        v
Future / existing EasyChatBot LLM service

The production EasyChatBot conversation and voice
architecture remains outside this hackathon runtime.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class ConversationRequest:
    """
    Immutable provider-independent conversation request.

    Attributes
    ----------
    system_instruction:
        Stable instruction defining how the character
        should behave.

    user_prompt:
        User-facing request to be answered.

    conversation_context:
        Optional contextual information, such as live
        intelligence retrieved by the Research Agent.

    character_id:
        Identifier of the character producing the response.

    character_name:
        Display name of the character.
    """

    system_instruction: str

    user_prompt: str

    conversation_context: str | None = None

    character_id: str = ""

    character_name: str = ""

    def __post_init__(self) -> None:
        """
        Validate request invariants.
        """

        if not isinstance(
            self.system_instruction,
            str,
        ):
            raise ValueError(
                "system_instruction must be a string."
            )

        if not isinstance(
            self.user_prompt,
            str,
        ):
            raise ValueError(
                "user_prompt must be a string."
            )

        system_instruction = (
            self.system_instruction.strip()
        )

        user_prompt = (
            self.user_prompt.strip()
        )

        if not system_instruction:
            raise ValueError(
                "system_instruction cannot be empty."
            )

        if not user_prompt:
            raise ValueError(
                "user_prompt cannot be empty."
            )

        object.__setattr__(
            self,
            "system_instruction",
            system_instruction,
        )

        object.__setattr__(
            self,
            "user_prompt",
            user_prompt,
        )

        if self.conversation_context is not None:
            context = (
                self.conversation_context.strip()
            )

            if not context:
                raise ValueError(
                    "conversation_context cannot be blank."
                )

            object.__setattr__(
                self,
                "conversation_context",
                context,
            )

        if not isinstance(
            self.character_id,
            str,
        ):
            raise ValueError(
                "character_id must be a string."
            )

        if not isinstance(
            self.character_name,
            str,
        ):
            raise ValueError(
                "character_name must be a string."
            )

        object.__setattr__(
            self,
            "character_id",
            self.character_id.strip(),
        )

        object.__setattr__(
            self,
            "character_name",
            self.character_name.strip(),
        )

    @property
    def has_conversation_context(self) -> bool:
        """
        Return whether contextual information exists.
        """

        return bool(
            self.conversation_context
        )

    @property
    def context_length(self) -> int:
        """
        Return the length of the conversation context.

        Useful for diagnostics and future token estimation.
        """

        if not self.conversation_context:
            return 0

        return len(
            self.conversation_context
        )

    def summary(self) -> dict[str, str]:
        """
        Return a safe diagnostic summary.

        Prompt and context content are deliberately
        excluded from the summary.
        """

        return {
            "character_id": self.character_id,
            "character_name": self.character_name,
            "has_context": str(
                self.has_conversation_context
            ),
            "system_instruction_length": str(
                len(self.system_instruction)
            ),
            "user_prompt_length": str(
                len(self.user_prompt)
            ),
            "context_length": str(
                self.context_length
            ),
        }
