def create_telegram_runtime() -> TelegramBotRuntime:
    """
    Create the production Beta Telegram runtime.

    Existing dependency seams remain exposed at this module level
    intentionally. This preserves deterministic test composition
    and allows controlled dependency injection without changing
    production behavior.
    """

    global _telegram_production

    if _telegram_production is None:
        _telegram_production = (
            create_production_beta_telegram_composition(
                database_engine_factory=(
                    create_database_engine
                ),
                session_factory_factory=(
                    create_session_factory
                ),
                conversation_runtime_factory=(
                    create_conversation_runtime
                ),
            )
        )

        logger.info(
            "Production Beta Telegram composition created."
        )

    return _telegram_production.beta_telegram.runtime


async def start_telegram_runtime() -> None:
    """
    Start the production Telegram runtime.
    """

    runtime = create_telegram_runtime()

    await runtime.start()

    logger.info(
        "Production Beta Telegram runtime started."
    )


async def stop_telegram_runtime() -> None:
    """
    Stop the production Telegram runtime.
    """

    if _telegram_production is None:
        return

    await _telegram_production.beta_telegram.runtime.stop()

    logger.info(
        "Production Beta Telegram runtime stopped."
    )


def reset_telegram_runtime() -> None:
    """
    Reset the Telegram production composition.

    Intended primarily for tests and controlled lifecycle
    management.
    """

    global _telegram_production

    _telegram_production = None


def get_telegram_production_composition() -> (
    BetaTelegramProductionComposition
):
    """
    Return the active production Beta Telegram composition.
    """

    if _telegram_production is None:
        raise RuntimeError(
            "Telegram production composition has not been created."
        )

    return _telegram_production

def create_production_beta_telegram_composition(
    *,
    expiration_minutes: int = 1440,
    database_engine: Any | None = None,
    session_factory: Any | None = None,
    database_engine_factory: Callable[[], Any] | None = None,
    session_factory_factory: Callable[[Any], Any] | None = None,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: Callable[[], ConversationRuntime] | None = None,
) -> BetaTelegramProductionComposition:
    if expiration_minutes <= 0:
        raise ValueError("expiration_minutes must be greater than zero.")

    if database_engine is not None and database_engine_factory is not None:
        raise ValueError(
            "Provide either database_engine or database_engine_factory, not both."
        )

    if session_factory is not None and session_factory_factory is not None:
        raise ValueError(
            "Provide either session_factory or session_factory_factory, not both."
        )

    if conversation_runtime is not None and conversation_runtime_factory is not None:
        raise ValueError(
            "Provide either conversation_runtime or conversation_runtime_factory, not both."
        )

    if database_engine is None:
        database_engine = (
            database_engine_factory or create_database_engine
        )()

    if session_factory is None:
        session_factory = (
            session_factory_factory or create_session_factory
        )(database_engine)

    beta_user_repository = PostgreSQLBetaUserRepository(session_factory)
    beta_token_repository = PostgreSQLBetaTokenRepository(session_factory)
    telegram_binding_repository = PostgreSQLTelegramBindingRepository(session_factory)
    conversation_event_repository = PostgreSQLConversationEventRepository(session_factory)

    # Durable provider usage ledger. The repository implements the existing
    # ProviderUsageRepository abstraction; accounting remains outside the
    # Operations Center.
    provider_usage_repository = PostgreSQLProviderUsageRepository(session_factory)

    privacy_identifier_service = PrivacyIdentifierService()
    conversation_event_validator = ConversationEventValidator()

    beta_telegram = create_beta_telegram_composition(
        user_repository=beta_user_repository,
        token_repository=beta_token_repository,
        binding_repository=telegram_binding_repository,
        conversation_event_repository=conversation_event_repository,
        privacy_identifier_service=privacy_identifier_service,
        conversation_event_validator=conversation_event_validator,
        expiration_minutes=expiration_minutes,
        conversation_runtime=conversation_runtime,
        conversation_runtime_factory=conversation_runtime_factory,
        provider_usage_repository=provider_usage_repository,
    )

    logger.info("Production Beta Telegram composition created with durable usage ledger.")

    return BetaTelegramProductionComposition(
        beta_telegram=beta_telegram,
        database_engine=database_engine,
        session_factory=session_factory,
    )

def _env_positive_int(name: str, default: int) -> int:
    raw = os.getenv(name)
    if raw is None:
        return default
    try:
        value = int(raw)
    except ValueError as exc:
        raise ValueError(f"{name} must be a positive integer.") from exc
    if value <= 0:
        raise ValueError(f"{name} must be a positive integer.")
    return value


def _env_positive_float(name: str, default: float) -> float:
    raw = os.getenv(name)
    if raw is None:
        return default
    try:
        value = float(raw)
    except ValueError as exc:
        raise ValueError(f"{name} must be a positive number.") from exc
    if value <= 0:
        raise ValueError(f"{name} must be a positive number.")
    return value


def create_beta_telegram_composition(
    *,
    user_repository: Any,
    token_repository: Any,
    binding_repository: Any,
    conversation_event_repository: Any,
    privacy_identifier_service: Any,
    conversation_event_validator: Any,
    expiration_minutes: int = 1440,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: Callable[[], ConversationRuntime] | None = None,
    rate_limiter: Any | None = None,
    abuse_protector: Any | None = None,
    observability: Any | None = None,
    provider_usage_capture: ProviderUsageCapture | None = None,
    provider_usage_repository: ProviderUsageRepository | None = None,
) -> BetaTelegramComposition:
    if expiration_minutes <= 0:
        raise ValueError("expiration_minutes must be greater than zero.")

    if conversation_runtime is not None and conversation_runtime_factory is not None:
        raise ValueError(
            "Provide either conversation_runtime or conversation_runtime_factory, not both."
        )

    beta_access_service = BetaAccessService(
        user_repository=user_repository,
        token_repository=token_repository,
        binding_repository=binding_repository,
        expiration_minutes=expiration_minutes,
    )

    resolved_usage_repository = provider_usage_repository
    if conversation_runtime is not None:
        resolved_conversation_runtime = conversation_runtime
    elif conversation_runtime_factory is not None:
        resolved_conversation_runtime = conversation_runtime_factory()
    else:
        if provider_usage_capture is None:
            if resolved_usage_repository is None:
                resolved_usage_repository = InMemoryProviderUsageRepository()
            provider_usage_service = ProviderUsageService(resolved_usage_repository)
            provider_usage_capture = ProviderUsageCapture(provider_usage_service)

        resolved_conversation_runtime = ConversationRuntime(
            llm_provider=LLMProviderFactory.create(),
            usage_capture=provider_usage_capture,
            usage_context=ProviderUsageContext(
                capability="conversation",
                functionality="chat",
                agent="conversation_agent",
            ),
        )

    conversation_service = TelegramConversationService(
        conversation_runtime=resolved_conversation_runtime,
        conversation_event_repository=conversation_event_repository,
        privacy_identifier_service=privacy_identifier_service,
        conversation_event_validator=conversation_event_validator,
    )

    resolved_rate_limiter = rate_limiter or InMemoryTelegramRateLimiter(
        max_messages=_env_positive_int("TELEGRAM_RATE_LIMIT_MAX_MESSAGES", 100),
        window_seconds=_env_positive_float(
            "TELEGRAM_RATE_LIMIT_WINDOW_SECONDS", 86400.0
        ),
    )
    resolved_abuse_protector = abuse_protector or InMemoryTelegramAbuseProtector(
        max_messages=_env_positive_int("TELEGRAM_ABUSE_MAX_MESSAGES", 10),
        window_seconds=_env_positive_float(
            "TELEGRAM_ABUSE_WINDOW_SECONDS", 60.0
        ),
    )
    resolved_observability = observability or LoggingTelegramObservability(
        privacy_identifier_service=privacy_identifier_service,
    )

    conversation_boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation_service,
        beta_access_service=beta_access_service,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,
        observability=resolved_observability,
    )
    private_chat_gate = BetaPrivateChatGate(beta_access_service=beta_access_service)
    start_handler = BetaTelegramStartHandler(beta_access_service=beta_access_service)
    adapter = BetaTelegramAdapter(
        start_handler=start_handler,
        private_chat_gate=private_chat_gate,
        conversation_boundary=conversation_boundary,
    )
    runtime = TelegramBotRuntime(
        adapter=adapter,
        conversation_boundary=conversation_boundary,
    )

    return BetaTelegramComposition(
        beta_access_service=beta_access_service,
        conversation_runtime=resolved_conversation_runtime,
        conversation_service=conversation_service,
        conversation_boundary=conversation_boundary,
        private_chat_gate=private_chat_gate,
        start_handler=start_handler,
        adapter=adapter,
        runtime=runtime,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,
        observability=resolved_observability,
        provider_usage_repository=resolved_usage_repository,
    )

CANONICAL PROVIDER RESULT

"""Canonical provider execution result for the live application provider path."""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any


@dataclass(frozen=True, slots=True)
class ProviderUsage:
    """Provider-neutral execution usage metadata.

    Values are optional because not every provider exposes every metric.
    Providers must never invent token counts or request IDs.
    """

    input_tokens: int | None = None
    output_tokens: int | None = None
    total_tokens: int | None = None

    def __post_init__(self) -> None:
        for name, value in (
            ("input_tokens", self.input_tokens),
            ("output_tokens", self.output_tokens),
            ("total_tokens", self.total_tokens),
        ):
            if value is not None and value < 0:
                raise ValueError(f"{name} cannot be negative.")

        if (
            self.total_tokens is not None
            and self.input_tokens is not None
            and self.output_tokens is not None
            and self.total_tokens < self.input_tokens + self.output_tokens
        ):
            raise ValueError(
                "total_tokens cannot be less than input_tokens + output_tokens."
            )


@dataclass(frozen=True, slots=True)
class ProviderResult:
    """Provider-independent result returned by every live LLM adapter.

    Provider SDK objects must never cross this boundary.

    ``content`` is the canonical generated text. ``output`` is retained as
    a compatibility property for consumers that already use that name.
    """

    success: bool
    provider_id: str
    model: str
    content: str = ""
    usage: ProviderUsage = field(default_factory=ProviderUsage)
    provider_request_id: str | None = None
    latency_ms: float | None = None
    completed_at: datetime = field(
        default_factory=lambda: datetime.now(timezone.utc)
    )
    error: str | None = None
    metadata: dict[str, Any] = field(default_factory=dict)

    def __post_init__(self) -> None:
        if not self.provider_id.strip():
            raise ValueError("ProviderResult requires provider_id.")
        if not self.model.strip():
            raise ValueError("ProviderResult requires model.")
        if self.latency_ms is not None and self.latency_ms < 0:
            raise ValueError("latency_ms cannot be negative.")
        if self.success and not self.content.strip():
            raise ValueError(
                "Successful ProviderResult requires non-empty content."
            )
        if not self.success and not self.error:
            raise ValueError(
                "Failed ProviderResult requires an error."
            )

    @property
    def output(self) -> str:
        """Backward-compatible alias for canonical ``content``."""
        return self.content

"""OpenAI adapter for the live application provider path."""

from __future__ import annotations

import os
import time

from openai import AsyncOpenAI

from backend.app.providers.llm.llm_provider import LLMProvider
from backend.app.providers.provider_result import ProviderResult, ProviderUsage


class OpenAILLMProvider(LLMProvider):
    """Infrastructure adapter that normalizes OpenAI responses."""

    def __init__(
        self,
        *,
        model: str | None = None,
        client: AsyncOpenAI | None = None,
    ) -> None:
        self.model = (
            model
            or os.getenv("OPENAI_MODEL")
            or "gpt-4o-mini"
        ).strip()

        if not self.model:
            raise ValueError("OpenAI model cannot be empty.")

        if client is not None:
            self.client = client
            return

        api_key = os.getenv("OPENAI_API_KEY")
        if not api_key:
            raise RuntimeError("OPENAI_API_KEY missing")

        self.client = AsyncOpenAI(api_key=api_key)

    async def generate(
        self,
        *,
        system_prompt: str,
        user_message: str,
    ) -> ProviderResult:
        started = time.perf_counter()

        try:
            response = await self.client.chat.completions.create(
                model=self.model,
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_message},
                ],
            )
        except Exception as exc:
            return ProviderResult(
                success=False,
                provider_id="openai",
                model=self.model,
                latency_ms=(time.perf_counter() - started) * 1000,
                error=str(exc),
            )

        content = (
            response.choices[0].message.content
            if response.choices
            else None
        ) or ""

        if not content.strip():
            return ProviderResult(
                success=False,
                provider_id="openai",
                model=self.model,
                provider_request_id=getattr(response, "id", None),
                latency_ms=(time.perf_counter() - started) * 1000,
                error="OpenAI returned an empty response.",
            )

        native_usage = getattr(response, "usage", None)

        usage = ProviderUsage(
            input_tokens=getattr(native_usage, "prompt_tokens", None),
            output_tokens=getattr(native_usage, "completion_tokens", None),
            total_tokens=getattr(native_usage, "total_tokens", None),
        )

        return ProviderResult(
            success=True,
            provider_id="openai",
            model=self.model,
            content=content,
            usage=usage,
            provider_request_id=getattr(response, "id", None),
            latency_ms=(time.perf_counter() - started) * 1000,
            metadata={
                "finish_reason": (
                    getattr(response.choices[0], "finish_reason", None)
                    if response.choices
                    else None
                )
            },
        )

"""Ollama adapter for the live application provider path."""

from __future__ import annotations

import logging
import os
import time
from typing import Any

import requests

from backend.app.providers.llm.llm_provider import LLMProvider
from backend.app.providers.provider_result import ProviderResult, ProviderUsage


logger = logging.getLogger(__name__)


class OllamaLLMProvider(LLMProvider):
    """Infrastructure adapter that normalizes Ollama responses."""

    def __init__(
        self,
        *,
        base_url: str | None = None,
        model: str | None = None,
    ) -> None:
        self.base_url = (
            base_url
            or os.getenv("OLLAMA_BASE_URL", "http://localhost:11434")
        ).rstrip("/")
        self.model = (
            model
            or os.getenv("OLLAMA_MODEL", "gpt-oss:120b-cloud")
        ).strip()

        if not self.model:
            raise ValueError("Ollama model cannot be empty.")

    async def generate(
        self,
        *,
        system_prompt: str,
        user_message: str,
        **kwargs: Any,
    ) -> ProviderResult:
        if not user_message:
            raise ValueError("User message cannot be empty.")

        started = time.perf_counter()

        payload = {
            "model": self.model,
            "prompt": (
                f"System:\n{system_prompt}\n\n"
                f"User:\n{user_message}\n\n"
                "Assistant:\n"
            ),
            "stream": False,
        }

        try:
            response = requests.post(
                f"{self.base_url}/api/generate",
                json=payload,
                timeout=300,
            )
            response.raise_for_status()
            data = response.json()
        except requests.RequestException as exc:
            logger.exception("Ollama request failed.")
            return ProviderResult(
                success=False,
                provider_id="ollama",
                model=self.model,
                latency_ms=(time.perf_counter() - started) * 1000,
                error="Ollama provider unavailable.",
            )
        except ValueError as exc:
            return ProviderResult(
                success=False,
                provider_id="ollama",
                model=self.model,
                latency_ms=(time.perf_counter() - started) * 1000,
                error=f"Invalid Ollama response: {exc}",
            )

        result = str(data.get("response") or "").strip()

        if not result:
            return ProviderResult(
                success=False,
                provider_id="ollama",
                model=self.model,
                latency_ms=(time.perf_counter() - started) * 1000,
                error="Ollama returned empty response.",
            )

        # Ollama exposes these counters for /api/generate when available.
        usage = ProviderUsage(
            input_tokens=_non_negative_int(data.get("prompt_eval_count")),
            output_tokens=_non_negative_int(data.get("eval_count")),
            total_tokens=_sum_if_available(
                data.get("prompt_eval_count"),
                data.get("eval_count"),
            ),
        )

        return ProviderResult(
            success=True,
            provider_id="ollama",
            model=self.model,
            content=result,
            usage=usage,
            provider_request_id=None,
            latency_ms=(time.perf_counter() - started) * 1000,
            metadata={
                "done": data.get("done"),
                "load_duration_ns": data.get("load_duration"),
                "prompt_eval_duration_ns": data.get("prompt_eval_duration"),
                "eval_duration_ns": data.get("eval_duration"),
            },
        )


def _non_negative_int(value: Any) -> int | None:
    if isinstance(value, bool):
        return None
    if isinstance(value, int) and value >= 0:
        return value
    return None


def _sum_if_available(left: Any, right: Any) -> int | None:
    left_value = _non_negative_int(left)
    right_value = _non_negative_int(right)
    if left_value is None or right_value is None:
        return None
    return left_value + right_value

from __future__ import annotations

from dataclasses import dataclass

from backend.app.application.providers.provider_usage_capture import (
    ProviderUsageCapture,
    ProviderUsageContext,
)
from backend.app.providers.llm.llm_provider import LLMProvider
from backend.app.providers.provider_result import ProviderResult


@dataclass(frozen=True, slots=True)
class ConversationRuntime:
    """Execute conversation requests and optionally capture provider usage."""

    llm_provider: LLMProvider
    usage_capture: ProviderUsageCapture | None = None
    usage_context: ProviderUsageContext | None = None

    async def respond(
        self,
        *,
        character_instruction: str,
        learner_message: str,
    ) -> str:
        result = await self.llm_provider.generate(
            system_prompt=character_instruction,
            user_message=learner_message,
        )

        if isinstance(result, ProviderResult):
            if self.usage_capture is not None:
                if self.usage_context is None:
                    raise RuntimeError(
                        "Provider usage capture requires usage context."
                    )
                self.usage_capture.capture(
                    result=result,
                    context=self.usage_context,
                )

            if not result.success:
                raise RuntimeError(result.error or "Provider request failed.")
            return result.content
