"""
EasyChatBot API Routes.

The API layer exposes the application.

Routes do not contain agent logic.

The API composition layer is responsible for wiring
infrastructure dependencies into application services.

Architecture:

API
|
v
CompanionService
|
v
ConversationAgent
|
v
LLMService
|
v
LLMProviderFactory
|
v
Concrete Provider
"""

from __future__ import annotations

from fastapi import APIRouter
from fastapi import HTTPException

from backend.agents.conversation_agent import (
    ConversationAgent,
)

from backend.api.schemas import (
    ChatRequest,
    ChatResponse,
)

from backend.characters.character_registry import (
    CharacterRegistry,
)

from backend.providers.llm.llm_provider_factory import (
    LLMProviderFactory,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)

from backend.services.companion_service import (
    CompanionService,
)

from backend.services.llm_service import (
    LLMService,
)


router = APIRouter()

character_registry = CharacterRegistry()


def create_llm_service() -> LLMService:
    """
    Create the application LLM service.

    Provider construction is isolated through the
    LLMProviderFactory.

    The LLMService remains provider-independent.
    """

    llm_provider = (
        LLMProviderFactory.create()
    )

    return LLMService(
        llm_provider
    )


def create_companion_service(
    character,
) -> CompanionService:
    """
    Create the application-level CompanionService.

    This function is the API composition boundary.

    Infrastructure dependencies are created here and
    injected into application services and agents.

    The application service remains unaware of:

    - OpenAI
    - Ollama
    - provider SDKs
    - API credentials
    """

    llm_service = create_llm_service()

    conversation_agent = ConversationAgent(
        character=character,
        llm_service=llm_service,
    )

    return CompanionService(
        character=character,
        conversation_agent=conversation_agent,
    )


@router.get("/characters")
def get_characters():
    """
    Return available AI companions.
    """

    characters = (
        character_registry.list_characters()
    )

    return {
        "characters": [
            {
                "id": character.character_id,
                "name": character.name,
                "description": character.description,
                "personality": character.personality,
                "learning_style": character.learning_style,
            }
            for character in characters
        ]
    }


@router.post(
    "/chat",
    response_model=ChatResponse,
)
async def chat(
    payload: ChatRequest,
) -> ChatResponse:
    """
    Execute an EasyChatBot companion interaction.

    Expected payload:

    {
        "user_id": "demo-user",
        "character": "yeti",
        "message": "Explain AI",
        "research": false
    }

    Request schema validation is handled by ChatRequest.

    The explicit message validation below preserves
    the existing EasyChatBot API contract:

        missing message -> HTTP 400

    Application orchestration remains delegated to
    CompanionService.
    """

    # ---------------------------------------------------------
    # API boundary validation
    #
    # ChatRequest intentionally allows message=None so that
    # the existing API contract can return HTTP 400 rather
    # than FastAPI's automatic HTTP 422 response.
    #
    # Invalid transport data must be rejected before it
    # reaches application/domain services.
    # ---------------------------------------------------------

    if payload.message is None:
        raise HTTPException(
            status_code=400,
            detail="message is required",
        )

    # Defensive empty-string validation.
    #
    # This preserves the previous behaviour for:
    #
    #     "message": ""
    #
    # and prevents invalid input from reaching the
    # MemoryContextBuilder.
    if not payload.message.strip():
        raise HTTPException(
            status_code=400,
            detail="message is required",
        )

    # ---------------------------------------------------------
    # Character validation
    # ---------------------------------------------------------

    try:
        character = (
            character_registry.get(
                payload.character
            )
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    # ---------------------------------------------------------
    # Application composition
    # ---------------------------------------------------------

    companion_service = (
        create_companion_service(
            character
        )
    )

    # ---------------------------------------------------------
    # Runtime context
    # ---------------------------------------------------------

    context = ExecutionContext(
        user_id=payload.user_id,
        request=payload.message,
    )

    # ---------------------------------------------------------
    # Application orchestration
    # ---------------------------------------------------------

    result = await companion_service.interact(
        context,
        use_research=payload.research,
    )

    # ---------------------------------------------------------
    # API response contract
    # ---------------------------------------------------------

    return ChatResponse(
        character=result["character"],
        conversation=result["conversation"],
        research=result["research"],
        quality=result["quality"],
        memory_count=result["memory_count"],
        memory_context_count=result.get(
            "memory_context_count",
            0,
        ),
    )
