from __future__ import annotations

from datetime import datetime, timezone
from uuid import UUID

from fastapi import APIRouter, HTTPException, Request, Response, status
from pydantic import BaseModel, EmailStr, Field

from backend.app.application.identity.authentication_service import (
    AuthenticationError,
    AuthenticationService,
)

# NOTE:
# The authentication service/repository must be composed into app.state
# by the application's existing bootstrap/composition root.


router = APIRouter(
    prefix="/auth",
    tags=["authentication"],
)


class RegisterRequest(BaseModel):
    email: EmailStr
    username: str = Field(
        min_length=2,
        max_length=100,
    )
    password: str = Field(
        min_length=12,
        max_length=256,
    )


class RegisterResponse(BaseModel):
    user_id: UUID
    detail: str


class LoginRequest(BaseModel):
    email: EmailStr
    password: str = Field(
        min_length=1,
        max_length=256,
    )


class LoginResponse(BaseModel):
    detail: str
    expires_at: str


@router.post(
    "/register",
    response_model=RegisterResponse,
    status_code=status.HTTP_201_CREATED,
)
async def register(
    request: RegisterRequest,
    http_request: Request,
) -> RegisterResponse:
    service = AuthenticationService(
        http_request.app.state.auth_repository
    )

    try:
        user_id = await service.register(
            email=str(request.email),
            username=request.username,
            password=request.password,
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(exc),
        ) from None

    return RegisterResponse(
        user_id=user_id,
        detail=(
            "Registration successful. "
            "Email verification is required."
        ),
    )


@router.post(
    "/login",
    response_model=LoginResponse,
)
async def login(
    request: LoginRequest,
    http_request: Request,
    response: Response,
) -> LoginResponse:
    service = AuthenticationService(
        http_request.app.state.auth_repository
    )

    try:
        result = await service.authenticate(
            email=str(request.email),
            password=request.password,
            device_name=http_request.headers.get("user-agent"),
            ip_address=(
                http_request.client.host
                if http_request.client
                else None
            ),
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=str(exc),
            headers={
                "WWW-Authenticate": "Session",
            },
        ) from None

    expires_in_seconds = max(
        0,
        int(
            (
                result.expires_at
                - datetime.now(timezone.utc)
            ).total_seconds()
        ),
    )

    response.set_cookie(
        key="easychatbot_session",
        value=str(result.session_id),
        httponly=True,
        secure=True,
        samesite="lax",
        max_age=expires_in_seconds,
        path="/",
    )

    return LoginResponse(
        detail="Login successful.",
        expires_at=result.expires_at.isoformat(),
    )


@router.post("/verify-email")
async def verify_email(
    token: str,
    http_request: Request,
) -> dict[str, str]:
    """Verify an application email address.

    Token validation and single-use enforcement remain inside the
    authentication repository.
    """

    service = AuthenticationService(
        http_request.app.state.auth_repository
    )

    try:
        await service.verify_email(token)
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(exc),
        ) from None

    return {
        "detail": "Email verification successful.",
    }


@router.post(
    "/logout",
    status_code=status.HTTP_204_NO_CONTENT,
)
async def logout(
    http_request: Request,
    response: Response,
) -> None:
    session_id = http_request.cookies.get(
        "easychatbot_session"
    )

    if session_id:
        service = AuthenticationService(
            http_request.app.state.auth_repository
        )
        await service.revoke_session(
            UUID(session_id)
        )

    response.delete_cookie(
        key="easychatbot_session",
        path="/",
    )


__all__ = [
    "router",
    "RegisterRequest",
    "RegisterResponse",
    "LoginRequest",
    "LoginResponse",
]

from __future__ import annotations

from collections.abc import Iterable
from uuid import uuid4

from .operational_event import (
    OperationalEvent,
)
from .operational_incident import (
    IncidentStatus,
    OperationalIncident,
)


class OperationalEventService:
    """
    Application service for operational event correlation
    and incident lifecycle management.

    This service intentionally uses in-memory state for WS-21.9.

    Persistence will be introduced separately once the event
    semantics are proven.

    Responsibilities:
        - accept sanitized operational events
        - correlate repeated events
        - create incidents
        - update occurrence counts
        - maintain incident lifecycle
        - expose dashboard-safe snapshots

    Does NOT:
        - perform remediation
        - send Telegram messages
        - modify production systems
        - execute AI
        - access learner conversation content
    """

    def __init__(
        self,
        *,
        initial_incidents: Iterable[
            OperationalIncident
        ] = (),
    ) -> None:
        self._incidents: dict[
            str,
            OperationalIncident,
        ] = {
            incident.incident_id: incident
            for incident in initial_incidents
        }

        self._correlation_index: dict[
            str,
            str,
        ] = {}

        for incident in self._incidents.values():
            self._correlation_index[
                incident.correlation_id
            ] = incident.incident_id

    def record_event(
        self,
        event: OperationalEvent,
    ) -> OperationalIncident:
        """
        Record an event.

        If an open incident already exists for the
        correlation ID, the event is aggregated into it.

        Otherwise a new incident is created.
        """

        existing_id = (
            self._correlation_index.get(
                event.correlation_id
            )
        )

        if existing_id is not None:
            incident = self._incidents[
                existing_id
            ]

            if not incident.is_open:
                incident = self._create_incident(
                    event
                )
            else:
                incident.add_event(event)

            return incident

        return self._create_incident(event)

    def get_incident(
        self,
        incident_id: str,
    ) -> OperationalIncident | None:
        return self._incidents.get(
            incident_id
        )

    def get_by_correlation_id(
        self,
        correlation_id: str,
    ) -> OperationalIncident | None:
        incident_id = (
            self._correlation_index.get(
                correlation_id
            )
        )

        if incident_id is None:
            return None

        return self._incidents.get(
            incident_id
        )

    def list_incidents(
        self,
        *,
        surface=None,
        status=None,
    ) -> tuple[
        OperationalIncident,
        ...
    ]:
        """
        Return incidents with optional operational filters.
        """

        incidents = tuple(
            self._incidents.values()
        )

        if surface is not None:
            incidents = tuple(
                incident
                for incident in incidents
                if incident.surface is surface
            )

        if status is not None:
            incidents = tuple(
                incident
                for incident in incidents
                if incident.status is status
            )

        return incidents

    def transition_incident(
        self,
        *,
        incident_id: str,
        status: IncidentStatus,
    ) -> OperationalIncident:
        """
        Explicitly transition an incident.
        """

        incident = self._incidents.get(
            incident_id
        )

        if incident is None:
            raise KeyError(
                f"Unknown incident: {incident_id}"
            )

        incident.transition(status)

        return incident

    def count_open_incidents(self) -> int:
        return sum(
            incident.is_open
            for incident
            in self._incidents.values()
        )

    def dashboard_snapshot(self) -> dict:
        """
        Return a dashboard-safe operational snapshot.
        """

        incidents = tuple(
            self._incidents.values()
        )

        return {
            "total_incidents": len(
                incidents
            ),
            "open_incidents": sum(
                incident.is_open
                for incident in incidents
            ),
            "critical_incidents": sum(
                incident.severity.value
                == "critical"
                and incident.is_open
                for incident in incidents
            ),
            "private_classroom_incidents": sum(
                incident.surface.value
                == "private_classroom"
                and incident.is_open
                for incident in incidents
            ),
            "announcement_group_incidents": sum(
                incident.surface.value
                == "announcement_group"
                and incident.is_open
                for incident in incidents
            ),
            "incidents": [
                incident.as_dict()
                for incident in incidents
            ],
        }

    def _create_incident(
        self,
        event: OperationalEvent,
    ) -> OperationalIncident:
        incident = OperationalIncident(
            incident_id=(
                f"incident-{uuid4().hex}"
            ),
            correlation_id=(
                event.correlation_id
            ),
            event_type=event.event_type,
            surface=event.surface,
            severity=event.severity,
            user_reference=(
                event.user_reference
            ),
            world_id=event.world_id,
            events=[event],
        )

        self._incidents[
            incident.incident_id
        ] = incident

        self._correlation_index[
            incident.correlation_id
        ] = incident.incident_id

        return incident


__all__ = [
    "OperationalEventService",
]

===== EXISTING CONTROL TESTS =====
import pytest

from backend.app.application.provider_management.functionality_operational_control_service import (
    FunctionalityOperationalControlService,
)


class Authorizer:
    def __init__(self, allowed=True):
        self.allowed = allowed

    def can_control_functionality(self, *, actor, environment, functionality_id):
        return self.allowed


class Auditor:
    def __init__(self):
        self.events = []

    def record_functionality_control(self, **event):
        self.events.append(event)


def make_service(allowed=True):
    auditor = Auditor()
    return FunctionalityOperationalControlService(
        authorizer=Authorizer(allowed), auditor=auditor
    ), auditor


def test_stop_requires_reason():
    service, _ = make_service()
    with pytest.raises(ValueError, match="reason is required"):
        service.stop(actor="operator", actor_name="operator",
                     environment="adults", functionality_id="conversation",
                     reason="   ")


def test_stop_disables_and_audits():
    service, auditor = make_service()
    result = service.stop(actor="operator", actor_name="operator",
                          environment="kids", functionality_id="conversation",
                          reason="Provider incident")
    assert result.availability == "disabled"
    assert service.get_availability(
        environment="kids", functionality_id="conversation"
    ) == "disabled"
    assert len(auditor.events) == 1
    assert auditor.events[0]["action"] == "stop"
    assert auditor.events[0]["reason"] == "Provider incident"


def test_resume_enables_and_audits():
    service, auditor = make_service()
    service.stop(actor="operator", actor_name="operator",
                 environment="adults", functionality_id="conversation",
                 reason="Temporary issue")
    result = service.resume(actor="operator", actor_name="operator",
                            environment="adults", functionality_id="conversation")
    assert result.availability == "enabled"
    assert service.get_availability(
        environment="adults", functionality_id="conversation"
    ) == "enabled"
    assert [e["action"] for e in auditor.events] == ["stop", "resume"]


def test_authorisation_is_enforced():
    service, _ = make_service(False)
    with pytest.raises(PermissionError):
        service.stop(actor="operator", actor_name="operator",
                     environment="adults", functionality_id="conversation",
                     reason="Temporary issue")


@pytest.mark.parametrize("environment", ["invalid", "", "KIDS"])
def test_environment_is_strict(environment):
    service, _ = make_service()
    with pytest.raises(ValueError):
        service.resume(actor="operator", actor_name="operator",
                       environment=environment, functionality_id="conversation")
