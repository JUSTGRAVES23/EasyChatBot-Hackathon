"""
Agent Manifest Definition.

The AgentManifest describes the identity and basic metadata
of an EasyChatBot AI Agent.

The manifest does not execute the agent.
It only defines:
- who the agent is
- what the agent represents
- its version information

Execution is handled by the AI Runtime.
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class AgentManifest:
    """
    Immutable definition of an AI Agent.

    Agents are identified through their manifest.
    The Runtime uses this information for controlled execution.

    Attributes:
        agent_id:
            Unique identifier of the agent.

        name:
            Human-readable agent name.

        description:
            Purpose and responsibility of the agent.

        version:
            Version of the agent definition.
    """

    agent_id: str
    name: str
    description: str
    version: str = "1.0.0"

"""
Agent Result Contract.

Defines the standard output returned by AI Agents.

All agents should return the same structure
so the Runtime can process results consistently.
"""

from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class AgentResult:
    """
    Immutable result returned after agent execution.

    Attributes:
        success:
            Indicates whether execution completed successfully.

        agent_id:
            Identifier of the agent that produced the result.

        output:
            Agent response payload.

        error:
            Error message when execution fails.
    """

    success: bool
    agent_id: str
    output: Any = None
    error: str | None = None

"""
Base AI Agent Contract.

All EasyChatBot AI Agents inherit this contract.

The BaseAgent defines:
- agent identity
- execution interface
- runtime interaction

Agents do not:
- call providers directly
- manage memory
- modify external systems

The AI Runtime controls execution.
"""

from abc import ABC, abstractmethod

from backend.runtime.context.execution_context import ExecutionContext
from backend.agents.agent_manifest import AgentManifest
from backend.agents.agent_result import AgentResult


class BaseAgent(ABC):
    """
    Abstract base class for EasyChatBot agents.
    """

    def __init__(
        self,
        manifest: AgentManifest,
    ):
        """
        Initialise an agent.

        Args:
            manifest:
                Agent identity definition.
        """

        self.manifest = manifest

    @abstractmethod
    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute the agent.

        Args:
            context:
                Runtime execution context.

        Returns:
            AgentResult.
        """

        pass

"""
Conversation Agent.

Provides character-aware conversational responses
for the EasyChatBot hackathon runtime.

The agent remains provider-independent.

Architecture:

ExecutionContext
|
v
ConversationAgent
|
+--> CharacterProfile
|
+--> Optional IntelligenceContext
|
v
ConversationRequest
|
v
AgentResult

The actual LLM execution remains outside this
hackathon agent and can later consume ConversationRequest
through the appropriate EasyChatBot LLM service.

The ConversationRequest remains part of the internal
agent result contract. Public API responses must decide
separately which fields are exposed externally.
"""

from __future__ import annotations

from backend.agents.agent_manifest import (
    AgentManifest,
)

from backend.agents.agent_result import (
    AgentResult,
)

from backend.agents.base_agent import (
    BaseAgent,
)

from backend.characters.character_profile import (
    CharacterProfile,
)

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class ConversationAgent(BaseAgent):
    """
    Character-aware conversation agent.

    Responsibilities:

    - apply character identity
    - construct a provider-independent conversation request
    - include optional live intelligence
    - return a standard AgentResult

    Does not:

    - call an LLM provider
    - call Parallel directly
    - manage voice
    - manage memory
    - execute external workflows
    """

    def __init__(
        self,
        character: CharacterProfile,
    ) -> None:

        super().__init__(
            AgentManifest(
                agent_id="conversation_agent",
                name="Conversation Agent",
                description=(
                    "Provides character-aware "
                    "conversation responses."
                ),
            )
        )

        self.character = character

    def build_request(
        self,
        context: ExecutionContext,
    ) -> ConversationRequest:
        """
        Build a provider-independent conversation request.

        This method translates runtime state and character
        identity into the application-level conversation
        contract.

        No provider-specific terminology is introduced.
        """

        system_instruction = (
            f"You are {self.character.name}, "
            f"a character in EasyChatBot. "
            f"Your personality is: "
            f"{self.character.personality} "
            f"Your learning style is: "
            f"{self.character.learning_style} "
            f"Communicate in a way that is consistent "
            f"with this character."
        )

        conversation_context = None

        if context.has_intelligence_context:
            conversation_context = (
                context.intelligence_context.to_context_text()
            )

        return ConversationRequest(
            system_instruction=system_instruction,
            user_prompt=context.request,
            conversation_context=conversation_context,
            character_id=self.character.character_id,
            character_name=self.character.name,
        )

    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute the conversation agent.

        The current hackathon implementation prepares a
        provider-independent ConversationRequest.

        Actual LLM generation remains outside this agent.

        The ConversationRequest is retained in the
        AgentResult as an internal runtime contract.
        Public API serialization should not expose this
        internal request directly.
        """

        conversation_request = self.build_request(
            context
        )

        output = {
            "character": self.character.name,

            "character_profile": {
                "character_id": (
                    self.character.character_id
                ),
                "name": self.character.name,
                "description": (
                    self.character.description
                ),
                "personality": (
                    self.character.personality
                ),
                "learning_style": (
                    self.character.learning_style
                ),
            },

            "personality": (
                self.character.personality
            ),

            "learning_style": (
                self.character.learning_style
            ),

            "message": (
                f"{self.character.name}: "
                f"Let's explore this together. "
                f"You asked: {context.request}"
            ),

            "intelligence_context": (
                conversation_request.conversation_context
            ),

            "conversation_request": (
                conversation_request
            ),
        }

        return AgentResult(
            success=True,
            agent_id=self.manifest.agent_id,
            output=output,
        )

"""
Research Agent.

Retrieves external information through approved
knowledge providers and converts the result into
provider-independent IntelligenceContext data.

Architecture:

Research Agent
        |
        v
ParallelSearchProvider
        |
        v
ParallelClientWrapper
        |
        v
IntelligenceContext
"""

from __future__ import annotations

from backend.agents.agent_manifest import AgentManifest
from backend.agents.agent_result import AgentResult
from backend.agents.base_agent import BaseAgent

from backend.intelligence.intelligence_context import (
    IntelligenceContext,
    IntelligenceSource,
)

from backend.providers.parallel.parallel_provider import (
    ParallelSearchProvider,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class ResearchAgent(BaseAgent):
    """
    Agent responsible for world information retrieval.
    """

    def __init__(
        self,
        provider=None,
    ):
        super().__init__(
            AgentManifest(
                agent_id="research_agent",
                name="Research Agent",
                description=(
                    "Retrieves live world information "
                    "for AI learning experiences."
                ),
            )
        )

        self.provider = (
            provider
            or ParallelSearchProvider()
        )

    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute research and produce a normalized
        IntelligenceContext.
        """

        result = await self.provider.execute(
            context
        )

        if not result.success:
            return AgentResult(
                success=False,
                agent_id=self.manifest.agent_id,
                output=None,
                error=result.error,
            )

        intelligence = (
            self._build_intelligence_context(
                query=context.request,
                provider_output=result.output,
            )
        )

        return AgentResult(
            success=True,
            agent_id=self.manifest.agent_id,
            output=intelligence,
        )

    @staticmethod
    def _build_intelligence_context(
        query: str,
        provider_output,
    ) -> IntelligenceContext:
        """
        Convert provider output into the
        provider-independent IntelligenceContext.
        """

        if not isinstance(
            provider_output,
            dict,
        ):
            raise TypeError(
                "Research provider output must be a dictionary."
            )

        raw_results = provider_output.get(
            "results",
            {},
        )

        if isinstance(
            raw_results,
            dict,
        ):
            raw_results = raw_results.get(
                "results",
                [],
            )

        if not isinstance(
            raw_results,
            list,
        ):
            raw_results = []

        sources: list[IntelligenceSource] = []

        for raw_result in raw_results:

            if not isinstance(
                raw_result,
                dict,
            ):
                continue

            title = str(
                raw_result.get(
                    "title",
                    "",
                )
            ).strip()

            url = str(
                raw_result.get(
                    "url",
                    "",
                )
            ).strip()

            if not title or not url:
                continue

            content = ResearchAgent._extract_content(
                raw_result
            )

            sources.append(
                IntelligenceSource(
                    title=title,
                    url=url,
                    source="parallel",
                    content=content,
                )
            )

        return IntelligenceContext(
            query=query,
            sources=tuple(sources),
        )

    @staticmethod
    def _extract_content(
        raw_result: dict,
    ) -> str:
        """
        Extract available content from a normalized
        Parallel result.
        """

        excerpts = raw_result.get(
            "excerpts",
        )

        if excerpts is None:
            excerpts = raw_result.get(
                "snippet",
                raw_result.get(
                    "content",
                    "",
                ),
            )

        if isinstance(
            excerpts,
            list,
        ):
            return "\n\n".join(
                str(item).strip()
                for item in excerpts
                if str(item).strip()
            )

        return str(
            excerpts or ""
        ).strip()

"""
Provider-independent conversation request.

Represents the information required to generate a
character-aware conversational response.

This contract belongs to the application/runtime boundary.

It deliberately does not know about:

- Gemini
- OpenAI
- Ollama
- Parallel
- voice
- HTTP
- persistence
- provider SDKs

Architecture:

CharacterProfile
        +
ExecutionContext
        +
Optional IntelligenceContext
        |
        v
ConversationRequest
        |
        v
Future / existing EasyChatBot LLM service

The production EasyChatBot conversation and voice
architecture remains outside this hackathon runtime.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class ConversationRequest:
    """
    Immutable provider-independent conversation request.

    Attributes
    ----------
    system_instruction:
        Stable instruction defining how the character
        should behave.

    user_prompt:
        User-facing request to be answered.

    conversation_context:
        Optional contextual information, such as live
        intelligence retrieved by the Research Agent.

    character_id:
        Identifier of the character producing the response.

    character_name:
        Display name of the character.
    """

    system_instruction: str

    user_prompt: str

    conversation_context: str | None = None

    character_id: str = ""

    character_name: str = ""

    def __post_init__(self) -> None:
        """
        Validate request invariants.
        """

        if not isinstance(
            self.system_instruction,
            str,
        ):
            raise ValueError(
                "system_instruction must be a string."
            )

        if not isinstance(
            self.user_prompt,
            str,
        ):
            raise ValueError(
                "user_prompt must be a string."
            )

        system_instruction = (
            self.system_instruction.strip()
        )

        user_prompt = (
            self.user_prompt.strip()
        )

        if not system_instruction:
            raise ValueError(
                "system_instruction cannot be empty."
            )

        if not user_prompt:
            raise ValueError(
                "user_prompt cannot be empty."
            )

        object.__setattr__(
            self,
            "system_instruction",
            system_instruction,
        )

        object.__setattr__(
            self,
            "user_prompt",
            user_prompt,
        )

        if self.conversation_context is not None:
            context = (
                self.conversation_context.strip()
            )

            if not context:
                raise ValueError(
                    "conversation_context cannot be blank."
                )

            object.__setattr__(
                self,
                "conversation_context",
                context,
            )

        if not isinstance(
            self.character_id,
            str,
        ):
            raise ValueError(
                "character_id must be a string."
            )

        if not isinstance(
            self.character_name,
            str,
        ):
            raise ValueError(
                "character_name must be a string."
            )

        object.__setattr__(
            self,
            "character_id",
            self.character_id.strip(),
        )

        object.__setattr__(
            self,
            "character_name",
            self.character_name.strip(),
        )

    @property
    def has_conversation_context(self) -> bool:
        """
        Return whether contextual information exists.
        """

        return bool(
            self.conversation_context
        )

    @property
    def context_length(self) -> int:
        """
        Return the length of the conversation context.

        Useful for diagnostics and future token estimation.
        """

        if not self.conversation_context:
            return 0

        return len(
            self.conversation_context
        )

    def summary(self) -> dict[str, str]:
        """
        Return a safe diagnostic summary.

        Prompt and context content are deliberately
        excluded from the summary.
        """

        return {
            "character_id": self.character_id,
            "character_name": self.character_name,
            "has_context": str(
                self.has_conversation_context
            ),
            "system_instruction_length": str(
                len(self.system_instruction)
            ),
            "user_prompt_length": str(
                len(self.user_prompt)
            ),
            "context_length": str(
                self.context_length
            ),
        }

"""
Intelligence Context.

Provider-independent representation of external
information prepared for AI execution.

Architecture:

Research Agent
        |
        v
IntelligenceContext
        |
        v
AI Runtime integration
        |
        v
LLMRequest.context

The context must not expose provider-specific
implementation details to the application layer.
"""

from __future__ import annotations

from dataclasses import dataclass, field


@dataclass(frozen=True, slots=True)
class IntelligenceSource:
    """
    Normalised external information source.
    """

    title: str
    url: str
    source: str
    content: str = ""

    def __post_init__(self) -> None:
        if not self.title.strip():
            raise ValueError(
                "Intelligence source title cannot be empty."
            )

        if not self.url.strip():
            raise ValueError(
                "Intelligence source URL cannot be empty."
            )

        if not self.source.strip():
            raise ValueError(
                "Intelligence source name cannot be empty."
            )


@dataclass(frozen=True, slots=True)
class IntelligenceContext:
    """
    Provider-independent intelligence context.

    Represents external information that may be supplied
    to an existing AI conversation.

    This object deliberately does not contain:
    - provider SDK objects
    - API credentials
    - model information
    - runtime state
    - persistence objects
    """

    query: str

    sources: tuple[IntelligenceSource, ...] = field(
        default_factory=tuple,
    )

    summary: str = ""

    def __post_init__(self) -> None:
        if not self.query.strip():
            raise ValueError(
                "Intelligence query cannot be empty."
            )

        if not isinstance(
            self.sources,
            tuple,
        ):
            raise ValueError(
                "Intelligence sources must be a tuple."
            )

        if not isinstance(
            self.summary,
            str,
        ):
            raise ValueError(
                "Intelligence summary must be a string."
            )

    @property
    def has_sources(self) -> bool:
        """
        Return whether external sources are available.
        """

        return bool(self.sources)

    @property
    def source_count(self) -> int:
        """
        Return the number of external sources.
        """

        return len(self.sources)

    @property
    def has_summary(self) -> bool:
        """
        Return whether a summary is available.
        """

        return bool(
            self.summary.strip()
        )

    def to_context_text(self) -> str:
        """
        Convert intelligence into a safe textual context.

        The result is suitable for eventual inclusion
        in an existing LLMRequest.context field.
        """

        sections: list[str] = [
            "LIVE INTELLIGENCE",
            f"Research query: {self.query.strip()}",
        ]

        if self.summary.strip():
            sections.extend(
                (
                    "",
                    "Summary:",
                    self.summary.strip(),
                )
            )

        if self.sources:
            sections.extend(
                (
                    "",
                    "Sources:",
                )
            )

            for index, source in enumerate(
                self.sources,
                start=1,
            ):
                sections.append(
                    f"{index}. {source.title}"
                )
                sections.append(
                    f"   URL: {source.url}"
                )
                sections.append(
                    f"   Source: {source.source}"
                )

                if source.content.strip():
                    sections.append(
                        f"   Content: {source.content.strip()}"
                    )

        return "\n".join(sections)

"""
Research Service.

Coordinates the Research Agent and exposes a
provider-independent intelligence capability.

Architecture:

Application / Runtime
        |
        v
ResearchService
        |
        v
ResearchAgent
        |
        v
Search Provider
        |
        v
IntelligenceContext

The service does not know about Parallel SDK details.
"""

from __future__ import annotations

from backend.agents.research_agent import (
    ResearchAgent,
)

from backend.intelligence.intelligence_context import (
    IntelligenceContext,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class ResearchService:
    """
    Application-facing service for live research.

    The service coordinates the Research Agent but does
    not implement provider-specific search behaviour.
    """

    def __init__(
        self,
        agent: ResearchAgent | None = None,
    ) -> None:
        self.agent = (
            agent
            or ResearchAgent()
        )

    async def research(
        self,
        user_id: str,
        query: str,
    ) -> IntelligenceContext:
        """
        Execute research and return normalized intelligence.

        Args:
            user_id:
                Identifier for the current user/session.

            query:
                Research question.

        Returns:
            Provider-independent IntelligenceContext.

        Raises:
            ValueError:
                If the query is empty.

            RuntimeError:
                If the Research Agent fails.
        """

        if not query.strip():
            raise ValueError(
                "Research query cannot be empty."
            )

        context = ExecutionContext(
            user_id=user_id,
            request=query,
        )

        result = await self.agent.execute(
            context
        )

        if not result.success:
            raise RuntimeError(
                result.error
                or "Research execution failed."
            )

        if not isinstance(
            result.output,
            IntelligenceContext,
        ):
            raise TypeError(
                "Research Agent did not return "
                "an IntelligenceContext."
            )

        return result.output

"""
Parallel Search API Client Wrapper.

This module isolates the official Parallel SDK.

The rest of EasyChatBot does not directly depend on
the Parallel SDK implementation.

Architecture:

Research Agent
        |
        v
Parallel Provider
        |
        v
Parallel Client Wrapper
        |
        v
Parallel API
"""

import asyncio
import os
from typing import Any

from dotenv import load_dotenv


load_dotenv()


class ParallelClientWrapper:
    """
    Wrapper around the official Parallel Search SDK.

    Responsibilities:
    - authenticate with Parallel
    - execute searches
    - normalize SDK responses

    Does not:
    - create learning content
    - manage agents
    - decide workflows
    """

    def __init__(
        self,
        api_key: str | None = None,
    ):
        """
        Initialise the Parallel SDK client.
        """

        self.api_key = (
            api_key
            or os.getenv("PARALLEL_API_KEY")
        )

        if not self.api_key:
            raise ValueError(
                "PARALLEL_API_KEY environment variable is required"
            )

        self._client = self._create_client()


    def _create_client(self):
        """
        Create the official Parallel SDK client.

        SDK:
            parallel-web 1.1.0
        """

        from parallel import Parallel

        return Parallel(
            api_key=self.api_key
        )


    @staticmethod
    def _normalize_response(
        response: Any,
    ) -> Any:
        """
        Convert the Parallel SDK response into
        application-safe data.

        Pydantic SDK models are converted to dictionaries.
        Other JSON-compatible values are returned unchanged.
        """

        if hasattr(response, "model_dump"):
            return response.model_dump()

        if hasattr(response, "dict"):
            return response.dict()

        return response


    async def search(
        self,
        query: str,
    ) -> dict[str, Any]:
        """
        Execute a Parallel web search.

        Args:
            query:
                Natural-language research objective.

        Returns:
            Normalized application-level search result.
        """

        search_query = self._build_search_query(
            query
        )

        response = await asyncio.to_thread(
            self._client.search,
            search_queries=[
                search_query
            ],
            mode="advanced",
            objective=query,
        )

        return {
            "query": query,
            "search_queries": [
                search_query
            ],
            "results": self._normalize_response(
                response
            ),
        }


    @staticmethod
    def _build_search_query(
        query: str,
    ) -> str:
        """
        Build a concise search query from the
        user's research request.

        The full request remains the Parallel
        objective while the search query is kept
        concise.
        """

        words = query.strip().split()

        if len(words) <= 6:
            return query.strip()

        return " ".join(words[:6])

    Coordinates AI companion experiences.

    provider-specific logic.

        self,
        character: CharacterProfile,
        conversation_agent=None,
        research_agent=None,
        quality_agent=None,
        memory_service=None,
    ):

        self.character = character

        self.conversation_agent = (
            conversation_agent
            or ConversationAgent(character)
        )

        self.research_agent = (
            research_agent
            or ResearchAgent()
        )

        self.quality_agent = (
            quality_agent
            or QualityAgent()
        )

        self.memory_service = (
            memory_service
            or MemoryService()
        )

    async def interact(
        self,
        context: ExecutionContext,
        use_research: bool = False,
    ):
        """
        Execute a complete companion interaction.

        When research is requested, the Research Agent
        produces an IntelligenceContext which is attached
        to a new immutable ExecutionContext before the
        Conversation Agent executes.

        The original ExecutionContext is never mutated.
        """

        memories = (
            self.memory_service.recall(
                context.user_id
            )
        )

        research_result = None

        conversation_context = context

        if use_research:

            research_result = (
                await self.research_agent.execute(
                    context
                )
            )

            if (
                research_result.success
                and research_result.output is not None
            ):
                conversation_context = replace(
                    context,
                    intelligence_context=(
                        research_result.output
                    ),
                )

        conversation_result = (
            await self.conversation_agent.execute(
                conversation_context
            )
        )

        quality_result = (
            await self.quality_agent.execute(
                conversation_context
            )
        )

        self.memory_service.remember(
            user_id=context.user_id,
            category="conversation",
            content=context.request,
        )

        return {
            "character": self.character.name,

            "conversation":
                conversation_result.output,

            "research":
                (
                    research_result.output
                    if research_result
                    else None
                ),

            "quality":
                quality_result.output,

            "memory_count":
                len(memories),
        }

"""
AI Runtime capability execution contract.

This module defines the interface between the AI Runtime
and concrete capability executors.

Architecture:

AI Runtime
    |
    v
Capability Executor Interface
    |
    +---- Live Intelligence Executor
    +---- Learning Executor
    +---- Evaluation Executor
    +---- Future Executors

The Runtime depends on this contract.

It does not depend directly on:
- agents
- providers
- SDKs
- external APIs

Concrete executors are responsible for translating the
provider-independent CapabilityRequest into the appropriate
capability implementation.
"""

from __future__ import annotations

from abc import ABC, abstractmethod

class CapabilityExecutor(ABC):
    """
    Abstract executor for one Runtime capability.

    Each concrete executor must declare which capability
    it handles and implement asynchronous execution.
    """

    @property
    @abstractmethod
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """
        raise NotImplementedError

    @abstractmethod
    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a capability request.

        Implementations must return a provider-independent
        CapabilityResult.

        They must not bypass the Runtime governance boundary.
        """
        raise NotImplementedError

"""
AI Runtime Capability Gateway.

Controlled execution boundary between applications and
concrete AI capabilities.

Flow:

Application
    ↓
CapabilityRequest
    ↓
CapabilityGateway
    ↓
Governance
    ↓
CapabilityRegistry
    ↓
CapabilityExecutor
    ↓
CapabilityResult
"""

from __future__ import annotations

from typing import Protocol

from src.ai_runtime.capabilities.errors import (
    CapabilityExecutionError,
    CapabilityGovernanceError,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
)

from src.ai_runtime.capabilities.registry import (
    CapabilityRegistry,
)


class GovernanceDecisionResult(Protocol):
    """
    Minimal governance result required by the Runtime.
    """

    @property
    def allowed(self) -> bool:
        """
        Whether execution is permitted.
        """
        ...


class CapabilityGovernance(Protocol):
    """
    Governance boundary required by the Runtime gateway.
    """

    def evaluate(
        self,
        request: CapabilityRequest,
    ) -> GovernanceDecisionResult:
        """
        Evaluate whether a capability may execute.
        """
        ...


class CapabilityGateway:
    """
    Controlled Runtime entry point for capability execution.
    """

    def __init__(
        self,
        registry: CapabilityRegistry,
        governance: CapabilityGovernance,
    ) -> None:
        self._registry = registry
        self._governance = governance

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute an approved capability.

        Governance is evaluated before resolving and executing
        the capability.
        """

        try:
            governance_result = (
                self._governance.evaluate(request)
            )
        except Exception as exc:
            raise CapabilityGovernanceError(
                "Governance evaluation failed."
            ) from exc

        if not governance_result.allowed:
            return CapabilityResult(
                request_id=request.request_id,
                capability=request.capability,
                status=CapabilityExecutionStatus.REJECTED,
                error=(
                    "Capability execution was rejected "
                    "by governance."
                ),
                metadata={
                    "governance_decision": "rejected",
                },
            )

        executor = self._registry.resolve(
            request.capability
        )

        try:
            return await executor.execute(request)
        except Exception as exc:
            raise CapabilityExecutionError(
                "Capability executor failed: "
                f"{request.capability.value}"
            ) from exc

class LiveIntelligenceExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for the existing LiveIntelligenceAgent.
    """

    def __init__(
        self,
        agent: LiveIntelligenceAgent,
    ) -> None:
        self._agent = agent

    @property
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """

        return CapabilityType.LIVE_INTELLIGENCE

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Translate and execute a Live Intelligence request.
        """

        if (
            request.capability
            != CapabilityType.LIVE_INTELLIGENCE
        ):
            raise ValueError(
                "LiveIntelligenceExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        agent_request = (
            self._build_agent_request(request)
        )

        response = await self._agent.execute(
            agent_request
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=response,
            metadata={
                "executor": (
                    "live_intelligence_executor"
                ),
            },
        )

    @staticmethod
    def _build_agent_request(
        request: CapabilityRequest,
    ) -> LiveIntelligenceRequest:
        """
        Translate Runtime payload into the existing
        LiveIntelligenceRequest contract.
        """

        payload = request.payload

        topic = payload.get("topic")

        if not isinstance(topic, str):
            raise ValueError(
                "Live Intelligence capability requires "
                "a string 'topic'."
            )

        language = payload.get(
            "language",
            "French",

"""
AI Runtime capability registry.

The registry maps logical Runtime capabilities to their
concrete executors.

The registry:
- resolves executors by capability
- prevents duplicate registrations
- provides explicit dependency injection
- does not execute capabilities
- does not perform governance decisions
- does not know about providers
"""

from __future__ import annotations

from src.ai_runtime.capabilities.errors import (
    CapabilityAlreadyRegisteredError,
    CapabilityNotRegisteredError,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityType,
)


class CapabilityRegistry:
    """
    Dependency-injection registry for Runtime capabilities.
    """

    def __init__(
        self,
        executors: list[CapabilityExecutor] | None = None,
    ) -> None:
        self._executors: dict[
            CapabilityType,
            CapabilityExecutor,
        ] = {}

        for executor in executors or []:
            self.register(executor)

    def register(
        self,
        executor: CapabilityExecutor,
    ) -> None:
        """
        Register an executor for its declared capability.
        """

        capability = executor.capability

        if capability in self._executors:
            raise CapabilityAlreadyRegisteredError(
                "Capability executor already registered: "
                f"{capability.value}"
            )

        self._executors[capability] = executor

    def resolve(
        self,
        capability: CapabilityType,
    ) -> CapabilityExecutor:
        """
        Resolve the executor responsible for a capability.
        """

        try:
            return self._executors[capability]
        except KeyError as exc:
            raise CapabilityNotRegisteredError(
                "No executor registered for capability: "
                f"{capability.value}"
            ) from exc

    def contains(
        self,
        capability: CapabilityType,
    ) -> bool:
        """
        Return whether an executor is registered.
        """

        return capability in self._executors

    def capabilities(
        self,
    ) -> tuple[CapabilityType, ...]:
        """
        Return all registered capabilities.
        """

        return tuple(self._executors.keys())

"""
Flash Deal Card presenter.

Pure presentation layer for the EasyChatBot hackathon.

Architecture:

FlashDealCard
      |
      v
FlashDealPresenter
      |
      v
Markdown / terminal output

Responsibilities:
- render an existing FlashDealCard
- present verified product and pricing information
- present character context
- present personal relevance
- expose the existing vendor actions

Non-responsibilities:
- discovery
- memory access
- provider calls
- database access
- commerce execution
- payment processing
- URL generation
- product data generation

The presenter must never mutate the FlashDealCard.
"""

from __future__ import annotations

from src.ai_runtime.discovery.flash_deal_card import (
    FlashDealCard,
)


class FlashDealPresenter:
    """
    Pure renderer for the user-facing Flash Deal Card.

    The presenter consumes the existing FlashDealCard contract
    and produces deterministic Markdown suitable for the
    hackathon terminal demonstration or a future chat UI.
    """

    def render(
        self,
        card: FlashDealCard,
    ) -> str:
        """
        Render a FlashDealCard as user-facing Markdown.

        No external calls or business decisions are performed.
        """

        sections: list[str] = []

        sections.append(
            f"## {card.character_name}"
        )

        sections.append(
            f"> {card.character_message}"
        )

        sections.append(
            f"### {card.badge}"
        )

        sections.append(
            f"**{card.product_name}**"
        )

        sections.append(
            f"*{card.tagline}*"
        )

        sections.append(
            self._render_highlights(card)
        )

        sections.append(
            self._render_price(card)
        )

        sections.append(
            "### WHY THIS IS RELEVANT TO YOU"
        )

        sections.append(
            card.relevance_reason
        )

        sections.append(
            "### AVAILABLE FROM"
        )

        sections.append(
            card.vendor_name
        )

        sections.append(
            self._render_actions(card)
        )

        return "\n\n".join(sections)

    @staticmethod
    def _render_highlights(
        card: FlashDealCard,
    ) -> str:
        """Render the product highlights."""

        lines = ["### HIGHLIGHTS"]

        lines.extend(
            f"- {highlight}"
            for highlight in card.highlights
        )

        return "\n".join(lines)

    @staticmethod
    def _render_price(
        card: FlashDealCard,
    ) -> str:
        """Render verified pricing information."""

        lines = ["### PRICE"]

        if card.previous_price is not None:
            lines.append(
                "Previous price: "
                f"{card.currency} "
                f"{card.previous_price:.2f}"
            )

        lines.append(
            "Current price: "
            f"{card.currency} "
            f"{card.current_price:.2f}"
        )

        if card.display_discount is not None:
            lines.append(
                f"Discount: {card.display_discount}"
            )

        if card.display_savings is not None:
            lines.append(
                f"Savings: {card.display_savings}"
            )

        return "\n".join(lines)

    @staticmethod
    def _render_actions(
        card: FlashDealCard,
    ) -> str:
        """
        Render vendor action destinations.

        The presenter only exposes URLs already present on the
        FlashDealCard. It does not create or transform them.
        """

        return "\n".join(
            [
                "### ACTIONS",
                "",
                f"[GET DEAL]({card.get_deal_url})",
                "",
                (
                    "[MORE INFORMATION]"
                    f"({card.more_information_url})"
                ),
            ]
        )

 External Vendor

from __future__ import annotations

class CommerceLinkType(str, Enum):
    """
    Type of outbound commerce link.
    """

    DIRECT = "direct"

    AFFILIATE = "affiliate"


@dataclass(frozen=True)
class CommerceLink:
    """
    Outbound link to the external vendor.

    EasyChatBot does not process the transaction.

    The destination remains the vendor's website.
    """

    vendor: str

    destination_url: str

    link_type: CommerceLinkType = (
        CommerceLinkType.DIRECT
    )

    tracking_url: str | None = None

    def __post_init__(self) -> None:
        """
        Validate the commerce link.
        """

        if not self.vendor.strip():
            raise ValueError(
                "Commerce vendor cannot be empty."
            )

        if not self.destination_url.strip():
            raise ValueError(
                "Commerce destination URL "
                "cannot be empty."
            )

        if not (
            self.destination_url.startswith(
                "https://"
            )
            or self.destination_url.startswith(
                "http://"
            )
        ):
            raise ValueError(
                "Commerce destination URL must "
                "use HTTP or HTTPS."
            )

        if self.tracking_url is not None:
            if not self.tracking_url.strip():
                raise ValueError(
                    "Commerce tracking URL cannot "
                    "be empty when provided."
                )

            if not (
                self.tracking_url.startswith(
                    "https://"
                )
                or self.tracking_url.startswith(
                    "http://"
                )
            ):
                raise ValueError(
                    "Commerce tracking URL must "
                    "use HTTP or HTTPS."
                )

    @property
    def click_url(self) -> str:
        """
        Return the URL that should be opened by the user.

        A tracking URL takes precedence when one is
        available.
        """

        return (
            self.tracking_url
            if self.tracking_url is not None
            else self.destination_url
        )

"""
Hackathon Memory persistence.

This module provides the minimal persistent Memory Store
required by the hackathon.

Architecture:

MemoryService
      |
      v
MemoryStore
      |
      v
SQLite

The store is deliberately small.

It is responsible only for persistence and retrieval of
MemoryEntry objects.

It does not:
- decide what should be remembered
- call an LLM
- perform semantic retrieval
- execute agents
- perform governance decisions
- manage purchases
"""

from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryStore:
    """
    SQLite-backed persistence for hackathon memories.

    The database schema is created automatically when the
    store is initialised.
    """

    def __init__(
        self,
        database_path: str | Path,
    ) -> None:
        self._database_path = str(database_path)

        self._initialise_database()

    def _connect(self) -> sqlite3.Connection:
        """
        Create a SQLite connection.
        """

        connection = sqlite3.connect(
            self._database_path
        )

        connection.row_factory = sqlite3.Row

        return connection

    def _initialise_database(self) -> None:
        """
        Create the Memory table if it does not exist.
        """

        with self._connect() as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    memory_id TEXT PRIMARY KEY,
                    user_id TEXT NOT NULL,
                    memory_type TEXT NOT NULL,
                    content TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    metadata_json TEXT NOT NULL
                )
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_id
                ON memories(user_id)
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_type
                ON memories(user_id, memory_type)
                """
            )

    def save(
        self,
        memory: MemoryEntry,
    ) -> MemoryEntry:
        """
        Persist a MemoryEntry.

        Existing entries with the same memory_id are
        replaced.
        """

        with self._connect() as connection:
            connection.execute(
                """
                INSERT OR REPLACE INTO memories (
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                )
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.memory_id,
                    memory.user_id,
                    memory.memory_type.value,
                    memory.content,
                    memory.created_at.isoformat(),
                    memory.updated_at.isoformat(),
                    json.dumps(
                        memory.metadata,
                        ensure_ascii=False,
                    ),
                ),
            )

        return memory

    def get(
        self,
        memory_id: str,
    ) -> MemoryEntry | None:
        """
        Retrieve a single memory by identifier.
        """

        with self._connect() as connection:
            row = connection.execute(
                """
                SELECT
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            ).fetchone()

        if row is None:
            return None

        return self._row_to_memory(row)

    def list_for_user(
        self,
        user_id: str,
        memory_type: MemoryType | None = None,
    ) -> list[MemoryEntry]:
        """
        Retrieve memories belonging to a user.

        Results are returned newest first.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        with self._connect() as connection:

            if memory_type is None:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                    ORDER BY updated_at DESC
                    """,
                    (user_id,),
                ).fetchall()

            else:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                      AND memory_type = ?
                    ORDER BY updated_at DESC
                    """,
                    (
                        user_id,
                        memory_type.value,
                    ),
                ).fetchall()

        return [
            self._row_to_memory(row)
            for row in rows
        ]

    def delete(
        self,
        memory_id: str,
    ) -> bool:
        """
        Delete a memory.

        Returns True when a memory was deleted.
        """

        with self._connect() as connection:
            cursor = connection.execute(
                """
                DELETE FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            )

        return cursor.rowcount > 0

    @staticmethod
    def _row_to_memory(
        row: sqlite3.Row,
    ) -> MemoryEntry:
        """
        Convert a SQLite row into a MemoryEntry.
        """

        return MemoryEntry(
            memory_id=row["memory_id"],
            user_id=row["user_id"],
            memory_type=MemoryType(
                row["memory_type"]
            ),
            content=row["content"],
            created_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["created_at"]
                )
            ),
            updated_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["updated_at"]
                )
            ),
            metadata=json.loads(
                row["metadata_json"]
            ),
        )

"""
Live Intelligence Agent Handler.

Responsible for executing the Live Intelligence Agent workflow.

Architecture:

AI Runtime
    |
    v
Live Intelligence Agent Handler
    |
    +--> Search Provider Interface
    |
    +--> Knowledge Processing
    |

Structured Intelligence Output

Responsibilities:

- Call approved provider interfaces
- Normalise research output
- Return explicit structured results

Non-responsibilities:
- Provider SDK management

- Autonomous actions
- Configuration changes
"""

from dataclasses import dataclass
from typing import List, Protocol

from src.providers.search_provider.interface import (
    SearchProvider,
    SearchResult,
)


@dataclass(frozen=True)
class LiveIntelligenceRequest:
    """
    Input contract for Live Intelligence Agent.
    """

    topic: str
    language: str = "French"
    learner_level: str = "B2"


@dataclass(frozen=True)
class LiveIntelligenceResponse:
    """
    Output contract from Live Intelligence Agent.
    """

    topic: str
    summary: str
    sources: List[SearchResult]
    key_information: List[str]


class IntelligenceProcessor(Protocol):
    """
    Contract for future LLM-based processing.

    Keeps the agent independent from
    any specific model/provider.
    """

    async def summarise(
        self,
        results: List[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, List[str]]:
        ...


class LiveIntelligenceAgent:
    """
    Live Intelligence Agent implementation.
    """

    def __init__(
        self,
        search_provider: SearchProvider,
        processor: IntelligenceProcessor,
    ) -> None:
        self._search_provider = search_provider
        self._processor = processor

    async def execute(
        self,
        request: LiveIntelligenceRequest,
    ) -> LiveIntelligenceResponse:
        """
        Execute research workflow.

        Flow:

        Request
          |
          v
        Search Provider
          |
          v
        Intelligence Processor
          |
          v
        Structured Response
        """

        self._validate_request(request)

        search_results = await self._search_provider.search(
            query=request.topic,
            max_results=5,
        )

        summary, key_information = await self._processor.summarise(
            results=search_results,
            language=request.language,
            learner_level=request.learner_level,
        )

        return LiveIntelligenceResponse(
            topic=request.topic,
            summary=summary,
            sources=search_results,
            key_information=key_information,
        )

    @staticmethod
    def _validate_request(
        request: LiveIntelligenceRequest,
    ) -> None:
        """
        Validate required input.
        """

        if not request.topic.strip():
            raise ValueError(
                "Topic cannot be empty"
            )

"""
Parallel Search Provider Adapter.

Bridges the provider-independent SearchProvider contract
to the existing EasyChatBot ParallelClientWrapper.

Architecture:

Live Intelligence Agent
        |
        v
SearchProvider Interface
        |
        v
ParallelSearchProvider
        |
        v
Existing ParallelClientWrapper
        |
        v
Parallel Search API

The official Parallel SDK remains isolated inside:

    backend/providers/parallel/parallel_client.py

This adapter must not import or instantiate the Parallel SDK
directly.
"""

from typing import Any

from backend.providers.parallel.parallel_client import (
    ParallelClientWrapper,
)

from src.providers.search_provider.interface import (
    SearchProvider,
    SearchResult,
)


class ParallelSearchProvider(SearchProvider):
    """
    SearchProvider implementation backed by the existing
    ParallelClientWrapper.
    """

    def __init__(
        self,
        api_key: str | None = None,
        client: ParallelClientWrapper | None = None,
    ) -> None:
        """
        Initialise the provider.

        Args:
            api_key:
                Optional Parallel API key.

                If omitted, the existing ParallelClientWrapper
                loads PARALLEL_API_KEY from the environment/.env.

            client:
                Optional injected client for testing.
        """

        self._client = client or ParallelClientWrapper(
            api_key=api_key
        )

    async def search(
        self,
        query: str,
        max_results: int = 5,
    ) -> list[SearchResult]:
        """
        Execute a Parallel search and normalise its results.

        Args:
            query:
                Natural-language research objective.

            max_results:
                Maximum number of results returned.

        Returns:
            Provider-independent SearchResult objects.
        """

        if not query.strip():
            raise ValueError(
                "Search query cannot be empty"
            )

        if max_results <= 0:
            raise ValueError(
                "max_results must be greater than zero"
            )

        response = await self._client.search(
            query=query
        )

        raw_results = self._extract_results(
            response
        )

        normalised_results: list[SearchResult] = []

        for raw_result in raw_results[:max_results]:
            result = self._normalise_result(
                raw_result
            )

            if result is not None:
                normalised_results.append(result)

        return normalised_results

    @staticmethod
    def _extract_results(
        response: Any,
    ) -> list[Any]:
        """
        Extract the actual result collection from the
        ParallelClientWrapper response.

        The existing wrapper returns:

            {
                "query": "...",
                "search_queries": [...],
                "results": <normalised SDK response>
            }

        The SDK response may itself contain:

            {
                "results": [...]
            }

        Therefore this method handles both levels of wrapping.
        """

        if not isinstance(response, dict):
            raise TypeError(
                "Parallel client response must be a dictionary"
            )

        raw_results = response.get(
            "results",
            []
        )

        # The SDK response itself may contain a nested
        # "results" collection.
        if isinstance(raw_results, dict):
            nested_results = raw_results.get(
                "results"
            )

            if isinstance(nested_results, list):
                return nested_results

            # Some SDK versions may expose the result
            # collection under another common field.
            nested_data = raw_results.get(
                "data"
            )

            if isinstance(nested_data, list):
                return nested_data

        if isinstance(raw_results, list):
            return raw_results

        # If the SDK returned an object that was not converted
        # to a dictionary, try to inspect its result attribute.
        nested_results = getattr(
            raw_results,
            "results",
            None,
        )

        if isinstance(nested_results, list):
            return nested_results

        raise TypeError(
            "Parallel search response does not contain "
            "a supported results collection"
        )

    @staticmethod
    def _normalise_result(
        raw_result: Any,
    ) -> SearchResult | None:
        """
        Convert one Parallel result into SearchResult.

        Parallel results may arrive either as dictionaries
        or SDK/Pydantic objects.
        """

        if isinstance(raw_result, dict):
            title = str(
                raw_result.get(
                    "title",
                    ""
                )
            ).strip()

            url = str(
                raw_result.get(
                    "url",
                    ""
                )
            ).strip()

            excerpts = raw_result.get(
                "excerpts",
                []
            )

            # Support providers/SDK versions that use
            # "snippet" or "content" instead.
            if not excerpts:
                excerpts = raw_result.get(
                    "snippet",
                    raw_result.get(
                        "content",
                        []
                    ),
                )

        else:
            title = str(
                getattr(
                    raw_result,
                    "title",
                    ""
                )
            ).strip()

            url = str(
                getattr(
                    raw_result,
                    "url",
                    ""
                )
            ).strip()

            excerpts = getattr(
                raw_result,
                "excerpts",
                None,
            )

            if not excerpts:
                excerpts = getattr(
                    raw_result,
                    "snippet",
                    getattr(
                        raw_result,
                        "content",
                        [],
                    ),
                )

        if not title or not url:
            return None

        content = ParallelSearchProvider._normalise_excerpts(
            excerpts
        )

        return SearchResult(
            title=title,
            url=url,
            content=content,
            source="parallel",
        )

    @staticmethod
    def _normalise_excerpts(
        excerpts: Any,
    ) -> str:
        """
        Convert Parallel excerpts/content into a single
        application-level content string.
        """

        if excerpts is None:
            return ""

        if isinstance(excerpts, str):
            return excerpts.strip()

        if isinstance(excerpts, list):
            return "\n\n".join(
                str(excerpt).strip()
                for excerpt in excerpts
                if str(excerpt).strip()
            )

        return str(excerpts).strip()
