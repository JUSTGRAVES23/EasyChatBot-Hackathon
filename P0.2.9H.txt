@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    """
    Manage application startup and shutdown lifecycle.

    Startup:
        - compose application dependencies;
        - create the authoritative functionality-control service;
        - start Telegram transport runtime.

    Shutdown:
        - stop Telegram transport runtime;
        - dispose of the functionality-control database engine.

    This layer does not:

    - execute agents;
    - execute capabilities;
    - create providers;
    - create Runtime components.
    """

    bootstrap_application()

    (
        functionality_control_service,
        functionality_control_engine,
    ) = create_standalone_functionality_control_service()

    application.state.functionality_control_service = (
        functionality_control_service
    )
    application.state.functionality_control_engine = (
        functionality_control_engine
    )

    try:
        await start_telegram_runtime(
            functionality_control_service=(
                functionality_control_service
            ),
        )

        yield

    finally:
        await stop_telegram_runtime()

        functionality_control_engine.dispose()

        application.state.functionality_control_service = None
        application.state.functionality_control_engine = None


def create_app() -> FastAPI:
    """
    Create the EasyChatBot FastAPI application.

    Application composition is intentionally separated
    from the HTTP layer.

    This function:

    - creates the FastAPI application;
    - configures local frontend access;
    - registers API routes;
    - attaches lifecycle management.

    It does not:

    - execute agents;
    - execute capabilities;
    - call providers directly;
    - create Runtime executors.
    """

    application = FastAPI(
        title="EasyChatBot",
        description="AI companion learning platform",
        lifespan=lifespan,
    )

    # =========================================================
    # Local Frontend CORS
    # =========================================================

    application.add_middleware(
        CORSMiddleware,
        allow_origins=[
            "http://127.0.0.1:5173",
            "http://localhost:5173",
            "http://127.0.0.1:5500",
            "http://localhost:5500",
            "http://127.0.0.1:5501",
            "http://localhost:5501",
            "http://localhost:3000",
            "http://127.0.0.1:3000",
        ],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    # =========================================================
    # Existing API Routes
    # =========================================================

    application.include_router(
        router,
    )

    # =========================================================
    # Agentic Cinema Routes
    # =========================================================
    #
    # Browser
    #     ↓
    # /api/cinema
    #     ↓
    # FastAPI Cinema Boundary
    #     ↓
    # Google Agent Runtime
    #     ↓
    # Gemini + Parallel + G5 workflow
    #
    # The API layer does not execute the agent directly.
    # The Cinema router owns the HTTP/application boundary.
    # =========================================================

    application.include_router(
        cinema_router,
    )

    # =========================================================
    # Character Theme Routes
    # =========================================================

    application.include_router(
        character_theme_router,
    )

    # =========================================================
    # Character Catalogue Routes
    # =========================================================

    application.include_router(
        character_router,
    )

    # =========================================================
    # AI Operations Center Routes
    # =========================================================

    application.include_router(
        operations_center_dashboard_router,
    )

    # =========================================================
    # Cybersecurity Intelligence Routes
    # =========================================================

    application.include_router(
        cybersecurity_intelligence_router,
    )

    return application


app = create_app()

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = (
        application.state.scheduler_runtime
    )

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )
    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )

def bootstrap_application() -> None:
    """
    Bootstrap application dependencies.

    Creates the ExternalRuntimeService and registers it
    inside the application runtime container.
    """

        return

        podcast_discovery_agent=(
            create_podcast_discovery_agent()

        podcast_providers=(
            create_podcast_providers()
        ),

            create_governance()
        ),

    )

    runtime_service = (
        factory.create_external_runtime_service(
            gateway=gateway,
        )
    )

    configure_external_runtime_service(
        runtime_service,
    )

    _bootstrapped = True

class BootstrapGovernanceDecision:
    """
    Runtime governance decision.
    """

    def __init__(
        self,
        allowed: bool,
    ) -> None:

        self.allowed = allowed


class BootstrapGovernance:
    """
    Temporary hackathon governance implementation.

    Sensitive internal-data capabilities fail closed.

    The Runtime receives a provider-independent
    RuntimeActorContext. Application/API identity remains
    outside the Runtime boundary.
    """

    def evaluate(
        self,
        request,
    ) -> BootstrapGovernanceDecision:

        capability = getattr(
            request,
            "capability",
            None,
        )

        capability_value = getattr(
            capability,
            "value",
            capability,
        )

        if capability_value == "internal_data_read":
            actor = getattr(
                request,
                "actor",
                None,
            )

            if not isinstance(
                actor,
                RuntimeActorContext,
            ):
                return BootstrapGovernanceDecision(
                    allowed=False,
                )

            allowed = has_permission(
                roles=actor.roles,
                permission=Permission.INTERNAL_DATA_READ,
            )

            return BootstrapGovernanceDecision(
                allowed=allowed,
            )

        return BootstrapGovernanceDecision(
            allowed=True,
        )


def create_governance():
    """
    Create Runtime governance dependency.
    """

    return BootstrapGovernance()

"""
Application Factory.

Creates application-layer services around the authoritative
external Runtime composition.

The factory delegates Runtime capability composition to
``src.ai_runtime.composition`` and wraps the resulting gateway
with application-layer services.

It does not create a second Runtime, registry, executor,
provider abstraction, or execution engine.
"""

from __future__ import annotations

from typing import Any, Sequence

from src.ai_runtime.capabilities.gateway import CapabilityGateway
from src.ai_runtime.composition import (
    create_external_runtime_gateway,
)

from backend.services.external_runtime_service import (
    ExternalRuntimeService,
)
from backend.services.runtime.runtime_application_service import (
    RuntimeApplicationService,
)


class ApplicationFactory:
    """Application composition factory."""

    @staticmethod
    def create_external_runtime(
        *,
        live_intelligence_agent: Any,
        podcast_discovery_agent: Any,
        podcast_providers: Sequence[Any],
        governance: Any,
        internal_documentation_agent: Any | None = None,
    ) -> ExternalRuntimeService:
        """
        Compose the authoritative external Runtime gateway and
        expose it through the application-facing service.

        Runtime internals remain owned by
        ``src.ai_runtime.composition``.

        This method does not create a parallel Runtime architecture.
        """

        gateway = create_external_runtime_gateway(
            live_intelligence_agent=live_intelligence_agent,
            podcast_discovery_agent=podcast_discovery_agent,
            podcast_providers=podcast_providers,
            governance=governance,
            internal_documentation_agent=(
                internal_documentation_agent
            ),
        )

        return ApplicationFactory.create_external_runtime_service(
            gateway=gateway,
        )

    @staticmethod
    def create_external_runtime_service(
        *,
        gateway: CapabilityGateway,
    ) -> ExternalRuntimeService:
        """
        Create the application-facing ExternalRuntimeService
        around an already-composed CapabilityGateway.
        """

        if gateway is None:
            raise ValueError(
                "CapabilityGateway cannot be None"
            )

        return ExternalRuntimeService(
            gateway=gateway,
        )

    @staticmethod
    def create_runtime_application_service(
        *,
        runtime_service: ExternalRuntimeService,
    ) -> RuntimeApplicationService:
        """
        Create the application RuntimeApplicationService.

        The RuntimeApplicationService receives an already-composed
        ExternalRuntimeService.

        This factory does not:

        - create providers;
        - create agents;
        - create gateways;
        - compose Runtime internals;
        - execute capabilities.
        """

        if runtime_service is None:
            raise ValueError(
                "ExternalRuntimeService cannot be None"
            )

        return RuntimeApplicationService(
            runtime_service=runtime_service,
        )

    *,

    The functionality-control service is supplied explicitly by the

    second control repository or control service.
    """

    global _telegram_production

    if functionality_control_service is None:
        raise ValueError(
            "functionality_control_service is required."
        )

    if _telegram_production is None:
        _telegram_production = (
            create_production_beta_telegram_composition(
                database_engine_factory=(
