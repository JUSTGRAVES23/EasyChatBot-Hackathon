from __future__ import annotations

from dataclasses import dataclass

import pytest

from app.application.beta_access.beta_telegram_conversation_boundary import (
    BetaTelegramConversationBoundary,
)


@dataclass(frozen=True)
class FakeConversationContext:
    world_id: str = "world-fr"
    character: str = "Alix"
    level: str = "B1"


class FakeBetaAccessService:
    """Deterministic authorization/context test double."""

    def __init__(self, *, approved: bool = True) -> None:
        self.approved = approved
        self.calls: list[int] = []

    async def get_conversation_context(
        self,
        telegram_user_id: int,
    ) -> FakeConversationContext:
        self.calls.append(telegram_user_id)

        if not self.approved:
            raise PermissionError(
                "Telegram user is not approved."
            )

        return FakeConversationContext()


class FakeConversationService:
    """Deterministic conversation-service test double."""

    def __init__(
        self,
        *,
        response: str = "Bonjour !",
        error: Exception | None = None,
    ) -> None:
        self.response = response
        self.error = error
        self.calls: list[dict] = []

    async def handle_telegram_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        self.calls.append(
            {
                "telegram_user_id": telegram_user_id,
                "message": message,
            }
        )

        if self.error is not None:
            raise self.error

        return self.response


def make_boundary(
    *,
    approved: bool = True,
    conversation_error: Exception | None = None,
):
    access = FakeBetaAccessService(
        approved=approved,
    )

    conversation_service = FakeConversationService(
        error=conversation_error,
    )

    boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation_service,
        beta_access_service=access,
    )

    return (
        boundary,
        access,
        conversation_service,
    )


@pytest.mark.asyncio
async def test_ws06_conversation_failure_returns_safe_fallback():
    (
        boundary,
        access,
        conversation_service,
    ) = make_boundary(
        conversation_error=RuntimeError(
            "internal conversation failure"
        ),
    )

    response = await boundary.handle_message(
        telegram_user_id=12345,
        message="Bonjour",
    )

    assert response

    assert (
        "internal conversation failure"
        not in response
    )

    assert "traceback" not in response.lower()

    assert access.calls == [12345]

    assert conversation_service.calls == [
        {
            "telegram_user_id": 12345,
            "message": "Bonjour",
        }
    ]

def _conversation_event_columns(
    *,
    engine,
) -> set[str]:
    """
    Discover the real production conversation_events schema.

    The test deliberately does not assume an event_id column.
    """

    inspector = inspect(engine)

    return {
        column["name"]
        for column in inspector.get_columns(
            "conversation_events",
        )
    }


def _find_existing_telegram_identities(
    *,
    engine,
    minimum: int = 1,
) -> list[dict[str, str]]:
    """
    Retrieve persisted Telegram classroom identities from PostgreSQL.
    """

    columns = _conversation_event_columns(
        engine=engine,
    )

    required = {
        "telegram_user_hash",
        "conversation_id",
        "world_id",
    }

    missing = required - columns

    if missing:
        raise AssertionError(
            "conversation_events is missing required "
            f"columns: {sorted(missing)}"
        )

    with engine.connect() as connection:
        rows = connection.execute(
            text(
                """
                SELECT DISTINCT
                    telegram_user_hash,
                    conversation_id,
                    world_id
                FROM conversation_events
                WHERE telegram_user_hash IS NOT NULL
                  AND world_id IS NOT NULL
                ORDER BY telegram_user_hash
                """
            )
        ).mappings().all()

    identities = [
        {
            "telegram_user_hash": row[
                "telegram_user_hash"
            ],
            "conversation_id": row[
                "conversation_id"
            ],
            "world_id": row[
                "world_id"
            ],
        }
        for row in rows
    ]

    if len(identities) < minimum:
        raise RuntimeError(
            "WS-19 requires at least "
            f"{minimum} persisted Telegram identity "
            "in PostgreSQL."
        )

    return identities


def _find_recent_event(
    *,
    engine,
    telegram_user_hash: str,
):
    """
    Retrieve the most recent persisted conversation event.

    The production table has no event_id column, so ordering uses created_at.
    """

    columns = _conversation_event_columns(
        engine=engine,
    )

    required = {
        "telegram_user_hash",
        "conversation_id",
        "world_id",
        "role",
        "content",
        "created_at",
    }

    missing = required - columns

    if missing:
        raise AssertionError(
            "conversation_events is missing required "
            f"columns: {sorted(missing)}"
        )

    with engine.connect() as connection:
        return connection.execute(
            text(
                """
                SELECT
                    telegram_user_hash,
                    conversation_id,
                    world_id,
                    role,
                    content,
                    created_at
                FROM conversation_events
                WHERE telegram_user_hash = :telegram_user_hash
                ORDER BY created_at DESC
                LIMIT 1
                """
            ),
            {
                "telegram_user_hash":
                    telegram_user_hash,
            },
        ).mappings().first()

def test_ws19_different_users_remain_isolated():
    """
    AC9:
        Different Telegram identities do not share persisted conversations.
    """

    composition = _build_composition()

    try:
        identities = (
            _find_existing_telegram_identities(
                engine=composition.database_engine,
                minimum=2,
            )
        )

        first = identities[0]
        second = identities[1]

        with composition.database_engine.connect() as connection:
            first_rows = connection.execute(
                text(
                    """
                    SELECT
                        telegram_user_hash,
                        conversation_id,
                        world_id,
                        role,
                        content
                    FROM conversation_events
                    WHERE telegram_user_hash = :user_hash
                    """
                ),
                {
                    "user_hash":
                        first[
                            "telegram_user_hash"
                        ],
                },
            ).mappings().all()

            second_rows = connection.execute(
                text(
                    """
                    SELECT
                        telegram_user_hash,
                        conversation_id,
                        world_id,
                        role,
                        content
                    FROM conversation_events
                    WHERE telegram_user_hash = :user_hash
                    """
                ),
                {
                    "user_hash":
                        second[
                            "telegram_user_hash"
                        ],
                },
            ).mappings().all()

        assert first_rows
        assert second_rows

        assert (
            first["telegram_user_hash"]
            != second["telegram_user_hash"]
        )

        first_conversations = {
            row["conversation_id"]
            for row in first_rows
        }

        second_conversations = {
            row["conversation_id"]
            for row in second_rows
        }

        assert first_conversations
        assert second_conversations

        assert not (
            first_conversations
            & second_conversations
        )

        assert all(
            row["telegram_user_hash"]
            == first["telegram_user_hash"]
            for row in first_rows
        )

        assert all(
            row["telegram_user_hash"]
            == second["telegram_user_hash"]
            for row in second_rows
        )

    finally:
        _dispose_composition(composition)
