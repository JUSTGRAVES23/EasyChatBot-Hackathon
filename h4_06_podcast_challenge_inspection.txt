"""
EasyChatBot Hackathon — Podcast Learning Challenge contracts.

Presentation/learning contract only. This module does not download audio,
transcribe podcasts, call providers, or execute external actions.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from uuid import uuid4


@dataclass(frozen=True)
class VocabularyItem:
    """A vocabulary item presented to the learner."""

    term: str
    meaning: str
    example: str

    def __post_init__(self) -> None:
        for name, value in {
            "term": self.term,
            "meaning": self.meaning,
            "example": self.example,
        }.items():
            if not value.strip():
                raise ValueError(
                    f"VocabularyItem {name} cannot be empty."
                )


@dataclass(frozen=True)
class ComprehensionQuestion:
    """A multiple-choice comprehension question."""

    question: str
    options: tuple[str, ...]
    correct_option: int

    def __post_init__(self) -> None:
        if not self.question.strip():
            raise ValueError(
                "ComprehensionQuestion question cannot be empty."
            )

        if len(self.options) < 2:
            raise ValueError(
                "ComprehensionQuestion requires at least two options."
            )

        if any(not option.strip() for option in self.options):
            raise ValueError(
                "ComprehensionQuestion options cannot be empty."
            )

        if not 0 <= self.correct_option < len(self.options):
            raise ValueError(
                "ComprehensionQuestion correct_option is out of range."
            )


@dataclass(frozen=True)
class PodcastLearningChallenge:
    """Learner-facing challenge derived from a verified podcast recommendation."""

    challenge_id: str
    user_id: str
    podcast_id: str
    podcast_title: str
    podcast_url: str
    target_language: str
    learner_level: str
    vocabulary: tuple[VocabularyItem, ...]
    comprehension_questions: tuple[ComprehensionQuestion, ...]
    discussion_prompt: str
    introduction: str
    metadata: dict[str, str] = field(default_factory=dict)

    def __post_init__(self) -> None:
        required = {
            "challenge_id": self.challenge_id,
            "user_id": self.user_id,
            "podcast_id": self.podcast_id,
            "podcast_title": self.podcast_title,
            "podcast_url": self.podcast_url,
            "target_language": self.target_language,
            "learner_level": self.learner_level,
            "discussion_prompt": self.discussion_prompt,
            "introduction": self.introduction,
        }

        for name, value in required.items():
            if not value.strip():
                raise ValueError(
                    f"PodcastLearningChallenge {name} cannot be empty."
                )

        if not (
            self.podcast_url.startswith("https://")
            or self.podcast_url.startswith("http://")
        ):
            raise ValueError(
                "PodcastLearningChallenge podcast_url must use HTTP or HTTPS."
            )

        if not self.vocabulary:
            raise ValueError(
                "PodcastLearningChallenge requires vocabulary."
            )

        if not self.comprehension_questions:
            raise ValueError(
                "PodcastLearningChallenge requires comprehension questions."
            )

    @classmethod
    def create(
        cls,
        *,
        user_id: str,
        podcast_id: str,
        podcast_title: str,
        podcast_url: str,
        target_language: str,
        learner_level: str,
        vocabulary: tuple[VocabularyItem, ...],
        comprehension_questions: tuple[ComprehensionQuestion, ...],
        discussion_prompt: str,
        introduction: str,
        metadata: dict[str, str] | None = None,
    ) -> "PodcastLearningChallenge":
        return cls(
            challenge_id=str(uuid4()),
            user_id=user_id,
            podcast_id=podcast_id,
            podcast_title=podcast_title,
            podcast_url=podcast_url,
            target_language=target_language,
            learner_level=learner_level,
            vocabulary=vocabulary,
            comprehension_questions=comprehension_questions,
            discussion_prompt=discussion_prompt,
            introduction=introduction,
            metadata=dict(metadata or {}),
        )


@dataclass(frozen=True)
class ChallengeResult:
    """Result of a learner completing the comprehension part."""

    challenge_id: str
    user_id: str
    score: int
    total_questions: int
    passed: bool
    vocabulary_review_required: bool
    message: str

    def __post_init__(self) -> None:
        if self.total_questions < 1:
            raise ValueError(
                "ChallengeResult total_questions must be at least 1."
            )

        if not 0 <= self.score <= self.total_questions:
            raise ValueError(
                "ChallengeResult score must be between 0 and total_questions."
            )

        if not self.message.strip():
            raise ValueError(
                "ChallengeResult message cannot be empty."
            )

"""
EasyChatBot Hackathon — Podcast Learning Challenge service.

MVP scope:
    verified podcast metadata
        -> level/language-aware challenge

It intentionally does not:
- download audio
- transcribe episodes
- infer facts from unavailable audio
- call Apple/Spotify
- write persistent memory
"""

from __future__ import annotations

from src.ai_runtime.podcast.challenge import (
    ChallengeResult,
    ComprehensionQuestion,
    PodcastLearningChallenge,
    VocabularyItem,
)
from src.ai_runtime.podcast.models import PodcastRecommendation


class PodcastLearningChallengeService:
    """Creates and evaluates small deterministic learning challenges."""

    _LEVEL_GUIDANCE = {
        "a1": "Keep the language simple and concrete.",
        "a2": "Use common vocabulary and short explanations.",
        "b1": "Use practical vocabulary and clear contextual questions.",
        "b2": "Use contextual vocabulary and inference-oriented questions.",
        "c1": "Use nuanced vocabulary and analytical questions.",
        "c2": "Use advanced nuance, register, and critical analysis.",
    }

    def create_challenge(
        self,
        *,
        recommendation: PodcastRecommendation,
    ) -> PodcastLearningChallenge:
        """
        Build a challenge from verified recommendation metadata.

        The content is intentionally deterministic for the hackathon demo.
        It does not claim to have analysed the podcast audio.
        """

        language = recommendation.result.languages[0]
        level = recommendation.result.metadata.get(
            "learner_level",
            "B2",
        )

        topic = recommendation.result.title

        vocabulary = (
            VocabularyItem(
                term="réglementation",
                meaning="Ensemble des règles qui encadrent une activité.",
                example=(
                    "La réglementation européenne encadre "
                    "l'intelligence artificielle."
                ),
            ),
            VocabularyItem(
                term="intelligence artificielle",
                meaning=(
                    "Technologies permettant à des machines "
                    "d'effectuer des tâches associées à l'intelligence humaine."
                ),
                example=(
                    "L'intelligence artificielle transforme "
                    "de nombreux secteurs."
                ),
            ),
            VocabularyItem(
                term="enjeu",
                meaning="Question importante qui présente un intérêt majeur.",
                example=(
                    "La protection des données est un enjeu important."
                ),
            ),
        )

        questions = (
            ComprehensionQuestion(
                question=(
                    f"Quel est le thème principal associé à « {topic} » ?"
                ),
                options=(
                    "L'intelligence artificielle et la réglementation",
                    "La cuisine française",
                    "Les transports urbains",
                ),
                correct_option=0,
            ),
            ComprehensionQuestion(
                question=(
                    "Pourquoi la réglementation peut-elle être importante "
                    "pour l'intelligence artificielle ?"
                ),
                options=(
                    "Pour encadrer son utilisation et ses risques",
                    "Pour supprimer toutes les technologies",
                    "Pour empêcher les utilisateurs d'apprendre",
                ),
                correct_option=0,
            ),
        )

        introduction = (
            f"Écoutez le podcast en {language}, puis revenez ici pour "
            f"votre défi {level}. Faites attention aux mots clés et aux "
            "idées principales."
        )

        discussion_prompt = (
            "Après votre écoute, expliquez en quelques phrases votre "
            "opinion sur le sujet et utilisez au moins deux nouveaux mots."
        )

        return PodcastLearningChallenge.create(
            user_id=recommendation.user_id,
            podcast_id=recommendation.result.external_id,
            podcast_title=recommendation.result.title,
            podcast_url=recommendation.result.url,
            target_language=language,
            learner_level=level,
            vocabulary=vocabulary,
            comprehension_questions=questions,
            discussion_prompt=discussion_prompt,
            introduction=introduction,
            metadata={
                "source": "podcast_recommendation",
                "provider": recommendation.result.provider.value,
                "challenge_type": "podcast_learning_mvp",
                "content_basis": "verified_podcast_metadata",
            },
        )

    def evaluate(
        self,
        *,
        challenge: PodcastLearningChallenge,
        answers: tuple[int, ...],
    ) -> ChallengeResult:
        """Evaluate comprehension answers without external side effects."""

        total = len(challenge.comprehension_questions)

        if len(answers) != total:
            raise ValueError(
                "Number of answers must equal the number of questions."
            )

        score = sum(
            answer == question.correct_option
            for answer, question in zip(
                answers,
                challenge.comprehension_questions,
            )
        )

        passed = score / total >= 0.5
        vocabulary_review_required = score < total

        if passed:
            message = (
                f"Challenge passed: {score}/{total}. "
                "Review the vocabulary and keep practising."
            )
        else:
            message = (
                f"Keep practising: {score}/{total}. "
                "Review the vocabulary and try again."
            )

        return ChallengeResult(
            challenge_id=challenge.challenge_id,
            user_id=challenge.user_id,
            score=score,
            total_questions=total,
            passed=passed,
            vocabulary_review_required=vocabulary_review_required,
            message=message,
        )

"""
EasyChatBot Hackathon — Podcast Learning Challenge presentation.
"""

from __future__ import annotations

from src.ai_runtime.podcast.challenge import (
    ChallengeResult,
    PodcastLearningChallenge,
)


class PodcastLearningChallengePresenter:
    """Render the learner-facing challenge for the hackathon demo."""

    def render(
        self,
        challenge: PodcastLearningChallenge,
    ) -> str:
        lines = [
            "",
            "=" * 72,
            "[PODCAST] PODCAST LEARNING CHALLENGE",
            "=" * 72,
            "",
            challenge.introduction,
            "",
            f"Podcast: {challenge.podcast_title}",
            f"Language: {challenge.target_language}",
            f"Level: {challenge.learner_level}",
            "",
            "[ LISTEN ]",
            f"  {challenge.podcast_url}",
            "",
            "VOCABULARY",
            "-" * 72,
        ]

        for index, item in enumerate(
            challenge.vocabulary,
            start=1,
        ):
            lines.extend(
                [
                    f"{index}. {item.term}",
                    f"   {item.meaning}",
                    f"   Example: {item.example}",
                    "",
                ]
            )

        lines.extend(
            [
                "COMPREHENSION",
                "-" * 72,
            ]
        )

        for index, question in enumerate(
            challenge.comprehension_questions,
            start=1,
        ):
            lines.append(f"{index}. {question.question}")
            for option_index, option in enumerate(
                question.options,
                start=1,
            ):
                lines.append(f"   {option_index}. {option}")
            lines.append("")

        lines.extend(
            [
                "DISCUSSION",
                "-" * 72,
                challenge.discussion_prompt,
                "",
            ]
        )

        return "\n".join(lines)

    def render_result(
        self,
        result: ChallengeResult,
    ) -> str:
        return "\n".join(
            [
                "",
                "CHALLENGE RESULT",
                "-" * 72,
                f"Score: {result.score}/{result.total_questions}",
                f"Status: {'PASS' if result.passed else 'REVIEW'}",
                result.message,
                (
                    "Vocabulary review: YES"
                    if result.vocabulary_review_required
                    else "Vocabulary review: NOT REQUIRED"
                ),
            ]
        )

from src.ai_runtime.memory.models import MemoryEntry, MemoryType
from src.ai_runtime.podcast.agent import PodcastDiscoveryAgent
from src.ai_runtime.podcast.models import LearningContext


def context() -> LearningContext:
    return LearningContext(
        world_id="French World",
        target_language="fr",
        learner_level="B2",
        learning_goal="Learn French through current affairs.",
    )


def test_agent_is_provider_agnostic_and_uses_preferences() -> None:
    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.PREFERENCE,
        content="Prefers technology podcasts.",
        metadata={
            "topics": "artificial intelligence, European regulation",
        },
    )

    intent = PodcastDiscoveryAgent().build_search_intent(
        user_id="user-001",
        learning_context=context(),
        memories=[memory],
    )

    assert intent.learning_context.target_language == "fr"
    assert intent.learning_context.world_id == "French World"
    assert "artificial intelligence" in intent.topics
    assert "European regulation" in intent.topics
    assert intent.language_required is True


def test_agent_ignores_non_preference_memory() -> None:
    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.DISCOVERY_CONTEXT,
        content="Discussed artificial intelligence.",
    )

    intent = PodcastDiscoveryAgent().build_search_intent(
        user_id="user-001",
        learning_context=context(),
        memories=[memory],
    )

    assert intent.topics == ("French World",)

class FakeResponse:
    def __init__(self, payload: dict):
        self.payload = payload

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc, tb):
        return False

    def read(self):
        return json.dumps(self.payload).encode("utf-8")

def candidate(title: str, language: str, description: str) -> PodcastSearchResult:
    return PodcastSearchResult(
        provider=PodcastProviderName.APPLE,

        description=description,

def test_wrong_language_is_rejected() -> None:
    ranked = PodcastRecommendationService.rank_candidates(
        candidates=[
            candidate(
                "AI Europe English",
                "en",
                "artificial intelligence European regulation",
            ),

                "IA Europe Français",
                "fr",
                "artificial intelligence European regulation",
            ),
        ],
        intent=intent(),
    )

                "fr",
                "artificial intelligence and European regulation",

        ],
        intent=intent(),

    assert ranked[0][0].title == "French AI Europe"
