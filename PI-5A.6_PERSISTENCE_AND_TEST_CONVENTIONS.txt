"""Infrastructure adapter for Platform Intelligence escalation email delivery."""

from __future__ import annotations

from typing import Any

from app.application.operations_center.platform_intelligence_escalation_email import (
    PlatformIntelligenceEscalationEmailRequest,
)
from cybersecurity.email.email_configuration import (
    PLATFORM_INTELLIGENCE_EMAIL_FROM,
    PLATFORM_INTELLIGENCE_ESCALATION_RECIPIENTS,
)
from cybersecurity.email.email_provider_manager import EmailProviderManager
from cybersecurity.email.resend_email_provider import ResendEmailProvider


class ResendPlatformIntelligenceEscalationEmailSender:
    """Translate governed PI email requests into the shared email provider."""

    def __init__(
        self,
        *,
        provider_manager: EmailProviderManager | None = None,
        recipients: dict[str, str] | None = None,
    ) -> None:
        self._recipients = dict(
            recipients
            if recipients is not None
            else PLATFORM_INTELLIGENCE_ESCALATION_RECIPIENTS
        )

        self._provider_manager = (
            provider_manager
            if provider_manager is not None
            else self._build_provider_manager()
        )

    @staticmethod
    def _build_provider_manager() -> EmailProviderManager:
        sender = PLATFORM_INTELLIGENCE_EMAIL_FROM.strip()

        if not sender:
            raise RuntimeError(
                "PLATFORM_INTELLIGENCE_EMAIL_FROM is not configured."
            )

        return EmailProviderManager(
            provider=ResendEmailProvider(
                sender=sender,
            ),
        )

    def send(
        self,
        request: PlatformIntelligenceEscalationEmailRequest,
        *,
        subject: str,
        body: str,
        html_body: str,
    ) -> object:
        if request is None:
            raise ValueError("request cannot be None.")

        recipient = self._recipients.get(request.route)

        if not recipient or not recipient.strip():
            raise RuntimeError(
                "No configured recipient for escalation route: "
                f"{request.route}"
            )

        if not subject.strip():
            raise ValueError("Email subject cannot be empty.")

        if not body.strip():
            raise ValueError("Email body cannot be empty.")

        if not html_body.strip():
            raise ValueError("HTML email body cannot be empty.")

        content: dict[str, Any] = {
            "recipient": recipient.strip(),
            "body": body,
            "html": html_body,
            "email_type": "platform_intelligence_escalation",
            "recommendation_id": request.recommendation_id,
            "risk_decision_id": request.risk_decision_id,
            "review_item_id": request.review_item_id,
            "escalation_route": request.route,
        }

        return self._provider_manager.send(
            subject.strip(),
            content,
        )


__all__ = [
    "ResendPlatformIntelligenceEscalationEmailSender",

        if record.occurred_at is None:

                    ORDER BY occurred_at ASC, audit_id ASC

                    """
                ),

            row = result.mappings().first()
            return None if row is None else self._to_record(row)

    async def list_retry_candidates(self, *, limit: int):
        if limit < 1:
            raise ValueError("limit must be at least 1.")
        async with self._session_factory() as session:
            result = await session.execute(

                    SELECT notification_key, recommendation_id,

                           attempt, status, provider_message_id, error_code,
                           created_at, updated_at
                    FROM {self._TABLE}
                    WHERE status IN (:failed, :started)
                    ORDER BY updated_at ASC NULLS FIRST, notification_key ASC
                    LIMIT :limit
                    """
                ),
                {
                    "failed": PlatformIntelligenceEscalationDeliveryStatus.FAILED.value,
                    "started": PlatformIntelligenceEscalationDeliveryStatus.STARTED.value,
                    "limit": limit,
                },
            )
            return tuple(self._to_record(row) for row in result.mappings().all())

    async def claim_retry_candidate(
        self, *, notification_key: str, claim_token: str,
        claimed_at: datetime, lease_expires_at: datetime,
    ) -> bool:
        if not notification_key.strip():
            raise ValueError("notification_key must be nonblank.")
        if not claim_token.strip():
            raise ValueError("claim_token must be nonblank.")
        if claimed_at.tzinfo is None or claimed_at.utcoffset() is None:
            raise ValueError("claimed_at must be timezone-aware.")
        if lease_expires_at.tzinfo is None or lease_expires_at.utcoffset() is None:
            raise ValueError("lease_expires_at must be timezone-aware.")
        if lease_expires_at <= claimed_at:
            raise ValueError("lease_expires_at must be after claimed_at.")

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    f"""
                    UPDATE {self._TABLE}
                    SET claim_token = :claim_token,
                        claimed_at = :claimed_at,
                        lease_expires_at = :lease_expires_at
                    WHERE notification_key = :notification_key
                      AND status IN (:failed, :started)
                      AND (
                          lease_expires_at IS NULL
                          OR lease_expires_at <= :claimed_at
                      )
                    """
                ),
                {
                    "notification_key": notification_key,
                    "claim_token": claim_token,
                    "claimed_at": claimed_at,
                    "lease_expires_at": lease_expires_at,
                    "failed": PlatformIntelligenceEscalationDeliveryStatus.FAILED.value,
                    "started": PlatformIntelligenceEscalationDeliveryStatus.STARTED.value,
                },
            )
            await session.commit()
            return result.rowcount == 1

                risk_decision_id=risk_decision_id,

                correlation_id=correlation_id,

            await session.commit()

        record = await self.get_by_notification_key(notification_key)
        if record is None:
            raise RuntimeError("Created PI escalation delivery could not be reloaded.")

                        updated_at = :updated_at,

        self,
        *,

        now: datetime,
    ) -> AcademyDeliveryRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                text(

                    INSERT INTO academy_notification_deliveries (
                        id,
                        user_id,
                        notification_type,
                        status,
                        attempt_count,
                        created_at,
                        updated_at
                    )
                    VALUES (
                        gen_random_uuid(),
                        :user_id,
                        :notification_type,
                        'SENDING',
                        1,
                        :now,
                        :now
                    )
                    ON CONFLICT (user_id, notification_type)
                    DO UPDATE SET
                        status = 'SENDING',
                        attempt_count =
                            academy_notification_deliveries.attempt_count + 1,
                        updated_at = EXCLUDED.updated_at
                    WHERE academy_notification_deliveries.status = 'FAILED'
                    RETURNING
                        user_id,
                        notification_type,
                        status,
                        attempt_count,
                        provider_message_id,
                        last_error
                    """
                ),
                {
                    "user_id": user_id,
                    "notification_type": notification_type,
                    "now": now,

            if row is None:
                await session.rollback()
                return None

            record = AcademyDeliveryRecord(
                user_id=row.user_id,
                notification_type=row.notification_type,
                status=AcademyDeliveryStatus(row.status),
                attempt_count=row.attempt_count,
                provider_message_id=row.provider_message_id,

class SqlAlchemyPublicationExecutionRepository(
    PublicationExecutionRepository
):
    """Durable repository using the existing injected async session factory."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    async def get_by_idempotency_key(
        self,
        idempotency_key: str,
    ) -> PublicationExecutionRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )
            model = result.scalar_one_or_none()

            return self._to_domain(model) if model else None

    async def create_started(
        self,
        *,
        publication_id,
        idempotency_key,
        correlation_id,
    ) -> PublicationExecutionRecord:
        """Create the first durable execution attempt."""

        now = datetime.now(timezone.utc)

        model = PublicationExecutionRecordModel(
            publication_id=str(publication_id),
            idempotency_key=idempotency_key,
            correlation_id=correlation_id,
            attempt=1,
            status=DurableExecutionStatus.STARTED.value,
            created_at=now,
            updated_at=now,
        )

        async with self._session_factory() as session:
            session.add(model)
            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def mark_succeeded(
        self,
        *,
        idempotency_key,
        provider_message_id,
    ) -> PublicationExecutionRecord:
        """Persist successful completion without changing the attempt count."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.SUCCEEDED.value,
            provider_message_id=provider_message_id,
            error_code=None,
        )

    async def mark_failed(
        self,
        *,
        idempotency_key,
        error_code,
    ) -> PublicationExecutionRecord:
        """Persist failure while retaining the current attempt number."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.FAILED.value,
            error_code=error_code,
        )

    async def start_retry(
        self,
        *,
        idempotency_key: str,
    ) -> PublicationExecutionRecord:
        """Increment the durable attempt counter for a retry.

        A retry is valid only for an existing FAILED execution. The same
        idempotency record is retained; no second database row is created.
        """

        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            if model.status != DurableExecutionStatus.FAILED.value:
                raise ValueError(
                    "Only failed executions can be retried."
                )

            model.attempt += 1
            model.status = DurableExecutionStatus.STARTED.value
            model.error_code = None
            model.provider_message_id = None
            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def _update(
        self,
        idempotency_key,
        **values,
    ) -> PublicationExecutionRecord:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            for key, value in values.items():
                setattr(model, key, value)

            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    @staticmethod
    def _to_domain(
        model: PublicationExecutionRecordModel,
    ) -> PublicationExecutionRecord:
        return PublicationExecutionRecord(
            publication_id=UUID(model.publication_id),
            idempotency_key=model.idempotency_key,
            correlation_id=model.correlation_id,
            attempt=model.attempt,
            status=DurableExecutionStatus(model.status),
            provider_message_id=model.provider_message_id,
            error_code=model.error_code,
            created_at=model.created_at,
            updated_at=model.updated_at,
        )

class SqlAlchemyPublicationRepository(PublicationRepository):
    """Async PostgreSQL adapter for the existing repository contract."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")
        self._session_factory = session_factory

    def _to_domain(self, model: PublicationModel) -> Publication:
        return Publication(
            publication_id=model.publication_id,
            template_id=model.template_id,
            template_version=model.template_version,
            publication_type=PublicationType(model.publication_type),
            title=model.title,
            content=model.content,
            status=PublicationStatus(model.status),
            created_at=model.created_at,
            approved_at=model.approved_at,
            published_at=model.published_at,
            failure_reason=model.failure_reason,
        )

    def _from_domain(self, publication: Publication) -> PublicationModel:
        return PublicationModel(
            publication_id=publication.publication_id,
            template_id=publication.template_id,
            template_version=publication.template_version,
            publication_type=publication.publication_type.value,
            title=publication.title,
            content=publication.content,
            status=publication.status.value,
            created_at=publication.created_at or datetime.now(timezone.utc),
            approved_at=publication.approved_at,
            published_at=publication.published_at,
            failure_reason=publication.failure_reason,
        )

    async def save_async(self, publication: Publication) -> None:
        async with self._session_factory() as session:
            model = await session.get(
                PublicationModel,
                publication.publication_id,
            )
            if model is None:
                session.add(self._from_domain(publication))
            else:
                model.template_id = publication.template_id
                model.template_version = publication.template_version
                model.publication_type = publication.publication_type.value
                model.title = publication.title
                model.content = publication.content
                model.status = publication.status.value
                model.created_at = publication.created_at or model.created_at
                model.approved_at = publication.approved_at
                model.published_at = publication.published_at
                model.failure_reason = publication.failure_reason
            await session.commit()

    async def get_async(self, publication_id: str) -> Publication | None:
        async with self._session_factory() as session:
            model = await session.get(PublicationModel, publication_id)
            return self._to_domain(model) if model else None

    async def list_all_async(self) -> tuple[Publication, ...]:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationModel).order_by(PublicationModel.created_at)
            )
            return tuple(self._to_domain(model) for model in result.scalars().all())

    # The existing repository contract is synchronous. These methods make
    # accidental async DB use explicit rather than silently blocking.
    def save(self, publication: Publication) -> None:
        raise RuntimeError(
            "SqlAlchemyPublicationRepository requires its async boundary; "
            "use save_async()."
        )

    def get(self, publication_id: str) -> Publication | None:
        raise RuntimeError(
            "SqlAlchemyPublicationRepository requires its async boundary; "
            "use get_async()."
        )

    def list_all(self) -> tuple[Publication, ...]:
        raise RuntimeError(
            "SqlAlchemyPublicationRepository requires its async boundary; "
            "use list_all_async()."
        )

class PostgreSQLAuthenticationRepository:
    """PostgreSQL adapter for the existing EasyChatBot auth tables.

    The adapter intentionally uses SQLAlchemy Core statements rather than
    importing speculative ORM models. This binds the application contract to
    the inspected database schema while keeping persistence isolated.
    """

    def __init__(self, session_factory: async_sessionmaker[AsyncSession]) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")
        self._session_factory = session_factory

    async def email_exists(self, email: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text("SELECT 1 FROM users WHERE email = :email LIMIT 1"),
                {"email": email},
            )
            return result.first() is not None

    async def username_exists(self, username: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text("SELECT 1 FROM users WHERE username = :username LIMIT 1"),
                {"username": username},
            )
            return result.first() is not None

    async def create_user(self, *, user_id: UUID, email: str, username: str) -> None:
        now = datetime.now(timezone.utc)
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO users
                            (id, email, username, status, trial_status, created_at,
                             email_verified)
                        VALUES
                            (:id, :email, :username, 'active', 'not_started', :created_at,
                             FALSE)
                        """
                    ),
                    {
                        "id": user_id,
                        "email": email,
                        "username": username,
                        "created_at": now,
                    },
                )
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_user_roles (user_id, role)
                        VALUES (:user_id, 'beta_user')
                        ON CONFLICT DO NOTHING
                        """
                    ),
                    {"user_id": user_id},
                )

    async def create_password_credential(self, *, user_id: UUID, password_hash: str) -> None:
        now = datetime.now(timezone.utc)
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO user_credentials
                            (id, user_id, password_hash, credential_type,
                             created_at, updated_at)
                        VALUES
                            (:id, :user_id, :password_hash, 'PASSWORD',
                             :created_at, :updated_at)
                        """
                    ),
                    {
                        "id": uuid4(),
                        "user_id": user_id,
                        "password_hash": password_hash,
                        "created_at": now,
                        "updated_at": now,
                    },
                )

    async def create_email_verification(self, *, user_id: UUID, email: str) -> str:
        """Create a one-time verification token.

        The raw token is returned to the application boundary. Only its SHA-256
        digest is persisted. Delivery is deliberately outside this repository.
        """
        raw_token = secrets.token_urlsafe(32)
        token_hash = hashlib.sha256(raw_token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)
        verification_id = uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_email_verification_tokens
                            (id, user_id, email, token_hash, expires_at, created_at)
                        VALUES
                            (:id, :user_id, :email, :token_hash,
                             :expires_at, :created_at)
                        """
                    ),
                    {
                        "id": verification_id,
                        "user_id": user_id,
                        "email": email,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=24),
                        "created_at": now,
                    },
                )
        return raw_token


    async def create_password_reset_token(
        self,
        *,
        user_id: UUID,
    ) -> str:
        """Create a secure one-time password-reset token.

        Only the SHA-256 digest is persisted. The raw token is returned
        exclusively to the application layer for email delivery.
        """
        raw_token = secrets.token_urlsafe(32)

        token_hash = hashlib.sha256(
            raw_token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)
        reset_id = uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                # A newly issued reset link supersedes older unused links.
                await session.execute(
                    text(
                        """
                        UPDATE auth_password_reset_tokens
                        SET used_at = :now
                        WHERE user_id = :user_id
                          AND used_at IS NULL
                        """
                    ),
                    {
                        "now": now,
                        "user_id": user_id,
                    },
                )

                await session.execute(
                    text(
                        """
                        INSERT INTO auth_password_reset_tokens
                            (
                                id,
                                user_id,
                                token_hash,
                                expires_at,
                                created_at
                            )
                        VALUES
                            (
                                :id,
                                :user_id,
                                :token_hash,
                                :expires_at,
                                :created_at
                            )
                        """
                    ),
                    {
                        "id": reset_id,
                        "user_id": user_id,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=1),
                        "created_at": now,
                    },
                )

        return raw_token

    async def count_recent_password_reset_tokens(
        self,
        *,
        user_id: UUID,
        since: datetime,
    ) -> int:
        """Count password-reset requests in the supplied time window."""
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT COUNT(*)
                    FROM auth_password_reset_tokens
                    WHERE user_id = :user_id
                      AND created_at >= :since
                    """
                ),
                {
                    "user_id": user_id,
                    "since": since,
                },
            )

            return int(result.scalar_one())

    async def get_password_reset_context(
        self,
        *,
        token: str,
    ) -> PasswordResetContextRecord | None:
        """Resolve a valid reset token to non-sensitive account context.

        The raw token is never persisted or returned. Only a valid, unused,
        unexpired token can resolve its account email and current credential
        hash. The credential hash is consumed only by the application service
        for the previous-password check and is never exposed by the HTTP API.
        """
        token_hash = hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()
        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT t.user_id, u.email, c.password_hash
                    FROM auth_password_reset_tokens AS t
                    INNER JOIN users AS u
                        ON u.id = t.user_id
                    INNER JOIN user_credentials AS c
                        ON c.user_id = t.user_id
                       AND c.credential_type = 'PASSWORD'
                    WHERE t.token_hash = :token_hash
                      AND t.used_at IS NULL
                      AND t.expires_at > :now
                    LIMIT 1
                    """
                ),
                {
                    "token_hash": token_hash,
                    "now": now,
                },
            )
            row = result.mappings().first()

            if row is None:
                return None

            return PasswordResetContextRecord(
                user_id=row["user_id"],
                email=row["email"],
                password_hash=row["password_hash"],
            )

    async def get_password_reset_user(
        self,
        token: str,
    ) -> PasswordResetContextRecord | None:
        """Backward-compatible repository name for password-reset context.

        The reset flow historically exposed this operation as
        ``get_password_reset_user``. Keep that contract as a thin delegate to
        the canonical context method so callers/tests do not need a second
        implementation or a duplicate SQL query.
        """
        return await self.get_password_reset_context(token=token)

    async def reset_password(
        self,
        *,
        token: str,
        password_hash: str,
        expected_password_hash: str | None = None,
    ) -> bool:
        """Atomically consume a reset token, update the password,
        and revoke existing sessions.
        """
        token_hash = hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    text(
                        """
                        SELECT id, user_id
                        FROM auth_password_reset_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now
                        FOR UPDATE
                        """
                    ),
                    {
                        "token_hash": token_hash,
                        "now": now,
                    },
                )

                row = result.mappings().first()

                if row is None:
                    return False

                user_id = row["user_id"]

                credential_result = await session.execute(
                    text(
                        """
                        SELECT id
                        FROM user_credentials
                        WHERE user_id = :user_id
                          AND credential_type = 'PASSWORD'
                        LIMIT 1
                        """
                    ),
                    {
                        "user_id": user_id,
                    },
                )

                credential = credential_result.mappings().first()

                if credential is None:
                    return False

                # Keep the optimistic-concurrency predicate out of a nullable
                # PostgreSQL bind expression.  asyncpg/PostgreSQL can otherwise
                # fail to infer the parameter type for ``:expected_password_hash
                # IS NULL`` on some database/driver combinations.  Branching
                # here also makes the two repository contracts explicit: a
                # normal reset and a reset guarded by the previously observed
                # password hash.
                if expected_password_hash is None:
                    update_result = await session.execute(
                        text(
                            """
                            UPDATE user_credentials
                            SET password_hash = :password_hash,
                                updated_at = :updated_at
                            WHERE id = :credential_id
                            """
                        ),
                        {
                            "password_hash": password_hash,
                            "updated_at": now,
                            "credential_id": credential["id"],
                        },
                    )
                else:
                    update_result = await session.execute(
                        text(
                            """
                            UPDATE user_credentials
                            SET password_hash = :password_hash,
                                updated_at = :updated_at
                            WHERE id = :credential_id
                              AND password_hash = :expected_password_hash
                            """
                        ),
                        {
                            "password_hash": password_hash,
                            "updated_at": now,
                            "credential_id": credential["id"],
                            "expected_password_hash": expected_password_hash,
                        },
                    )

                if update_result.rowcount != 1:
                    return False

                await session.execute(
                    text(
                        """
                        UPDATE auth_password_reset_tokens
                        SET used_at = :used_at
                        WHERE id = :id
                        """
                    ),
                    {
                        "used_at": now,
                        "id": row["id"],
                    },
                )

                await session.execute(
                    text(
                        """
                        UPDATE user_sessions
                        SET is_revoked = TRUE,
                            status = 'revoked'
                        WHERE user_id = :user_id
                          AND is_revoked = FALSE
                        """
                    ),
                    {
                        "user_id": user_id,
                    },
                )

                return True
    async def verify_email(self, token: str) -> bool:
        token_hash = hashlib.sha256(token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    text(
                        """
                        SELECT id, user_id
                        FROM auth_email_verification_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now
                        FOR UPDATE
                        """
                    ),
                    {"token_hash": token_hash, "now": now},
                )
                row = result.mappings().first()
                if row is None:
                    return False

                await session.execute(
                    text(
                        "UPDATE users SET email_verified = TRUE WHERE id = :user_id"
                    ),
                    {"user_id": row["user_id"]},
                )
                await session.execute(
                    text(
                        "UPDATE auth_email_verification_tokens SET used_at = :now WHERE id = :id"
                    ),
                    {"now": now, "id": row["id"]},
                )
                return True

    async def create_session(
        self,
        *,
        session_id: UUID,
        user_id: UUID,
        created_at: datetime,
        expires_at: datetime,
        device_name: str | None,

                return None


            row = session.get(
                BetaUserRecord,
                binding.user_id,
            )


            if row is None:
                return None


            user = _user_from_record(row)


            #
            # Telegram binding is authoritative.
            #
            # Do not depend on beta_users.telegram_id.
            #
            return BetaUserState(
                user_id=user.user_id,
                email=user.email,
                email_verified=user.email_verified,
                beta_approved=user.beta_approved,
                telegram_id=int(
                    binding.telegram_user_id
                ),
                active_world_id=user.active_world_id,
            )


    async def save(
        self,
        user: BetaUserState,
    ) -> BetaUserState:

        now = _utcnow()

        with self._session_factory() as session:

            row = session.get(
                BetaUserRecord,
                user.user_id,
            )


            if row is None:

                row = BetaUserRecord(
                    id=user.user_id,
                    email=user.email,
                    email_verified=user.email_verified,
                    approved=user.beta_approved,
                    active_world_id=user.active_world_id,
                    telegram_id=(
                        None
                        if user.telegram_id is None
                        else str(user.telegram_id)
                    ),
                    created_at=now,
                    updated_at=now,
                )

                session.add(row)


            else:

                row.email = user.email
                row.email_verified = user.email_verified
                row.approved = user.beta_approved
                row.active_world_id = user.active_world_id

                row.telegram_id = (
                    None
                    if user.telegram_id is None
                    else str(user.telegram_id)
                )

                row.updated_at = now


            session.commit()


        return user



class PostgreSQLBetaTokenRepository(
    BetaTokenRepository
):

    def __init__(
        self,
        session_factory,
    ):
        self._session_factory = session_factory


    async def get_by_hash(
        self,
        token_hash: str,
    ) -> BetaConnectionToken | None:

        with self._session_factory() as session:

            row = session.scalar(
                select(BetaConnectionTokenRecord).where(
                    BetaConnectionTokenRecord.token_hash
                    == token_hash
                )
            )

            return (
                None
                if row is None
                else _token_from_record(row)
            )


    async def save(
        self,
        token: BetaConnectionToken,
    ) -> BetaConnectionToken:

        now = _utcnow()

        with self._session_factory() as session:

            row = session.get(
                BetaConnectionTokenRecord,
                token.token_id,
            )


            if row is None:

                session.add(
                    BetaConnectionTokenRecord(
                        id=token.token_id,
                        token_hash=token.token_hash,
                        user_id=token.user_id,
                        expires_at=token.expires_at,
                        used_at=token.used_at,
                        created_at=now,
                    )
                )

            else:

                row.used_at = token.used_at


            session.commit()


        return token


    async def mark_used(
        self,
        token: BetaConnectionToken,
    ) -> BetaConnectionToken:

        return await self.save(token)

class Discovery:
    def __init__(self, candidates) -> None:
        self.candidates = tuple(candidates)
        self.calls = 0
        self.limit = None

    async def discover(self, *, limit: int):
        self.calls += 1
        self.limit = limit
        return PlatformIntelligenceEscalationRetryCandidateBatch(
            candidates=self.candidates,
            limit=limit,
        )


class Orchestrator:
    def __init__(self, results=None, errors=None) -> None:
        self.results = results or {}
        self.errors = errors or {}
        self.calls = []

    async def run_candidate(self, *, candidate, reference_at):
        self.calls.append((candidate.notification_key, reference_at))
        if candidate.notification_key in self.errors:
            raise self.errors[candidate.notification_key]
        return self.results[candidate.notification_key]
