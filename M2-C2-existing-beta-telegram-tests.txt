async def test_telegram_identity_cannot_be_owned_by_two_users():

    tokens = InMemoryBetaTokenRepository()
    bindings = InMemoryTelegramBindingRepository()

        user_repository=users,
        token_repository=tokens,
        binding_repository=bindings,
        world_resolver=create_world_resolver(),
    )

    for user_id in ("a", "b"):
        await users.save(
            BetaUserState(
                user_id=user_id,
                email=f"{user_id}@example.com",
                email_verified=True,
                beta_approved=True,
                active_world_id="world-fr",
            )
        )

    first = await service.create_connection_token("a")

    await service.validate_and_bind(
        token=first.token,
        telegram_id=111,
        chat_id=111,
    )

    second = await service.create_connection_token("b")

    with pytest.raises(ValueError, match="already connected"):
        await service.validate_and_bind(
            token=second.token,
            telegram_id=111,
            chat_id=111,
        )

from decimal import Decimal

import pytest

from app.ai_runtime.conversation.conversation_runtime import ConversationRuntime
from app.application.providers.provider_usage_capture import (
    ProviderUsageCapture,
    ProviderUsageContext,
)
from app.providers.llm.llm_provider import LLMProvider
from app.providers.provider_result import ProviderResult, ProviderUsage
from src.ai_runtime.providers.usage.repository import InMemoryProviderUsageRepository
from src.ai_runtime.providers.usage.service import ProviderUsageService


class FakeProvider(LLMProvider):
    def __init__(self, result: ProviderResult):
        self.result = result

    async def generate(self, *, system_prompt: str, user_message: str):
        return self.result


def _capture():
    repository = InMemoryProviderUsageRepository()
    service = ProviderUsageService(repository)
    return ProviderUsageCapture(service), repository


@pytest.mark.asyncio
async def test_runtime_captures_successful_provider_result():
    capture, repository = _capture()
    result = ProviderResult(
        success=True,
        provider_id="openai",
        model="gpt-5-mini",
        content="Bonjour",
        usage=ProviderUsage(input_tokens=10, output_tokens=5, total_tokens=15),
        provider_request_id="req_test_1",
        latency_ms=120,
    )
    runtime = ConversationRuntime(
        llm_provider=FakeProvider(result),
        usage_capture=capture,
        usage_context=ProviderUsageContext(
            capability="conversation",
            functionality="chat",
            agent="conversation_agent",
            estimated_cost=Decimal("0"),
        ),
    )

    assert await runtime.respond(
        character_instruction="You are a French teacher.",
        learner_message="Bonjour",
    ) == "Bonjour"

    events = repository.get_all()
    assert len(events) == 1
    assert events[0].provider == "openai"
    assert events[0].input_tokens == 10
    assert events[0].output_tokens == 5
    assert events[0].success is True


@pytest.mark.asyncio
async def test_runtime_captures_failed_provider_result_before_raising():
    capture, repository = _capture()
    result = ProviderResult(
        success=False,
        provider_id="openai",
        model="gpt-5-mini",
        error="provider unavailable",
        provider_request_id="req_test_2",
    )
    runtime = ConversationRuntime(

    assert events[0].success is False

        effective_user=FakeUser(111),

        update,

    assert call["telegram_user_id"] == 111
    assert call["chat_id"] == 111
    assert call["chat_type"] == "private"

    # Telegram transport owns reply delivery.
    # The adapter passes an explicit reply callback
    # into the start-handler boundary.
    assert "reply" in call
    assert callable(call["reply"])

    def __init__(self, availability: str = "enabled") -> None:
        self.availability = availability
        self.calls: list[dict[str, Any]] = []

    def get_availability(
        self,
        *,
        environment: str,
        functionality_id: str,
    ) -> str:
        self.calls.append(
            {
                "environment": environment,
                "functionality_id": functionality_id,
            }
        )
        return self.availability


def create_boundary(
    *,
    availability: str = "enabled",
) -> tuple[
    BetaTelegramConversationBoundary,
    FakeFunctionalityControlService,

        availability=availability,
    )

    beta_access_service = MagicMock(name="beta_access_service")

    boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation_service,
        beta_access_service=beta_access_service,
        functionality_control_service=control_service,
    )

    return boundary, control_service

        self.events.append(event)
        return event

    async def get_recent(
        self,
        *,
        telegram_user_hash: str,
        conversation_id: str,
        world_id: str,
        limit: int = 20,
    ):
        self.get_recent_calls.append(
            {
                "telegram_user_hash": telegram_user_hash,
                "conversation_id": conversation_id,
                "world_id": world_id,
                "limit": limit,
            }
        )

        if limit <= 0:
            return []

        events = [
            event
            for event in self.events
            if (
                event.telegram_user_hash
                == telegram_user_hash
                and event.conversation_id
                == conversation_id
                and event.world_id
                == world_id
            )
        ]

        return events[-limit:]

        self,

class FakeConversationService:
    """
    Deterministic WS-08 conversation/session double.

    Conversation state is deliberately scoped by Telegram
    identity so the integration tests can detect accidental
    cross-user state sharing.
    """

    def __init__(self) -> None:
        self.calls: list[dict] = []
        self.sessions: dict[int, list[str]] = {}

    async def handle_telegram_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        self.calls.append(
            {
                "telegram_user_id": telegram_user_id,
                "message": message,
            }
        )

        history = self.sessions.setdefault(
            telegram_user_id,
            [],
        )

        history.append(message)

        return (
            f"response:{telegram_user_id}:"
            f"{len(history)}"
        )

    async def get_recent_history(
        self,
        *,
        telegram_user_id: int,
        limit: int = 20,
    ) -> list[str]:
        history = self.sessions.get(
            telegram_user_id,
            [],
        )

        return history[-limit:]

    from the conversation service and from Telegram transport.
    """

class FakeObservability:
    """
    Deterministic WS-13 observability/audit event sink.

    Events intentionally contain only operational metadata. Raw
    learner message content is not accepted by the event contract.
    """

    def __init__(
        self,
        *,
        error: Exception | None = None,
    ) -> None:
        self.error = error
        self.events: list[dict] = []

    async def record(
        self,
        *,
        event_type: str,
        telegram_user_id: int,
        outcome: str,
    ) -> None:
        if self.error is not None:
            raise self.error

        self.events.append(
            {
                "event_type": event_type,
                "telegram_user_id": telegram_user_id,
                "outcome": outcome,
            }
        )


class ObservableBoundary:
    """
    Minimal WS-13 integration composition.

    Observability is deliberately separated from the conversation
    service. An observability failure must not become a classroom
    application failure.
    """

    SAFE_ERROR_RESPONSE = (
        "Sorry, something went wrong while processing your message. "
        "Please try again later."
    )

    UNAUTHORIZED_RESPONSE = (
        "Sorry, this Telegram account is not authorized "
        "for Beta access."
    )

    def __init__(
        self,
        *,
        conversation_service: FakeConversationService,
        beta_access_service: FakeBetaAccessService,
        observability: FakeObservability,
    ) -> None:
        self._conversation_service = conversation_service
        self._beta_access_service = beta_access_service
        self._observability = observability

    async def _record(
        self,
        *,
        event_type: str,
        telegram_user_id: int,
        outcome: str,
    ) -> None:
        try:
            await self._observability.record(
                event_type=event_type,
                telegram_user_id=telegram_user_id,
                outcome=outcome,
            )
        except Exception:
            # Observability is non-blocking infrastructure. Failure
            # must never break the classroom flow.
            return

    async def handle_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        if isinstance(telegram_user_id, bool):
            raise ValueError(
                "Telegram user ID must be a positive integer."
            )

        if not isinstance(telegram_user_id, int):
            raise TypeError(
                "Telegram user ID must be a positive integer."
            )

        if telegram_user_id <= 0:
            raise ValueError(
                "Telegram user ID must be a positive integer."
            )

        if message is None:
            raise ValueError(
                "Telegram message cannot be empty."
            )

        if not isinstance(message, str):
            raise TypeError(
                "Telegram message must be a string."
            )

        if not message.strip():
            raise ValueError(
                "Telegram message cannot be empty."
            )

        clean_message = message.strip()

        if len(clean_message) > 4096:
            raise ValueError(
                "Telegram message is too long. "
                "Maximum length is 4096 characters."
            )

        try:
            context = await (
                self._beta_access_service
                .get_conversation_context(
                    telegram_user_id
                )
            )
        except PermissionError:
            await self._record(
                event_type="telegram_classroom_authorization",
                telegram_user_id=telegram_user_id,
                outcome="denied",
            )

            return self.UNAUTHORIZED_RESPONSE

        # Context is operational metadata and is never injected into
        # the learner message or observability event content.
        _ = context

        try:
            response = await (
                self._conversation_service
                .handle_telegram_message(
                    telegram_user_id=telegram_user_id,
                    message=clean_message,
                )
            )
        except Exception:
            await self._record(
                event_type="telegram_classroom_conversation",
                telegram_user_id=telegram_user_id,
                outcome="error",
            )

            return self.SAFE_ERROR_RESPONSE

        await self._record(
            event_type="telegram_classroom_conversation",
            telegram_user_id=telegram_user_id,
            outcome="success",
        )

        return response


def make_boundary(
    *,
    approved: bool = True,
    conversation_error: Exception | None = None,
    observability_error: Exception | None = None,
):
    access = FakeBetaAccessService(

    )

    conversation_service = FakeConversationService(
        error=conversation_error,
    )

    )

    boundary = ObservableBoundary(
        conversation_service=conversation_service,
        beta_access_service=access,
        observability=observability,
    )

    return (
        boundary,
        access,
        conversation_service,
        observability,
    )

    def validate(
        policy: TelegramClassroomPolicy,
    ) -> None:
        if not isinstance(policy, TelegramClassroomPolicy):
            raise TypeError(
                "Telegram classroom policy must be a "
                "TelegramClassroomPolicy."
            )

        if not isinstance(policy.max_message_length, int):
            raise TypeError(
                "max_message_length must be an integer."
            )

        if policy.max_message_length <= 0:
            raise ValueError(
                "max_message_length must be greater than zero."
            )

        if not isinstance(policy.max_messages_per_user, int):
            raise TypeError(
                "max_messages_per_user must be an integer."
            )

        if policy.max_messages_per_user <= 0:
            raise ValueError(
                "max_messages_per_user must be greater than zero."
            )

        if not isinstance(policy.enabled, bool):
            raise TypeError(
                "enabled must be a boolean."
            )


class FakePolicyEnforcer:
    """
    Deterministic policy enforcer.

    State is explicitly scoped by Telegram identity.
    """

    def __init__(
        self,
        *,
        policy: TelegramClassroomPolicy,
    ) -> None:
        TelegramClassroomPolicyValidator.validate(policy)

        self.policy = policy
        self.calls: list[int] = []
        self._counts: dict[int, int] = {}

    async def allow(
        self,
        *,
        telegram_user_id: int,
    ) -> bool:
        self.calls.append(telegram_user_id)

        if not self.policy.enabled:
            return False

        count = self._counts.get(
            telegram_user_id,
            0,
        )

        if count >= self.policy.max_messages_per_user:
            return False

        self._counts[telegram_user_id] = count + 1
        return True


class PolicyControlledBoundary:
    """
    Minimal WS-14 integration composition.

    Policy validation and enforcement are deliberately kept outside
    the Telegram transport adapter and conversation service.
    """

    POLICY_DISABLED_RESPONSE = (
        "Sorry, the classroom is temporarily unavailable."
    )

    POLICY_LIMIT_RESPONSE = (
        "Sorry, you have reached the classroom message limit. "
        "Please try again later."
    )

    def __init__(
        self,
        *,
        conversation_service: FakeConversationService,
        beta_access_service: FakeBetaAccessService,
        policy: TelegramClassroomPolicy,
    ) -> None:
        TelegramClassroomPolicyValidator.validate(policy)

        self.policy = policy
        self._conversation_service = conversation_service
        self._beta_access_service = beta_access_service
        self._policy_enforcer = FakePolicyEnforcer(
            policy=policy,
        )

        self._conversation_boundary = BetaTelegramConversationBoundary(
            conversation_service=conversation_service,
            beta_access_service=beta_access_service,
        )

    async def handle_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        # Reuse the established WS-10 input validation contract before
        # consuming any configured classroom allowance.
        BetaTelegramConversationBoundary._validate_telegram_user_id(
            telegram_user_id,
        )

        clean_message = (
            BetaTelegramConversationBoundary._validate_message(
                message,
            )
        )

        if not self.policy.enabled:
            return self.POLICY_DISABLED_RESPONSE

        if len(clean_message) > self.policy.max_message_length:
            raise ValueError(
                "Telegram message is too long for the configured "
                "classroom policy."
            )

        if not await self._policy_enforcer.allow(
            telegram_user_id=telegram_user_id,
        ):
            return self.POLICY_LIMIT_RESPONSE

        return await self._conversation_boundary.handle_message(
            telegram_user_id=telegram_user_id,
            message=clean_message,
        )


def make_boundary(
    *,
    approved: bool = True,
    policy: TelegramClassroomPolicy | None = None,
):
    if policy is None:
        policy = TelegramClassroomPolicy()

    access = FakeBetaAccessService(
        approved=approved,
    )
    conversation_service = FakeConversationService()

    boundary = PolicyControlledBoundary(
        conversation_service=conversation_service,
        beta_access_service=access,
        policy=policy,
    )

    return (
        boundary,
        access,
        conversation_service,
    )

def _find_existing_telegram_identities(
    *,
    engine,
    minimum: int = 1,
) -> list[dict[str, str]]:
    """
    Retrieve persisted Telegram classroom identities from PostgreSQL.
    """

    columns = _conversation_event_columns(
        engine=engine,
    )

    required = {
        "telegram_user_hash",
        "conversation_id",
        "world_id",
    }

    missing = required - columns

    if missing:
        raise AssertionError(
            "conversation_events is missing required "
            f"columns: {sorted(missing)}"
        )

    with engine.connect() as connection:
        rows = connection.execute(
            text(
                """
                SELECT DISTINCT
                    telegram_user_hash,
                    conversation_id,
                    world_id
                FROM conversation_events
                WHERE telegram_user_hash IS NOT NULL
                  AND world_id IS NOT NULL
                ORDER BY telegram_user_hash
                """
            )
        ).mappings().all()

    identities = [
        {
            "telegram_user_hash": row[
                "telegram_user_hash"
            ],
            "conversation_id": row[
                "conversation_id"
            ],
            "world_id": row[
                "world_id"
            ],
        }
        for row in rows
    ]

    if len(identities) < minimum:
        raise RuntimeError(
            "WS-19 requires at least "
            f"{minimum} persisted Telegram identity "
            "in PostgreSQL."
        )

    return identities

@dataclass(frozen=True)
class FakeConversationContext:
    world_id: str = "world-fr"
    name: str = "French World"
    language_code: str = "fr"
    character: str = "Alix"
    level: str = "B1"


class FakeBetaAccessService:
    """Deterministic authorization/context double."""

    def __init__(
        self,
        *,
        context: FakeConversationContext | None = None,
        error: Exception | None = None,
    ) -> None:
        self.context = context or FakeConversationContext()
        self.error = error
        self.calls: list[int] = []

    async def get_conversation_context(
        self,
        telegram_user_id: int,
    ) -> FakeConversationContext:
        self.calls.append(telegram_user_id)

        if self.error is not None:
            raise self.error

        return self.context


class FakeConversationService:
    """Deterministic conversation-service double."""

    def __init__(
        self,
        *,
        response: str = "Bonjour !",
        error: Exception | None = None,
    ) -> None:
        self.response = response
        self.error = error
        self.calls: list[dict[str, Any]] = []

    async def handle_telegram_message_with_context(
        self,
        *,
        telegram_user_id: int,
        message: str,
        context: FakeConversationContext,
    ) -> str:
        self.calls.append(
            {
                "telegram_user_id": telegram_user_id,
                "message": message,
                "context": context,
            }
        )

        if self.error is not None:
            raise self.error

        return self.response


class FakeObservability:
    """Deterministic observability double."""

    def __init__(
        self,
        *,
        error: Exception | None = None,
    ) -> None:
        self.error = error
        self.events: list[dict[str, Any]] = []

    async def record(self, **event: Any) -> None:
        self.events.append(event)

        if self.error is not None:
            raise self.error


class FakePolicy:
    """Deterministic policy double."""

    def __init__(
        self,
        *,
        allowed: bool = True,
        error: Exception | None = None,
    ) -> None:
        self.allowed = allowed
        self.error = error
        self.calls: list[int] = []

    async def allow(
        self,
        *,
        telegram_user_id: int,
    ) -> bool:
        self.calls.append(telegram_user_id)

        if self.error is not None:
            raise self.error

        return self.allowed


# ============================================================================
# Boundary factory
# ============================================================================


def make_boundary(
    *,
    access_error: Exception | None = None,
    conversation_error: Exception | None = None,
    observability_error: Exception | None = None,
    rate_limiter: Any | None = None,
    abuse_protector: Any | None = None,
):
    access = FakeBetaAccessService(
        error=access_error,
    )

    conversation = FakeConversationService(
        error=conversation_error,
    )

    observability = (
        FakeObservability(
            error=observability_error,
        )
        if observability_error is not None
        else None
    )

    boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation,
        beta_access_service=access,
        rate_limiter=rate_limiter,
        abuse_protector=abuse_protector,
        observability=observability,
    )

    return (
        boundary,
        access,
        conversation,
        observability,
    )

class RecoverableAccessService:
    """Beta-access dependency that can be toggled between failure/healthy."""

    def __init__(self) -> None:
        self.available = False
        self.calls: list[int] = []

    async def get_conversation_context(
        self,
        telegram_user_id: int,
    ) -> FakeConversationContext:
        self.calls.append(telegram_user_id)

        if not self.available:
            raise RuntimeError(
                "temporary PostgreSQL connection failure"
            )

        return FakeConversationContext()


class RecoverableConversationService:
    """Conversation/AI dependency that can recover between requests."""

    def __init__(self) -> None:
        self.available = False
        self.calls: list[dict[str, Any]] = []
        self.history: dict[int, list[str]] = {}

    async def handle_telegram_message_with_context(
        self,
        *,
        telegram_user_id: int,
        message: str,
        context: FakeConversationContext,
    ) -> str:
        self.calls.append(
            {
                "telegram_user_id": telegram_user_id,
                "message": message,
                "context": context,
            }
        )

        if not self.available:
            raise RuntimeError(
                "temporary AI provider failure"
            )

        user_history = self.history.setdefault(
            telegram_user_id,
            [],
        )
        user_history.append(message)

        return (
            f"response:{telegram_user_id}:"
            f"{len(user_history)}"
        )


class RecoverablePersistenceConversationService:
    """
    Conversation double where persistence can fail once and then recover.

    The failed request is deliberately not added to history.
    """

    def __init__(self) -> None:
        self.persistence_available = False
        self.calls: list[dict[str, Any]] = []
        self.history: dict[int, list[str]] = []

    async def handle_telegram_message_with_context(
        self,
        *,
        telegram_user_id: int,
        message: str,
        context: FakeConversationContext,
    ) -> str:
        self.calls.append(
            {
                "telegram_user_id": telegram_user_id,
                "message": message,
                "context": context,
            }
        )

        if not self.persistence_available:
            raise RuntimeError(
                "temporary conversation persistence failure"
            )

        self.history.append(message)

        return f"response:{telegram_user_id}:{len(self.history)}"


class RecordingObservability:
    def __init__(self) -> None:
        self.events: list[dict[str, Any]] = []

    async def record(self, **event: Any) -> None:
        self.events.append(event)

def make_boundary(
    *,
    access: Any | None = None,
    conversation: Any | None = None,
    observability: Any | None = None,
    rate_limiter: Any | None = None,
    abuse_protector: Any | None = None,
):
    resolved_access = access or RecoverableAccessService()
    resolved_conversation = (
        conversation or RecoverableConversationService()
    )

    boundary = BetaTelegramConversationBoundary(
        conversation_service=resolved_conversation,
        beta_access_service=resolved_access,
        rate_limiter=rate_limiter,
        abuse_protector=abuse_protector,
        observability=observability,
    )

    return (
        boundary,
        resolved_access,
        resolved_conversation,
    )
