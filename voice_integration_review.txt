===== livekit_adapter.py =====
"""
LiveKit Infrastructure Adapter.

Responsible for communication with LiveKit.

This layer is infrastructure only.

Responsibilities:
- Prepare LiveKit connection data.
- Abstract LiveKit provider details.
- Create room connection metadata.

Does NOT:
- Decide when a call starts.
- Manage learner logic.
- Execute AI agents.
- Store memory.
"""

from __future__ import annotations

import logging

from app.domain.uai.voice.voice_session import (
    VoiceSession,
)


logger = logging.getLogger(__name__)


class LiveKitAdapter:
    """
    Infrastructure adapter for LiveKit.

    Beta 1 implementation:
    - Provider abstraction only.
    - Real API connection added later.
    """

    def create_room_configuration(
        self,
        *,
        session: VoiceSession,
    ) -> dict[str, str]:
        """
        Prepare LiveKit room configuration.

        Parameters
        ----------
        session:
            Voice session information.

        Returns
        -------
        dict
            Provider-independent room data.
        """

        logger.info(
            "Preparing LiveKit room configuration",
            extra={
                "session_id": session.session_id,
                "room_name": session.room_name,
            },
        )


        return {
            "room_name": (
                session.room_name
            ),
            "character_id": (
                session.character_id
            ),
            "language_code": (
                session.language_code
            ),
            "provider": "livekit",
        }

===== test_livekit_adapter.py =====
"""
Tests for LiveKit Infrastructure Adapter.
"""

from app.domain.uai.voice.voice_session import (
    VoiceSession,
)

from app.infrastructure.voice.livekit_adapter import (
    LiveKitAdapter,
)


def test_create_room_configuration():

    adapter = LiveKitAdapter()


    session = VoiceSession(
        session_id="session-001",
        room_name="yeti-session-001",
        character_id="yeti",
        language_code="fr",
    )


    configuration = (
        adapter.create_room_configuration(
            session=session,
        )
    )


    assert configuration["room_name"] == (
        "yeti-session-001"
    )


    assert configuration["character_id"] == (
        "yeti"
    )


    assert configuration["language_code"] == (
        "fr"
    )


    assert configuration["provider"] == (
        "livekit"
    )

===== voice_session.py =====
"""
Voice Session Domain Model.

Represents one character voice interaction.

Does NOT:
- Connect to LiveKit.
- Generate tokens.
- Execute AI.
- Handle audio.

It is only the domain state.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(
    frozen=True,
    slots=True,
)
class VoiceSession:
    """
    Immutable voice session state.
    """

    session_id: str

    room_name: str

    character_id: str

    language_code: str

    status: str = "created"


    def __post_init__(self) -> None:
        """
        Validate session state.
        """

        if not self.session_id.strip():
            raise ValueError(
                "session_id cannot be empty."
            )

        if not self.room_name.strip():
            raise ValueError(
                "room_name cannot be empty."
            )

        if not self.character_id.strip():
            raise ValueError(
                "character_id cannot be empty."
            )

        if not self.language_code.strip():
            raise ValueError(
                "language_code cannot be empty."
            )

===== voice_character_session_service.py =====
"""
Voice Character Session Service.

Application service responsible for
creating character voice sessions.

Responsibilities:
- Coordinate character runtime.
- Coordinate speech runtime.
- Request voice session creation.

Does NOT:
- Call speech providers.
- Manage LiveKit directly.
- Store configuration.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.ai_runtime.voice.voice_character_runtime import (
    VoiceCharacterRuntime,
)

from app.ai_runtime.voice.voice_speech_runtime import (
    VoiceSpeechRuntime,
)

from app.domain.uai.context.interaction_context import (
    InteractionContext,
)

from app.domain.uai.voice.voice_session import (
    VoiceSession,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCharacterSessionService:
    """
    Creates voice sessions for AI characters.
    """

    livekit_adapter: object

    speech_runtime: VoiceSpeechRuntime



    def create_session(
        self,
        *,
        context: InteractionContext,
    ) -> VoiceSession:
        """
        Create a character voice session.
        """


        character_runtime = (
            VoiceCharacterRuntime(
                context=context,
                speech_runtime=(
                    self.speech_runtime
                ),
            )
        )


        return (
            self.livekit_adapter
            .create_session(
                character_id=(
                    character_runtime
                    .character_id()
                ),
                language_code=(
                    context.voice.language_code
                ),
            )
        )
