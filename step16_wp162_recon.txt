"""
AI Runtime capability errors.

These exceptions provide explicit Runtime-level failure semantics.

The Runtime must distinguish between:

- invalid requests
- unknown capabilities
- duplicate capability registration
- governance rejection
- capability execution failure

Provider-specific exceptions must not leak through the Runtime
boundary.
"""

from __future__ import annotations


class RuntimeCapabilityError(Exception):
    """
    Base exception for Runtime capability failures.
    """


class InvalidCapabilityRequestError(
    RuntimeCapabilityError
):
    """
    Raised when a capability request is invalid.
    """


class CapabilityNotRegisteredError(
    RuntimeCapabilityError
):
    """
    Raised when no executor is registered for a capability.
    """


class CapabilityAlreadyRegisteredError(
    RuntimeCapabilityError
):
    """
    Raised when an executor is registered more than once.
    """


class CapabilityExecutionError(
    RuntimeCapabilityError
):
    """
    Raised when a capability executor fails during execution.

    The original provider/implementation exception is retained
    as the exception cause, but callers receive a Runtime-level
    error type.
    """


class CapabilityGovernanceError(
    RuntimeCapabilityError
):
    """
    Raised when governance evaluation itself fails.

    This is deliberately different from a governance rejection.

    Rejection:
        governance worked and said no.

    Governance error:
        governance could not produce a valid decision.
    """
"""

class CapabilityExecutor(ABC):
    """
    Abstract executor for one Runtime capability.

    Each concrete executor must:

    - declare the capability it handles;
    - implement asynchronous execution;
    - return a provider-independent result.

    Executors must not bypass:

        CapabilityGateway
            |
            v
        Governance
    """

    @property
    @abstractmethod
    def capability(
        self,
    ) -> CapabilityType:
        """
        Return the capability handled by this executor.

        Returns:

            CapabilityType
        """

        raise NotImplementedError(
            "Capability must be implemented."
        )

    @abstractmethod
    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a capability request.

        Implementations must:

        - consume CapabilityRequest;
        - return CapabilityResult;
        - remain provider-independent.

        Implementations must not:

        - bypass Runtime governance;
        - directly modify application state;
        - create uncontrolled external side effects.
        """

        raise NotImplementedError(
            "Executor execution must be implemented."
        )

        self,

AI Runtime Capability Gateway.

Controlled execution boundary between applications and
concrete AI capabilities.

Flow:

Application
    |
    v
CapabilityRequest
    |
    v
CapabilityGateway
    |
    v
Governance
    |
    v
CapabilityRegistry
    |
    v
CapabilityExecutor
    |
    v
CapabilityResult

The gateway:

- validates execution requests;
- enforces governance before execution;
- resolves registered capabilities;
- delegates execution.

The gateway does not:

- implement capability logic;
- call providers;
- manage memory;
- modify agents;
- perform external side effects.
"""

from __future__ import annotations

from typing import Protocol

from src.ai_runtime.capabilities.errors import (
    CapabilityExecutionError,
    CapabilityGovernanceError,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
)

from src.ai_runtime.capabilities.registry import (
    CapabilityRegistry,
)


class GovernanceDecisionResult(Protocol):
    """
    Minimal governance result required by Runtime.
    """

    @property
    def allowed(self) -> bool:
        """
        Whether execution is permitted.
        """
        ...


class CapabilityGovernance(Protocol):
    """
    Governance boundary required by Runtime gateway.
    """

    def evaluate(
        self,
        request: CapabilityRequest,
    ) -> GovernanceDecisionResult:
        """
        Evaluate whether a capability may execute.
        """
        ...


class CapabilityGateway:
    """
    Controlled Runtime entry point for capability execution.

    Responsibilities:

    - governance enforcement;
    - capability resolution;
    - execution delegation.

    It is intentionally not a business layer.
    """

    def __init__(
        self,
        registry: CapabilityRegistry,
        governance: CapabilityGovernance,
    ) -> None:
        """
        Create capability gateway.

        Dependencies are injected from composition.
        """

        if registry is None:
            raise ValueError(
                "CapabilityRegistry cannot be None."
            )

        if governance is None:
            raise ValueError(
                "CapabilityGovernance cannot be None."
            )

        self._registry = registry
        self._governance = governance

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute an approved capability.

        Execution order:

        1. Validate request
        2. Evaluate governance
        3. Resolve executor
        4. Execute capability

        Governance is always evaluated before
        capability execution.
        """

        if request is None:
            raise ValueError(
                "CapabilityRequest cannot be None."
            )

        try:
            governance_result = (
                self._governance.evaluate(
                    request,
                )
            )

        except Exception as exc:
            raise CapabilityGovernanceError(
                "Governance evaluation failed."
            ) from exc

        if not governance_result.allowed:
            return CapabilityResult(
                request_id=request.request_id,
                capability=request.capability,
                status=(
                    CapabilityExecutionStatus.REJECTED
                ),
                error=(
                    "Capability execution was rejected "
                    "by governance."
                ),
                metadata={
                    "governance_decision": "rejected",
                },
            )

        try:
            executor = self._registry.resolve(
                request.capability,
            )

        except Exception as exc:
            raise CapabilityExecutionError(
                "Capability resolution failed: "
                f"{request.capability.value}"
            ) from exc

        try:
            return await executor.execute(
                request,
            )

        except Exception as exc:
            raise CapabilityExecutionError(
                "Capability executor failed: "
                f"{request.capability.value}"
            ) from exc

        self._agent = agent

    def execute(self, payload: dict[str, Any]) -> dict[str, Any]:
        query = payload.get("query")
        max_results = payload.get("max_results", 5)

            InternalDocumentationRequest(
                query=query,
                max_results=max_results,

            "result_count": response.result_count,
            "results": [
                {
                    "source_id": result.source_id,
                    "title": result.title,
                    "content": result.content,
                    "location": result.location,
                    "relevance": result.relevance,
                }
                for result in response.results
            ],
        }

class ListeningProposalExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for Podcast Listening Proposal flow.
    """

    def __init__(
        self,
        service: ListeningProposalService | None = None,
    ) -> None:
        """
        Initialise executor.

        Service injection keeps Runtime testing isolated.
        """

        self._service = (
            service
            if service is not None
            else ListeningProposalService()
        )

        if self._service is None:
            raise ValueError(
                "ListeningProposalService cannot be None."
            )

    @property
    def capability(
        self,
    ) -> CapabilityType:
        """
        Return capability handled by this executor.
        """

        return (
            CapabilityType.PODCAST_LISTENING_PROPOSAL
        )

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute provider-independent listening proposal.
        """

        if request is None:
            raise ValueError(
                "CapabilityRequest cannot be None."
            )

        if (
            request.capability
            != CapabilityType.PODCAST_LISTENING_PROPOSAL
        ):
            raise ValueError(
                "ListeningProposalExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        recommendation = (
            self._build_recommendation(
                request.payload,
            )
        )

        target_language = (
            self._build_string(
                request.payload,
                "target_language",
            )
        )

        learner_level = (
            self._build_string(
                request.payload,
                "learner_level",
            )
        )

        action = self._build_action(
            request.payload,
        )

        proposal = (
            self._service.create_proposal(
                recommendation,
                target_language=target_language,
                learner_level=learner_level,
                action=action,
            )
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=(
                CapabilityExecutionStatus.COMPLETED
            ),
            output=proposal,
            metadata={
                "executor": (
                    "listening_proposal_executor"
                ),
                "passive": True,
                "side_effects": False,
            },
        )

    @staticmethod
    def _build_recommendation(
        payload: Mapping[str, Any],
    ) -> PodcastRecommendation:
        """
        Validate PodcastRecommendation input.

        Discovery must already have produced the
        recommendation.

        This executor does not perform discovery.
        """

        recommendation = payload.get(
            "recommendation",
        )

        if not isinstance(
            recommendation,
            PodcastRecommendation,
        ):
            raise ValueError(
                "Podcast Listening Proposal capability "
                "requires a PodcastRecommendation under "
                "'recommendation'."
            )

        return recommendation

    @staticmethod
    def _build_string(
        payload: Mapping[str, Any],
        name: str,
    ) -> str:
        """
        Validate required string input.
        """

        value = payload.get(name)

        if not isinstance(
            value,
            str,
        ):
            raise ValueError(
                f"Podcast Listening Proposal '{name}' "
                "must be a string."
            )

        if not value.strip():
            raise ValueError(
                f"Podcast Listening Proposal '{name}' "
                "cannot be empty."
            )

        return value

    @staticmethod
    def _build_action(
        payload: Mapping[str, Any],
    ) -> ListeningProposalAction:
        """
        Convert Runtime action input into enum.
        """

        value = payload.get(
            "action",
            ListeningProposalAction.LISTEN.value,
        )

        if isinstance(
            value,
            ListeningProposalAction,
        ):
            return value

        if not isinstance(
            value,
            str,
        ):
            raise ValueError(
                "Podcast Listening Proposal 'action' "
                "must be a string."
            )

        try:
            return ListeningProposalAction(
                value,
            )

        except ValueError as exc:
            raise ValueError(
                "Podcast Listening Proposal 'action' "
                "must be 'listen' or 'open'."
            ) from exc

"""
AI Runtime capability contracts.

This module defines the provider-independent contracts used by
the AI Runtime to request and represent execution of approved
capabilities.

Architecture:

Application
    |
    v
AI Runtime
    |
    v
Capability Request
    |
    v
Governance
    |
    v
Capability Execution
    |
    v
Capability Result


This module contains contracts only.

It does not:

- execute agents;
- call providers;
- evaluate governance policies;
- perform external side effects.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import StrEnum
from typing import Any, Mapping


class CapabilityType(StrEnum):
    """
    Supported AI Runtime capability categories.

    Capabilities are explicit Runtime contracts.

    Adding a new capability requires:

    - registration;
    - governance review;
    - executor implementation;
    - testing.

    The capability itself does not imply autonomous execution.
    """

    LIVE_INTELLIGENCE = (
        "live_intelligence"
    )

    PODCAST_DISCOVERY = (
        "podcast_discovery"
    )

    PODCAST_LISTENING_PROPOSAL = (
        "podcast_listening_proposal"
    )

    LEARNING_GENERATION = (
        "learning_generation"
    )

    FLASHCARD_GENERATION = (
        "flashcard_generation"
    )

    EVALUATION = (
        "evaluation"
    )

CapabilityName = CapabilityType


class CapabilityExecutionStatus(StrEnum):
    """
    Runtime-level execution status.
    """

    ACCEPTED = "accepted"

    COMPLETED = "completed"

    REJECTED = "rejected"

    FAILED = "failed"


@dataclass(frozen=True, slots=True)
class CapabilityRequest:
    """
    Provider-independent request submitted to AI Runtime.

    The application/API boundary may provide either:

    - CapabilityType enum values;
    - string capability values.

    Strings are normalised to CapabilityType internally.
    """

    capability: CapabilityType

    request_id: str

    payload: Mapping[str, Any]

    metadata: Mapping[str, Any] = field(
        default_factory=dict
    )

    requested_by: str = "application"

    def __post_init__(self) -> None:
        """
        Validate and normalise the Runtime request.
        """

        if not isinstance(
            self.request_id,
            str,
        ):
            raise ValueError(
                "Capability request_id must be a string."
            )

        if not self.request_id.strip():
            raise ValueError(
                "Capability request_id cannot be empty."
            )

        if not isinstance(
            self.requested_by,
            str,
        ):
            raise ValueError(
                "Capability requested_by must be a string."
            )

        if not self.requested_by.strip():
            raise ValueError(
                "Capability requested_by cannot be empty."
            )

        if not isinstance(
            self.payload,
            Mapping,
        ):
            raise ValueError(
                "Capability payload must be a mapping."
            )

        if not isinstance(
            self.metadata,
            Mapping,
        ):
            raise ValueError(
                "Capability metadata must be a mapping."
            )

        if isinstance(
            self.capability,
            str,
        ):
            try:
                object.__setattr__(
                    self,
                    "capability",
                    CapabilityType(
                        self.capability,
                    ),
                )

            except ValueError as exc:
                raise ValueError(
                    "Unsupported capability: "
                    f"{self.capability}"
                ) from exc

        elif not isinstance(
            self.capability,
            CapabilityType,
        ):
            raise ValueError(
                "Capability must be a valid CapabilityType."
            )


@dataclass(frozen=True, slots=True)
class CapabilityResult:
    """
    Provider-independent result returned by AI Runtime.

    Provider-specific responses remain hidden behind
    capability executors.
    """

    request_id: str

    capability: CapabilityType

    status: CapabilityExecutionStatus

    output: Any = None

    error: str | None = None

    metadata: Mapping[str, Any] = field(
        default_factory=dict
    )

    @property
    def succeeded(self) -> bool:
        """
        Return True when execution completed successfully.
        """

        return (
            self.status
            == CapabilityExecutionStatus.COMPLETED
        )

    @property
    def failed(self) -> bool:
        """
        Return True when execution failed.
        """

        return (
            self.status
            == CapabilityExecutionStatus.FAILED
        )

    @property
    def rejected(self) -> bool:
        """
        Return True when governance rejected execution.
        """

        return (
            self.status
            == CapabilityExecutionStatus.REJECTED
        )

class PodcastDiscoveryExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for the existing Podcast Discovery flow.
    """

    def __init__(
        self,
        agent: PodcastDiscoveryAgent,
        providers: Sequence[PodcastProvider],
    ) -> None:
        """
        Create Podcast Discovery executor.

        Dependencies are injected from Runtime composition.
        """

        if agent is None:
            raise ValueError(
                "PodcastDiscoveryAgent cannot be None."
            )

        if not providers:
            raise ValueError(
                "PodcastDiscoveryExecutor requires "
                "at least one provider."
            )

        self._agent = agent

        self._recommendation_service = (
            PodcastRecommendationService(
                providers=list(providers),
            )
        )

    @property
    def capability(
        self,
    ) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """

        return CapabilityType.PODCAST_DISCOVERY

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a provider-independent podcast discovery request.
        """

        if request is None:
            raise ValueError(
                "CapabilityRequest cannot be None."
            )

        if (
            request.capability
            != CapabilityType.PODCAST_DISCOVERY
        ):
            raise ValueError(
                "PodcastDiscoveryExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        intent = self._build_search_intent(
            request,
        )

        recommendation = (
            self._recommendation_service.recommend(
                intent=intent,
                character_id=self._required_string(
                    request.payload,
                    "character_id",
                ),
                character_name=self._required_string(
                    request.payload,
                    "character_name",
                ),
                character_message=self._required_string(
                    request.payload,
                    "character_message",
                ),
            )
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=(
                CapabilityExecutionStatus.COMPLETED
            ),
            output=recommendation,
            metadata={
                "executor": (
                    "podcast_discovery_executor"
                ),
            },
        )

    def _build_search_intent(
        self,
        request: CapabilityRequest,
    ):
        """
        Translate Runtime payload into the existing
        PodcastSearchIntent through PodcastDiscoveryAgent.
        """

        payload = request.payload

        user_id = self._required_string(
            payload,
            "user_id",
        )

        learning_context = (
            self._build_learning_context(
                payload,
            )
        )

        memories = self._build_memories(
            payload,
        )

        max_results = payload.get(
            "max_results",
            10,
        )

        if not isinstance(
            max_results,
            int,
        ):
            raise ValueError(
                "Podcast Discovery 'max_results' "
                "must be an integer."
            )

        if max_results < 1:
            raise ValueError(
                "Podcast Discovery 'max_results' "
                "must be at least 1."
            )

        return self._agent.build_search_intent(
            user_id=user_id,
            learning_context=learning_context,
            memories=memories,
            max_results=max_results,
        )

    @staticmethod
    def _build_learning_context(
        payload: Mapping[str, Any],
    ) -> LearningContext:
        """
        Build provider-independent LearningContext.
        """

        world_id = payload.get(
            "world_id",
            "French World",
        )

        target_language = payload.get(
            "target_language",
            "fr",
        )

        learner_level = payload.get(
            "learner_level",
            "B2",
        )

        learning_goal = payload.get(
            "learning_goal",
            "",
        )

        country_code = payload.get(
            "country_code",
            "LU",
        )

        preferred_languages = payload.get(
            "preferred_languages",
            (),
        )

        values = {
            "world_id": world_id,
            "target_language": target_language,
            "learner_level": learner_level,
            "learning_goal": learning_goal,
            "country_code": country_code,
        }

        for name, value in values.items():
            if not isinstance(
                value,
                str,
            ):
                raise ValueError(
                    f"Podcast Discovery '{name}' "
                    "must be a string."
                )

        if not isinstance(
            preferred_languages,
            (list, tuple),
        ):
            raise ValueError(
                "Podcast Discovery "
                "'preferred_languages' "
                "must be a list or tuple."
            )

        if not all(
            isinstance(language, str)
            for language in preferred_languages
        ):
            raise ValueError(
                "Podcast Discovery "
                "'preferred_languages' "
                "must contain only strings."
            )

        return LearningContext(
            world_id=world_id,
            target_language=target_language,
            learner_level=learner_level,
            learning_goal=learning_goal,
            preferred_languages=tuple(
                preferred_languages
            ),
            country_code=country_code,
        )

    @staticmethod
    def _build_memories(
        payload: Mapping[str, Any],
    ) -> list[MemoryEntry]:
        """
        Validate explicit learner memories.

        Memory retrieval remains outside this executor.
        """

        memories = payload.get(
            "memories",
            (),
        )

        if not isinstance(
            memories,
            (list, tuple),
        ):
            raise ValueError(
                "Podcast Discovery 'memories' "
                "must be a list or tuple."
            )

        for memory in memories:
            if not isinstance(
                memory,
                MemoryEntry,
            ):
                raise ValueError(
                    "Podcast Discovery 'memories' "
                    "must contain only "
                    "MemoryEntry objects."
                )

        return list(memories)

    @staticmethod
    def _required_string(
        payload: Mapping[str, Any],
        field_name: str,
    ) -> str:
        """
        Return a required non-empty string payload value.
        """

        value = payload.get(
            field_name,
        )

        if not isinstance(
            value,
            str,
        ):
            raise ValueError(
                f"Podcast Discovery '{field_name}' "
                "must be a string."
            )

        if not value.strip():
            raise ValueError(
                f"Podcast Discovery '{field_name}' "
                "cannot be empty."
            )

        return value

class PodcastLearningChallengeExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for Podcast Learning Challenge generation.
    """

    def __init__(
        self,
        service: PodcastLearningChallengeService | None = None,
    ) -> None:
        """
        Create learning challenge executor.

        Service dependency may be injected for testing.
        """

        self._service = (
            service
            if service is not None
            else PodcastLearningChallengeService()
        )

    @property
    def capability(
        self,
    ) -> CapabilityType:
        """
        Return the Runtime capability handled by this executor.
        """

        return CapabilityType.LEARNING_GENERATION

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Generate a Podcast Learning Challenge.

        The request must contain a verified
        PodcastRecommendation under:

            recommendation

        This executor does not perform discovery.
        """

        if request is None:
            raise ValueError(
                "CapabilityRequest cannot be None."
            )

        if (
            request.capability
            != CapabilityType.LEARNING_GENERATION
        ):
            raise ValueError(
                "PodcastLearningChallengeExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        recommendation = (
            self._build_recommendation(
                request.payload,
            )
        )

        challenge = (
            self._service.create_challenge(
                recommendation=recommendation,
            )
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=(
                CapabilityExecutionStatus.COMPLETED
            ),
            output=challenge,
            metadata={
                "executor": (
                    "podcast_learning_challenge_executor"
                ),
                "challenge_type": (
                    "podcast_learning_mvp"
                ),
                "content_basis": (
                    "verified_podcast_metadata"
                ),
            },
        )

    @staticmethod
    def _build_recommendation(
        payload: Mapping[str, Any],
    ) -> PodcastRecommendation:
        """
        Validate and extract PodcastRecommendation.

        The recommendation must already have been produced
        by Podcast Discovery.

        This executor does not perform discovery.
        """

        recommendation = payload.get(
            "recommendation",
        )

        if not isinstance(
            recommendation,
            PodcastRecommendation,
        ):
            raise ValueError(
                "Learning Generation capability requires "
                "a PodcastRecommendation under "
                "'recommendation'."
            )

        return recommendation
"""
AI Runtime capability registry.

The registry maps logical Runtime capabilities to their
concrete executors.

The registry:

- resolves executors by capability;
- prevents duplicate registrations;
- provides explicit dependency injection;
- validates registrations.

The registry does not:

- execute capabilities;
- perform governance decisions;
- know about providers;
- manage application state.
"""

from __future__ import annotations

from src.ai_runtime.capabilities.errors import (
    CapabilityAlreadyRegisteredError,
    CapabilityNotRegisteredError,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityType,
)


class CapabilityRegistry:
    """
    Dependency-injection registry for Runtime capabilities.

    The registry is intentionally simple.

    Composition creates it.
    Gateway consumes it.

    It does not own lifecycle management.
    """

    def __init__(
        self,
        executors: list[CapabilityExecutor] | None = None,
    ) -> None:
        """
        Create capability registry.

        Args:

            executors:
                Optional initial executor collection.
        """

        self._executors: dict[
            CapabilityType,
            CapabilityExecutor,
        ] = {}

        for executor in executors or []:
            self.register(
                executor,
            )

    def register(
        self,
        executor: CapabilityExecutor,
    ) -> None:
        """
        Register an executor for its declared capability.

        Raises:

            ValueError:
                If executor is invalid.

            CapabilityAlreadyRegisteredError:
                If capability already exists.
        """

        if executor is None:
            raise ValueError(
                "CapabilityExecutor cannot be None."
            )

        capability = executor.capability

        if not isinstance(
            capability,
            CapabilityType,
        ):
            raise ValueError(
                "Executor capability must be a valid "
                "CapabilityType."
            )

        if capability in self._executors:
            raise CapabilityAlreadyRegisteredError(
                "Capability executor already registered: "
                f"{capability.value}"
            )

        self._executors[capability] = executor

    def resolve(
        self,
        capability: CapabilityType,
    ) -> CapabilityExecutor:
        """
        Resolve the executor responsible for a capability.

        Raises:

            CapabilityNotRegisteredError:
                When no executor exists.
        """

        if not isinstance(
            capability,
            CapabilityType,
        ):
            raise ValueError(
                "Capability must be a valid CapabilityType."
            )

        try:
            return self._executors[capability]

        except KeyError as exc:
            raise CapabilityNotRegisteredError(
                "No executor registered for capability: "
                f"{capability.value}"
            ) from exc

    def contains(
        self,
        capability: CapabilityType,
    ) -> bool:
        """
        Return whether an executor is registered.
        """

        return capability in self._executors

    def capabilities(
        self,
    ) -> tuple[CapabilityType, ...]:
        """
        Return all registered capabilities.

        Returned as immutable tuple to prevent
        external mutation.
        """

        return tuple(
            self._executors.keys()
        )

This adapter bridges the provider-independent AI Runtime
capability contract with the existing LiveIntelligenceAgent.

Architecture:

The adapter:

- validates the Runtime capability;
- translates Runtime payloads into LiveIntelligenceRequest;
- invokes the existing Live Intelligence Agent;
- translates the agent response into CapabilityResult.

It does not:

- implement intelligence logic;
- call providers directly;
- modify the LiveIntelligenceAgent;
- perform governance decisions.

This adapter bridges the provider-independent AI Runtime capability
contract with the existing Podcast Discovery Agent and
Podcast Recommendation Service.

Architecture:

AI Runtime
    |
    v

    |
    v
PodcastDiscoveryExecutor
    |
    +--> PodcastDiscoveryAgent
    |
    +--> PodcastRecommendationService
    |

    |

The executor:

- validates the Runtime capability;
- validates explicit capability inputs;
- builds the existing LearningContext;
- passes explicit learner memories to the PodcastDiscoveryAgent;
- invokes the existing PodcastRecommendationService;
- returns a provider-independent CapabilityResult.

It does not:

- call Apple directly;
- call Spotify directly;
- implement provider logic;
- retrieve memory itself;
- modify MemoryService;
- perform purchases;
- bypass the AI Runtime;
- make governance decisions.

AI Runtime executor for Podcast Discovery.

This adapter bridges the provider-independent AI Runtime capability
contract with the existing Podcast Discovery Agent and
Podcast Recommendation Service.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
PodcastDiscoveryExecutor
    |
    +--> PodcastDiscoveryAgent
    |
    +--> PodcastRecommendationService
    |
    v
PodcastRecommendation
    |
    v
CapabilityResult

The executor:

- validates the Runtime capability;
- validates explicit capability inputs;
- builds the existing LearningContext;
- passes explicit learner memories to the PodcastDiscoveryAgent;
- invokes the existing PodcastRecommendationService;
- returns a provider-independent CapabilityResult.

It does not:

- call Apple directly;
- call Spotify directly;
- implement provider logic;
- retrieve memory itself;
- modify MemoryService;
- perform purchases;
- bypass the AI Runtime;
- make governance decisions.
