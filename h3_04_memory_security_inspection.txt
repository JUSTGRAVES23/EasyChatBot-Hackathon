class ConversationAgent(BaseAgent):
    """
    Character-aware conversation agent.

    The agent consumes the provider-independent
    ExecutionContext.

    Memory and Live Intelligence remain application/runtime
    concerns. The agent only receives their prepared context.
    """

    def __init__(
        self,
        character: CharacterProfile,
        llm_service: LLMService | None = None,
        llm_provider: LLMProvider | None = None,
    ) -> None:

        super().__init__(
            AgentManifest(
                agent_id="conversation_agent",
                name="Conversation Agent",
                description=(
                    "Provides character-aware "
                    "conversation responses."
                ),
            )
        )

        if (
            llm_service is not None
            and llm_provider is not None
        ):
            raise ValueError(
                "Provide either llm_service or "
                "llm_provider, not both."
            )

        self.character = character

        if llm_service is not None:
            self.llm_service = llm_service

        elif llm_provider is not None:
            self.llm_service = LLMService(
                llm_provider
            )

        else:
            self.llm_service = None

    def build_request(
        self,
        context: ExecutionContext,
    ) -> ConversationRequest:
        """
        Build provider-independent conversation request.

        Memory and intelligence are explicitly represented
        as contextual information.

        Neither is treated as an instruction.
        """

        system_instruction = (
            f"You are {self.character.name}, "
            f"a character in EasyChatBot. "
            f"Your personality is: "
            f"{self.character.personality} "
            f"Your learning style is: "
            f"{self.character.learning_style} "
            f"Communicate in a way that is consistent "
            f"with this character."
        )

        context_sections: list[str] = []

        # -------------------------------------------------
        # Learner memory
        # -------------------------------------------------

        if context.has_memory_context:
            memory_text = (
                context.memory_context.text
            )

            context_sections.append(
                memory_text
            )

        # -------------------------------------------------
        # Live intelligence
        # -------------------------------------------------

        if context.has_intelligence_context:
            intelligence_text = (
                context.intelligence_context
                .to_context_text()
            )

            context_sections.append(
                "Current live intelligence:\n"
                + intelligence_text
            )

        conversation_context = None

        if context_sections:
            conversation_context = (
                "\n\n".join(
                    context_sections
                )
            )

        return ConversationRequest(
            system_instruction=system_instruction,
            user_prompt=context.request,
            conversation_context=(
                conversation_context
            ),
            character_id=(
                self.character.character_id
            ),
            character_name=(
                self.character.name
            ),
        )

    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute the conversation agent.
        """

        conversation_request = (
            self.build_request(context)
        )

        provider_result = None

        if self.llm_service is not None:

            provider_result = (
                await self.llm_service.generate(
                    conversation_request
                )
            )

            if not provider_result.success:
                return AgentResult(
                    success=False,
                    agent_id=(
                        self.manifest.agent_id
                    ),
                    output={
                        "character": (
                            self.character.name
                        ),

                        "character_profile": {
                            "character_id": (
                                self.character.character_id
                            ),
                            "name": (
                                self.character.name
                            ),
                            "description": (
                                self.character.description
                            ),
                            "personality": (
                                self.character.personality
                            ),
                            "learning_style": (
                                self.character.learning_style
                            ),
                        },

                        "personality": (
                            self.character.personality
                        ),

                        "learning_style": (
                            self.character.learning_style
                        ),

                        "message": (
                            "The conversation provider "
                            "could not generate a response."
                        ),

                        "intelligence_context": (
                            conversation_request
                            .conversation_context
                        ),

                        "conversation_request": (
                            conversation_request
                        ),

                        "provider_id": (
                            provider_result.provider_id
                        ),

                        "provider_error": (
                            provider_result.error
                        ),
                    },
                )

        if provider_result is not None:
            message = provider_result.output

        else:
            message = (
                f"{self.character.name}: "
                f"Let's explore this together. "
                f"You asked: "
                f"{context.request}"
            )

        output = {
            "character": (
                self.character.name
            ),

            "character_profile": {
                "character_id": (
                    self.character.character_id
                ),
                "name": (
                    self.character.name
                ),
                "description": (
                    self.character.description
                ),
                "personality": (
                    self.character.personality
                ),
                "learning_style": (
                    self.character.learning_style
                ),
            },

            "personality": (
                self.character.personality
            ),

            "learning_style": (
                self.character.learning_style
            ),

            "message": message,

            "intelligence_context": (
                conversation_request
                .conversation_context
            ),

            "conversation_request": (
                conversation_request
            ),
        }

        if provider_result is not None:
            output["provider_id"] = (
                provider_result.provider_id
            )

        return AgentResult(
            success=True,
            agent_id=(
                self.manifest.agent_id
            ),
            output=output,
        )

"""
OpenAI LLM Provider.

Concrete LLM provider implementation for the EasyChatBot
hackathon runtime.

Architecture:

ConversationAgent
|

LLMService

LLMProvider
|

|

OpenAI Responses API
|
v
ProviderResult

Responsibilities:

- execute ConversationRequest through OpenAI
- use the configured OpenAI model
- translate the OpenAI response into ProviderResult
- isolate the OpenAI SDK from the application layer

Does not:

- build prompts
- manage characters
- manage memory
- perform research
- perform routing
- modify application state
"""

from __future__ import annotations

import logging
import os

from openai import AsyncOpenAI

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.providers.provider_result import (
    ProviderResult,
)


logger = logging.getLogger(__name__)


class OpenAIProvider(LLMProvider):
    """
    OpenAI implementation of the LLMProvider contract.

    The OpenAI SDK is deliberately isolated inside this
    infrastructure adapter.
    """

    def __init__(
        self,
        model: str = "gpt-5-mini",
        client: AsyncOpenAI | None = None,
    ) -> None:
        """
        Initialise the OpenAI provider.

        Parameters
        ----------
        model:
            OpenAI model identifier.

        client:
            Optional injected AsyncOpenAI client.

            Dependency injection allows unit tests to
            provide a fake client without making a real
            API call.
        """

        super().__init__(
            provider_id="openai",
        )

        if not isinstance(
            model,
            str,
        ):
            raise ValueError(
                "OpenAI model must be a string."
            )

        model = model.strip()

        if not model:
            raise ValueError(
                "OpenAI model cannot be empty."
            )

        self.model = model

        if client is not None:
            self._client = client
            return

        api_key = os.getenv(
            "OPENAI_API_KEY"
        )

        if not api_key:
            raise ValueError(
                "OPENAI_API_KEY is not configured."
            )

        self._client = AsyncOpenAI(
            api_key=api_key,
        )

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate a response using OpenAI.

        The provider translates the provider-independent
        ConversationRequest into the OpenAI Responses API
        request.

        Parameters
        ----------
        request:
            Provider-independent conversation request.

        Returns
        -------
        ProviderResult
            Provider-independent execution result.

        Raises
        ------
        ValueError
            If request is None.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        messages = [
            {
                "role": "system",
                "content": request.system_instruction,
            },
        ]

        if request.conversation_context:
            messages.append(
                {
                    "role": "user",
                    "content": (
                        "Additional context:\n"
                        f"{request.conversation_context}"
                    ),
                }
            )

        messages.append(
            {
                "role": "user",
                "content": request.user_prompt,
            }
        )

        try:
            response = (
                await self._client.responses.create(
                    model=self.model,
                    input=messages,
                )
            )

        except Exception as exc:
            logger.exception(
                "OpenAI provider execution failed."
            )

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=str(exc),
            )

        output_text = getattr(
            response,
            "output_text",
            None,
        )

        if not output_text:
            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=(
                    "OpenAI returned an empty response."
                ),
            )

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output=output_text,
        )

"""
Runtime Execution Context.

Provides immutable runtime context shared with agents.

Responsibilities:
- identify an execution
- carry user request information
- provide runtime metadata
- optionally carry live intelligence context
- optionally carry bounded learner memory context

Does not:
- execute providers
- manage memory persistence
- modify systems
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any
from uuid import uuid4

from src.ai_runtime.memory.context_builder import (
    MemoryContext,
)


@dataclass(frozen=True)
class ExecutionContext:
    """
    Runtime execution context shared with agents.

    The context is immutable. Enrichment must therefore use
    dataclasses.replace() rather than mutating an existing
    instance.
    """

    user_id: str

    request: str

    execution_id: str = field(
        default_factory=lambda: str(uuid4())
    )

    created_at: datetime = field(
        default_factory=lambda: datetime.now(timezone.utc)
    )

    metadata: dict[str, Any] = field(
        default_factory=dict
    )

    intelligence_context: Any | None = None

    memory_context: MemoryContext | None = None

    @property
    def has_intelligence_context(self) -> bool:
        """
        Return whether live intelligence is attached.
        """

        return (
            self.intelligence_context is not None
        )

    @property
    def has_memory_context(self) -> bool:
        """
        Return whether bounded learner memory context
        is attached.
        """

        return (
            self.memory_context is not None
            and not self.memory_context.is_empty
        )

class CompanionService:
    """
    Coordinates AI companion experiences.

    The service orchestrates agents but does not contain
    provider-specific logic.

    Providers are injected into agents by the composition
    layer.

    Memory retrieval remains owned by MemoryService.

    MemoryContextBuilder converts already-retrieved memory
    into bounded runtime context.

    ExecutionContext remains immutable. Enrichment is
    performed using dataclasses.replace().
    """

    def __init__(
        self,
        character: CharacterProfile,
        conversation_agent: ConversationAgent | None = None,
        research_agent: ResearchAgent | None = None,
        quality_agent: QualityAgent | None = None,
        memory_service: MemoryService | None = None,
        memory_context_builder: (
            MemoryContextBuilder | None
        ) = None,
    ) -> None:
        """
        Initialise the CompanionService.

        Dependencies are injectable so the service can be
        tested without invoking real infrastructure.
        """

        self.character = character

        self.conversation_agent = (
            conversation_agent
            if conversation_agent is not None
            else ConversationAgent(
                character
            )
        )

        self.research_agent = (
            research_agent
            if research_agent is not None
            else ResearchAgent()
        )

        self.quality_agent = (
            quality_agent
            if quality_agent is not None
            else QualityAgent()
        )

        self.memory_service = (
            memory_service
            if memory_service is not None
            else MemoryService()
        )

        self.memory_context_builder = (
            memory_context_builder
            if memory_context_builder is not None
            else MemoryContextBuilder()
        )

    async def interact(
        self,
        context: ExecutionContext,
        use_research: bool = False,
    ) -> dict[str, Any]:
        """
        Execute a complete companion interaction.

        Flow:

        MemoryService
            ↓
        MemoryContextBuilder
            ↓
        optional memory-enriched ExecutionContext
            ↓
        optional ResearchAgent
            ↓
        optional intelligence-enriched ExecutionContext
            ↓
        ConversationAgent
            ↓
        QualityAgent
            ↓
        MemoryService.remember()

        The original ExecutionContext is never mutated.

        Parameters
        ----------
        context:
            Runtime execution context.

        use_research:
            Whether live research should be performed
            before the conversation response.
        """

        # -----------------------------------------------------
        # Step 1 — Retrieve existing learner memory
        # -----------------------------------------------------

        memories = self.memory_service.recall(
            context.user_id
        )

        # -----------------------------------------------------
        # Step 2 — Build bounded learner memory context
        # -----------------------------------------------------

        memory_context = (
            self.memory_context_builder.build(
                user_id=context.user_id,
                request=context.request,
                memories=memories,
            )
        )

        # -----------------------------------------------------
        # Step 3 — Attach memory only when relevant memory
        # exists.
        #
        # If there is no relevant memory, preserve the
        # original ExecutionContext object.
        # -----------------------------------------------------

        if memory_context.is_empty:
            conversation_context = context
        else:
            conversation_context = replace(
                context,
                memory_context=memory_context,
            )

        # -----------------------------------------------------
        # Step 4 — Optional Live Intelligence
        #
        # Research receives the current runtime context but
        # does not receive or manage memory itself.
        #
        # If research succeeds, create another immutable
        # context preserving the existing memory_context.
        # -----------------------------------------------------

        research_result = None

        if use_research:
            research_result = (
                await self.research_agent.execute(
                    conversation_context
                )
            )

            if (
                research_result.success
                and research_result.output is not None
            ):
                conversation_context = replace(
                    conversation_context,
                    intelligence_context=(
                        research_result.output
                    ),
                )

        # -----------------------------------------------------
        # Step 5 — Conversation Agent
        #
        # ConversationAgent can now receive:
        #
        #   memory_context
        #   intelligence_context
        #
        # independently or together.
        # -----------------------------------------------------

        conversation_result = (
            await self.conversation_agent.execute(
                conversation_context
            )
        )

        # -----------------------------------------------------
        # Step 6 — Quality Agent
        # -----------------------------------------------------

        quality_result = (
            await self.quality_agent.execute(
                conversation_context
            )
        )

        # -----------------------------------------------------
        # Step 7 — Persist interaction
        #
        # Memory persistence remains owned by MemoryService.
        # -----------------------------------------------------

        self.memory_service.remember(
            user_id=context.user_id,
            category="conversation",
            content=context.request,
        )

        # -----------------------------------------------------
        # Step 8 — Return combined result
        # -----------------------------------------------------

        return {
            "character": self.character.name,

            "conversation": (
                conversation_result.output
            ),

            "research": (
                research_result.output
                if (
                    research_result is not None
                    and research_result.success
                )
                else None
            ),

            "quality": (
                quality_result.output
            ),

            "memory_count": len(memories),

            "memory_context_count": (
                memory_context.memory_count
            ),
        }

"""
AI Runtime capability registry.

The registry maps logical Runtime capabilities to their
concrete executors.

The registry:
- resolves executors by capability
- prevents duplicate registrations
- provides explicit dependency injection
- does not execute capabilities
- does not perform governance decisions
- does not know about providers
"""

from __future__ import annotations

from src.ai_runtime.capabilities.errors import (
    CapabilityAlreadyRegisteredError,
    CapabilityNotRegisteredError,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityType,
)


class CapabilityRegistry:
    """
    Dependency-injection registry for Runtime capabilities.
    """

    def __init__(
        self,
        executors: list[CapabilityExecutor] | None = None,
    ) -> None:
        self._executors: dict[
            CapabilityType,
            CapabilityExecutor,
        ] = {}

        for executor in executors or []:
            self.register(executor)

    def register(
        self,
        executor: CapabilityExecutor,
    ) -> None:
        """
        Register an executor for its declared capability.
        """

        capability = executor.capability

        if capability in self._executors:
            raise CapabilityAlreadyRegisteredError(
                "Capability executor already registered: "
                f"{capability.value}"
            )

        self._executors[capability] = executor

    def resolve(
        self,
        capability: CapabilityType,
    ) -> CapabilityExecutor:
        """
        Resolve the executor responsible for a capability.
        """

        try:
            return self._executors[capability]
        except KeyError as exc:
            raise CapabilityNotRegisteredError(
                "No executor registered for capability: "
                f"{capability.value}"
            ) from exc

    def contains(
        self,
        capability: CapabilityType,
    ) -> bool:
        """
        Return whether an executor is registered.
        """

        return capability in self._executors

    def capabilities(
        self,
    ) -> tuple[CapabilityType, ...]:
        """
        Return all registered capabilities.
        """

        return tuple(self._executors.keys())

class DiscoveryService:
    """
    Orchestrates personalised discovery.

    This service intentionally contains no provider logic and no
    persistence logic. It composes the existing DealCard,
    MemoryEntry and FlashDealCard contracts.
    """

    def create_flash_deal(
        self,
        request: DiscoveryRequest,
        memories: list[MemoryEntry] | None = None,
    ) -> FlashDealCard:
        """
        Create a personalised FlashDealCard.

        Memory is read-only input to this operation.

        Preference memory is used to generate the personal
        relevance explanation. Other memory types are not
        automatically treated as preferences.
        """

        memory_context = memories or []

        relevance_reason = self._build_relevance_reason(
            memory_context
        )

        return FlashDealCard.from_deal(
            deal=request.deal,
            character_id=request.character.character_id,
            character_name=request.character.character_name,
            character_message=request.character.message,
            relevance_reason=relevance_reason,
            product_image_url=request.product_image_url,
            tagline=request.tagline,
            highlights=request.highlights,
            metadata={
                "user_id": request.user_id,
                "discovery_service": "hackathon",
            },
        )

    @staticmethod
    def _build_relevance_reason(
        memories: list[MemoryEntry],
    ) -> str:
        """
        Build a deterministic relevance explanation.

        Only explicit preference memories are used.

        This is deliberately deterministic for the hackathon;
        an LLM-based personalisation layer can be introduced later
        behind an explicit interface.
        """

        preferences = [
            memory
            for memory in memories
            if memory.memory_type
            == MemoryType.PREFERENCE
        ]

        if not preferences:
            return (
                "This discovery may be relevant to your "

"""
EasyChatBot — Memory Context Builder.

H3-01 — Memory-Aware Live Intelligence.

The builder operates on the APPLICATION memory contract:

    MemoryService
        ↓
    MemoryRecord
        ↓
    MemoryContextBuilder
        ↓
    MemoryContext

It deliberately does NOT consume MemoryEntry directly.

MemoryEntry belongs to the persistence/runtime storage boundary.
MemoryRecord is the application-facing memory contract.

Responsibilities:
- validate already-retrieved memory
- enforce user isolation
- determine deterministic relevance
- bound memory count
- bound context size
- produce provider-independent context

Does not:
- retrieve memory
- persist memory
- call an LLM
- call a provider
- execute an agent
- mutate ExecutionContext
- perform semantic/vector search
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
import re

from backend.memory.memory_record import (
    MemoryRecord,
)


DEFAULT_MAX_MEMORY_ITEMS = 5
DEFAULT_MAX_CONTEXT_CHARACTERS = 4000

_WORD_PATTERN = re.compile(
    r"[\wÀ-ÿ'-]+",
    re.UNICODE,
)


# Application-level memory categories.
#
# These are deliberately strings because MemoryService exposes
# MemoryRecord.category rather than MemoryEntry.memory_type.
_MEMORY_CATEGORY_PRIORITY: dict[str, int] = {
    "preference": 30,
    "learning_goal": 30,
    "discovery_context": 20,
    "saved_item": 10,
    "conversation": 10,
}


@dataclass(frozen=True)
class MemoryContext:
    """
    Bounded, provider-independent learner context.
    """

    entries: tuple[MemoryRecord, ...]
    text: str

    @property
    def is_empty(self) -> bool:
        """Return True when no memory was selected."""
        return not self.entries

    @property
    def memory_count(self) -> int:
        """Return the number of selected memories."""
        return len(self.entries)


class MemoryContextBuilder:
    """
    Build bounded learner context from MemoryRecord objects.

    The builder is stateless and deterministic.
    """

    def __init__(
        self,
        *,
        max_memory_items: int = DEFAULT_MAX_MEMORY_ITEMS,
        max_context_characters: int = (
            DEFAULT_MAX_CONTEXT_CHARACTERS
        ),
    ) -> None:
        if max_memory_items < 1:
            raise ValueError(
                "max_memory_items must be greater than zero."
            )

        if max_context_characters < 1:
            raise ValueError(
                "max_context_characters must be greater than zero."
            )

        self._max_memory_items = max_memory_items
        self._max_context_characters = (
            max_context_characters
        )

    def build(
        self,
        *,
        user_id: str,
        request: str,
        memories: list[MemoryRecord]
        | tuple[MemoryRecord, ...],
    ) -> MemoryContext:
        """
        Build bounded learner context.

        Memory is supplied by the caller.

        The builder does not retrieve or persist memory.
        """

        if not user_id.strip():
            raise ValueError(
                "user_id cannot be empty."
            )

        if not request.strip():
            raise ValueError(
                "request cannot be empty."
            )

        if not memories:
            return MemoryContext(
                entries=(),
                text="",
            )

        request_tokens = self._tokens(
            request
        )

        candidates: list[
            tuple[
                int,
                datetime,
                str,
                MemoryRecord,
            ]
        ] = []

        for memory in memories:

            if not isinstance(
                memory,
                MemoryRecord,
            ):
                continue

            # Defence in depth.
            if memory.user_id != user_id:
                continue

            if not memory.category.strip():
                continue

            if not memory.content.strip():
                continue

            score = self._score(
                memory=memory,
                request_tokens=request_tokens,
            )

            # A memory category alone does NOT make the memory
            # relevant to the current request.
            if score <= 0:
                continue

            candidates.append(
                (
                    score,
                    memory.created_at,
                    memory.content,
                    memory,
                )
            )

        # Deterministic ordering:
        #
        # 1. relevance descending
        # 2. newest memory first
        # 3. content as stable tie-breaker
        candidates.sort(
            key=lambda item: (
                -item[0],
                -item[1].timestamp(),
                item[2].casefold(),
            )
        )

        selected: list[MemoryRecord] = []
        rendered_items: list[str] = []

        current_length = len(
            "Learner context:\n"
        )

        for (
            _score,
            _created_at,
            _content,
            memory,
        ) in candidates:

            if (
                len(selected)
                >= self._max_memory_items
            ):
                break

            line = self._render_memory(
                memory
            )

            separator_length = (
                1
                if rendered_items
                else 0
            )

            projected_length = (
                current_length
                + separator_length
                + len(line)
            )

            if (
                projected_length
                > self._max_context_characters
            ):
                # Do not partially truncate an individual
                # memory because that could alter meaning.
                continue

            selected.append(memory)
            rendered_items.append(line)

            current_length = projected_length

        if not selected:
            return MemoryContext(
                entries=(),
                text="",
            )

        text = (
            "Learner context:\n"
            + "\n".join(
                f"- {item}"
                for item in rendered_items
            )
        )

        if (
            len(text)
            > self._max_context_characters
        ):
            raise AssertionError(
                "MemoryContextBuilder produced "
                "an oversized context."
            )

        return MemoryContext(
            entries=tuple(selected),
            text=text,
        )

    @staticmethod
    def _tokens(
        value: str,
    ) -> set[str]:
        """
        Return normalised tokens for deterministic matching.
        """

        return {
            token.casefold()
            for token in _WORD_PATTERN.findall(
                value
            )
            if len(token) > 1
        }

    @classmethod
    def _score(
        cls,
        *,
        memory: MemoryRecord,
        request_tokens: set[str],
    ) -> int:
        """
        Calculate deterministic relevance.

        Relevance requires actual token overlap.

        This prevents an unrelated preference such as
        "Prefers hiking" from being injected into a request
        about "European AI regulation" simply because it is
        a preference.
        """

        content_tokens = cls._tokens(
            memory.content
        )

        category_tokens = cls._tokens(
            memory.category
        )

        overlap = request_tokens & (
            content_tokens
            | category_tokens
        )

        if not overlap:
            return 0

        category = (
            memory.category.strip().casefold()
        )

        category_priority = (
            _MEMORY_CATEGORY_PRIORITY.get(
                category,
                10,
            )
        )

        return (
            category_priority
            + len(overlap)
        )

    @staticmethod
    def _render_memory(
        memory: MemoryRecord,
    ) -> str:
        """
        Render memory explicitly as learner context.

        Memory content is treated as data, not instructions.
        """

        category = (
            memory.category.strip()
        )

        content = " ".join(
            memory.content.split()
        )

        return (
            f"[{category}] {content}"
        )

"""
Tests for ConversationAgent LLM provider integration.

These tests verify the architecture:

ConversationAgent
    |
    v
LLMService
    |
    v
Injected LLMProvider
    |
    v
ProviderResult

No real provider API is called.

from backend.agents.conversation_agent import (
    ConversationAgent,

    CharacterProfile,

    ProviderResult,

    ExecutionContext,

class FakeLLMProvider:
    """
    Deterministic provider for ConversationAgent tests.

    The provider is injected into ConversationAgent and
    therefore wrapped by LLMService internally.
    """

    def __init__(
        self,
        result: ProviderResult,
    ) -> None:
        self.result = result
        self.received_request = None

    async def generate(
        self,
        request,
    ) -> ProviderResult:
        """
        Capture the request and return the configured result.
        """

        self.received_request = request

        return self.result


def create_character() -> CharacterProfile:
    """
    Create the deterministic test character.
    """

    return CharacterProfile(
        character_id="yeti",
        name="Yeti",
        description=(

        ),

        ),
    )
