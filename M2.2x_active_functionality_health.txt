export type FunctionalityStatus =
  | "healthy"
  | "degraded"
  | "unavailable"
  | "maintenance"
  | "disabled"
  | "unknown";

export type OperationsEnvironment = "kids" | "adults";
export type UserImpact = "low" | "medium" | "high" | "unknown";

export interface FunctionalityHealth {
  functionality_id: string;
  name: string;
  category: string;
  description: string;
  status: FunctionalityStatus;
  health_score: number | null;
  latency_ms: number | null;
  provider: string | null;
  model: string | null;
  capability: string | null;
  checked_at: string | null;
  controllable: boolean;
  risk_level: string;
  reason: string;
  environment: OperationsEnvironment;
  availability: "enabled" | "restricted" | "disabled" | "unknown";
  policy_status: "compliant" | "not_evaluated" | "restricted" | "unknown";
}

export interface FunctionalityHealthPayload {
  generated_at: string;
  environment: OperationsEnvironment;
  platform_health: "healthy" | "degraded" | "incident" | "unknown";
  platform_health_score: number | null;
  user_impact: UserImpact;
  counts: Record<"healthy" | "degraded" | "incident" | "maintenance" | "disabled" | "unknown", number>;
  functionalities: FunctionalityHealth[];
  attention: FunctionalityHealth[];
}

export async function getFunctionalityHealth(
  environment?: OperationsEnvironment,
): Promise<FunctionalityHealthPayload> {
  const query = environment
    ? `?environment=${encodeURIComponent(environment)}`
    : "";

  const response = await fetch(
    `/api/operations-center/functionality-health${query}`,
    { headers: { Accept: "application/json" } },
  );

  if (!response.ok) {
    throw new Error(`Functionality health API failed: ${response.status}`);
  }

  return response.json() as Promise<FunctionalityHealthPayload>;
}

import type { OperationsEnvironment } from "./functionalityHealth";

export interface FunctionalityControlResult {
  functionality_id: string;
  environment: OperationsEnvironment;
  availability: "enabled" | "disabled";
  action: "stop" | "resume";
  reason: string | null;
  changed_at: string;
  changed_by: string;
}

async function postControl(
  functionalityId: string,
  action: "stop" | "resume",
  payload: { environment: OperationsEnvironment; reason?: string },
): Promise<FunctionalityControlResult> {
  const response = await fetch(
    `/api/operations-center/functionality/${encodeURIComponent(functionalityId)}/${action}`,
    {
      method: "POST",
      headers: {
        Accept: "application/json",
        "Content-Type": "application/json",
      },
      body: JSON.stringify(payload),
    },
  );

  if (!response.ok) {
    let message = `Functionality ${action} failed: ${response.status}`;
    try {
      const body = (await response.json()) as { detail?: string };
      if (body.detail) message = body.detail;
    } catch {
      // Preserve the safe status-based error.
    }
    throw new Error(message);
  }
  return (await response.json()) as FunctionalityControlResult;
}

export function stopFunctionality(
  functionalityId: string,
  environment: OperationsEnvironment,
  reason: string,
) {
  return postControl(functionalityId, "stop", { environment, reason });
}

export function resumeFunctionality(
  functionalityId: string,
  environment: OperationsEnvironment,
) {
  return postControl(functionalityId, "resume", { environment });
}
