===== WORLD CONTEXT =====
"""
World Context.

Represents the active learning world
for an AI interaction.

Examples:

- French World
- Spanish World
- Business French World

The world determines:
- language
- learning environment
- available experience rules
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(
    frozen=True,
    slots=True,
)
class WorldContext:
    """
    Immutable learning world context.
    """

    world_id: str

    name: str

    language_code: str

    available_characters: tuple[str, ...] = ()

    def __post_init__(self) -> None:
        """
        Validate world context.
        """

        if not self.world_id.strip():
            raise ValueError(
                "world_id cannot be empty."
            )

        if not self.name.strip():
            raise ValueError(
                "world name cannot be empty."
            )

        if not self.language_code.strip():
            raise ValueError(
                "language_code cannot be empty."
            )

===== WORLD REPOSITORY =====
"""
YAML World Repository.

Infrastructure implementation for loading
World configurations from YAML files.

Implements:

WorldRepository
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

import yaml


class WorldConfiguration:
    """
    Infrastructure representation of a world.

    Validates the shape and required values of the
    loaded YAML configuration before it can be consumed
    by the application layer.
    """

    def __init__(
        self,
        data: dict[str, Any],
    ) -> None:

        if not isinstance(data, dict):
            raise ValueError(
                "World configuration must be a mapping."
            )

        self.id = self._required_string(
            data,
            "id",
            "World id cannot be empty.",
        )

        self.name = self._required_string(
            data,
            "name",
            "World name cannot be empty.",
        )

        self.language_code = self._required_string(
            data,
            "language_code",
            "Language code cannot be empty.",
        )

        available_characters = data.get(
            "available_characters",
            [],
        )

        if not isinstance(
            available_characters,
            list,
        ):
            raise ValueError(
                "available_characters must be a list."
            )

        validated_characters: list[str] = []

        for character_id in available_characters:

            if not isinstance(
                character_id,
                str,
            ):
                raise ValueError(
                    "Character id must be a string."
                )

            if not character_id.strip():
                raise ValueError(
                    "Character id cannot be empty."
                )

            validated_characters.append(
                character_id
            )

        self.available_characters = (
            validated_characters
        )

    @staticmethod
    def _required_string(
        data: dict[str, Any],
        field_name: str,
        error_message: str,
    ) -> str:
        """
        Return a required non-empty string field.
        """

        value = data.get(field_name)

        if not isinstance(value, str):
            raise ValueError(error_message)

        if not value.strip():
            raise ValueError(error_message)

        return value


class YamlWorldRepository:
    """
    Loads worlds from YAML files.
    """

    def __init__(
        self,
        *,
        worlds_path: Path,
    ) -> None:

        self._worlds_path = worlds_path

    def get_by_id(
        self,
        world_id: str,
    ) -> WorldConfiguration | None:
        """
        Load a world configuration by its identifier.
        """

        file_path = (
            self._worlds_path.resolve()
            / f"{world_id}.yaml"
        )

        if not file_path.exists():
            return None

        with file_path.open(
            "r",
            encoding="utf-8",
        ) as file:
            data = yaml.safe_load(file)

        return WorldConfiguration(data)

===== WORLD RESOLVER =====
"""
World Resolver.

Application service responsible for resolving
the learner's active AI learning world.

Responsibilities
----------------

- Retrieve world configuration.
- Validate world availability.
- Create WorldContext.

Does NOT
--------

- Execute AI models.
- Manage characters.
- Create conversations.
- Handle persistence directly.
"""

from __future__ import annotations

import logging
from typing import Protocol

from app.domain.uai.context.world_context import (
    WorldContext,
)


logger = logging.getLogger(__name__)


class WorldRepository(Protocol):
    """
    Repository contract for world retrieval.
    """

    def get_by_id(
        self,
        world_id: str,
    ):
        """
        Retrieve a world configuration.
        """
        ...


class WorldResolver:
    """
    Resolves a learning world into runtime context.
    """

    def __init__(
        self,
        *,
        world_repository: WorldRepository,
    ) -> None:
        self._world_repository = world_repository

    def resolve(
        self,
        world_id: str,
    ) -> WorldContext:
        """
        Resolve world configuration.

        Returns
        -------
        WorldContext
            Validated world interaction context.
        """

        logger.info(
            "Resolving world context",
            extra={
                "world_id": world_id,
            },
        )

        world = (
            self._world_repository
            .get_by_id(world_id)
        )

        if world is None:
            raise ValueError(
                f"World not found: {world_id}"
            )

        return WorldContext(
            world_id=world.id,
            name=world.name,
            language_code=world.language_code,
            available_characters=(
                tuple(
                    world.available_characters
                )
            ),
        )


===== WORLD CATALOGUE MODELS =====
"""Application model for the authoritative Learning World catalogue."""
from __future__ import annotations
from dataclasses import dataclass

@dataclass(frozen=True, slots=True)
class LearningWorldCatalogueEntry:
    identifier: str
    name: str
    description: str
    language: str
    version: str
    status: str
    enabled: bool
    world_type: str
    learning_domain: str
    runtime_world_id: str | None

    def __post_init__(self) -> None:
        for field_name in ("identifier", "name", "version", "status", "world_type", "learning_domain"):
            if not getattr(self, field_name).strip():
                raise ValueError(f"Learning World catalogue {field_name} cannot be empty.")
        if self.runtime_world_id is not None and not self.runtime_world_id.strip():
            raise ValueError("runtime_world_id cannot be blank when provided.")


===== WORLD CATALOGUE SERVICE =====
"""Application service for catalogue-to-runtime World resolution."""
from __future__ import annotations
from app.application.uai.world.world_resolver import WorldResolver
from .repositories import LearningWorldCatalogueRepository

class LearningWorldCatalogueService:
    """Apply catalogue policy and delegate runtime resolution to WorldResolver."""
    def __init__(self, *, repository: LearningWorldCatalogueRepository, world_resolver: WorldResolver) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")
        if world_resolver is None:
            raise ValueError("world_resolver cannot be None.")
        self._repository = repository
        self._world_resolver = world_resolver

    def get_by_identifier(self, identifier: str):
        if not identifier or not identifier.strip():
            raise ValueError("catalogue identifier is required.")
        return self._repository.get_by_identifier(identifier.strip())

    def resolve_runtime_world(self, catalogue_identifier: str):
        entry = self.get_by_identifier(catalogue_identifier)
        if entry is None:
            raise ValueError(f"Learning World not found: {catalogue_identifier.strip()}")
        if not entry.enabled:
            raise ValueError(f"Learning World is disabled: {entry.identifier}")
        if not entry.runtime_world_id:
            raise ValueError(f"Learning World has no runtime configuration: {entry.identifier}")
        return self._world_resolver.resolve(entry.runtime_world_id)
