"""
Voice Call Domain Aggregate.

Represents a persistent AI character
voice call identity.

Responsibilities:
- Maintain voice call identity.
- Track lifecycle state.
- Validate reconnect availability.
- Support disconnect/reconnect lifecycle.

Does NOT:
- Handle HTTP.
- Manage LiveKit.
- Generate tokens.
- Manage providers.
- Persist data.
"""

from __future__ import annotations


from dataclasses import dataclass, replace


from datetime import datetime, timezone


from backend.app.domain.uai.voice.voice_call_state import (
    VoiceCallState,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCall:
    """
    Persistent voice call aggregate.

    A VoiceCall represents the business
    identity of a conversation.

    Multiple execution sessions can exist
    for the same VoiceCall.
    """

    id: str

    learner_id: str

    character_id: str

    world_id: str

    language_code: str

    state: VoiceCallState

    created_at: datetime

    expires_at: datetime



    def __post_init__(
        self,
    ) -> None:
        """
        Validate aggregate invariants.
        """


        if not self.id:
            raise ValueError(
                "Voice call id is required."
            )


        if not self.learner_id:
            raise ValueError(
                "Learner id is required."
            )


        if not self.character_id:
            raise ValueError(
                "Character id is required."
            )


        if not self.world_id:
            raise ValueError(
                "World id is required."
            )


        if self.expires_at < self.created_at:
            raise ValueError(
                "Voice call expiration "
                "cannot be before creation."
            )



    def can_reconnect(
        self,
        *,
        now: datetime,
    ) -> bool:
        """
        Check whether this voice call
        can reconnect.

        Allowed states:

        ACTIVE
        DISCONNECTED
        RECONNECTABLE

        The reconnect window must
        not have expired.
        """


        now = self._ensure_utc(
            now
        )


        expires_at = self._ensure_utc(
            self.expires_at
        )


        return (
            now < expires_at
            and self.state
            in {
                VoiceCallState.ACTIVE,
                VoiceCallState.DISCONNECTED,
                VoiceCallState.RECONNECTABLE,
            }
        )



    def mark_disconnected(
        self,
    ) -> "VoiceCall":
        """
        Mark current execution as
        temporarily disconnected.
        """


        return replace(
            self,
            state=VoiceCallState.DISCONNECTED,
        )



    def mark_reconnectable(
        self,
    ) -> "VoiceCall":
        """
        Mark voice call as available
        for reconnection.
        """


        return replace(
            self,
            state=VoiceCallState.RECONNECTABLE,
        )



    def mark_active(
        self,
    ) -> "VoiceCall":
        """
        Mark voice call active again
        after reconnect.
        """


        return replace(
            self,
            state=VoiceCallState.ACTIVE,
        )



    def mark_completed(
        self,
    ) -> "VoiceCall":
        """
        Complete voice call lifecycle.
        """


        return replace(
            self,
            state=VoiceCallState.COMPLETED,
        )



    def mark_failed(
        self,
    ) -> "VoiceCall":
        """
        Mark voice call as failed.
        """


        return replace(
            self,
            state=VoiceCallState.FAILED,
        )



    @staticmethod
    def _ensure_utc(
        value: datetime,
    ) -> datetime:
        """
        Normalize datetime values.

        Existing code may still create
        naive datetime values.

        Domain boundary normalizes them
        to UTC.
        """


        if value.tzinfo is None:

            return value.replace(
                tzinfo=timezone.utc
            )


        return value.astimezone(
            timezone.utc
        )



__all__ = [
    "VoiceCall",
]

    """

    character_id: str

    world_id: str

    language_code: str

    state: VoiceCallState = (

            raise ValueError(

            raise ValueError(
                "language_code cannot be empty."
            )

        """
        Return whether call is active.

        return (
            self.state

"""
Voice Call Room Domain Model.

Represents the logical room composition
for a voice call.

Responsibilities:
- Define room identity.
- Hold call participants.
- Validate participant composition.

Does NOT:
- Connect to LiveKit.
- Manage audio streams.
- Execute AI.
- Manage providers.
- Handle networking.
"""

from __future__ import annotations


from dataclasses import dataclass


from backend.app.domain.uai.voice.voice_call_participant import (
    VoiceCallParticipant,
    VoiceParticipantRole,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallRoom:
    """
    Immutable voice call room.

    Supports:

    Beta:
        Learner + Character

    Future:
        Learner + Learner + Character
    """

    room_id: str

    participants: tuple[
        VoiceCallParticipant,
        ...

    ]


    def __post_init__(self) -> None:
        """
        Validate room composition.
        """

        if not self.room_id.strip():
            raise ValueError(
                "room_id cannot be empty."
            )


        if not self.participants:
            raise ValueError(
                "Voice call room requires "
                "at least one participant."
            )


        character_count = sum(
            1
            for participant in self.participants
            if participant.role
            ==
            VoiceParticipantRole.CHARACTER
        )


        if character_count > 1:
            raise ValueError(
                "Voice call room supports "
                "only one character participant."
            )


    def has_character(self) -> bool:
        """
        Check whether a character exists
        in the room.
        """

        return any(
            participant.role
            ==
            VoiceParticipantRole.CHARACTER
            for participant in self.participants
        )


    def learner_count(self) -> int:
        """
        Return number of learners.
        """

        return sum(
            1
            for participant in self.participants
            if participant.role
            ==
            VoiceParticipantRole.LEARNER
        )

Coordinates the preparation of a voice call.

Responsibilities:

from __future__ import annotations

from dataclasses import dataclass

from backend.app.application.uai.context.interaction_context_builder import (
    InteractionContextBuilder,
)
from backend.app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)
from backend.app.application.uai.voice.voice_character_session_service import (
    VoiceCharacterSessionService,
)

from backend.app.ai_runtime.voice.voice_call_runtime import (
    VoiceCallRuntime,
)
from backend.app.domain.uai.context.learner_context import (
    LearnerContext,
)
from backend.app.domain.uai.context.voice_context import (
    VoiceContext,
)
from backend.app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)


@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallOrchestrator:
    """
    Voice call workflow coordinator.
    """

    context_builder: InteractionContextBuilder

    session_service: VoiceCharacterSessionService

    voice_runtime: VoiceCallRuntime = (
        VoiceCallRuntime()
    )

    def prepare_call(
        self,
        *,
        request: VoiceCallRequest,
    ) -> VoiceCallSession:
        """
        Prepare voice call session.

        Flow:

        Request
          |
          ↓
        Interaction Context
          |
          ↓
        Character Voice Session
          |
          ↓
        Voice Call Session
          |
          ↓
        Runtime Preparation
          |
          ↓
        READY state
        """

        learner = LearnerContext(
            learner_id=request.learner_id,
            learning_goals=[
                "conversation",
            ],
        )

        voice = VoiceContext(
            language_code=request.language_code,
            voice_provider="openai",
            voice_profile=request.character_id,
        )

        context = (
            self.context_builder
            .build(
                learner=learner,
                world_id=request.world_id,
                character_id=request.character_id,
                voice=voice,
            )
        )

        voice_session = (
            self.session_service
            .create_session(
                context=context,
            )
        )

        call_session = VoiceCallSession(
            session_id=(
                voice_session.session_id
            ),
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
        )

        return (
            self.voice_runtime
            .prepare(
                session=call_session,
            )
        )

    """

        self,
        *,
        voice_session_service: VoiceSessionService,
        livekit_adapter: LiveKitAdapter,

"""

Application facade for the complete

from __future__ import annotations

from dataclasses import dataclass

from backend.app.application.uai.voice.voice_call_execution_coordinator import (

from backend.app.application.uai.voice.voice_connection_service import (

)


@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallLifecycleService:
    """
    Application lifecycle facade.
    """

    execution_coordinator: (
        VoiceCallExecutionCoordinator
    )

    connection_service: (
        VoiceConnectionService | None
    ) = None

    voice_call_management_service: (
        VoiceCallManagementService | None
    ) = None

    voice_call_session_service: (
        VoiceCallSessionService | None
    ) = None

    voice_call_event_service: (
        VoiceCallEventService | None
    ) = None

    def start_call(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> dict:
        """
        Prepare room, execute call,
        create connection and record event.

        Identity rules
        --------------
        VoiceCall.id:
            Persistent business/lifecycle identity.

        VoiceCallSession.session_id:
            Execution/session identity.

        Lifecycle events must use the persistent
        VoiceCall.id when a VoiceCall is available.
        """

        result = (
            self.execution_coordinator
            .start(
                context=context,
            )
        )

        session = result["session"]

        if self.voice_call_event_service is not None:
            voice_call_id = (
                context.voice_call.id
                if context.voice_call is not None
                else session.session_id
            )

            self.voice_call_event_service.record_event(
                voice_call_id=voice_call_id,
                learner_id=session.learner_id,
                character_id=session.character_id,
                event_type=VoiceCallEventType.STARTED,
            )

        if self.connection_service is not None:
            connection = (
                self.connection_service
                .create_connection(
                    room_name=(
                        result["room"]["room_name"]
                    ),
                )
            )

            result["connection"] = connection

        return result

    def create_voice_call(
        self,
        *,
        learner_id: str,
        character_id: str,
        world_id: str,
        language_code: str,
    ) -> VoiceCall:
        """
        Create persistent voice identity.
        """

        if (
            self.voice_call_management_service
            is None
        ):
            raise RuntimeError(
                "Voice call management service "
                "not configured."
            )

        return (
            self.voice_call_management_service
            .create_call(
                learner_id=learner_id,
                character_id=character_id,
                world_id=world_id,
                language_code=language_code,
            )
        )

    def create_session(
        self,
        *,
        voice_call: VoiceCall,
    ) -> VoiceCallSession:
        """
        Create execution session.
        """

        if (
            self.voice_call_session_service
            is None
        ):
            raise RuntimeError(
                "Voice call session service "
                "not configured."
            )

        return (
            self.voice_call_session_service
            .create_session(
                voice_call=voice_call,
            )

            learner_id=voice_call.learner_id,
            character_id=voice_call.character_id,

                VoiceCallEventType.ACCEPTED

from __future__ import annotations

@dataclass(frozen=True, slots=True)
class VoiceCallManagementService:
    """Application service for voice-call lifecycle management."""

    repository: VoiceCallRepository

    def create_call(
        self,
        *,
        learner_id: str,
        character_id: str,
        world_id: str,
        language_code: str,
    ) -> VoiceCall:
        """
        Create and persist a new active voice call.

        VoiceCall owns the lifecycle state and invariants.
        The application service is responsible for creation,
        persistence and orchestration only.
        """

        now = datetime.now(timezone.utc)

        voice_call = VoiceCall(
            id=f"vc-{uuid4()}",
            learner_id=learner_id,
            character_id=character_id,
            world_id=world_id,
            language_code=language_code,
            state=VoiceCallState.CREATED,
            created_at=now,
            expires_at=now + timedelta(minutes=30),
        )

        return self.repository.save(
            voice_call=voice_call,
        )

    def get_call(
        self,
        *,
        voice_call_id: str,
    ) -> VoiceCall | None:
        """Retrieve a voice call by persistent identity."""

        return self.repository.get_by_id(
            voice_call_id=voice_call_id,
        )

    def can_reconnect(
        self,
        *,
        voice_call_id: str,
        now: datetime | None = None,
    ) -> bool:
        """Determine whether an existing call can reconnect."""

        voice_call = self.get_call(
            voice_call_id=voice_call_id,
        )

        if voice_call is None:
            return False

        current_time = (
            now
            if now is not None
            else datetime.now(timezone.utc)
        )

        return voice_call.can_reconnect(
            now=current_time,
        )

    def mark_disconnected(
        self,
        *,
        voice_call_id: str,
    ) -> VoiceCall:
        """Mark an existing voice call as disconnected."""

        voice_call = self.get_call(
            voice_call_id=voice_call_id,
        )

        if voice_call is None:
            raise ValueError(
                f"Unknown voice call: {voice_call_id}"
            )

        disconnected_call = (
            voice_call.mark_disconnected()
        )

        return self.repository.save(
            voice_call=disconnected_call,
        )

    def complete_call(
        self,
        *,
        voice_call_id: str,
    ) -> VoiceCall:
        """Complete an existing voice call."""

        voice_call = self.get_call(
            voice_call_id=voice_call_id,
        )

        if voice_call is None:
            raise ValueError(
                f"Unknown voice call: {voice_call_id}"
            )

        completed_call = (
            voice_call.mark_completed()
        )

        return self.repository.save(
            voice_call=completed_call,
        )
