"""

- Create voice session.
- Prepare voice infrastructure.
- Return call configuration.

Does NOT:
- Run AI agents.
- Handle audio streams.
- Manage LiveKit lifecycle.
"""

from __future__ import annotations

import logging

from backend.app.application.uai.voice.voice_session_service import (
    VoiceSessionService,

    InteractionContext,

    LiveKitAdapter,
)

class VoiceSessionOrchestrator:
    """
    Coordinates voice session preparation.
    """

    def __init__(
        self,
        *,
        voice_session_service: VoiceSessionService,
        livekit_adapter: LiveKitAdapter,
    ) -> None:

        self._voice_session_service = (
            voice_session_service
        )

        self._livekit_adapter = (
            livekit_adapter
        )


    def prepare_session(
        self,
        *,
        interaction_context: InteractionContext,
    ) -> dict[str, str]:
        """
        Prepare a voice session.

        Returns provider configuration
        required to start a call.
        """

        session = (
            self._voice_session_service
            .create_session(
                interaction_context=interaction_context,
            )
        )


        configuration = (
            self._livekit_adapter
            .create_room_configuration(
                session=session,
            )
        )


        logger.info(
            "Voice session prepared",
            extra={
                "session_id": session.session_id,
            },
        )


        return configuration

"""
Voice Call Orchestrator.

Coordinates the preparation of a voice call.

Responsibilities:
- Build interaction context.
- Create character voice session.
- Create domain voice call session.
- Apply voice call runtime lifecycle.

Does NOT:
- Execute AI providers.
- Manage LiveKit.
- Handle HTTP.
- Persist data.
"""

from __future__ import annotations

from dataclasses import dataclass

from backend.app.application.uai.context.interaction_context_builder import (
    InteractionContextBuilder,
)
from backend.app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)
from backend.app.application.uai.voice.voice_character_session_service import (
    VoiceCharacterSessionService,
)

from backend.app.ai_runtime.voice.voice_call_runtime import (
    VoiceCallRuntime,
)
from backend.app.domain.uai.context.learner_context import (
    LearnerContext,

from backend.app.domain.uai.context.voice_context import (
    VoiceContext,
)
from backend.app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)


@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallOrchestrator:
    """
    Voice call workflow coordinator.
    """

    context_builder: InteractionContextBuilder

    session_service: VoiceCharacterSessionService

    voice_runtime: VoiceCallRuntime = (
        VoiceCallRuntime()
    )

    def prepare_call(
        self,
        *,
        request: VoiceCallRequest,
    ) -> VoiceCallSession:
        """
        Prepare voice call session.

        Flow:

        Request
          |
          ↓
        Interaction Context
          |
          ↓
        Character Voice Session
          |
          ↓
        Voice Call Session
          |
          ↓
        Runtime Preparation
          |
          ↓
        READY state
        """

        learner = LearnerContext(
            learner_id=request.learner_id,
            learning_goals=[
                "conversation",
            ],
        )

        voice = VoiceContext(
            language_code=request.language_code,
            voice_provider="openai",
            voice_profile=request.character_id,
        )

        context = (
            self.context_builder
            .build(
                learner=learner,
                world_id=request.world_id,
                character_id=request.character_id,
                voice=voice,
            )
        )

        voice_session = (
            self.session_service
            .create_session(
                context=context,
            )
        )

        call_session = VoiceCallSession(
            session_id=(
                voice_session.session_id
            ),
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
        )

        return (
            self.voice_runtime
            .prepare(
                session=call_session,
            )
        )

class VoiceSessionService:
    """
    Creates voice sessions from interaction context.
    """

    def create_session(
        self,
        *,
        interaction_context: InteractionContext,
    ) -> VoiceSession:
        """
        Create a new voice session.

        Parameters
        ----------
        interaction_context:
            Complete UAI interaction context.

        Returns
        -------
        VoiceSession
            Created voice session.
        """

        session_id = (
            str(
                uuid.uuid4()
            )
        )


        character_id = (
            interaction_context
            .character
            .character_id
        )


        language_code = (
            interaction_context
            .world
            .language_code
        )


        room_name = (
            f"{character_id}-{session_id[:8]}"
        )


        session = VoiceSession(
            session_id=session_id,
            room_name=room_name,
            character_id=character_id,
            language_code=language_code,
        )


        logger.info(
            "Voice session created",
            extra={
                "session_id": session_id,
                "character_id": character_id,
            },
        )


        return session

    context_builder = InteractionContextBuilder(

        livekit_adapter=livekit_adapter,
        speech_runtime=speech_runtime,
    )

        context_builder=context_builder,
        session_service=session_service,
    )


    #
    # Keep the interaction-context bridge available
    # through the composition root for realtime voice
    # infrastructure.
    #
    # The existing VoiceCallService contract remains
    # unchanged.
    #

    _ = interaction_context_service


    return VoiceCallService(
        orchestrator=orchestrator,
    )
