"""
EasyChatBot AI Operations Center provider health probe runner.

Coordinates provider-independent health probes with the
existing ProviderHealthApplicationService.

Architecture:

ProviderHealthProbe
        |
        v
ProviderHealthProbeRunner
        |
        v
ProviderHealthApplicationService
        |
        v
ProviderHealthMonitorService
        |
        v
Operations Center

The runner is orchestration only.

It does NOT:

- implement provider-specific API calls;
- calculate health state;
- create incidents directly;
- modify Runtime routing;
- select alternative providers;
- perform autonomous remediation.
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone
from time import perf_counter
from typing import Iterable

from app.application.provider_management.provider_health_application_service import (
    ProviderHealthApplicationResult,
    ProviderHealthApplicationService,
)
from app.application.provider_management.provider_health_models import (
    ProviderHealthObservation,
    ProviderHealthStatus,
)
from app.application.provider_management.provider_health_probe import (
    ProviderHealthProbe,
)


@dataclass(frozen=True)
class ProviderHealthProbeResult:
    """
    Result of processing one provider health probe.

    Contains both the raw provider-independent observation
    and the application result produced by the existing
    health application boundary.
    """

    observation: ProviderHealthObservation

    application_result: ProviderHealthApplicationResult


class ProviderHealthProbeRunner:
    """
    Coordinates provider health probes.

    The runner receives probes through dependency injection.
    It never constructs provider-specific clients itself.
    """

    def __init__(
        self,
        *,
        health_application_service: (
            ProviderHealthApplicationService
        ),
        probes: Iterable[ProviderHealthProbe],
    ) -> None:
        """
        Initialise the probe runner.
        """

        if health_application_service is None:
            raise ValueError(
                "Provider health application service "
                "cannot be None."
            )

        if probes is None:
            raise ValueError(
                "Provider health probes cannot be None."
            )

        self._health_application_service = (
            health_application_service
        )

        self._probes = tuple(probes)

        self._validate_probes()

    async def run_probe(
        self,
        probe: ProviderHealthProbe,
    ) -> ProviderHealthProbeResult:
        """
        Execute one health probe and submit its observation
        through the existing application boundary.
        """

        if probe is None:
            raise ValueError(
                "Provider health probe cannot be None."
            )

        started = perf_counter()

        try:
            observation = await probe.check()

        except Exception as exc:
            elapsed_ms = int(
                round(
                    (perf_counter() - started)
                    * 1000
                )
            )

            observation = (
                self._build_failure_observation(
                    probe=probe,
                    latency_ms=elapsed_ms,
                    error=exc,
                )
            )

        if observation.provider != probe.provider:
            raise ValueError(
                "Provider health probe returned an "
                "observation for a different provider."
            )

        result = (
            self._health_application_service.observe(
                observation
            )
        )

        return ProviderHealthProbeResult(
            observation=observation,
            application_result=result,
        )

    async def run_all(
        self,
    ) -> tuple[ProviderHealthProbeResult, ...]:
        """
        Execute all configured probes.

        Probes are executed sequentially for deterministic
        application behaviour and predictable provider load.

        Parallel execution can be introduced later only if
        measurement demonstrates that it is required.
        """

        results: list[
            ProviderHealthProbeResult
        ] = []

        for probe in self._probes:
            results.append(
                await self.run_probe(probe)
            )

        return tuple(results)

    def _validate_probes(self) -> None:
        """
        Validate configured probe identities.
        """

        providers: set[str] = set()

        for probe in self._probes:
            if probe is None:
                raise ValueError(
                    "Provider health probes cannot contain None."
                )

            provider = probe.provider.strip()

            if not provider:
                raise ValueError(
                    "Provider health probe provider "
                    "cannot be empty."
                )

            if provider in providers:
                raise ValueError(
                    "Duplicate provider health probe "
                    f"configured: {provider}."
                )

            providers.add(provider)

    @staticmethod
    def _build_failure_observation(
        *,
        probe: ProviderHealthProbe,
        latency_ms: int,
        error: Exception,
    ) -> ProviderHealthObservation:
        """
        Convert a probe exception into a provider-independent
        unavailable observation.

        Provider-specific exceptions do not cross the
        application boundary.
        """

        now = datetime.now(
            timezone.utc
        )

        return ProviderHealthObservation(
            provider=probe.provider,
            model=probe.model,
            capability=probe.capability,
            status=ProviderHealthStatus.UNAVAILABLE,
            observed_at=now,
            latency_ms=max(
                latency_ms,
                0,
            ),
            error_type=type(error).__name__,
            error_message=str(error),
            consecutive_failures=1,
            consecutive_successes=0,
        )

"""
EasyChatBot AI Operations Center provider health monitor.

Coordinates provider health observations and their
read-only operational projections.

Architecture:

Provider Health Observation
        |
        v
ProviderHealthService
        |
        v
ProviderHealthMonitorService
        |
        v
ProviderHealthReadModel
        |
        v
Operations Center


Responsibilities:

- accept provider health observations;
- delegate health calculation to ProviderHealthService;
- retain the latest observation metadata;
- expose dashboard-safe provider health read models;
- expose all currently observed providers.

Does NOT:

- execute providers;
- perform health probes;
- persist incidents;
- send emails;
- change Runtime routing;
- approve recommendations;
- perform autonomous failover.

ProviderHealthService remains the authority for health
state calculation.
"""

from __future__ import annotations

from app.application.provider_management.provider_health_models import (
    ProviderHealthObservation,
    ProviderHealthState,
    ProviderHealthTransitionEvent,
)

from app.application.provider_management.provider_health_read_model import (
    ProviderHealthReadModel,
    ProviderHealthReadModelService,
)

from app.application.provider_management.provider_health_service import (
    ProviderHealthService,
)


class ProviderHealthMonitorService:
    """
    Coordinates provider health state and operational
    read-model generation.

    The service deliberately keeps the latest observation
    separately from calculated health state because the
    current ProviderHealthState contract does not contain
    model or capability metadata.
    """

    def __init__(
        self,
        *,
        health_service: ProviderHealthService | None = None,
        read_model_service: (
            ProviderHealthReadModelService | None
        ) = None,
    ) -> None:
        """
        Initialise the provider health monitor.
        """

        self._health_service = (
            health_service
            if health_service is not None
            else ProviderHealthService()
        )

        self._read_model_service = (
            read_model_service
            if read_model_service is not None
            else ProviderHealthReadModelService()
        )

        self._latest_observations: dict[
            str,
            ProviderHealthObservation,
        ] = {}

    def observe(
        self,
        observation: ProviderHealthObservation,
    ) -> ProviderHealthTransitionEvent:
        """
        Process a provider health observation.

        Health calculation remains delegated to
        ProviderHealthService.
        """

        if observation is None:
            raise ValueError(
                "Provider health observation cannot be None."
            )

        transition = self._health_service.observe(
            observation
        )

        self._latest_observations[
            observation.provider
        ] = observation

        return transition

    def get_provider_health(
        self,
        provider: str,
    ) -> ProviderHealthReadModel | None:
        """
        Return the current operational health projection
        for one provider.
        """

        if not provider.strip():
            raise ValueError(
                "Provider cannot be empty."
            )

        state = self._health_service.get_state(
            provider
        )

        observation = self._latest_observations.get(
            provider
        )

        if (
            state is None
            or observation is None
        ):
            return None

        return self._read_model_service.build(
            observation=observation,
            state=state,
        )

    def get_all_provider_health(
        self,
    ) -> tuple[
        ProviderHealthReadModel,
        ...
    ]:
        """
        Return operational health projections for all
        observed providers.

        Results are deterministic and sorted by provider.
        """

        results: list[
            ProviderHealthReadModel
        ] = []

        for state in (
            self._health_service.get_all_states()
        ):
            observation = (
                self._latest_observations.get(
                    state.provider
                )
            )

            if observation is None:
                continue

            results.append(
                self._read_model_service.build(
                    observation=observation,
                    state=state,
                )
            )

        return tuple(
            sorted(
                results,
                key=lambda item: item.provider,
            )
        )

    def clear(self) -> None:
        """
        Clear tracked health state and observations.

        Intended for isolated tests and demonstrations.
        """

        self._health_service.clear()

        self._latest_observations.clear()

"""
EasyChatBot provider health application service.

Provides the controlled application boundary for ingesting
provider health observations.

Architecture:

Provider Health Probe / Adapter
        |
        v
ProviderHealthApplicationService
        |
        +-----------------------------+
        |                             |
        v                             v
ProviderHealthMonitorService   ProviderIncidentService
        |                             |
        v                             v
Health State / Read Model       Incident Lifecycle
        |                             |
        +-------------+---------------+
                      |
                      v
               Operations Center


Responsibilities:

- accept explicit provider health observations;
- delegate health processing to the existing monitor;
- preserve ProviderHealthService as the authority for
  health-state calculation;
- process operational incident transitions;
- return a stable application result;
- keep provider probes and HTTP concerns outside this service.

Does NOT:

- execute providers;
- perform health probes;
- change Runtime routing;
- select alternative providers;
- send notifications directly;
- approve operational actions;
- modify infrastructure;
- perform autonomous remediation.

This is an application orchestration service.
"""

from __future__ import annotations

from dataclasses import dataclass

from app.application.provider_management.provider_health_models import (
    ProviderHealthObservation,
    ProviderHealthStatus,
    ProviderHealthTransition,
)

from app.application.provider_management.provider_health_monitor_service import (
    ProviderHealthMonitorService,
)

from app.application.provider_management.provider_incident_service import (
    ProviderIncident,
    ProviderIncidentService,
)


@dataclass(frozen=True)
class ProviderHealthApplicationResult:
    """
    Stable result returned after processing a provider health
    observation.
    """

    provider: str

    model: str

    capability: str

    status: ProviderHealthStatus

    transition: ProviderHealthTransition

    incident: ProviderIncident | None


class ProviderHealthApplicationService:
    """
    Application boundary for provider health ingestion.

    The service coordinates the existing provider health
    monitor and provider incident service.

    It deliberately does not create a second
    ProviderHealthService. The monitor remains the single
    owner of provider health state.
    """

    def __init__(
        self,
        *,
        health_monitor: ProviderHealthMonitorService,
        incident_service: ProviderIncidentService,
    ) -> None:
        """
        Initialise the provider health application service.
        """

        if health_monitor is None:
            raise ValueError(
                "Provider health monitor "
                "cannot be None."
            )

        if incident_service is None:
            raise ValueError(
                "Provider incident service "
                "cannot be None."
            )

        self._health_monitor = (
            health_monitor
        )

        self._incident_service = (
            incident_service
        )

    def observe(
        self,
        observation: ProviderHealthObservation,
    ) -> ProviderHealthApplicationResult:
        """
        Process one provider health observation.

        Health calculation and provider read-model state remain
        owned by ProviderHealthMonitorService.

        Incident lifecycle remains owned by
        ProviderIncidentService.
        """

        if observation is None:
            raise ValueError(
                "Provider health observation "
                "cannot be None."
            )

        transition = (
            self._health_monitor
            .observe(observation)
        )

        incident = (
            self._incident_service
            .process_transition(transition)
        )

        return ProviderHealthApplicationResult(
            provider=observation.provider,
            model=observation.model,
            capability=observation.capability,
            status=observation.status,
            transition=transition.transition,
            incident=incident,
        )
