registered on the existing scheduler. No second scheduler is created.

def create_cybersecurity_scheduler_runtime(
    workflow_service=None,
    *,
    academy_one_week_workflow=None,
) -> CybersecuritySchedulerRuntime:
    """Create the single shared scheduler runtime."""

    if workflow_service is None:
        email_service = CybersecurityEmailService(
            recipient="davdbarbero1999@gmail.com",
        )

        workflow_service = CybersecuritySchedulerWorkflow(
            email_service=email_service,
        )

    scheduler = AsyncIOScheduler(
        timezone=SCHEDULER_TIMEZONE,
    )

    # Academy production composition is explicit and injectable.
    #
    # It is deliberately NOT created here automatically because the
    # application import path must remain safe when PostgreSQL is not
    # configured in a development/test environment.
    #
    # Production composition may inject the completed Academy workflow.

    adapter = CybersecurityAPSchedulerAdapter(
        scheduler=scheduler,
        workflow_service=workflow_service,
        academy_one_week_workflow=academy_one_week_workflow,
    )

    return CybersecuritySchedulerRuntime(
        scheduler_adapter=adapter,
    )


__all__ = [
    "create_cybersecurity_scheduler_runtime",
]

class DisabledAcademyTelegramSender:
    """Safe default when no existing Telegram Bot is supplied.

    Academy email delivery remains independently testable.
    No second Telegram Bot is created.
    """

    async def send(
        self,
        *,
        chat_id: str,
        text: str,
        portal_url: str,
    ) -> dict:
        return {
            "provider": "telegram",
            "status": "not_configured",
            "message_id": None,
        }


def _required_config(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for Academy production composition."
        )

    return value


def create_academy_one_week_workflow(
    *,
    session_factory=None,
    email_provider_manager: EmailProviderManager | None = None,
    telegram_bot=None,
    portal_url: str | None = None,
    registration_email: str | None = None,
) -> AcademyOneWeekWorkflow:
    """Compose the production Academy one-week workflow.

    Composition:

        PostgreSQL
            ↓
        AcademyOneWeekWorkflow
            ↓
        OneWeekNotificationOrchestrator
            ↓
        EmailProviderManager
            ↓
        Existing Telegram Bot adapter

    The scheduler itself is intentionally not created here.
    """

    if session_factory is None:
        engine = create_async_database_engine()
        session_factory = create_async_session_factory(engine)

    resolved_portal_url = (
        portal_url.strip()
        if portal_url is not None
        else _required_config("EASYCHATBOT_PORTAL_URL")
    )

    resolved_registration_email = (
        registration_email.strip()
        if registration_email is not None
        else _required_config("EASYCHATBOT_REGISTRATION_EMAIL")
    )

    email_sender = AcademyEmailNotificationSender(
        provider_manager=email_provider_manager
        or EmailProviderManager()
    )

    if telegram_bot is not None:
        telegram_sender: TelegramNotificationSender = (
            ExistingBotTelegramNotificationSender(
                telegram_bot,
            )
        )
    else:
        telegram_sender = DisabledAcademyTelegramSender()

    notifications = OneWeekNotificationOrchestrator(
        email_sender=email_sender,
        telegram_sender=telegram_sender,
        email_formatter=AcademyEmailFormatter(),
        telegram_formatter=AcademyTelegramFormatter(),
        registration_email=resolved_registration_email,
        portal_url=resolved_portal_url,
    )

    users = PostgreSQLAcademyUserQuery(
        session_factory,
    )

    deliveries = PostgreSQLAcademyDeliveryRepository(
        session_factory,
    )

    return AcademyOneWeekWorkflow(
        users=users,
        deliveries=deliveries,
        notifications=notifications,
    )


__all__ = [
    "create_academy_one_week_workflow",
]

from unittest.mock import MagicMock

from backend.cybersecurity.scheduler.cybersecurity_scheduler_factory import (
    create_cybersecurity_scheduler_runtime,
)


def test_scheduler_factory_accepts_existing_academy_workflow():
    academy_workflow = MagicMock()

    runtime = create_cybersecurity_scheduler_runtime(
        workflow_service=MagicMock(),
        academy_one_week_workflow=academy_workflow,
    )

    adapter = runtime.scheduler_adapter

    assert adapter.academy_one_week_workflow is academy_workflow


def test_scheduler_factory_creates_one_shared_scheduler():
    academy_workflow = MagicMock()

    runtime = create_cybersecurity_scheduler_runtime(
        workflow_service=MagicMock(),
        academy_one_week_workflow=academy_workflow,
    )

    adapter = runtime.scheduler_adapter

    assert adapter.scheduler is not None
    assert adapter.academy_one_week_workflow is academy_workflow
    assert adapter.scheduler.running is False
