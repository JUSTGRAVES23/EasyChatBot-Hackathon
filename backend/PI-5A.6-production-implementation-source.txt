def _required_environment(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for authentication email composition."
        )

    return value


def _create_announcement_components():
    telegram_client = TelegramAnnouncementClient()

    announcement_publisher = TelegramAnnouncementPublisher(
        telegram_client=telegram_client,
    )

    return (
        telegram_client,
        announcement_publisher,
    )


@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )


def create_application() -> FastAPI:
    academy_config = (
        AcademyActivationConfig.from_environment()
    )

    auth_engine = create_async_database_engine()

    auth_session_factory = (
        create_async_session_factory(
            auth_engine,
        )
    )

    auth_repository = (
        PostgreSQLAuthenticationRepository(
            auth_session_factory,
        )
    )

    email_provider_manager = EmailProviderManager()

    authentication_email_sender = AuthenticationEmailSenderAdapter(
        provider_manager=email_provider_manager,
    )

    authentication_email_service = AuthenticationEmailService(
        authentication_email_sender,
        frontend_base_url=_required_environment(
            "EASYCHATBOT_PORTAL_URL",
        ),
    )

    authentication_service = AuthenticationService(
        auth_repository,
        email_service=authentication_email_service,
    )

    # ------------------------------------------------------------------
    # Developer Administration
    #
    # Production path:
    #
    #   DeveloperAdminService
    #          |
    #          +--> DeveloperAdminUnitOfWorkFactory
    #                  |
    #                  +--> fresh PostgreSQLDeveloperAdminUnitOfWork
    #                          |
    #                          +--> fresh AsyncSession
    #                          |
    #                          +--> fresh transaction
    #                          |
    #                          +--> auth_user_roles
    #                          |
    #                          +--> developer_admin_audit_events
    #          |
    #          +--> DeveloperAdminAuditService
    #
    # Each Developer Admin mutation receives a fresh Unit-of-Work
    # instance. The role mutation and its security audit evidence
    # therefore remain within the same database transaction while
    # mutable UoW/session state is never shared between operations.
    #
    # The standalone audit repository owns transactions only when its
    # standalone ``record`` operation is used. The production Developer
    # Admin mutation uses ``record_in_session`` through the UoW.
    # ------------------------------------------------------------------

    developer_admin_unit_of_work_factory = (
        PostgreSQLDeveloperAdminUnitOfWorkFactory(
            auth_session_factory,
        )
    )

    developer_admin_audit_service = (
        DeveloperAdminAuditService()
    )

    developer_admin_service = DeveloperAdminService(
        repository=auth_repository,
        unit_of_work_factory=developer_admin_unit_of_work_factory,
        audit_service=developer_admin_audit_service,
    )

    # ------------------------------------------------------------------
    # W15.2.1 Transaction Authorization
    #
    # Transaction authorization is composed once at the application root.
    # The registry and consumer are application-lifetime components and
    # provide their own synchronization boundaries.
    #
    # Authentication assurance remains a separate trusted-evidence
    # boundary. This composition does not manufacture MFA/A2 evidence.
    # ------------------------------------------------------------------

    authorization_service = AuthorizationService()

    transaction_authorization_registry = (
        TransactionAuthorizationRegistry()
    )

    transaction_authorization_consumer = (
        TransactionAuthorizationConsumer(
            transaction_authorization_registry,
        )
    )

    transaction_authorization_service = (
        TransactionAuthorizationApplicationService(
            authorization_service=authorization_service,
            registry=transaction_authorization_registry,
            consumer=transaction_authorization_consumer,
        )
    )

    trusted_clock = SystemTrustedClock()

    developer_admin_transaction_service = (
        DeveloperAdminTransactionService(
            developer_admin_service=developer_admin_service,
            transaction_authorization_service=(
                transaction_authorization_service
            ),
            clock=trusted_clock,
        )
    )

    # ------------------------------------------------------------------
    # Provider usage / synchronous application persistence
    #
    # The provider-usage repository is synchronous by contract.
    # It therefore receives a synchronous engine/session factory.
    #
    # It must not receive the async authentication session factory.
    # ------------------------------------------------------------------

    provider_usage_engine = (
        create_database_engine()
    )

    persistence_diagnostics_service = PersistenceDiagnosticsService(
        metadata=Base.metadata,
        project_root=Path(__file__).resolve().parents[1],
        engine=provider_usage_engine,
    )

    provider_usage_session_factory = (
        create_session_factory(
            provider_usage_engine,
        )
    )

    # ------------------------------------------------------------------
    # P5B.6.2B.2 - User Policy Profile Persistence
    #
    # Reuse the existing synchronous application persistence boundary.
    # The policy-profile repository is synchronous by contract and must
    # not receive the async authentication session factory.
    # ------------------------------------------------------------------
    user_policy_profile_repository = (
        PostgreSQLUserPolicyProfileRepository(
            provider_usage_session_factory,
        )
    )

    # ------------------------------------------------------------------
    # Q7 / M4 - Questionnaire Runtime
    #
    # Reuse the existing synchronous application persistence boundary.
    # The runtime composition owns the assessment repositories and
    # AssessmentService wiring; the API remains responsible for the
    # authenticated learner boundary.
    # ------------------------------------------------------------------

    questionnaire_runtime = create_questionnaire_runtime(
        session_factory=provider_usage_session_factory,
    )

    # ------------------------------------------------------------------
    # M2-C.2 / Authenticated Private Chat
    #
    # The ConversationRuntime remains the shared AI-runtime boundary.
    # Private-chat persistence receives a request-scoped SQLAlchemy session
    # through the composition factory; no session is shared between requests.
    # ------------------------------------------------------------------

    private_chat_runtime = create_conversation_runtime()

    private_chat_character_instruction = os.getenv(
        "EASYCHATBOT_PRIVATE_CHAT_CHARACTER_INSTRUCTION",
        (
            "You are a helpful EasyChatBot learning character. "
            "Stay within the selected Learning World and follow its "
            "application-controlled language policy. "

            "or other application-controlled settings."

    private_chat_service_factory = build_private_chat_service_factory(

    # ------------------------------------------------------------------

    learner_context_service = build_production_learner_context_service(

    # ------------------------------------------------------------------
    # M2-C5 / Study Materials

    app_root = Path(__file__).resolve().parents[2]

    world_repository = YamlWorldRepository(

    # ------------------------------------------------------------------

from __future__ import annotations

from dataclasses import asdict
from typing import Any

from app.application.operations_center.platform_intelligence_authoritative_state import (
    PlatformIntelligenceAuthoritativeStateRepository,
)
from app.application.operations_center.platform_intelligence_escalation_retry_orchestrator import (
    PlatformIntelligenceRetryStateReader,
)
from app.application.operations_center.platform_intelligence_escalation_retry_orchestration import (
    PlatformIntelligenceRetryState,
)
from app.application.operations_center.platform_intelligence_review_source import (
    PlatformIntelligenceReviewSource,
)


class PostgreSQLPlatformIntelligenceReviewSource(
    PlatformIntelligenceReviewSource,
    PlatformIntelligenceRetryStateReader,
):
    """
    Production read adapter over authoritative Platform Intelligence state.

    Architecture boundary:

        PostgreSQL authoritative PI state
                    |
                    v
        PostgreSQLPlatformIntelligenceReviewSource
                    |
              +-----+------+
              |            |
              v            v
        Review projection  Retry state
              |            |
              v            v
        Review API     Retry Orchestrator

    Responsibilities:
    - read authoritative PI state;
    - expose the governed review projection;
    - expose recommended escalations;
    - rehydrate the existing retry-state contract.

    Non-responsibilities:
    - no retry policy;
    - no dispatch;
    - no email sending;
    - no delivery persistence;
    - no scheduler;
    - no mutation of risk/review state.
    """

    def __init__(
        self,
        repository: PlatformIntelligenceAuthoritativeStateRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository must not be None.")

        self._repository = repository

    # ------------------------------------------------------------------
    # Existing ReviewSource synchronous contract
    # ------------------------------------------------------------------
    #
    # The current ReviewSource application boundary is synchronous,
    # while authoritative persistence is asynchronous.
    #
    # The async methods below are therefore the production persistence
    # adapter entry points. These synchronous methods intentionally fail
    # closed rather than performing unsafe event-loop manipulation.
    #
    # Composition must use the async methods when binding this adapter
    # to the async PostgreSQL repository.

    def list_reviews(self) -> tuple[Any, ...]:
        raise RuntimeError(
            "Use list_reviews_async() from the async application composition."
        )

    def get_review(
        self,
        review_item_id: str,
    ) -> Any | None:
        raise RuntimeError(
            "Use get_review_async() from the async application composition."
        )

    def list_escalations(self) -> tuple[Any, ...]:
        raise RuntimeError(
            "Use list_escalations_async() from the async application composition."
        )

    # ------------------------------------------------------------------
    # Async authoritative-state API
    # ------------------------------------------------------------------

    async def list_reviews_async(
        self,
    ) -> tuple[dict[str, Any], ...]:
        states = await self._repository.list_all()

        return tuple(
            self._projection(state)
            for state in states
        )

    async def get_review_async(
        self,
        review_item_id: str,
    ) -> dict[str, Any] | None:
        if not review_item_id or not review_item_id.strip():
            raise ValueError("review_item_id must not be blank.")

        state = await self._repository.get(review_item_id)

        if state is None:
            return None

        return self._projection(state)

    async def list_escalations_async(
        self,
    ) -> tuple[dict[str, Any], ...]:
        states = await self._repository.list_all()

        return tuple(
            self._projection(state)
            for state in states
            if state.recommendation_status == "RECOMMENDED"
        )

    # ------------------------------------------------------------------
    # Retry State Reader
    # ------------------------------------------------------------------

    async def get_retry_state(
        self,
        notification_key: str,
        recommendation_id: str,
    ) -> PlatformIntelligenceRetryState | None:
        if not notification_key or not notification_key.strip():
            raise ValueError("notification_key must not be blank.")

        if not recommendation_id or not recommendation_id.strip():
            raise ValueError("recommendation_id must not be blank.")

        state = await self._repository.get_by_recommendation_id(
            recommendation_id
        )

        if state is None:
            return None

        if state.recommendation_id != recommendation_id:
            return None

        return PlatformIntelligenceRetryState(
            notification_key=notification_key,
            recommendation_id=recommendation_id,
            recommendation_status=state.recommendation_status or "",
            review_status=state.review_status,
            decision_status=state.decision_status,
            quality_gate_status=state.quality_gate_status,
            dispatch_request=self._dispatch_request(
                state,
                notification_key,
            ),
            email_request=self._email_request(
                state,
                notification_key,
            ),
        )

    # ------------------------------------------------------------------
    # Projection
    # ------------------------------------------------------------------

    @staticmethod
    def _projection(
        state: Any,
    ) -> dict[str, Any]:
        """
        Return the persisted authoritative state as the governed
        Operations Center projection.

        The aggregate is immutable, so asdict() is safe here and keeps
        this adapter free from a second projection model.
        """
        return asdict(state)

    # ------------------------------------------------------------------
    # Existing canonical dispatch contract
    # ------------------------------------------------------------------

    @staticmethod
    def _dispatch_request(
        state: Any,
        notification_key: str,
    ) -> Any:
        """
        Rehydrate the existing canonical PI escalation dispatch request.

        This adapter does not implement dispatch behaviour. It only
        reconstructs the immutable request consumed by the existing
        dispatch service.
        """
        from app.application.operations_center.platform_intelligence_escalation_dispatch import (
            PlatformIntelligenceEscalationDispatchRequest,
        )

        return PlatformIntelligenceEscalationDispatchRequest(
            notification_key=notification_key,
            recommendation_id=state.recommendation_id or "",
            risk_decision_id=state.risk_decision_id,
            review_item_id=state.review_item_id,
            risk_level=state.risk_level,
            risk_score=state.risk_score,
            quality_gate_status=state.quality_gate_status,
            review_status=state.review_status,
            decision_status=state.decision_status,
            correlation_id=notification_key,
            route=state.recommendation_route or "",
            reasons=state.recommendation_reasons,
        )

    # ------------------------------------------------------------------
    # Existing canonical email contract
    # ------------------------------------------------------------------

    @staticmethod
    def _email_request(
        state: Any,
        notification_key: str,
    ) -> Any:
        """
        Rehydrate the existing canonical PI escalation email request.

        The authoritative state contains created_at, not updated_at.
        created_at is therefore the authoritative timestamp used for
        generated_at during retry-state reconstruction.
        """
        from app.application.operations_center.platform_intelligence_escalation_email import (
            PlatformIntelligenceEscalationEmailRequest,
        )

        return PlatformIntelligenceEscalationEmailRequest(
            recommendation_id=state.recommendation_id or "",
            risk_decision_id=state.risk_decision_id,
            review_item_id=state.review_item_id,
            scope=state.scope,
            risk_level=state.risk_level,
            risk_score=state.risk_score,
            quality_gate_status=state.quality_gate_status,
            review_status=state.review_status,
            decision_status=state.decision_status,
            route=state.recommendation_route or "",
            reasons=state.recommendation_reasons,
            evidence_references=state.evidence_references,
            historical_references=state.historical_references,
            policy_id=state.policy_id,
            policy_version=state.policy_version,
            rationale=state.rationale,
            generated_at=state.created_at.isoformat(),
            notification_key=notification_key,
        )
