"""
Voice Call Event Repository Interface.

Application boundary for storing
voice lifecycle events.

Responsibilities:
- Define event persistence contract.
- Allow infrastructure replacement.

Does NOT:
- Store data directly.
- Know database technology.
- Execute workflows.
"""

from __future__ import annotations



from abc import (
    ABC,
    abstractmethod,
)



from app.domain.uai.voice.voice_call_event import (
    VoiceCallEvent,
)



class VoiceCallEventRepository(
    ABC,
):
    """
    Repository contract for
    voice call events.
    """



    @abstractmethod
    def save(
        self,
        *,
        event: VoiceCallEvent,
    ) -> VoiceCallEvent:
        """
        Persist a voice call event.
        """

        raise NotImplementedError



    @abstractmethod
    def get_by_voice_call_id(
        self,
        *,
        voice_call_id: str,
    ) -> list[VoiceCallEvent]:
        """
        Retrieve events for a voice call.
        """

        raise NotImplementedError



__all__ = [
    "VoiceCallEventRepository",
]
"""
In-Memory Voice Call Event Repository.

Infrastructure implementation for
voice lifecycle event storage.

Responsibilities:
- Store VoiceCallEvent objects in memory.
- Retrieve events by voice_call_id.
- Provide Beta persistence adapter.

Does NOT:
- Contain business logic.
- Execute workflows.
- Send notifications.
- Call external providers.
- Replace production event storage.
"""

from __future__ import annotations



from app.application.uai.voice.voice_call_event_repository import (
    VoiceCallEventRepository,
)



from app.domain.uai.voice.voice_call_event import (
    VoiceCallEvent,
)



class InMemoryVoiceCallEventRepository(
    VoiceCallEventRepository,
):
    """
    In-memory repository for Beta.

    Future replacement:
    PostgreSQL/event store implementation.
    """



    def __init__(
        self,
    ) -> None:
        """
        Initialise repository storage.
        """

        self._events: list[
            VoiceCallEvent
        ] = []



    def save(
        self,
        *,
        event: VoiceCallEvent,
    ) -> VoiceCallEvent:
        """
        Store a voice call event.
        """


        self._events.append(
            event
        )


        return event



    def get_by_voice_call_id(
        self,
        *,
        voice_call_id: str,
    ) -> list[VoiceCallEvent]:
        """
        Retrieve events for a voice call.
        """


        return [
            event
            for event in self._events
            if event.voice_call_id
            ==
            voice_call_id
        ]



    def get_all(
        self,
    ) -> list[VoiceCallEvent]:
        """
        Return all stored events.

        Useful for Beta diagnostics
        and tests.
        """


        return list(
            self._events
        )



__all__ = [
    "InMemoryVoiceCallEventRepository",
]
"""
Voice Call Event Domain Model.

Represents immutable lifecycle events
generated during a voice call.

Responsibilities:
- Define voice lifecycle events.
- Preserve audit information.
- Represent business state changes.

Does NOT:
- Persist events.
- Send notifications.
- Execute AI.
- Manage providers.
- Handle infrastructure.
"""

from __future__ import annotations


from dataclasses import dataclass


from datetime import datetime, timezone


from enum import Enum



class VoiceCallEventType(
    str,
    Enum,
):
    """
    Supported voice lifecycle events.
    """


    CREATED = (
        "voice_call_created"
    )


    TRIGGERED = (
        "voice_call_triggered"
    )


    ACCEPTED = (
        "voice_call_accepted"
    )


    STARTED = (
        "voice_call_started"
    )


    RECONNECTED = (
        "voice_call_reconnected"
    )


    ENDED = (
        "voice_call_ended"
    )


    COMPLETED = (
        "voice_call_completed"
    )


    FAILED = (
        "voice_call_failed"
    )



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallEvent:
    """
    Immutable voice lifecycle event.
    """


    event_id: str

    voice_call_id: str

    learner_id: str

    character_id: str

    event_type: VoiceCallEventType

    created_at: datetime



    def __post_init__(
        self,
    ) -> None:
        """
        Validate event invariants.
        """


        if not self.event_id.strip():

            raise ValueError(
                "event_id cannot be empty."
            )


        if not self.voice_call_id.strip():

            raise ValueError(
                "voice_call_id cannot be empty."
            )


        if not self.learner_id.strip():

            raise ValueError(
                "learner_id cannot be empty."
            )


        if not self.character_id.strip():

            raise ValueError(
                "character_id cannot be empty."
            )


        if not isinstance(
            self.event_type,
            VoiceCallEventType,
        ):

            raise ValueError(
                "Invalid voice call event type."
            )


        if self.created_at.tzinfo is None:

            raise ValueError(
                "created_at must be timezone aware."
            )



    @classmethod
    def create(
        cls,
        *,
        event_id: str,
        voice_call_id: str,
        learner_id: str,
        character_id: str,
        event_type: VoiceCallEventType,
    ) -> "VoiceCallEvent":
        """
        Factory method.
        """


        return cls(
            event_id=event_id,
            voice_call_id=voice_call_id,
            learner_id=learner_id,
            character_id=character_id,
            event_type=event_type,
            created_at=(
                datetime.now(
                    timezone.utc
                )
            ),
        )



__all__ = [
    "VoiceCallEvent",
    "VoiceCallEventType",
]
