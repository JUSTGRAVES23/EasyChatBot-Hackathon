    )

    return (

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = (
        application.state.scheduler_runtime
    )

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )
    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )


def create_application() -> FastAPI:
    academy_config = (
        AcademyActivationConfig.from_environment()
    )

    auth_engine = create_async_database_engine()

    auth_session_factory = (
        create_async_session_factory(
            auth_engine,
        )
    )

    auth_repository = (
        PostgreSQLAuthenticationRepository(
            auth_session_factory,
        )
    )

    # Provider-usage repository is synchronous by
    # contract. It therefore receives a synchronous
    # engine/session factory.
    #
    # It must not receive the async authentication
    # session factory.
    provider_usage_engine = (
        create_database_engine()
    )

    provider_usage_session_factory = (
        create_session_factory(
            provider_usage_engine,
        )
    )

    provider_usage_repository = (
        PostgreSQLProviderUsageRepository(
            provider_usage_session_factory,
        )
    )

    provider_usage_dashboard_service = (
        ProviderUsageDashboardService(
            repository=provider_usage_repository,
        )
    )

    functionality_control_repository = (
        PostgreSQLFunctionalityControlRepository(
            provider_usage_session_factory,
        )
    )

    functionality_control_authorizer = (
        AuthorizationFunctionalityControlAuthorizer()
    )

    functionality_control_service = (
        FunctionalityOperationalControlService(
            authorizer=functionality_control_authorizer,
            repository=functionality_control_repository,
        )
    )

    academy_one_week_workflow = None
    announcement_telegram_client = None

    if academy_config.activation_allowed:
        academy_session_factory = (
            auth_session_factory
        )

        (
            announcement_telegram_client,
            _announcement_publisher,

def _required_config(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for Academy production composition."
        )

    return value


def create_academy_one_week_workflow(
    *,
    session_factory=None,
    email_provider_manager: EmailProviderManager | None = None,
    telegram_bot=None,
    portal_url: str | None = None,
    registration_email: str | None = None,
) -> AcademyOneWeekWorkflow:
    """Compose the production Academy one-week workflow.

    Composition:

        PostgreSQL
            ↓
        AcademyOneWeekWorkflow
            ↓
        OneWeekNotificationOrchestrator
            ↓
        EmailProviderManager
            ↓
        Existing Telegram Bot adapter

    The scheduler itself is intentionally not created here.
    """

    if session_factory is None:
        engine = create_async_database_engine()
        session_factory = create_async_session_factory(engine)

    resolved_portal_url = (
        portal_url.strip()
        if portal_url is not None
        else _required_config("EASYCHATBOT_PORTAL_URL")
    )

    resolved_registration_email = (
        registration_email.strip()
        if registration_email is not None
        else _required_config("EASYCHATBOT_REGISTRATION_EMAIL")
    )

    email_sender = AcademyEmailNotificationSender(
        provider_manager=email_provider_manager
        or EmailProviderManager(),
    )

    if telegram_bot is not None:
        telegram_sender: TelegramNotificationSender = (
            ExistingBotTelegramNotificationSender(
                telegram_bot,
            )
        )
    else:
        telegram_sender = DisabledAcademyTelegramSender()

    notifications = OneWeekNotificationOrchestrator(
        email_sender=email_sender,
        telegram_sender=telegram_sender,
        email_formatter=AcademyEmailFormatter(),
        telegram_formatter=AcademyTelegramFormatter(),
        registration_email=resolved_registration_email,
        portal_url=resolved_portal_url,
    )

    users = PostgreSQLAcademyUserQuery(
        session_factory,
    )

    deliveries = PostgreSQLAcademyDeliveryRepository(
        session_factory,
    )

    return AcademyOneWeekWorkflow(
        users=users,
        deliveries=deliveries,
        notifications=notifications,
    )

from __future__ import annotations

from datetime import datetime, timedelta, timezone

from backend.app.application.academy_notification_service import (
    OneWeekNotificationOrchestrator,
)
from backend.app.application.academy.academy_notification_delivery import (
    AcademyDeliveryRepository,
    AcademyUserQuery,
)


class AcademyOneWeekWorkflow:
    """Execute the Academy 1-week workflow without owning scheduler state."""

    NOTIFICATION_TYPE = "1_WEEK_MILESTONE"

    def __init__(
        self,
        *,
        users: AcademyUserQuery,
        deliveries: AcademyDeliveryRepository,
        notifications: OneWeekNotificationOrchestrator,
        clock=None,
    ) -> None:
        self._users = users
        self._deliveries = deliveries
        self._notifications = notifications
        self._clock = clock or (lambda: datetime.now(timezone.utc))

    async def run(self) -> dict[str, int]:
        now = self._clock()
        if now.tzinfo is None:
            raise ValueError("Workflow clock must return a timezone-aware datetime.")

        window_start = now - timedelta(days=8)
        window_end = now - timedelta(days=7)

        candidates = await self._users.list_one_week_candidates(
            window_start=window_start,
            window_end=window_end,
        )

        attempted = 0
        succeeded = 0
        skipped = 0
        failed = 0

        for user in candidates:
            # The query is deliberately bounded by the same eligibility window,
            # while the domain rule remains the final guard.
            if not self._notifications.is_one_week_eligible(
                registered_at=user.created_at,
                now=now,
            ):
                skipped += 1
                continue

            claim = await self._deliveries.claim(
                user_id=user.user_id,
                notification_type=self.NOTIFICATION_TYPE,
                now=now,
            )

            if claim is None:
                skipped += 1
                continue

            attempted += 1

            try:
                result = await self._notifications.send_one_week(
                    email=user.email,
                    username=user.username,
                    telegram_chat_id=None,
                )
                email_result = result.get("email") or {}
                provider_message_id = email_result.get("message_id")

                await self._deliveries.mark_succeeded(
                    user_id=user.user_id,
                    notification_type=self.NOTIFICATION_TYPE,
                    provider_message_id=(
                        str(provider_message_id)
                        if provider_message_id is not None
                        else None
                    ),
                    now=now,
                )
                succeeded += 1
            except Exception as exc:
                await self._deliveries.mark_failed(
                    user_id=user.user_id,
                    notification_type=self.NOTIFICATION_TYPE,
                    error=str(exc)[:2000],
                    now=now,
                )
                failed += 1

        return {
            "candidates": len(candidates),
            "attempted": attempted,
            "succeeded": succeeded,
            "failed": failed,
            "skipped": skipped,
        }

from __future__ import annotations

from datetime import datetime, timedelta

from .academy_notification_formatters import AcademyEmailFormatter, AcademyTelegramFormatter
from .academy_notification_ports import EmailNotificationSender, TelegramNotificationSender
from ..domain.academy_notification import (
    AcademyNotification,
    AcademyNotificationType,
)


class OneWeekNotificationOrchestrator:
    """Coordinates the 1-week milestone across Email and Telegram."""

    def __init__(
        self,
        *,
        email_sender: EmailNotificationSender,
        telegram_sender: TelegramNotificationSender,
        email_formatter: AcademyEmailFormatter,
        telegram_formatter: AcademyTelegramFormatter,
        registration_email: str,
        portal_url: str,
    ) -> None:
        if not registration_email.strip():
            raise ValueError("registration_email cannot be empty.")
        if not portal_url.strip():
            raise ValueError("portal_url cannot be empty.")

        self._email_sender = email_sender
        self._telegram_sender = telegram_sender
        self._email_formatter = email_formatter
        self._telegram_formatter = telegram_formatter
        self._registration_email = registration_email
        self._portal_url = portal_url

    @staticmethod
    def is_one_week_eligible(
        *,
        registered_at: datetime,
        now: datetime,
    ) -> bool:
        if registered_at.tzinfo is None or now.tzinfo is None:
            raise ValueError("timestamps must be timezone-aware.")
        age = now - registered_at
        return timedelta(days=7) <= age < timedelta(days=8)

    async def send_one_week(
        self,
        *,
        email: str,
        username: str,
        telegram_chat_id: str | None,
    ) -> dict:
        notification = AcademyNotification(
            notification_type=AcademyNotificationType.ONE_WEEK_MILESTONE,
            username=username,
            portal_url=self._portal_url,
        )

        rendered_email = self._email_formatter.render(notification)

        email_result = await self._email_sender.send(
            "ðŸŽ“ One Week at EasyChatBot, "
            f"@{username.strip()}!",
            {
                "recipient": email,
                "email_type": "academy_notification",
                "notification_type": notification.notification_type.value,
                "from": self._registration_email,
                "body": rendered_email["body"],
                "html": rendered_email["html"],
                "attachments": rendered_email.get(
                    "attachments",
                    [],
                ),
            },
        )

        telegram_result = None
        if telegram_chat_id:
            telegram_result = await self._telegram_sender.send(
                chat_id=telegram_chat_id,
                text=self._telegram_formatter.render(notification),
                portal_url=self._portal_url,
            )

        return {
            "notification_type": notification.notification_type.value,
            "email": email_result,
            "telegram": telegram_result,
        }

from __future__ import annotations

from datetime import datetime
from typing import Any
from uuid import UUID

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker


class PostgreSQLAcademyUserQuery:
    """PostgreSQL adapter for authoritative Academy registration data."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        self._session_factory = session_factory

    async def list_one_week_candidates(
        self,
        *,
        window_start: datetime,
        window_end: datetime,
    ):
        from backend.app.application.academy.academy_notification_delivery import (
            AcademyEligibleUser,
        )

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT id, email, username, created_at
                    FROM users
                    WHERE created_at >= :window_start
                      AND created_at < :window_end
                      AND status = 'active'
                    ORDER BY created_at ASC
                    """
                ),
                {
                    "window_start": window_start,
                    "window_end": window_end,
                },
            )

            return [
                AcademyEligibleUser(
                    user_id=row.id,
                    email=row.email,
                    username=row.username,
                    created_at=row.created_at,
                )
                for row in result
            ]


class PostgreSQLAcademyDeliveryRepository:
    """Durable Academy delivery/idempotency adapter."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        self._session_factory = session_factory

    async def claim(
        self,
        *,
        user_id: UUID,
        notification_type: str,
        now: datetime,
    ):
        from backend.app.application.academy.academy_notification_delivery import (
            AcademyDeliveryRecord,
            AcademyDeliveryStatus,
        )

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    INSERT INTO academy_notification_deliveries (
                        id,
                        user_id,
                        notification_type,
                        status,
                        attempt_count,
                        created_at,
                        updated_at
                    )
                    VALUES (
                        gen_random_uuid(),
                        :user_id,
                        :notification_type,
                        'SENDING',
                        1,
                        :now,
                        :now
                    )
                    ON CONFLICT (user_id, notification_type)
                    DO UPDATE SET
                        status = 'SENDING',
                        attempt_count =
                            academy_notification_deliveries.attempt_count + 1,
                        updated_at = EXCLUDED.updated_at
                    WHERE academy_notification_deliveries.status = 'FAILED'
                    RETURNING
                        user_id,
                        notification_type,
                        status,
                        attempt_count,
                        provider_message_id,
                        last_error
                    """
                ),
                {
                    "user_id": user_id,
                    "notification_type": notification_type,
                    "now": now,
                },
            )

            row = result.first()

            if row is None:
                await session.rollback()
                return None

            record = AcademyDeliveryRecord(
                user_id=row.user_id,
                notification_type=row.notification_type,
                status=AcademyDeliveryStatus(row.status),
                attempt_count=row.attempt_count,
                provider_message_id=row.provider_message_id,
                last_error=row.last_error,
            )

            await session.commit()

            return record

    async def mark_succeeded(
        self,
        *,
        user_id: UUID,
        notification_type: str,
        provider_message_id: str | None,
        now: datetime,
    ) -> None:
        async with self._session_factory() as session:
            await session.execute(
                text(
                    """
                    UPDATE academy_notification_deliveries
                    SET status = 'SUCCEEDED',
                        provider_message_id = :provider_message_id,
                        last_error = NULL,
                        sent_at = :now,
                        updated_at = :now
                    WHERE user_id = :user_id
                      AND notification_type = :notification_type
                    """
                ),
                {
                    "user_id": user_id,
                    "notification_type": notification_type,
                    "provider_message_id": provider_message_id,
                    "now": now,
                },
            )

            await session.commit()

    async def mark_failed(
        self,
        *,
        user_id: UUID,
        notification_type: str,
        error: str,
        now: datetime,
    ) -> None:
        async with self._session_factory() as session:
            await session.execute(
                text(
                    """
                    UPDATE academy_notification_deliveries
                    SET status = 'FAILED',
                        last_error = :error,
                        updated_at = :now
                    WHERE user_id = :user_id
                      AND notification_type = :notification_type
                    """
                ),
                {
                    "user_id": user_id,
                    "notification_type": notification_type,
                    "error": error[:2000],
                    "now": now,
                },
            )

            await session.commit()

from __future__ import annotations

from typing import Any

from backend.cybersecurity.email.email_provider_manager import (
    EmailProviderManager,
)


class AcademyEmailNotificationSender:
    """Academy email adapter over the existing provider boundary.

    Application code depends on the Academy email port.
    Provider selection remains owned by EmailProviderManager.

    Does NOT:
    - create a second email provider;
    - access provider credentials;
    - send email directly through Resend;
    - make Academy eligibility decisions.
    """

    def __init__(
        self,
        *,
        provider_manager: EmailProviderManager | None = None,
    ) -> None:
        self._provider_manager = (
            provider_manager
            if provider_manager is not None
            else EmailProviderManager()
        )

    async def send(
        self,
        subject: str,
        content: dict[str, Any],
    ) -> dict[str, Any]:
        if not isinstance(subject, str) or not subject.strip():
            raise ValueError("Email subject cannot be empty.")

        if not isinstance(content, dict):
            raise TypeError("Email content must be a dictionary.")

        recipient = content.get("recipient")

        if not isinstance(recipient, str) or not recipient.strip():
            raise ValueError("Email recipient is required.")

        result = await self._provider_manager.send(
            subject.strip(),
            content,
        )

        if not isinstance(result, dict):
            raise TypeError(
                "Email provider returned an invalid result."
            )

        return result

    "AcademyEmailNotificationSender",
]

from __future__ import annotations

from typing import Any


class ExistingBotTelegramNotificationSender:
    """Outbound adapter around the existing Telegram Bot instance.

    No Bot is created here. The composition root injects the existing
    `TelegramBotRuntime.bot`.
    """

    def __init__(self, bot: Any) -> None:
        if bot is None:
            raise ValueError("bot cannot be None.")
        self._bot = bot

    async def send(
        self,
        *,
        chat_id: str,
        text: str,
        portal_url: str,
    ) -> dict:
        if not chat_id.strip():
            raise ValueError("chat_id cannot be empty.")
        if not portal_url.strip():
            raise ValueError("portal_url cannot be empty.")

        from telegram import InlineKeyboardButton, InlineKeyboardMarkup

        markup = InlineKeyboardMarkup(
            [[InlineKeyboardButton("ðŸš€ Continue Learning", url=portal_url)]]
        )

        message = await self._bot.send_message(
            chat_id=chat_id,
            text=text,
            parse_mode="Markdown",
            reply_markup=markup,
        )

        return {
            "provider": "telegram",
            "status": "accepted",
            "message_id": getattr(message, "message_id", None),
        }
