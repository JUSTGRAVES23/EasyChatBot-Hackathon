    """A live provider observation must be reflected by the read model."""
    monitor = ProviderHealthMonitorService()

    monitor.observe(
        ProviderHealthObservation(
            provider="openai",
            model="gpt-5-mini",

            observed_at=datetime.now(timezone.utc),
            latency_ms=240,
            consecutive_successes=1,

    payload = FunctionalityHealthService(
        health_monitor=monitor,

from __future__ import annotations

import asyncio
import os
from typing import Any

from backend.app.application.health.health_result import (
    HealthCheckResult,
    HealthStatus,
)


class LLMProviderHealthCheck:
    """Read-only health probe for the configured LLM provider.

    Ollama uses /api/tags.

    OpenAI uses the models endpoint.

    Neither probe generates a learner response or sends
    learner content to the provider.
    """

    name = "llm_provider"

    def __init__(
        self,
        *,
        provider: str | None = None,
        request_get: Any | None = None,
        openai_client_factory: Any | None = None,
    ) -> None:
        self._provider = (
            provider
            or os.getenv(
                "LLM_PROVIDER",
                "ollama",
            )
        ).strip().lower()

        self._request_get = request_get
        self._openai_client_factory = (
            openai_client_factory
        )

    async def check(self) -> HealthCheckResult:
        if self._provider == "ollama":
            return await self._check_ollama()

        if self._provider == "openai":
            return await self._check_openai()

        return HealthCheckResult(
            name=self.name,
            status=HealthStatus.UNHEALTHY,
            details={
                "reason": "unsupported_provider",
            },
        )

    async def _check_ollama(
        self,
    ) -> HealthCheckResult:
        base_url = os.getenv(
            "OLLAMA_BASE_URL",
            "http://localhost:11434",
        ).rstrip("/")

        try:
            if self._request_get is not None:
                response = await self._request_get(
                    f"{base_url}/api/tags"
                )

                status_code = getattr(
                    response,
                    "status_code",
                    200,
                )

                if status_code >= 400:
                    raise RuntimeError(
                        "Ollama health request failed."
                    )

            else:
                import requests

                response = await asyncio.to_thread(
                    requests.get,
                    f"{base_url}/api/tags",
                    timeout=10,
                )

                response.raise_for_status()

            return HealthCheckResult(
                name=self.name,
                status=HealthStatus.HEALTHY,
                details={
                    "provider": "ollama",
                    "reachable": True,
                },
            )

        except Exception:
            return HealthCheckResult(
                name=self.name,
                status=HealthStatus.UNHEALTHY,
                details={
                    "provider": "ollama",
                    "reason": "provider_unavailable",
                },
            )

    async def _check_openai(
        self,
    ) -> HealthCheckResult:
        if not os.getenv(
            "OPENAI_API_KEY"
        ):
            return HealthCheckResult(
                name=self.name,
                status=HealthStatus.NOT_CONFIGURED,
                details={
                    "provider": "openai",
                    "configured": False,
                },
            )

        try:
            if self._openai_client_factory is None:
                from openai import AsyncOpenAI

                client = AsyncOpenAI(
                    api_key=os.environ[
                        "OPENAI_API_KEY"
                    ],
                )

            else:
                client = (
                    self._openai_client_factory()
                )

            await client.models.list()

            return HealthCheckResult(
                name=self.name,
                status=HealthStatus.HEALTHY,
                details={
                    "provider": "openai",
                    "reachable": True,
                },
            )

        except Exception:
            return HealthCheckResult(
                name=self.name,
                status=HealthStatus.UNHEALTHY,
                details={
                    "provider": "openai",
                    "reason": "provider_unavailable",
                },
            )

def _create_ai_operations_center_service(
    *,
    repositories,
    runtime_services,
):
    """
    Compose the verified I14 Operations Center service.

    This function is intentionally kept inside the application composition
    boundary.

    The Operations Center remains a read-only administrative/control surface.

    It does NOT:
    - execute Agents
    - execute Skills
    - create a second runtime
    - create a second repository bundle
    - define Agents
    - define Skills
    - bypass RuntimeServices
    """

    from i14_1_registry.discovery_registry import (
        DynamicCapabilityRegistry,
    )

    from i14_4_application.capability_graph_service import (
        CapabilityGraphService,
    )

    from i14_5_application.operational_detail_service import (
        OperationalDetailService,
    )

    from i14_6_application.operations_center_service import (
        OperationsCenterService,
    )

    # -------------------------------------------------------------------------
    # I14 application capability composition
    #
    # This factory is the integration boundary for the existing I14
    # discovery/registry implementation.
    #
    # No Agent or Skill definitions are hard-coded here.
    # -------------------------------------------------------------------------

    from app.bootstrap.compositions.i14_capability_composition import (
        create_i14_capability_services,
    )

    i14_services = create_i14_capability_services(
        repositories=repositories,
        runtime_services=runtime_services,
    )

    if i14_services is None:
        raise RuntimeError(
            "I14 capability services composition returned None."
        )

    snapshot = i14_services.snapshot

    # -------------------------------------------------------------------------
    # I14.1 registry remains the discovery boundary.
    #
    # The registry itself does not define application Agents/Skills here.
    # -------------------------------------------------------------------------

    registry = DynamicCapabilityRegistry()

    del registry

    # -------------------------------------------------------------------------
    # I14.4 / I14.5 are read-only projections over the authoritative
    # I14.1 application snapshot.
    # -------------------------------------------------------------------------

    snapshot_provider = lambda: snapshot

    capability_graph = CapabilityGraphService(
        snapshot_provider=snapshot_provider,
    )

    operational_detail = OperationalDetailService(
        snapshot_provider=snapshot_provider,
    )

    return OperationsCenterService(
        capability_graph=capability_graph,
        operational_detail=operational_detail,
    )

def create_application() -> FastAPI:
    """
    Create and configure the EasyChatBot Academy application.

    Composition happens here only.

    Business logic remains inside the application/domain layers.

    Returns
    -------
    FastAPI
        Configured EasyChatBot Academy application.

    Raises
    ------
    Exception
        Propagates composition failures so application startup fails closed
        rather than running with incomplete dependencies.
    """

    logger.info(
        "Creating EasyChatBot Academy application."
    )

    try:
        # =====================================================================
        # Repository Composition
        #
        # Infrastructure implementations are composed behind repository
        # contracts.
        # =====================================================================

        repositories = create_repositories()

        if repositories is None:
            raise RuntimeError(
                "Repository composition returned None."
            )

        # =====================================================================
        # Runtime Composition
        #
        # Runtime owns orchestration and application lifecycle concerns.
        # =====================================================================

        runtime_services = create_runtime_services(
            repositories=repositories,
        )

        if runtime_services is None:
            raise RuntimeError(
                "Runtime services composition returned None."
            )

        # =====================================================================
        # Questionnaire Runtime Composition
        #
        # Compose the existing QuestionnaireRuntime once at the application
        # boundary. The HTTP router resolves this instance from application
        # state.
        #
        # This does not create another FastAPI application.
        # =====================================================================

        questionnaire_runtime = QuestionnaireRuntime()

        # =====================================================================
        # FastAPI Application
        # =====================================================================

        application = FastAPI(
            title="EasyChatBot Academy",
            version="0.1.0-beta",
            description=(
                "AI-powered learning platform runtime."
            ),
        )

        # =====================================================================
        # Browser CORS Boundary
        #
        # The frontend runs as a separate origin during local development.
        #
        # Keep the trust boundary explicit. Do not use:
        #
        #     allow_origins=["*"]
        #
        # This middleware belongs at the application composition boundary,
        # rather than inside CMP routers or application services.
        # =====================================================================

        application.add_middleware(
            CORSMiddleware,
            allow_origins=[
                "http://localhost:3000",
                "http://127.0.0.1:3000",
            ],
            allow_credentials=True,
            allow_methods=["*"],
            allow_headers=["*"],
        )

        # =====================================================================
        # Application State
        #
        # Store the already-composed boundaries.
        #
        # No duplicate repositories or services are created by API routes.
        # =====================================================================

        application.state.repositories = repositories

        application.state.runtime_services = runtime_services

        application.state.questionnaire_runtime = (
            questionnaire_runtime
        )

        # =====================================================================
        # M2.2 World Questionnaire repository boundary
        # =====================================================================

        learning_world_repository = getattr(
            repositories,
            "learning_world_repository",
            None,
        )

        if learning_world_repository is None:
            raise RuntimeError(
                "LearningWorldRepository was not composed."
            )

        application.state.learning_world_repository = (
            learning_world_repository
        )

        # =====================================================================
        # Authentication API
        #
        # Authentication implementation remains behind the existing Academy
        # application/infrastructure boundaries.
        # =====================================================================

        application.include_router(
            authentication_router,
        )

        # =====================================================================
        # Academy API
        # =====================================================================

        application.include_router(
            academy_router,
        )

        # =====================================================================
        # QUESTIONNAIRE API
        #
        # Existing QuestionnaireRuntime is exposed through its thin Academy
        # HTTP presentation boundary.
        #
        # Exact frontend contract:
        #
        # GET
        # /api/academy/questionnaires/{questionnaire_id}
        #
        # POST
        # /api/academy/questionnaires/{questionnaire_id}/answers
        #
        # POST
        # /api/academy/questionnaires/{questionnaire_id}/complete
        #
        # GET
        # /api/academy/questionnaire-results/{result_id}
        # =====================================================================

        application.include_router(
            questionnaire_router,
        )

        # =====================================================================
        # M2.2 WORLD QUESTIONNAIRE API
        # =====================================================================

        application.include_router(
            world_questionnaire_router,
        )

        # =====================================================================
        # CMP OPERATIONS CENTER API
        #
        # Existing Compliance Framework Operations Center remains separate
        # from the AI Operations Center.
        #
        # Do NOT merge these two bounded contexts.
        # =====================================================================

        application.include_router(
            cmp_operations_center_router,
        )

        application.include_router(
            cmp_population_router,
        )

        # =====================================================================
        # AI OPERATIONS CENTER
        #
        # I14.1 -> I14.6
        #
        # Dedicated EasyChatBot Academy administrative/control presentation
        # boundary.
        #
        # The existing Academy authentication dependency is injected.
        #
        # No second authentication system is created.
        # =====================================================================

        from app.api.routes.ai_operations_center_routes import (
            create_ai_operations_center_router,
