=== ORCHESTRATOR ===
"""Orchestrate subscription entitlement persistence and World access.

This service is the async application boundary between:
    payment entitlement state
        -> Learning World access

It deliberately does not contain Stripe/provider logic.
"""

from __future__ import annotations

from uuid import UUID

from app.application.learning.subscription_world_access_service import (
    SubscriptionWorldAccessService,
)


class SubscriptionEntitlementWorldAccessOrchestrator:
    """Synchronize World access from an authoritative subscription entitlement."""

    def __init__(
        self,
        *,
        subscription_world_access_service: SubscriptionWorldAccessService,
    ) -> None:
        if subscription_world_access_service is None:
            raise ValueError(
                "subscription_world_access_service cannot be None."
            )

        self._subscription_world_access_service = (
            subscription_world_access_service
        )

    async def synchronize(
        self,
        *,
        user_id: UUID,
        plan_code: str,
        active: bool,
    ) -> None:
        """Reconcile World access for the supplied entitlement state.

        The underlying SubscriptionWorldAccessService remains responsible
        for entitlement policy and WorldAccessService authorization.
        """
        if not isinstance(user_id, UUID):
            raise ValueError("user_id must be a UUID.")

        if not isinstance(plan_code, str) or not plan_code.strip():
            raise ValueError("plan_code is required.")

        await self._subscription_world_access_service.synchronize(
            user_id=user_id,
            plan_code=plan_code,
            active=active,
        )


__all__ = ["SubscriptionEntitlementWorldAccessOrchestrator"]

from __future__ import annotations

from datetime import datetime
from typing import Any

from app.domain.payment.models import PaymentEvent


class SubscriptionEntitlementService:
    """Apply verified Stripe subscription events to EASYWORLD persistence.

    Stripe is the payment system of record.
    EASYWORLD subscription and entitlement persistence is the access system
    of record.

    This service deliberately contains no Stripe SDK calls.
    """

    SUBSCRIPTION_EVENTS = {
        "customer.subscription.created",
        "customer.subscription.updated",
        "customer.subscription.deleted",
    }

    def __init__(
        self,
        subscription_repository,
        entitlement_repository,
    ) -> None:
        if subscription_repository is None:
            raise ValueError("subscription_repository cannot be None.")

        if entitlement_repository is None:
            raise ValueError("entitlement_repository cannot be None.")

        self.subscription_repository = subscription_repository
        self.entitlement_repository = entitlement_repository

    def handle_payment_event(self, event: PaymentEvent) -> None:
        if not isinstance(event, PaymentEvent):
            raise TypeError("event must be a PaymentEvent.")

        if event.event_type not in self.SUBSCRIPTION_EVENTS:
            return

        subscription = self._extract_subscription(event)

        beneficiary_user_id = self._metadata_value(
            subscription,
            "easyworld_user_id",
        )

        if not beneficiary_user_id:
            raise ValueError(
                "Stripe subscription is missing easyworld_user_id metadata."
            )

        payer_user_id = self._metadata_value(
            subscription,
            "easyworld_payer_user_id",
        ) or beneficiary_user_id

        plan_code = self._metadata_value(
            subscription,
            "easyworld_plan_code",
        )

        if not plan_code:
            raise ValueError(
                "Stripe subscription is missing easyworld_plan_code metadata."
            )

        stripe_subscription_id = subscription.get("id")
        stripe_customer_id = self._customer_id(subscription.get("customer"))
        stripe_price_id = self._price_id(subscription)

        if not stripe_subscription_id:
            raise ValueError("Stripe subscription is missing id.")

        if not stripe_customer_id:
            raise ValueError("Stripe subscription is missing customer.")

        if not stripe_price_id:
            raise ValueError("Stripe subscription is missing price.")

        status = str(subscription.get("status") or "unknown")

        current_period_start = self._timestamp(
            subscription.get("current_period_start")
        )
        current_period_end = self._timestamp(
            subscription.get("current_period_end")
        )

        cancel_at_period_end = bool(
            subscription.get("cancel_at_period_end", False)
        )

        self.subscription_repository.upsert(
            beneficiary_user_id=beneficiary_user_id,
            payer_user_id=payer_user_id,
            stripe_customer_id=stripe_customer_id,
            stripe_subscription_id=stripe_subscription_id,
            stripe_price_id=stripe_price_id,
            plan_code=plan_code,
            status=status,
            current_period_start=current_period_start,
            current_period_end=current_period_end,
            cancel_at_period_end=cancel_at_period_end,
        )

        active = status in {
            "trialing",
            "active",
        }

        self.entitlement_repository.upsert(
            user_id=beneficiary_user_id,
            plan_code=plan_code,
            active=active,
        )

    @staticmethod
    def _extract_subscription(event: PaymentEvent) -> dict[str, Any]:
        data = event.payload.get("data")

        if not isinstance(data, dict):
            raise ValueError("Stripe event is missing data.")

        obj = data.get("object")

        if not isinstance(obj, dict):
            raise ValueError("Stripe event is missing subscription object.")

        return obj

    @staticmethod
    def _metadata_value(
        subscription: dict[str, Any],
        key: str,
    ) -> str | None:
        metadata = subscription.get("metadata")

        if not isinstance(metadata, dict):
            return None

        value = metadata.get(key)

        if value is None:
            return None

        value = str(value).strip()

        return value or None

    @staticmethod
    def _customer_id(customer: Any) -> str | None:
        if isinstance(customer, str):
            return customer

        if isinstance(customer, dict):
            value = customer.get("id")
            return str(value) if value else None

        return None

    @staticmethod
    def _price_id(subscription: dict[str, Any]) -> str | None:
        items = subscription.get("items")

        if not isinstance(items, dict):
            return None

        data = items.get("data")

        if not isinstance(data, list) or not data:
            return None

        first_item = data[0]

        if not isinstance(first_item, dict):
            return None

        price = first_item.get("price")

        if isinstance(price, str):
            return price

        if isinstance(price, dict):
            value = price.get("id")
            return str(value) if value else None

        return None

    @staticmethod
    def _timestamp(value: Any) -> datetime | None:
        if value is None:
            return None

        try:
            return datetime.fromtimestamp(float(value))
        except (TypeError, ValueError, OSError):
            raise ValueError(
                f"Invalid Stripe timestamp: {value!r}."
            )

from __future__ import annotations

from sqlalchemy import select
from sqlalchemy.orm import Session, sessionmaker

from app.infrastructure.persistence.billing_models import EntitlementRecord


class PostgreSQLEntitlementRepository:
    """Durable PostgreSQL repository for EASYWORLD entitlements."""

    def __init__(
        self,
        session_factory: sessionmaker[Session],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    def get(
        self,
        *,
        user_id: str,
        plan_code: str,
    ) -> EntitlementRecord | None:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if not plan_code:
            raise ValueError("plan_code cannot be empty.")

        with self._session_factory() as session:
            return session.scalar(
                select(EntitlementRecord).where(
                    EntitlementRecord.user_id == user_id,
                    EntitlementRecord.plan_code == plan_code,
                )
            )

    def get_for_user(
        self,
        user_id: str,
    ) -> list[EntitlementRecord]:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        with self._session_factory() as session:
            return list(
                session.scalars(
                    select(EntitlementRecord)
                    .where(EntitlementRecord.user_id == user_id)
                    .order_by(EntitlementRecord.created_at.desc())
                ).all()
            )

    def upsert(
        self,
        *,
        user_id: str,
        plan_code: str,
        ai_credits: int = 0,
        agent_access: bool = False,
        premium_models: bool = False,
        priority_processing: bool = False,
        active: bool = True,
    ) -> EntitlementRecord:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if not plan_code:
            raise ValueError("plan_code cannot be empty.")

        if ai_credits < 0:
            raise ValueError("ai_credits cannot be negative.")

        with self._session_factory() as session:
            record = session.scalar(
                select(EntitlementRecord).where(
                    EntitlementRecord.user_id == user_id,
                    EntitlementRecord.plan_code == plan_code,
                )
            )

            if record is None:
                record = EntitlementRecord(
                    user_id=user_id,
                    plan_code=plan_code,
                    ai_credits=ai_credits,
                    agent_access=agent_access,
                    premium_models=premium_models,
                    priority_processing=priority_processing,
                    active=active,
                )
                session.add(record)
            else:
                record.ai_credits = ai_credits
                record.agent_access = agent_access
                record.premium_models = premium_models
                record.priority_processing = priority_processing
                record.active = active

            session.commit()
            session.refresh(record)

            return record

from __future__ import annotations

class PostgreSQLSubscriptionRepository:
    """Durable PostgreSQL repository for EASYWORLD subscriptions."""

    def __init__(
        self,
        session_factory: sessionmaker[Session],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    def get_by_stripe_subscription_id(
        self,
        stripe_subscription_id: str,
    ) -> SubscriptionRecord | None:
        if not stripe_subscription_id:
            raise ValueError("stripe_subscription_id cannot be empty.")

        with self._session_factory() as session:
            return session.scalar(
                select(SubscriptionRecord).where(
                    SubscriptionRecord.stripe_subscription_id
                    == stripe_subscription_id
                )
            )

    def get_by_beneficiary_user_id(
        self,
        beneficiary_user_id: str,
    ) -> list[SubscriptionRecord]:
        if not beneficiary_user_id:
            raise ValueError("beneficiary_user_id cannot be empty.")

        with self._session_factory() as session:
            return list(
                session.scalars(
                    select(SubscriptionRecord)
                    .where(
                        SubscriptionRecord.beneficiary_user_id
                        == beneficiary_user_id
                    )
                    .order_by(SubscriptionRecord.created_at.desc())
                ).all()
            )

    def upsert(
        self,
        *,
        beneficiary_user_id: str,
        payer_user_id: str,
        stripe_customer_id: str,
        stripe_subscription_id: str,
        stripe_price_id: str,
        plan_code: str,
        status: str,
        current_period_start: datetime | None = None,
        current_period_end: datetime | None = None,
        cancel_at_period_end: bool = False,
    ) -> SubscriptionRecord:
        required_values = {
            "beneficiary_user_id": beneficiary_user_id,
            "payer_user_id": payer_user_id,
            "stripe_customer_id": stripe_customer_id,
            "stripe_subscription_id": stripe_subscription_id,
            "stripe_price_id": stripe_price_id,
            "plan_code": plan_code,
            "status": status,
        }

        for field, value in required_values.items():
            if not value:
                raise ValueError(f"{field} cannot be empty.")

        with self._session_factory() as session:
            record = session.scalar(
                select(SubscriptionRecord).where(
                    SubscriptionRecord.stripe_subscription_id
                    == stripe_subscription_id
                )
            )

            if record is None:
                record = SubscriptionRecord(
                    beneficiary_user_id=beneficiary_user_id,
                    payer_user_id=payer_user_id,
                    stripe_customer_id=stripe_customer_id,
                    stripe_subscription_id=stripe_subscription_id,
                    stripe_price_id=stripe_price_id,
                    plan_code=plan_code,
                    status=status,
                    current_period_start=current_period_start,
                    current_period_end=current_period_end,
                    cancel_at_period_end=cancel_at_period_end,
                )
                session.add(record)
            else:
                record.beneficiary_user_id = beneficiary_user_id
                record.payer_user_id = payer_user_id
                record.stripe_customer_id = stripe_customer_id
                record.stripe_price_id = stripe_price_id
                record.plan_code = plan_code
                record.status = status
                record.current_period_start = current_period_start
                record.current_period_end = current_period_end
                record.cancel_at_period_end = cancel_at_period_end

            session.commit()
            session.refresh(record)

            return record

    def delete_by_stripe_subscription_id(
        self,
        stripe_subscription_id: str,
    ) -> None:
        if not stripe_subscription_id:
            raise ValueError("stripe_subscription_id cannot be empty.")

        with self._session_factory() as session:
            record = session.scalar(
                select(SubscriptionRecord).where(
                    SubscriptionRecord.stripe_subscription_id
                    == stripe_subscription_id
                )
            )

            if record is not None:
                session.delete(record)
                session.commit()

from __future__ import annotations

from datetime import datetime, timezone

from sqlalchemy import select
from sqlalchemy.orm import Session, sessionmaker

from app.application.payment.ports import PaymentEventRepository
from app.domain.payment.models import PaymentEvent
from app.infrastructure.persistence.billing_models import StripeWebhookEventRecord


class PostgreSQLPaymentEventRepository(PaymentEventRepository):
    """Durable PostgreSQL implementation of Stripe webhook event storage.

    Responsibilities:
    - Persist verified Stripe webhook events.
    - Provide durable idempotency state.
    - Track processing failures.

    Does not:
    - Verify Stripe signatures.
    - Grant entitlements.
    - Apply payment business rules.
    """

    def __init__(
        self,
        session_factory: sessionmaker[Session],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    def has_processed(self, event_id: str) -> bool:
        if not event_id:
            raise ValueError("event_id cannot be empty.")

        with self._session_factory() as session:
            record = session.get(
                StripeWebhookEventRecord,
                event_id,
            )

            return record is not None and record.processed

    def record_received(self, event: PaymentEvent) -> None:
        if not isinstance(event, PaymentEvent):
            raise TypeError("event must be a PaymentEvent.")

        with self._session_factory() as session:
            existing = session.get(
                StripeWebhookEventRecord,
                event.event_id,
            )

            if existing is None:
                session.add(
                    StripeWebhookEventRecord(
                        stripe_event_id=event.event_id,
                        event_type=event.event_type,
                        processed=False,
                        processed_at=None,
                        payload=event.payload,
                        created_at=datetime.now(timezone.utc),
                    )
                )

            session.commit()

    def mark_processed(self, event_id: str) -> None:
        if not event_id:
            raise ValueError("event_id cannot be empty.")

        with self._session_factory() as session:
            record = session.get(
                StripeWebhookEventRecord,
                event_id,
            )

            if record is None:
                raise ValueError(
                    f"Stripe webhook event not found: {event_id!r}."
                )

            record.processed = True
            record.processed_at = datetime.now(timezone.utc)

            session.commit()

    def mark_failed(
        self,
        event_id: str,
        error: str,
    ) -> None:
        """Record that processing failed without deleting the event.

        The current persistence schema has no dedicated error column.
        Therefore the durable event remains recorded and unprocessed.
        The caller can retry the event safely.
        """

        if not event_id:
            raise ValueError("event_id cannot be empty.")

        if not error:
            raise ValueError("error cannot be empty.")

        with self._session_factory() as session:
            record = session.get(
                StripeWebhookEventRecord,
                event_id,
            )

            if record is None:
                raise ValueError(
                    f"Stripe webhook event not found: {event_id!r}."
                )

            record.processed = False
            record.processed_at = None

            session.commit()

"""PostgreSQL adapter for learner-to-World access."""

from __future__ import annotations

from sqlalchemy import select
from sqlalchemy.orm import Session, sessionmaker

from app.domain.learning.learner_world_access import LearnerWorldAccess
from app.domain.learning.learner_world_access_repository import (
    LearnerWorldAccessRepository,
)
from app.infrastructure.persistence.learner_world_access_model import (
    LearnerWorldAccessRecord,
)


class PostgreSQLLearnerWorldAccessRepository(LearnerWorldAccessRepository):
    """Synchronous repository; application async callers execute it off-loop."""

    def __init__(self, session_factory: sessionmaker[Session]) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")
        self._session_factory = session_factory

    def save(self, access: LearnerWorldAccess) -> LearnerWorldAccess:
        with self._session_factory() as session:
            record = session.get(LearnerWorldAccessRecord, access.access_id)

            if record is None:
                record = LearnerWorldAccessRecord(
                    id=access.access_id,
                    user_id=access.user_id,
                    world_id=access.world_id,
                    status=access.status,
                    access_source=access.access_source,
                    created_at=access.created_at,
                    updated_at=access.updated_at,
                )
                session.add(record)
            else:
                record.status = access.status
                record.access_source = access.access_source
                record.updated_at = access.updated_at

            session.commit()
            session.refresh(record)

            return self._to_domain(record)

    def get_by_id(self, access_id):
        with self._session_factory() as session:
            record = session.get(LearnerWorldAccessRecord, access_id)
            return self._to_domain(record) if record is not None else None

    def get_by_user_and_world(self, *, user_id, world_id):
        with self._session_factory() as session:
            record = session.scalar(
                select(LearnerWorldAccessRecord).where(
                    LearnerWorldAccessRecord.user_id == user_id,
                    LearnerWorldAccessRecord.world_id == world_id,
                )
            )
            return self._to_domain(record) if record is not None else None

    def list_by_user(self, *, user_id):
        with self._session_factory() as session:
            records = session.scalars(
                select(LearnerWorldAccessRecord)
                .where(LearnerWorldAccessRecord.user_id == user_id)
                .order_by(LearnerWorldAccessRecord.created_at)
            ).all()

            return tuple(self._to_domain(record) for record in records)

    @staticmethod
    def _to_domain(record: LearnerWorldAccessRecord) -> LearnerWorldAccess:
        return LearnerWorldAccess(
            access_id=record.id,
            user_id=record.user_id,
            world_id=record.world_id,
            status=record.status,
            access_source=record.access_source,
            created_at=record.created_at,
            updated_at=record.updated_at,
        )
