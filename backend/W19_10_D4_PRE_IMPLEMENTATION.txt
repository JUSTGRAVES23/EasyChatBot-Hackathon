from __future__ import annotations

from dataclasses import dataclass

from app.providers.provider_result import ProviderResult


@dataclass(frozen=True, slots=True)
class ProviderFailure(RuntimeError):
    """Application-level typed representation of a failed LLM request."""

    code: str
    provider_id: str
    provider_request_id: str | None
    message: str
    retryable: bool = True

    def __post_init__(self) -> None:
        if self.code != "PROVIDER_TEMPORARILY_UNAVAILABLE":
            raise ValueError(
                "W19.10-D4 ProviderFailure currently supports only "
                "PROVIDER_TEMPORARILY_UNAVAILABLE."
            )
        if not self.provider_id.strip():
            raise ValueError("provider_id is required.")
        if not self.message.strip():
            raise ValueError("message is required.")

        RuntimeError.__init__(self, self.message)

    @classmethod
    def from_result(cls, result: ProviderResult) -> "ProviderFailure":
        if result.success:
            raise ValueError(
                "ProviderFailure cannot be created from a successful ProviderResult."
            )

        return cls(
            code="PROVIDER_TEMPORARILY_UNAVAILABLE",
            provider_id=result.provider_id,
            provider_request_id=result.provider_request_id,
            message=(
                "The configured AI provider is temporarily unavailable."
            ),
            retryable=True,
        )


__all__ = ["ProviderFailure"]

class BetaTelegramConversationBoundary:
    """Application policy boundary for Beta Telegram conversations."""

    MAX_TELEGRAM_MESSAGE_LENGTH = 4096

    TELEGRAM_FUNCTIONALITY_ID = "telegram"
    TELEGRAM_OPERATIONS_ENVIRONMENT = "adults"

    SAFE_ERROR_RESPONSE = (
        "Sorry, something went wrong while processing your message. "
        "Please try again later."
    )

    UNAUTHORIZED_RESPONSE = (
        "Sorry, this Telegram account is not authorized "
        "and is not approved for Beta access."
    )

    FUNCTIONALITY_DISABLED_RESPONSE = (
        "Sorry, the Telegram classroom is temporarily unavailable. "
        "Please try again later."
    )

    def __init__(
        self,
        *,
        conversation_service: Any,
        beta_access_service: BetaAccessService,
        rate_limiter: TelegramPolicy | None = None,
        abuse_protector: TelegramPolicy | None = None,
        observability: TelegramObservability | None = None,
        policy_message_service: TelegramPolicyMessageService | None = None,
        functionality_control_service: (
            FunctionalityOperationalControlService | None
        ) = None,
        private_chat_bridge: TelegramPrivateChatBridge | None = None,
    ) -> None:
        if conversation_service is None:
            raise ValueError(
                "conversation_service cannot be None."
            )

        if beta_access_service is None:
            raise ValueError(
                "beta_access_service cannot be None."
            )

        self._conversation_service = conversation_service
        self._beta_access_service = beta_access_service
        self._rate_limiter = rate_limiter
        self._abuse_protector = abuse_protector
        self._observability = observability
        self._policy_message_service = (
            policy_message_service
            or TelegramPolicyMessageService()
        )
        self._functionality_control_service = (
            functionality_control_service
        )
        self._private_chat_bridge = private_chat_bridge

    async def _record(
        self,
        *,
        event_type: str,
        telegram_user_id: int,
        outcome: str,
        metadata: dict[str, Any] | None = None,
    ) -> None:
        if self._observability is None:
            return

        try:
            await self._observability.record(
                event_type=event_type,
                telegram_user_id=telegram_user_id,
                outcome=outcome,
                metadata=metadata,
            )
        except Exception:
            logger.exception(
                "Telegram observability failure for event_type=%s.",
                event_type,
            )

    async def _functionality_is_enabled(
        self,
        *,
        telegram_user_id: int,
    ) -> bool:
        control_service = self._functionality_control_service

        # Legacy/unit-test compositions may omit operational control.
        # Production composition always supplies it.
        if control_service is None:
            return True

        environment = self.TELEGRAM_OPERATIONS_ENVIRONMENT
        environment_value = getattr(
            environment,
            "value",
            environment,
        )

        try:
            availability = await asyncio.to_thread(
                control_service.get_availability,
                environment=environment,
                functionality_id=self.TELEGRAM_FUNCTIONALITY_ID,
            )

        except Exception as exc:
            await self._record(
                event_type="telegram_classroom_functionality_control",
                telegram_user_id=telegram_user_id,
                outcome="error",
                metadata={
                    "functionality_id": self.TELEGRAM_FUNCTIONALITY_ID,
                    "environment": environment_value,
                    "error_type": type(exc).__name__,
                },
            )

            logger.exception(
                "Telegram functionality-control lookup failed."
            )

            # Fail closed: an unavailable control plane must not
            # permit Telegram classroom processing.
            return False

        enabled = availability == "enabled"

        await self._record(
            event_type="telegram_classroom_functionality_control",
            telegram_user_id=telegram_user_id,
            outcome="allowed" if enabled else "blocked",
            metadata={
                "functionality_id": self.TELEGRAM_FUNCTIONALITY_ID,
                "environment": environment_value,
            },
        )

        return enabled

    async def handle_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> Any:
        self._validate_telegram_user_id(
            telegram_user_id,
        )

        clean_message = self._validate_message(
            message,
        )

        # ---------------------------------------------------------------
        # Beta authorization / World context
        # ---------------------------------------------------------------
        try:
            context = await (
                self._beta_access_service
                .get_conversation_context(
                    telegram_user_id,
                )
            )

        except PermissionError:
            await self._record(
                event_type="telegram_classroom_authorization",
                telegram_user_id=telegram_user_id,
                outcome="denied",
            )

            return self.UNAUTHORIZED_RESPONSE

        except Exception as exc:
            await self._record(
                event_type="telegram_classroom_authorization",
                telegram_user_id=telegram_user_id,
                outcome="error",
                metadata={
                    "error_type": type(exc).__name__,
                },
            )

            logger.exception(
                "Telegram Beta access/context resolution failed."
            )

            return self.SAFE_ERROR_RESPONSE

        # ---------------------------------------------------------------
        # Functionality operational control
        # ---------------------------------------------------------------
        #
        # This deliberately occurs after authorization/context resolution
        # but before rate limiting, abuse protection, persistence or AI.
        #
        if not await self._functionality_is_enabled(
            telegram_user_id=telegram_user_id,
        ):
            return self.FUNCTIONALITY_DISABLED_RESPONSE

        # ---------------------------------------------------------------
        # Rate limiting
        # ---------------------------------------------------------------
        if self._rate_limiter is not None:
            try:
                allowed = await self._rate_limiter.allow(
                    telegram_user_id=telegram_user_id,
                )

            except Exception as exc:
                await self._record(
                    event_type="telegram_classroom_rate_limit",
                    telegram_user_id=telegram_user_id,
                    outcome="error",
                    metadata={
                        "error_type": type(exc).__name__,
                    },
                )

                logger.exception(
                    "Telegram rate limiter failed."
                )

                return self.SAFE_ERROR_RESPONSE

            if not allowed:
                await self._record(
                    event_type="telegram_classroom_rate_limit",
                    telegram_user_id=telegram_user_id,
                    outcome="blocked",
                    metadata={
                        "world_id": context.world_id,
                    },
                )

                return (
                    self._policy_message_service
                    .rate_limit_message(
                        context=context,
                        retry_after_seconds=(
                            self._get_rate_limit_retry_after_seconds()
                        ),
                    )
                )

        # ---------------------------------------------------------------
        # Abuse protection
        # ---------------------------------------------------------------
        if self._abuse_protector is not None:
            try:
                allowed = await self._abuse_protector.allow(
                    telegram_user_id=telegram_user_id,
                )

            except Exception as exc:
                await self._record(
                    event_type=(
                        "telegram_classroom_abuse_protection"
                    ),
                    telegram_user_id=telegram_user_id,
                    outcome="error",
                    metadata={
                        "error_type": type(exc).__name__,
                    },
                )

                logger.exception(
                    "Telegram abuse protection failed."
                )

                return self.SAFE_ERROR_RESPONSE

            if not allowed:
                await self._record(
                    event_type=(
                        "telegram_classroom_abuse_protection"
                    ),
                    telegram_user_id=telegram_user_id,
                    outcome="blocked",
                    metadata={
                        "world_id": context.world_id,
                    },
                )

                return (
                    self._policy_message_service
                    .abuse_protection_message(
                        context=context,
                        retry_after_seconds=(
                            self._get_abuse_retry_after_seconds()
                        ),
                    )
                )

        # ---------------------------------------------------------------
        # Accepted classroom message
        # ---------------------------------------------------------------
        await self._record(
            event_type="telegram_classroom_message",
            telegram_user_id=telegram_user_id,
            outcome="accepted",
            metadata={
                "world_id": context.world_id,
                "character": context.character,
                "level": context.level,
            },
        )

        # ---------------------------------------------------------------
        # Conversation / AI Runtime
        # ---------------------------------------------------------------
        try:
            if self._private_chat_bridge is not None:
                user = getattr(context, "user", None)
                user_id = getattr(user, "id", None)

                if user_id is None:
                    raise RuntimeError(
                        "Telegram conversation context has no authenticated user id."
                    )

                response = await self._private_chat_bridge.handle_message(
                    user_id=str(user_id),
                    world_id=context.world_id,
                    message=clean_message,
                )

            else:
                context_handler = getattr(
                    self._conversation_service,
                    "handle_telegram_message_with_context",
                    None,
                )

                if context_handler is not None:
                    response = await context_handler(
                        telegram_user_id=telegram_user_id,
                        message=clean_message,
                        context=context,
                    )

                else:
                    response = await (
                        self._conversation_service
                        .handle_telegram_message(
                            telegram_user_id=telegram_user_id,
                            message=clean_message,
                        )
                    )

        except Exception as exc:
            await self._record(
                event_type="telegram_classroom_conversation",
                telegram_user_id=telegram_user_id,
                outcome="error",
                metadata={
                    "world_id": context.world_id,
                    "error_type": type(exc).__name__,
                },
            )

            logger.exception(
                "Telegram conversation processing failed."
            )

            return self.SAFE_ERROR_RESPONSE

        # ---------------------------------------------------------------
        # Successful completion
        # ---------------------------------------------------------------
        await self._record(
            event_type="telegram_classroom_conversation",
            telegram_user_id=telegram_user_id,
            outcome="success",
            metadata={
                "world_id": context.world_id,
                "character": context.character,
                "level": context.level,
            },
        )

        return response

    def _get_rate_limit_retry_after_seconds(self) -> int:
        limiter = self._rate_limiter

        if limiter is None:
            return 60

        window = getattr(
            limiter,
            "window_seconds",
            60,
        )

        try:
            return max(
                1,
                int(float(window)),
            )
        except (TypeError, ValueError):
            return 60

    def _get_abuse_retry_after_seconds(self) -> int:
        protector = self._abuse_protector

        if protector is None:
            return 60

        window = getattr(
            protector,
            "window_seconds",
            60,
        )

        try:
            return max(
                1,
                int(float(window)),
            )
        except (TypeError, ValueError):
            return 60

    @classmethod
    def _validate_telegram_user_id(
        cls,
        telegram_user_id: int,
    ) -> None:
        if isinstance(telegram_user_id, bool):
            raise ValueError(
                "Telegram user ID must be a positive integer."
            )

        if not isinstance(
            telegram_user_id,
            int,
        ):
            raise TypeError(
                "Telegram user ID must be a positive integer."
            )

        if telegram_user_id <= 0:
            raise ValueError(
                "Telegram user ID must be a positive integer."
            )

    @classmethod
    def _validate_message(
        cls,
        message: str,
    ) -> str:
        if message is None:
            raise ValueError(
                "Telegram message cannot be empty."
            )

        if not isinstance(
            message,
            str,
        ):
            raise TypeError(
                "Telegram message must be a string."
            )

        if not message or not message.strip():
            raise ValueError(
                "Telegram message cannot be empty."
            )

        clean_message = message.strip()

        if len(clean_message) > cls.MAX_TELEGRAM_MESSAGE_LENGTH:
            raise ValueError(
                "Telegram message is too long. Maximum length is "
                f"{cls.MAX_TELEGRAM_MESSAGE_LENGTH} characters."
            )

        return clean_message

from __future__ import annotations

import logging
from typing import Any

from app.application.beta_access.beta_telegram_conversation_boundary import (
    BetaTelegramConversationBoundary,
)
from app.application.world_telegram.world_telegram_start_router import (
    WorldTelegramStartRouter,
)
from app.interfaces.telegram.private_chat_gate import (
    BetaPrivateChatGate,
)


logger = logging.getLogger(__name__)


class BetaTelegramAdapter:
    """
    Telegram transport adapter for the Beta private classroom.

    Responsibilities:
    - handle Telegram transport metadata;
    - enforce private-chat-only classroom access;
    - authorize Telegram identity through BetaPrivateChatGate;
    - forward authorized messages to BetaTelegramConversationBoundary;
    - deliver safe user-facing Telegram responses.

    Response ownership:

        Conversation Boundary
                |
                v
             response
                |
                v
        BetaTelegramAdapter
                |
                +----> Telegram reply_text()
                |
                v
             response

    IMPORTANT:

    The adapter is the SINGLE owner of classroom Telegram response
    delivery.

    The returned response is provided to the direct caller for
    testing, observability, and application-level inspection.

    The caller MUST NOT send the returned response to Telegram again.

    This adapter MUST NOT:
    - call an LLM/provider directly;
    - access persistence directly;
    - modify Beta user state directly;
    - bypass BetaAccessService;
    - process group/channel messages through the AI path.
    """

    def __init__(
        self,
        *,
        start_handler: Any,
        private_chat_gate: BetaPrivateChatGate,
        conversation_boundary: (
            BetaTelegramConversationBoundary | None
        ) = None,
        world_start_router: WorldTelegramStartRouter | None = None,
    ) -> None:
        if start_handler is None:
            raise ValueError(
                "start_handler cannot be None."
            )

        if private_chat_gate is None:
            raise ValueError(
                "private_chat_gate cannot be None."
            )

        self._start_handler = start_handler
        self._private_chat_gate = private_chat_gate
        self._conversation_boundary = conversation_boundary
        self._world_start_router = world_start_router

    # =========================================================
    # Telegram /start
    # =========================================================

    async def handle_start(
        self,
        update: Any,
        context: Any,
    ) -> None:
        user = getattr(
            update,
            "effective_user",
            None,
        )

        chat = getattr(
            update,
            "effective_chat",
            None,
        )

        message = getattr(
            update,
            "effective_message",
            None,
        )

        if user is None or chat is None:
            return

        telegram_user_id = getattr(
            user,
            "id",
            None,
        )

        chat_id = getattr(
            chat,
            "id",
            None,
        )

        chat_type = getattr(
            chat,
            "type",
            None,
        )

        if (
            telegram_user_id is None
            or chat_id is None
        ):
            return

        if chat_type != "private":
            await self._reply(
                message,
                (
                    "Please connect to the "
                    "EasyChatBot Academy from a "
                    "private Telegram chat."
                ),
            )
            return

        token = self._extract_start_token(
            context,
        )

        async def reply(
            text: str,
            user_override: Any = None,
        ) -> None:
            await self._reply(
                message,
                text,
            )

        if self._world_start_router is not None:
            try:
                world_route = self._world_start_router.resolve(token)
            except ValueError as exc:
                await reply(str(exc))
                return
            except Exception:
                logger.exception(
                    "Telegram World start routing failed."
                )
                await reply(
                    "We could not resolve that Learning World. "
                    "Please try again."
                )
                return

            if world_route is not None:
                await reply(
                    "ðŸŒ Learning World route resolved: "
                    f"{world_route.world_identifier}"
                )
                return

        await self._start_handler.handle(
            token=token,
            telegram_user_id=telegram_user_id,
            chat_id=chat_id,
            chat_type=chat_type,
            reply=reply,
        )

    # =========================================================
    # Private classroom message flow
    # =========================================================

    async def handle_message(
        self,
        update: Any,
        context: Any = None,
    ) -> str | None:
        """
        Process one Telegram classroom message.

        Response ownership is exclusively inside this adapter.

        Successful flow:

            transport validation
                    |
                    v
            private-chat authorization
                    |
                    v
            conversation boundary
                    |
                    v
              application response
                    |
                    +----> Telegram reply_text()
                    |
                    v
              return response

        The response is delivered exactly once.

        The caller may inspect the returned response but MUST NOT
        send it to Telegram again.
        """

        if self._conversation_boundary is None:
            raise RuntimeError(
                "conversation_boundary is not configured."
            )

        user = getattr(
            update,
            "effective_user",
            None,
        )

        chat = getattr(
            update,
            "effective_chat",
            None,
        )

        message = getattr(
            update,
            "effective_message",
            None,
        )

        if (
            user is None
            or chat is None
            or message is None
        ):
            return None

        telegram_user_id = getattr(
            user,
            "id",
            None,
        )

        chat_type = getattr(
            chat,
            "type",
            None,
        )

        if telegram_user_id is None:
            return None

        # =====================================================
        # HARD TRANSPORT BOUNDARY
        # =====================================================

        if chat_type != "private":
            return None

        # =====================================================
        # MESSAGE VALIDATION
        # =====================================================

        text = getattr(
            message,
            "text",
            None,
        )

        if not isinstance(
            text,
            str,
        ):
            return None

        text = text.strip()

        if not text:
            return None

        # =====================================================
        # BETA ACCESS AUTHORIZATION
        # =====================================================

        try:
            authorized = await (
                self._private_chat_gate.authorize(
                    telegram_user_id=telegram_user_id,
                    chat_type=chat_type,
                )
            )

        except Exception:
            logger.exception(
                "Telegram private-chat authorization failed."
            )

            await self._reply(
                message,
                (
                    "We could not verify your "
                    "EasyChatBot Academy access. "
                    "Please try again."
                ),
            )

            return None

        if not authorized:
            await self._reply(
                message,
                (
                    "Your EasyChatBot Academy Telegram "
                    "access is not authorized and is not "
                    "approved for Beta access."
                ),
            )

            return None

        # =====================================================
        # CONVERSATION BOUNDARY
        # =====================================================

        try:
            response = await (
                self._conversation_boundary.handle_message(
                    telegram_user_id=telegram_user_id,
                    message=text,
                )
            )

        except Exception:
            logger.exception(
                "Telegram classroom conversation processing failed."
            )

            await self._reply(
                message,
                (
                    "Sorry, I could not process "
                    "your classroom message. "
                    "Please try again."
                ),
            )

            return None

        # =====================================================
        # SINGLE RESPONSE DELIVERY POINT
        # =====================================================

        if (
            isinstance(response, str)
            and response.strip()
        ):
            await self._reply(
                message,
                response,
            )

            # Delivery has already happened exactly once.
            #
            # Return only for observability/testing.
            #
            # The runtime MUST ignore this value.
            return response

        return None

    # =========================================================
    # Helpers
    # =========================================================

    @staticmethod
    def _extract_start_token(
        context: Any,
    ) -> str | None:
        args = getattr(
            context,
            "args",
            None,
        )

        if not args:
            return None

        if not isinstance(
            args,
            (list, tuple),
        ):
            return None

        if not args[0]:
            return None

        return str(
            args[0]
        )

    @staticmethod
    async def _reply(
        message: Any,
        text: str,
    ) -> None:
        if message is None:
            return

        reply_text = getattr(
            message,
            "reply_text",
            None,
        )

        if reply_text is None:
            return

        await reply_text(
            text,
        )
