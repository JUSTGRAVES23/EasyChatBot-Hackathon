from __future__ import annotations

from datetime import datetime, timezone
from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException, Request, Response, status
from pydantic import BaseModel, EmailStr, Field

from app.api.dependencies.identity import get_current_principal
from app.application.identity.authentication_service import (
    AuthenticationError,
    AuthenticationService,
)
from app.domain.identity.principal import Principal


# NOTE:
# The authentication service/repository must be composed into app.state
# by the application's existing bootstrap/composition root.


router = APIRouter(
    prefix="/auth",
    tags=["authentication"],
)


class RegisterRequest(BaseModel):
    email: EmailStr
    username: str = Field(
        min_length=2,
        max_length=100,
    )
    password: str = Field(
        min_length=12,
        max_length=256,
    )


class RegisterResponse(BaseModel):
    user_id: UUID
    detail: str


class LoginRequest(BaseModel):
    email: EmailStr
    password: str = Field(
        min_length=1,
        max_length=256,
    )


class LoginResponse(BaseModel):
    detail: str
    expires_at: str


class SessionResponse(BaseModel):
    authenticated: bool


@router.get(
    "/session",
    response_model=SessionResponse,
    summary="Check the current authentication session",
    description=(
        "Return whether the current request has a valid authenticated "
        "application session. The session identifier and other sensitive "
        "authentication data are never returned."
    ),
)
async def get_session(
    principal: Principal = Depends(get_current_principal),
) -> SessionResponse:
    """
    Return the current authentication state.

    Authentication is resolved exclusively through the existing
    get_current_principal() dependency and SessionIdentityProvider.

    This endpoint intentionally exposes only authentication state.
    It does not expose the session identifier, credentials, or other
    sensitive authentication material.
    """
    return SessionResponse(
        authenticated=principal.is_active,
    )


@router.post(
    "/register",
    response_model=RegisterResponse,
    status_code=status.HTTP_201_CREATED,
)
async def register(
    request: RegisterRequest,
    http_request: Request,
) -> RegisterResponse:
    service = AuthenticationService(
        http_request.app.state.auth_repository
    )

    try:
        user_id = await service.register(
            email=str(request.email),
            username=request.username,
            password=request.password,
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(exc),
        ) from None

    return RegisterResponse(
        user_id=user_id,
        detail=(
            "Registration successful. "
            "Email verification is required."
        ),
    )


@router.post(
    "/login",
    response_model=LoginResponse,
)
async def login(
    request: LoginRequest,
    http_request: Request,
    response: Response,
) -> LoginResponse:
    service = AuthenticationService(
        http_request.app.state.auth_repository
    )

    try:
        result = await service.authenticate(
            email=str(request.email),
            password=request.password,
            device_name=http_request.headers.get("user-agent"),
            ip_address=(
                http_request.client.host
                if http_request.client
                else None
            ),
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=str(exc),
            headers={
                "WWW-Authenticate": "Session",
            },
        ) from None

    expires_in_seconds = max(
        0,
        int(
            (
                result.expires_at
                - datetime.now(timezone.utc)
            ).total_seconds()
        ),
    )

    response.set_cookie(
        key="easychatbot_session",
        value=str(result.session_id),
        httponly=True,
        secure=True,
        samesite="lax",
        max_age=expires_in_seconds,
        path="/",
    )

    return LoginResponse(
        detail="Login successful.",
        expires_at=result.expires_at.isoformat(),
    )


@router.post("/verify-email")
async def verify_email(
    token: str,
    http_request: Request,
) -> dict[str, str]:
    """Verify an application email address.

    Token validation and single-use enforcement remain inside the
    authentication repository.
    """
    service = AuthenticationService(
        http_request.app.state.auth_repository
    )

    try:
        await service.verify_email(token)
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(exc),
        ) from None

    return {
        "detail": "Email verification successful.",
    }


@router.post(
    "/logout",
    status_code=status.HTTP_204_NO_CONTENT,
)
async def logout(
    http_request: Request,
    response: Response,
) -> None:
    session_id = http_request.cookies.get(
        "easychatbot_session"
    )

    if session_id:
        service = AuthenticationService(
            http_request.app.state.auth_repository
        )
        await service.revoke_session(
            UUID(session_id)
        )

    response.delete_cookie(
        key="easychatbot_session",
        path="/",
    )

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from uuid import UUID, uuid4
import hashlib
import hmac
import secrets


class AuthenticationError(ValueError):
    """Expected authentication failure."""


@dataclass(frozen=True, slots=True)
class SessionResult:
    session_id: UUID
    user_id: UUID
    expires_at: datetime


class PasswordHasher:
    """PBKDF2 password hashing using the Python standard library."""

    algorithm = "pbkdf2_sha256"
    iterations = 600_000
    salt_bytes = 32

    def hash(self, password: str) -> str:
        if len(password) < 12:
            raise AuthenticationError(
                "Password must contain at least 12 characters."
            )

        salt = secrets.token_bytes(self.salt_bytes)
        digest = hashlib.pbkdf2_hmac(
            "sha256",
            password.encode("utf-8"),
            salt,
            self.iterations,
        )
        return (
            f"{self.algorithm}${self.iterations}$"
            f"{salt.hex()}${digest.hex()}"
        )

    def verify(self, password: str, encoded: str) -> bool:
        try:
            algorithm, iterations, salt_hex, digest_hex = encoded.split("$")
            if algorithm != self.algorithm:
                return False

            candidate = hashlib.pbkdf2_hmac(
                "sha256",
                password.encode("utf-8"),
                bytes.fromhex(salt_hex),
                int(iterations),
            )
            return hmac.compare_digest(candidate.hex(), digest_hex)
        except (ValueError, TypeError):
            return False


class AuthenticationService:
    """Application service for registration/login/session lifecycle.

    Persistence is supplied through a repository protocol so the service does
    not depend on a particular ORM implementation.
    """

    def __init__(
        self,
        repository,
        *,
        password_hasher: PasswordHasher | None = None,
        session_ttl: timedelta = timedelta(hours=12),
    ) -> None:
        self._repository = repository
        self._password_hasher = password_hasher or PasswordHasher()
        self._session_ttl = session_ttl

    async def register(
        self,
        *,
        email: str,
        username: str,
        password: str,
    ):
        normalized_email = email.strip().lower()
        normalized_username = username.strip()

        if not normalized_email or "@" not in normalized_email:
            raise AuthenticationError("A valid email is required.")

        if not normalized_username:
            raise AuthenticationError("Username is required.")

        if await self._repository.email_exists(normalized_email):
            raise AuthenticationError("An account with this email already exists.")

        if await self._repository.username_exists(normalized_username):
            raise AuthenticationError("That username is already in use.")

        user_id = uuid4()
        password_hash = self._password_hasher.hash(password)

        await self._repository.create_user(
            user_id=user_id,
            email=normalized_email,
            username=normalized_username,
        )
        await self._repository.create_password_credential(
            user_id=user_id,
            password_hash=password_hash,
        )
        await self._repository.create_email_verification(
            user_id=user_id,
            email=normalized_email,
        )

        return user_id

    async def authenticate(
        self,
        *,
        email: str,
        password: str,
        device_name: str | None = None,
        ip_address: str | None = None,
    ) -> SessionResult:
        user = await self._repository.get_user_by_email(
            email.strip().lower()
        )

        if user is None:
            raise AuthenticationError("Invalid email or password.")

        if user.status != "active":
            raise AuthenticationError("Account is not active.")

        if not user.email_verified:
            raise AuthenticationError("Email verification is required.")

        credential = await self._repository.get_password_credential(
            user.id
        )

        if credential is None or not self._password_hasher.verify(
            password,
            credential.password_hash,
        ):
            raise AuthenticationError("Invalid email or password.")

        now = datetime.now(timezone.utc)
        expires_at = now + self._session_ttl
        session_id = uuid4()

        await self._repository.create_session(
            session_id=session_id,
            user_id=user.id,
            created_at=now,
            expires_at=expires_at,
            device_name=device_name,
            ip_address=ip_address,
        )

        return SessionResult(
            session_id=session_id,
            user_id=user.id,
            expires_at=expires_at,
        )

    async def verify_email(
        self,
        token: str,
    ) -> None:
        """Verify an email address using a repository-managed token.

        Token generation, hashing, expiration, single-use enforcement,
        and persistence remain the responsibility of the repository.
        The application service exposes the business operation without
        depending on the persistence implementation.
        """

        normalized_token = token.strip()

        if not normalized_token:
            raise AuthenticationError(
                "Email verification token is required."
            )

        verified = await self._repository.verify_email(
            normalized_token,
        )

        if not verified:
            raise AuthenticationError(
                "Invalid or expired email verification token."
            )

    async def revoke_session(self, session_id: UUID) -> None:
        await self._repository.revoke_session(session_id)

def _registration_email_sender() -> str:
    """Return the dedicated EasyChatBot registration sender."""

    repository_root = Path(__file__).resolve().parents[4]
    env_file = repository_root / ".env"

        raise RuntimeError(
            "EASYCHATBOT_REGISTRATION_EMAIL is not configured."
        )

    return sender.strip()

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from uuid import UUID

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker


@dataclass(frozen=True, slots=True)
class AuthUserRecord:
    id: UUID
    email: str
    username: str
    status: str
    email_verified: bool


@dataclass(frozen=True, slots=True)
class AuthCredentialRecord:
    user_id: UUID
    password_hash: str
    credential_type: str


@dataclass(frozen=True, slots=True)
class AuthSessionRecord:
    session_id: UUID
    user_id: UUID
    status: str
    created_at: datetime
    expires_at: datetime
    last_activity_at: datetime
    device_name: str | None
    ip_address: str | None
    is_revoked: bool
    metadata_json: dict


class PostgreSQLAuthenticationRepository:
    """PostgreSQL adapter for the existing EasyChatBot auth tables.

    The adapter intentionally uses SQLAlchemy Core statements rather than
    importing speculative ORM models. This binds the application contract to
    the inspected database schema while keeping persistence isolated.
    """

    def __init__(self, session_factory: async_sessionmaker[AsyncSession]) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")
        self._session_factory = session_factory

    async def email_exists(self, email: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text("SELECT 1 FROM users WHERE email = :email LIMIT 1"),
                {"email": email},
            )
            return result.first() is not None

    async def username_exists(self, username: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text("SELECT 1 FROM users WHERE username = :username LIMIT 1"),
                {"username": username},
            )
            return result.first() is not None

    async def create_user(self, *, user_id: UUID, email: str, username: str) -> None:
        now = datetime.now(timezone.utc)
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO users
                            (id, email, username, status, trial_status, created_at,
                             email_verified)
                        VALUES
                            (:id, :email, :username, 'active', 'not_started', :created_at,
                             FALSE)
                        """
                    ),
                    {
                        "id": user_id,
                        "email": email,
                        "username": username,
                        "created_at": now,
                    },
                )
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_user_roles (user_id, role)
                        VALUES (:user_id, 'beta_user')
                        ON CONFLICT DO NOTHING
                        """
                    ),
                    {"user_id": user_id},
                )

    async def create_password_credential(self, *, user_id: UUID, password_hash: str) -> None:
        now = datetime.now(timezone.utc)
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO user_credentials
                            (id, user_id, password_hash, credential_type,
                             created_at, updated_at)
                        VALUES
                            (:id, :user_id, :password_hash, 'PASSWORD',
                             :created_at, :updated_at)
                        """
                    ),
                    {
                        "id": __import__("uuid").uuid4(),
                        "user_id": user_id,
                        "password_hash": password_hash,
                        "created_at": now,
                        "updated_at": now,
                    },
                )

    async def create_email_verification(self, *, user_id: UUID, email: str) -> str:
        """Create a one-time verification token.

        The raw token is returned to the application boundary. Only its SHA-256
        digest is persisted. Delivery is deliberately outside this repository.
        """
        import hashlib
        import secrets

        raw_token = secrets.token_urlsafe(32)
        token_hash = hashlib.sha256(raw_token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)
        verification_id = __import__("uuid").uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_email_verification_tokens
                            (id, user_id, email, token_hash, expires_at, created_at)
                        VALUES
                            (:id, :user_id, :email, :token_hash,
                             :expires_at, :created_at)
                        """
                    ),
                    {
                        "id": verification_id,
                        "user_id": user_id,
                        "email": email,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=24),
                        "created_at": now,
                    },
                )
        return raw_token

    async def verify_email(self, token: str) -> bool:
        import hashlib

        token_hash = hashlib.sha256(token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    text(
                        """
                        SELECT id, user_id
                        FROM auth_email_verification_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now
                        FOR UPDATE
                        """
                    ),
                    {"token_hash": token_hash, "now": now},
                )
                row = result.mappings().first()
                if row is None:
                    return False

                await session.execute(
                    text(
                        "UPDATE users SET email_verified = TRUE WHERE id = :user_id"
                    ),
                    {"user_id": row["user_id"]},
                )
                await session.execute(
                    text(
                        "UPDATE auth_email_verification_tokens SET used_at = :now WHERE id = :id"
                    ),
                    {"now": now, "id": row["id"]},
                )
                return True

    async def get_user_by_email(self, email: str) -> AuthUserRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT id, email, username, status, email_verified
                    FROM users
                    WHERE email = :email
                    LIMIT 1
                    """
                ),
                {"email": email},
            )
            row = result.mappings().first()
            return self._user(row) if row else None

    async def get_password_credential(self, user_id: UUID) -> AuthCredentialRecord | None:

from __future__ import annotations

from datetime import datetime, timezone

from app.application.identity.identity_provider import IdentityProvider
from app.domain.identity.principal import AccountStatus, Principal


class SessionIdentityProvider(IdentityProvider):
    """Resolve the authenticated Principal from a server-side session."""

    def __init__(
        self,
        *,
        session_id: str | None,
        repository,
    ) -> None:
        self._session_id = session_id
        self._repository = repository

    async def get_current_principal(self) -> Principal | None:
        if not self._session_id:
            return None

        session = await self._repository.get_session(self._session_id)

        if session is None:
            return None

        now = datetime.now(timezone.utc)

        if session.is_revoked or session.expires_at <= now:
            return None

        user = await self._repository.get_user_for_identity(
            session.user_id
        )

        if user is None:
            return None

        roles = await self._repository.get_user_roles(user.id)

        try:
            status = AccountStatus(user.status)
        except ValueError:
            return None

        return Principal(
            user_id=str(user.id),
            email=user.email,
            email_verified=bool(user.email_verified),
            account_status=status,
            roles=frozenset(roles),
        )

    """Safe default when no existing Telegram Bot is supplied.

    Academy email delivery remains independently testable.

def _required_config(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for Academy production composition."
        )

    return value


def create_academy_one_week_workflow(
    *,
    session_factory=None,
    email_provider_manager: EmailProviderManager | None = None,
    telegram_bot=None,
    portal_url: str | None = None,
    registration_email: str | None = None,
) -> AcademyOneWeekWorkflow:
    """Compose the production Academy one-week workflow.

    Composition:

        PostgreSQL
            ?
        AcademyOneWeekWorkflow
            ?
        OneWeekNotificationOrchestrator
            ?
        EmailProviderManager
            ?
        Existing Telegram Bot adapter

    The scheduler itself is intentionally not created here.
    """

    if session_factory is None:
        engine = create_async_database_engine()
        session_factory = create_async_session_factory(engine)

    resolved_portal_url = (
        portal_url.strip()
        if portal_url is not None
        else _required_config("EASYCHATBOT_PORTAL_URL")
    )

    resolved_registration_email = (
        registration_email.strip()
        if registration_email is not None
        else _required_config("EASYCHATBOT_REGISTRATION_EMAIL")
    )

    email_sender = AcademyEmailNotificationSender(
        provider_manager=email_provider_manager
        or EmailProviderManager(),
    )

    if telegram_bot is not None:
        telegram_sender: TelegramNotificationSender = (
            ExistingBotTelegramNotificationSender(
                telegram_bot,
            )
        )
    else:
        telegram_sender = DisabledAcademyTelegramSender()

    notifications = OneWeekNotificationOrchestrator(
        email_sender=email_sender,
        telegram_sender=telegram_sender,
        email_formatter=AcademyEmailFormatter(),
        telegram_formatter=AcademyTelegramFormatter(),
        registration_email=resolved_registration_email,
        portal_url=resolved_portal_url,
    )

    users = PostgreSQLAcademyUserQuery(
        session_factory,
    )

    deliveries = PostgreSQLAcademyDeliveryRepository(
        session_factory,
    )

    return AcademyOneWeekWorkflow(
        users=users,
        deliveries=deliveries,
        notifications=notifications,
    )
