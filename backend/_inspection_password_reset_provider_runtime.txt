"""
Resend email provider.

Infrastructure adapter responsible only for email delivery.

The provider supports multiple controlled email contracts:

- Cybersecurity notifications:
  rendered by the existing cybersecurity HTML renderer.

- Authentication verification emails:
  application service supplies the authoritative HTML and
  plain-text content.

- Authentication password-reset emails:
  application service supplies the authoritative HTML and
  plain-text content.

- Privacy Governance reminders:
  the application service supplies the authoritative HTML and
  plain-text content. The provider delivers that content without
  replacing it with the cybersecurity template.

- Privacy Governance email threading:
  optional In-Reply-To and References headers may be supplied
  by the application layer when a notification is replying to
  an existing governance email thread.

The provider does not:
- make governance decisions;
- modify governance state;
- execute remediation;
- generate application decisions;
- access provider credentials outside configuration;
- decide which governance emails belong to a thread.
"""

from __future__ import annotations

import base64
import mimetypes
from pathlib import Path
from typing import Any

import resend

from cybersecurity.email.email_configuration import (
    EMAIL_FROM,
    RESEND_API_KEY,
)


class ResendEmailProvider:
    """
    Provider adapter for Resend email delivery.

    Routing rule:

        authentication_verification
            -> supplied html

        authentication_password_reset
            -> supplied html

        academy_notification
            -> supplied html

        privacy_governance_reminder
            -> supplied html_body

        all other email types
            -> existing cybersecurity renderer

    Email threading rule:

        in_reply_to
            -> In-Reply-To header

        references
            -> References header

    Threading metadata is supplied by the application workflow.
    This provider only translates that metadata into the provider
    request.
    """

    provider_id = "resend"

    def __init__(
        self,
        *,
        api_key: str | None = None,
        sender: str | None = None,
    ) -> None:
        self.api_key = api_key or RESEND_API_KEY
        self.sender = sender or EMAIL_FROM

        if not self.api_key:
            raise ValueError(
                "RESEND_API_KEY is not configured."
            )

        if not self.sender:
            raise ValueError(
                "Email sender is not configured."
            )

        resend.api_key = self.api_key

    async def send(
        self,
        subject: str,
        content: dict[str, Any],
    ) -> dict[str, Any]:
        """Send an email through Resend."""

        if not isinstance(subject, str) or not subject.strip():
            raise ValueError(
                "Email subject cannot be empty."
            )

        if not isinstance(content, dict):
            raise TypeError(
                "Email content must be a dictionary."
            )

        recipient = content.get("recipient")

        if not isinstance(recipient, str) or not recipient.strip():
            raise ValueError(
                "Email recipient is required."
            )

        recipient = recipient.strip()

        email_type = content.get(
            "email_type",
            "",
        )

        # --------------------------------------------------------
        # Presentation
        # --------------------------------------------------------

        if email_type in {
            "authentication_verification",
            "authentication_password_reset",
            "academy_notification",
        }:
            supplied_html = content.get("html")

            if (
                not isinstance(supplied_html, str)
                or not supplied_html.strip()
            ):
                raise ValueError(
                    f"{email_type} requires "
                    "html to be a non-empty string."
                )

            html = supplied_html

        elif email_type == "privacy_governance_reminder":
            html = self._get_supplied_html_body(content)

        else:
            # Preserve existing Cybersecurity behaviour.
            html = self._build_html(content)

        text_body = content.get("body")

        if text_body is not None:
            if not isinstance(text_body, str):
                raise TypeError(
                    "body must be a string when provided."
                )

            if not text_body.strip():
                raise ValueError(
                    "body cannot be empty when provided."
                )

        # --------------------------------------------------------
        # Base Resend request
        # --------------------------------------------------------

        sender = self.sender

        if email_type in {
            "authentication_verification",
            "authentication_password_reset",
            "academy_notification",
        }:
            supplied_sender = content.get("from")

            if (
                not isinstance(supplied_sender, str)
                or not supplied_sender.strip()
            ):
                raise ValueError(
                    f"{email_type} sender is required."
                )

            sender = supplied_sender.strip()

        params: dict[str, Any] = {
            "from": sender,
            "to": [recipient],
            "subject": subject.strip(),
            "html": html,
        }

        if isinstance(text_body, str) and text_body.strip():
            params["text"] = text_body

        # --------------------------------------------------------
        # Optional email threading
        # --------------------------------------------------------

        in_reply_to = self._normalize_header_value(
            content.get("in_reply_to"),
            field_name="in_reply_to",
        )

        references = self._normalize_header_value(
            content.get("references"),
            field_name="references",
        )

        headers: dict[str, str] = {}

        if in_reply_to:
            headers["In-Reply-To"] = in_reply_to

        if references:
            headers["References"] = references

        if headers:
            params["headers"] = headers

        # --------------------------------------------------------
        # Attachments
        # --------------------------------------------------------

        attachments = content.get("attachments")

        if attachments is not None:
            params["attachments"] = self._build_attachments(
                attachments
            )

        # --------------------------------------------------------
        # Provider delivery
        # --------------------------------------------------------

        response = await resend.Emails.send_async(
            params
        )

        if isinstance(response, dict):
            message_id = response.get("id")
        else:
            message_id = getattr(
                response,
                "id",
                None,
            )

        return {
            "provider": self.provider_id,
            "status": "accepted",
            "message_id": message_id,
        }

    @staticmethod
    def _normalize_header_value(
        value: Any,
        *,
        field_name: str,
    ) -> str | None:
        """
        Validate and normalize optional email-header metadata.
        """

        if value is None:
            return None

        if not isinstance(value, str):
            raise TypeError(
                f"{field_name} must be a string when provided."
            )

        normalized = value.strip()

        if not normalized:
            return None

        return normalized

    @staticmethod
    def _get_supplied_html_body(
        content: dict[str, Any],
    ) -> str:
        """
        Return application-owned Privacy Governance HTML.
        """

        html_body = content.get("html_body")

        if not isinstance(html_body, str):
            raise TypeError(
                "Privacy Governance email requires "
                "html_body to be a string."
            )

        if not html_body.strip():
            raise ValueError(
                "Privacy Governance email html_body cannot be empty."
            )

        return html_body

    @staticmethod
    def _build_attachments(
        attachments: Any,
    ) -> list[dict[str, Any]]:
        """
        Convert local application attachment metadata into the
        Resend attachment representation.

        Inline images use the application-level ``inline`` flag as a
        validation/control signal and the provider-level ``content_id``
        to match a ``cid:...`` reference in the supplied HTML.

        The MIME content type is preserved explicitly because email
        clients use it when deciding how to render an inline resource.
        """

        if not isinstance(attachments, list):
            raise TypeError(
                "attachments must be a list when provided."
            )

        normalized: list[dict[str, Any]] = []

        for attachment in attachments:
            if not isinstance(attachment, dict):
                raise TypeError(
                    "Each email attachment must be a dictionary."
                )

            path_value = attachment.get("path")

            if not path_value:
                raise ValueError(
                    "Email attachment path is required."
                )

            path = Path(str(path_value))

            if not path.exists():
                raise FileNotFoundError(
                    f"Email attachment not found: {path}"
                )

            if not path.is_file():
                raise ValueError(
                    f"Email attachment is not a file: {path}"
                )

            filename = attachment.get("filename")

            resolved_filename = (
                str(filename).strip()
                if filename
                else path.name
            )

            if not resolved_filename:
                raise ValueError(
                    "Email attachment filename is required."
                )

            content_id = attachment.get("content_id")

            inline = bool(
                attachment.get(
                    "inline",
                    False,
                )
            )

            if inline and not content_id:
                raise ValueError(
                    "Inline email attachments require "
                    "content_id."
                )

            encoded_content = base64.b64encode(
                path.read_bytes()
            ).decode("ascii")

            content_type = attachment.get("content_type")

            if content_type:
                resolved_content_type = str(
                    content_type
                ).strip()
            else:
                resolved_content_type = (
                    mimetypes.guess_type(
                        resolved_filename
                    )[0]
                    or "application/octet-stream"
                )

            resend_attachment: dict[str, Any] = {
                "filename": resolved_filename,
                "content": encoded_content,
                "content_type": resolved_content_type,
            }

            if content_id:
                resend_attachment["content_id"] = str(
                    content_id
                )

            normalized.append(
                resend_attachment
            )

        return normalized

    @staticmethod
    def _build_html(
        content: dict[str, Any],
    ) -> str:
        """
        Existing Cybersecurity HTML renderer.

        This remains the Cybersecurity fallback and is intentionally
        not used for authentication or Privacy Governance emails.
        """

        posture = str(
            content.get(
                "posture",
                "Security notification",
            )
        )

        critical_count = content.get(
            "critical_count",
            0,
        )

        high_count = content.get(
            "high_count",
            0,
        )

        priority_signals = content.get(
            "priority_signals",
            [],
        )

        signal_rows = ""

        if isinstance(
            priority_signals,
            list,
        ):
            for signal in priority_signals:
                if not isinstance(
                    signal,
                    dict,
                ):
                    continue

                title = str(
                    signal.get(
                        "title",
                        "Security signal",
                    )
                )

                severity = str(
                    signal.get(
                        "severity",
                        "",
                    )
                ).upper()

                signal_rows += (
                    "<li>"
                    f"<strong>{title}</strong>"
                    f" — {severity}"
                    "</li>"
                )

        return f"""
<!DOCTYPE html>
<html>
<head>
    <meta charset="utf-8">
    <title>EasyChatBot Cybersecurity</title>
</head>
<body>
    <h2>EasyChatBot Cybersecurity Notification</h2>

    <p>
        <strong>Security posture:</strong>
        {posture}
    </p>

    <p>
        <strong>Critical findings:</strong>
        {critical_count}
    </p>

    <p>
        <strong>High findings:</strong>
        {high_count}
    </p>

    <h3>Priority Security Signals</h3>

    <ul>
        {signal_rows}
    </ul>

    <hr>

    <p>
        This notification is decision support only.
        No remediation action was executed by EasyChatBot.
    </p>

    <p>
        <strong>Environment:</strong> DEMO
    </p>
</body>
</html>
"""


__all__ = [
    "ResendEmailProvider",
]

"""
Email Provider Manager.

Controls provider selection, configuration, availability,
and controlled email delivery.

The manager is deliberately provider-independent.
Concrete providers such as Resend are injected into this
boundary rather than called directly by application services.

Does NOT:
- generate cybersecurity reports;
- schedule emails;
- analyse security;
- execute remediation;
- expose provider credentials.
"""

from __future__ import annotations

from typing import Any

from cybersecurity.email.resend_email_provider import (
    ResendEmailProvider,
)


class EmailProviderManager:
    """
    Provider abstraction for controlled email delivery.
    """

    def __init__(
        self,
        provider: Any | None = None,
    ) -> None:

        self.provider = (
            provider
            if provider is not None
            else self._build_default_provider()
        )

    @staticmethod
    def _build_default_provider() -> Any | None:
        """
        Build the configured default email provider.

        Resend is currently the approved hackathon provider.

        If Resend is not configured, return None so the
        application can report provider_not_configured
        rather than failing during application import.
        """

        try:
            return ResendEmailProvider()

        except ValueError:
            return None

    def status(self) -> dict[str, Any]:
        """
        Return non-sensitive provider status.

        This method never exposes:
        - API keys;
        - secrets;
        - credentials.

        Status is determined from the provider already
        configured in the manager. No external provider
        request is made.
        """

        if self.provider is None:

            return {
                "provider": None,
                "configured": False,
                "available": False,
                "status": "not_configured",
            }

        provider_id = getattr(
            self.provider,
            "provider_id",
            self.provider.__class__.__name__,
        )

        return {
            "provider": provider_id,
            "configured": True,
            "available": True,
            "status": "ready",
        }

    async def send(
        self,
        subject: str,
        content: dict,
    ) -> dict:

        if self.provider is None:

            return {
                "status": "provider_not_configured",
            }

from __future__ import annotations

from typing import Any

from cybersecurity.email.email_provider_manager import EmailProviderManager


class AuthenticationEmailSenderAdapter:
    """Authentication email adapter over the shared provider boundary.

    Authentication application services depend on the authentication email
    sender contract. Provider selection and credentials remain owned by the
    shared EmailProviderManager.

    This adapter does not:
    - access provider credentials;
    - call Resend directly;
    - select providers;
    - make authentication or verification decisions.
    """

    def __init__(
        self,
        *,
        provider_manager: EmailProviderManager | None = None,
    ) -> None:
        self._provider_manager = (
            provider_manager
            if provider_manager is not None
            else EmailProviderManager()
        )

    async def send(
        self,
        subject: str,
        content: dict[str, Any],
    ) -> dict[str, Any]:
        """Send an authentication email through the shared provider boundary."""
        if not isinstance(subject, str) or not subject.strip():
            raise ValueError("Email subject cannot be empty.")

        if not isinstance(content, dict):
            raise TypeError("Email content must be a dictionary.")

        recipient = content.get("recipient")
        if not isinstance(recipient, str) or not recipient.strip():
            raise ValueError("Email recipient is required.")

        result = await self._provider_manager.send(
            subject.strip(),
            content,
        )

        if not isinstance(result, dict):
            raise TypeError("Email provider returned an invalid result.")

        return result

def _required_config(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for Academy production composition."
        )

    return value


def create_academy_one_week_workflow(
    *,
    session_factory=None,
    email_provider_manager: EmailProviderManager | None = None,
    telegram_bot=None,
    portal_url: str | None = None,
    registration_email: str | None = None,
) -> AcademyOneWeekWorkflow:
    """Compose the production Academy one-week workflow.

    Composition:

        PostgreSQL
            ?
        AcademyOneWeekWorkflow
            ?
        OneWeekNotificationOrchestrator
            ?
        EmailProviderManager
            ?
        Existing Telegram Bot adapter

    The scheduler itself is intentionally not created here.
    """

    if session_factory is None:
        engine = create_async_database_engine()
        session_factory = create_async_session_factory(engine)

    resolved_portal_url = (
        portal_url.strip()
        if portal_url is not None
        else _required_config("EASYCHATBOT_PORTAL_URL")
    )

    resolved_registration_email = (
        registration_email.strip()
        if registration_email is not None
        else _required_config("EASYCHATBOT_REGISTRATION_EMAIL")
    )

    email_sender = AcademyEmailNotificationSender(
        provider_manager=email_provider_manager
        or EmailProviderManager(),
    )

    if telegram_bot is not None:
        telegram_sender: TelegramNotificationSender = (
            ExistingBotTelegramNotificationSender(
                telegram_bot,
            )
        )
    else:
        telegram_sender = DisabledAcademyTelegramSender()

    notifications = OneWeekNotificationOrchestrator(
        email_sender=email_sender,
        telegram_sender=telegram_sender,
        email_formatter=AcademyEmailFormatter(),
        telegram_formatter=AcademyTelegramFormatter(),
        registration_email=resolved_registration_email,
        portal_url=resolved_portal_url,
    )

    users = PostgreSQLAcademyUserQuery(
        session_factory,
    )

    deliveries = PostgreSQLAcademyDeliveryRepository(
        session_factory,
    )

    return AcademyOneWeekWorkflow(
        users=users,
        deliveries=deliveries,
        notifications=notifications,
    )
