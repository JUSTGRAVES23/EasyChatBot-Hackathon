from __future__ import annotations

from app.application.authorization.authorization_service import (
    AuthorizationService,
    PermissionDeniedError,
)
from app.domain.authorization.permissions import Permission
from app.domain.identity.principal import Principal


class AuthorizationFunctionalityControlAuthorizer:
    """Application adapter for functionality-control authorization."""

    def __init__(
        self,
        authorization_service: AuthorizationService | None = None,
    ) -> None:
        self._authorization_service = (
            authorization_service or AuthorizationService()
        )

    def can_control_functionality(
        self,
        *,
        actor: object,
        environment: str,
        functionality_id: str,
    ) -> bool:
        if not isinstance(actor, Principal):
            return False

        try:
            self._authorization_service.require_permission(
                principal=actor,
                permission=Permission.FUNCTIONALITY_CONTROL,
            )
        except PermissionDeniedError:
            return False

        return True


__all__ = ["AuthorizationFunctionalityControlAuthorizer"]

from app.application.provider_management.functionality_control_authorizer import (
    AuthorizationFunctionalityControlAuthorizer,
)
from app.application.provider_management.functionality_operational_control_service import (
    FunctionalityOperationalControlService,
)
from app.infrastructure.persistence.postgresql_functionality_control_repository import (
    PostgreSQLFunctionalityControlRepository,
)
from app.infrastructure.persistence.database import (
    create_database_engine,
    create_session_factory,
)


def create_functionality_control_service(
    *,
    session_factory,
) -> FunctionalityOperationalControlService:
    """Create the approved functionality-control application service.

    The caller owns the database engine/session lifecycle.
    This helper only composes the application service boundary.
    """

    repository = PostgreSQLFunctionalityControlRepository(
        session_factory,
    )

    authorizer = AuthorizationFunctionalityControlAuthorizer()

    return FunctionalityOperationalControlService(
        authorizer=authorizer,
        repository=repository,
    )


def create_standalone_functionality_control_service() -> tuple[
    FunctionalityOperationalControlService,
    object,
]:
    """Create a control service for a standalone runtime process.

    Returns both the service and its synchronous database engine so
    the caller can dispose the engine during shutdown.
    """

    engine = create_database_engine()
    session_factory = create_session_factory(engine)

    service = create_functionality_control_service(
        session_factory=session_factory,
    )

    return service, engine

    _FUNCTIONALITIES = (

        "pdf_presentation": "Content",

        "payments": "Platform",
        "authentication": "Platform",
        "scheduler": "Platform",
        "provider_runtime": "Infrastructure",
        "database": "Infrastructure",
    }

    def __init__(
        self,
        *,
        readiness_service: ProductionReadinessService,
        functionality_control_service: FunctionalityOperationalControlService | None = None,

        if readiness_service is None:
            raise ValueError("readiness_service cannot be None.")
        self._readiness_service = readiness_service
        self._functionality_control_service = functionality_control_service

    async def get_health(
        self,
        *,
        environment: OperationsEnvironment | None = None,
    ) -> dict[str, Any]:
        resolved_environment = resolve_environment(environment)
        readiness = await self._readiness_service.check()
        checks_by_name = {check.name: check for check in readiness.checks}
        generated_at = datetime.now(timezone.utc).isoformat()

        functionalities = [
            self._build_functionality(
                functionality_id=functionality_id,
                name=name,
                check=self._resolve_check(
                    functionality_id=functionality_id,
                    checks_by_name=checks_by_name,
                ),
                environment=resolved_environment,
                checked_at=generated_at,
                availability=self._get_availability(
                    functionality_id=functionality_id,
                    environment=resolved_environment,
                ),
            )
            for functionality_id, name in self._FUNCTIONALITIES
        ]

        attention = [
            item for item in functionalities
            if item["status"] in {"degraded", "unavailable"}
        ]

        counts = {
            "total": len(functionalities),
            "healthy": sum(item["status"] == "healthy" for item in functionalities),
            "degraded": sum(item["status"] == "degraded" for item in functionalities),
            "incident": sum(item["status"] == "unavailable" for item in functionalities),
            "maintenance": 0,
            "disabled": 0,
            "unknown": sum(item["status"] == "unknown" for item in functionalities),
        }

        if counts["incident"] > 0:
            platform_health, score, impact = "incident", 0.0, "high"
        elif counts["degraded"] > 0:
            platform_health, score, impact = "degraded", 50.0, "medium"
        elif counts["healthy"] > 0:
            platform_health, score, impact = "healthy", 100.0, "low"
        else:
            platform_health, score, impact = "unknown", None, "unknown"

        return {
            "generated_at": generated_at,
            "environment": resolved_environment.value,
            "platform_health": platform_health,
            "platform_health_score": score,
            "user_impact": impact,
            "counts": counts,
            "functionalities": functionalities,
            "attention": attention,
        }

    @classmethod
    def _resolve_check(
        cls,
        *,
        functionality_id: str,
        checks_by_name: dict[str, HealthCheckResult],
    ) -> HealthCheckResult | None:
        for check_name, functionality_ids in cls._CHECK_TO_FUNCTIONALITIES.items():
            if functionality_id in functionality_ids:
                return checks_by_name.get(check_name)
        return None

    @classmethod
    def _build_functionality(
        cls,
        *,
        functionality_id: str,
        name: str,
        check: HealthCheckResult | None,
        environment: OperationsEnvironment,
        checked_at: str,
        availability: str,
    ) -> dict[str, Any]:
        if check is None:
            status = "unknown"
            health_score = None
            risk_level = "unknown"
            provider = model = latency_ms = None
            reason = "No dedicated health check is currently available."
        else:
            status = cls._map_status(check.status)
            health_score = (
                100.0 if status == "healthy"
                else 50.0 if status == "degraded"
                else 0.0 if status == "unavailable"
                else None
            )
            risk_level = {
                "healthy": "low",
                "degraded": "medium",
                "unavailable": "high",
                "unknown": "unknown",
            }[status]
            provider = check.details.get("provider")
            model = check.details.get("model")
            latency_ms = check.details.get("latency_ms")
            reason = check.details.get("reason", "Health check completed.")

        return {
            "functionality_id": functionality_id,
            "name": name,
            "category": cls._CATEGORY_BY_FUNCTIONALITY[functionality_id],
            "description": f"Operational health for {name}.",
            "status": status,
            "health_score": health_score,
            "latency_ms": latency_ms,
            "provider": provider,
            "model": model,
            "capability": functionality_id,
            "checked_at": checked_at,
            "controllable": False,
            "risk_level": risk_level,
            "reason": reason,
            "environment": environment.value,
            "availability": availability,
            "policy_status": "not_evaluated",
        }

    def _get_availability(
        self,
        *,
        functionality_id: str,
        environment: OperationsEnvironment,
    ) -> str:
        if self._functionality_control_service is None:
            return "enabled"

        return self._functionality_control_service.get_availability(
            environment=environment.value,
            functionality_id=functionality_id,
        )

    @staticmethod
    def _map_status(status: HealthStatus) -> str:
        if status is HealthStatus.HEALTHY:
            return "healthy"
        if status is HealthStatus.UNHEALTHY:
            return "unavailable"
        return "unknown"

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone
from threading import RLock
from typing import Protocol

from app.application.provider_management.functionality_registry import (
    is_known_functionality,
)


class FunctionalityControlAuthorizer(Protocol):
    def can_control_functionality(
        self, *, actor: object, environment: str, functionality_id: str
    ) -> bool: ...


class FunctionalityControlAuditor(Protocol):
    def record_functionality_control(
        self,
        *,
        actor: object,
        environment: str,
        functionality_id: str,
        action: str,
        reason: str | None,
        changed_at: datetime,
    ) -> None: ...


@dataclass(frozen=True)
class OperationalControlResult:
    functionality_id: str
    environment: str
    availability: str
    action: str
    reason: str | None
    changed_at: datetime
    changed_by: str


class FunctionalityControlRepository(Protocol):
    def get_availability(
        self, *, environment: str, functionality_id: str
    ) -> str: ...

    def get_last_action(
        self, *, environment: str, functionality_id: str
    ) -> OperationalControlResult | None: ...

    def record_transition(
        self,
        *,
        functionality_id: str,
        environment: str,
        availability: str,
        action: str,
        reason: str | None,
        changed_at: datetime,
        changed_by: str,
    ) -> OperationalControlResult: ...


class FunctionalityOperationalControlService:
    VALID_ENVIRONMENTS = {"kids", "adults"}

    def __init__(
        self,
        *,
        authorizer: FunctionalityControlAuthorizer,
        auditor: FunctionalityControlAuditor | None = None,
        repository: FunctionalityControlRepository | None = None,
    ) -> None:
        if auditor is None and repository is None:
            raise ValueError("auditor or repository is required.")

        self._authorizer = authorizer
        self._auditor = auditor
        self._repository = repository
        self._lock = RLock()

        # Compatibility fallback for existing unit-test/in-memory usage.
        self._availability: dict[tuple[str, str], str] = {}
        self._last_action: dict[
            tuple[str, str], OperationalControlResult
        ] = {}

    def stop(
        self,
        *,
        actor: object,
        actor_name: str,
        environment: str,
        functionality_id: str,
        reason: str,
    ) -> OperationalControlResult:
        self._validate(environment, functionality_id)

        clean_reason = reason.strip()
        if not clean_reason:
            raise ValueError("A reason is required to stop a functionality.")

        self._authorize(actor, environment, functionality_id)

        return self._transition(
            actor=actor,
            actor_name=actor_name,
            environment=environment,
            functionality_id=functionality_id,
            action="stop",
            availability="disabled",
            reason=clean_reason,
        )

    def resume(
        self,
        *,
        actor: object,
        actor_name: str,
        environment: str,
        functionality_id: str,
    ) -> OperationalControlResult:
        self._validate(environment, functionality_id)
        self._authorize(actor, environment, functionality_id)

        return self._transition(
            actor=actor,
            actor_name=actor_name,
            environment=environment,
            functionality_id=functionality_id,
            action="resume",
            availability="enabled",
            reason=None,
        )

    def get_availability(
        self,
        *,
        environment: str,
        functionality_id: str,
    ) -> str:
        self._validate(environment, functionality_id)

        if self._repository is not None:
            return self._repository.get_availability(
                environment=environment,
                functionality_id=functionality_id,
            )

        with self._lock:
            return self._availability.get(
                (environment, functionality_id),
                "enabled",
            )

    def get_last_action(
        self,
        *,
        environment: str,
        functionality_id: str,
    ) -> OperationalControlResult | None:
        self._validate(environment, functionality_id)

        if self._repository is not None:
            return self._repository.get_last_action(
                environment=environment,
                functionality_id=functionality_id,
            )

        with self._lock:
            return self._last_action.get(
                (environment, functionality_id)
            )

    def _authorize(
        self,
        actor: object,
        environment: str,
        functionality_id: str,
    ) -> None:
        if not self._authorizer.can_control_functionality(
            actor=actor,
            environment=environment,
            functionality_id=functionality_id,
        ):
            raise PermissionError(
                "Not authorised to control this functionality."
            )

    def _transition(
        self,
        *,
        actor: object,
        actor_name: str,
        environment: str,
        functionality_id: str,
        action: str,
        availability: str,
        reason: str | None,
    ) -> OperationalControlResult:
        changed_at = datetime.now(timezone.utc)

        if self._repository is not None:
            return self._repository.record_transition(
                functionality_id=functionality_id,
                environment=environment,
                availability=availability,
                action=action,
                reason=reason,
                changed_at=changed_at,
                changed_by=actor_name,
            )

        result = OperationalControlResult(
            functionality_id=functionality_id,
            environment=environment,
            availability=availability,
            action=action,
            reason=reason,
            changed_at=changed_at,
            changed_by=actor_name,
        )

        with self._lock:
            self._availability[
                (environment, functionality_id)
            ] = availability
            self._last_action[
                (environment, functionality_id)
            ] = result

        if self._auditor is None:
            raise RuntimeError(
                "Control auditor is required for in-memory operation."
            )

        self._auditor.record_functionality_control(
            actor=actor,
            environment=environment,
            functionality_id=functionality_id,
            action=action,
            reason=reason,
            changed_at=changed_at,
        )

        return result

    @classmethod
    def _validate(
        cls,
        environment: str,
        functionality_id: str,
    ) -> None:
        if environment not in cls.VALID_ENVIRONMENTS:
            raise ValueError("environment must be 'kids' or 'adults'")

        if not functionality_id.strip():
            raise ValueError("functionality_id is required")

        if not is_known_functionality(functionality_id):
            raise ValueError(
                f"Unknown functionality_id: {functionality_id}"
            )

"""
AI Runtime capability errors.

These exceptions provide explicit Runtime-level failure semantics.

The Runtime must distinguish between:

- invalid requests
- unknown capabilities
- duplicate capability registration
- governance rejection
- capability execution failure

Provider-specific exceptions must not leak through the Runtime
boundary.
"""

from __future__ import annotations


class RuntimeCapabilityError(Exception):
    """
    Base exception for Runtime capability failures.
    """


class InvalidCapabilityRequestError(
    RuntimeCapabilityError
):
    """
    Raised when a capability request is invalid.
    """


class CapabilityNotRegisteredError(
    RuntimeCapabilityError
):
    """
    Raised when no executor is registered for a capability.
    """


class CapabilityAlreadyRegisteredError(
    RuntimeCapabilityError
):
    """
    Raised when an executor is registered more than once.
    """


class CapabilityExecutionError(
    RuntimeCapabilityError
):
    """
    Raised when a capability executor fails during execution.

    The original provider/implementation exception is retained
    as the exception cause, but callers receive a Runtime-level
    error type.
    """


class CapabilityGovernanceError(
    RuntimeCapabilityError
):
    """
    Raised when governance evaluation itself fails.

    This is deliberately different from a governance rejection.

    Rejection:
        governance worked and said no.

    Governance error:
        governance could not produce a valid decision.
    """

"""
AI Runtime capability execution contract.

This module defines the interface between the AI Runtime
and concrete capability executors.

Architecture:

AI Runtime
    |
    v
Capability Executor Interface
    |
    +---- Live Intelligence Executor
    +---- Podcast Discovery Executor
    +---- Learning Executor
    +---- Evaluation Executor
    +---- Future Executors

The Runtime depends on this contract.

It does not depend directly on:

- agents;
- providers;
- SDKs;
- external APIs.

Concrete executors are responsible for translating the
provider-independent CapabilityRequest into the appropriate
capability implementation.

This module defines contracts only.

It does not:

- execute capabilities;
- call providers;
- apply governance;
- manage runtime state.
"""

from __future__ import annotations

from abc import ABC, abstractmethod

from src.ai_runtime.capabilities.models import (
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)


class CapabilityExecutor(ABC):
    """
    Abstract executor for one Runtime capability.

    Each concrete executor must:

    - declare the capability it handles;
    - implement asynchronous execution;
    - return a provider-independent result.

    Executors must not bypass:

        CapabilityGateway
            |
            v
        Governance
    """

    @property
    @abstractmethod
    def capability(
        self,
    ) -> CapabilityType:
        """
        Return the capability handled by this executor.

        Returns:

            CapabilityType
        """

        raise NotImplementedError(
            "Capability must be implemented."
        )

    @abstractmethod
    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a capability request.

        Implementations must:

        - consume CapabilityRequest;
        - return CapabilityResult;
        - remain provider-independent.

        Implementations must not:

        - bypass Runtime governance;
        - directly modify application state;
        - create uncontrolled external side effects.
        """

        raise NotImplementedError(
            "Executor execution must be implemented."
        )

This adapter connects the provider-independent AI Runtime

Architecture:

FlashcardGenerationExecutor

"""
AI Runtime Capability Gateway.

Controlled execution boundary between applications and
concrete AI capabilities.

Flow:

- manage memory;
- modify agents;
- perform external side effects.

from __future__ import annotations

from typing import Protocol

from src.ai_runtime.capabilities.errors import (

    CapabilityGovernanceError,

class GovernanceDecisionResult(Protocol):
    """
    Minimal governance result required by Runtime.
    """

    @property
    def allowed(self) -> bool:
        """
        Whether execution is permitted.
        """
        ...


class CapabilityGovernance(Protocol):
    """
    Governance boundary required by Runtime gateway.
    """

    def evaluate(
        self,
        request: CapabilityRequest,
    ) -> GovernanceDecisionResult:
        """
        Evaluate whether a capability may execute.
        """
        ...


class CapabilityGateway:
    """
    Controlled Runtime entry point for capability execution.

    Responsibilities:

    - governance enforcement;
    - capability resolution;
    - execution delegation.

    It is intentionally not a business layer.
    """

    def __init__(
        self,
        registry: CapabilityRegistry,
        governance: CapabilityGovernance,
    ) -> None:
        """
        Create capability gateway.

        Dependencies are injected from composition.
        """

        if registry is None:
            raise ValueError(
                "CapabilityRegistry cannot be None."
            )

        if governance is None:
            raise ValueError(
                "CapabilityGovernance cannot be None."
            )

        self._registry = registry
        self._governance = governance

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute an approved capability.

        Execution order:

        1. Validate request
        2. Evaluate governance
        3. Resolve executor
        4. Execute capability

        Governance is always evaluated before
        capability execution.
        """

        if request is None:
            raise ValueError(
                "CapabilityRequest cannot be None."
            )

        try:
            governance_result = (
                self._governance.evaluate(
                    request,
                )
            )

        except Exception as exc:
            raise CapabilityGovernanceError(
                "Governance evaluation failed."
            ) from exc

        if not governance_result.allowed:
            return CapabilityResult(
                request_id=request.request_id,
                capability=request.capability,
                status=(
                    CapabilityExecutionStatus.REJECTED
                ),
                error=(
                    "Capability execution was rejected "
                    "by governance."
                ),
                metadata={
                    "governance_decision": "rejected",
                },
            )

        try:
            executor = self._registry.resolve(
                request.capability,
            )

        except Exception as exc:
            raise CapabilityExecutionError(
                "Capability resolution failed: "
                f"{request.capability.value}"
            ) from exc

        try:
            return await executor.execute(
                request,
            )

        except Exception as exc:
            raise CapabilityExecutionError(
                "Capability executor failed: "
                f"{request.capability.value}"
            ) from exc

"""
AI Runtime capability registry.

The registry maps logical Runtime capabilities to their
concrete executors.

The registry:

- resolves executors by capability;
- prevents duplicate registrations;
- provides explicit dependency injection;
- validates registrations.

The registry does not:

- execute capabilities;
- perform governance decisions;
- know about providers;
- manage application state.
"""

from __future__ import annotations

from src.ai_runtime.capabilities.errors import (
    CapabilityAlreadyRegisteredError,
    CapabilityNotRegisteredError,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityType,
)


class CapabilityRegistry:
    """
    Dependency-injection registry for Runtime capabilities.

    The registry is intentionally simple.

    Composition creates it.
    Gateway consumes it.

    It does not own lifecycle management.
    """

    def __init__(
        self,
        executors: list[CapabilityExecutor] | None = None,
    ) -> None:
        """
        Create capability registry.

        Args:

            executors:
                Optional initial executor collection.
        """

        self._executors: dict[
            CapabilityType,
            CapabilityExecutor,
        ] = {}

        for executor in executors or []:
            self.register(
                executor,
            )

    def register(
        self,
        executor: CapabilityExecutor,
    ) -> None:
        """
        Register an executor for its declared capability.

        Raises:

            ValueError:
                If executor is invalid.

            CapabilityAlreadyRegisteredError:
                If capability already exists.
        """

        if executor is None:
            raise ValueError(
                "CapabilityExecutor cannot be None."
            )

        capability = executor.capability

        if not isinstance(
            capability,
            CapabilityType,
        ):
            raise ValueError(
                "Executor capability must be a valid "
                "CapabilityType."
            )

        if capability in self._executors:
            raise CapabilityAlreadyRegisteredError(
                "Capability executor already registered: "
                f"{capability.value}"
            )

        self._executors[capability] = executor

    def resolve(
        self,
        capability: CapabilityType,
    ) -> CapabilityExecutor:
        """
        Resolve the executor responsible for a capability.

        Raises:

            CapabilityNotRegisteredError:
                When no executor exists.
        """

        if not isinstance(
            capability,
            CapabilityType,
        ):
            raise ValueError(
                "Capability must be a valid CapabilityType."
            )

        try:
            return self._executors[capability]

        except KeyError as exc:
            raise CapabilityNotRegisteredError(
                "No executor registered for capability: "
                f"{capability.value}"
            ) from exc

    def contains(
        self,
        capability: CapabilityType,
    ) -> bool:
        """
        Return whether an executor is registered.
        """

        return capability in self._executors

    def capabilities(
        self,
    ) -> tuple[CapabilityType, ...]:
        """
        Return all registered capabilities.

        Returned as immutable tuple to prevent
        external mutation.
        """

        return tuple(
            self._executors.keys()
        )

class FlashDealCard:

            "character_message": self.character_message,
            "product_name": self.product_name,
            "product_image_url": self.product_image_url,

        }

        for field_name, value in required_text_fields.items():
            if not value.strip():
                raise ValueError(

        if not self.highlights:
            raise ValueError(

            raise ValueError(

            self.previous_price is not None

        ):
            raise ValueError(

            )

        if (
            self.discount_percentage is not None
            and not 0 <= self.discount_percentage <= 100
        ):
            raise ValueError(
                "FlashDealCard discount_percentage must be "

        self._validate_url(
            self.more_information_url,
            "more_information_url",
        )
        self._validate_url(

            "get_deal_url",

        *,

        character_id: str,
        character_name: str,
        character_message: str,
        relevance_reason: str,
        product_image_url: str,
        tagline: str,

        Product, pricing and vendor information comes from the
        DealCard. Character and personalisation information is

        if (
            deal.previous_price is not None
            and deal.previous_price > deal.price
        ):
            savings = round(
                deal.previous_price - deal.price,
                2,
            )

        return cls(
            deal_id=deal.deal_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            product_name=deal.product_name,
            product_image_url=product_image_url,

            metadata=(
                dict(metadata)
                if metadata is not None
                else {}
            ),
        )

    @property
    def has_discount(self) -> bool:
        """Return whether a verified discount exists."""

        return (

    def display_savings(self) -> str | None:
        """Return a UI-friendly savings label."""

        if self.savings is None:

    def render(
        self,
        card: FlashDealCard,
    ) -> str:
        """
        Render a FlashDealCard as user-facing Markdown.

        No external calls or business decisions are performed.
        """

        sections: list[str] = []

        sections.append(
            f"## {card.character_name}"
        )

        sections.append(
            f"> {card.character_message}"
        )

            f"### {card.badge}"
        )

            f"**{card.product_name}**"
        )

            f"*{card.tagline}*"
        )

        )

            card.relevance_reason

        sections.append(
            "### AVAILABLE FROM"

        )

class CommerceLinkType(str, Enum):
    """
    Type of outbound commerce link.
    """

    DIRECT = "direct"

    AFFILIATE = "affiliate"


@dataclass(frozen=True)
class CommerceLink:
    """
    Outbound link to the external vendor.

    EasyChatBot does not process the transaction.

    The destination remains the vendor's website.
    """

    vendor: str

    destination_url: str

    link_type: CommerceLinkType = (
        CommerceLinkType.DIRECT
    )

    tracking_url: str | None = None

    def __post_init__(self) -> None:
        """
        Validate the commerce link.
        """

        if not self.vendor.strip():
            raise ValueError(
                "Commerce vendor cannot be empty."
            )

        if not self.destination_url.strip():
            raise ValueError(
                "Commerce destination URL "
                "cannot be empty."
            )

        if not (
            self.destination_url.startswith(
                "https://"
            )
            or self.destination_url.startswith(
                "http://"
            )
        ):
            raise ValueError(
                "Commerce destination URL must "
                "use HTTP or HTTPS."
            )

        if self.tracking_url is not None:
            if not self.tracking_url.strip():
                raise ValueError(
                    "Commerce tracking URL cannot "
                    "be empty when provided."
                )

            if not (
                self.tracking_url.startswith(
                    "https://"
                )
                or self.tracking_url.startswith(
                    "http://"
                )
            ):
                raise ValueError(
                    "Commerce tracking URL must "
                    "use HTTP or HTTPS."
                )

    @property
    def click_url(self) -> str:
        """
        Return the URL that should be opened by the user.

        A tracking URL takes precedence when one is
        available.
        """

        return (
            self.tracking_url
            if self.tracking_url is not None
            else self.destination_url
        )

"""
EasyChatBot â€” Memory Context Builder.

H3-01 â€” Memory-Aware Live Intelligence.

The builder operates on the APPLICATION memory contract:

    MemoryService
        â†“
    MemoryRecord
        â†“
    MemoryContextBuilder
        â†“
    MemoryContext

It deliberately does NOT consume MemoryEntry directly.

MemoryEntry belongs to the persistence/runtime storage boundary.
MemoryRecord is the application-facing memory contract.

Responsibilities:
- validate already-retrieved memory
- enforce user isolation
- determine deterministic relevance
- bound memory count
- bound context size
- produce provider-independent context

Does not:
- retrieve memory
- persist memory
- call an LLM
- call a provider
- execute an agent
- mutate ExecutionContext
- perform semantic/vector search
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
import re

from backend.memory.memory_record import (
    MemoryRecord,
)


DEFAULT_MAX_MEMORY_ITEMS = 5
DEFAULT_MAX_CONTEXT_CHARACTERS = 4000

_WORD_PATTERN = re.compile(
    r"[\wÃ€-Ã¿'-]+",
    re.UNICODE,
)


# Application-level memory categories.
#
# These are deliberately strings because MemoryService exposes
# MemoryRecord.category rather than MemoryEntry.memory_type.
_MEMORY_CATEGORY_PRIORITY: dict[str, int] = {
    "preference": 30,
    "learning_goal": 30,
    "discovery_context": 20,
    "saved_item": 10,
    "conversation": 10,
}


@dataclass(frozen=True)
class MemoryContext:
    """
    Bounded, provider-independent learner context.
    """

    entries: tuple[MemoryRecord, ...]
    text: str

    @property
    def is_empty(self) -> bool:
        """Return True when no memory was selected."""
        return not self.entries

    @property
    def memory_count(self) -> int:
        """Return the number of selected memories."""
        return len(self.entries)


class MemoryContextBuilder:
    """
    Build bounded learner context from MemoryRecord objects.

    The builder is stateless and deterministic.
    """

    def __init__(
        self,
        *,
        max_memory_items: int = DEFAULT_MAX_MEMORY_ITEMS,
        max_context_characters: int = (
            DEFAULT_MAX_CONTEXT_CHARACTERS
        ),
    ) -> None:
        if max_memory_items < 1:
            raise ValueError(
                "max_memory_items must be greater than zero."
            )

        if max_context_characters < 1:
            raise ValueError(
                "max_context_characters must be greater than zero."
            )

        self._max_memory_items = max_memory_items
        self._max_context_characters = (
            max_context_characters
        )

    def build(
        self,
        *,
        user_id: str,
        request: str,
        memories: list[MemoryRecord]
        | tuple[MemoryRecord, ...],
    ) -> MemoryContext:
        """
        Build bounded learner context.

        Memory is supplied by the caller.

        The builder does not retrieve or persist memory.
        """

        if not user_id.strip():
            raise ValueError(
                "user_id cannot be empty."
            )

        if not request.strip():
            raise ValueError(
                "request cannot be empty."
            )

        if not memories:
            return MemoryContext(
                entries=(),
                text="",
            )

        request_tokens = self._tokens(
            request
        )

        candidates: list[
            tuple[
                int,
                datetime,
                str,
                MemoryRecord,
            ]
        ] = []

        for memory in memories:

            if not isinstance(
                memory,
                MemoryRecord,
            ):
                continue

            # Defence in depth.
            if memory.user_id != user_id:
                continue

            if not memory.category.strip():
                continue

            if not memory.content.strip():
                continue

            score = self._score(
                memory=memory,
                request_tokens=request_tokens,
            )

            # A memory category alone does NOT make the memory
            # relevant to the current request.
            if score <= 0:
                continue

            candidates.append(
                (
                    score,
                    memory.created_at,
                    memory.content,
                    memory,
                )
            )

        # Deterministic ordering:
        #
        # 1. relevance descending
        # 2. newest memory first
        # 3. content as stable tie-breaker
        candidates.sort(
            key=lambda item: (
                -item[0],
                -item[1].timestamp(),
                item[2].casefold(),
            )
        )

        selected: list[MemoryRecord] = []
        rendered_items: list[str] = []

        current_length = len(
            "Learner context:\n"
        )

        for (
            _score,
            _created_at,
            _content,
            memory,
        ) in candidates:

            if (
                len(selected)
                >= self._max_memory_items
            ):
                break

            line = self._render_memory(
                memory
            )

            separator_length = (
                1
                if rendered_items
                else 0
            )

            projected_length = (
                current_length
                + separator_length
                + len(line)
            )

            if (
                projected_length
                > self._max_context_characters
            ):
                # Do not partially truncate an individual
                # memory because that could alter meaning.
                continue

            selected.append(memory)
            rendered_items.append(line)

            current_length = projected_length

        if not selected:
            return MemoryContext(
                entries=(),
                text="",
            )

        text = (
            "Learner context:\n"
            + "\n".join(
                f"- {item}"
                for item in rendered_items
            )
        )

        if (
            len(text)
            > self._max_context_characters
        ):
            raise AssertionError(
                "MemoryContextBuilder produced "
                "an oversized context."
            )

        return MemoryContext(
            entries=tuple(selected),
            text=text,
        )

    @staticmethod
    def _tokens(
        value: str,
    ) -> set[str]:
        """
        Return normalised tokens for deterministic matching.
        """

        return {
            token.casefold()
            for token in _WORD_PATTERN.findall(
                value
            )
            if len(token) > 1
        }

    @classmethod
    def _score(
        cls,
        *,
        memory: MemoryRecord,
        request_tokens: set[str],
    ) -> int:
        """
        Calculate deterministic relevance.

        Relevance requires actual token overlap.

        This prevents an unrelated preference such as
        "Prefers hiking" from being injected into a request
        about "European AI regulation" simply because it is
        a preference.
        """

        content_tokens = cls._tokens(
            memory.content
        )

        category_tokens = cls._tokens(
            memory.category
        )

        overlap = request_tokens & (
            content_tokens
            | category_tokens
        )

        if not overlap:
            return 0

        category = (
            memory.category.strip().casefold()
        )

        category_priority = (
            _MEMORY_CATEGORY_PRIORITY.get(
                category,
                10,
            )
        )

        return (
            category_priority
            + len(overlap)
        )

    @staticmethod
    def _render_memory(
        memory: MemoryRecord,
    ) -> str:
        """
        Render memory explicitly as learner context.

        Memory content is treated as data, not instructions.
        """

        category = (
            memory.category.strip()
        )

        content = " ".join(
            memory.content.split()
        )

        return (
            f"[{category}] {content}"
        )

"""
Hackathon Memory persistence.

This module provides the minimal persistent Memory Store
required by the hackathon.

Architecture:

MemoryService
      |
      v
MemoryStore
      |
      v
SQLite

The store is deliberately small.

It is responsible only for persistence and retrieval of
MemoryEntry objects.

It does not:
- decide what should be remembered
- call an LLM
- perform semantic retrieval
- execute agents
- perform governance decisions
- manage purchases
"""

from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryStore:
    """
    SQLite-backed persistence for hackathon memories.

    The database schema is created automatically when the
    store is initialised.
    """

    def __init__(
        self,
        database_path: str | Path,
    ) -> None:
        self._database_path = str(database_path)

        self._initialise_database()

    def _connect(self) -> sqlite3.Connection:
        """
        Create a SQLite connection.
        """

        connection = sqlite3.connect(
            self._database_path
        )

        connection.row_factory = sqlite3.Row

        return connection

    def _initialise_database(self) -> None:
        """
        Create the Memory table if it does not exist.
        """

        with self._connect() as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    memory_id TEXT PRIMARY KEY,
                    user_id TEXT NOT NULL,
                    memory_type TEXT NOT NULL,
                    content TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    metadata_json TEXT NOT NULL
                )
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_id
                ON memories(user_id)
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_type
                ON memories(user_id, memory_type)
                """
            )

    def save(
        self,
        memory: MemoryEntry,
    ) -> MemoryEntry:
        """
        Persist a MemoryEntry.

        Existing entries with the same memory_id are
        replaced.
        """

        with self._connect() as connection:
            connection.execute(
                """
                INSERT OR REPLACE INTO memories (
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                )
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.memory_id,
                    memory.user_id,
                    memory.memory_type.value,
                    memory.content,
                    memory.created_at.isoformat(),
                    memory.updated_at.isoformat(),
                    json.dumps(
                        memory.metadata,
                        ensure_ascii=False,
                    ),
                ),
            )

        return memory

    def get(
        self,
        memory_id: str,
    ) -> MemoryEntry | None:
        """
        Retrieve a single memory by identifier.
        """

        with self._connect() as connection:
            row = connection.execute(
                """
                SELECT
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            ).fetchone()

        if row is None:
            return None

        return self._row_to_memory(row)

    def list_for_user(
        self,
        user_id: str,
        memory_type: MemoryType | None = None,
    ) -> list[MemoryEntry]:
        """
        Retrieve memories belonging to a user.

        Results are returned newest first.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        with self._connect() as connection:

            if memory_type is None:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                    ORDER BY updated_at DESC
                    """,
                    (user_id,),
                ).fetchall()

            else:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                      AND memory_type = ?
                    ORDER BY updated_at DESC
                    """,
                    (
                        user_id,
                        memory_type.value,
                    ),
                ).fetchall()

        return [
            self._row_to_memory(row)
            for row in rows
        ]

    def delete(
        self,
        memory_id: str,
    ) -> bool:
        """
        Delete a memory.

        Returns True when a memory was deleted.
        """

        with self._connect() as connection:
            cursor = connection.execute(
                """
                DELETE FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            )

        return cursor.rowcount > 0

    @staticmethod
    def _row_to_memory(
        row: sqlite3.Row,
    ) -> MemoryEntry:
        """
        Convert a SQLite row into a MemoryEntry.
        """

        return MemoryEntry(
            memory_id=row["memory_id"],
            user_id=row["user_id"],
            memory_type=MemoryType(
                row["memory_type"]
            ),
            content=row["content"],
            created_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["created_at"]
                )
            ),
            updated_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["updated_at"]
                )
            ),
            metadata=json.loads(
                row["metadata_json"]
            ),
        )

"""
EasyChatBot Hackathon â€” Podcast Discovery Agent.

The Agent converts explicit learner context and preference memory into
provider-neutral search intent.

It does not call Apple/Spotify, write memory, select a provider, or invent
podcast metadata.
"""

from __future__ import annotations

import re

from src.ai_runtime.memory.models import MemoryEntry, MemoryType
from src.ai_runtime.podcast.models import (
    LearningContext,
    PodcastSearchIntent,
)


class PodcastDiscoveryAgent:
    """Build provider-neutral podcast discovery intent."""

    _STOP_WORDS = {
        "a", "an", "and", "are", "be", "for", "from", "i", "in", "is",
        "like", "of", "on", "podcasts", "prefers", "prefer", "the", "to",
        "with", "you", "your",
    }

    def build_search_intent(
        self,
        *,
        user_id: str,
        learning_context: LearningContext,
        memories: list[MemoryEntry],
        max_results: int = 10,
    ) -> PodcastSearchIntent:
        """
        Convert explicit preferences + learning-world context into intent.

        Only PREFERENCE memories are used, matching the existing
        EasyChatBot discovery contract.
        """

        if not user_id.strip():
            raise ValueError("Podcast Agent user_id cannot be empty.")

        preferences = [
            memory
            for memory in memories
            if memory.user_id == user_id
            and memory.memory_type == MemoryType.PREFERENCE
        ]

        topics: list[str] = []
        keywords: list[str] = []

        for memory in preferences:
            topics.extend(
                self._metadata_terms(
                    memory.metadata,
                    "topics",
                    "interests",
                    "preferred_topics",
                )
            )
            keywords.extend(
                self._metadata_terms(
                    memory.metadata,
                    "keywords",
                )
            )

            if not topics and not keywords:
                keywords.extend(
                    self._content_keywords(memory.content)
                )

        # The learning world is always part of the search context.
        topics.append(learning_context.world_id)

        # Target language is deliberately a hard eligibility constraint,
        # not just another search keyword.
        return PodcastSearchIntent(
            user_id=user_id,
            learning_context=learning_context,
            topics=tuple(self._dedupe(topics)),
            keywords=tuple(self._dedupe(keywords)),
            language_required=True,
            max_results=max_results,
        )

    @staticmethod
    def _metadata_terms(
        metadata: dict[str, str],
        *keys: str,
    ) -> list[str]:
        terms: list[str] = []
        for key in keys:
            raw = metadata.get(key, "")
            if raw:
                terms.extend(
                    item.strip()
                    for item in re.split(r"[,;|]", raw)
                    if item.strip()
                )
        return terms

    @classmethod
    def _content_keywords(cls, content: str) -> list[str]:
        words = re.findall(
            r"[A-Za-zÃ€-Ã¿0-9][A-Za-zÃ€-Ã¿0-9'-]{2,}",
            content.lower(),
        )
        return [
            word for word in words
            if word not in cls._STOP_WORDS
        ][:8]

    @staticmethod
    def _dedupe(values: list[str]) -> list[str]:
        seen: set[str] = set()
        result: list[str] = []
        for value in values:
            key = value.strip().lower()
            if key and key not in seen:
                seen.add(key)
                result.append(value.strip())
        return result

"""

Uses Apple's iTunes Search API for read-only podcast catalogue discovery.
"""

from __future__ import annotations

import json
from urllib.parse import urlencode
from urllib.request import Request, urlopen

from src.ai_runtime.podcast.models import (
    PodcastProviderName,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)


class ApplePodcastProvider:
    """Read-only Apple podcast catalogue provider."""

    BASE_URL = "https://itunes.apple.com/search"

    def __init__(self, *, timeout_seconds: float = 10.0) -> None:
        self._timeout_seconds = timeout_seconds

    def search(
        self,
        intent: PodcastSearchIntent,
    ) -> list[PodcastSearchResult]:
        query_terms = list(intent.topics) + list(intent.keywords)
        params = {
            "term": " ".join(query_terms[:6]),
            "country": intent.learning_context.country_code.upper(),
            "media": "podcast",
            "entity": "podcast",
            "limit": str(intent.max_results),
            "explicit": "No",
        }

        request = Request(
            f"{self.BASE_URL}?{urlencode(params)}",
            headers={
                "Accept": "application/json",
                "User-Agent": "EasyChatBot-Hackathon/1.0",
            },
            method="GET",
        )

        with urlopen(
            request,
            timeout=self._timeout_seconds,
        ) as response:
            payload = json.loads(
                response.read().decode("utf-8")
            )

        return [
            self._normalise(item)
            for item in payload.get("results", [])
            if self._is_podcast_result(item)
        ]

    @staticmethod
    def _is_podcast_result(item: dict[str, object]) -> bool:
        return (
            str(item.get("kind", "")).lower() == "podcast"
            or str(item.get("wrapperType", "")).lower() == "track"
        )

    @staticmethod
    def _normalise(
        item: dict[str, object],
    ) -> PodcastSearchResult:
        language = normalise_language(
            str(item.get("language") or "unknown")
        )

        return PodcastSearchResult(
            provider=PodcastProviderName.APPLE,
            external_id=str(
                item.get("collectionId")
                or item.get("trackId")
                or ""
            ),
            title=str(
                item.get("collectionName")
                or item.get("trackName")
                or ""
            ),
            description=str(
                item.get("description")
                or item.get("longDescription")
                or ""
            ),
            publisher=str(
                item.get("artistName")
                or "Unknown publisher"
            ),
            languages=(language,),
            url=str(
                item.get("collectionViewUrl")
                or item.get("trackViewUrl")
                or ""
            ),
            image_url=str(
                item.get("artworkUrl600")
                or item.get("artworkUrl100")
                or ""
            ) or None,
            published_at=str(
                item.get("releaseDate") or ""
            ) or None,
            explicit=(
                str(
                    item.get("trackExplicitness", "")
                ).lower() == "explicit"
            ),
            metadata={"provider": "apple_podcasts"},
        )

from __future__ import annotations

from dataclasses import dataclass, field
from uuid import uuid4


@dataclass(frozen=True)
class VocabularyItem:
    """A vocabulary item presented to the learner."""

    term: str
    meaning: str
    example: str

    def __post_init__(self) -> None:
        for name, value in {
            "term": self.term,
            "meaning": self.meaning,
            "example": self.example,
        }.items():
            if not value.strip():
                raise ValueError(
                    f"VocabularyItem {name} cannot be empty."
                )


@dataclass(frozen=True)
class ComprehensionQuestion:
    """A multiple-choice comprehension question."""

    question: str
    options: tuple[str, ...]
    correct_option: int

    def __post_init__(self) -> None:
        if not self.question.strip():
            raise ValueError(
                "ComprehensionQuestion question cannot be empty."
            )

        if len(self.options) < 2:
            raise ValueError(
                "ComprehensionQuestion requires at least two options."
            )

        if any(not option.strip() for option in self.options):
            raise ValueError(
                "ComprehensionQuestion options cannot be empty."
            )

        if not 0 <= self.correct_option < len(self.options):
            raise ValueError(
                "ComprehensionQuestion correct_option is out of range."
            )


@dataclass(frozen=True)
class PodcastLearningChallenge:
    """Learner-facing challenge derived from a verified podcast recommendation."""

    challenge_id: str
    user_id: str
    podcast_id: str
    podcast_title: str
    podcast_url: str
    target_language: str
    learner_level: str
    vocabulary: tuple[VocabularyItem, ...]
    comprehension_questions: tuple[ComprehensionQuestion, ...]
    discussion_prompt: str
    introduction: str
    metadata: dict[str, str] = field(default_factory=dict)

    def __post_init__(self) -> None:
        required = {
            "challenge_id": self.challenge_id,
            "user_id": self.user_id,
            "podcast_id": self.podcast_id,
            "podcast_title": self.podcast_title,
            "podcast_url": self.podcast_url,
            "target_language": self.target_language,
            "learner_level": self.learner_level,
            "discussion_prompt": self.discussion_prompt,
            "introduction": self.introduction,
        }

        for name, value in required.items():
            if not value.strip():
                raise ValueError(
                    f"PodcastLearningChallenge {name} cannot be empty."
                )

        if not (
            self.podcast_url.startswith("https://")
            or self.podcast_url.startswith("http://")
        ):
            raise ValueError(
                "PodcastLearningChallenge podcast_url must use HTTP or HTTPS."
            )

        if not self.vocabulary:
            raise ValueError(
                "PodcastLearningChallenge requires vocabulary."
            )

        if not self.comprehension_questions:
            raise ValueError(
                "PodcastLearningChallenge requires comprehension questions."
            )

    @classmethod
    def create(
        cls,

        user_id: str,
        podcast_id: str,
        podcast_title: str,
        podcast_url: str,
        target_language: str,
        learner_level: str,

        comprehension_questions: tuple[ComprehensionQuestion, ...],
        discussion_prompt: str,
        introduction: str,
        metadata: dict[str, str] | None = None,
    ) -> "PodcastLearningChallenge":
        return cls(
            challenge_id=str(uuid4()),
            user_id=user_id,
            podcast_id=podcast_id,
            podcast_title=podcast_title,
            podcast_url=podcast_url,
            target_language=target_language,
            learner_level=learner_level,
            vocabulary=vocabulary,
            comprehension_questions=comprehension_questions,
            discussion_prompt=discussion_prompt,
            introduction=introduction,
            metadata=dict(metadata or {}),
        )

            challenge.comprehension_questions,

            for option_index, option in enumerate(
                question.options,
                start=1,

            [

ListeningProposal

Learner decision

from dataclasses import dataclass

class ListeningProposalAction(StrEnum):

    Actions that may be presented to the learner.

        user_id:
            Learner receiving the proposal.

        podcast_id:
            External podcast identifier.

        podcast_title:
            Human-readable podcast title.

        podcast_url:
            Verified podcast URL.

        character_id:
            Character presenting the proposal.

        character_name:
            Human-readable character name.

        message:
            Character-facing proposal message.

        action:
            Proposed learner action.

        reason:
            Explanation of why the podcast is being proposed.

        target_language:
            Language associated with the learning experience.

        learner_level:
            Learner level associated with the recommendation.
    """

    user_id: str
    podcast_id: str
    podcast_title: str
    podcast_url: str
    character_id: str
    character_name: str
    message: str
    action: ListeningProposalAction
    reason: str
    target_language: str
    learner_level: str

            "user_id": self.user_id,
            "podcast_id": self.podcast_id,
            "podcast_title": self.podcast_title,
            "podcast_url": self.podcast_url,
            "character_id": self.character_id,
            "character_name": self.character_name,
            "message": self.message,
            "reason": self.reason,
            "target_language": self.target_language,
            "learner_level": self.learner_level,
        }

        for name, value in values.items():

                "https://",

            raise ValueError(
                "ListeningProposal podcast_url must use "
                "HTTP or HTTPS."
            )

    @property
    def is_passive(self) -> bool:
        """
        Return True because this object represents a proposal only.

        Creating a ListeningProposal never causes playback or
        another external side effect.
        """

        return True

class PodcastRecommendationService:
    """Search providers and select a personalised candidate."""

    def __init__(self, providers: list[PodcastProvider]) -> None:
        if not providers:
            raise ValueError(
                "PodcastRecommendationService requires at least one provider."
            )
        self._providers = list(providers)

    def recommend(
        self,
        *,
        intent: PodcastSearchIntent,
        character_id: str,
        character_name: str,
        character_message: str,
    ) -> PodcastRecommendation:
        candidates: list[PodcastSearchResult] = []

        for provider in self._providers:
            candidates.extend(provider.search(intent))

        ranked = self.rank_candidates(
            candidates=candidates,
            intent=intent,
        )

        if not ranked:
            raise LookupError(
                "No podcast matched the learner's language and preferences."
            )

        result, score, reason = ranked[0]

        return PodcastRecommendation(
            recommendation_id=str(uuid4()),
            user_id=intent.user_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            result=result,
            relevance_reason=reason,
            match_score=score,
        )

    @classmethod
    def rank_candidates(
        cls,
        *,
        candidates: list[PodcastSearchResult],
        intent: PodcastSearchIntent,
    ) -> list[tuple[PodcastSearchResult, float, str]]:
        ranked: list[tuple[PodcastSearchResult, float, str]] = []

        target_language = normalise_language(
            intent.learning_context.target_language
        )

        preference_terms = [
            term.lower()
            for term in list(intent.topics) + list(intent.keywords)
            if term.strip()
        ]

        for candidate in candidates:
            candidate_languages = {
                normalise_language(language)
                for language in candidate.languages
            }

            # Critical learning-platform rule:
            # unknown/wrong-language content is not eligible.
            if (
                intent.language_required
                and target_language not in candidate_languages
            ):
                continue

            haystack = " ".join(
                [
                    candidate.title,
                    candidate.description,
                    candidate.publisher,
                ]
            ).lower()

            matched_terms = [
                term
                for term in preference_terms
                if cls._term_matches(term, haystack)
            ]

            score = 50.0

            if matched_terms:
                score += min(30.0, len(matched_terms) * 10.0)

            if intent.learning_context.world_id.lower() in haystack:
                score += 10.0

            level = intent.learning_context.learner_level.lower()
            if level and level in haystack:
                score += 10.0

            reason = cls._build_reason(
                language=intent.learning_context.target_language,
                matched_terms=matched_terms,
                world_id=intent.learning_context.world_id,
            )

            ranked.append(
                (candidate, min(score, 100.0), reason)
            )

        ranked.sort(
            key=lambda item: (
                item[1],
                item[0].published_at or "",
            ),
            reverse=True,
        )

        return ranked[: intent.max_results]

    @staticmethod
    def _term_matches(term: str, haystack: str) -> bool:
        term = term.strip().lower()
        if not term:
            return False
        if term in haystack:
            return True

        words = [
            word
            for word in re.findall(r"[a-z0-9Ã€-Ã¿'-]+", term)
            if len(word) > 2
        ]
        return bool(words) and any(
            word in haystack
            for word in words
        )

    @staticmethod
    def _build_reason(
        *,
        language: str,
        matched_terms: list[str],
        world_id: str,
    ) -> str:
        reasons = [f"It is available in {language}."]

        if matched_terms:
            reasons.append(
                "It matches your interests in "
                + ", ".join(matched_terms[:3])
                + "."
            )

        reasons.append(
            f"It fits your {world_id} learning world."
        )

        return " ".join(reasons)

from src.ai_runtime.podcast.models import (

class SpotifyPodcastProvider:
    """Read-only Spotify catalogue provider."""

    BASE_URL = "https://api.spotify.com/v1/search"

    def __init__(
        self,
        *,
        access_token: str,
        timeout_seconds: float = 10.0,
        market: str | None = None,
    ) -> None:
        if not access_token.strip():
            raise ValueError(
                "SpotifyPodcastProvider access_token cannot be empty."
            )
        self._access_token = access_token
        self._timeout_seconds = timeout_seconds
        self._market = market

    def search(
        self,
        intent: PodcastSearchIntent,
    ) -> list[PodcastSearchResult]:
        query_terms = list(intent.topics) + list(intent.keywords)
        params = {
            "q": " ".join(query_terms[:6]),
            "type": "episode",
            "limit": str(intent.max_results),
        }

        market = self._market or intent.learning_context.country_code
        if market:
            params["market"] = market.upper()

        request = Request(
            f"{self.BASE_URL}?{urlencode(params)}",
            headers={
                "Accept": "application/json",
                "Authorization": f"Bearer {self._access_token}",
                "User-Agent": "EasyChatBot-Hackathon/1.0",
            },
            method="GET",
        )

        with urlopen(
            request,
            timeout=self._timeout_seconds,
        ) as response:
            payload = json.loads(
                response.read().decode("utf-8")
            )

        episodes = payload.get("episodes", {})
        if not isinstance(episodes, dict):
            return []

        items = episodes.get("items", [])
        if not isinstance(items, list):
            return []

        results: list[PodcastSearchResult] = []

        for item in items:
            if not isinstance(item, dict):
                continue

            result = self._normalise_episode(item)
            if result is not None:
                results.append(result)

        return results

    @staticmethod
    def _normalise_episode(
        item: dict[str, object],
    ) -> PodcastSearchResult | None:
        external_id = str(item.get("id") or "")
        title = str(item.get("name") or "")

        external_urls = item.get("external_urls")
        url = ""
        if isinstance(external_urls, dict):
            url = str(
                external_urls.get("spotify") or ""
            )

        if not external_id or not title or not url:
            return None

        raw_languages = item.get("languages", [])
        if not isinstance(raw_languages, list):
            return None

        languages = tuple(
            normalise_language(str(language))
            for language in raw_languages
            if str(language).strip()
        )

        # Unknown language is not invented. The recommendation service
        # will reject a candidate if language_required=True.
        if not languages:
            return None

        images = item.get("images", [])
        image_url = None
        if isinstance(images, list) and images:
            first = images[0]
            if isinstance(first, dict):
                image_url = str(first.get("url") or "") or None

        duration_ms = item.get("duration_ms")
        duration_seconds = (
            int(duration_ms) // 1000
            if isinstance(duration_ms, (int, float))
            else None
        )

        return PodcastSearchResult(
            provider=PodcastProviderName.SPOTIFY,
            external_id=external_id,
            title=title,
            description=str(item.get("description") or ""),
            publisher="Spotify Podcast",
            languages=languages,
            url=url,
            image_url=image_url,
            published_at=str(
                item.get("release_date") or ""
            ) or None,
            duration_seconds=duration_seconds,
            explicit=bool(item.get("explicit", False)),
            metadata={
                "provider": "spotify",
                "content_type": "episode",
            },
        )

@dataclass(frozen=True)
class CostCalculationInput:
    """
    Input required to calculate provider cost.

    Token values may be None when the provider did not supply

    None means unknown.
    0 means the provider explicitly reported zero.
    """

    provider: str

    model: str

    input_tokens: int | None

    output_tokens: int | None

    def __post_init__(self) -> None:

            )

        if (
            self.input_tokens is not None
            and self.input_tokens < 0
        ):
            raise ValueError(
                "Input tokens cannot be negative."
            )

        if (
            self.output_tokens is not None
            and self.output_tokens < 0
        ):
            raise ValueError(
                "Output tokens cannot be negative."
            )


@dataclass(frozen=True)
class ProviderCostResult:
    """
    Result of provider cost calculation.

    If either token value is unknown, the corresponding cost
    cannot be calculated reliably and is represented as None.

    This prevents unknown usage from being reported as zero cost.
    """

    provider: str

    model: str

    input_tokens: int | None

    output_tokens: int | None

    input_cost: Decimal | None

    output_cost: Decimal | None

    @property
    def total_cost(self) -> Decimal | None:
        """
        Return total calculated cost.

        Returns None when either required token cost is unknown.
        """

        if (
            self.input_cost is None
            or self.output_cost is None
        ):
            return None

        return self.input_cost + self.output_cost

    """
    Calculates provider execution costs.

    The service is provider-independent.

    Pricing is supplied explicitly rather than embedded
    in provider-specific logic.
    """

    def calculate(
        self,
        *,
        usage: CostCalculationInput,
        input_price_per_million_tokens: Decimal,
        output_price_per_million_tokens: Decimal,
    ) -> ProviderCostResult:
        """
        Calculate provider execution cost.

        None token usage means that the corresponding cost
        cannot be calculated reliably.

        0 token usage remains a real zero-cost calculation.

        Raises:
            ValueError: If usage is None or a supplied price
                is negative.

            )

            raise ValueError(
                "Input token price cannot be negative."
            )

        if output_price_per_million_tokens < 0:
            raise ValueError(
                "Output token price cannot be negative."
            )

        input_cost: Decimal | None

        if usage.input_tokens is None:
            input_cost = None
        else:
            input_cost = (
                Decimal(usage.input_tokens)
                / Decimal("1000000")
            ) * input_price_per_million_tokens

        output_cost: Decimal | None

        if usage.output_tokens is None:
            output_cost = None
        else:
            output_cost = (
                Decimal(usage.output_tokens)
                / Decimal("1000000")
            ) * output_price_per_million_tokens

        return ProviderCostResult(
            provider=usage.provider,
            model=usage.model,
            input_tokens=usage.input_tokens,
            output_tokens=usage.output_tokens,
            input_cost=input_cost,
            output_cost=output_cost,
        )

AI Runtime provider usage service.

Application service responsible for collecting
and aggregating AI Runtime usage events.

Responsibilities:

- record usage events;
- retrieve usage events through repository;
- generate usage summaries.

Non-responsibilities:

- provider execution;
- API calls;
- billing APIs;
- database implementation;
- dashboard rendering;
- agent execution.

Persistence is handled through the repository
boundary.
"""

from __future__ import annotations

from decimal import Decimal

from src.ai_runtime.providers.usage.models import (
    ProviderUsageEvent,
    ProviderUsageSummary,
)

from src.ai_runtime.providers.usage.repository import (
    ProviderUsageRepository,
)


class ProviderUsageService:
    """
    Application service for AI Runtime usage intelligence.

    The service depends on the repository abstraction.

    It does not know whether storage is:

    - memory;
    - SQLite;
    - PostgreSQL;
    - analytics warehouse.
    """

    def __init__(
        self,
        repository: ProviderUsageRepository,
    ) -> None:

        if repository is None:
            raise ValueError(
                "ProviderUsageRepository cannot be None."
            )

        self._repository = repository

    def record(
        self,
        event: ProviderUsageEvent,
    ) -> None:
        """
        Record one Runtime usage event.

        The event is delegated to the repository.

        The service does not:

        - call providers;
        - enrich events;
        - calculate external billing.
        """

        if event is None:
            raise ValueError(
                "Provider usage event cannot be None."
            )

        self._repository.save(
            event
        )

    def summary(
        self,
    ) -> ProviderUsageSummary:
        """
        Generate aggregated usage information.
        """

        events = (
            self._repository.get_all()
        )

        total_requests = len(
            events
        )

        total_input_tokens = sum(
            event.input_tokens
            for event in events
        )

        total_output_tokens = sum(
            event.output_tokens
            for event in events
        )

        successful_requests = sum(
            1
            for event in events
            if event.success
        )

        failed_requests = (
            total_requests
            - successful_requests
        )

        providers = tuple(
            sorted(
                {
                    event.provider
                    for event in events
                }
            )
        )

        total_cost = sum(
            (
                event.estimated_cost
                for event in events
            ),
            Decimal("0"),
        )

        return ProviderUsageSummary(
            total_requests=(
                total_requests
            ),
            total_input_tokens=(
                total_input_tokens
            ),
            total_output_tokens=(
                total_output_tokens
            ),
            successful_requests=(
                successful_requests
            ),
            failed_requests=(
                failed_requests
            ),
            providers=providers,
            total_cost=total_cost,
        )

    def clear(
        self,
    ) -> None:
        """
        Clear repository data.

        Mainly useful for:

        - tests;
        - demonstrations.

        Production systems should not rely
        on clearing runtime usage history.
        """

        self._repository.clear()
