"""
Voice Call Lifecycle Composition.

Composition root for the complete
voice call lifecycle application service.

Responsibilities:
- Build lifecycle dependencies.
- Wire execution coordinator.
- Wire connection services.
- Wire persistent voice call services.
- Wire voice event services.
- Return ready application facade.

Does NOT:
- Handle HTTP.
- Execute voice calls.
- Manage providers.
- Persist data directly.
"""

from __future__ import annotations



from app.application.uai.voice.voice_call_lifecycle_service import (
    VoiceCallLifecycleService,
)



from app.application.uai.voice.voice_call_execution_coordinator import (
    VoiceCallExecutionCoordinator,
)



from app.application.uai.voice.voice_room_service import (
    VoiceRoomService,
)



from app.application.uai.voice.voice_call_management_service import (
    VoiceCallManagementService,
)



from app.application.uai.voice.voice_call_session_service import (
    VoiceCallSessionService,
)



from app.application.uai.voice.voice_call_event_service import (
    VoiceCallEventService,
)



from app.application.composition.voice_connection_composition import (
    create_voice_connection_service,
)



from app.application.composition.voice_event_composition import (
    build_voice_call_event_service,
)



from app.infrastructure.voice.voice_room_adapter import (
    VoiceRoomAdapter,
)



from app.infrastructure.persistence.voice_call_repository import (
    InMemoryVoiceCallRepository,
)



from app.ai_runtime.voice.voice_call_execution_runtime import (
    VoiceCallExecutionRuntime,
)



def create_voice_call_lifecycle_service(
) -> VoiceCallLifecycleService:
    """
    Create configured voice lifecycle service.
    """



    room_service = VoiceRoomService(
        room_adapter=(
            VoiceRoomAdapter()
        ),
    )



    execution_runtime = (
        VoiceCallExecutionRuntime()
    )



    execution_coordinator = (
        VoiceCallExecutionCoordinator(
            room_service=room_service,
            execution_runtime=execution_runtime,
        )
    )



    connection_service = (
        create_voice_connection_service()
    )



    voice_call_repository = (
        InMemoryVoiceCallRepository()
    )



    voice_call_management_service = (
        VoiceCallManagementService(
            repository=(
                voice_call_repository
            ),
        )
    )



    voice_call_session_service = (
        VoiceCallSessionService()
    )



    voice_call_event_service = (
        build_voice_call_event_service()
    )



    return VoiceCallLifecycleService(

        execution_coordinator=(
            execution_coordinator
        ),


        connection_service=(
            connection_service
        ),


        voice_call_management_service=(
            voice_call_management_service
        ),


        voice_call_session_service=(
            voice_call_session_service
        ),


        voice_call_event_service=(
            voice_call_event_service
        ),
    )



def build_voice_call_lifecycle_service(
) -> VoiceCallLifecycleService:
    """
    Dependency builder for application adapters.

    Keeps API composition isolated from
    internal dependency creation.
    """


    return (
        create_voice_call_lifecycle_service()
    )



__all__ = [
    "create_voice_call_lifecycle_service",
    "build_voice_call_lifecycle_service",
]

"""
Voice Call Lifecycle Service.

Application facade for the complete
voice call lifecycle.

Responsibilities:
- Coordinate voice call execution.
- Create persistent voice call identity.
- Create execution sessions.
- Generate voice connections.
- Handle reconnect lifecycle.
- Record lifecycle events.

Does NOT:
- Handle HTTP.
- Manage providers directly.
- Connect LiveKit directly.
- Persist directly.
"""

from __future__ import annotations


from dataclasses import dataclass


from backend.app.application.uai.voice.voice_call_execution_coordinator import (
    VoiceCallExecutionCoordinator,
)


from backend.app.application.uai.voice.voice_connection_service import (
    VoiceConnectionService,
)


from backend.app.application.uai.voice.voice_call_management_service import (
    VoiceCallManagementService,
)


from backend.app.application.uai.voice.voice_call_session_service import (
    VoiceCallSessionService,
)


from backend.app.application.uai.voice.voice_call_event_service import (
    VoiceCallEventService,
)


from app.domain.uai.voice.voice_call_execution_context import (
    VoiceCallExecutionContext,
)


from app.domain.uai.voice.voice_call import (
    VoiceCall,
)


from app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)


from app.domain.uai.voice.voice_call_event import (
    VoiceCallEventType,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallLifecycleService:
    """
    Application lifecycle facade.
    """


    execution_coordinator: (
        VoiceCallExecutionCoordinator
    )


    connection_service: (
        VoiceConnectionService | None
    ) = None


    voice_call_management_service: (
        VoiceCallManagementService | None
    ) = None


    voice_call_session_service: (
        VoiceCallSessionService | None
    ) = None


    voice_call_event_service: (
        VoiceCallEventService | None
    ) = None



    def start_call(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> dict:
        """
        Prepare room, execute call,
        create connection and record event.

        Identity rules
        --------------
        VoiceCall.id:
            Persistent business/lifecycle identity.

        VoiceCallSession.session_id:
            Execution/session identity.

        Lifecycle events must use the persistent
        VoiceCall.id when a VoiceCall is available.
        """


        result = (
            self.execution_coordinator
            .start(
                context=context,
            )
        )


        session = result["session"]


        if self.voice_call_event_service is not None:

            #
            # Preserve the persistent VoiceCall
            # identity for lifecycle events.
            #
            # The execution session has its own
            # session_id, but that must not replace
            # the VoiceCall business identity.
            #
            voice_call_id = (
                context.voice_call.id
                if context.voice_call is not None
                else session.session_id
            )


            self.voice_call_event_service.record_event(
                voice_call_id=(
                    voice_call_id
                ),
                learner_id=(
                    session.learner_id
                ),
                character_id=(
                    session.character_id
                ),
                event_type=(
                    VoiceCallEventType.STARTED
                ),
            )


        if self.connection_service is not None:

            connection = (
                self.connection_service
                .create_connection(
                    room_name=(
                        result["room"]["room_name"]
                    ),
                )
            )


            result["connection"] = (
                connection
            )


        return result



    def create_voice_call(
        self,
        *,
        learner_id: str,
        character_id: str,
        world_id: str,
        language_code: str,
    ) -> VoiceCall:
        """
        Create persistent voice identity.
        """


        if (
            self.voice_call_management_service
            is None
        ):
            raise RuntimeError(
                "Voice call management service "
                "not configured."
            )


        return (
            self.voice_call_management_service
            .create_call(
                learner_id=learner_id,
                character_id=character_id,
                world_id=world_id,
                language_code=language_code,
            )
        )



    def create_session(
        self,
        *,
        voice_call: VoiceCall,
    ) -> VoiceCallSession:
        """
        Create execution session.
        """


        if (
            self.voice_call_session_service
            is None
        ):
            raise RuntimeError(
                "Voice call session service "
                "not configured."
            )


        return (
            self.voice_call_session_service
            .create_session(
                voice_call=voice_call,
            )
        )



    def reconnect_call(
        self,
        *,
        voice_call: VoiceCall,
    ) -> VoiceCallSession:
        """
        Reconnect existing voice call.
        """


        if (
            self.voice_call_session_service
            is None
        ):
            raise RuntimeError(
                "Voice call session service "
                "not configured."
            )


        session = (
            self.voice_call_session_service
            .reconnect_session(
                voice_call=voice_call,
            )
        )


        if self.voice_call_event_service is not None:

            self.voice_call_event_service.record_event(
                voice_call_id=(
                    voice_call.id
                ),
                learner_id=(
                    voice_call.learner_id
                ),
                character_id=(
                    voice_call.character_id
                ),
                event_type=(
                    VoiceCallEventType.RECONNECTED
                ),
            )


        return session



__all__ = [
    "VoiceCallLifecycleService",
]

"""
Voice Calls API Router.

HTTP boundary for preparing
character voice calls.

Responsibilities:
- Receive API requests.
- Validate request payload.
- Delegate to application services.
- Return HTTP responses.

Does NOT:
- Execute AI agents.
- Connect to LiveKit directly.
- Manage conversations.
- Select providers.
- Create runtime dependencies.
- Build domain execution contexts.
"""

from __future__ import annotations

from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
)

from pydantic import BaseModel

from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)

from app.application.uai.voice.voice_call_lifecycle_service import (
    VoiceCallLifecycleService,
)

from app.application.uai.voice.voice_call_context_factory import (
    VoiceCallContextFactory,
)

from app.application.composition.voice_call_lifecycle_composition import (
    build_voice_call_lifecycle_service,
)


router = APIRouter(
    prefix="/voice",
    tags=[
        "voice",
    ],
)


class VoiceCallRequestModel(
    BaseModel
):
    """
    HTTP request model.
    """

    learner_id: str
    world_id: str
    character_id: str
    language_code: str


class VoiceCallResponseModel(
    BaseModel
):
    """
    HTTP response model.
    """

    room_name: str
    character_id: str
    language_code: str
    provider: str
    state: str
    token: str
    expires_at: str


def get_voice_call_lifecycle_service(
) -> VoiceCallLifecycleService:
    """
    Dependency provider.

    Composition decisions remain
    outside the HTTP adapter.
    """

    return (
        build_voice_call_lifecycle_service()
    )


def get_voice_call_context_factory(
) -> VoiceCallContextFactory:
    """
    Dependency provider for context creation.

    Domain creation remains inside
    application layer.
    """

    return VoiceCallContextFactory()


@router.post(
    "/call",
    response_model=VoiceCallResponseModel,
)
def prepare_voice_call(
    request: VoiceCallRequestModel,
    service: VoiceCallLifecycleService = Depends(
        get_voice_call_lifecycle_service
    ),
    context_factory: VoiceCallContextFactory = Depends(
        get_voice_call_context_factory
    ),
) -> VoiceCallResponseModel:
    """
    Prepare and start a character voice call.

    A persistent VoiceCall is created first so that
    the execution context retains the business
    identity throughout the lifecycle.
    """

    try:
        call_request = VoiceCallRequest(
            learner_id=request.learner_id,
            world_id=request.world_id,
            character_id=request.character_id,
            language_code=request.language_code,
        )

        voice_call = (
            service.create_voice_call(
                learner_id=call_request.learner_id,
                character_id=call_request.character_id,
                world_id=call_request.world_id,
                language_code=call_request.language_code,
            )
        )

        context = (
            context_factory.create(
                request=call_request,
                voice_call=voice_call,
            )
        )

        result = (
            service.start_call(
                context=context,
            )
        )

        room = result["room"]
        session = result["session"]
        connection = result["connection"]

        return VoiceCallResponseModel(
            room_name=room["room_name"],
            character_id=session.character_id,
            language_code=session.language_code,
            provider=room["provider"],
            state=session.state.value,
            token=connection.token,
            expires_at=(
                connection.expires_at
                .isoformat()
            ),
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc


__all__ = [
    "router",
]
