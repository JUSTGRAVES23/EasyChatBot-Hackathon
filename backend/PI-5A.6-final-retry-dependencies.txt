from __future__ import annotations

from app.application.operations_center.platform_intelligence_escalation_delivery import (
    PlatformIntelligenceEscalationDeliveryStatus,
)
from app.application.operations_center.platform_intelligence_escalation_retry_candidates import (
    PlatformIntelligenceEscalationRetryCandidate,
    PlatformIntelligenceEscalationRetryCandidateBatch,
    PlatformIntelligenceEscalationRetryCandidateRepository,
)


class PlatformIntelligenceEscalationRetryCandidateDiscoveryService:
    """Discover durable retry work without executing it.

    This boundary deliberately does not:
    - call the retry policy;
    - mutate delivery state;
    - send email;
    - schedule work;
    - claim/lock candidates.

    Policy eligibility and execution remain separate steps.
    """

    def __init__(
        self,
        *,
        repository: PlatformIntelligenceEscalationRetryCandidateRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")

        self._repository = repository

    async def discover(
        self,
        *,
        limit: int = 100,
    ) -> PlatformIntelligenceEscalationRetryCandidateBatch:
        if limit < 1:
            raise ValueError("limit must be at least 1.")

        records = await self._repository.list_retry_candidates(
            limit=limit,
        )

        candidates: list[PlatformIntelligenceEscalationRetryCandidate] = []

        for record in records:
            if record.status not in (
                PlatformIntelligenceEscalationDeliveryStatus.FAILED,
                PlatformIntelligenceEscalationDeliveryStatus.STARTED,
            ):
                continue

            candidates.append(
                PlatformIntelligenceEscalationRetryCandidate(
                    notification_key=record.notification_key,
                    recommendation_id=record.recommendation_id,
                    risk_decision_id=record.risk_decision_id,
                    review_item_id=record.review_item_id,
                    correlation_id=record.correlation_id,
                    attempt=record.attempt,
                    status=record.status.value,
                    error_code=record.error_code,
                    updated_at=record.updated_at,
                )
            )

        candidates.sort(
            key=lambda candidate: (
                candidate.updated_at is None,
                candidate.updated_at,
                candidate.notification_key,
            )
        )

        return PlatformIntelligenceEscalationRetryCandidateBatch(
            candidates=tuple(candidates),
            limit=limit,
        )

from __future__ import annotations

from datetime import datetime, timedelta, timezone
from typing import Protocol
from uuid import uuid4

from app.application.operations_center.platform_intelligence_escalation_retry_claims import (
    PlatformIntelligenceEscalationRetryClaim,
    RetryCandidateClaimStatus,
)


class PlatformIntelligenceEscalationRetryClaimRepository(Protocol):
    async def claim_retry_candidate(
        self,
        *,
        notification_key: str,
        claim_token: str,
        claimed_at: datetime,
        lease_expires_at: datetime,
    ) -> bool: ...


class PlatformIntelligenceEscalationRetryClaimService:
    """Claims a candidate for one bounded execution attempt.

    Claiming is coordination only. It does not evaluate retry policy, send
    email, mutate the delivery lifecycle beyond the lease, or authorize the
    escalation itself.
    """

    def __init__(
        self,
        repository: PlatformIntelligenceEscalationRetryClaimRepository,
        *,
        lease_seconds: int = 300,
    ) -> None:
        if lease_seconds < 1:
            raise ValueError("lease_seconds must be at least 1.")
        self._repository = repository
        self._lease_seconds = lease_seconds

    async def claim(
        self,
        *,
        notification_key: str,
        claimed_at: datetime | None = None,
    ) -> PlatformIntelligenceEscalationRetryClaim | None:
        if not isinstance(notification_key, str) or not notification_key.strip():
            raise ValueError("notification_key must be nonblank.")

        reference_at = claimed_at or datetime.now(timezone.utc)
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("claimed_at must be timezone-aware.")

        claim_token = str(uuid4())
        lease_expires_at = reference_at + timedelta(seconds=self._lease_seconds)

        claimed = await self._repository.claim_retry_candidate(
            notification_key=notification_key,
            claim_token=claim_token,
            claimed_at=reference_at,
            lease_expires_at=lease_expires_at,
        )

        if not claimed:
            return None

        return PlatformIntelligenceEscalationRetryClaim(
            notification_key=notification_key,
            claim_token=claim_token,
            claimed_at=reference_at,
            lease_expires_at=lease_expires_at,
            status=RetryCandidateClaimStatus.CLAIMED,
        )

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from enum import StrEnum

from app.application.operations_center.platform_intelligence_escalation_delivery import (
    PlatformIntelligenceEscalationDeliveryStatus,
)
from app.application.operations_center.platform_intelligence_escalation_dispatch import (
    PlatformIntelligenceEscalationDispatchRequest,
    PlatformIntelligenceEscalationDispatchResult,
    PlatformIntelligenceEscalationDispatchService,
)
from app.application.operations_center.platform_intelligence_escalation_retry import (
    PlatformIntelligenceEscalationRetryDecision,
    PlatformIntelligenceEscalationRetryPolicyService,
)


class PlatformIntelligenceEscalationRetryExecutionStatus(StrEnum):
    NOT_ELIGIBLE = "NOT_ELIGIBLE"
    NOT_DUE = "NOT_DUE"
    RETRIED = "RETRIED"


@dataclass(frozen=True, slots=True)
class PlatformIntelligenceEscalationRetryExecutionResult:
    status: PlatformIntelligenceEscalationRetryExecutionStatus
    notification_key: str
    attempt: int
    reason: str
    dispatch_result: PlatformIntelligenceEscalationDispatchResult | None = None


class PlatformIntelligenceEscalationRetryExecutionService:
    """Execute one policy-authorized retry without owning scheduling.

    This service:
    - retrieves the authoritative durable delivery record;
    - evaluates retry policy;
    - enforces the policy-derived due time;
    - increments FAILED attempts only through the repository;
    - preserves the attempt number for STARTED recovery;
    - delegates actual delivery to the existing dispatch service.

    It does not:
    - schedule future work;
    - sleep;
    - call an email provider directly;
    - mutate delivery state directly.
    """

    def __init__(
        self,
        *,
        delivery_repository: object,
        retry_policy: PlatformIntelligenceEscalationRetryPolicyService,
        dispatch_service: PlatformIntelligenceEscalationDispatchService,
    ) -> None:
        if delivery_repository is None:
            raise ValueError(
                "delivery_repository cannot be None."
            )

        if retry_policy is None:
            raise ValueError(
                "retry_policy cannot be None."
            )

        if dispatch_service is None:
            raise ValueError(
                "dispatch_service cannot be None."
            )

        self._repository = delivery_repository
        self._retry_policy = retry_policy
        self._dispatch_service = dispatch_service

    async def retry(
        self,
        *,
        request: PlatformIntelligenceEscalationDispatchRequest,
        email_request: object,
        reference_at: datetime,
    ) -> PlatformIntelligenceEscalationRetryExecutionResult:
        record = await self._repository.get_by_notification_key(
            request.notification_key
        )

        if record is None:
            raise ValueError(
                "Retry requires an existing durable delivery record."
            )

        if record.notification_key != request.notification_key:
            raise ValueError(
                "Delivery record notification key does not match request."
            )

        policy_result = self._retry_policy.evaluate(
            record=record,
            reference_at=reference_at,
        )

        if (
            policy_result.decision
            is PlatformIntelligenceEscalationRetryDecision.NOT_ELIGIBLE
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_ELIGIBLE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=policy_result.reason,
            )

        if (
            policy_result.next_eligible_at is not None
            and reference_at < policy_result.next_eligible_at
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_DUE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=(
                    "Retry is eligible but the policy-derived "
                    "backoff window has not elapsed."
                ),
            )

        if (
            record.status
            is PlatformIntelligenceEscalationDeliveryStatus.FAILED
        ):
            await self._repository.start_retry(
                notification_key=record.notification_key,
            )

        # STARTED recovery deliberately does not call start_retry().
        # The durable attempt remains the same because this is recovery
        # of the interrupted attempt rather than a new FAILED retry.
        dispatch_result = await self._dispatch_service.dispatch(
            request=request,
            email_request=email_request,
        )

        return PlatformIntelligenceEscalationRetryExecutionResult(
            status=(
                PlatformIntelligenceEscalationRetryExecutionStatus
                .RETRIED
            ),
            notification_key=record.notification_key,
            attempt=(
                dispatch_result.attempt
                if dispatch_result.attempt is not None
                else record.attempt
            ),
            reason=policy_result.reason,
            dispatch_result=dispatch_result,
        )
