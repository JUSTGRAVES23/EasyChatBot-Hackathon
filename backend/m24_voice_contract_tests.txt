"""
Tests for Voice Call Repository contract.

Validates that the repository boundary
defines the expected persistence contract
for VoiceCall aggregates.
"""

from datetime import datetime, timedelta


from backend.app.application.uai.voice.voice_call_repository import (
    VoiceCallRepository,
)


from app.domain.uai.voice.voice_call import (
    VoiceCall,
)


from app.domain.uai.voice.voice_call_state import (
    VoiceCallState,
)



class InMemoryVoiceCallRepository(
    VoiceCallRepository,
):
    """
    Test implementation of repository contract.
    """

    def __init__(self):

        self.storage = {}



    def save(
        self,
        *,
        voice_call: VoiceCall,
    ) -> VoiceCall:

        self.storage[
            voice_call.id
        ] = voice_call

        return voice_call



    def get_by_id(
        self,
        *,
        voice_call_id: str,
    ) -> VoiceCall | None:

        return self.storage.get(
            voice_call_id
        )



def create_voice_call() -> VoiceCall:
    """
    Create valid voice call.
    """

    created_at = datetime.utcnow()


    return VoiceCall(
        id="vc-001",
        learner_id="learner-001",
        character_id="yeti",
        world_id="french_world",
        language_code="fr",
        state=VoiceCallState.ACTIVE,
        created_at=created_at,
        expires_at=(
            created_at
            + timedelta(
                minutes=30
            )
        ),
    )



def test_repository_contract_can_save_voice_call():

    repository = (
        InMemoryVoiceCallRepository()
    )


    voice_call = create_voice_call()


    saved = repository.save(
        voice_call=voice_call,
    )


    assert saved.id == (
        "vc-001"
    )


    assert saved.character_id == (
        "yeti"
    )



def test_repository_contract_can_retrieve_voice_call():

    repository = (
        InMemoryVoiceCallRepository()
    )


    voice_call = create_voice_call()


    repository.save(
        voice_call=voice_call,
    )


    result = repository.get_by_id(
        voice_call_id="vc-001",
    )


    assert result is not None


    assert result.id == (
        "vc-001"
    )


    assert result.learner_id == (
        "learner-001"
    )



def test_repository_returns_none_for_unknown_call():

    repository = (
        InMemoryVoiceCallRepository()
    )


    result = repository.get_by_id(
        voice_call_id="unknown",
    )


    assert result is None



def test_repository_implements_contract():

    repository = (
        InMemoryVoiceCallRepository()
    )


    assert isinstance(
        repository,
        VoiceCallRepository,
    )

from backend.app.application.uai.voice.voice_call_event_service import (
    VoiceCallEventService,
)

    InMemoryVoiceCallEventRepository,

    VoiceCallEventType,
)



def create_service():

    repository = (
        InMemoryVoiceCallEventRepository()
    )


    return (
        VoiceCallEventService(
            repository=repository,
        )
    )



def test_record_event():

    service = (
        create_service()
    )


    event = (
        service.record_event(
            voice_call_id="vc-001",
            learner_id="learner-001",
            character_id="yeti",
            event_type=(
                VoiceCallEventType.CREATED
            ),
        )
    )


    assert event.voice_call_id == (
        "vc-001"
    )


    assert event.learner_id == (
        "learner-001"
    )


    assert event.character_id == (
        "yeti"
    )


    assert event.event_type == (
        VoiceCallEventType.CREATED
    )



def test_record_event_persists_event():

    service = (
        create_service()
    )


    created = (
        service.record_event(
            voice_call_id="vc-001",
            learner_id="learner-001",
            character_id="yeti",
            event_type=(
                VoiceCallEventType.CREATED
            ),
        )
    )


    events = (
        service.get_events(
            voice_call_id="vc-001",
        )
    )


    assert len(events) == 1


    assert events[0].event_id == (
        created.event_id
    )



def test_multiple_events_are_retrieved():

    service = (
        create_service()
    )


    service.record_event(
        voice_call_id="vc-001",
        learner_id="learner-001",
        character_id="yeti",
        event_type=(
            VoiceCallEventType.CREATED
        ),
    )


    service.record_event(
        voice_call_id="vc-001",
        learner_id="learner-001",
        character_id="yeti",
        event_type=(
            VoiceCallEventType.ACCEPTED
        ),
    )


    events = (
        service.get_events(
            voice_call_id="vc-001",
        )
    )


    assert len(events) == 2


    assert events[0].event_type == (
        VoiceCallEventType.CREATED
    )


    assert events[1].event_type == (
        VoiceCallEventType.ACCEPTED
    )



def test_events_are_isolated_by_voice_call():

    service = (
        create_service()
    )


    service.record_event(
        voice_call_id="vc-001",
        learner_id="learner-001",
        character_id="yeti",
        event_type=(
            VoiceCallEventType.CREATED
        ),
    )


    service.record_event(
        voice_call_id="vc-002",

        event_type=(
            VoiceCallEventType.CREATED
        ),
    )


    events = (
        service.get_events(
            voice_call_id="vc-001",
        )
    )


    assert len(events) == 1


    assert events[0].voice_call_id == (
        "vc-001"
    )

from datetime import datetime


from backend.app.application.uai.voice.voice_call_management_service import (
    VoiceCallManagementService,
)


from app.infrastructure.persistence.voice_call_repository import (
    InMemoryVoiceCallRepository,
)


from app.domain.uai.voice.voice_call_state import (
    VoiceCallState,
)



def create_service() -> VoiceCallManagementService:
    """
    Create management service.
    """

    repository = (
        InMemoryVoiceCallRepository()
    )


    return VoiceCallManagementService(
        repository=repository,
    )



def test_create_voice_call():

    service = create_service()


    voice_call = service.create_call(
        learner_id="learner-001",
        character_id="yeti",
        world_id="french_world",
        language_code="fr",
    )


    assert voice_call.id.startswith(
        "vc-"
    )


    assert voice_call.learner_id == (
        "learner-001"
    )


    assert voice_call.character_id == (
        "yeti"
    )


    assert voice_call.state == (
        VoiceCallState.CREATED
    )



def test_get_existing_voice_call():

    service = create_service()


    created = service.create_call(
        learner_id="learner-001",
        character_id="yeti",
        world_id="french_world",
        language_code="fr",
    )


    result = service.get_call(
        voice_call_id=created.id,
    )


    assert result is not None


    assert result.id == (
        created.id
    )



def test_unknown_voice_call_returns_none():

    service = create_service()


    result = service.get_call(
        voice_call_id="unknown",
    )


    assert result is None



def test_active_voice_call_can_reconnect():

    service = create_service()


    voice_call = service.create_call(
        learner_id="learner-001",
        character_id="yeti",
        world_id="french_world",
        language_code="fr",
    )


    reconnectable = (
        voice_call.mark_disconnected()
    )


    service.repository.save(
        voice_call=reconnectable,
    )


    assert service.can_reconnect(
        voice_call_id=voice_call.id,
        now=datetime.utcnow(),
    )



def test_mark_disconnected_preserves_identity():

    service = create_service()


    voice_call = service.create_call(
        learner_id="learner-001",
        character_id="yeti",
        world_id="french_world",
        language_code="fr",
    )


    disconnected = service.mark_disconnected(
        voice_call_id=voice_call.id,
    )


    assert disconnected.id == (
        voice_call.id
    )


    assert disconnected.state == (
        VoiceCallState.DISCONNECTED
    )



def test_complete_call_preserves_identity():

    service = create_service()


    voice_call = service.create_call(
        learner_id="learner-001",
        character_id="yeti",
        world_id="french_world",
        language_code="fr",
    )


    completed = service.complete_call(
        voice_call_id=voice_call.id,
    )


    assert completed.id == (
        voice_call.id
    )


    assert completed.state == (
        VoiceCallState.COMPLETED
    )



def test_disconnect_unknown_call_is_rejected():

    service = create_service()


    try:

        service.mark_disconnected(

        assert False


    except ValueError:
