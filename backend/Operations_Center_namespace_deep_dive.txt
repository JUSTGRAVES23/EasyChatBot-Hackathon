"""
WS-21.18 Announcement Group Operations application API.

Read-only, fresh operational detail for the Announcement Group surface.

The API deliberately reuses the existing OperationalDashboardService
and the authoritative OperationalEventService supplied by the Operations
Center composition root.

No second incident/event store is created.

Security boundary:
    Raw operational event metadata is never exposed through this API.
    Only explicitly approved operational metadata keys are returned.
"""

from __future__ import annotations

from typing import Any

from backend.app.application.health.operational_dashboard_service import (
    OperationalDashboardService,
)
from backend.app.application.health.operational_event import (
    OperationalSurface,
)
from backend.app.application.health.operational_event_service import (
    OperationalEventService,
)
from backend.app.application.health.operational_incident import (
    OperationalIncident,
)


# Operational metadata explicitly approved for the Operations Center UI.
#
# Sensitive learner content, prompts, AI responses, credentials, tokens,
# and arbitrary provider payloads are deliberately excluded.
SAFE_EVENT_METADATA_KEYS = frozenset(
    {
        "safe_status",
    }
)


class AnnouncementGroupOperationsAPI:
    """Read-only application boundary for WS-21.18."""

    def __init__(
        self,
        *,
        dashboard_service: OperationalDashboardService,
        event_service: OperationalEventService | None = None,
    ) -> None:
        if dashboard_service is None:
            raise ValueError(
                "dashboard_service cannot be None."
            )

        self._dashboard_service = dashboard_service

        self._event_service = (
            event_service
            if event_service is not None
            else dashboard_service.event_service
        )

    async def get_details(self) -> dict[str, Any]:
        """
        Build a fresh Announcement Group snapshot.

        Every request performs fresh dependency health evaluation and reads
        the current operational incident/event state.
        """

        snapshot = (
            await self._dashboard_service.create_dashboard_snapshot()
        )

        incidents = self._event_service.list_incidents(
            surface=OperationalSurface.ANNOUNCEMENT_GROUP,
        )

        events = tuple(
            event
            for incident in incidents
            for event in incident.events
        )

        return {
            "generated_at": snapshot.generated_at.isoformat(),
            "surface": "announcement_group",
            "status": (
                snapshot.announcement_group.status.value
            ),
            "health_score": snapshot.health_score,
            "dependencies": [
                dependency.as_dict()
                for dependency
                in snapshot.announcement_group.dependencies
            ],
            "metrics": (
                snapshot.announcement_group.metrics.as_dict()
            ),
            "incidents": {
                "summary": (
                    snapshot.announcement_group
                    .incidents
                    .as_dict()
                ),
                "items": [
                    _incident_to_dict(incident)
                    for incident in incidents
                ],
            },
            "events": [
                _event_to_dict(event)
                for event in events
            ],
        }


def _incident_to_dict(
    incident: OperationalIncident,
) -> dict[str, Any]:
    """
    Return an existing incident in dashboard-safe form.

    Raw event metadata is deliberately not serialized.
    """

    return {
        "incident_id": incident.incident_id,
        "correlation_id": incident.correlation_id,
        "event_type": incident.event_type.value,
        "surface": incident.surface.value,
        "severity": incident.severity.value,
        "status": incident.status.value,
        "first_seen": incident.first_seen.isoformat(),
        "last_seen": incident.last_seen.isoformat(),
        "occurrence_count": incident.occurrence_count,
        "user_reference": incident.user_reference,
        "world_id": incident.world_id,
        "events": [
            _event_to_dict(event)
            for event in incident.events
        ],
    }


def _event_to_dict(event) -> dict[str, Any]:
    """
    Serialize an operational event through an explicit security boundary.

    Do not call event.as_dict() here because that currently includes the
    complete raw metadata dictionary.
    """

    metadata = getattr(event, "metadata", {}) or {}

    safe_metadata = {
        key: metadata[key]
        for key in SAFE_EVENT_METADATA_KEYS
        if key in metadata
    }

    return {
        "event_type": event.event_type.value,
        "surface": event.surface.value,
        "correlation_id": event.correlation_id,
        "severity": event.severity.value,
        "occurred_at": event.occurred_at.isoformat(),
        "user_reference": event.user_reference,
        "world_id": event.world_id,
        "metadata": safe_metadata,
    }


__all__ = [
    "AnnouncementGroupOperationsAPI",
]

from __future__ import annotations

from typing import Any


class EmailWorkflowStatusService:
    """Read-only operational snapshot for scheduled email workflows."""

    _WORKFLOW_METADATA: dict[str, dict[str, Any]] = {
        "cybersecurity_daily_report": {
            "name": "Cybersecurity Daily Report",
            "category": "Cybersecurity",
            "schedule": "Daily at 08:00",
        },
        "cybersecurity_weekly_report": {
            "name": "Cybersecurity Weekly Report",
            "category": "Cybersecurity",
            "schedule": "Monday at 08:00",
        },
        "privacy_governance_reminder": {
            "name": "Privacy Governance Reminder",
            "category": "Privacy",
            "schedule": "Every 60 minutes",
        },
        "academy_one_week_milestone": {
            "name": "Academy One Week Milestone",
            "category": "Academy",
            "schedule": "Daily at 08:15",
        },
    }

    def __init__(self, scheduler_runtime, provider_manager=None) -> None:
        if scheduler_runtime is None:
            raise ValueError("scheduler_runtime cannot be None.")
        self.scheduler_runtime = scheduler_runtime
        self.provider_manager = provider_manager

    def snapshot(self) -> dict[str, Any]:
        provider = self._provider_status()
        workflows = [
            self._workflow_payload(job, provider)
            for job in self._production_jobs()
        ]
        return {
            "status": "ready",
            "scheduler": self._scheduler_status(),
            "provider": provider,
            "workflows": workflows,
            "count": len(workflows),
        }

    def _production_jobs(self) -> list[Any]:
        adapter = getattr(self.scheduler_runtime, "scheduler_adapter", None)
        scheduler = getattr(adapter, "scheduler", None)
        if scheduler is not None and hasattr(scheduler, "get_jobs"):
            return list(scheduler.get_jobs())
        return []

    def _scheduler_status(self) -> dict[str, Any]:
        adapter = getattr(self.scheduler_runtime, "scheduler_adapter", None)
        scheduler = getattr(adapter, "scheduler", None)
        return {
            "running": bool(getattr(scheduler, "running", False)),
            "adapter_enabled": adapter is not None,
        }

    def _provider_status(self) -> dict[str, Any]:
        if self.provider_manager is None:
            return {
                "provider": None,
                "configured": False,
                "available": False,
                "status": "not_configured",
            }
        return dict(self.provider_manager.status())

    def _workflow_payload(self, job: Any, provider: dict[str, Any]) -> dict[str, Any]:
        job_id = str(getattr(job, "id", "unknown"))
        metadata = self._WORKFLOW_METADATA.get(
            job_id,
            {
                "name": job_id,
                "category": "Other",
                "schedule": "Configured scheduler job",
            },
        )
        next_run = getattr(job, "next_run_time", None)
        return {
            "id": job_id,
            "name": metadata["name"],
            "category": metadata["category"],
            "schedule": metadata["schedule"],
            "registered": True,
            "enabled": True,
            "operational_status": (
                "ready"
                if provider.get("available", False)
                else "provider_not_configured"
            ),
            "next_run": next_run.isoformat() if next_run is not None else None,
        }

"""
WS-21.17 Private Classroom Operations application API.

Read-only, fresh operational detail for the Private Classroom surface.

The API reuses the existing Operations Center dashboard service and
operational incident/event abstractions.

No second monitoring model is created.
No infrastructure is accessed directly.
No operational side effects are performed.
"""

from __future__ import annotations

from typing import Any

from backend.app.application.health.operational_dashboard_service import (
    OperationalDashboardService,
)
from backend.app.application.health.operational_incident import (
    OperationalIncident,
)


_PRIVATE_CLASSROOM_SURFACE = "private_classroom"


class PrivateClassroomOperationsAPI:
    """Read-only application boundary for WS-21.17."""

    def __init__(
        self,
        *,
        dashboard_service: OperationalDashboardService,
    ) -> None:
        if dashboard_service is None:
            raise ValueError(
                "dashboard_service cannot be None."
            )

        self._dashboard_service = dashboard_service

    async def get_details(self) -> dict[str, Any]:
        """
        Build a fresh Private Classroom operational snapshot.

        Dependency probes are executed by the existing dashboard service.

        Incident/event state is read from the dashboard service's
        existing application-owned event abstraction.
        """

        snapshot = (
            await self._dashboard_service.create_dashboard_snapshot()
        )

        # Retrieve the authoritative current incident collection.
        #
        # We intentionally do not pass OperationalSurface here because
        # the repository can be reached through multiple legacy package
        # namespaces in the current repository. Filtering by the stable
        # serialized surface value avoids enum identity mismatches.
        all_incidents = self._dashboard_service.list_incidents()

        incidents = tuple(
            incident
            for incident in all_incidents
            if _surface_value(incident.surface)
            == _PRIVATE_CLASSROOM_SURFACE
        )

        events = tuple(
            event
            for incident in incidents
            for event in incident.events
        )

        return {
            "generated_at": snapshot.generated_at.isoformat(),
            "surface": _PRIVATE_CLASSROOM_SURFACE,
            "status": snapshot.private_classroom.status.value,
            "health_score": snapshot.health_score,
            "dependencies": [
                dependency.as_dict()
                for dependency
                in snapshot.private_classroom.dependencies
            ],
            "metrics": (
                snapshot.private_classroom
                .metrics
                .as_dict()
            ),
            "incidents": {
                "summary": (
                    snapshot.private_classroom
                    .incidents
                    .as_dict()
                ),
                "items": [
                    _incident_to_dict(incident)
                    for incident in incidents
                ],
            },
            "events": [
                _event_to_dict(event)
                for event in events
            ],
        }


def _surface_value(surface: Any) -> str:
    """
    Return a stable surface value.

    Supports both Enum instances and string-compatible values.
    """

    value = getattr(surface, "value", surface)

    return str(value).lower()


def _incident_to_dict(
    incident: OperationalIncident,
) -> dict[str, Any]:
    """
    Return a dashboard-safe incident representation.
    """

    return {
        "incident_id": incident.incident_id,
        "correlation_id": incident.correlation_id,
        "event_type": incident.event_type.value,
        "surface": _surface_value(incident.surface),
        "severity": incident.severity.value,
        "status": incident.status.value,
        "first_seen": incident.first_seen.isoformat(),
        "last_seen": incident.last_seen.isoformat(),
        "occurrence_count": incident.occurrence_count,
        "user_reference": incident.user_reference,
        "world_id": incident.world_id,
        "events": [
            _event_to_dict(event)
            for event in incident.events
        ],
    }


def _event_to_dict(event: Any) -> dict[str, Any]:
    """
    Return a sanitized operational event.

    Only explicitly approved operational metadata is exposed.
    Learner content, prompts, responses and secrets are excluded.
    """

    payload = event.as_dict()

    metadata = payload.get("metadata", {})

    if not isinstance(metadata, dict):
        metadata = {}

    safe_metadata = {
        key: value
        for key, value in metadata.items()
        if key in {
            "safe_status",
            "status",
            "reason",
            "error_code",
            "error_type",
            "dependency",
            "provider",
            "model",
            "capability",
        }
    }

    payload["metadata"] = safe_metadata

    return payload

Creates the application-owned dependency graph for the read-only
Private Classroom operational detail API.

Architecture:

    ProductionHealthConfiguration
        -> ProductionReadinessService
        -> OperationalHealthSnapshotService
        -> OperationalEventService
        -> OperationalDashboardService
        -> PrivateClassroomOperationsAPI

This composition deliberately reuses the existing Operations Center
health and operational-event abstractions.

It does not:

- create a second incident store;
- create a second event service;
- execute agents;
- execute providers;
- modify Runtime state;
- perform remediation;
- persist operational events.

The snapshot is generated fresh for every API request.
"""

class PrivateClassroomOperationsComposition:
    """
    Application-owned WS-21.17 dependency graph.

    The composition owns one OperationalEventService instance so the
    current in-memory operational state remains consistent across
    requests within the running application process.
    """

        configuration = ProductionHealthConfiguration()

        readiness_service = ProductionReadinessService(
            checks=configuration.build_checks(),
        )

        health_snapshot_service = OperationalHealthSnapshotService(
            readiness_service=readiness_service,
        )

        event_service = OperationalEventService()

        dashboard_service = OperationalDashboardService(

            event_service=event_service,

        self._event_service = event_service

        self._dashboard_service = dashboard_service

        self._private_classroom_api = PrivateClassroomOperationsAPI(
            dashboard_service=dashboard_service,
        )

    @property
    def event_service(self) -> OperationalEventService:
        """Return the application-owned operational event service."""
        return self._event_service

    @property
    def dashboard_service(self) -> OperationalDashboardService:

        return self._dashboard_service

    @property
    def private_classroom_api(self) -> PrivateClassroomOperationsAPI:
        """Return the WS-21.17 Private Classroom application API."""
        return self._private_classroom_api


def create_private_classroom_operations_api() -> (
    PrivateClassroomOperationsAPI
):
    """
    Create the WS-21.17 Private Classroom application boundary.

    return (

        -> OperationalHealthSnapshotService
        -> OperationalEventService
        -> OperationalDashboardService
        -> PrivateClassroomOperationsAPI

This composition deliberately reuses the existing Operations Center
health and operational-event abstractions.

It does not:

- create a second incident store;
- create a second event service;
- execute agents;
- execute providers;
- modify Runtime state;
- perform remediation;
- persist operational events.

The snapshot is generated fresh for every API request.
"""

from __future__ import annotations

from app.application.health.operational_dashboard_service import (
    OperationalDashboardService,
)
from backend.app.application.health.operational_event_service import (
    OperationalEventService,
)
from backend.app.application.health.operational_health_snapshot import (
    OperationalHealthSnapshotService,
)
from backend.app.application.health.production_health_configuration import (
    ProductionHealthConfiguration,
)
from backend.app.application.health.production_readiness_service import (
    ProductionReadinessService,
)
from backend.app.application.operations_center.private_classroom_operations_api import (
    PrivateClassroomOperationsAPI,
)


class PrivateClassroomOperationsComposition:
    """
    Application-owned WS-21.17 dependency graph.

    The composition owns one OperationalEventService instance so the
    current in-memory operational state remains consistent across
    requests within the running application process.
    """

    def __init__(self) -> None:
        configuration = ProductionHealthConfiguration()

        readiness_service = ProductionReadinessService(
            checks=configuration.build_checks(),
        )

        health_snapshot_service = OperationalHealthSnapshotService(
            readiness_service=readiness_service,
        )

        event_service = OperationalEventService()

        dashboard_service = OperationalDashboardService(
            health_snapshot_service=health_snapshot_service,
            event_service=event_service,
        )

        self._event_service = event_service

        self._dashboard_service = dashboard_service

        self._private_classroom_api = PrivateClassroomOperationsAPI(
            dashboard_service=dashboard_service,
        )

    @property
    def event_service(self) -> OperationalEventService:
        """Return the application-owned operational event service."""
        return self._event_service

    @property
    def dashboard_service(self) -> OperationalDashboardService:
        """Return the shared Operations Center dashboard service."""
        return self._dashboard_service

    @property
    def private_classroom_api(self) -> PrivateClassroomOperationsAPI:
        """Return the WS-21.17 Private Classroom application API."""
        return self._private_classroom_api


def create_private_classroom_operations_api() -> (
    PrivateClassroomOperationsAPI
):
    """
    Create the WS-21.17 Private Classroom application boundary.
    """
    return (
        PrivateClassroomOperationsComposition()
        .private_classroom_api
    )


__all__ = [
    "PrivateClassroomOperationsComposition",
    "create_private_classroom_operations_api",
]
