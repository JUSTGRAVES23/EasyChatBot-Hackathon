import os

from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker


class Base(DeclarativeBase):
    """Canonical SQLAlchemy metadata registry for the platform."""

    pass


# The authentication subsystem owns the physical ``users`` table and
# persists it through its raw-SQL repository.  This metadata-only table
# exists solely so SQLAlchemy can resolve foreign keys from ORM models
# that reference users.id.  It is not an ORM-mapped User entity and does
# not create or manage the physical table.
users_table = Table(
    "users",
    Base.metadata,
    Column(
        "id",
        PGUUID(as_uuid=True),
        nullable=False,
    ),
)


def database_url() -> str:
    """Return the canonical PostgreSQL connection URL."""

    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for PostgreSQL persistence."
        )

    return value


def create_database_engine():
    """Create the existing synchronous PostgreSQL engine.

    This API is preserved for the existing Beta 1 persistence layer.
    """

    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(
    engine,
) -> sessionmaker[Session]:
    """Create the existing synchronous session factory.

    Preserved for existing Beta 1 repositories.
    """

    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )


def async_database_url() -> str:
    """Return DATABASE_URL adapted for SQLAlchemy async PostgreSQL.

    The environment remains the single source of truth. We only adapt the
    driver at the infrastructure boundary when the configured URL uses the
    standard PostgreSQL SQLAlchemy scheme.
    """

    value = database_url()

    if value.startswith("postgresql+asyncpg://"):
        return value

    if value.startswith("postgresql://"):
        return value.replace(
            "postgresql://",
            "postgresql+asyncpg://",
            1,
        )

    raise RuntimeError(
        "DATABASE_URL must use a PostgreSQL URL compatible with "
        "the synchronous or async PostgreSQL persistence boundary."
    )


def create_async_database_engine() -> AsyncEngine:
    """Create the canonical asynchronous PostgreSQL engine.

    This is used by durable WS-22 persistence while sharing the same
    DATABASE_URL and canonical Base as the existing persistence layer.
    """

    return create_async_engine(
        async_database_url(),
        pool_pre_ping=True,
    )


def create_async_session_factory(
    engine: AsyncEngine,
) -> async_sessionmaker[AsyncSession]:
    """Create the canonical asynchronous PostgreSQL session factory."""

    return async_sessionmaker(
        bind=engine,
        class_=AsyncSession,
        expire_on_commit=False,
    )

class PostgreSQLBetaConnectionTokenRepository(BetaConnectionTokenRepository):
    def __init__(self, session: Session) -> None:
        self._session = session

    def add(self, token: BetaConnectionToken) -> None:
        self._session.add(BetaConnectionTokenModel(
            id=token.id, user_id=token.user_id, token_hash=token.token_hash,
            expires_at=token.expires_at, used_at=token.used_at, created_at=token.created_at,
        ))
        self._session.flush()

    def get_by_hash(self, token_hash: str) -> BetaConnectionToken | None:
        row = self._session.scalar(select(BetaConnectionTokenModel).where(
            BetaConnectionTokenModel.token_hash == token_hash
        ))
        if row is None:
            return None
        return BetaConnectionToken(
            id=row.id, user_id=row.user_id, token_hash=row.token_hash,
            expires_at=row.expires_at, created_at=row.created_at, used_at=row.used_at,
        )

    def mark_used(self, token_id: str, used_at: datetime) -> None:
        self._session.execute(
            update(BetaConnectionTokenModel)
            .where(BetaConnectionTokenModel.id == token_id)
            .where(BetaConnectionTokenModel.used_at.is_(None))
            .values(used_at=used_at)
        )
        self._session.flush()

class PostgreSQLConversationEventRepository(
    ConversationEventRepository
):
    """
    PostgreSQL implementation of conversation event storage.

    Privacy isolation:

        telegram_user_hash
                +
          conversation_id
                +
             world_id

    Governance:

        privacy_hash_version
                +
        retention_policy_id

    Responsibility:

    - persist conversation events
    - retrieve conversation history

    Does NOT:

    - validate business rules
    - create memories
    - analyse learning behaviour
    - create RAG knowledge
    """

    def __init__(
        self,
        session_factory,
    ) -> None:
        if session_factory is None:
            raise ValueError(
                "session_factory cannot be None."
            )

        self._session_factory = session_factory

    async def save(
        self,
        event: ConversationEvent,
    ) -> ConversationEvent:
        """
        Persist an anonymised conversation event.

        Privacy and retention governance values are taken
        from the validated event metadata and persisted into
        their dedicated database columns.
        """

        metadata = event.metadata or {}

        privacy_hash_version = metadata.get(
            "privacy_hash_version",
        )

        retention_policy_id = metadata.get(
            "retention_policy_id",
        )

        if not privacy_hash_version:
            raise ValueError(
                "Conversation event is missing "
                "privacy_hash_version."
            )

        if not retention_policy_id:
            raise ValueError(
                "Conversation event is missing "
                "retention_policy_id."
            )

        with self._session_factory() as session:
            record = ConversationEventRecord(
                id=event.event_id,
                conversation_id=event.conversation_id,
                telegram_user_hash=(
                    event.telegram_user_hash
                ),
                world_id=event.world_id,
                channel=event.channel,
                role=event.role,
                content=event.content,
                metadata_json=metadata,
                privacy_hash_version=(
                    privacy_hash_version
                ),
                retention_policy_id=(
                    retention_policy_id
                ),
                created_at=event.created_at,
            )

            session.add(record)
            session.commit()

        return event

    async def get_recent(
        self,
        *,
        telegram_user_hash: str,
        conversation_id: str,
        world_id: str,
        limit: int = 20,
    ) -> list[ConversationEvent]:
        """
        Retrieve recent conversation events for one
        pseudonymised Telegram identity, conversation,
        and learning world.

        Isolation key:

            telegram_user_hash
                    +
            conversation_id
                    +
                world_id

        Results are returned in chronological order so
        callers can construct conversation context safely.
        """

        if limit <= 0:
            return []

        if not telegram_user_hash:
            raise ValueError(
                "telegram_user_hash cannot be empty."
            )

        if not conversation_id:
            raise ValueError(
                "conversation_id cannot be empty."
            )

        if not world_id:
            raise ValueError(
                "world_id cannot be empty."
            )

        with self._session_factory() as session:
            rows = session.scalars(
                select(
                    ConversationEventRecord
                )
                .where(
                    ConversationEventRecord.telegram_user_hash
                    == telegram_user_hash
                )
                .where(
                    ConversationEventRecord.conversation_id
                    == conversation_id
                )
                .where(
                    ConversationEventRecord.world_id
                    == world_id
                )
                .order_by(
                    ConversationEventRecord.created_at.desc()
                )
                .limit(limit)
            ).all()

        return [
            _event_from_record(row)
            for row in reversed(rows)
        ]


def _event_from_record(
    row: ConversationEventRecord,
) -> ConversationEvent:
    """
    Convert the persistence record back into the
    application-domain conversation event.

    Governance fields remain available through metadata
    so the domain event retains its original contract.
    """

    metadata = dict(
        row.metadata_json or {}
    )

    metadata.setdefault(
        "privacy_hash_version",
        row.privacy_hash_version,
    )

    metadata.setdefault(
        "retention_policy_id",
        row.retention_policy_id,
    )

    return ConversationEvent(
        event_id=row.id,
        conversation_id=row.conversation_id,
        telegram_user_hash=(
            row.telegram_user_hash
        ),
        world_id=row.world_id,
        channel=row.channel,
        role=row.role,
        content=row.content,
        metadata=metadata,
        created_at=row.created_at,
    )

class PostgreSQLEmailWorkflowRegistry(EmailWorkflowRegistry):
    """PostgreSQL implementation of the authoritative email workflow registry."""

    def __init__(self, session_factory) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")

        self._session_factory = session_factory

    def create(
        self,
        workflow: EmailWorkflowDefinition,
    ) -> EmailWorkflowDefinition:
        if workflow is None:
            raise ValueError("workflow is required.")

        with self._session_factory() as session:
            existing = session.get(
                EmailWorkflowRecord,
                workflow.workflow_id,
            )

            if existing is not None:
                raise ValueError(
                    f"Workflow already exists: {workflow.workflow_id}"
                )

            session.add(self._to_record(workflow))
            session.commit()

        return workflow

    def get(
        self,
        workflow_id: str,
    ) -> EmailWorkflowDefinition | None:
        with self._session_factory() as session:
            record = session.scalar(
                select(EmailWorkflowRecord).where(
                    EmailWorkflowRecord.workflow_id == workflow_id,
                )
            )

        if record is None:
            return None

        return self._to_domain(record)

    def list(self):
        with self._session_factory() as session:
            records = session.scalars(
                select(EmailWorkflowRecord).order_by(
                    EmailWorkflowRecord.workflow_id.asc(),
                )
            ).all()

        return [self._to_domain(record) for record in records]

    def update(
        self,
        workflow: EmailWorkflowDefinition,
    ) -> EmailWorkflowDefinition:
        if workflow is None:
            raise ValueError("workflow is required.")

        with self._session_factory() as session:
            record = session.get(
                EmailWorkflowRecord,
                workflow.workflow_id,
            )

            if record is None:
                raise KeyError(
                    f"Workflow not found: {workflow.workflow_id}"
                )

            record.name = workflow.name
            record.description = workflow.description
            record.workflow_type = workflow.workflow_type.value
            record.lifecycle = workflow.lifecycle.value
            record.owner = workflow.owner
            record.trigger = workflow.trigger
            record.schedule = workflow.schedule
            record.template_id = workflow.template_id
            record.version = workflow.version
            record.updated_at = workflow.updated_at

            session.commit()

        return workflow

    @staticmethod
    def _to_record(
        workflow: EmailWorkflowDefinition,
    ) -> EmailWorkflowRecord:
        return EmailWorkflowRecord(
            workflow_id=workflow.workflow_id,
            name=workflow.name,
            description=workflow.description,
            workflow_type=workflow.workflow_type.value,
            lifecycle=workflow.lifecycle.value,
            owner=workflow.owner,
            trigger=workflow.trigger,
            schedule=workflow.schedule,
            template_id=workflow.template_id,
            version=workflow.version,
            created_at=workflow.created_at,
            updated_at=workflow.updated_at,
        )

    @staticmethod
    def _to_domain(
        record: EmailWorkflowRecord,
    ) -> EmailWorkflowDefinition:
        return EmailWorkflowDefinition(
            workflow_id=record.workflow_id,
            name=record.name,
            description=record.description,
            workflow_type=EmailWorkflowType(record.workflow_type),
            lifecycle=EmailWorkflowLifecycle(record.lifecycle),
            owner=record.owner,
            trigger=record.trigger,
            schedule=record.schedule,
            template_id=record.template_id,
            version=record.version,
            created_at=record.created_at,
            updated_at=record.updated_at,
        )

class PostgreSQLEntitlementRepository:
    """Durable PostgreSQL repository for EASYWORLD entitlements."""

    def __init__(
        self,
        session_factory: sessionmaker[Session],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    def get(
        self,
        *,
        user_id: str,
        plan_code: str,
    ) -> EntitlementRecord | None:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if not plan_code:
            raise ValueError("plan_code cannot be empty.")

        with self._session_factory() as session:
            return session.scalar(
                select(EntitlementRecord).where(
                    EntitlementRecord.user_id == user_id,
                    EntitlementRecord.plan_code == plan_code,
                )
            )

    def get_for_user(
        self,
        user_id: str,
    ) -> list[EntitlementRecord]:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        with self._session_factory() as session:
            return list(
                session.scalars(
                    select(EntitlementRecord)
                    .where(EntitlementRecord.user_id == user_id)
                    .order_by(EntitlementRecord.created_at.desc())
                ).all()
            )

    def upsert(
        self,
        *,
        user_id: str,
        plan_code: str,
        ai_credits: int = 0,
        agent_access: bool = False,
        premium_models: bool = False,
        priority_processing: bool = False,
        active: bool = True,
    ) -> EntitlementRecord:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if not plan_code:
            raise ValueError("plan_code cannot be empty.")

        if ai_credits < 0:
            raise ValueError("ai_credits cannot be negative.")

        with self._session_factory() as session:
            record = session.scalar(
                select(EntitlementRecord).where(
                    EntitlementRecord.user_id == user_id,
                    EntitlementRecord.plan_code == plan_code,
                )
            )

            if record is None:
                record = EntitlementRecord(
                    user_id=user_id,
                    plan_code=plan_code,
                    ai_credits=ai_credits,
                    agent_access=agent_access,
                    premium_models=premium_models,
                    priority_processing=priority_processing,
                    active=active,
                )
                session.add(record)
            else:
                record.ai_credits = ai_credits
                record.agent_access = agent_access
                record.premium_models = premium_models
                record.priority_processing = priority_processing
                record.active = active

            session.commit()
            session.refresh(record)

            return record

class PostgreSQLPaymentEventRepository(PaymentEventRepository):
    """Durable PostgreSQL implementation of Stripe webhook event storage.

    Responsibilities:
    - Persist verified Stripe webhook events.
    - Provide durable idempotency state.
    - Track processing failures.

    Does not:
    - Verify Stripe signatures.
    - Grant entitlements.
    - Apply payment business rules.
    """

    def __init__(
        self,
        session_factory: sessionmaker[Session],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    def has_processed(self, event_id: str) -> bool:
        if not event_id:
            raise ValueError("event_id cannot be empty.")

        with self._session_factory() as session:
            record = session.get(
                StripeWebhookEventRecord,
                event_id,
            )

            return record is not None and record.processed

    def record_received(self, event: PaymentEvent) -> None:
        if not isinstance(event, PaymentEvent):
            raise TypeError("event must be a PaymentEvent.")

        with self._session_factory() as session:
            existing = session.get(
                StripeWebhookEventRecord,
                event.event_id,
            )

            if existing is None:
                session.add(
                    StripeWebhookEventRecord(
                        stripe_event_id=event.event_id,
                        event_type=event.event_type,
                        processed=False,
                        processed_at=None,
                        payload=event.payload,
                        created_at=datetime.now(timezone.utc),
                    )
                )

            session.commit()

    def mark_processed(self, event_id: str) -> None:
        if not event_id:
            raise ValueError("event_id cannot be empty.")

        with self._session_factory() as session:
            record = session.get(
                StripeWebhookEventRecord,
                event_id,
            )

            if record is None:
                raise ValueError(
                    f"Stripe webhook event not found: {event_id!r}."
                )

            record.processed = True
            record.processed_at = datetime.now(timezone.utc)

            session.commit()

    def mark_failed(
        self,
        event_id: str,
        error: str,
    ) -> None:
        """Record that processing failed without deleting the event.

        The current persistence schema has no dedicated error column.
        Therefore the durable event remains recorded and unprocessed.
        The caller can retry the event safely.
        """

        if not event_id:
            raise ValueError("event_id cannot be empty.")

        if not error:
            raise ValueError("error cannot be empty.")

        with self._session_factory() as session:
            record = session.get(
                StripeWebhookEventRecord,
                event_id,
            )

            if record is None:
                raise ValueError(
                    f"Stripe webhook event not found: {event_id!r}."
                )

            record.processed = False
            record.processed_at = None

            session.commit()

                )
            ).all()

        return [self._to_domain(record) for record in records]

    def save(self, project: ProjectDefinition) -> ProjectDefinition:
        if project is None:
            raise ValueError("project is required.")

        with self._session_factory() as session:
            record = session.get(
                ProjectRegistryRecord,
                project.project_id,
            )

            if record is None:
                record = ProjectRegistryRecord(
                    project_id=project.project_id,
                    name=project.name,
                    description=project.description,
                    owner=project.owner,
                    active=project.active,
                    created_at=project.created_at,
                )
                session.add(record)
            else:
                record.name = project.name
                record.description = project.description
                record.owner = project.owner
                record.active = project.active

            session.commit()

        self,
        projects: list[ProjectDefinition],
    ) -> list[ProjectDefinition]:
        return self._persist_many(projects, overwrite=False)

    def upsert_many(
        self,
        projects: list[ProjectDefinition],
    ) -> list[ProjectDefinition]:
        return self._persist_many(projects, overwrite=True)

    def _persist_many(
        self,
        projects: list[ProjectDefinition],
        *,
        overwrite: bool,
    ) -> list[ProjectDefinition]:
        if not projects:
            raise ValueError("projects are required.")

        with self._session_factory() as session:

                    record.owner = project.owner
                    record.active = project.active

                    )

            session.commit()

        return projects

class PostgreSQLStudyMaterialRepository(
    StudyMaterialRepository,
):
    """
    PostgreSQL implementation of the Study Material repository.

    The application-facing World identifier is the authoritative
    Learning World `identifier` (for example, `french_world`).

    The Study Material persistence table stores the corresponding
    Learning World UUID in `study_materials.world_id`.

    This adapter therefore joins against the authoritative
    Learning World catalogue when reading materials.

    Learner identity, Telegram identifiers, conversation identifiers,
    progress, AI/RAG state and content ingestion are deliberately
    outside this repository.
    """

    def __init__(
        self,
        session_factory: sessionmaker[Session],
    ) -> None:
        if session_factory is None:
            raise ValueError(
                "session_factory cannot be None."
            )

        self._session_factory = session_factory

    def list_published_by_world(
        self,
        world_id: str,
    ) -> list[StudyMaterial]:
        """
        Return published Study Materials for one Learning World.

        Results are deterministically ordered by:

        1. display_order ascending
        2. material_id ascending
        """

        normalized_world_id = self._validate_world_id(
            world_id
        )

        with self._session_factory() as session:
            rows = session.scalars(
                select(StudyMaterialRecord)
                .join(
                    LearningWorldModel,
                    StudyMaterialRecord.world_id
                    == LearningWorldModel.id,
                )
                .where(
                    LearningWorldModel.identifier
                    == normalized_world_id,
                    StudyMaterialRecord.status
                    == StudyMaterialStatus.PUBLISHED.value,
                )
                .order_by(
                    StudyMaterialRecord.display_order.asc(),
                    StudyMaterialRecord.material_id.asc(),
                )
            ).all()

            return [
                self._to_domain(
                    row,
                    world_id=normalized_world_id,
                )
                for row in rows
            ]

    def get_published(
        self,
        material_id: UUID,
        world_id: str,
    ) -> StudyMaterial | None:
        """
        Return one published Study Material belonging to
        the supplied Learning World.

        The World identifier is part of the query boundary so a
        material belonging to another World cannot be returned
        merely because its UUID is known.
        """

        if not isinstance(material_id, UUID):
            raise ValueError(
                "material_id must be a UUID."
            )

        normalized_world_id = self._validate_world_id(
            world_id
        )

        with self._session_factory() as session:
            row = session.scalar(
                select(StudyMaterialRecord)
                .join(
                    LearningWorldModel,
                    StudyMaterialRecord.world_id
                    == LearningWorldModel.id,
                )
                .where(
                    StudyMaterialRecord.material_id
                    == material_id,
                    LearningWorldModel.identifier
                    == normalized_world_id,
                    StudyMaterialRecord.status
                    == StudyMaterialStatus.PUBLISHED.value,
                )
            )

            if row is None:
                return None

            return self._to_domain(
                row,
                world_id=normalized_world_id,
            )

    @staticmethod
    def _validate_world_id(
        world_id: str,
    ) -> str:
        if not isinstance(world_id, str):
            raise TypeError(
                "world_id must be a string."
            )

        normalized = world_id.strip()

        if not normalized:
            raise ValueError(
                "world_id cannot be empty."
            )

        return normalized

    @staticmethod
    def _to_domain(
        row: StudyMaterialRecord,
        *,
        world_id: str,
    ) -> StudyMaterial:
        try:
            material_type = StudyMaterialType(
                row.material_type
            )
        except ValueError as exc:
            raise ValueError(
                "Unsupported Study Material type: "
                f"{row.material_type!r}."
            ) from exc

        try:
            status = StudyMaterialStatus(
                row.status
            )
        except ValueError as exc:
            raise ValueError(
                "Unsupported Study Material status: "
                f"{row.status!r}."
            ) from exc

        return StudyMaterial(
            material_id=row.material_id,
            world_id=world_id,
            title=row.title,
            description=row.description,
            material_type=material_type,
            status=status,
            display_order=row.display_order,
            created_at=row.created_at,
            updated_at=row.updated_at,
        )

class PrivateChatConversationRecord(Base):
    """Durable ownership boundary for an authenticated private chat."""

    __tablename__ = "private_chat_conversations"

    conversation_id: Mapped[UUID] = mapped_column(
        UUID(as_uuid=True),
        primary_key=True,
        default=uuid4,
    )

    user_id: Mapped[str] = mapped_column(
        String(255),
        nullable=False,
        index=True,
    )

    world_id: Mapped[str] = mapped_column(
        String(255),
        nullable=False,
        index=True,
    )

    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
    )

    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
    )

    messages: Mapped[list["PrivateChatMessageRecord"]] = relationship(
        back_populates="conversation",
        cascade="all, delete-orphan",
        passive_deletes=True,
        order_by="PrivateChatMessageRecord.created_at",
    )

    active_conversation: Mapped["PrivateChatActiveConversationRecord | None"] = (
        relationship(
            back_populates="conversation",
            uselist=False,
            passive_deletes=True,
        )
    )

    __table_args__ = (
        Index(
            "ix_private_chat_conversations_user_world",
            "user_id",
            "world_id",
        ),
    )

class DisabledAcademyTelegramSender:
    """Safe default when no existing Telegram Bot is supplied.

    Academy email delivery remains independently testable.
    No second Telegram Bot is created.
    """

    async def send(
        self,
        *,
        chat_id: str,
        text: str,
        portal_url: str,
    ) -> dict:

            "provider": "telegram",
            "status": "not_configured",

def _required_config(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for Academy production composition."
        )

    return value


def create_academy_one_week_workflow(
    *,
    session_factory=None,
    email_provider_manager: EmailProviderManager | None = None,
    telegram_bot=None,
    portal_url: str | None = None,
    registration_email: str | None = None,
) -> AcademyOneWeekWorkflow:
    """Compose the production Academy one-week workflow.

    Composition:

        PostgreSQL
            ?
        AcademyOneWeekWorkflow
            ?
        OneWeekNotificationOrchestrator
            ?
        EmailProviderManager
            ?
        Existing Telegram Bot adapter

    The scheduler itself is intentionally not created here.
    """

    if session_factory is None:
        engine = create_async_database_engine()
        session_factory = create_async_session_factory(engine)

    resolved_portal_url = (
        portal_url.strip()
        if portal_url is not None
        else _required_config("EASYCHATBOT_PORTAL_URL")
    )

    resolved_registration_email = (
        registration_email.strip()
        if registration_email is not None
        else _required_config("EASYCHATBOT_REGISTRATION_EMAIL")
    )

    email_sender = AcademyEmailNotificationSender(
        provider_manager=email_provider_manager
        or EmailProviderManager(),
    )

    if telegram_bot is not None:
        telegram_sender: TelegramNotificationSender = (
            ExistingBotTelegramNotificationSender(
                telegram_bot,
            )
        )
    else:
        telegram_sender = DisabledAcademyTelegramSender()

    notifications = OneWeekNotificationOrchestrator(
        email_sender=email_sender,
        telegram_sender=telegram_sender,
        email_formatter=AcademyEmailFormatter(),
        telegram_formatter=AcademyTelegramFormatter(),
        registration_email=resolved_registration_email,
        portal_url=resolved_portal_url,
    )

    users = PostgreSQLAcademyUserQuery(
        session_factory,
    )

    deliveries = PostgreSQLAcademyDeliveryRepository(
        session_factory,
    )

    return AcademyOneWeekWorkflow(
        users=users,
        deliveries=deliveries,
        notifications=notifications,
    )

    "create_academy_one_week_workflow",
]

def _env_positive_int(name: str, default: int) -> int:
    raw = os.getenv(name)

    if raw is None:
        return default

    try:
        value = int(raw)
    except ValueError as exc:
        raise ValueError(
            f"{name} must be a positive integer."
        ) from exc

    if value <= 0:
        raise ValueError(
            f"{name} must be a positive integer."
        )

    return value


def _env_positive_float(name: str, default: float) -> float:
    raw = os.getenv(name)

    if raw is None:
        return default

    try:
        value = float(raw)
    except ValueError as exc:
        raise ValueError(
            f"{name} must be a positive number."
        ) from exc

    if value <= 0:
        raise ValueError(
            f"{name} must be a positive number."
        )

    return value


def create_beta_telegram_composition(
    *,
    user_repository: Any,
    token_repository: Any,
    binding_repository: Any,
    conversation_event_repository: Any,
    privacy_identifier_service: Any,
    conversation_event_validator: Any,
    beta_world_access_service: BetaWorldAccessService,
    expiration_minutes: int = 1440,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: (
        Callable[[], ConversationRuntime] | None
    ) = None,
    rate_limiter: Any | None = None,
    abuse_protector: Any | None = None,
    observability: Any | None = None,
    provider_usage_capture: ProviderUsageCapture | None = None,
    provider_usage_repository: ProviderUsageRepository | None = None,
    functionality_control_service: (
        FunctionalityOperationalControlService | None
    ) = None,
    world_resolver: WorldResolver | None = None,
    language_response_policy: LanguageResponsePolicy | None = None,
    language_detector: LanguageDetector | None = None,
    private_chat_service_factory: Callable[[], Any] | None = None,
    world_start_router: WorldTelegramStartRouter | None = None,
) -> BetaTelegramComposition:
    """Create the Beta Telegram application composition.

    BetaAccessService owns Telegram identity binding.

    BetaWorldAccessService owns durable provisioning of the complete
    Learning World portfolio after successful Beta onboarding.

    Telegram never selects or changes the user's active World.
    """

    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    if beta_world_access_service is None:
        raise ValueError(
            "beta_world_access_service is required."
        )

    if (
        conversation_runtime is not None
        and conversation_runtime_factory is not None
    ):
        raise ValueError(
            "Provide either conversation_runtime or "
            "conversation_runtime_factory, not both."
        )

    # Application configuration root:
    #
    # backend/app
    #
    app_root = (
        Path(__file__)
        .resolve()
        .parents[2]
    )

    world_repository = YamlWorldRepository(
        worlds_path=(
            app_root
            / "configuration"
            / "worlds"
        )
    )

    if world_resolver is None:
        world_resolver = WorldResolver(
            world_repository=world_repository,
        )

    beta_access_service = BetaAccessService(
        user_repository=user_repository,
        token_repository=token_repository,
        binding_repository=binding_repository,
        world_resolver=world_resolver,
        expiration_minutes=expiration_minutes,
    )

    resolved_usage_repository = provider_usage_repository

    if conversation_runtime is not None:
        resolved_conversation_runtime = conversation_runtime

    elif conversation_runtime_factory is not None:
        resolved_conversation_runtime = conversation_runtime_factory()

        # P1.6.1: when production supplies a durable usage repository,
        # attach usage capture to the factory-created ConversationRuntime.
        # Keep the existing factory API and explicit runtime injection seam.
        if (
            provider_usage_repository is not None
            and provider_usage_capture is None
            and isinstance(
                resolved_conversation_runtime,
                ConversationRuntime,
            )
            and resolved_conversation_runtime.usage_capture is None
        ):
            provider_usage_service = ProviderUsageService(
                provider_usage_repository
            )

            provider_usage_capture = ProviderUsageCapture(
                provider_usage_service
            )

            resolved_conversation_runtime = replace(
                resolved_conversation_runtime,
                usage_capture=provider_usage_capture,
                usage_context=ProviderUsageContext(
                    capability="conversation",
                    functionality="chat",
                    agent="conversation_agent",
                ),
            )

    else:
        if provider_usage_capture is None:
            if resolved_usage_repository is None:
                resolved_usage_repository = (
                    InMemoryProviderUsageRepository()
                )

            provider_usage_service = ProviderUsageService(
                resolved_usage_repository
            )

            provider_usage_capture = ProviderUsageCapture(
                provider_usage_service
            )

        resolved_conversation_runtime = ConversationRuntime(
            llm_provider=LLMProviderFactory.create(),
            usage_capture=provider_usage_capture,
            usage_context=ProviderUsageContext(
                capability="conversation",
                functionality="chat",
                agent="conversation_agent",
            ),
        )

    resolved_language_response_policy = (
        language_response_policy
        or LanguageResponsePolicy()
    )

    if language_detector is None:
        raise ValueError(
            "language_detector is required for "
            "Beta Telegram composition."
        )

    conversation_service = TelegramConversationService(
        conversation_runtime=resolved_conversation_runtime,
        conversation_event_repository=conversation_event_repository,
        privacy_identifier_service=privacy_identifier_service,
        conversation_event_validator=conversation_event_validator,
        language_response_policy=resolved_language_response_policy,
        language_detector=language_detector,
        world_resolver=world_resolver,
    )

    resolved_rate_limiter = (
        rate_limiter
        or InMemoryTelegramRateLimiter(
            max_messages=_env_positive_int(
                "TELEGRAM_RATE_LIMIT_MAX_MESSAGES",
                100,
            ),
            window_seconds=_env_positive_float(
                "TELEGRAM_RATE_LIMIT_WINDOW_SECONDS",
                86400.0,
            ),
        )
    )

    resolved_abuse_protector = (
        abuse_protector
        or InMemoryTelegramAbuseProtector(
            max_messages=_env_positive_int(
                "TELEGRAM_ABUSE_MAX_MESSAGES",
                10,
            ),
            window_seconds=_env_positive_float(
                "TELEGRAM_ABUSE_WINDOW_SECONDS",
                60.0,
            ),
        )
    )

    resolved_observability = (
        observability
        or LoggingTelegramObservability(
            privacy_identifier_service=privacy_identifier_service,
        )
    )

    private_chat_bridge = None

    if private_chat_service_factory is not None:
        private_chat_bridge = TelegramPrivateChatBridge(
            private_chat_service_factory=private_chat_service_factory,
        )

    conversation_boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation_service,
        beta_access_service=beta_access_service,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,
        observability=resolved_observability,
        functionality_control_service=functionality_control_service,
        private_chat_bridge=private_chat_bridge,
    )

    private_chat_gate = BetaPrivateChatGate(
        beta_access_service=beta_access_service,
    )

    start_handler = BetaTelegramStartHandler(
    beta_access_service=beta_access_service,
    beta_world_access_service=beta_world_access_service,
    privacy_identifier_service=privacy_identifier_service,
)

    adapter = BetaTelegramAdapter(
        start_handler=start_handler,
        private_chat_gate=private_chat_gate,
        conversation_boundary=conversation_boundary,
        world_start_router=world_start_router,
    )

    runtime = TelegramBotRuntime(
        adapter=adapter,
        conversation_boundary=conversation_boundary,
    )

    return BetaTelegramComposition(
        beta_access_service=beta_access_service,
        beta_world_access_service=beta_world_access_service,
        conversation_runtime=resolved_conversation_runtime,
        conversation_service=conversation_service,
        conversation_boundary=conversation_boundary,
        private_chat_bridge=private_chat_bridge,
        private_chat_gate=private_chat_gate,
        start_handler=start_handler,
        adapter=adapter,
        runtime=runtime,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,

      composition.

    existing private-chat conversation handling.

    Telegram API, grant World access, or invoke the AI runtime.

            "Provide either conversation_runtime or "

            database_engine_factory
            or create_database_engine

            session_factory_factory
            or create_session_factory
        )(database_engine)

    if async_database_engine is None:

        session = session_factory()
        repository = PostgreSQLPrivateChatRepository(session)

            language_response_policy=language_response_policy,
            language_detector=language_detector,
            conversation_runtime=conversation_runtime,
            character_instruction=character_instruction,
        )
