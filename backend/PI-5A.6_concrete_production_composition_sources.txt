class PostgreSQLPlatformIntelligenceReviewSource(
    PlatformIntelligenceReviewSource,
    PlatformIntelligenceRetryStateReader,
):
    """
    Production read adapter over authoritative Platform Intelligence state.

    Architecture boundary:

        PostgreSQL authoritative PI state
                    |
                    v
        PostgreSQLPlatformIntelligenceReviewSource
                    |
              +-----+------+
              |            |
              v            v
        Review projection  Retry state
              |            |
              v            v
        Review API     Retry Orchestrator

    Responsibilities:
    - read authoritative PI state;
    - expose the governed review projection;
    - expose recommended escalations;
    - rehydrate the existing retry-state contract.

    Non-responsibilities:
    - no retry policy;
    - no dispatch;
    - no email sending;
    - no delivery persistence;
    - no scheduler;
    - no mutation of risk/review state.
    """

    def __init__(
        self,
        repository: PlatformIntelligenceAuthoritativeStateRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository must not be None.")

        self._repository = repository

    # ------------------------------------------------------------------
    # Existing ReviewSource synchronous contract
    # ------------------------------------------------------------------
    #
    # The current ReviewSource application boundary is synchronous,
    # while authoritative persistence is asynchronous.
    #
    # The async methods below are therefore the production persistence
    # adapter entry points. These synchronous methods intentionally fail
    # closed rather than performing unsafe event-loop manipulation.
    #
    # Composition must use the async methods when binding this adapter
    # to the async PostgreSQL repository.

    def list_reviews(self) -> tuple[Any, ...]:
        raise RuntimeError(
            "Use list_reviews_async() from the async application composition."
        )

    def get_review(
        self,
        review_item_id: str,
    ) -> Any | None:
        raise RuntimeError(
            "Use get_review_async() from the async application composition."
        )

    def list_escalations(self) -> tuple[Any, ...]:
        raise RuntimeError(
            "Use list_escalations_async() from the async application composition."
        )

    # ------------------------------------------------------------------
    # Async authoritative-state API
    # ------------------------------------------------------------------

    async def list_reviews_async(
        self,
    ) -> tuple[dict[str, Any], ...]:
        states = await self._repository.list_all()

        return tuple(
            self._projection(state)
            for state in states
        )

    async def get_review_async(
        self,
        review_item_id: str,
    ) -> dict[str, Any] | None:
        if not review_item_id or not review_item_id.strip():
            raise ValueError("review_item_id must not be blank.")

        state = await self._repository.get(review_item_id)

        if state is None:
            return None

        return self._projection(state)

    async def list_escalations_async(
        self,
    ) -> tuple[dict[str, Any], ...]:
        states = await self._repository.list_all()

        return tuple(
            self._projection(state)
            for state in states
            if state.recommendation_status == "RECOMMENDED"
        )

    # ------------------------------------------------------------------
    # Retry State Reader
    # ------------------------------------------------------------------

    async def get_retry_state(
        self,
        notification_key: str,
        recommendation_id: str,
    ) -> PlatformIntelligenceRetryState | None:
        if not notification_key or not notification_key.strip():
            raise ValueError("notification_key must not be blank.")

        if not recommendation_id or not recommendation_id.strip():
            raise ValueError("recommendation_id must not be blank.")

        state = await self._repository.get_by_recommendation_id(
            recommendation_id
        )

        if state is None:
            return None

        if state.recommendation_id != recommendation_id:
            return None

        return PlatformIntelligenceRetryState(
            notification_key=notification_key,
            recommendation_id=recommendation_id,
            recommendation_status=state.recommendation_status or "",
            review_status=state.review_status,
            decision_status=state.decision_status,
            quality_gate_status=state.quality_gate_status,
            dispatch_request=self._dispatch_request(
                state,
                notification_key,
            ),
            email_request=self._email_request(
                state,
                notification_key,
            ),
        )

    # ------------------------------------------------------------------
    # Projection
    # ------------------------------------------------------------------

    @staticmethod
    def _projection(
        state: Any,
    ) -> dict[str, Any]:
        """
        Return the persisted authoritative state as the governed
        Operations Center projection.

        The aggregate is immutable, so asdict() is safe here and keeps
        this adapter free from a second projection model.
        """
        return asdict(state)

    # ------------------------------------------------------------------
    # Existing canonical dispatch contract
    # ------------------------------------------------------------------

    @staticmethod
    def _dispatch_request(
        state: Any,
        notification_key: str,
    ) -> Any:
        """
        Rehydrate the existing canonical PI escalation dispatch request.

        This adapter does not implement dispatch behaviour. It only
        reconstructs the immutable request consumed by the existing
        dispatch service.
        """
        from app.application.operations_center.platform_intelligence_escalation_dispatch import (
            PlatformIntelligenceEscalationDispatchRequest,
        )

        return PlatformIntelligenceEscalationDispatchRequest(
            notification_key=notification_key,
            recommendation_id=state.recommendation_id or "",
            risk_decision_id=state.risk_decision_id,
            review_item_id=state.review_item_id,
            risk_level=state.risk_level,
            risk_score=state.risk_score,
            quality_gate_status=state.quality_gate_status,
            review_status=state.review_status,
            decision_status=state.decision_status,
            correlation_id=notification_key,
            route=state.recommendation_route or "",
            reasons=state.recommendation_reasons,
        )

    # ------------------------------------------------------------------
    # Existing canonical email contract
    # ------------------------------------------------------------------

    @staticmethod
    def _email_request(
        state: Any,
        notification_key: str,
    ) -> Any:
        """
        Rehydrate the existing canonical PI escalation email request.

        The authoritative state contains created_at, not updated_at.
        created_at is therefore the authoritative timestamp used for
        generated_at during retry-state reconstruction.
        """
        from app.application.operations_center.platform_intelligence_escalation_email import (
            PlatformIntelligenceEscalationEmailRequest,

        return PlatformIntelligenceEscalationEmailRequest(

            route=state.recommendation_route or "",
            reasons=state.recommendation_reasons,
            evidence_references=state.evidence_references,
            historical_references=state.historical_references,
            policy_id=state.policy_id,
            policy_version=state.policy_version,
            rationale=state.rationale,

            notification_key=notification_key,

class PlatformIntelligenceEscalationRetryExecutionService:
    """Execute one policy-authorized retry without owning scheduling.

    This service:
    - retrieves the authoritative durable delivery record;
    - evaluates retry policy;
    - enforces the policy-derived due time;
    - increments FAILED attempts only through the repository;
    - preserves the attempt number for STARTED recovery;
    - delegates actual delivery to the existing dispatch service.

    It does not:
    - schedule future work;
    - sleep;
    - call an email provider directly;
    - mutate delivery state directly.
    """

    def __init__(
        self,
        *,
        delivery_repository: object,
        retry_policy: PlatformIntelligenceEscalationRetryPolicyService,
        dispatch_service: PlatformIntelligenceEscalationDispatchService,
    ) -> None:
        if delivery_repository is None:
            raise ValueError(
                "delivery_repository cannot be None."
            )

        if retry_policy is None:
            raise ValueError(
                "retry_policy cannot be None."
            )

        if dispatch_service is None:
            raise ValueError(
                "dispatch_service cannot be None."
            )

        self._repository = delivery_repository
        self._retry_policy = retry_policy
        self._dispatch_service = dispatch_service

    async def retry(
        self,
        *,
        request: PlatformIntelligenceEscalationDispatchRequest,
        email_request: object,
        reference_at: datetime,
    ) -> PlatformIntelligenceEscalationRetryExecutionResult:
        record = await self._repository.get_by_notification_key(
            request.notification_key
        )

        if record is None:
            raise ValueError(
                "Retry requires an existing durable delivery record."
            )

        if record.notification_key != request.notification_key:
            raise ValueError(
                "Delivery record notification key does not match request."
            )

        policy_result = self._retry_policy.evaluate(
            record=record,
            reference_at=reference_at,
        )

        if (
            policy_result.decision
            is PlatformIntelligenceEscalationRetryDecision.NOT_ELIGIBLE
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_ELIGIBLE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=policy_result.reason,
            )

        if (
            policy_result.next_eligible_at is not None
            and reference_at < policy_result.next_eligible_at
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_DUE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=(
                    "Retry is eligible but the policy-derived "
                    "backoff window has not elapsed."
                ),
            )

        if (
            record.status
            is PlatformIntelligenceEscalationDeliveryStatus.FAILED
        ):
            await self._repository.start_retry(
                notification_key=record.notification_key,
            )

        # STARTED recovery deliberately does not call start_retry().
        # The durable attempt remains the same because this is recovery
        # of the interrupted attempt rather than a new FAILED retry.
        dispatch_result = await self._dispatch_service.dispatch(
            request=request,
            email_request=email_request,
        )

        return PlatformIntelligenceEscalationRetryExecutionResult(
            status=(
                PlatformIntelligenceEscalationRetryExecutionStatus
                .RETRIED
            ),
            notification_key=record.notification_key,
            attempt=(
                dispatch_result.attempt
                if dispatch_result.attempt is not None
                else record.attempt
            ),
            reason=policy_result.reason,
            dispatch_result=dispatch_result,
        )

    This service deliberately does not discover candidates, implement retry
    policy, send email directly, or mutate human-review/risk state.

    The canonical retry execution service remains the owner of retry-policy
    enforcement and existing dispatch integration. The orchestrator only
    supplies the authoritative dispatch context after the durable claim and
    final state revalidation.
    """

        self,
        *,
        state_reader: PlatformIntelligenceRetryStateReader,
        policy_evaluator: PlatformIntelligenceRetryPolicyEvaluator,
        claim_service: PlatformIntelligenceEscalationRetryClaimService,
        retry_execution: PlatformIntelligenceRetryExecutor,
    ) -> None:
        self._state_reader = state_reader
        self._policy_evaluator = policy_evaluator
        self._claim_service = claim_service
        self._retry_execution = retry_execution

    async def run_candidate(
        self,
        *,
        candidate: PlatformIntelligenceEscalationRetryCandidate,
        reference_at: datetime,
    ) -> PlatformIntelligenceRetryRunResult:
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("reference_at must be timezone-aware.")

        eligible = await self._policy_evaluator.evaluate(
            candidate=candidate,
            reference_at=reference_at,

            notification_key=candidate.notification_key,
            recommendation_id=candidate.recommendation_id,

            return self._skipped(candidate, "NOTIFICATION_KEY_MISMATCH")
        if state.recommendation_id != candidate.recommendation_id:

class PlatformIntelligenceRetryWorker:
    """Process one bounded candidate batch through the governed retry chain.

    This class deliberately does not:
    - schedule itself;
    - sleep;
    - create background tasks;
    - access providers;
    - implement retry policy;
    - claim candidates directly;
    - mutate risk or human-review state.

    The caller owns invocation timing and supplies the reference timestamp.
    """

    def __init__(
        self,
        *,
        discovery_service: PlatformIntelligenceEscalationRetryCandidateDiscoveryService,
        orchestrator: PlatformIntelligenceEscalationRetryOrchestrator,
    ) -> None:
        if discovery_service is None:
            raise ValueError("discovery_service cannot be None.")

        if orchestrator is None:
            raise ValueError("orchestrator cannot be None.")

        self._discovery_service = discovery_service
        self._orchestrator = orchestrator

    async def run(
        self,
        *,
        request: PlatformIntelligenceRetryWorkerRequest,
        reference_at: datetime,
    ) -> PlatformIntelligenceRetryWorkerResult:
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("reference_at must be timezone-aware.")

        batch = await self._discovery_service.discover(
            limit=request.limit,
        )

        results: list[PlatformIntelligenceRetryWorkerItemResult] = []

        for candidate in batch.candidates:
            try:
                execution = await self._orchestrator.run_candidate(
                    candidate=candidate,
                    reference_at=reference_at,
                )

            except Exception as exc:
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=candidate.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .FAILED
                        ),
                        reason="CANDIDATE_EXECUTION_ERROR",
                        attempt=candidate.attempt,
                        error_type=type(exc).__name__,
                    )
                )
                continue

            if (
                execution.status
                is PlatformIntelligenceRetryRunStatus.SUCCEEDED
            ):
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .SUCCEEDED
                        ),
                        reason=execution.reason,
                        attempt=execution.attempt,
                        provider_message_id=(
                            execution.provider_message_id
                        ),
                    )
                )

            elif (
                execution.status
                is PlatformIntelligenceRetryRunStatus.SKIPPED
            ):
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .SKIPPED
                        ),
                        reason=execution.reason,
                        attempt=execution.attempt,
                    )
                )

            else:
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .FAILED
                        ),
                        reason=execution.reason,
                        attempt=execution.attempt,
                    )
                )

        succeeded = sum(
            item.status
            is PlatformIntelligenceRetryWorkerItemStatus.SUCCEEDED
            for item in results
        )

        skipped = sum(
            item.status
            is PlatformIntelligenceRetryWorkerItemStatus.SKIPPED
            for item in results
        )

        failed = sum(
            item.status
            is PlatformIntelligenceRetryWorkerItemStatus.FAILED
            for item in results
        )

        return PlatformIntelligenceRetryWorkerResult(
            requested_limit=request.limit,
            discovered_count=len(batch.candidates),
            processed_count=len(results),
            succeeded_count=succeeded,
            skipped_count=skipped,
            failed_count=failed,
            items=tuple(results),
        )
