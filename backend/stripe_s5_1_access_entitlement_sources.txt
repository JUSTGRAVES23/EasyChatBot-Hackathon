from __future__ import annotations

from datetime import datetime
from typing import Any

from app.domain.payment.models import PaymentEvent


class SubscriptionEntitlementService:
    """Apply verified Stripe subscription events to EASYWORLD persistence.

    Stripe is the payment system of record.
    EASYWORLD subscription and entitlement persistence is the access system
    of record.

    This service deliberately contains no Stripe SDK calls.
    """

    SUBSCRIPTION_EVENTS = {
        "customer.subscription.created",
        "customer.subscription.updated",
        "customer.subscription.deleted",
    }

    def __init__(
        self,
        subscription_repository,
        entitlement_repository,
    ) -> None:
        if subscription_repository is None:
            raise ValueError("subscription_repository cannot be None.")

        if entitlement_repository is None:
            raise ValueError("entitlement_repository cannot be None.")

        self.subscription_repository = subscription_repository
        self.entitlement_repository = entitlement_repository

    def handle_payment_event(self, event: PaymentEvent) -> None:
        if not isinstance(event, PaymentEvent):
            raise TypeError("event must be a PaymentEvent.")

        if event.event_type not in self.SUBSCRIPTION_EVENTS:
            return

        subscription = self._extract_subscription(event)

        beneficiary_user_id = self._metadata_value(
            subscription,
            "easyworld_user_id",
        )

        if not beneficiary_user_id:
            raise ValueError(
                "Stripe subscription is missing easyworld_user_id metadata."
            )

        payer_user_id = self._metadata_value(
            subscription,
            "easyworld_payer_user_id",
        ) or beneficiary_user_id

        plan_code = self._metadata_value(
            subscription,
            "easyworld_plan_code",
        )

        if not plan_code:
            raise ValueError(
                "Stripe subscription is missing easyworld_plan_code metadata."
            )

        stripe_subscription_id = subscription.get("id")
        stripe_customer_id = self._customer_id(subscription.get("customer"))
        stripe_price_id = self._price_id(subscription)

        if not stripe_subscription_id:
            raise ValueError("Stripe subscription is missing id.")

        if not stripe_customer_id:
            raise ValueError("Stripe subscription is missing customer.")

        if not stripe_price_id:
            raise ValueError("Stripe subscription is missing price.")

        status = str(subscription.get("status") or "unknown")

        current_period_start = self._timestamp(
            subscription.get("current_period_start")
        )
        current_period_end = self._timestamp(
            subscription.get("current_period_end")
        )

        cancel_at_period_end = bool(
            subscription.get("cancel_at_period_end", False)
        )

        self.subscription_repository.upsert(
            beneficiary_user_id=beneficiary_user_id,
            payer_user_id=payer_user_id,
            stripe_customer_id=stripe_customer_id,
            stripe_subscription_id=stripe_subscription_id,
            stripe_price_id=stripe_price_id,
            plan_code=plan_code,
            status=status,
            current_period_start=current_period_start,
            current_period_end=current_period_end,
            cancel_at_period_end=cancel_at_period_end,
        )

        active = status in {
            "trialing",
            "active",
        }

        self.entitlement_repository.upsert(
            user_id=beneficiary_user_id,
            plan_code=plan_code,
            active=active,
        )

    @staticmethod
    def _extract_subscription(event: PaymentEvent) -> dict[str, Any]:
        data = event.payload.get("data")

        if not isinstance(data, dict):
            raise ValueError("Stripe event is missing data.")

        obj = data.get("object")

        if not isinstance(obj, dict):
            raise ValueError("Stripe event is missing subscription object.")

        return obj

    @staticmethod
    def _metadata_value(
        subscription: dict[str, Any],
        key: str,
    ) -> str | None:
        metadata = subscription.get("metadata")

        if not isinstance(metadata, dict):
            return None

        value = metadata.get(key)

        if value is None:
            return None

        value = str(value).strip()

        return value or None

    @staticmethod
    def _customer_id(customer: Any) -> str | None:
        if isinstance(customer, str):
            return customer

        if isinstance(customer, dict):
            value = customer.get("id")
            return str(value) if value else None

        return None

    @staticmethod
    def _price_id(subscription: dict[str, Any]) -> str | None:
        items = subscription.get("items")

        if not isinstance(items, dict):
            return None

        data = items.get("data")

        if not isinstance(data, list) or not data:
            return None

        first_item = data[0]

        if not isinstance(first_item, dict):
            return None

        price = first_item.get("price")

        if isinstance(price, str):
            return price

        if isinstance(price, dict):
            value = price.get("id")
            return str(value) if value else None

        return None

    @staticmethod
    def _timestamp(value: Any) -> datetime | None:
        if value is None:
            return None

        try:
            return datetime.fromtimestamp(float(value))
        except (TypeError, ValueError, OSError):
            raise ValueError(
                f"Invalid Stripe timestamp: {value!r}."
            )


__all__ = ["SubscriptionEntitlementService"]

class PaymentEventRouter(EntitlementService):
    """Route verified payment events to the appropriate application handler.

    This class does not:
    - verify Stripe signatures;
    - call the Stripe SDK;
    - persist webhook events;
    - decide whether Checkout succeeded.

    Those responsibilities remain in their existing layers.
    """

    def __init__(
        self,
        subscription_entitlement_service,
    ) -> None:
        if subscription_entitlement_service is None:
            raise ValueError(
                "subscription_entitlement_service cannot be None."
            )

        self.subscription_entitlement_service = (
            subscription_entitlement_service
        )

    def handle_payment_event(self, event: PaymentEvent) -> None:
        if not isinstance(event, PaymentEvent):
            raise TypeError("event must be a PaymentEvent.")

        self.subscription_entitlement_service.handle_payment_event(event)

"""Application service for learner-to-learning-world authorization."""

from __future__ import annotations

import asyncio
from typing import Protocol
from uuid import UUID

from app.domain.learning.learner_world_access import BETA, LearnerWorldAccess
from app.domain.learning.learner_world_access_repository import (
    LearnerWorldAccessRepository,
)


class CanonicalUserIdentityReader(Protocol):
    """Minimal provider-neutral read boundary over canonical user identity."""

    async def get_user_for_identity(self, user_id: UUID):
        """Return the canonical user record or None."""
        ...


class WorldAccessError(ValueError):
    """Expected World access validation failure."""


class UserNotFoundError(WorldAccessError):
    """The requested canonical user does not exist."""


class LearningWorldNotFoundError(WorldAccessError):
    """The requested Learning World does not exist."""


class LearningWorldDisabledError(WorldAccessError):
    """The requested Learning World is disabled for ordinary new grants."""


class WorldAccessService:
    """
    Authorize learners for one or more Learning Worlds.

    Important boundary:
    - World access is durable authorization/membership.
    - World ``enabled`` remains runtime availability.
    - Ordinary direct grants still reject disabled Worlds.
    - The explicit ``allow_disabled_world`` flag is reserved for a
      higher-level provisioning policy such as Beta portfolio provisioning.
    """

    def __init__(
        self,
        *,
        access_repository: LearnerWorldAccessRepository,
        user_identity_reader: CanonicalUserIdentityReader,
        learning_world_repository,
    ) -> None:
        if access_repository is None:
            raise ValueError("access_repository is required.")
        if user_identity_reader is None:
            raise ValueError("user_identity_reader is required.")
        if learning_world_repository is None:
            raise ValueError("learning_world_repository is required.")

        self._access_repository = access_repository
        self._user_identity_reader = user_identity_reader
        self._learning_world_repository = learning_world_repository

    async def grant_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
        access_source: str = BETA,
        allow_disabled_world: bool = False,
    ) -> LearnerWorldAccess:
        self._validate_user_id(user_id)
        identifier = self._normalize_world_identifier(world_identifier)

        user = await self._user_identity_reader.get_user_for_identity(user_id)
        if user is None:
            raise UserNotFoundError("Canonical user does not exist.")

        world = await asyncio.to_thread(
            self._learning_world_repository.get_by_identifier,
            identifier,
        )

        if world is None:
            raise LearningWorldNotFoundError(
                f"Learning World does not exist: {identifier}"
            )

        if not world.enabled and not allow_disabled_world:
            raise LearningWorldDisabledError(
                f"Learning World is disabled: {identifier}"
            )

        world_id = self._world_id(world)

        existing = await asyncio.to_thread(
            self._access_repository.get_by_user_and_world,
            user_id=user_id,
            world_id=world_id,
        )

        if existing is not None:
            if existing.is_active():
                return existing

            revived = LearnerWorldAccess.grant(
                user_id=user_id,
                world_id=world_id,
                access_source=access_source,
                access_id=existing.access_id,
            )
            return await asyncio.to_thread(
                self._access_repository.save,
                revived,
            )

        access = LearnerWorldAccess.grant(
            user_id=user_id,
            world_id=world_id,
            access_source=access_source,
        )

        return await asyncio.to_thread(
            self._access_repository.save,
            access,
        )

    async def has_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> bool:
        access = await self.get_access(
            user_id=user_id,
            world_identifier=world_identifier,
        )
        return access is not None and access.is_active()

    async def require_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> LearnerWorldAccess:
        access = await self.get_access(
            user_id=user_id,
            world_identifier=world_identifier,
        )

        if access is None or not access.is_active():
            raise PermissionError(
                "Learner does not have active access to this Learning World."
            )

        return access

    async def get_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> LearnerWorldAccess | None:
        self._validate_user_id(user_id)
        identifier = self._normalize_world_identifier(world_identifier)

        world = await asyncio.to_thread(
            self._learning_world_repository.get_by_identifier,
            identifier,
        )

        if world is None:
            return None

        world_id = self._world_id(world)

        return await asyncio.to_thread(
            self._access_repository.get_by_user_and_world,
            user_id=user_id,
            world_id=world_id,
        )

    async def list_world_access(
        self,
        *,
        user_id: UUID,
    ) -> tuple[LearnerWorldAccess, ...]:
        self._validate_user_id(user_id)

        return await asyncio.to_thread(
            self._access_repository.list_by_user,
            user_id=user_id,
        )

    @staticmethod
    def _world_id(world) -> UUID:
        world_id = getattr(world, "id", None)
        if not isinstance(world_id, UUID):
            raise WorldAccessError(
                "Learning World catalogue entry must expose the "
                "authoritative World UUID as 'id'."
            )
        return world_id

    @staticmethod
    def _validate_user_id(user_id: UUID) -> None:
        if not isinstance(user_id, UUID):
            raise ValueError("user_id must be a UUID.")

    @staticmethod
    def _normalize_world_identifier(world_identifier: str) -> str:
        if not isinstance(world_identifier, str):
            raise ValueError("world_identifier must be a string.")

        identifier = world_identifier.strip()

        if not identifier:
            raise ValueError("world_identifier is required.")

        if len(identifier) > 100:
            raise ValueError("world_identifier is too long.")

        return identifier

    "LearningWorldDisabledError",
    "LearningWorldNotFoundError",
    "UserNotFoundError",
    "WorldAccessError",
    "WorldAccessService",
]

"""Domain model for durable learner-to-learning-world access."""

from __future__ import annotations

from dataclasses import dataclass, replace
from datetime import datetime, timezone
from uuid import UUID, uuid4


ACTIVE = "active"
REVOKED = "revoked"

BETA = "beta"
SUBSCRIPTION = "subscription"
ADMIN = "admin"
TRIAL = "trial"

_VALID_STATUSES = frozenset({ACTIVE, REVOKED})
_VALID_SOURCES = frozenset({BETA, SUBSCRIPTION, ADMIN, TRIAL})


@dataclass(frozen=True, slots=True)
class LearnerWorldAccess:
    """Durable authorization record for one learner and one World."""

    access_id: UUID
    user_id: UUID
    world_id: UUID
    status: str
    access_source: str
    created_at: datetime
    updated_at: datetime

    @classmethod
    def grant(
        cls,
        *,
        user_id: UUID,
        world_id: UUID,
        access_source: str,
        access_id: UUID | None = None,
        now: datetime | None = None,
    ) -> "LearnerWorldAccess":
        if not isinstance(user_id, UUID):
            raise ValueError("user_id must be a UUID.")
        if not isinstance(world_id, UUID):
            raise ValueError("world_id must be a UUID.")
        if access_source not in _VALID_SOURCES:
            raise ValueError(f"Unsupported access source: {access_source!r}.")

        timestamp = now or datetime.now(timezone.utc)

        return cls(
            access_id=access_id or uuid4(),
            user_id=user_id,
            world_id=world_id,
            status=ACTIVE,
            access_source=access_source,
            created_at=timestamp,
            updated_at=timestamp,
        )

    def is_active(self) -> bool:
        return self.status == ACTIVE

    def revoke(self, *, now: datetime | None = None) -> "LearnerWorldAccess":
        if self.status == REVOKED:
            return self

        if self.status not in _VALID_STATUSES:
            raise ValueError(f"Unsupported access status: {self.status!r}.")

        timestamp = now or datetime.now(timezone.utc)
        return replace(self, status=REVOKED, updated_at=timestamp)

        self,

    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    def get(
        self,
        *,
        user_id: str,
        plan_code: str,
    ) -> EntitlementRecord | None:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if not plan_code:
            raise ValueError("plan_code cannot be empty.")

        with self._session_factory() as session:
            return session.scalar(
                select(EntitlementRecord).where(
                    EntitlementRecord.user_id == user_id,
                    EntitlementRecord.plan_code == plan_code,
                )
            )

    def get_for_user(
        self,
        user_id: str,
    ) -> list[EntitlementRecord]:
        if not user_id:
            raise ValueError("user_id cannot be empty.")

        with self._session_factory() as session:
            return list(
                session.scalars(
                    select(EntitlementRecord)
                    .where(EntitlementRecord.user_id == user_id)
                    .order_by(EntitlementRecord.created_at.desc())
                ).all()
            )

    def upsert(
        self,
        *,
        user_id: str,
        plan_code: str,
        ai_credits: int = 0,
        agent_access: bool = False,
        premium_models: bool = False,
        priority_processing: bool = False,

    ) -> EntitlementRecord:

        if not plan_code:
            raise ValueError("plan_code cannot be empty.")

        if ai_credits < 0:
            raise ValueError("ai_credits cannot be negative.")

        with self._session_factory() as session:
            record = session.scalar(
                select(EntitlementRecord).where(
                    EntitlementRecord.user_id == user_id,

            else:
                record.ai_credits = ai_credits
                record.agent_access = agent_access
                record.premium_models = premium_models
                record.priority_processing = priority_processing
                record.active = active

            session.refresh(record)

        if session_factory is None:
            raise ValueError("session_factory is required.")
        self._session_factory = session_factory

    def save(self, access: LearnerWorldAccess) -> LearnerWorldAccess:
        with self._session_factory() as session:
            record = session.get(LearnerWorldAccessRecord, access.access_id)

            if record is None:
                record = LearnerWorldAccessRecord(
                    id=access.access_id,
                    user_id=access.user_id,
                    world_id=access.world_id,
                    status=access.status,
                    access_source=access.access_source,
                    created_at=access.created_at,
                    updated_at=access.updated_at,
                )
                session.add(record)
            else:
                record.status = access.status
                record.access_source = access.access_source
                record.updated_at = access.updated_at

            session.commit()
            session.refresh(record)

            return self._to_domain(record)

    def get_by_id(self, access_id):
        with self._session_factory() as session:
            record = session.get(LearnerWorldAccessRecord, access_id)
            return self._to_domain(record) if record is not None else None

            record = session.scalar(
                select(LearnerWorldAccessRecord).where(
                    LearnerWorldAccessRecord.user_id == user_id,
                    LearnerWorldAccessRecord.world_id == world_id,
                )
            )
            return self._to_domain(record) if record is not None else None

                select(LearnerWorldAccessRecord)
                .where(LearnerWorldAccessRecord.user_id == user_id)
                .order_by(LearnerWorldAccessRecord.created_at)

            return tuple(self._to_domain(record) for record in records)

def create_production_beta_telegram_composition(
    *,
    expiration_minutes: int = 1440,
    database_engine: Any | None = None,
    session_factory: Any | None = None,
    database_engine_factory: Callable[[], Any] | None = None,
    session_factory_factory: Callable[[Any], Any] | None = None,
    async_database_engine: Any | None = None,
    async_session_factory: Any | None = None,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: (
        Callable[[], ConversationRuntime] | None
    ) = None,
    functionality_control_service: (
        FunctionalityOperationalControlService | None
    ) = None,
    language_detector: LanguageDetector | None = None,
) -> BetaTelegramProductionComposition:
    """
    Create the production Beta Telegram composition.

    W18.3 production provisioning is wired here.

    W19.3 controlled World Telegram community access and W19.4
    community conversation boundary are composed here, but remain
    outside the private Telegram transport/runtime boundary.

    Responsibilities:
    - Build the existing synchronous Beta Telegram persistence boundary.
    - Build the canonical asynchronous authentication boundary.
    - Reuse the existing learner/world access repository.
    - Build WorldAccessService from the canonical repositories.
    - Build BetaWorldAccessService from the canonical Beta user,
      World catalogue and World access services.
    - Build the W19.3 World Telegram destination and membership
      persistence boundaries.
    - Build the W19.3 controlled community access application service.
    - Build the W19.4 community conversation boundary.
    - Inject BetaWorldAccessService into the canonical Telegram
      composition.

    Telegram remains responsible only for onboarding/binding and
    existing private-chat conversation handling.

    W19.3 does not register Telegram group handlers, call the
    Telegram API, grant World access, or invoke the AI runtime.

    W19.4 only authorizes registered community conversation context;
    it does not register Telegram handlers, call the Telegram API,
    create membership, grant World access, or invoke the AI runtime.
    """

    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    if functionality_control_service is None:
        raise ValueError(
            "functionality_control_service is required "
            "for production Telegram composition."
        )

    if language_detector is None:
        raise ValueError(
            "language_detector is required "
            "for production Telegram composition."
        )

    if (
        database_engine is not None
        and database_engine_factory is not None
    ):
        raise ValueError(
            "Provide either database_engine or "
            "database_engine_factory, not both."
        )

    if (
        session_factory is not None
        and session_factory_factory is not None
    ):
        raise ValueError(
            "Provide either session_factory or "
            "session_factory_factory, not both."
        )

    if (
        conversation_runtime is not None
        and conversation_runtime_factory is not None
    ):
        raise ValueError(
            "Provide either conversation_runtime or "
            "conversation_runtime_factory, not both."
        )

    if database_engine is None:
        database_engine = (
            database_engine_factory
            or create_database_engine
        )()

    if session_factory is None:
        session_factory = (
            session_factory_factory
            or create_session_factory
        )(database_engine)

    if async_database_engine is None:
        async_database_engine = create_async_database_engine()

    if async_session_factory is None:
        async_session_factory = create_async_session_factory(
            async_database_engine
        )

    # ------------------------------------------------------------------
    # Existing Beta Telegram persistence
    # ------------------------------------------------------------------

    beta_user_repository = PostgreSQLBetaUserRepository(
        session_factory,
    )

    beta_token_repository = PostgreSQLBetaTokenRepository(
        session_factory,
    )

    telegram_binding_repository = (
        PostgreSQLTelegramBindingRepository(
            session_factory,
        )
    )

    conversation_event_repository = (
        PostgreSQLConversationEventRepository(
            session_factory,
        )
    )

    provider_usage_repository = (
        PostgreSQLProviderUsageRepository(
            session_factory,
        )
    )

    # ------------------------------------------------------------------
    # Canonical World infrastructure
    # ------------------------------------------------------------------

    app_root = Path(__file__).resolve().parents[2]

    world_repository = YamlWorldRepository(
        worlds_path=(
            app_root
            / "configuration"
            / "worlds"
        )
    )

    world_resolver = WorldResolver(
        world_repository=world_repository,
    )

    catalogue_repository = (
        PostgreSQLLearningWorldCatalogueRepository(
            session_factory,
        )
    )

    catalogue_service = LearningWorldCatalogueService(
        repository=catalogue_repository,
        world_resolver=world_resolver,
    )

    # ------------------------------------------------------------------
    # W18.3 World authorization/provisioning
    # ------------------------------------------------------------------

    authentication_repository = (
        PostgreSQLAuthenticationRepository(
            async_session_factory,
        )
    )

    learner_world_access_repository = (
        PostgreSQLLearnerWorldAccessRepository(
            session_factory,
        )
    )

    world_access_service = WorldAccessService(
        access_repository=learner_world_access_repository,
        user_identity_reader=authentication_repository,
        learning_world_repository=catalogue_repository,
    )

    # ------------------------------------------------------------------
    # W19.3 Controlled World Telegram community access
    # ------------------------------------------------------------------

    world_telegram_destination_repository = (
        PostgreSQLWorldTelegramDestinationRepository(
            session_factory,
        )
    )

    world_telegram_membership_repository = (
        PostgreSQLWorldTelegramMembershipRepository(
            session_factory,
        )
    )

    world_telegram_membership_service = (
        WorldTelegramMembershipService(
