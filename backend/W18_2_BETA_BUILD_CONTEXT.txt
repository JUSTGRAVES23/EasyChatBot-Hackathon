from __future__ import annotations

from enum import StrEnum


class Permission(StrEnum):
    ANNOUNCEMENT_VIEW = "announcement.view"
    ANNOUNCEMENT_PUBLISH = "announcement.publish"
    FUNCTIONALITY_CONTROL = "functionality.control"
    INTERNAL_DATA_READ = "internal_data.read"
    DEVELOPER_ADMIN_MANAGE = "developer_admin.manage"


class Role(StrEnum):
    BETA_USER = "beta_user"
    OPERATOR = "operator"
    ADMIN = "admin"
    DEVELOPER_ADMIN = "developer_admin"


ROLE_PERMISSIONS: dict[Role, frozenset[Permission]] = {
    Role.BETA_USER: frozenset(),

    Role.OPERATOR: frozenset(
        {
            Permission.ANNOUNCEMENT_VIEW,
            Permission.ANNOUNCEMENT_PUBLISH,
            Permission.FUNCTIONALITY_CONTROL,
            Permission.INTERNAL_DATA_READ,
        }
    ),

    Role.ADMIN: frozenset(
        {
            Permission.ANNOUNCEMENT_VIEW,
            Permission.ANNOUNCEMENT_PUBLISH,
            Permission.FUNCTIONALITY_CONTROL,
            Permission.INTERNAL_DATA_READ,
        }
    ),

    Role.DEVELOPER_ADMIN: frozenset(
        {
            Permission.DEVELOPER_ADMIN_MANAGE,
            Permission.INTERNAL_DATA_READ,
            Permission.FUNCTIONALITY_CONTROL,
        }
    ),
}


def has_permission(
    *,
    roles: frozenset[str],
    permission: Permission,
) -> bool:
    valid_roles = {role.value for role in Role}

    return any(
        permission in ROLE_PERMISSIONS.get(
            Role(raw_role),
            frozenset(),
        )
        for raw_role in roles
        if raw_role in valid_roles
    )

from __future__ import annotations

import hashlib
import logging
import secrets

from dataclasses import dataclass, replace
from datetime import datetime, timedelta, timezone
from typing import Iterator

from app.application.uai.world.world_resolver import WorldResolver

from .models import (
    BetaConnectionToken,
    BetaUserState,
    TelegramBinding,
)

from .repositories import (
    BetaTokenRepository,
    BetaUserRepository,
    TelegramBindingRepository,
)


logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class BetaConnectionTokenResult:
    token: str
    connection: BetaConnectionToken

    def __iter__(self) -> Iterator[object]:
        yield self.token
        yield self.connection


@dataclass(frozen=True)
class BetaConversationContext:
    user_id: str
    telegram_user_id: int
    email: str
    world_id: str
    language_code: str | None
    character: str
    level: str


class BetaAccessService:
    """
    Application boundary for Beta eligibility,
    Telegram binding and authorization.

    Telegram identity invariant:

        One Telegram identity
        -> one Beta account

    A Telegram identity that is already connected cannot
    be rebound to another Beta account through a new
    connection token.

    World assignment invariant:

        A world must be resolved through the authoritative
        WorldResolver before its ID can be persisted as the
        user's active world.

    World conversation invariant:

        A conversation context must be derived from the
        user's persisted active World through WorldResolver.

        Missing or invalid World configuration fails closed.
        The service never silently substitutes a default World.

    World compatibility:

        Legacy Beta/Telegram world identifier:
            world-fr

        Canonical World identifier:
            french_world

    Legacy identifiers are accepted only at this application
    compatibility boundary. New persisted assignments always
    use the canonical identifier returned by WorldResolver.
    """

    LEGACY_WORLD_ALIASES = {
        "world-fr": "french_world",
    }

    def __init__(
        self,
        *,
        user_repository: BetaUserRepository,
        token_repository: BetaTokenRepository,
        binding_repository: TelegramBindingRepository,
        world_resolver: WorldResolver,
        expiration_minutes: int = 1440,
    ) -> None:
        if expiration_minutes <= 0:
            raise ValueError(
                "expiration_minutes must be positive."
            )

        self._users = user_repository
        self._tokens = token_repository
        self._bindings = binding_repository
        self._world_resolver = world_resolver
        self._expiration_minutes = expiration_minutes

    async def approve_beta(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before Beta approval."
            )

        return await self._users.save(
            replace(
                user,
                beta_approved=True,
            )
        )

    async def assign_world(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> BetaUserState:
        if not world_id:
            raise ValueError(
                "world_id is required."
            )

        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before World assignment."
            )

        if not user.beta_approved:
            raise ValueError(
                "User is not Beta approved."
            )

        # Resolve legacy application identifiers to the canonical
        # World identifier before crossing the authoritative
        # WorldResolver boundary.
        canonical_world_id = self.LEGACY_WORLD_ALIASES.get(
            world_id,
            world_id,
        )

        # Resolve the requested World before persistence.
        #
        # This is the authoritative World catalogue boundary.
        # If the World does not exist or is invalid, WorldResolver
        # raises and active_world_id is not modified.
        world = self._world_resolver.resolve(
            canonical_world_id,
        )

        return await self._users.save(
            replace(
                user,
                active_world_id=world.world_id,
            )
        )

    async def create_connection_token(
        self,
        user_id: str,
    ) -> BetaConnectionTokenResult:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email is not verified."
            )

        if not user.beta_approved:
            raise ValueError(
                "User is not Beta approved."
            )

        if not user.active_world_id:
            raise ValueError(
                "User must have a Learning World before Telegram access."
            )

        raw_token = secrets.token_urlsafe(32)

        token = BetaConnectionToken(
            token_id=secrets.token_hex(16),
            user_id=user.user_id,
            token_hash=self.hash_token(
                raw_token
            ),
            expires_at=(
                datetime.now(timezone.utc)
                + timedelta(
                    minutes=self._expiration_minutes
                )
            ),
        )

        await self._tokens.save(token)

        return BetaConnectionTokenResult(
            token=raw_token,
            connection=token,
        )

    async def validate_and_bind(
        self,
        *,
        token: str,
        telegram_id: int,
        chat_id: int,
    ) -> BetaUserState:
        record = await self._tokens.get_by_hash(
            self.hash_token(token)
        )

        if record is None:
            raise ValueError(
                "Invalid connection token."
            )

        if record.used:
            raise ValueError(
                "Connection token already used."
            )

        expires_at = record.expires_at

        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(
                tzinfo=timezone.utc
            )

        if expires_at <= datetime.now(timezone.utc):
            raise ValueError(
                "Connection token expired."
            )

        user = await self._require_user(
            record.user_id
        )

        if not user.eligible:
            raise ValueError(
                "User is not eligible."
            )

        # -------------------------------------------------
        # Telegram identity ownership invariant
        #
        # A Telegram identity may belong to only one
        # Beta account. Never silently transfer ownership.
        # -------------------------------------------------

        existing_binding = (
            await self._bindings.get_by_telegram_user_id(
                str(telegram_id)
            )
        )

        if existing_binding is not None:
            raise ValueError(
                "Telegram identity is already connected."
            )

        await self._bindings.save(
            TelegramBinding(
                binding_id=secrets.token_hex(16),
                user_id=user.user_id,
                telegram_user_id=str(
                    telegram_id
                ),
                chat_id=str(
                    chat_id
                ),
                connected_at=datetime.now(
                    timezone.utc
                ),
            )
        )

        await self._tokens.mark_used(
            replace(
                record,
                used_at=datetime.now(
                    timezone.utc
                ),
            )
        )

        return await self._users.save(
            replace(
                user,
                telegram_id=telegram_id,
            )
        )

    async def authorize_private_chat(
        self,
        telegram_id: int,
    ) -> BetaUserState:
        user = await self._users.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            raise PermissionError(
                "Telegram account is not connected."
            )

        if not user.active:
            raise PermissionError(
                "User is not active."
            )

        return user

    async def get_conversation_context(
        self,
        telegram_user_id: int,
    ) -> BetaConversationContext:
        """
        Resolve the authoritative conversation context for
        an authorized Telegram learner.

        The persisted active_world_id is treated only as the
        World identifier. The authoritative World configuration,
        including its language code, is always obtained through
        WorldResolver.

        This method deliberately fails closed when no active World
        is assigned or when the assigned World cannot be resolved.
        """

        user = await self.authorize_private_chat(
            telegram_user_id
        )

        if not user.active_world_id:
            raise ValueError(
                "Active learning World is required."
            )

        canonical_world_id = self._canonical_world_id(
            user.active_world_id
        )

        # WorldResolver is the authoritative runtime World boundary.
        # No language or World defaults are inferred here.
        world = self._world_resolver.resolve(
            canonical_world_id
        )

        return BetaConversationContext(
            user_id=user.user_id,
            telegram_user_id=telegram_user_id,
            email=user.email,
            world_id=world.world_id,
            language_code=world.language_code,
            character="Alix",
            level="B1",
        )

    @classmethod
    def _canonical_world_id(
        cls,
        world_id: str | None,
    ) -> str:
        """
        Convert an accepted legacy application identifier to the
        canonical World identifier.

        A missing identifier is intentionally not replaced with a
        default World. Callers must fail closed instead.
        """

        if not world_id:
            raise ValueError(
                "Active learning World is required."
            )

        return cls.LEGACY_WORLD_ALIASES.get(
            world_id,
            world_id,
        )

    @staticmethod
    def hash_token(
        token: str,
    ) -> str:
        return hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

    async def _require_user(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._users.get(
            user_id
        )

        if user is None:
            raise ValueError(
                "Academy user does not exist."
            )

        return user

from __future__ import annotations

from typing import Any

from app.application.internal_data.beta_user_read_repository import (
    BetaUserReadRepository,
)


class BetaUserReadService:
    """Application service exposing the approved Beta-user read boundary.

    Security properties:
    - read-only;
    - no SQL;
    - no SQLAlchemy dependency;
    - no ORM objects returned;
    - no repository write methods exposed.
    """

    MAX_USER_ID_LENGTH = 128
    MAX_OUTPUT_STRING_LENGTH = 512

    def __init__(
        self,
        repository: BetaUserReadRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")

        self._repository = repository

    async def get_beta_user(
        self,
        user_id: str,
    ) -> dict[str, Any] | None:
        self._validate_user_id(user_id)

        user = await self._repository.get(user_id)

        if user is None:
            return None

        return self._to_safe_dto(user)

    async def get_beta_user_by_telegram_id(
        self,
        telegram_id: int,
    ) -> dict[str, Any] | None:
        self._validate_telegram_id(telegram_id)

        user = await self._repository.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            return None

        return self._to_safe_dto(user)

    @classmethod
    def _validate_user_id(
        cls,
        user_id: str,
    ) -> None:
        if not isinstance(user_id, str):
            raise ValueError("user_id must be a string.")

        user_id = user_id.strip()

        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if len(user_id) > cls.MAX_USER_ID_LENGTH:
            raise ValueError(
                "user_id exceeds the maximum allowed length."
            )

    @staticmethod
    def _validate_telegram_id(
        telegram_id: int,
    ) -> None:
        if isinstance(telegram_id, bool):
            raise ValueError(
                "telegram_id must be an integer."
            )

        if not isinstance(telegram_id, int):
            raise ValueError(
                "telegram_id must be an integer."
            )

        if telegram_id <= 0:
            raise ValueError(
                "telegram_id must be greater than zero."
            )

    @classmethod
    def _to_safe_dto(
        cls,
        user: Any,
    ) -> dict[str, Any]:
        """Convert the domain object to a controlled DTO.

        Do not return the domain/ORM object itself.
        """

        return {
            "user_id": cls._safe_string(
                getattr(user, "user_id", None)
            ),
            "email": cls._safe_string(
                getattr(user, "email", None)
            ),
            "email_verified": bool(
                getattr(user, "email_verified", False)
            ),
            "beta_approved": bool(
                getattr(user, "beta_approved", False)
            ),
            "telegram_id": getattr(
                user,
                "telegram_id",
                None,
            ),
            "active_world_id": cls._safe_string(
                getattr(user, "active_world_id", None)
            ),
        }

    @classmethod
    def _safe_string(
        cls,
        value: Any,
    ) -> str | None:
        if value is None:
            return None

        value = str(value)

        if len(value) > cls.MAX_OUTPUT_STRING_LENGTH:
            return value[
                : cls.MAX_OUTPUT_STRING_LENGTH
            ]

        return value

"""Beta policy for provisioning access to the complete World catalogue."""

from __future__ import annotations

from typing import Any, Protocol
from uuid import UUID

from app.domain.learning.learner_world_access import BETA, LearnerWorldAccess


class BetaUserReadBoundary(Protocol):
    async def get_beta_user(
        self,
        user_id: str,
    ) -> dict[str, Any] | None:
        """Return the controlled Beta-user DTO or None."""
        ...


class CanonicalUserIdentityReader(Protocol):
    async def get_user_for_identity(
        self,
        user_id: UUID,
    ):
        """Return the canonical user record or None."""
        ...


class WorldCatalogueBoundary(Protocol):
    def list_all(self):
        """Return all configured catalogue Worlds."""
        ...


class WorldAccessBoundary(Protocol):
    async def grant_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
        access_source: str = BETA,
        allow_disabled_world: bool = False,
    ) -> LearnerWorldAccess:
        ...


class BetaWorldAccessError(ValueError):
    """Expected Beta World-access provisioning failure."""


class BetaUserNotFoundError(BetaWorldAccessError):
    """The requested Beta user does not exist."""


class BetaUserNotEligibleError(BetaWorldAccessError):
    """The Beta user is not verified and approved."""


class CanonicalUserNotFoundError(BetaWorldAccessError):
    """The Beta user cannot be resolved through canonical identity."""


class BetaWorldAccessService:
    """
    Provision durable Beta access for every World in the authoritative catalogue.

    Policy:
        verified + Beta-approved learner
            -> all catalogue Worlds

    ``enabled`` is intentionally NOT filtered here. The service uses the
    explicit WorldAccessService provisioning flag so disabled Worlds can
    receive durable portfolio access. Runtime/Study Mode entry must still
    enforce current World availability.
    """

    def __init__(
        self,
        *,
        beta_user_reader: BetaUserReadBoundary,
        user_identity_reader: CanonicalUserIdentityReader,
        world_catalogue: WorldCatalogueBoundary,
        world_access: WorldAccessBoundary,
    ) -> None:
        if beta_user_reader is None:
            raise ValueError("beta_user_reader is required.")
        if user_identity_reader is None:
            raise ValueError("user_identity_reader is required.")
        if world_catalogue is None:
            raise ValueError("world_catalogue is required.")
        if world_access is None:
            raise ValueError("world_access is required.")

        self._beta_user_reader = beta_user_reader
        self._user_identity_reader = user_identity_reader
        self._world_catalogue = world_catalogue
        self._world_access = world_access

    async def provision_all_worlds(
        self,
        *,
        beta_user_id: str,
    ) -> tuple[LearnerWorldAccess, ...]:
        if not isinstance(beta_user_id, str) or not beta_user_id.strip():
            raise ValueError("beta_user_id is required.")

        beta_user = await self._beta_user_reader.get_beta_user(
            beta_user_id.strip()
        )

        if beta_user is None:
            raise BetaUserNotFoundError(
                "Beta user does not exist."
            )

        if not bool(beta_user.get("email_verified")):
            raise BetaUserNotEligibleError(
                "Email must be verified before World access provisioning."
            )

        if not bool(beta_user.get("beta_approved")):
            raise BetaUserNotEligibleError(
                "User is not Beta approved."
            )

        canonical_user_id = self._canonical_uuid(
            beta_user.get("user_id")
        )

        canonical_user = await self._user_identity_reader.get_user_for_identity(
            canonical_user_id
        )

        if canonical_user is None:
            raise CanonicalUserNotFoundError(
                "Canonical user does not exist."
            )

        worlds = tuple(self._world_catalogue.list_all())

        # The authoritative catalogue is the only source of World scope.
        # There is deliberately no language/World allow-list here.
        granted: list[LearnerWorldAccess] = []

        for world in worlds:
            identifier = getattr(world, "identifier", None)

            if not isinstance(identifier, str) or not identifier.strip():
                raise BetaWorldAccessError(
                    "Authoritative World catalogue contains an invalid identifier."
                )

            access = await self._world_access.grant_access(
                user_id=canonical_user_id,
                world_identifier=identifier.strip(),
                access_source=BETA,
                allow_disabled_world=True,
            )
            granted.append(access)

        return tuple(granted)

    @staticmethod
    def _canonical_uuid(value: Any) -> UUID:
        if isinstance(value, UUID):
            return value

        if not isinstance(value, str) or not value.strip():
            raise BetaWorldAccessError(
                "Beta user must expose a canonical UUID user_id."
            )

        try:
            return UUID(value.strip())
        except ValueError as exc:
            raise BetaWorldAccessError(
                "Beta user user_id is not a valid canonical UUID."
            ) from exc

    "BetaUserNotFoundError",
    "BetaWorldAccessError",
    "BetaWorldAccessService",

"""Application service for learner-to-learning-world authorization."""

from __future__ import annotations

import asyncio
from typing import Protocol
from uuid import UUID

from app.domain.learning.learner_world_access import BETA, LearnerWorldAccess
from app.domain.learning.learner_world_access_repository import (
    LearnerWorldAccessRepository,
)


class CanonicalUserIdentityReader(Protocol):
    """Minimal provider-neutral read boundary over canonical user identity."""

    async def get_user_for_identity(self, user_id: UUID):
        """Return the canonical user record or None."""
        ...


class WorldAccessError(ValueError):
    """Expected World access validation failure."""


class UserNotFoundError(WorldAccessError):
    """The requested canonical user does not exist."""


class LearningWorldNotFoundError(WorldAccessError):
    """The requested Learning World does not exist."""


class LearningWorldDisabledError(WorldAccessError):
    """The requested Learning World is disabled for ordinary new grants."""


class WorldAccessService:
    """
    Authorize learners for one or more Learning Worlds.

    Important boundary:
    - World access is durable authorization/membership.
    - World ``enabled`` remains runtime availability.
    - Ordinary direct grants still reject disabled Worlds.
    - The explicit ``allow_disabled_world`` flag is reserved for a
      higher-level provisioning policy such as Beta portfolio provisioning.
    """

    def __init__(
        self,
        *,
        access_repository: LearnerWorldAccessRepository,
        user_identity_reader: CanonicalUserIdentityReader,
        learning_world_repository,
    ) -> None:
        if access_repository is None:
            raise ValueError("access_repository is required.")
        if user_identity_reader is None:
            raise ValueError("user_identity_reader is required.")
        if learning_world_repository is None:
            raise ValueError("learning_world_repository is required.")

        self._access_repository = access_repository
        self._user_identity_reader = user_identity_reader
        self._learning_world_repository = learning_world_repository

    async def grant_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
        access_source: str = BETA,
        allow_disabled_world: bool = False,
    ) -> LearnerWorldAccess:
        self._validate_user_id(user_id)
        identifier = self._normalize_world_identifier(world_identifier)

        user = await self._user_identity_reader.get_user_for_identity(user_id)
        if user is None:
            raise UserNotFoundError("Canonical user does not exist.")

        world = await asyncio.to_thread(
            self._learning_world_repository.get_by_identifier,
            identifier,
        )

        if world is None:
            raise LearningWorldNotFoundError(
                f"Learning World does not exist: {identifier}"
            )

        if not world.enabled and not allow_disabled_world:
            raise LearningWorldDisabledError(
                f"Learning World is disabled: {identifier}"
            )

        world_id = self._world_id(world)

        existing = await asyncio.to_thread(
            self._access_repository.get_by_user_and_world,
            user_id=user_id,
            world_id=world_id,
        )

        if existing is not None:
            if existing.is_active():
                return existing

            revived = LearnerWorldAccess.grant(
                user_id=user_id,
                world_id=world_id,
                access_source=access_source,
                access_id=existing.access_id,
            )
            return await asyncio.to_thread(
                self._access_repository.save,
                revived,
            )

        access = LearnerWorldAccess.grant(
            user_id=user_id,
            world_id=world_id,
            access_source=access_source,
        )

        return await asyncio.to_thread(
            self._access_repository.save,
            access,
        )

    async def has_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> bool:
        access = await self.get_access(
            user_id=user_id,
            world_identifier=world_identifier,
        )
        return access is not None and access.is_active()

    async def require_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> LearnerWorldAccess:
        access = await self.get_access(
            user_id=user_id,
            world_identifier=world_identifier,
        )

        if access is None or not access.is_active():
            raise PermissionError(
                "Learner does not have active access to this Learning World."
            )

        return access

    async def get_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> LearnerWorldAccess | None:
        self._validate_user_id(user_id)
        identifier = self._normalize_world_identifier(world_identifier)

        world = await asyncio.to_thread(
            self._learning_world_repository.get_by_identifier,
            identifier,
        )

        if world is None:
            return None

        world_id = self._world_id(world)

        return await asyncio.to_thread(
            self._access_repository.get_by_user_and_world,
            user_id=user_id,
            world_id=world_id,
        )

    async def list_world_access(
        self,
        *,
        user_id: UUID,
    ) -> tuple[LearnerWorldAccess, ...]:
        self._validate_user_id(user_id)

        return await asyncio.to_thread(
            self._access_repository.list_by_user,
            user_id=user_id,
        )

    @staticmethod
    def _world_id(world) -> UUID:
        world_id = getattr(world, "id", None)
        if not isinstance(world_id, UUID):
            raise WorldAccessError(
                "Learning World catalogue entry must expose the "
                "authoritative World UUID as 'id'."
            )
        return world_id

    @staticmethod
    def _validate_user_id(user_id: UUID) -> None:
        if not isinstance(user_id, UUID):
            raise ValueError("user_id must be a UUID.")

    @staticmethod
    def _normalize_world_identifier(world_identifier: str) -> str:
        if not isinstance(world_identifier, str):
            raise ValueError("world_identifier must be a string.")

        identifier = world_identifier.strip()

        if not identifier:
            raise ValueError("world_identifier is required.")

        if len(identifier) > 100:
            raise ValueError("world_identifier is too long.")

        return identifier

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )

            "EASYCHATBOT_PORTAL_URL",
        ),
    )

    authentication_service = AuthenticationService(
        auth_repository,
        email_service=authentication_email_service,
    )

    # ------------------------------------------------------------------
    # Developer Administration
    #
    # Production path:
    #
    #   DeveloperAdminService
    #          |
    #          +--> DeveloperAdminUnitOfWorkFactory
    #                  |
    #                  +--> fresh PostgreSQLDeveloperAdminUnitOfWork
    #                          |
    #                          +--> fresh AsyncSession
    #                          |
    #                          +--> fresh transaction
    #                          |
    #                          +--> auth_user_roles
    #                          |
    #                          +--> developer_admin_audit_events
    #          |
    #          +--> DeveloperAdminAuditService
    #
    # Each Developer Admin mutation receives a fresh Unit-of-Work
    # instance. The role mutation and its security audit evidence
    # therefore remain within the same database transaction while
    # mutable UoW/session state is never shared between operations.
    #
    # The standalone audit repository owns transactions only when its
    # standalone ``record`` operation is used. The production Developer
    # Admin mutation uses ``record_in_session`` through the UoW.
    # ------------------------------------------------------------------

    developer_admin_unit_of_work_factory = (
        PostgreSQLDeveloperAdminUnitOfWorkFactory(
            auth_session_factory,
        )
    )

    developer_admin_audit_service = (
        DeveloperAdminAuditService()
    )

    developer_admin_service = DeveloperAdminService(
        repository=auth_repository,
        unit_of_work_factory=developer_admin_unit_of_work_factory,
        audit_service=developer_admin_audit_service,
    )

    # W15.2.1 Transaction Authorization
    #
    # Transaction authorization is composed once at the application root.
    # The registry and consumer are application-lifetime components and
    # provide their own synchronization boundaries.
    #
    # Authentication assurance remains a separate trusted-evidence
    # boundary. This composition does not manufacture MFA/A2 evidence.
    # ------------------------------------------------------------------

    authorization_service = AuthorizationService()

    transaction_authorization_registry = (
        TransactionAuthorizationRegistry()
    )

    transaction_authorization_consumer = (
        TransactionAuthorizationConsumer(
            transaction_authorization_registry,

        TransactionAuthorizationApplicationService(
            authorization_service=authorization_service,
            registry=transaction_authorization_registry,
            consumer=transaction_authorization_consumer,
        )
    )

                transaction_authorization_service

            clock=trusted_clock,

    # Provider usage / synchronous application persistence

    # The provider-usage repository is synchronous by contract.
    # It therefore receives a synchronous engine/session factory.

    # It must not receive the async authentication session factory.
