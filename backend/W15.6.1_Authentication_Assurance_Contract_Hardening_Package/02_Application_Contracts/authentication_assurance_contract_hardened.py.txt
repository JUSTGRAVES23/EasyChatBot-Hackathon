# W15.6.1 CONTRACT-ONLY REFERENCE — NOT PRODUCTION CODE

from dataclasses import dataclass
from datetime import datetime
from enum import Enum, IntEnum


class AuthenticationAssuranceLevel(IntEnum):
    A0 = 0
    A1 = 1
    A2 = 2
    A3 = 3


# Contract invariant:
# A3 > A2 > A1 > A0


class AuthenticationAssuranceDecision(str, Enum):
    ALLOW = "ALLOW"
    DENY = "DENY"


class AuthenticationAssuranceFailure(str, Enum):
    UNAUTHENTICATED = "UNAUTHENTICATED"
    INVALID_AUTHENTICATION_EVIDENCE = "INVALID_AUTHENTICATION_EVIDENCE"
    INVALID_PRINCIPAL_BINDING = "INVALID_PRINCIPAL_BINDING"
    STALE_AUTHENTICATION_ASSURANCE = (
        "STALE_AUTHENTICATION_ASSURANCE"
    )
    INSUFFICIENT_AUTHENTICATION_ASSURANCE = (
        "INSUFFICIENT_AUTHENTICATION_ASSURANCE"
    )


@dataclass(frozen=True, slots=True)
class ValidatedAuthenticationEvidence:
    """
    Trusted internal authentication evidence.

    This object is created only after the Identity Adapter has validated
    the external identity-provider evidence.

    Callers must not be able to self-assert assurance properties such as
    A3, phishing resistance, or user verification.
    """

    principal_id: str
    issuer: str
    subject: str
    authentication_time: datetime
    authentication_method: str
    authentication_strength: str
    user_verified: bool
    phishing_resistant: bool
    evidence_id: str
    policy_version: str


@dataclass(frozen=True, slots=True)
class AuthenticationAssuranceContext:
    """
    Immutable authentication-assurance result.

    Assurance is derived from validated evidence and centrally controlled
    policy. Callers cannot mutate assurance-relevant fields after validation.
    """

    principal_id: str
    assurance_level: AuthenticationAssuranceLevel
    authentication_time: datetime
    assurance_time: datetime
    authentication_method: str
    authentication_strength: str
    user_verified: bool
    phishing_resistant: bool
    issuer: str
    subject: str
    evidence_id: str
    policy_version: str


@dataclass(frozen=True, slots=True)
class AuthenticationAssuranceDecisionResult:
    """
    Structured assurance-policy decision.

    ALLOW means the requested assurance level has been satisfied.

    DENY means the requested assurance level has not been satisfied.
    The failure field identifies the deterministic reason when applicable.
    """

    decision: AuthenticationAssuranceDecision
    context: AuthenticationAssuranceContext | None
    failure: AuthenticationAssuranceFailure | None


class AuthenticationAssuranceServiceContract:
    """
    Contract-only reference.

    This is NOT production implementation.

    Freshness MUST use trusted server/application time rather than
    caller-controlled current time.

    The implementation MUST accept only validated internal evidence
    produced by the Identity Adapter.

    Higher authentication assurance does NOT imply authorization.
    Authorization remains a separate application security decision.
    """

    def require(
        self,
        *,
        principal_id: str,
        required_level: AuthenticationAssuranceLevel,
        evidence: ValidatedAuthenticationEvidence,
    ) -> AuthenticationAssuranceDecisionResult:
        raise NotImplementedError
