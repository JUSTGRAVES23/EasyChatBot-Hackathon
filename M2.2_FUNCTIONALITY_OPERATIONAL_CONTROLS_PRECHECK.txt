from __future__ import annotations

from datetime import datetime, timezone
from typing import Any

from backend.app.application.health.health_result import (
    HealthCheckResult,
    HealthStatus,
)
from backend.app.application.health.production_readiness_service import (
    ProductionReadinessService,
)
from backend.app.application.provider_management.environment import (
    OperationsEnvironment,
    resolve_environment,
)


class FunctionalityHealthService:
    _FUNCTIONALITIES = (
        ("conversation", "AI Conversation"),
        ("telegram", "Telegram"),
        ("voice_tts", "Voice / TTS"),
        ("lessons", "Lessons"),
        ("review", "Review / Spaced Repetition"),
        ("pdf_presentation", "PDF / Presentation"),
        ("email_workflows", "Email Workflows"),
        ("payments", "Payments"),
        ("authentication", "Authentication"),
        ("scheduler", "Scheduler"),
        ("provider_runtime", "Provider / LLM Runtime"),
        ("database", "Database"),
    )

    _CHECK_TO_FUNCTIONALITIES = {
        "llm_provider": {"conversation", "provider_runtime"},
        "telegram": {"telegram"},
        "postgresql": {"database"},
    }

    _CATEGORY_BY_FUNCTIONALITY = {
        "conversation": "AI",
        "telegram": "Communication",
        "voice_tts": "AI",
        "lessons": "Learning",
        "review": "Learning",
        "pdf_presentation": "Content",
        "email_workflows": "Communication",
        "payments": "Platform",
        "authentication": "Platform",
        "scheduler": "Platform",
        "provider_runtime": "Infrastructure",
        "database": "Infrastructure",
    }

    def __init__(self, *, readiness_service: ProductionReadinessService) -> None:
        if readiness_service is None:
            raise ValueError("readiness_service cannot be None.")
        self._readiness_service = readiness_service

    async def get_health(
        self,
        *,
        environment: OperationsEnvironment | None = None,
    ) -> dict[str, Any]:
        resolved_environment = resolve_environment(environment)
        readiness = await self._readiness_service.check()
        checks_by_name = {check.name: check for check in readiness.checks}
        generated_at = datetime.now(timezone.utc).isoformat()

        functionalities = [
            self._build_functionality(
                functionality_id=functionality_id,
                name=name,
                check=self._resolve_check(
                    functionality_id=functionality_id,
                    checks_by_name=checks_by_name,
                ),
                environment=resolved_environment,
                checked_at=generated_at,
            )
            for functionality_id, name in self._FUNCTIONALITIES
        ]

        attention = [
            item for item in functionalities
            if item["status"] in {"degraded", "unavailable"}
        ]

        counts = {
            "total": len(functionalities),
            "healthy": sum(item["status"] == "healthy" for item in functionalities),
            "degraded": sum(item["status"] == "degraded" for item in functionalities),
            "incident": sum(item["status"] == "unavailable" for item in functionalities),
            "maintenance": 0,
            "disabled": 0,
            "unknown": sum(item["status"] == "unknown" for item in functionalities),
        }

        if counts["incident"] > 0:
            platform_health, score, impact = "incident", 0.0, "high"
        elif counts["degraded"] > 0:
            platform_health, score, impact = "degraded", 50.0, "medium"
        elif counts["healthy"] > 0:
            platform_health, score, impact = "healthy", 100.0, "low"
        else:
            platform_health, score, impact = "unknown", None, "unknown"

        return {
            "generated_at": generated_at,
            "environment": resolved_environment.value,
            "platform_health": platform_health,
            "platform_health_score": score,
            "user_impact": impact,
            "counts": counts,
            "functionalities": functionalities,
            "attention": attention,
        }

    @classmethod
    def _resolve_check(
        cls,
        *,
        functionality_id: str,
        checks_by_name: dict[str, HealthCheckResult],
    ) -> HealthCheckResult | None:
        for check_name, functionality_ids in cls._CHECK_TO_FUNCTIONALITIES.items():
            if functionality_id in functionality_ids:
                return checks_by_name.get(check_name)
        return None

    @classmethod
    def _build_functionality(
        cls,
        *,
        functionality_id: str,
        name: str,
        check: HealthCheckResult | None,
        environment: OperationsEnvironment,
        checked_at: str,
    ) -> dict[str, Any]:
        if check is None:
            status = "unknown"
            health_score = None
            risk_level = "unknown"
            provider = model = latency_ms = None
            reason = "No dedicated health check is currently available."
        else:
            status = cls._map_status(check.status)
            health_score = (
                100.0 if status == "healthy"
                else 50.0 if status == "degraded"
                else 0.0 if status == "unavailable"
                else None
            )
            risk_level = {
                "healthy": "low",
                "degraded": "medium",
                "unavailable": "high",
                "unknown": "unknown",
            }[status]
            provider = check.details.get("provider")
            model = check.details.get("model")
            latency_ms = check.details.get("latency_ms")
            reason = check.details.get("reason", "Health check completed.")

        return {
            "functionality_id": functionality_id,
            "name": name,
            "category": cls._CATEGORY_BY_FUNCTIONALITY[functionality_id],
            "description": f"Operational health for {name}.",
            "status": status,
            "health_score": health_score,
            "latency_ms": latency_ms,
            "provider": provider,
            "model": model,
            "capability": functionality_id,
            "checked_at": checked_at,
            "controllable": False,
            "risk_level": risk_level,
            "reason": reason,
            "environment": environment.value,
            "availability": "enabled",
            "policy_status": "not_evaluated",
        }

    @staticmethod
    def _map_status(status: HealthStatus) -> str:
        if status is HealthStatus.HEALTHY:
            return "healthy"
        if status is HealthStatus.UNHEALTHY:
            return "unavailable"
        return "unknown"

def create_operations_center_router(
    *,
    composition: OperationsCenterComposition | None = None,
) -> APIRouter:
    resolved_composition = composition or create_operations_center_composition()

    dashboard_api = resolved_composition.dashboard_api
    chat_operations_api = resolved_composition.chat_operations_api
    functionality_health_service = resolved_composition.functionality_health_service
    event_service = resolved_composition.event_service
    provider_usage_dashboard_service = resolved_composition.provider_usage_dashboard_service

    router = APIRouter(prefix="/api/operations-center", tags=["operations-center"])

    @router.get("/dashboard")
    async def get_dashboard() -> dict[str, Any]:
        return await dashboard_api.get_dashboard()

    @router.get("/chat-operations")
    async def get_chat_operations_dashboard() -> dict[str, Any]:
        return await chat_operations_api.get_dashboard()

    @router.get("/chat-operations/private-classroom")
    async def get_private_classroom_operations() -> dict[str, Any]:
        return await chat_operations_api.get_private_classroom_details()

    @router.get("/chat-operations/announcement-group")
    async def get_announcement_group_operations() -> dict[str, Any]:
        return await chat_operations_api.get_announcement_group_details()

    @router.get("/functionality-health")
    async def get_functionality_health(
        environment: OperationsEnvironment | None = None,
    ) -> dict[str, Any]:
        return await functionality_health_service.get_health(environment=environment)

    @router.get("/operational-events")
    def get_operational_events() -> list[dict[str, Any]]:
        incidents = event_service.list_incidents()
        return [
            event.as_dict()
            for incident in incidents
            for event in incident.events
        ]

    @router.get("/provider-usage")
    def get_provider_usage() -> dict[str, Any]:
        if provider_usage_dashboard_service is None:
            return {
                "status": "not_configured",
                "reason": "provider_usage_dashboard_service_not_configured",
            }
        return provider_usage_dashboard_service.get_snapshot()

    return router


router = create_operations_center_router()

function HealthRing({ score }: { score: number | null }) {
  const safeScore = score == null ? 0 : Math.max(0, Math.min(100, score));
  const circumference = 2 * Math.PI * 42;
  const offset = circumference - (safeScore / 100) * circumference;

  return (
    <div className={styles.healthRing}>
      <svg viewBox="0 0 100 100" className={styles.healthRingSvg}>
        <circle
          cx="50"
          cy="50"
          r="42"
          className={styles.healthRingTrack}
        />
        <circle
          cx="50"
          cy="50"
          r="42"
          className={styles.healthRingValue}
          strokeDasharray={circumference}
          strokeDashoffset={offset}
        />
      </svg>
      <div className={styles.healthRingLabel}>
        <strong>{score == null ? "—" : `${score}%`}</strong>
        <span>Health</span>
      </div>
    </div>
  );
}

function CategoryChart({
  functionalities,
}: {
  functionalities: FunctionalityHealth[];
}) {
  const categories = useMemo(() => {
    const grouped = new Map<string, number>();

    functionalities.forEach((item) => {
      grouped.set(
        item.category,
        (grouped.get(item.category) ?? 0) + 1,
      );
    });

    return Array.from(grouped.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, 6);
  }, [functionalities]);

  const max = Math.max(...categories.map(([, count]) => count), 1);

  return (
    <div className={styles.chartCard}>
      <div className={styles.cardHeader}>
        <div>
          <h2>Status by Category</h2>
          <p>Current functionality distribution</p>
        </div>
      </div>

      <div className={styles.barChart}>
        {categories.map(([category, count]) => (
          <div className={styles.barColumn} key={category}>
            <div className={styles.barValue}>{count}</div>
            <div className={styles.barTrack}>
              <div
                className={styles.barFill}
                style={{
                  height: `${Math.max(12, (count / max) * 100)}%`,
                }}
              />
            </div>
            <div className={styles.barLabel}>{category}</div>
          </div>
        ))}
      </div>
    </div>
  );
}

    void refresh();

      void refresh();

            </p>
          </div>

          <div className={styles.headerActions}>

              {(["kids", "adults"] as const).map((option) => (
                <button key={option} type="button" aria-pressed={environment === option} onClick={() => setEnvironment(option)} className={`rounded-md px-3 py-2 text-sm font-semibold transition ${environment === option ? "bg-white text-slate-900" : "text-slate-300 hover:bg-white/10"}`}>

          />

        {data.attention.length > 0 && (
          <section className={styles.attentionBanner}>
            <div className={styles.attentionIcon}>
              <Icon type="warning" />
            </div>
            <div>
              <strong>
                {data.attention.length} functionality
                {data.attention.length === 1 ? "" : "ies"} require attention
              </strong>
              <span>
                Review the exception before taking operational action.
              </span>
            </div>
          </section>
        )}

        <section className={styles.chartGrid}>
          <StatusDistribution
            counts={counts}
            functionalities={data.functionalities}
          />
          <CategoryChart functionalities={data.functionalities} />
        </section>

        <section className={styles.workspace}>
          <FunctionalityTable
            functionalities={data.functionalities}
            selectedId={selectedId}
            onSelect={setSelectedId}
          />

          {selected && (
            <aside className={styles.detailsPanel}>
              <div className={styles.detailsHeader}>
                <div>
                  <span>Functionality Details</span>
                  <h2>{selected.name}</h2>
                </div>

                <button
                  type="button"
                  aria-label="Close functionality details"
                  className={styles.iconButton}
                  onClick={() => setSelectedId(null)}
                >
                  <Icon type="close" />
                </button>
              </div>

              <StatusBadge status={selected.status} />

              <p className={styles.detailsDescription}>
                {selected.description}
              </p>

              <div className={styles.detailsGrid}>
                <div>
                  <span>Functionality ID</span>
                  <strong>{selected.functionality_id}</strong>
                </div>
                <div>
                  <span>Category</span>
                  <strong>{selected.category}</strong>
                </div>
                <div>
                  <span>Health Score</span>
                  <strong>{selected.health_score ?? "—"}%</strong>
                </div>
                <div>
                  <span>Latency</span>
                  <strong>{selected.latency_ms ?? "—"} ms</strong>
                </div>
                <div>
                  <span>Provider</span>
                  <strong>{selected.provider ?? "—"}</strong>
                </div>
                <div>
                  <span>Model</span>
                  <strong>{selected.model ?? "—"}</strong>
                </div>
                <div>
                  <span>Risk Level</span>
                  <strong>{selected.risk_level}</strong>
                </div>
                <div>
                  <span>Last Checked</span>
                  <strong>
                    {selected.checked_at
                      ? new Date(selected.checked_at).toLocaleString()
                      : "—"}
                  </strong>
                </div>
              </div>

              <div className={styles.assessment}>
                <span>Current assessment</span>
                <p>{selected.reason}</p>
              </div>

              <div className={styles.controlNotice}>
                <strong>Operational controls</strong>
                <span>
                  M1 is read-only. Operational actions will be introduced
                  through backend-authorised controls with confirmation and
                  audit logging.
                </span>
              </div>
            </aside>
          )}
        </section>

        <footer className={styles.footer}>
          <span>EasyChatBot Operations Center</span>
          <span>Read-only operational visibility</span>
        </footer>
      </div>
    </main>
  );
}




===== FRONTEND CONTROL API =====
import type { OperationsEnvironment } from "./functionalityHealth";

export interface FunctionalityControlResult {
  functionality_id: string;
  environment: OperationsEnvironment;
  availability: "enabled" | "disabled";
  action: "stop" | "resume";
  reason: string | null;
  changed_at: string;
  changed_by: string;
}

async function postControl(
  functionalityId: string,
  action: "stop" | "resume",
  payload: { environment: OperationsEnvironment; reason?: string },
): Promise<FunctionalityControlResult> {
  const response = await fetch(
    `/api/operations-center/functionality/${encodeURIComponent(functionalityId)}/${action}`,
    {
      method: "POST",
      headers: {
        Accept: "application/json",
        "Content-Type": "application/json",
      },
      body: JSON.stringify(payload),
    },
  );

  if (!response.ok) {
    let message = `Functionality ${action} failed: ${response.status}`;
    try {
      const body = (await response.json()) as { detail?: string };
      if (body.detail) message = body.detail;
    } catch {
      // Preserve the safe status-based error.
    }
    throw new Error(message);
  }
  return (await response.json()) as FunctionalityControlResult;
}

export function stopFunctionality(
  functionalityId: string,
  environment: OperationsEnvironment,
  reason: string,
) {
  return postControl(functionalityId, "stop", { environment, reason });
}

export function resumeFunctionality(
  functionalityId: string,
  environment: OperationsEnvironment,
) {
  return postControl(functionalityId, "resume", { environment });
}
