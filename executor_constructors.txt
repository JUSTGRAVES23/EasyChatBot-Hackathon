> """
AI Runtime executor for Live Intelligence.

This adapter bridges the provider-independent AI Runtime
capability contract with the existing LiveIntelligenceAgent.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
LiveIntelligenceExecutor
    |
    v
LiveIntelligenceAgent
    |
    v
LiveIntelligenceResponse

The adapter:

- validates the Runtime capability
- translates Runtime payloads into LiveIntelligenceRequest
- invokes the existing Live Intelligence Agent
- translates the agent response into CapabilityResult

It does not:

- implement intelligence logic
- call Parallel directly
- modify the LiveIntelligenceAgent
- perform governance decisions
"""

from __future__ import annotations

from src.ai_agents.live_intelligence_agent.handler import (
    LiveIntelligenceAgent,
    LiveIntelligenceRequest,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)


class LiveIntelligenceExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for the existing LiveIntelligenceAgent.
    """

    def __init__(
        self,
        agent: LiveIntelligenceAgent,
    ) -> None:
        self._agent = agent

    @property
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """

        return CapabilityType.LIVE_INTELLIGENCE

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Translate and execute a Live Intelligence request.
        """

        if (
            request.capability
            != CapabilityType.LIVE_INTELLIGENCE
        ):
            raise ValueError(
                "LiveIntelligenceExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        agent_request = (
            self._build_agent_request(request)
        )

        response = await self._agent.execute(
            agent_request
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=response,
            metadata={
                "executor": (
                    "live_intelligence_executor"
                ),
            },
        )

    @staticmethod
    def _build_agent_request(
        request: CapabilityRequest,
    ) -> LiveIntelligenceRequest:
        """
        Translate Runtime payload into the existing
        LiveIntelligenceRequest contract.
        """

        payload = request.payload

        topic = payload.get("topic")

        if not isinstance(topic, str):
            raise ValueError(
                "Live Intelligence capability requires "
                "a string 'topic'."
            )

        language = payload.get(
            "language",
            "French",
        )

        learner_level = payload.get(
            "learner_level",
            "B2",
        )

        if not isinstance(language, str):
            raise ValueError(
                "Live Intelligence 'language' "
                "must be a string."
            )

        if not isinstance(
            learner_level,
            str,
        ):
            raise ValueError(
                "Live Intelligence 'learner_level' "
                "must be a string."
            )

        return LiveIntelligenceRequest(
            topic=topic,
            language=language,
            learner_level=learner_level,
        )

> """
AI Runtime executor for Podcast Discovery.

This adapter bridges the provider-independent AI Runtime capability
contract with the existing Podcast Discovery Agent and
Podcast Recommendation Service.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
PodcastDiscoveryExecutor
    |
    +--> PodcastDiscoveryAgent
    |
    +--> PodcastRecommendationService
    |
    v
PodcastRecommendation
    |
    v
CapabilityResult

The executor:

- validates the Runtime capability;
- validates explicit capability inputs;
- builds the existing LearningContext;
- passes explicit learner memories to the PodcastDiscoveryAgent;
- invokes the existing PodcastRecommendationService;
- returns a provider-independent CapabilityResult.

It does not:

- call Apple directly;
- call Spotify directly;
- implement provider logic;
- retrieve memory itself;
- modify MemoryService;
- perform purchases;
- bypass the AI Runtime;
- make governance decisions.
"""

from __future__ import annotations

from collections.abc import Mapping, Sequence
from typing import Any

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
)

from src.ai_runtime.podcast.agent import (
    PodcastDiscoveryAgent,
)

from src.ai_runtime.podcast.models import (
    LearningContext,
)

from src.ai_runtime.podcast.providers import (
    PodcastProvider,
)

from src.ai_runtime.podcast.service import (
    PodcastRecommendationService,
)


class PodcastDiscoveryExecutor(CapabilityExecutor):
    """
    Runtime adapter for the existing Podcast Discovery flow.
    """

    def __init__(
        self,
        agent: PodcastDiscoveryAgent,
        providers: Sequence[PodcastProvider],
    ) -> None:
        if not providers:
            raise ValueError(
                "PodcastDiscoveryExecutor requires at least one provider."
            )

        self._agent = agent
        self._recommendation_service = PodcastRecommendationService(
            providers=list(providers),
        )

    @property
    def capability(self) -> CapabilityType:
        """Return the capability handled by this executor."""

        return CapabilityType.PODCAST_DISCOVERY

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a provider-independent podcast discovery request.
        """

        if request.capability != CapabilityType.PODCAST_DISCOVERY:
            raise ValueError(
                "PodcastDiscoveryExecutor received an unsupported "
                "capability: "
                f"{request.capability.value}"
            )

        intent = self._build_search_intent(request)

        recommendation = self._recommendation_service.recommend(
            intent=intent,
            character_id=self._required_string(
                request.payload,
                "character_id",
            ),
            character_name=self._required_string(
                request.payload,
                "character_name",
            ),
            character_message=self._required_string(
                request.payload,
                "character_message",
            ),
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=recommendation,
            metadata={
                "executor": "podcast_discovery_executor",
            },
        )

    def _build_search_intent(
        self,
        request: CapabilityRequest,
    ):
        """
        Translate the Runtime payload into the existing
        PodcastSearchIntent through PodcastDiscoveryAgent.
        """

        payload = request.payload

        user_id = self._required_string(
            payload,
            "user_id",
        )

        learning_context = self._build_learning_context(
            payload
        )

        memories = self._build_memories(
            payload
        )

        max_results = payload.get(
            "max_results",
            10,
        )

        if not isinstance(max_results, int):
            raise ValueError(
                "Podcast Discovery 'max_results' must be an integer."
            )

        if max_results < 1:
            raise ValueError(
                "Podcast Discovery 'max_results' must be at least 1."
            )

        return self._agent.build_search_intent(
            user_id=user_id,
            learning_context=learning_context,
            memories=memories,
            max_results=max_results,
        )

    @staticmethod
    def _build_learning_context(
        payload: Mapping[str, Any],
    ) -> LearningContext:
        """
        Build the existing provider-independent LearningContext.
        """

        world_id = payload.get(
            "world_id",
            "French World",
        )

        target_language = payload.get(
            "target_language",
            "fr",
        )

        learner_level = payload.get(
            "learner_level",
            "B2",
        )

        learning_goal = payload.get(
            "learning_goal",
            "",
        )

        country_code = payload.get(
            "country_code",
            "LU",
        )

        preferred_languages = payload.get(
            "preferred_languages",
            (),
        )

        values = {
            "world_id": world_id,
            "target_language": target_language,
            "learner_level": learner_level,
            "learning_goal": learning_goal,
            "country_code": country_code,
        }

        for name, value in values.items():
            if not isinstance(value, str):
                raise ValueError(
                    f"Podcast Discovery '{name}' must be a string."
                )

        if not isinstance(
            preferred_languages,
            (list, tuple),
        ):
            raise ValueError(
                "Podcast Discovery 'preferred_languages' "
                "must be a list or tuple."
            )

        if not all(
            isinstance(language, str)
            for language in preferred_languages
        ):
            raise ValueError(
                "Podcast Discovery 'preferred_languages' "
                "must contain only strings."
            )

        return LearningContext(
            world_id=world_id,
            target_language=target_language,
            learner_level=learner_level,
            learning_goal=learning_goal,
            preferred_languages=tuple(
                preferred_languages
            ),
            country_code=country_code,
        )

    @staticmethod
    def _build_memories(
        payload: Mapping[str, Any],
    ) -> list[MemoryEntry]:
        """
        Validate explicit learner memories.

        Memory retrieval remains outside this executor. The caller
        supplies the memories that the Runtime has approved for context.
        """

        memories = payload.get(
            "memories",
            (),
        )

        if not isinstance(
            memories,
            (list, tuple),
        ):
            raise ValueError(
                "Podcast Discovery 'memories' must be a list or tuple."
            )

        for memory in memories:
            if not isinstance(
                memory,
                MemoryEntry,
            ):
                raise ValueError(
                    "Podcast Discovery 'memories' must contain only "
                    "MemoryEntry objects."
                )

        return list(memories)

    @staticmethod
    def _required_string(
        payload: Mapping[str, Any],
        field_name: str,
    ) -> str:
        """Return a required non-empty string payload value."""

        value = payload.get(field_name)

        if not isinstance(value, str):
            raise ValueError(
                f"Podcast Discovery '{field_name}' must be a string."
            )

        if not value.strip():
            raise ValueError(
                f"Podcast Discovery '{field_name}' cannot be empty."
            )

        return value

> """
AI Runtime executor for Podcast Learning Challenges.

This adapter connects the provider-independent AI Runtime

CapabilityResult

The executor:

- validates the Runtime capability;
- validates the explicit PodcastRecommendation payload;

- returns a provider-independent CapabilityResult.

It does not:

- download podcast audio;
- transcribe podcasts;
- call Apple;
- call Spotify;
- retrieve memory;

- bypass the AI Runtime.
"""

from __future__ import annotations

from typing import Any, Mapping

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)

from src.ai_runtime.podcast.challenge import (
    PodcastLearningChallenge,
)

from src.ai_runtime.podcast.challenge_service import (
    PodcastLearningChallengeService,
)

from src.ai_runtime.podcast.models import (
    PodcastRecommendation,
)


class PodcastLearningChallengeExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for Podcast Learning Challenge generation.
    """

    def __init__(
        self,
        service: PodcastLearningChallengeService | None = None,
    ) -> None:
        self._service = (
            service
            if service is not None
            else PodcastLearningChallengeService()
        )

    @property
    def capability(self) -> CapabilityType:
        """
        Return the Runtime capability handled by this executor.
        """

        return CapabilityType.LEARNING_GENERATION

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Generate a Podcast Learning Challenge.

        The request must contain a verified
        PodcastRecommendation under the ``recommendation``
        payload key.
        """

        if (
            request.capability
            != CapabilityType.LEARNING_GENERATION
        ):
            raise ValueError(
                "PodcastLearningChallengeExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        recommendation = (
            self._build_recommendation(
                request.payload
            )
        )

        challenge = (
            self._service.create_challenge(
                recommendation=recommendation,
            )
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=challenge,
            metadata={
                "executor": (
                    "podcast_learning_challenge_executor"
                ),
                "challenge_type": (
                    "podcast_learning_mvp"
                ),
                "content_basis": (
                    "verified_podcast_metadata"
                ),
            },
        )

    @staticmethod
    def _build_recommendation(
        payload: Mapping[str, Any],
    ) -> PodcastRecommendation:
        """
        Validate and extract the PodcastRecommendation.

        The recommendation must already have been produced by
        the Podcast Discovery flow. This executor does not
        perform discovery itself.
        """

        recommendation = payload.get(
            "recommendation"
        )

        if not isinstance(
            recommendation,
            PodcastRecommendation,
        ):
            raise ValueError(
                "Learning Generation capability requires "
                "a PodcastRecommendation under "
                "'recommendation'."
            )

        return recommendation

> """
AI Runtime executor for Podcast Listening Proposals.

This adapter bridges the provider-independent AI Runtime
capability contract with the Podcast Listening Proposal
Service.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
ListeningProposalExecutor
    |
    v
ListeningProposalService
    |
    v
ListeningProposal
    |
    v
Application / UI

The executor:

- validates the Runtime capability;
- validates the explicit payload;
- extracts the existing PodcastRecommendation;
- extracts explicit learner language and level;
- invokes ListeningProposalService;
- returns a provider-independent CapabilityResult.

It does not:

- start playback;
- call Apple;
- call Spotify;
- access external accounts;
- subscribe the learner;
- purchase content;
- perform external side effects;
- bypass the AI Runtime;
- make governance decisions.
"""

from __future__ import annotations

from typing import Any, Mapping

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)

from src.ai_runtime.podcast.listening import (
    ListeningProposalAction,
)

from src.ai_runtime.podcast.listening_service import (
    ListeningProposalService,
)

from src.ai_runtime.podcast.models import (
    PodcastRecommendation,
)


class ListeningProposalExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for the Podcast Listening Proposal flow.
    """

    def __init__(
        self,
        service: ListeningProposalService | None = None,
    ) -> None:
        """
        Initialise the executor.

        The service is injectable so the executor can be tested
        without replacing the Runtime boundary.
        """

        self._service = (
            service
            if service is not None
            else ListeningProposalService()
        )

    @property
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """

        return CapabilityType.PODCAST_LISTENING_PROPOSAL

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a provider-independent listening proposal request.
        """

        if (
            request.capability
            != CapabilityType.PODCAST_LISTENING_PROPOSAL
        ):
            raise ValueError(
                "ListeningProposalExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        recommendation = (
            self._build_recommendation(
                request.payload,
            )
        )

        target_language = (
            self._build_string(
                request.payload,
                "target_language",
            )
        )

        learner_level = (
            self._build_string(
                request.payload,
                "learner_level",
            )
        )

        action = self._build_action(
            request.payload,
        )

        proposal = self._service.create_proposal(
            recommendation,
            target_language=target_language,
            learner_level=learner_level,
            action=action,
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=proposal,
            metadata={
                "executor": (
                    "listening_proposal_executor"
                ),
                "passive": True,
            },
        )

    @staticmethod
    def _build_recommendation(
        payload: Mapping[str, Any],
    ) -> PodcastRecommendation:
        """
        Validate and extract the PodcastRecommendation.

        The recommendation must already have been produced
        by the Podcast Discovery capability.
        """

        recommendation = payload.get(
            "recommendation",
        )

        if not isinstance(
            recommendation,
            PodcastRecommendation,
        ):
            raise ValueError(
                "Podcast Listening Proposal capability "
                "requires a PodcastRecommendation under "
                "'recommendation'."
            )

        return recommendation

    @staticmethod
    def _build_string(
        payload: Mapping[str, Any],
        name: str,
    ) -> str:
        """
        Validate a required string payload value.
        """

        value = payload.get(name)

        if not isinstance(
            value,
            str,
        ):
            raise ValueError(
                f"Podcast Listening Proposal '{name}' "
                "must be a string."
            )

        if not value.strip():
            raise ValueError(
                f"Podcast Listening Proposal '{name}' "
                "cannot be empty."
            )

        return value

    @staticmethod
    def _build_action(
        payload: Mapping[str, Any],
    ) -> ListeningProposalAction:
        """
        Translate the optional Runtime action value.
        """

        value = payload.get(
            "action",
            ListeningProposalAction.LISTEN.value,
        )

        if isinstance(
            value,
            ListeningProposalAction,
        ):
            return value

        if not isinstance(
            value,
            str,
        ):
            raise ValueError(
                "Podcast Listening Proposal 'action' "
                "must be a string."
            )

        try:
            return ListeningProposalAction(
                value,
            )
        except ValueError as exc:
            raise ValueError(
                "Podcast Listening Proposal 'action' "
                "must be 'listen' or 'open'."
            ) from exc
