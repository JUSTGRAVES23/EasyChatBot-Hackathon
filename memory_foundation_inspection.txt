class DemoSearchProvider:
    """
    Controlled search provider for the unified demo.

    The real Parallel provider is already validated separately
    by H1 integration tests.

    This keeps the unified demo deterministic and suitable for
    repeatable hackathon demonstrations.
    """

    async def search(
        self,
        query: str,
        max_results: int = 5,
    ) -> list[SearchResult]:

        return [
            SearchResult(
                title="European AI Act — Demo Source",
                url="https://example.com/ai-act",
                content=(
                    "The European Union AI Act establishes "
                    "a risk-based framework for artificial "
                    "intelligence."
                ),
                source="demo-provider",
            )
        ]


class DemoIntelligenceProcessor:
    """
    Controlled intelligence processor.

    The real provider-backed intelligence flow is validated
    separately in H1.
    """

    async def summarise(
        self,
        results: list[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, list[str]]:

        return (
            "The European Union AI Act establishes "
            "a risk-based framework for artificial "
            "intelligence.",
            [
                "AI governance",
                "risk management",
                "compliance",
            ],
        )


class DemoChallengeContentGenerator:
    """
    Controlled educational content generator.
    """

    async def generate_vocabulary(
        self,
        summary: str,
        language: str,
        learner_level: str,
    ) -> list[VocabularyItem]:

        return [
            VocabularyItem(
                term="gouvernance",
                definition=(
                    "La manière dont une organisation "
                    "est dirigée et contrôlée."

                example=(
                    "La gouvernance de l'IA devient "

                term="réglementation",

                    "Ensemble des règles qui encadrent "
                    "une activité."

                example=(

                ),
            ),
        ]

    async def generate_questions(
        self,
        summary: str,
        language: str,

        return [
            DiscussionQuestion(
                question=(
                    "Pourquoi la réglementation de "
                    "l'IA est-elle importante ?"

                difficulty=learner_level,
            ),
            DiscussionQuestion(
                question=(
                    "Quels sont les avantages d'une "
                    "réglementation européenne de l'IA ?"
                ),
                difficulty=learner_level,
            ),
        ]


def create_governance_policy() -> GovernancePolicy:
    """
    Governance policy for the live intelligence demo.

    Live research may access external information but
    must never create external side effects.
    """

    return GovernancePolicy(
        capability="live_intelligence_research",
        risk_level=RiskLevel.MEDIUM,
        requires_human_review=False,
        allows_external_calls=True,
        allows_side_effects=False,
    )


async def create_learning_challenge() -> LearningChallenge:
    """
    Execute the H1 learning challenge flow.
    """

    intelligence_agent = LiveIntelligenceAgent(
        search_provider=DemoSearchProvider(),
        processor=DemoIntelligenceProcessor(),
    )

    intelligence = await intelligence_agent.execute(
        LiveIntelligenceRequest(
            topic="AI Act",
            language="French",
            learner_level="B2",
        )
    )

    challenge_generator = ChallengeGenerator(
        content_generator=DemoChallengeContentGenerator()
    )

    return await challenge_generator.generate(
        intelligence=intelligence,
        language="French",
        learner_level="B2",
    )


async def run_demo() -> None:
    """
    Execute the complete hackathon demonstration.
    """

    print()
    print("=" * 72)
    print("EasyChatBot — HACKATHON DEMONSTRATION")
    print("=" * 72)
    print()

    # -----------------------------------------------------------------------
    # STEP 1 — Governance
    # -----------------------------------------------------------------------

    print("STEP 1 — GOVERNANCE")
    print("-" * 72)

    governance_engine = GovernancePolicyEngine()

    governance_result = governance_engine.evaluate(
        create_governance_policy()
    )

    print(
        f"Capability: {governance_result.capability}"
    )

    print(
        f"Risk level: "
        f"{governance_result.risk_level.value}"
    )

    print(
        f"Decision: "
        f"{governance_result.decision.value.upper()}"
    )

    if governance_result.decision != GovernanceDecision.ALLOW:
        raise RuntimeError(
            "Hackathon demo stopped because governance "
            "did not allow the capability."
        )

    print("Governance check: PASSED")
    print()

    # -----------------------------------------------------------------------
    # STEP 2 — Live Intelligence + Challenge
    # -----------------------------------------------------------------------

    print("STEP 2 — LIVE INTELLIGENCE")
    print("-" * 72)

    challenge = await create_learning_challenge()

    print(f"Topic: {challenge.topic}")
    print(f"Language: {challenge.language}")
    print(f"Learner level: {challenge.learner_level}")
    print()

    print("Summary:")
    print(challenge.summary)
    print()

    print("Vocabulary:")

    for item in challenge.vocabulary:
        print(f"  • {item.term}: {item.definition}")

    print()

    print("Discussion questions:")

    for index, question in enumerate(
        challenge.discussion_questions,
        start=1,
    ):
        print(
            f"  {index}. {question.question}"
        )

    print()

    # -----------------------------------------------------------------------
    # STEP 3 — Quality Evaluation
    # -----------------------------------------------------------------------

    print("STEP 3 — AI QUALITY EVALUATION")
    print("-" * 72)

    evaluator = DeterministicEvaluator()

    criterion_scores = evaluator.evaluate(
        challenge
    )

    evaluation_engine = EvaluationEngine(
        criteria=[
            COMPLETENESS,
            LANGUAGE_LEVEL,
            EDUCATIONAL_VALUE,
            SOURCE_COVERAGE,
        ],
        passing_score=0.70,
    )

    evaluation_result = evaluation_engine.evaluate(
        criterion_scores
    )

    report = EvaluationReport().generate(
        evaluation_result
    )

    print(report)

    # -----------------------------------------------------------------------
    # STEP 4 — PERSONALISED FLASH DEAL
    # -----------------------------------------------------------------------

    print("STEP 4 — PERSONALISED DISCOVERY")
    print("-" * 72)

    # Build the same domain inputs that a real discovery
    # application flow would receive.
    demo_deal = build_demo_deal()
    demo_memory = build_demo_memory()

    discovery_request = DiscoveryRequest(
        user_id=demo_memory.user_id,
        deal=demo_deal,
        character=CharacterContext(
            character_id="yeti",
            character_name="Yeti",
            message=(
                "Look what I found! "
                "This could be perfect for you."
            ),
        ),
        product_image_url=(
            "https://example.com/demo/iphone17-pro.jpg"
        ),
        tagline="Pro. Beautiful. Powerful.",
        highlights=(
            "Pro camera system",
            "High-performance chip",
            "All-day battery life",
            "Premium display",
        ),
    )

    flash_deal_card = DiscoveryService().create_flash_deal(
        discovery_request,
        memories=[demo_memory],
    )

    print(
        render_demo_flash_deal_card(
            flash_deal_card
        )
    )

    assert flash_deal_card.character_id == "yeti"
    assert flash_deal_card.deal_id == demo_deal.deal_id
    assert flash_deal_card.product_name
    assert demo_memory.user_id == "demo-user"
    assert flash_deal_card.relevance_reason
    assert flash_deal_card.get_deal_url
    assert flash_deal_card.more_information_url

    # -----------------------------------------------------------------------
    # STEP 5 — COMMERCE ACTION
    # -----------------------------------------------------------------------

    print("STEP 5 — COMMERCE ACTION")
    print("-" * 72)

    commerce_action = FlashDealCommerce().create_action(
        user_id="demo-user",
        card=flash_deal_card,
    )

    print(
        f"Product: {flash_deal_card.product_name}"
    )
    print(
        f"Vendor: {flash_deal_card.vendor_name}"
    )
    print(
        f"Commerce action: "
        f"{commerce_action.action_type.value.upper()}"
    )

    commerce_result = CommerceService().initiate(
        commerce_action
    )

    print(
        f"Status: "
        f"{commerce_result.status.value.upper()}"
    )

    print()
    print("GET DEAL")
    print("  -> Approved commerce boundary")
    print("  -> Vendor handoff")
    print()
    print(
        "Payment is not executed in the hackathon demo."
    )

    assert (
        commerce_result.status
        == CommerceActionStatus.REDIRECT
    )
    assert (
        commerce_result.deal_id
        == flash_deal_card.deal_id
    )
    assert (
        commerce_result.next_url
        == flash_deal_card.get_deal_url
    )

    # -----------------------------------------------------------------------
    # STEP 6 — Final result
    # -----------------------------------------------------------------------

    print()
    print("=" * 72)

    if evaluation_result.passed:
        print("HACKATHON DEMO: SUCCESS")
        print()
        # ASCII-safe output keeps the executable reliable when
        # stdout is captured by Windows pytest/subprocess.
        # Preserve the stable hackathon readiness marker.
        print(
            "Governance -> Intelligence -> Learning -> Evaluation"
        )
        print(
            "Governance -> Intelligence -> Learning -> "
            "Evaluation -> Personalised Discovery"
        )
        print(
            "All demonstrated capabilities completed successfully."
        )
    else:
        print("HACKATHON DEMO: QUALITY CHECK FAILED")

    print("=" * 72)
    print()

    assert governance_result.decision == GovernanceDecision.ALLOW
    assert challenge.topic == "AI Act"
    assert evaluation_result.passed is True


def main() -> None:
    """
    Application entry point.
    """

    asyncio.run(run_demo())


if __name__ == "__main__":
    main()

"""
Base AI Agent Contract.

All EasyChatBot AI Agents inherit this contract.

The BaseAgent defines:
- agent identity
- execution interface
- runtime interaction

Agents do not:
- call providers directly
- manage memory
- modify external systems

The AI Runtime controls execution.
"""

from abc import ABC, abstractmethod

class ConversationAgent(BaseAgent):
    """
    Character-aware conversation agent.

    The agent depends primarily on the application-level
    LLMService boundary.

    The LLMService is responsible for delegating
    execution to the configured LLMProvider.

    A legacy ``llm_provider`` argument remains accepted
    as a compatibility seam for existing tests and
    callers. When supplied, it is wrapped by LLMService.

    New composition code should inject ``llm_service``.
    """

    def __init__(
        self,
        character: CharacterProfile,
        llm_service: LLMService | None = None,
        llm_provider: LLMProvider | None = None,
    ) -> None:
        """
        Initialise the Conversation Agent.

        Parameters
        ----------
        character:
            Approved EasyChatBot character profile.

        llm_service:
            Application-level LLM execution service.

        llm_provider:
            Compatibility injection seam.

            When supplied and ``llm_service`` is not supplied,
            the provider is wrapped inside LLMService.

        Raises
        ------
        ValueError
            If both ``llm_service`` and ``llm_provider``
            are supplied simultaneously.
        """

        super().__init__(
            AgentManifest(
                agent_id="conversation_agent",
                name="Conversation Agent",
                description=(
                    "Provides character-aware "
                    "conversation responses."
                ),
            )
        )

        if (
            llm_service is not None
            and llm_provider is not None
        ):
            raise ValueError(
                "Provide either llm_service or "
                "llm_provider, not both."
            )

        self.character = character

        if llm_service is not None:
            self.llm_service = llm_service

        elif llm_provider is not None:
            self.llm_service = LLMService(
                llm_provider
            )

        else:
            self.llm_service = None

    def build_request(
        self,
        context: ExecutionContext,
    ) -> ConversationRequest:
        """
        Build a provider-independent conversation request.

        Runtime state and character identity are translated
        into the ConversationRequest contract.

        No provider-specific implementation is introduced.
        """

        system_instruction = (
            f"You are {self.character.name}, "
            f"a character in EasyChatBot. "
            f"Your personality is: "
            f"{self.character.personality} "
            f"Your learning style is: "
            f"{self.character.learning_style} "
            f"Communicate in a way that is consistent "
            f"with this character."
        )

        conversation_context = None

        if context.has_intelligence_context:
            conversation_context = (
                context.intelligence_context.to_context_text()

        return ConversationRequest(
            system_instruction=system_instruction,
            user_prompt=context.request,
            conversation_context=conversation_context,
            character_id=self.character.character_id,

        When an LLMService is configured, execution
        passes through the application service boundary.
        """

        conversation_request = self.build_request(
            context

        provider_result = None

        if self.llm_service is not None:

                await self.llm_service.generate(

            if not provider_result.success:

                    agent_id=self.manifest.agent_id,
                    output={
                        "character": self.character.name,

                        "character_profile": {

                            ),
                            "name": self.character.name,
                            "description": (
                                self.character.description
                            ),
                            "personality": (
                                self.character.personality
                            ),
                            "learning_style": (

                            ),
                        },

                        "personality": (
                            self.character.personality
                        ),

                        "learning_style": (
                            self.character.learning_style
                        ),

                        "message": (
                            "The conversation provider "
                            "could not generate a response."
                        ),

                        "intelligence_context": (
                            conversation_request
                            .conversation_context
                        ),

                        "conversation_request": (
                            conversation_request
                        ),

                        "provider_id": (
                            provider_result.provider_id
                        ),

                        "provider_error": (
                            provider_result.error
                        ),
                    },
                )

        if provider_result is not None:
            message = provider_result.output

        else:
            message = (
                f"{self.character.name}: "
                f"Let's explore this together. "
                f"You asked: {context.request}"
            )

        output = {
            "character": self.character.name,

            "character_profile": {
                "character_id": (
                    self.character.character_id
                ),
                "name": self.character.name,
                "description": (
                    self.character.description
                ),

                    self.character.personality

                "learning_style": (

                self.character.personality
            ),

            "learning_style": (
                self.character.learning_style
            ),

            "message": message,

            "intelligence_context": (
                conversation_request
                .conversation_context
            ),

            "conversation_request": (
                conversation_request
            ),
        }

        if provider_result is not None:
            output["provider_id"] = (
                provider_result.provider_id
            )

        return AgentResult(
            success=True,
            agent_id=self.manifest.agent_id,
            output=output,
        )

"""
Memory Service.

Application-layer memory service.

This service adapts the legacy backend MemoryRecord contract
to the hackathon AI Runtime MemoryEntry / MemoryStore
persistence contract.

Architecture:

Application
    |
    v
MemoryService

MemoryEntry / MemoryType
    |

MemoryStore
    |

SQLite

Responsibilities:
- save application memories
- retrieve application memories
- translate application memory categories
- isolate persistence details from callers

Does not:
- perform semantic retrieval
- call an LLM
- execute agents
- make autonomous memory decisions
"""

from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from backend.memory.memory_record import (
    MemoryRecord,
)

from backend.memory.memory_store import (
    MemoryStore,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryService:
    """
    Controls application memory operations.

    The service keeps the existing MemoryRecord interface
    for the backend application while delegating persistence
    to the current AI Runtime MemoryStore.
    """

    DEFAULT_DATABASE_PATH = (
        Path("data") / "memory.db"
    )

    def __init__(
        self,
        store: MemoryStore | None = None,
        database_path: str | Path | None = None,
    ) -> None:
        """
        Initialise the memory service.

        Args:
            store:
                Optional injected MemoryStore.

                Primarily useful for tests and controlled
                application configuration.

            database_path:
                Optional SQLite database path.

                Ignored when an explicit store is supplied.
        """

        if store is not None:
            self.store = store
            return

        path = (
            Path(database_path)
            if database_path is not None
            else self.DEFAULT_DATABASE_PATH
        )

        path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        self.store = MemoryStore(
            database_path=path
        )

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> MemoryRecord:
        """
        Save an application memory.

        The legacy category is preserved in metadata while
        the persistence layer receives a supported MemoryType.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        if not category.strip():
            raise ValueError(
                "Memory category cannot be empty."
            )

        if not content.strip():
            raise ValueError(
                "Memory content cannot be empty."
            )

        created_at = datetime.now(
            timezone.utc
        )

        record = MemoryRecord(
            user_id=user_id,
            category=category,
            content=content,
            created_at=created_at,
        )

        memory_entry = MemoryEntry(
            user_id=user_id,
            memory_type=self._to_memory_type(
                category
            ),
            content=content,
            created_at=created_at,
            updated_at=created_at,
            metadata={
                "category": category,
            },
        )

        self.store.save(
            memory_entry
        )

        return record

    def recall(
        self,
        user_id: str,
    ) -> list[MemoryRecord]:
        """
        Retrieve memories for a user.

        Persistence-specific MemoryEntry objects are
        translated back into the application-facing
        MemoryRecord contract.
        """

        entries = self.store.list_for_user(
            user_id
        )

        return [
            self._to_memory_record(entry)
            for entry in entries
        ]

    @staticmethod
    def _to_memory_type(
        category: str,
    ) -> MemoryType:
        """
        Translate application categories into the
        supported AI Runtime memory types.

        The original category remains available in
        MemoryEntry.metadata.
        """

        normalized = category.strip().lower()

        if normalized == "saved_item":
            return MemoryType.SAVED_ITEM

        if normalized == "discovery_context":
            return MemoryType.DISCOVERY_CONTEXT

        return MemoryType.PREFERENCE

    @staticmethod
    def _to_memory_record(
        entry: MemoryEntry,
    ) -> MemoryRecord:
        """
        Translate a persistent MemoryEntry into the
        application-facing MemoryRecord.
        """

        category = entry.metadata.get(
            "category",
            entry.memory_type.value,
        )

        return MemoryRecord(
            user_id=entry.user_id,
            category=category,
            content=entry.content,
            created_at=entry.created_at,
        )

"""
Hackathon Memory persistence.

This module provides the minimal persistent Memory Store
required by the hackathon.

Architecture:

MemoryService
    |
    v
MemoryStore
    |
    v
SQLite

The store is deliberately small.

It is responsible only for persistence and retrieval of
MemoryEntry objects.

It does not:

- decide what should be remembered
- call an LLM
- perform semantic retrieval
- execute agents
- perform governance decisions
- manage products
- manage vendor links
- process payments
"""

from __future__ import annotations

import json
import sqlite3
from datetime import datetime
from pathlib import Path

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryStore:
    """
    SQLite-backed persistence for hackathon memories.

    The database schema is created automatically when the
    store is initialised.
    """

    def __init__(
        self,
        database_path: str | Path,
    ) -> None:
        self._database_path = str(database_path)

        self._initialise_database()

    def _connect(self) -> sqlite3.Connection:
        """
        Create a SQLite connection.
        """

        connection = sqlite3.connect(
            self._database_path
        )

        connection.row_factory = sqlite3.Row

        return connection

    def _initialise_database(self) -> None:
        """
        Create the Memory table if it does not exist.
        """

        with self._connect() as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    memory_id TEXT PRIMARY KEY,
                    user_id TEXT NOT NULL,
                    memory_type TEXT NOT NULL,
                    content TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    metadata_json TEXT NOT NULL
                )
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_id
                ON memories(user_id)
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_type
                ON memories(user_id, memory_type)
                """
            )

    def save(
        self,
        memory: MemoryEntry,
    ) -> MemoryEntry:
        """
        Persist a MemoryEntry.

        Existing entries with the same memory_id are
        replaced.
        """

        with self._connect() as connection:
            connection.execute(
                """
                INSERT OR REPLACE INTO memories (
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                )
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.memory_id,
                    memory.user_id,
                    memory.memory_type.value,
                    memory.content,
                    memory.created_at.isoformat(),
                    memory.updated_at.isoformat(),
                    json.dumps(
                        memory.metadata,
                        ensure_ascii=False,
                    ),
                ),
            )

        return memory

    def get(
        self,
        memory_id: str,
    ) -> MemoryEntry | None:
        """
        Retrieve a single memory by identifier.
        """

        with self._connect() as connection:
            row = connection.execute(
                """
                SELECT
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            ).fetchone()

        if row is None:
            return None

        return self._row_to_memory(row)

    def list_for_user(
        self,
        user_id: str,
        memory_type: MemoryType | None = None,
    ) -> list[MemoryEntry]:
        """
        Retrieve memories belonging to a user.

        Results are returned newest first.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        with self._connect() as connection:
            if memory_type is None:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                    ORDER BY updated_at DESC
                    """,
                    (user_id,),
                ).fetchall()

            else:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                      AND memory_type = ?
                    ORDER BY updated_at DESC
                    """,
                    (
                        user_id,
                        memory_type.value,
                    ),
                ).fetchall()

        return [
            self._row_to_memory(row)
            for row in rows
        ]

    def delete(
        self,
        memory_id: str,
    ) -> bool:
        """
        Delete a memory.

        Returns True when a memory was deleted.
        """

        with self._connect() as connection:
            cursor = connection.execute(
                """
                DELETE FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            )

        return cursor.rowcount > 0

    @staticmethod
    def _row_to_memory(
        row: sqlite3.Row,
    ) -> MemoryEntry:
        """
        Convert a SQLite row into a MemoryEntry.
        """

        return MemoryEntry(
            memory_id=row["memory_id"],
            user_id=row["user_id"],
            memory_type=MemoryType(
                row["memory_type"]
            ),
            content=row["content"],
            created_at=datetime.fromisoformat(
                row["created_at"]
            ),
            updated_at=datetime.fromisoformat(
                row["updated_at"]
            ),
            metadata=json.loads(
                row["metadata_json"]
            ),
        )

    LLMProvider,

from backend.providers.llm.openai_provider import (
    OpenAIProvider,
)


class LLMProviderFactory:
    """
    Factory responsible for creating LLM providers.

    Provider selection is centralised here so the
    application layer does not depend directly on
    concrete infrastructure implementations.
    """

    @staticmethod
    def create(
        provider: str = "openai",
    ) -> LLMProvider:
        """
        Create an LLM provider.

        Parameters
        ----------
        provider:
            Provider identifier.

        Supported providers
        -------------------
        openai:
            OpenAI LLM provider.

        Returns
        -------
        LLMProvider
            Configured provider implementation.

        Raises
        ------
        ValueError
            If provider is invalid or unsupported.
        """

        if provider is None:
            raise ValueError(
                "Provider name cannot be empty."
            )

        if not isinstance(
            provider,
            str,
        ):
            raise ValueError(
                "Provider name must be a string."
            )

        provider_name = (
            provider.strip().lower()
        )

        if not provider_name:
            raise ValueError(
                "Provider name cannot be empty."
            )

        if provider_name == "openai":
            return OpenAIProvider(
                model="gpt-5-mini",
            )

        raise ValueError(
            (
                "Unsupported LLM provider: "
                f"{provider}"
            )
        )

class OpenAIProvider(LLMProvider):
    """
    OpenAI implementation of the LLMProvider contract.

    The OpenAI SDK is deliberately isolated inside this
    infrastructure adapter.
    """

    def __init__(
        self,
        model: str = "gpt-5-mini",
        client: AsyncOpenAI | None = None,
    ) -> None:
        """
        Initialise the OpenAI provider.

        Args:
            model:
                OpenAI model identifier.

            client:
                Optional injected AsyncOpenAI client.

                Dependency injection allows unit tests to
                provide a fake client without making a real
                API call.
        """

        super().__init__(
            provider_id="openai",
        )

        if not model.strip():
            raise ValueError(
                "OpenAI model cannot be empty."
            )

        self.model = model.strip()

        self._client = (
            client
            if client is not None
            else AsyncOpenAI(
                api_key=[REDACTED]"OPENAI_API_KEY"
                )
            )
        )

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate a response using OpenAI.

        The provider translates the provider-independent
        ConversationRequest into the OpenAI Responses API
        request.

        Args:
            request:
                Provider-independent conversation request.

        Returns:
            ProviderResult containing the generated text
            or an error description.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        messages = [
            {
                "role": "system",
                "content": request.system_instruction,
            },
        ]

        if request.conversation_context:
            messages.append(
                {
                    "role": "user",
                    "content": (
                        "Additional context:\n"
                        f"{request.conversation_context}"
                    ),
                }
            )

        messages.append(
            {
                "role": "user",
                "content": request.user_prompt,
            }
        )

        try:
            response = await self._client.responses.create(
                model=self.model,
                input=messages,
            )

        except Exception as exc:
            logger.exception(
                "OpenAI provider execution failed."
            )

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=str(exc),
            )

        output_text = (
            getattr(
                response,
                "output_text",
                None,
            )
        )

        if not output_text:
            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=(
                    "OpenAI returned an empty response."
                ),
            )

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output=output_text,
        )

        memory_service: MemoryService | None = None,

    async def interact(
        self,
        context: ExecutionContext,
        use_research: bool = False,
    ) -> dict[str, Any]:

        Execute a complete companion interaction.

        produces an IntelligenceContext which is attached

        Conversation Agent executes.

        The original ExecutionContext is never mutated.

        context:

class LLMService:
    """
    Application service for LLM execution.

    The service depends only on the provider-independent
    LLMProvider contract.

    Provider construction and provider selection remain
    outside this service.
    """

    def __init__(
        self,
        llm_provider: LLMProvider,
    ) -> None:
        """
        Initialise the LLM service.

        Parameters
        ----------
        llm_provider:
            Provider implementing the LLMProvider contract.
        """

        if llm_provider is None:
            raise ValueError(
                "LLM provider cannot be None."
            )

        self.llm_provider = llm_provider

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate an LLM response.

        The service delegates execution to the injected
        provider without knowing the provider's concrete
        implementation.

        Parameters
        ----------
        request:
            Provider-independent conversation request.

        Returns
        -------
        ProviderResult
            Standard provider execution result.

        Raises
        ------
        ValueError
            If the conversation request is None.

        RuntimeError
            If the provider returns no result.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        logger.info(
            "LLM generation started."
        )

        result = await self.llm_provider.generate(
            request
        )

        if result is None:
            raise RuntimeError(
                "LLM provider returned no result."
            )

        logger.info(
            (
                "LLM generation completed. "
                "success=%s provider=%s"
            ),
            result.success,
            result.provider_id,
        )

        return result

"""
Hackathon Flash Deal Card presentation contract.

This module defines the user-facing data contract for the
EasyChatBot Flash Deal Card shown by a character.

Architecture:

Parallel / Discovery
        |
        v
     DealCard
        |
        +---- Memory Context
        |
        +---- Character
        |
        v
  FlashDealCard
        |
        +---- GET DEAL
        |
        +---- MORE INFORMATION
        |
        v
 External Vendor / Future Commerce Flow

The model is presentation-oriented.

It does not:
- call providers
- access SQLite
- perform payments
- execute agents
- scrape vendor pages
- invent product information
"""

from __future__ import annotations

from dataclasses import dataclass, field

from src.ai_runtime.discovery.models import DealCard


@dataclass(frozen=True)
class FlashDealCard:
    """
    User-facing Flash Deal Card.

    This contract maps directly to the hackathon UX:

    - character discovery
    - product information
    - pricing
    - discount
    - personal relevance
    - vendor
    - GET DEAL action
    - MORE INFORMATION action

    The card deliberately contains no payment implementation.
    """

    deal_id: str
    character_id: str
    character_name: str
    character_message: str

    product_name: str
    product_image_url: str
    tagline: str
    highlights: tuple[str, ...]

    current_price: float
    currency: str
    previous_price: float | None
    discount_percentage: float | None
    savings: float | None

    relevance_reason: str

    vendor_name: str
    more_information_url: str
    get_deal_url: str

    badge: str = "NEW FIND!"
    card_id: str | None = None
    metadata: dict[str, str] = field(
        default_factory=dict
    )

    def __post_init__(self) -> None:
        """
        Validate the presentation contract.
        """

        required_text_fields = {
            "deal_id": self.deal_id,
            "character_id": self.character_id,
            "character_name": self.character_name,
            "character_message": self.character_message,
            "product_name": self.product_name,
            "product_image_url": self.product_image_url,
            "tagline": self.tagline,
            "currency": self.currency,
            "relevance_reason": self.relevance_reason,
            "vendor_name": self.vendor_name,
            "more_information_url": self.more_information_url,
            "get_deal_url": self.get_deal_url,
        }

        for field_name, value in required_text_fields.items():
            if not value.strip():
                raise ValueError(
                    f"FlashDealCard {field_name} cannot be empty."
                )

        if not self.highlights:
            raise ValueError(
                "FlashDealCard must contain at least one highlight."
            )

        if any(
            not highlight.strip()
            for highlight in self.highlights
        ):
            raise ValueError(
                "FlashDealCard highlights cannot contain empty values."
            )

        if self.current_price < 0:
            raise ValueError(
                "FlashDealCard current_price cannot be negative."
            )

        if (
            self.previous_price is not None
            and self.previous_price < 0
        ):
            raise ValueError(
                "FlashDealCard previous_price cannot be negative."
            )

        if (
            self.discount_percentage is not None
            and not 0 <= self.discount_percentage <= 100
        ):
            raise ValueError(
                "FlashDealCard discount_percentage must be "
                "between 0 and 100."
            )

        if self.savings is not None and self.savings < 0:
            raise ValueError(
                "FlashDealCard savings cannot be negative."
            )

        self._validate_url(
            self.product_image_url,
            "product_image_url",
        )
        self._validate_url(
            self.more_information_url,
            "more_information_url",
        )
        self._validate_url(
            self.get_deal_url,
            "get_deal_url",
        )

    @staticmethod
    def _validate_url(
        url: str,
        field_name: str,
    ) -> None:
        """
        Validate a user-facing URL.
        """

        if not (
            url.startswith("https://")
            or url.startswith("http://")
        ):
            raise ValueError(
                f"FlashDealCard {field_name} must use "
                "HTTP or HTTPS."
            )

    @classmethod
    def from_deal(
        cls,
        *,
        deal: DealCard,
        character_id: str,
        character_name: str,
        character_message: str,
        relevance_reason: str,
        product_image_url: str,
        tagline: str,
        highlights: list[str] | tuple[str, ...],
        badge: str = "NEW FIND!",
        metadata: dict[str, str] | None = None,
    ) -> "FlashDealCard":
        """
        Build a presentation card from a domain DealCard.

        Product, pricing and vendor information comes from the

        self,
        card: FlashDealCard,
    ) -> str:
        """
        Render a FlashDealCard as user-facing Markdown.

        sections: list[str] = []

        sections.append(

        )

    def remember(

            ),
        )

        return self._store.save(
            memory
        )

    def get(
        self,
        memory_id: str,
    ) -> MemoryEntry | None:
        """
        Retrieve a memory by identifier.
        """

        return self._store.get(
            memory_id
        )

    def list_for_user(
        self,

"""
Hackathon Memory persistence.

This module provides the minimal persistent Memory Store
required by the hackathon.

Architecture:

MemoryService
      |
      v
MemoryStore
      |
      v
SQLite

The store is deliberately small.

It is responsible only for persistence and retrieval of
MemoryEntry objects.

It does not:
- decide what should be remembered
- call an LLM
- perform semantic retrieval
- execute agents
- perform governance decisions
- manage purchases
"""

from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryStore:
    """
    SQLite-backed persistence for hackathon memories.

    The database schema is created automatically when the
    store is initialised.
    """

    def __init__(
        self,
        database_path: str | Path,
    ) -> None:
        self._database_path = str(database_path)

        self._initialise_database()

    def _connect(self) -> sqlite3.Connection:
        """
        Create a SQLite connection.
        """

        connection = sqlite3.connect(
            self._database_path
        )

        connection.row_factory = sqlite3.Row

        return connection

    def _initialise_database(self) -> None:
        """
        Create the Memory table if it does not exist.
        """

        with self._connect() as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    memory_id TEXT PRIMARY KEY,
                    user_id TEXT NOT NULL,
                    memory_type TEXT NOT NULL,
                    content TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    metadata_json TEXT NOT NULL
                )
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_id
                ON memories(user_id)
                """
            )

            connection.execute(
                """
                CREATE INDEX IF NOT EXISTS
                idx_memories_user_type
                ON memories(user_id, memory_type)
                """
            )

    def save(
        self,
        memory: MemoryEntry,
    ) -> MemoryEntry:
        """
        Persist a MemoryEntry.

        Existing entries with the same memory_id are
        replaced.
        """

        with self._connect() as connection:
            connection.execute(
                """
                INSERT OR REPLACE INTO memories (
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                )
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.memory_id,
                    memory.user_id,
                    memory.memory_type.value,
                    memory.content,
                    memory.created_at.isoformat(),
                    memory.updated_at.isoformat(),
                    json.dumps(
                        memory.metadata,
                        ensure_ascii=False,
                    ),
                ),
            )

        return memory

    def get(
        self,
        memory_id: str,
    ) -> MemoryEntry | None:
        """
        Retrieve a single memory by identifier.
        """

        with self._connect() as connection:
            row = connection.execute(
                """
                SELECT
                    memory_id,
                    user_id,
                    memory_type,
                    content,
                    created_at,
                    updated_at,
                    metadata_json
                FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            ).fetchone()

        if row is None:
            return None

        return self._row_to_memory(row)

    def list_for_user(
        self,
        user_id: str,
        memory_type: MemoryType | None = None,
    ) -> list[MemoryEntry]:
        """
        Retrieve memories belonging to a user.

        Results are returned newest first.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        with self._connect() as connection:

            if memory_type is None:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                    ORDER BY updated_at DESC
                    """,
                    (user_id,),
                ).fetchall()

            else:
                rows = connection.execute(
                    """
                    SELECT
                        memory_id,
                        user_id,
                        memory_type,
                        content,
                        created_at,
                        updated_at,
                        metadata_json
                    FROM memories
                    WHERE user_id = ?
                      AND memory_type = ?
                    ORDER BY updated_at DESC
                    """,
                    (
                        user_id,
                        memory_type.value,
                    ),
                ).fetchall()

        return [
            self._row_to_memory(row)
            for row in rows
        ]

    def delete(
        self,
        memory_id: str,
    ) -> bool:
        """
        Delete a memory.

        Returns True when a memory was deleted.
        """

        with self._connect() as connection:
            cursor = connection.execute(
                """
                DELETE FROM memories
                WHERE memory_id = ?
                """,
                (memory_id,),
            )

        return cursor.rowcount > 0

    @staticmethod
    def _row_to_memory(
        row: sqlite3.Row,
    ) -> MemoryEntry:
        """
        Convert a SQLite row into a MemoryEntry.
        """

        return MemoryEntry(
            memory_id=row["memory_id"],
            user_id=row["user_id"],
            memory_type=MemoryType(
                row["memory_type"]
            ),
            content=row["content"],
            created_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["created_at"]
                )
            ),
            updated_at=(
                __import__("datetime")
                .datetime.fromisoformat(
                    row["updated_at"]
                )
            ),
            metadata=json.loads(
                row["metadata_json"]
            ),
        )

"""
EasyChatBot Hackathon — Podcast Discovery Agent.

The Agent converts explicit learner context and preference memory into
provider-neutral search intent.

It does not call Apple/Spotify, write memory, select a provider, or invent
podcast metadata.
"""

from __future__ import annotations

import re

from src.ai_runtime.memory.models import MemoryEntry, MemoryType
from src.ai_runtime.podcast.models import (
    LearningContext,
    PodcastSearchIntent,
)


class PodcastDiscoveryAgent:
    """Build provider-neutral podcast discovery intent."""

    _STOP_WORDS = {
        "a", "an", "and", "are", "be", "for", "from", "i", "in", "is",
        "like", "of", "on", "podcasts", "prefers", "prefer", "the", "to",
        "with", "you", "your",
    }

    def build_search_intent(
        self,
        *,
        user_id: str,
        learning_context: LearningContext,
        memories: list[MemoryEntry],
        max_results: int = 10,
    ) -> PodcastSearchIntent:
        """
        Convert explicit preferences + learning-world context into intent.

        Only PREFERENCE memories are used, matching the existing
        EasyChatBot discovery contract.
        """

        if not user_id.strip():
            raise ValueError("Podcast Agent user_id cannot be empty.")

        preferences = [
            memory
            for memory in memories
            if memory.user_id == user_id
            and memory.memory_type == MemoryType.PREFERENCE
        ]

        topics: list[str] = []
        keywords: list[str] = []

        for memory in preferences:
            topics.extend(
                self._metadata_terms(
                    memory.metadata,
                    "topics",
                    "interests",
                    "preferred_topics",
                )
            )
            keywords.extend(
                self._metadata_terms(
                    memory.metadata,
                    "keywords",
                )
            )

            if not topics and not keywords:
                keywords.extend(
                    self._content_keywords(memory.content)
                )

        # The learning world is always part of the search context.
        topics.append(learning_context.world_id)

        # Target language is deliberately a hard eligibility constraint,
        # not just another search keyword.
        return PodcastSearchIntent(
            user_id=user_id,
            learning_context=learning_context,
            topics=tuple(self._dedupe(topics)),
            keywords=tuple(self._dedupe(keywords)),
            language_required=True,
            max_results=max_results,
        )

    @staticmethod
    def _metadata_terms(
        metadata: dict[str, str],
        *keys: str,
    ) -> list[str]:
        terms: list[str] = []
        for key in keys:
            raw = metadata.get(key, "")
            if raw:
                terms.extend(
                    item.strip()
                    for item in re.split(r"[,;|]", raw)
                    if item.strip()
                )
        return terms

    @classmethod
    def _content_keywords(cls, content: str) -> list[str]:
        words = re.findall(
            r"[A-Za-zÀ-ÿ0-9][A-Za-zÀ-ÿ0-9'-]{2,}",
            content.lower(),
        )
        return [
            word for word in words
            if word not in cls._STOP_WORDS
        ][:8]

    @staticmethod
    def _dedupe(values: list[str]) -> list[str]:
        seen: set[str] = set()
        result: list[str] = []
        for value in values:
            key = value.strip().lower()
            if key and key not in seen:
                seen.add(key)
                result.append(value.strip())
        return result

                options=(

                    "Pour supprimer toutes les technologies",
                    "Pour empêcher les utilisateurs d'apprendre",
                ),
                correct_option=0,
            ),
        )

        introduction = (
            f"Écoutez le podcast en {language}, puis revenez ici pour "
            f"votre défi {level}. Faites attention aux mots clés et aux "
            "idées principales."
        )

        discussion_prompt = (
            "Après votre écoute, expliquez en quelques phrases votre "
            "opinion sur le sujet et utilisez au moins deux nouveaux mots."
        )

        return PodcastLearningChallenge.create(
            user_id=recommendation.user_id,
            podcast_id=recommendation.result.external_id,
            podcast_title=recommendation.result.title,
            podcast_url=recommendation.result.url,
            target_language=language,
            learner_level=level,
            vocabulary=vocabulary,
            comprehension_questions=questions,
            discussion_prompt=discussion_prompt,

        self,
        *,
        challenge: PodcastLearningChallenge,
        answers: tuple[int, ...],
    ) -> ChallengeResult:

        total = len(challenge.comprehension_questions)

        if len(answers) != total:
            raise ValueError(
                "Number of answers must equal the number of questions."
            )

        score = sum(
            answer == question.correct_option
            for answer, question in zip(
                answers,
                challenge.comprehension_questions,
            )
        )

        vocabulary_review_required = score < total

        if passed:
            message = (
                f"Challenge passed: {score}/{total}. "
                "Review the vocabulary and keep practising."
            )
        else:
            message = (
                f"Keep practising: {score}/{total}. "
                "Review the vocabulary and try again."
            )

        return ChallengeResult(
            challenge_id=challenge.challenge_id,
            user_id=challenge.user_id,
            score=score,
            total_questions=total,
            passed=passed,
            vocabulary_review_required=vocabulary_review_required,
            message=message,
        )

from src.ai_runtime.discovery.flash_idea import (
    FlashIdeaBuilder,
)

from src.ai_runtime.discovery.models import (
    CommerceLink,
    DealCard,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


def make_deal() -> DealCard:
    link = CommerceLink(
        vendor="Example Vendor",
        destination_url=(
            "https://example.com/product/backpack"
        ),
    )

    return DealCard(
        product_name="Minimalist Travel Backpack",
        vendor="Example Vendor",
        price=89.0,
        currency="EUR",
        previous_price=129.0,
        discount_percentage=31.0,
        commerce_link=link,
        source="parallel",
    )


def test_flash_idea_uses_user_preference():

    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.PREFERENCE,
        content="prefers minimalist travel products",
    )

    idea = FlashIdeaBuilder().build(
        character_id="yeti",
        deal=make_deal(),
        memories=[memory],
    )

    assert idea.character_id == "yeti"

    assert (
        idea.deal_id
        == make_deal().deal_id
        or idea.deal_id
    )

    assert (
        "minimalist travel products"
        in idea.message
    )

    assert (
        "Minimalist Travel Backpack"
        in idea.message
    )

    assert (
        "31% off"
        in idea.message
    )


def test_flash_idea_works_without_memory():

    idea = FlashIdeaBuilder().build(
        character_id="yeti",
        deal=make_deal(),
        memories=[],
    )

    assert (
        "I found something you might like"
        in idea.message
    )

    assert (
        "Minimalist Travel Backpack"
        in idea.message
    )


def test_flash_idea_contains_commerce_url():

    idea = FlashIdeaBuilder().build(
        character_id="yeti",
        deal=make_deal(),
        memories=[],
    )

    assert (
        idea.metadata["commerce_url"]
        == "https://example.com/product/backpack"
    )


def test_non_preference_memory_does_not_become_preference():

    memory = MemoryEntry(
        user_id="user-001",
        memory_type=MemoryType.DISCOVERY_CONTEXT,
        content="Looking for travel gear",
    )

    idea = FlashIdeaBuilder().build(
        character_id="yeti",
        deal=make_deal(),
        memories=[memory],
    )

    assert (
        "I found something you might like"
        in idea.message
    )

    """

                content="Previous conversation",
            ),

        self.recall_calls = []
        self.remember_calls = []

    def recall(
        self,
        user_id: str,

        self.recall_calls.append(

            for memory in self.memories
            if memory.user_id == user_id

        self,
        user_id: str,
        category: str,
        content: str,
    ) -> None:
        self.remember_calls.append(
            {
                "user_id": user_id,
                "category": category,
                "content": content,
            }
        )

@dataclass
class FakeMemory:
    """
    Deterministic memory record for tests.
    """

    user_id: str
    category: str
    content: str


class FakeMemoryService:
    """
    Deterministic MemoryService test double.
    """

    def __init__(self) -> None:
        self.memories = [
            FakeMemory(
                user_id="demo-user",
                category="conversation",
                content="Previous conversation",
            ),
        ]

        self.recall_calls = []
        self.remember_calls = []

    def recall(
        self,
        user_id: str,
    ):
        self.recall_calls.append(
            user_id
        )

        return [
            memory
            for memory in self.memories
            if memory.user_id == user_id
        ]

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> None:
        self.remember_calls.append(
            {
                "user_id": user_id,
                "category": category,
                "content": content,
            }
        )
