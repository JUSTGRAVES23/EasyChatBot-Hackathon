from __future__ import annotations

from datetime import datetime, timedelta

from .academy_notification_formatters import AcademyEmailFormatter, AcademyTelegramFormatter
from .academy_notification_ports import EmailNotificationSender, TelegramNotificationSender
from ..domain.academy_notification import (
    AcademyNotification,
    AcademyNotificationType,
)


class OneWeekNotificationOrchestrator:
    """Coordinates the 1-week milestone across Email and Telegram."""

    def __init__(
        self,
        *,
        email_sender: EmailNotificationSender,
        telegram_sender: TelegramNotificationSender,
        email_formatter: AcademyEmailFormatter,
        telegram_formatter: AcademyTelegramFormatter,
        registration_email: str,
        portal_url: str,
    ) -> None:
        if not registration_email.strip():
            raise ValueError("registration_email cannot be empty.")
        if not portal_url.strip():
            raise ValueError("portal_url cannot be empty.")

        self._email_sender = email_sender
        self._telegram_sender = telegram_sender
        self._email_formatter = email_formatter
        self._telegram_formatter = telegram_formatter
        self._registration_email = registration_email
        self._portal_url = portal_url

    @staticmethod
    def is_one_week_eligible(
        *,
        registered_at: datetime,
        now: datetime,
    ) -> bool:
        if registered_at.tzinfo is None or now.tzinfo is None:
            raise ValueError("timestamps must be timezone-aware.")
        age = now - registered_at
        return timedelta(days=7) <= age < timedelta(days=8)

    async def send_one_week(
        self,
        *,
        email: str,
        username: str,
        telegram_chat_id: str | None,
    ) -> dict:
        notification = AcademyNotification(
            notification_type=AcademyNotificationType.ONE_WEEK_MILESTONE,
            username=username,
            portal_url=self._portal_url,
        )

        rendered_email = self._email_formatter.render(notification)

        email_result = await self._email_sender.send(
            "🎓 One Week at EasyChatBot, "
            f"@{username.strip()}!",
            {
                "recipient": email,
                "email_type": "academy_notification",
                "notification_type": notification.notification_type.value,
                "from": self._registration_email,
                "body": rendered_email["body"],
                "html": rendered_email["html"],
                "attachments": rendered_email.get(
                    "attachments",
                    [],
                ),
            },
        )

        telegram_result = None
        if telegram_chat_id:
            telegram_result = await self._telegram_sender.send(
                chat_id=telegram_chat_id,
                text=self._telegram_formatter.render(notification),
                portal_url=self._portal_url,
            )

        return {
            "notification_type": notification.notification_type.value,
            "email": email_result,
            "telegram": telegram_result,
        }

class AcademyEmailNotificationSender:
    """Academy email adapter over the existing provider boundary.

    Application code depends on the Academy email port.
    Provider selection remains owned by EmailProviderManager.

    Does NOT:
    - create a second email provider;
    - access provider credentials;
    - send email directly through Resend;
    - make Academy eligibility decisions.
    """

    def __init__(
        self,
        *,
        provider_manager: EmailProviderManager | None = None,
    ) -> None:
        self._provider_manager = (
            provider_manager
            if provider_manager is not None
            else EmailProviderManager()
        )

    async def send(
        self,
        subject: str,
        content: dict[str, Any],
    ) -> dict[str, Any]:
        if not isinstance(subject, str) or not subject.strip():
            raise ValueError("Email subject cannot be empty.")

        if not isinstance(content, dict):
            raise TypeError("Email content must be a dictionary.")

        recipient = content.get("recipient")

        if not isinstance(recipient, str) or not recipient.strip():
            raise ValueError("Email recipient is required.")

        result = await self._provider_manager.send(
            subject.strip(),
            content,
        )

        if not isinstance(result, dict):
            raise TypeError(
                "Email provider returned an invalid result."
            )

        return result

from __future__ import annotations

from datetime import datetime, timedelta, timezone

from backend.app.application.academy_notification_service import (
    OneWeekNotificationOrchestrator,
)
from backend.app.application.academy.academy_notification_delivery import (
    AcademyDeliveryRepository,
    AcademyUserQuery,
)


class AcademyOneWeekWorkflow:
    """Execute the Academy 1-week workflow without owning scheduler state."""

    NOTIFICATION_TYPE = "1_WEEK_MILESTONE"

    def __init__(
        self,
        *,
        users: AcademyUserQuery,
        deliveries: AcademyDeliveryRepository,
        notifications: OneWeekNotificationOrchestrator,
        clock=None,
    ) -> None:
        self._users = users
        self._deliveries = deliveries
        self._notifications = notifications
        self._clock = clock or (lambda: datetime.now(timezone.utc))

    async def run(self) -> dict[str, int]:
        now = self._clock()
        if now.tzinfo is None:
            raise ValueError("Workflow clock must return a timezone-aware datetime.")

        window_start = now - timedelta(days=8)
        window_end = now - timedelta(days=7)

        candidates = await self._users.list_one_week_candidates(
            window_start=window_start,
            window_end=window_end,
        )

        attempted = 0
        succeeded = 0
        skipped = 0
        failed = 0

        for user in candidates:
            # The query is deliberately bounded by the same eligibility window,
            # while the domain rule remains the final guard.
            if not self._notifications.is_one_week_eligible(
                registered_at=user.created_at,
                now=now,
            ):
                skipped += 1
                continue

            claim = await self._deliveries.claim(
                user_id=user.user_id,
                notification_type=self.NOTIFICATION_TYPE,
                now=now,
            )

            if claim is None:
                skipped += 1
                continue

            attempted += 1

            try:
                result = await self._notifications.send_one_week(
                    email=user.email,
                    username=user.username,
                    telegram_chat_id=None,
                )
                email_result = result.get("email") or {}
                provider_message_id = email_result.get("message_id")

                await self._deliveries.mark_succeeded(
                    user_id=user.user_id,
                    notification_type=self.NOTIFICATION_TYPE,
                    provider_message_id=(
                        str(provider_message_id)
                        if provider_message_id is not None
                        else None
                    ),
                    now=now,
                )
                succeeded += 1
            except Exception as exc:
                await self._deliveries.mark_failed(
                    user_id=user.user_id,
                    notification_type=self.NOTIFICATION_TYPE,
                    error=str(exc)[:2000],
                    now=now,
                )
                failed += 1

        return {
            "candidates": len(candidates),
            "attempted": attempted,
            "succeeded": succeeded,
            "failed": failed,
            "skipped": skipped,
        }

from __future__ import annotations

from typing import Any


class ExistingBotTelegramNotificationSender:
    """Outbound adapter around the existing Telegram Bot instance.

    No Bot is created here. The composition root injects the existing
    `TelegramBotRuntime.bot`.
    """

    def __init__(self, bot: Any) -> None:
        if bot is None:
            raise ValueError("bot cannot be None.")
        self._bot = bot

    async def send(
        self,
        *,
        chat_id: str,
        text: str,
        portal_url: str,
    ) -> dict:
        if not chat_id.strip():
            raise ValueError("chat_id cannot be empty.")
        if not portal_url.strip():
            raise ValueError("portal_url cannot be empty.")

        from telegram import InlineKeyboardButton, InlineKeyboardMarkup

        markup = InlineKeyboardMarkup(
            [[InlineKeyboardButton("🚀 Continue Learning", url=portal_url)]]
        )

        message = await self._bot.send_message(
            chat_id=chat_id,
            text=text,
            parse_mode="Markdown",
            reply_markup=markup,
        )

        return {
            "provider": "telegram",
            "status": "accepted",
            "message_id": getattr(message, "message_id", None),
        }

from __future__ import annotations

import logging
import os
from typing import Any

from telegram import Update
from telegram.ext import (
    Application,
    CommandHandler,
    ContextTypes,
    MessageHandler,
    filters,
)

from backend.app.interfaces.telegram.beta_telegram_adapter import (
    BetaTelegramAdapter,
)


logger = logging.getLogger(__name__)


class TelegramBotRuntime:
    """
    Infrastructure adapter for python-telegram-bot.

    Responsibilities:
    - Telegram application lifecycle;
    - Telegram handler registration;
    - delegation to BetaTelegramAdapter;
    - exposing the already-created Telegram bot client to
      controlled infrastructure adapters.

    The runtime MUST NOT:
    - perform Beta authorization;
    - execute conversation logic;
    - deliver classroom responses returned by the adapter;
    - create a second Telegram client;
    - bypass the controlled publication boundary.

    IMPORTANT RESPONSE OWNERSHIP:

        Telegram update
              |
              v
        TelegramBotRuntime
              |
              v
        BetaTelegramAdapter
              |
              +----> reply_text()
              |
              v
          response

    The adapter owns Telegram classroom response delivery.

    Therefore the runtime MUST NOT call reply_text() with the
    response returned by the adapter.

    This prevents duplicate Telegram messages.

    PUBLICATION CLIENT OWNERSHIP:

        TelegramBotRuntime
              |
              +----> existing Application
                           |
                           +----> existing Bot client
                                      |
                                      v
                         Controlled publication adapters

    The runtime remains the owner of Telegram client creation
    and lifecycle. Consumers receive a reference to the existing
    client through the read-only ``bot`` property.
    """

    def __init__(
        self,
        *,
        adapter: BetaTelegramAdapter,
        conversation_boundary: Any = None,
    ) -> None:
        if adapter is None:
            raise ValueError(
                "adapter cannot be None."
            )

        self._adapter = adapter

        # Retained for constructor compatibility with existing
        # composition. Runtime does not execute the boundary.
        self._conversation_boundary = conversation_boundary

        self._application: Application | None = None
        self._running = False

    # =========================================================
    # Application construction
    # =========================================================

    def build(self) -> Application:
        token = os.getenv(
            "TELEGRAM_BOT_TOKEN"
        )

        if not token:
            raise RuntimeError(
                "TELEGRAM_BOT_TOKEN is missing."
            )

        application = (
            Application
            .builder()
            .token(token)
            .build()
        )

        application.add_handler(
            CommandHandler(
                "start",
                self._handle_start,
            )
        )

        application.add_handler(
            MessageHandler(
                filters.ChatType.PRIVATE
                & filters.TEXT
                & ~filters.COMMAND,
                self._handle_private_message,
            )
        )

        self._application = application

        logger.info(
            "Telegram bot runtime created."
        )

        return application

    # =========================================================
    # Existing Telegram bot client
    # =========================================================

    @property
    def bot(self) -> Any:
        """
        Return the existing Telegram bot client.

        The TelegramBotRuntime remains responsible for creating
        and owning the Telegram Application and its Bot client.

        This property does not create another client and does not
        expose credentials.

        Consumers may use this reference for controlled outbound
        operations such as the announcement publication adapter.

        Raises:
            RuntimeError: If the Telegram application has not yet
                been built.
        """

        if self._application is None:
            raise RuntimeError(
                "Telegram bot runtime has not been built."
            )

        return self._application.bot

    # =========================================================
    # Lifecycle
    # =========================================================

    async def start(self) -> None:
        if self._running:
            logger.warning(
                "Telegram runtime already running."
            )
            return

        if self._application is None:
            self.build()

        assert self._application is not None

        await self._application.initialize()
        await self._application.start()

        if self._application.updater is not None:
            await self._application.updater.start_polling(
                drop_pending_updates=False,
            )

        self._running = True

        logger.info(
            "Telegram bot polling started."
        )

    async def stop(self) -> None:
        if self._application is None:
            return

        if (
            self._application.updater is not None
            and self._running
        ):
            await self._application.updater.stop()

        await self._application.stop()
        await self._application.shutdown()

        self._running = False

        logger.info(
            "Telegram bot runtime stopped."
        )

    # =========================================================
    # Telegram handlers
    # =========================================================

    async def _handle_start(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        await self._adapter.handle_start(
            update,
            context,
        )

    async def _handle_private_message(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        if (
            update.effective_user is None
            or update.effective_chat is None
            or update.effective_message is None
        ):
            return

        try:
            # =================================================
            # SINGLE DELEGATION POINT
            # =================================================
            #
            # BetaTelegramAdapter:
            #
            #   1. validates transport
            #   2. authorizes access
            #   3. invokes conversation boundary
            #   4. sends Telegram response
            #   5. returns response for observation/testing
            #
            # The returned value MUST NOT be sent again.
            #

            await self._adapter.handle_message(
                update,
                context,
            )

        except Exception:
            logger.exception(
                "Unexpected Telegram adapter failure."
            )

            # This is ONLY an infrastructure-level fallback for
            # failures escaping the adapter itself.
            #
            # Normal conversation failures are already contained
            # by BetaTelegramAdapter.
            await update.effective_message.reply_text(
                "EasyChatBot is temporarily unavailable. "
                "Please try again."
            )

    "TelegramBotRuntime",

"""
Email Provider Manager.

Controls provider selection, configuration, availability,
and controlled email delivery.

The manager is deliberately provider-independent.
Concrete providers such as Resend are injected into this
boundary rather than called directly by application services.

Does NOT:
- generate cybersecurity reports;
- schedule emails;
- analyse security;
- execute remediation;
- expose provider credentials.
"""

from __future__ import annotations

from typing import Any

from backend.cybersecurity.email.resend_email_provider import (
    ResendEmailProvider,
)


class EmailProviderManager:
    """
    Provider abstraction for controlled email delivery.
    """

    def __init__(
        self,
        provider: Any | None = None,
    ) -> None:

        self.provider = (
            provider
            if provider is not None
            else self._build_default_provider()
        )

    @staticmethod
    def _build_default_provider() -> Any | None:
        """
        Build the configured default email provider.

        Resend is currently the approved hackathon provider.

        If Resend is not configured, return None so the
        application can report provider_not_configured
        rather than failing during application import.
        """

        try:
            return ResendEmailProvider()

        except ValueError:
            return None

    def status(self) -> dict[str, Any]:
        """
        Return non-sensitive provider status.

        This method never exposes:
        - API keys;
        - secrets;
        - credentials.

        Status is determined from the provider already
        configured in the manager. No external provider
        request is made.
        """

        if self.provider is None:

            return {
                "provider": None,
                "configured": False,
                "available": False,
                "status": "not_configured",
            }

        provider_id = getattr(
            self.provider,
            "provider_id",
            self.provider.__class__.__name__,
        )

        return {
            "provider": provider_id,
            "configured": True,
            "available": True,
            "status": "ready",
        }

    async def send(
        self,
        subject: str,
        content: dict,
    ) -> dict:

        if self.provider is None:

            return {
                "status": "provider_not_configured",
            }

        return await self.provider.send(
            subject,
            content,
        )

class CybersecurityAPSchedulerAdapter:
    """APScheduler implementation boundary.

    All scheduled application workflows share this single scheduler.

    Optional workflows remain dependency-injected so existing
    cybersecurity-only tests and deployments remain backwards compatible.
    """

    def __init__(
        self,
        scheduler,
        workflow_service,
        governance_reminder_workflow=None,
        governance_reminder_interval_minutes: int = 60,
        academy_one_week_workflow=None,
        academy_one_week_hour: int = 8,
        academy_one_week_minute: int = 15,
    ) -> None:
        if scheduler is None:
            raise ValueError("scheduler cannot be None.")

        if workflow_service is None:
            raise ValueError("workflow_service cannot be None.")

        if governance_reminder_interval_minutes <= 0:
            raise ValueError(
                "governance_reminder_interval_minutes must be > 0."
            )

        if not 0 <= academy_one_week_hour <= 23:
            raise ValueError(
                "academy_one_week_hour must be between 0 and 23."
            )

        if not 0 <= academy_one_week_minute <= 59:
            raise ValueError(
                "academy_one_week_minute must be between 0 and 59."
            )

        self.scheduler = scheduler
        self.workflow_service = workflow_service

        self.governance_reminder_workflow = (
            governance_reminder_workflow
        )
        self.governance_reminder_interval_minutes = (
            governance_reminder_interval_minutes
        )

        self.academy_one_week_workflow = (
            academy_one_week_workflow
        )
        self.academy_one_week_hour = academy_one_week_hour
        self.academy_one_week_minute = academy_one_week_minute

    def register_jobs(self) -> None:
        """Register all enabled workflows on the single scheduler."""

        self.scheduler.add_job(
            self.workflow_service.create_daily_report,
            trigger="cron",
            hour=DAILY_REPORT_HOUR,
            minute=DAILY_REPORT_MINUTE,
            id="cybersecurity_daily_report",
            replace_existing=True,
        )

        self.scheduler.add_job(
            self.workflow_service.create_weekly_report,
            trigger="cron",
            day_of_week=WEEKLY_REPORT_DAY,
            hour=WEEKLY_REPORT_HOUR,
            minute=WEEKLY_REPORT_MINUTE,
            id="cybersecurity_weekly_report",
            replace_existing=True,
        )

        if self.governance_reminder_workflow is not None:
            self.scheduler.add_job(
                self.governance_reminder_workflow.run,
                trigger="interval",
                minutes=self.governance_reminder_interval_minutes,
                id="privacy_governance_reminder",
                replace_existing=True,
            )

        if self.academy_one_week_workflow is not None:
            self.scheduler.add_job(
                self.academy_one_week_workflow.run,
                trigger="cron",
                hour=self.academy_one_week_hour,
                minute=self.academy_one_week_minute,
                id="academy_one_week_milestone",
                replace_existing=True,
            )

    def register_critical_handler(self, handler) -> None:
        self.critical_handler = handler

    async def trigger_critical_event(self, event: dict):
        if not hasattr(self, "critical_handler"):
            return None

        return await self.critical_handler(event)

    def start(self) -> None:
        if not self.scheduler.running:
            self.scheduler.start()

    def shutdown(self) -> None:
        if self.scheduler.running:
            self.scheduler.shutdown()

from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker


class Base(DeclarativeBase):
    """Canonical SQLAlchemy metadata registry for the platform."""

    pass


def database_url() -> str:
    """Return the canonical PostgreSQL connection URL."""

    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for PostgreSQL persistence."
        )

    return value


def create_database_engine():
    """Create the existing synchronous PostgreSQL engine.

    This API is preserved for the existing Beta 1 persistence layer.
    """

    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(
    engine,
) -> sessionmaker[Session]:
    """Create the existing synchronous session factory.

    Preserved for existing Beta 1 repositories.
    """

    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )


def async_database_url() -> str:
    """Return DATABASE_URL adapted for SQLAlchemy async PostgreSQL.

    The environment remains the single source of truth. We only adapt the
    driver at the infrastructure boundary when the configured URL uses the
    standard PostgreSQL SQLAlchemy scheme.
    """

    value = database_url()

    if value.startswith("postgresql+asyncpg://"):
        return value

    if value.startswith("postgresql://"):
        return value.replace(
            "postgresql://",
            "postgresql+asyncpg://",
            1,
        )

    raise RuntimeError(
        "DATABASE_URL must use a PostgreSQL URL compatible with "
        "the synchronous or async PostgreSQL persistence boundary."
    )


def create_async_database_engine() -> AsyncEngine:
    """Create the canonical asynchronous PostgreSQL engine.

    This is used by durable WS-22 persistence while sharing the same
    DATABASE_URL and canonical Base as the existing persistence layer.
    """

    return create_async_engine(
        async_database_url(),
        pool_pre_ping=True,
    )


def create_async_session_factory(
    engine: AsyncEngine,
) -> async_sessionmaker[AsyncSession]:
    """Create the canonical asynchronous PostgreSQL session factory."""

    return async_sessionmaker(
        bind=engine,
        class_=AsyncSession,
        expire_on_commit=False,
    )

    - Telegram application lifecycle;
    - Telegram handler registration;
    - delegation to BetaTelegramAdapter;
    - exposing the already-created Telegram bot client to
      controlled infrastructure adapters.

    The runtime MUST NOT:
    - perform Beta authorization;
    - execute conversation logic;
    - deliver classroom responses returned by the adapter;
    - create a second Telegram client;
    - bypass the controlled publication boundary.

    IMPORTANT RESPONSE OWNERSHIP:

        Telegram update
              |
              v
        TelegramBotRuntime
              |
              v
        BetaTelegramAdapter
              |
              +----> reply_text()
              |
              v
          response

    The adapter owns Telegram classroom response delivery.

    Therefore the runtime MUST NOT call reply_text() with the
    response returned by the adapter.

    This prevents duplicate Telegram messages.

    PUBLICATION CLIENT OWNERSHIP:

        TelegramBotRuntime
              |
              +----> existing Application
                           |
                           +----> existing Bot client
                                      |
                                      v
                         Controlled publication adapters

    The runtime remains the owner of Telegram client creation
    and lifecycle. Consumers receive a reference to the existing
    client through the read-only ``bot`` property.
    """

    def __init__(
        self,
        *,
        adapter: BetaTelegramAdapter,
        conversation_boundary: Any = None,
    ) -> None:
        if adapter is None:
            raise ValueError(
                "adapter cannot be None."
            )

        self._adapter = adapter

        # Retained for constructor compatibility with existing
        # composition. Runtime does not execute the boundary.
        self._conversation_boundary = conversation_boundary

        self._application: Application | None = None
        self._running = False

    # =========================================================
    # Application construction
    # =========================================================

    def build(self) -> Application:
        token = os.getenv(
            "TELEGRAM_BOT_TOKEN"
        )

        if not token:
            raise RuntimeError(
                "TELEGRAM_BOT_TOKEN is missing."
            )

        application = (
            Application
            .builder()
            .token(token)
            .build()
        )

        application.add_handler(
            CommandHandler(
                "start",
                self._handle_start,
            )
        )

        application.add_handler(
            MessageHandler(
                filters.ChatType.PRIVATE
                & filters.TEXT
                & ~filters.COMMAND,
                self._handle_private_message,
            )
        )

        self._application = application

        logger.info(
            "Telegram bot runtime created."
        )

        return application

    # =========================================================
    # Existing Telegram bot client
    # =========================================================

    @property
    def bot(self) -> Any:
        """
        Return the existing Telegram bot client.

        The TelegramBotRuntime remains responsible for creating
        and owning the Telegram Application and its Bot client.

        This property does not create another client and does not
        expose credentials.

        Consumers may use this reference for controlled outbound
        operations such as the announcement publication adapter.

        Raises:
            RuntimeError: If the Telegram application has not yet
                been built.
        """

        if self._application is None:
            raise RuntimeError(
                "Telegram bot runtime has not been built."
            )

        return self._application.bot

    # =========================================================
    # Lifecycle
    # =========================================================

    async def start(self) -> None:
        if self._running:
            logger.warning(
                "Telegram runtime already running."
            )
            return

        if self._application is None:
            self.build()

        assert self._application is not None

        await self._application.initialize()
        await self._application.start()

        if self._application.updater is not None:
            await self._application.updater.start_polling(
                drop_pending_updates=False,
            )

        self._running = True

        logger.info(
            "Telegram bot polling started."
        )

    async def stop(self) -> None:
        if self._application is None:
            return

        if (
            self._application.updater is not None
            and self._running
        ):
            await self._application.updater.stop()

        await self._application.stop()
        await self._application.shutdown()

        self._running = False

        logger.info(
            "Telegram bot runtime stopped."
        )

    # =========================================================
    # Telegram handlers
    # =========================================================

    async def _handle_start(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        await self._adapter.handle_start(
            update,
            context,
        )

    async def _handle_private_message(
        self,
        update: Update,
        context: ContextTypes.DEFAULT_TYPE,
    ) -> None:
        if (
            update.effective_user is None
            or update.effective_chat is None
            or update.effective_message is None
        ):
            return

        try:
            # =================================================
            # SINGLE DELEGATION POINT
            # =================================================
            #
            # BetaTelegramAdapter:
            #
            #   1. validates transport
            #   2. authorizes access
            #   3. invokes conversation boundary
            #   4. sends Telegram response
            #   5. returns response for observation/testing
            #
            # The returned value MUST NOT be sent again.
            #

            await self._adapter.handle_message(
                update,
                context,
            )

        except Exception:
            logger.exception(
                "Unexpected Telegram adapter failure."
            )

            # This is ONLY an infrastructure-level fallback for
            # failures escaping the adapter itself.
            #
            # Normal conversation failures are already contained
            # by BetaTelegramAdapter.
            await update.effective_message.reply_text(
                "EasyChatBot is temporarily unavailable. "
                "Please try again."
            )
