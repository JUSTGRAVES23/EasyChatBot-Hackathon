"""
EasyChatBot API Application.

Hosts the user interaction layer.

The API application is responsible only for:

- creating the FastAPI application;
- registering API routes;
- exposing the application health endpoint;
- triggering application composition bootstrap during startup;
- allowing the local frontend to communicate with the API.

Business logic belongs to application services.

Agent execution belongs to the agent/runtime layer.

Provider construction belongs to the composition layer.
"""

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    """
    Manage application startup and shutdown lifecycle.

    Startup:
        - compose application dependencies;
        - start Telegram transport runtime.

    Shutdown:
        - stop Telegram transport runtime.

    This layer does not:

    - execute agents;
    - execute capabilities;
    - create providers;
    - create Runtime components.
    """

    bootstrap_application()

    await start_telegram_runtime()

    try:
        yield

    finally:
        await stop_telegram_runtime()


def create_app() -> FastAPI:
    """
    Create the EasyChatBot FastAPI application.

    Application composition is intentionally separated
    from the HTTP layer.

    This function:

    - creates the FastAPI application;
    - configures local frontend access;
    - registers API routes;
    - attaches lifecycle management.

    It does not:

    - execute agents;
    - execute capabilities;
    - call providers directly;
    - create Runtime executors.
    """

    application = FastAPI(
        title="EasyChatBot",
        description="AI companion learning platform",
        lifespan=lifespan,
    )

    # =========================================================
    # Local Frontend CORS
    # =========================================================

    application.add_middleware(
        CORSMiddleware,
        allow_origins=[
            "http://127.0.0.1:5500",
            "http://localhost:5500",
            "http://127.0.0.1:5501",
            "http://localhost:5501",
            "http://localhost:3000",
            "http://127.0.0.1:3000",
        ],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    # =========================================================
    # Existing API Routes
    # =========================================================

    application.include_router(
        router,
    )

    # =========================================================
    # Character Theme Routes
    # =========================================================

    application.include_router(
        character_theme_router,
    )

    # =========================================================
    # Character Catalogue Routes
    # =========================================================

    application.include_router(
        character_router,
    )

    # =========================================================
    # AI Operations Center Routes
    # =========================================================

    application.include_router(
        operations_center_dashboard_router,
    )

    # =========================================================
    # Cybersecurity Intelligence Routes
    # =========================================================

    application.include_router(
        cybersecurity_intelligence_router,
    )

    return application


app = create_app()


@app.get("/")
def health() -> dict[str, str]:
    """
    Return application health status.
    """

    return {
        "status": "EasyChatBot running",
        "architecture": (
            "Runtime + Agents + Memory"
        ),
    }

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    """
    Application lifecycle.

    Starts:
    - cybersecurity scheduler runtime.

    Stops:
    - cybersecurity scheduler runtime.

    No business logic belongs here.
    """

    try:
        await scheduler_runtime.start()

    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()

    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

@dataclass(frozen=True)
class BetaTelegramComposition:
    """
    Fully composed Beta Telegram classroom application graph.

    This object contains application/infrastructure components only.
    It does not start Telegram polling.
    """

    beta_access_service: BetaAccessService
    conversation_runtime: ConversationRuntime
    conversation_service: TelegramConversationService
    conversation_boundary: BetaTelegramConversationBoundary
    private_chat_gate: BetaPrivateChatGate
    start_handler: BetaTelegramStartHandler
    adapter: BetaTelegramAdapter
    runtime: TelegramBotRuntime


def create_beta_telegram_composition(
    *,
    user_repository: Any,
    token_repository: Any,
    binding_repository: Any,
    conversation_event_repository: Any,
    privacy_identifier_service: Any,
    conversation_event_validator: Any,
    expiration_minutes: int = 1440,
) -> BetaTelegramComposition:
    """
    Compose the production Beta Telegram classroom graph.

    Dependency direction:

        Infrastructure
              â†“
        Application services
              â†“
        AI Runtime
              â†“
        Provider abstraction

    No Telegram component creates an LLM provider directly.
    """

    beta_access_service = BetaAccessService(
        user_repository=user_repository,
        token_repository=token_repository,
        binding_repository=binding_repository,
        expiration_minutes=expiration_minutes,
    )

    conversation_runtime = ConversationRuntime(
        llm_provider=LLMProviderFactory.create(),
    )

    conversation_service = TelegramConversationService(
        conversation_runtime=conversation_runtime,
        conversation_event_repository=(
            conversation_event_repository
        ),
        privacy_identifier_service=(
            privacy_identifier_service
        ),
        conversation_event_validator=(
            conversation_event_validator
        ),
    )

    conversation_boundary = (
        BetaTelegramConversationBoundary(
            conversation_service=conversation_service,
            beta_access_service=beta_access_service,
        )
    )

    private_chat_gate = BetaPrivateChatGate(
        beta_access_service=beta_access_service,
    )

    start_handler = BetaTelegramStartHandler(
        beta_access_service=beta_access_service,
    )

    adapter = BetaTelegramAdapter(
        start_handler=start_handler,
        private_chat_gate=private_chat_gate,
        conversation_boundary=conversation_boundary,
    )

    runtime = TelegramBotRuntime(
        adapter=adapter,
        conversation_boundary=conversation_boundary,
    )

    return BetaTelegramComposition(
        beta_access_service=beta_access_service,
        conversation_runtime=conversation_runtime,
        conversation_service=conversation_service,
        conversation_boundary=conversation_boundary,
        private_chat_gate=private_chat_gate,
        start_handler=start_handler,
        adapter=adapter,
        runtime=runtime,
    )


__all__ = [
    "BetaTelegramComposition",
    "create_beta_telegram_composition",
]

@dataclass(frozen=True)
class BetaTelegramProductionComposition:
    """Complete production dependency graph for Beta Telegram."""

    database_engine: Any
    session_factory: Any
    beta_user_repository: PostgreSQLBetaUserRepository
    beta_token_repository: PostgreSQLBetaTokenRepository
    telegram_binding_repository: PostgreSQLTelegramBindingRepository
    conversation_event_repository: PostgreSQLConversationEventRepository
    privacy_identifier_service: PrivacyIdentifierService
    conversation_event_validator: ConversationEventValidator
    beta_telegram: BetaTelegramComposition


def create_production_beta_telegram_composition(
    *,
    expiration_minutes: int = 1440,
) -> BetaTelegramProductionComposition:
    """Build the production Beta Telegram dependency graph.

    The function creates dependencies only; it does not start
    Telegram polling or perform external side effects.
    """
    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    database_engine = create_database_engine()
    session_factory = create_session_factory(database_engine)

    beta_user_repository = PostgreSQLBetaUserRepository(
        session_factory,
    )
    beta_token_repository = PostgreSQLBetaTokenRepository(
        session_factory,
    )
    telegram_binding_repository = PostgreSQLTelegramBindingRepository(
        session_factory,
    )
    conversation_event_repository = PostgreSQLConversationEventRepository(
        session_factory,
    )

    privacy_identifier_service = PrivacyIdentifierService()
    conversation_event_validator = ConversationEventValidator()

    beta_telegram = create_beta_telegram_composition(
        user_repository=beta_user_repository,
        token_repository=beta_token_repository,
        binding_repository=telegram_binding_repository,
        conversation_event_repository=conversation_event_repository,
        privacy_identifier_service=privacy_identifier_service,
        conversation_event_validator=conversation_event_validator,
        expiration_minutes=expiration_minutes,
    )

    return BetaTelegramProductionComposition(
        database_engine=database_engine,
        session_factory=session_factory,
        beta_user_repository=beta_user_repository,
        beta_token_repository=beta_token_repository,
        telegram_binding_repository=telegram_binding_repository,
        conversation_event_repository=conversation_event_repository,
        privacy_identifier_service=privacy_identifier_service,
        conversation_event_validator=conversation_event_validator,
        beta_telegram=beta_telegram,
    )


__all__ = [
