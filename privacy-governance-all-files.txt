function formatDate(value?: string | null): string {
  if (!value) {
    return "â€”";
  }

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return new Intl.DateTimeFormat(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(date);
}

function statusLabel(value?: string | null): string {
  if (!value) {
    return "Not yet decided";
  }

  return value.replaceAll("_", " ");
}

export function PrivacyGovernanceCaseDetail({

    caseData?.approval_history ??
    caseData?.approval_records ??
    [];

    caseData?.processing_ids ?? [];

    caseData?.data_map_ids ?? [];

  | "ALL"

function statusLabel(
  value?: string | null,
): string {
  if (!value) {
    return "Unknown";
  }

  return value.replaceAll("_", " ");
}

function formatDate(
  value?: string | null,
): string {
  if (!value) {
    return "â€”";
  }

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return new Intl.DateTimeFormat(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(date);
}

function matchesFilter(
  caseItem: PrivacyGovernanceCase,
  filter: CaseFilter,
): boolean {
  if (filter === "ALL") {
    return true;
  }

  return (
    caseItem.governance_status ===
    filter
  );
}

function normalizeStatusChart(
  value: PrivacyGovernanceStatusItem[] | null,
): PrivacyGovernanceStatusItem[] {
  if (!value) {
    return [];
  }

  return value.map((item) => ({
    status: item.status,
    count: Number(item.count ?? 0),
  }));
}

function createApprovalId(
  caseId: string,
  role: GovernanceRole,
): string {
  const suffix =
    typeof crypto !== "undefined" &&
    "randomUUID" in crypto
      ? crypto.randomUUID()
      : `${Date.now()}-${Math.random()
          .toString(36)
          .slice(2, 10)}`;

  return `APR-UI-${role}-${caseId}-${suffix}`;
}

function statusLabel(
  value?: string | null,
): string {
  if (!value) {
    return "Unknown";
  }

  return value.replaceAll("_", " ");
}

  decision: GovernanceDecision,
): string {
  switch (decision) {
    case "approve":
      return "Approve";

    case "reject":
      return "Reject";

    case "request-changes":
      return "Request changes";
  }
}

async function requestJson<T>(
  url: string,
): Promise<T> {
  const response = await fetch(url, {
    headers: {
      Accept: "application/json",
    },
  });

  if (!response.ok) {
    throw new Error(
      `Privacy governance API request failed (${response.status}).`,
    );
  }

  return response.json() as Promise<T>;
}

export function getPrivacyGovernanceSummary(): Promise<PrivacyGovernanceSummary> {

  );
}

import type {
  PrivacyGovernanceCaseDetail,
} from "./types";

export type GovernanceDecision = "approve" | "reject" | "request-changes";
export type GovernanceRole = "DPO" | "GRC";

export interface GovernanceDecisionRequest {
  change_id: string;
  approval_id: string;
  reviewer: string;
  role: GovernanceRole;
  comment: string;
}

export interface GovernanceDecisionResponse {
  case: PrivacyGovernanceCaseDetail;
  approval_record: Record<string, unknown>;
}

const API_BASE = "/api/privacy/governance";

export async function submitGovernanceDecision(
  caseId: string,
  decision: GovernanceDecision,
  request: GovernanceDecisionRequest,
  signal?: AbortSignal,
): Promise<GovernanceDecisionResponse> {
  const response = await fetch(
    `${API_BASE}/cases/${encodeURIComponent(caseId)}/${decision}`,
    {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify(request),
      signal,
    },
  );

  if (!response.ok) {
    let message = `Governance decision failed (${response.status}).`;

    try {
      const payload = (await response.json()) as { detail?: string };
      if (payload.detail) {
        message = payload.detail;
      }
    } catch {
      // Keep the stable HTTP error message when the body is not JSON.
    }

    throw new Error(message);
  }

  return (await response.json()) as GovernanceDecisionResponse;
}
