==================== BASE PROVIDER ====================

"""
Base Provider Contract.

Defines the interface for external AI and knowledge providers.

Examples:
- Gemini
- Parallel Search
- Future LLM providers

The Runtime communicates with providers through this contract.

Providers are replaceable implementations.
The Agent layer must remain independent from provider details.
"""

from abc import ABC, abstractmethod

from backend.providers.provider_result import ProviderResult
from backend.runtime.context.execution_context import ExecutionContext


class BaseProvider(ABC):
    """
    Abstract base class for EasyChatBot providers.

    Providers are responsible for communicating
    with external services.

    They are NOT responsible for:
    - agent decisions
    - workflow orchestration
    - memory management
    - user interaction
    """

    provider_id: str

    def __init__(
        self,
        provider_id: str,
    ):
        """
        Initialise provider.

        Args:
            provider_id:
                Unique provider identifier.
        """

        self.provider_id = provider_id

    @abstractmethod
    async def execute(
        self,
        context: ExecutionContext,
    ) -> ProviderResult:
        """
        Execute provider operation.

        Args:
            context:
                Runtime execution context.

        Returns:
            ProviderResult containing provider output.
        """

        pass


==================== PROVIDER RESULT ====================

"""
Provider Result Contract.

Defines the standard response returned
by external providers.

All providers return this structure so
the Runtime can process results consistently.
"""

from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class ProviderResult:
    """
    Immutable provider execution result.

    Attributes:

        success:
            Indicates whether provider execution succeeded.

        provider_id:
            Identifier of the provider.

        output:
            Provider response payload.

        error:
            Error details if execution failed.
    """

    success: bool
    provider_id: str
    output: Any = None
    error: str | None = None


==================== PARALLEL CLIENT ====================

"""
Parallel Search API Client Wrapper.

This module isolates the official Parallel SDK.

The rest of EasyChatBot does not directly depend on
the Parallel SDK implementation.

Architecture:

Research Agent
        |
        v
Parallel Provider
        |
        v
Parallel Client Wrapper
        |
        v
Parallel API
"""

import asyncio
import os
from typing import Any

from dotenv import load_dotenv


load_dotenv()


class ParallelClientWrapper:
    """
    Wrapper around the official Parallel Search SDK.

    Responsibilities:
    - authenticate with Parallel
    - execute searches
    - normalize SDK responses

    Does not:
    - create learning content
    - manage agents
    - decide workflows
    """

    def __init__(
        self,
        api_key: str | None = None,
    ):
        """
        Initialise the Parallel SDK client.
        """

        self.api_key = (
            api_key
            or os.getenv("PARALLEL_API_KEY")
        )

        if not self.api_key:
            raise ValueError(
                "PARALLEL_API_KEY environment variable is required"
            )

        self._client = self._create_client()


    def _create_client(self):
        """
        Create the official Parallel SDK client.

        SDK:
            parallel-web 1.1.0
        """

        from parallel import Parallel

        return Parallel(
            api_key=self.api_key
        )


    @staticmethod
    def _normalize_response(
        response: Any,
    ) -> Any:
        """
        Convert the Parallel SDK response into
        application-safe data.

        Pydantic SDK models are converted to dictionaries.
        Other JSON-compatible values are returned unchanged.
        """

        if hasattr(response, "model_dump"):
            return response.model_dump()

        if hasattr(response, "dict"):
            return response.dict()

        return response


    async def search(
        self,
        query: str,
    ) -> dict[str, Any]:
        """
        Execute a Parallel web search.

        Args:
            query:
                Natural-language research objective.

        Returns:
            Normalized application-level search result.
        """

        search_query = self._build_search_query(
            query
        )

        response = await asyncio.to_thread(
            self._client.search,
            search_queries=[
                search_query
            ],
            mode="advanced",
            objective=query,
        )

        return {
            "query": query,
            "search_queries": [
                search_query
            ],
            "results": self._normalize_response(
                response
            ),
        }


    @staticmethod
    def _build_search_query(
        query: str,
    ) -> str:
        """
        Build a concise search query from the
        user's research request.

        The full request remains the Parallel
        objective while the search query is kept
        concise.
        """

        words = query.strip().split()

        if len(words) <= 6:
            return query.strip()

        return " ".join(words[:6])


==================== PARALLEL PROVIDER ====================

"""
Parallel Search Provider.

Provides live world information
through the Parallel Search API.
"""

from backend.providers.base_provider import (
    BaseProvider,
)

from backend.providers.provider_result import (
    ProviderResult,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class ParallelSearchProvider(BaseProvider):
    """
    Provider implementation using Parallel Search.

    The SDK client is created only when required.
    This allows tests to run without API credentials.
    """

    def __init__(
        self,
        client=None,
    ):
        super().__init__(
            provider_id="parallel_search"
        )

        self.client = client


    def _get_client(self):
        """
        Lazy-load the real Parallel client.

        Avoids requiring API keys during unit tests.
        """

        if self.client is None:

            from backend.providers.parallel.parallel_client import (
                ParallelClientWrapper,
            )

            self.client = ParallelClientWrapper()

        return self.client


    async def execute(
        self,
        context: ExecutionContext,
    ) -> ProviderResult:
        """
        Execute a live Parallel search.
        """

        try:

            client = self._get_client()

            result = await client.search(
                query=context.request
            )

            return ProviderResult(
                success=True,
                provider_id=self.provider_id,
                output=result,
            )

        except Exception as exc:

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error=str(exc),
            )


==================== RUNTIME ENGINE ====================

"""
EasyChatBot Runtime Engine.

The Runtime is responsible for orchestrating
agent execution.

Responsibilities:
- receive execution requests
- provide execution context
- call approved agents
- return structured results

The Runtime does not:
- implement business logic
- call providers directly
- manage user interfaces
"""

from backend.agents.base_agent import BaseAgent
from backend.agents.agent_result import AgentResult
from backend.runtime.context.execution_context import ExecutionContext


class RuntimeEngine:
    """
    Central execution orchestrator.

    Agents execute through the Runtime.
    """

    async def execute(
        self,
        agent: BaseAgent,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute an agent using runtime control.

        Args:
            agent:
                Agent implementation.

            context:
                Runtime execution context.

        Returns:
            AgentResult.
        """

        return await agent.execute(context)


==================== API ROUTES ====================

"""
EasyChatBot API Routes.

The API layer exposes the application.

Routes do not contain agent logic.
They delegate execution to application services.
"""

from fastapi import APIRouter, HTTPException

from backend.characters.character_registry import (
    CharacterRegistry,
)

from backend.services.companion_service import (
    CompanionService,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


router = APIRouter()

character_registry = CharacterRegistry()


def create_companion_service(
    character,
) -> CompanionService:
    """
    Create the application-level CompanionService.

    Kept as a small dependency seam so API tests can
    replace the service without invoking external providers.

    Production behaviour remains the same.
    """

    return CompanionService(
        character=character
    )


@router.get("/characters")
def get_characters():
    """
    Return available AI companions.
    """

    characters = (
        character_registry.list_characters()
    )

    return {
        "characters": [
            {
                "id": character.character_id,
                "name": character.name,
                "description": character.description,
                "personality": character.personality,
                "learning_style": character.learning_style,
            }
            for character in characters
        ]
    }


@router.post("/chat")
async def chat(
    payload: dict,
):
    """
    Execute an EasyChatBot companion interaction.

    Expected payload:

    {
        "user_id": "demo-user",
        "character": "yeti",
        "message": "Explain AI",
        "research": false
    }
    """

    message = payload.get("message")

    if not message:
        raise HTTPException(
            status_code=400,
            detail="message is required",
        )

    character_id = payload.get(
        "character",
        "yeti",
    )

    try:
        character = character_registry.get(
            character_id
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    companion_service = (
        create_companion_service(
            character
        )
    )

    context = ExecutionContext(
        user_id=payload.get(
            "user_id",
            "demo-user",
        ),
        request=message,
    )

    result = await companion_service.interact(
        context,
        use_research=payload.get(
            "research",
            False,
        ),
    )

    return result
