export type GovernanceStatus = {
  yeti: {
    message: string;
  };
  controls: Array<{
    component: string;
    status: string;
  }>;
  recent_events: Array<{
    event: string;
    actor: string;
    reason: string;
    component: string;
    timestamp: string;
  }>;
};

export async function getGovernanceStatus()
: Promise<GovernanceStatus> {

  const response = await fetch(
    "/operations/governance/status"
  );

  if (!response.ok) {
    throw new Error(
      "Unable to load governance status"
    );
  }

  return response.json();
}

export async function getGovernanceTimeline() {

    const response = await fetch(
        "/governance/timeline"
    );

    if (!response.ok) {
        throw new Error(
            "Unable to load governance timeline"
        );
    }

    return response.json();
}

import { useEffect, useState } from "react";

import {
  getGovernanceStatus,
  GovernanceStatus,
} from "../services/governanceApi";


export function useGovernanceStatus() {

  const [data, setData] =
    useState<GovernanceStatus | null>(null);

  const [loading, setLoading] =
    useState(true);

  const [error, setError] =
    useState<string | null>(null);


  useEffect(() => {

    getGovernanceStatus()
      .then(setData)
      .catch(
        (error) =>
          setError(error.message)
      )
      .finally(
        () => setLoading(false)
      );

  }, []);


  return {
    data,
    loading,
    error,
  };
}
