"""
Voice Call Application Service.

Application boundary for voice calls.

Responsibilities:
- Receive voice call requests.
- Delegate preparation workflow.
- Return prepared call information.

Does NOT:
- Execute speech.
- Manage providers.
- Manage HTTP.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.voice.voice_call_orchestrator import (
    VoiceCallOrchestrator,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallService:
    """
    Voice call application service.
    """

    orchestrator: VoiceCallOrchestrator



    def prepare_call(
        self,
        *,
        request: VoiceCallRequest,
    ) -> dict:
        """
        Prepare voice call.
        """


        session = (
            self.orchestrator
            .prepare_call(
                request=request
            )
        )


        return {
            "room_name": (
                session.session_id
            ),
            "character_id": (
                session.character_id
            ),
            "language_code": (
                session.language_code
            ),
            "provider": (
                "livekit"
            ),
        }

@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallOrchestrator:
    """
    Voice call workflow coordinator.
    """

    context_builder: InteractionContextBuilder

    session_service: VoiceCharacterSessionService

    voice_runtime: VoiceCallRuntime = (
        VoiceCallRuntime()
    )



    def prepare_call(
        self,
        *,
        request: VoiceCallRequest,
    ) -> VoiceCallSession:
        """
        Prepare voice call session.

        Flow:

        Request
          |
          â†“
        Interaction Context
          |
          â†“
        Character Voice Session
          |
          â†“
        Voice Call Session
          |
          â†“
        Runtime Preparation
          |
          â†“
        READY state
        """


        learner = LearnerContext(
            learner_id=request.learner_id,
            learning_goals=[
                "conversation",
            ],
        )


        voice = VoiceContext(
            language_code=request.language_code,
            voice_provider="openai",
            voice_profile="verse",
        )


        context = (
            self.context_builder
            .build(
                learner=learner,
                world_id=request.world_id,
                character_id=request.character_id,
                voice=voice,
            )
        )


        voice_session = (
            self.session_service
            .create_session(
                context=context,
            )
        )


        call_session = VoiceCallSession(
            session_id=(
                voice_session.session_id
            ),
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
        )


        return (
            self.voice_runtime
            .prepare(
                session=call_session,
            )
        )
"""
Voice Character Session Service.

Application service responsible for
creating character voice sessions.

- Coordinate character runtime.
- Coordinate speech runtime.

from __future__ import annotations


from dataclasses import dataclass
from uuid import uuid4


from app.ai_runtime.voice.voice_character_runtime import (
    VoiceCharacterRuntime,
)


from app.ai_runtime.voice.voice_speech_runtime import (
    VoiceSpeechRuntime,
)

    InteractionContext,
)

    VoiceSession,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCharacterSessionService:
    """
    Creates voice sessions for AI characters.
    """

    livekit_adapter: object

    speech_runtime: VoiceSpeechRuntime



    def create_session(
        self,
        *,
        context: InteractionContext,
    ) -> VoiceSession:
        """
        Create a character voice session.
        """


        character_runtime = (
            VoiceCharacterRuntime(
                context=context,
                speech_runtime=(
                    self.speech_runtime
                ),
            )
        )


        character_id = (
            character_runtime
            .character_id()
        )


        session = VoiceSession(
            session_id=str(
                uuid4()
            ),
            room_name=(
                f"{character_id}-voice-room"
            ),
            character_id=character_id,
            language_code=(
                context.voice.language_code
            ),
        )


        self.livekit_adapter.create_room_configuration(
            session=session,
        )


        return session

@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallExecutionService:
    """
    Application boundary for
    voice call execution.
    """


    execution_runtime: (
        VoiceCallExecutionRuntime
    )



    def _build_context(
        self,
        *,
        session: VoiceCallSession,
        voice_call: VoiceCall | None = None,
    ) -> VoiceCallExecutionContext:
        """
        Build runtime execution context.

        Beta composition:

        Learner
            +
        Character
        """


        learner = VoiceCallParticipant(
            participant_id=(
                session.learner_id
            ),
            role=(
                VoiceParticipantRole.LEARNER
            ),
            display_name="Learner",
        )


        character = VoiceCallParticipant(
            participant_id=(
                session.character_id
            ),
            role=(
                VoiceParticipantRole.CHARACTER
            ),
            display_name=(
                session.character_id
            ),
        )


        room = VoiceCallRoom(
            room_id=(
                session.session_id
            ),
            participants=(
                learner,
                character,
            ),
        )


        return VoiceCallExecutionContext(
            session=session,
            room=room,
            voice_call=voice_call,
        )



    def start_call(
        self,
        *,
        session: VoiceCallSession,
        voice_call: VoiceCall | None = None,
    ) -> VoiceCallSession:
        """
        Start voice call execution.
        """


        context = self._build_context(
            session=session,
            voice_call=voice_call,
        )


        return (
            self.execution_runtime
            .start(
                context=context,
            )
        )



    def stop_call(
        self,
        *,
        session: VoiceCallSession,
        voice_call: VoiceCall | None = None,
    ) -> VoiceCallSession:
        """
        Stop voice call execution.
        """


        context = self._build_context(
            session=session,
            voice_call=voice_call,
        )


        return (
            self.execution_runtime
            .stop(
                context=context,
            )
        )



    def fail_call(
        self,
        *,
        session: VoiceCallSession,
        voice_call: VoiceCall | None = None,
    ) -> VoiceCallSession:
        """
        Mark voice call execution failed.
        """


        context = self._build_context(
            session=session,
            voice_call=voice_call,
        )


        return (
            self.execution_runtime
            .fail(
                context=context,
            )
        )

    "VoiceCallExecutionService",

"""
Voice Call Execution Coordinator.

Application orchestration boundary
for voice call execution.

Responsibilities:
- Prepare voice room.
- Coordinate runtime execution.

Does NOT:
- Handle HTTP.
- Manage providers.
- Persist data.
- Connect LiveKit directly.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.application.uai.voice.voice_room_service import (
    VoiceRoomService,
)


from app.ai_runtime.voice.voice_call_execution_runtime import (
    VoiceCallExecutionRuntime,
)


from app.domain.uai.voice.voice_call_execution_context import (
    VoiceCallExecutionContext,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallExecutionCoordinator:
    """
    Coordinates voice call execution.

    Keeps compatibility with the current
    application contracts.

    Future migration target:

        Lifecycle
            |
            â†“
        ExecutionService
            |
            â†“
        Runtime
    """


    room_service: VoiceRoomService

    execution_runtime: VoiceCallExecutionRuntime



    def start(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> dict:
        """
        Prepare room and execute runtime.
        """


        room = (
            self.room_service
            .prepare_room(
                context=context,
            )
        )


        session = (
            self.execution_runtime
            .start(
                context=context,
            )
        )


        return {
            "room": room,
            "session": session,
        }



    def prepare(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> dict:
        """
        Prepare room only.
        """


        return (
            self.room_service
            .prepare_room(
                context=context,
            )
        )

def build_voice_call_service() -> VoiceCallService:
    """
    Create configured voice call service.
    """

    #
    # Application configuration root:
    #
    # backend/app
    #

    app_root = (
        Path(__file__)
        .resolve()
        .parents[3]
    )


    world_repository = YamlWorldRepository(
        worlds_path=(
            app_root
            / "configuration"
            / "worlds"
        )
    )


    character_repository = YamlCharacterRepository(
        profiles_path=(
            Path(
                r"C:\AI Folder\FrenchTeacherAI"
            )
            / "docs"
            / "specifications"
            / "characters"
            / "profiles"
        )
    )


    context_builder = InteractionContextBuilder(
        world_resolver=WorldResolver(
            world_repository=world_repository,
        ),
        character_resolver=CharacterResolver(
            character_repository=character_repository,
        ),
    )


    #
    # Voice interaction-context bridge.
    #
    # Converts the existing voice call execution
    # context into the established UAI InteractionContext.
    #

    interaction_context_service = (
        VoiceInteractionContextService(
            context_builder=context_builder,
        )
    )


    #
    # LiveKit infrastructure boundary.
    #

    livekit_adapter = LiveKitAdapter()


    #
    # Speech provider configuration.
    #

    speech_config = SpeechProviderConfig(
        provider_name="openai",
        model_name="tts-1",
        default_language="fr",
    )


    speech_provider = (
        SpeechProviderFactory.create(
            config=speech_config,
        )
    )


    speech_runtime = VoiceSpeechRuntime(
        speech_provider=speech_provider,
    )


    session_service = VoiceCharacterSessionService(
        livekit_adapter=livekit_adapter,
        speech_runtime=speech_runtime,
    )


    orchestrator = VoiceCallOrchestrator(
        context_builder=context_builder,
        session_service=session_service,
    )


    #
    # Keep the interaction-context bridge available
    # through the composition root for realtime voice
    # infrastructure.
    #
    # The existing VoiceCallService contract remains
    # unchanged.
    #

    _ = interaction_context_service


    return VoiceCallService(
        orchestrator=orchestrator,
    )
