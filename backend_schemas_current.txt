"""
EasyChatBot API Schemas.

Defines the HTTP request and response contracts
for the EasyChatBot API.

The API schema layer is responsible only for
transport validation and serialisation.

It does not:

- execute agents;
- select providers;
- build prompts;
- manage memory;
- perform research;
- call provider SDKs;
- execute Runtime capabilities.
"""

from __future__ import annotations

from typing import Any

from pydantic import BaseModel
from pydantic import Field


class ChatRequest(BaseModel):
    """
    Request payload for a companion interaction.

    The message field remains optional at the Pydantic
    transport layer so the API route can preserve the
    existing EasyChatBot HTTP contract:

        missing message -> HTTP 400

    This maintains backward compatibility with the
    existing API behaviour and tests.
    """

    user_id: str = Field(
        default="demo-user",
        min_length=1,
        description="Learner/user identifier.",
    )

    character: str = Field(
        default="yeti",
        min_length=1,
        description="Selected EasyChatBot character identifier.",
    )

    message: str | None = Field(
        default=None,
        description="Learner message.",
    )

    research: bool = Field(
        default=False,
        description="Whether live research should be performed.",
    )


class ChatResponse(BaseModel):
    """
    Response payload for a companion interaction.

    Agent output remains flexible during Beta 1 because
    the underlying result contracts are intentionally
    extensible.
    """

    character: str

    conversation: Any

    research: Any = None

    quality: Any

    memory_count: int

    memory_context_count: int = 0


class PodcastDiscoveryRequest(BaseModel):
    """
    HTTP request contract for Podcast Discovery.

    This schema is responsible only for transport
    validation.

    It does not:

    - execute discovery;
    - select providers;
    - apply governance;
    - execute agents.
    """

    query: str = Field(
        min_length=1,
        description="Podcast discovery query.",
    )

    target_language: str = Field(
        min_length=1,
        description="Target language for the learner.",
    )

    learner_level: str = Field(
        min_length=1,
        description="Learner proficiency level.",
    )


class PodcastDiscoveryResponse(BaseModel):
    """
    HTTP response contract for Podcast Discovery.

    The Runtime remains the source of the capability
    result. The API schema only serialises the result
    for HTTP transport.
    """

    request_id: str

    capability: Any

    status: Any

    output: Any = None


class RuntimeExecuteRequest(BaseModel):
    """
    HTTP request contract for External Runtime execution.

    The API schema layer only validates transport data.

    It does not:

    - execute capabilities;
    - call providers;
    - execute agents;
    - apply governance decisions.
    """

    capability: str = Field(
        min_length=1,
        description="Requested Runtime capability.",
    )

    payload: dict[str, Any] = Field(
        default_factory=dict,
        description="Capability execution payload.",
    )

    user_id: str = Field(
        default="demo-user",
        min_length=1,
        description="Learner/user identifier.",
    )


class RuntimeExecuteResponse(BaseModel):
    """
    HTTP response contract for External Runtime execution.

    The Runtime remains responsible for execution.
    This model only serialises the result.
    """

    request_id: str | None = None

    capability: str | None = None

    status: str

    output: Any = None

    error: str | None = None

class HackathonDemoRequest(BaseModel):
    """
    Request contract for the hackathon demonstration flow.

    This endpoint is intentionally separated from the
    production ChatRequest contract.

    Purpose:

    - allow hackathon iteration;
    - protect existing chat functionality;
    - provide a stable demo entry point.

    This schema does not:

    - execute agents;
    - call providers;
    - bypass Runtime controls.
    """

    user_id: str = Field(
        default="demo-user",
        min_length=1,
        description="Demo learner identifier.",
    )

    message: str = Field(
        min_length=1,
        description="Hackathon demo instruction.",
    )


class HackathonDemoResponse(BaseModel):
    """
    Response contract for the hackathon vertical slice.

    The response represents the application result
    returned after Runtime execution.
    """

    status: str

    capability: str

    output: Any = None

    request_id: str | None = None
