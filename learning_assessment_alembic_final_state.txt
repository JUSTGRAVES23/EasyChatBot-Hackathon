        Set-Location "C:\AI Folder\EasyChatBot-Hackathon\backend"

        Get-ChildItem ".\alembic\versions" -File |
            Where-Object { $_.Extension -eq ".py" } |
            ForEach-Object {

                $Content = Get-Content $_.FullName -Raw

                $Revision = [regex]::Match(
                    $Content,
                    '(?m)^\s*revision\s*:\s*str\s*=\s*["'']([^"'']+)["'']'
                )

                if (-not $Revision.Success) {
                    $Revision = [regex]::Match(
                        $Content,
                        '(?m)^\s*revision\s*=\s*["'']([^"'']+)["'']'
                    )
                }

                $DownRevision = [regex]::Match(
                    $Content,
                    '(?m)^\s*down_revision[^=]*=\s*(.+)$'
                )

                $BranchLabels = [regex]::Match(
                    $Content,
                    '(?m)^\s*branch_labels[^=]*=\s*(.+)$'
                )

                    File = $_.Name
                    Revision = $Revision.Groups[1].Value
                    DownRevision = $DownRevision.Groups[1].Value.Trim()

                }
            } |
            Sort-Object Revision |
            Format-Table -AutoSize
