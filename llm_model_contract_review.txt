==================== MODEL ID ====================

"""
Model Identifier.

Defines a stable identifier for AI models.

The identifier is independent from provider execution logic.
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class ModelId:
    """
    Unique identifier for an AI model.

    Examples:

    gemini-2.5-flash
    gpt-5-mini
    llama3

    """

    provider: str
    name: str

    def full_name(self) -> str:
        """
        Returns the complete model identifier.
        """

        return f"{self.provider}:{self.name}"


==================== MODEL DESCRIPTOR ====================

"""
Model Descriptor.

Combines model identity and capabilities.

The Runtime uses this information
without depending on a specific provider.
"""

from dataclasses import dataclass

from backend.runtime.models.model_id import ModelId
from backend.runtime.models.model_capabilities import ModelCapabilities


@dataclass(frozen=True)
class ModelDescriptor:
    """
    Describes an available AI model.

    Attributes:

        model_id:
            Unique model identity.

        capabilities:
            Supported capabilities.

        description:
            Human-readable explanation.
    """

    model_id: ModelId

    capabilities: ModelCapabilities

    description: str = ""


==================== MODEL CAPABILITIES ====================

"""
Model Capability Definition.

Describes what an AI model can support.

Used by the Runtime to select appropriate models.
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class ModelCapabilities:
    """
    Capabilities supported by a model.
    """

    text_generation: bool = True
    reasoning: bool = False
    tool_usage: bool = False
    vision: bool = False
    audio: bool = False


==================== MODEL TEST ====================

from backend.runtime.models.model_id import ModelId
from backend.runtime.models.model_capabilities import ModelCapabilities
from backend.runtime.models.model_descriptor import ModelDescriptor


def test_model_id():

    model = ModelId(
        provider="google",
        name="gemini-2.5-flash",
    )

    assert model.full_name() == (
        "google:gemini-2.5-flash"
    )


def test_model_descriptor():

    descriptor = ModelDescriptor(
        model_id=ModelId(
            provider="google",
            name="gemini-2.5-flash",
        ),
        capabilities=ModelCapabilities(
            reasoning=True,
            tool_usage=True,
        ),
        description="Fast reasoning model",
    )

    assert descriptor.model_id.provider == "google"

    assert descriptor.capabilities.reasoning is True

    assert descriptor.capabilities.tool_usage is True


==================== PROVIDER CONTRACT TEST ====================

import pytest

from backend.providers.base_provider import BaseProvider
from backend.providers.provider_result import ProviderResult
from backend.runtime.context.execution_context import ExecutionContext


class DemoProvider(BaseProvider):

    async def execute(
        self,
        context: ExecutionContext,
    ) -> ProviderResult:

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output="Provider response",
        )


@pytest.mark.asyncio
async def test_provider_execution():

    provider = DemoProvider(
        provider_id="demo_provider"
    )

    context = ExecutionContext(
        user_id="user1",
        request="Generate response"
    )

    result = await provider.execute(context)

    assert result.success is True
    assert result.provider_id == "demo_provider"
    assert result.output == "Provider response"
