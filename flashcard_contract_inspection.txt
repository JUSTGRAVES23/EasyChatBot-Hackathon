===== FLASHCARD GENERATOR =====

from __future__ import annotations

from src.application.experience.flashcards.models import (
    CharacterPresentationContext,
    FlashCard,
)

from src.application.experience.flashcards.types import (
    FlashCardType,
)

from src.application.experience.recommendations.models import (
    ProductRecommendation,
)


class FlashCardGenerator:
    """
    Converts recommendations into user-facing cards.

    The generator applies character personality.
    """

    def create_product_card(
        self,
        *,
        recommendation: ProductRecommendation,
        character: CharacterPresentationContext,
    ) -> FlashCard:

        return FlashCard(
            card_id=(
                recommendation.recommendation_id
            ),

            card_type=(
                FlashCardType.PRODUCT_RECOMMENDATION
            ),

            character=character,

            title=(
                character.opening_phrase
            ),

            subtitle=(
                recommendation.product_name
            ),

            highlights=(
                recommendation.relevance_reasons
            ),

            actions=tuple(
                {
                    "type": action.action_type,
                    "label": action.label,
                    "url": action.url or "",
                }
                for action in recommendation.actions
            ),
        )

===== FLASHCARD MODELS =====

from __future__ import annotations

from dataclasses import dataclass

from src.application.experience.flashcards.types import (
    FlashCardType,
)


@dataclass(frozen=True, slots=True)
class CharacterPresentationContext:
    """
    Character personality input.

    The character influences style,
    not recommendation decisions.
    """

    character_id: str

    character_name: str

    opening_phrase: str

    tone: str


@dataclass(frozen=True, slots=True)
class FlashCard:
    """
    User-facing recommendation card.

    This is consumed by:

    - Web App
    - Mobile App
    - Telegram Adapter
    - Voice Adapter
    """

    card_id: str

    card_type: FlashCardType

    character: CharacterPresentationContext

    title: str

    subtitle: str

    highlights: tuple[str, ...]

    actions: tuple[dict[str, str], ...]

===== FLASHCARD TYPES =====

from __future__ import annotations

from enum import StrEnum


class FlashCardType(StrEnum):
    """
    Supported user experience cards.

    Cards are presentation contracts.

    They do not contain business logic.
    """

    PRODUCT_RECOMMENDATION = (
        "product_recommendation"
    )

    CONTENT_RECOMMENDATION = (
        "content_recommendation"
    )

    PODCAST_RECOMMENDATION = (
        "podcast_recommendation"
    )

    LEARNING_CHALLENGE = (
        "learning_challenge"
    )

    NEWS_ALERT = (
        "news_alert"
    )

===== RECOMMENDATION MODELS =====

from __future__ import annotations

from dataclasses import dataclass
from decimal import Decimal


@dataclass(frozen=True, slots=True)
class RecommendationAction:
    """
    User action available on a recommendation.
    """

    action_type: str

    label: str

    url: str | None = None


@dataclass(frozen=True, slots=True)
class ProductRecommendation:
    """
    Provider-independent product recommendation.

    This is the intelligence output.

    It does not know about UI rendering.
    """

    recommendation_id: str

    product_name: str

    description: str

    vendor: str

    vendor_url: str

    price: Decimal

    currency: str

    discount_percentage: int

    relevance_reasons: tuple[str, ...]

    actions: tuple[RecommendationAction, ...]
