===== backend/runtime/context/execution_context.py =====
"""
Runtime Execution Context.

Provides immutable runtime context shared with agents.

Responsibilities:
- identify an execution
- carry user request information
- provide runtime metadata
- optionally carry live intelligence context

Does not:
- execute providers
- manage memory
- modify systems
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any
from uuid import uuid4


@dataclass(frozen=True)
class ExecutionContext:
    """
    Runtime execution context shared with agents.

    Attributes
    ----------
    user_id:
        Identifier of the user initiating the execution.

    request:
        User request being processed.

    execution_id:
        Unique identifier for this execution.

    created_at:
        UTC timestamp when the context was created.

    metadata:
        Runtime metadata.

    intelligence_context:
        Optional provider-independent live intelligence
        generated by the Research Agent.

    Responsibilities:
    - identify an execution
    - carry user request information
    - provide runtime metadata
    - optionally carry live intelligence context

    Does not:
    - execute providers
    - manage memory
    - modify systems
    """

    user_id: str

    request: str

    execution_id: str = field(
        default_factory=lambda: str(uuid4())
    )

    created_at: datetime = field(
        default_factory=lambda: datetime.now(timezone.utc)
    )

    metadata: dict[str, Any] = field(
        default_factory=dict
    )

    intelligence_context: Any | None = None

    @property
    def has_intelligence_context(self) -> bool:
        """
        Return whether live intelligence is attached
        to this execution.
        """

        return self.intelligence_context is not None

===== backend/services/companion_service.py =====
"""
Companion Service.

Application orchestration layer for EasyChatBot.

Coordinates:

Character
Conversation Agent
Research Agent
Quality Agent
Memory Service

Live Intelligence flow:

CompanionService
|
+--> MemoryService
|
+--> ResearchAgent
|        |
|        v
|   IntelligenceContext
|
+--> ConversationAgent
|        |
|        v
|   LLMService
|        |
|        v
|   LLMProvider
|
+--> QualityAgent
|
v
Interaction Result

Responsibilities:

- orchestrate companion interactions
- retrieve memories
- optionally execute research
- enrich the execution context with live intelligence
- execute the ConversationAgent
- execute the QualityAgent
- persist the interaction
- return the combined result

Does not:

- create LLM providers
- select LLM providers
- call provider SDKs
- call OpenAI directly
- call Ollama directly
- manage provider routing
- build provider-specific prompts
"""

from __future__ import annotations

from dataclasses import replace
from typing import Any

from backend.agents.conversation_agent import (
    ConversationAgent,
)

from backend.agents.research_agent import (
    ResearchAgent,
)

from backend.agents.quality_agent import (
    QualityAgent,
)

from backend.characters.character_profile import (
    CharacterProfile,
)

from backend.memory.memory_service import (
    MemoryService,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class CompanionService:
    """
    Coordinates AI companion experiences.

    The service orchestrates agents but does not contain
    provider-specific logic.

    Providers are injected into agents by the composition
    layer.
    """

    def __init__(
        self,
        character: CharacterProfile,
        conversation_agent: ConversationAgent | None = None,
        research_agent: ResearchAgent | None = None,
        quality_agent: QualityAgent | None = None,
        memory_service: MemoryService | None = None,
    ) -> None:
        """
        Initialise the CompanionService.

        Dependencies are injectable so the service can be
        tested without invoking real infrastructure.
        """

        self.character = character

        self.conversation_agent = (
            conversation_agent
            if conversation_agent is not None
            else ConversationAgent(
                character
            )
        )

        self.research_agent = (
            research_agent
            if research_agent is not None
            else ResearchAgent()
        )

        self.quality_agent = (
            quality_agent
            if quality_agent is not None
            else QualityAgent()
        )

        self.memory_service = (
            memory_service
            if memory_service is not None
            else MemoryService()
        )

    async def interact(
        self,
        context: ExecutionContext,
        use_research: bool = False,
    ) -> dict[str, Any]:
        """
        Execute a complete companion interaction.

        When research is requested, the Research Agent
        produces an IntelligenceContext which is attached
        to a new immutable ExecutionContext before the
        Conversation Agent executes.

        The original ExecutionContext is never mutated.

        Parameters
        ----------
        context:
            Runtime execution context.

        use_research:
            Whether live research should be performed
            before the conversation response.
        """

        memories = self.memory_service.recall(
            context.user_id
        )

        research_result = None

        conversation_context = context

        if use_research:
            research_result = (
                await self.research_agent.execute(
                    context
                )
            )

            if (
                research_result.success
                and research_result.output is not None
            ):
                conversation_context = replace(
                    context,
                    intelligence_context=(
                        research_result.output
                    ),
                )

        conversation_result = (
            await self.conversation_agent.execute(
                conversation_context
            )
        )

        quality_result = (
            await self.quality_agent.execute(
                conversation_context
            )
        )

        self.memory_service.remember(
            user_id=context.user_id,
            category="conversation",
            content=context.request,
        )

        return {
            "character": self.character.name,

            "conversation": (
                conversation_result.output
            ),

            "research": (
                research_result.output
                if research_result is not None
                else None
            ),

            "quality": (
                quality_result.output
            ),

            "memory_count": len(memories),
        }

===== backend/agents/conversation_agent.py =====
"""
Conversation Agent.

Provides character-aware conversational responses
for the EasyChatBot hackathon runtime.

The agent remains provider-independent.

Architecture:

ExecutionContext
|
v
ConversationAgent
|
+--> CharacterProfile
|
+--> Optional IntelligenceContext
|
v
ConversationRequest
|
v
LLMService
|
v
LLMProvider
|
v
Concrete LLM Provider
|
v
ProviderResult
|
v
AgentResult

Responsibilities:

- apply character identity
- construct ConversationRequest
- include optional live intelligence
- execute an injected LLMService
- preserve deterministic demo behaviour when no
  LLM service is configured
- return a standard AgentResult

Does not:

- depend directly on OpenAI
- depend directly on Ollama
- call Parallel directly
- manage memory
- perform research
- manage voice
- execute external workflows
- select providers
- execute provider SDKs directly

Architecture boundary:

ConversationAgent
-> LLMService
-> LLMProvider
-> Concrete Provider
"""

from __future__ import annotations

from backend.agents.agent_manifest import (
    AgentManifest,
)

from backend.agents.agent_result import (
    AgentResult,
)

from backend.agents.base_agent import (
    BaseAgent,
)

from backend.characters.character_profile import (
    CharacterProfile,
)

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)

from backend.services.llm_service import (
    LLMService,
)


class ConversationAgent(BaseAgent):
    """
    Character-aware conversation agent.

    The agent depends primarily on the application-level
    LLMService boundary.

    The LLMService is responsible for delegating
    execution to the configured LLMProvider.

    A legacy ``llm_provider`` argument remains accepted
    as a compatibility seam for existing tests and
    callers. When supplied, it is wrapped by LLMService.

    New composition code should inject ``llm_service``.
    """

    def __init__(
        self,
        character: CharacterProfile,
        llm_service: LLMService | None = None,
        llm_provider: LLMProvider | None = None,
    ) -> None:
        """
        Initialise the Conversation Agent.

        Parameters
        ----------
        character:
            Approved EasyChatBot character profile.

        llm_service:
            Application-level LLM execution service.

        llm_provider:
            Compatibility injection seam.

            When supplied and ``llm_service`` is not supplied,
            the provider is wrapped inside LLMService.

        Raises
        ------
        ValueError
            If both ``llm_service`` and ``llm_provider``
            are supplied simultaneously.
        """

        super().__init__(
            AgentManifest(
                agent_id="conversation_agent",
                name="Conversation Agent",
                description=(
                    "Provides character-aware "
                    "conversation responses."
                ),
            )
        )

        if (
            llm_service is not None
            and llm_provider is not None
        ):
            raise ValueError(
                "Provide either llm_service or "
                "llm_provider, not both."
            )

        self.character = character

        if llm_service is not None:
            self.llm_service = llm_service

        elif llm_provider is not None:
            self.llm_service = LLMService(
                llm_provider
            )

        else:
            self.llm_service = None

    def build_request(
        self,
        context: ExecutionContext,
    ) -> ConversationRequest:
        """
        Build a provider-independent conversation request.

        Runtime state and character identity are translated
        into the ConversationRequest contract.

        No provider-specific implementation is introduced.
        """

        system_instruction = (
            f"You are {self.character.name}, "
            f"a character in EasyChatBot. "
            f"Your personality is: "
            f"{self.character.personality} "
            f"Your learning style is: "
            f"{self.character.learning_style} "
            f"Communicate in a way that is consistent "
            f"with this character."
        )

        conversation_context = None

        if context.has_intelligence_context:
            conversation_context = (
                context.intelligence_context.to_context_text()
            )

        return ConversationRequest(
            system_instruction=system_instruction,
            user_prompt=context.request,
            conversation_context=conversation_context,
            character_id=self.character.character_id,
            character_name=self.character.name,
        )

    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute the conversation agent.

        Flow:

        ExecutionContext
            |
            v
        ConversationRequest
            |
            v
        LLMService
            |
            v
        LLMProvider
            |
            v
        ProviderResult
            |
            v
        AgentResult

        When no LLMService is configured, the existing
        deterministic hackathon behaviour is preserved.

        When an LLMService is configured, execution
        passes through the application service boundary.
        """

        conversation_request = self.build_request(
            context
        )

        provider_result = None

        if self.llm_service is not None:
            provider_result = (
                await self.llm_service.generate(
                    conversation_request
                )
            )

            if not provider_result.success:
                return AgentResult(
                    success=False,
                    agent_id=self.manifest.agent_id,
                    output={
                        "character": self.character.name,

                        "character_profile": {
                            "character_id": (
                                self.character.character_id
                            ),
                            "name": self.character.name,
                            "description": (
                                self.character.description
                            ),
                            "personality": (
                                self.character.personality
                            ),
                            "learning_style": (
                                self.character.learning_style
                            ),
                        },

                        "personality": (
                            self.character.personality
                        ),

                        "learning_style": (
                            self.character.learning_style
                        ),

                        "message": (
                            "The conversation provider "
                            "could not generate a response."
                        ),

                        "intelligence_context": (
                            conversation_request
                            .conversation_context
                        ),

                        "conversation_request": (
                            conversation_request
                        ),

                        "provider_id": (
                            provider_result.provider_id
                        ),

                        "provider_error": (
                            provider_result.error
                        ),
                    },
                )

        if provider_result is not None:
            message = provider_result.output

        else:
            message = (
                f"{self.character.name}: "
                f"Let's explore this together. "
                f"You asked: {context.request}"
            )

        output = {
            "character": self.character.name,

            "character_profile": {
                "character_id": (
                    self.character.character_id
                ),
                "name": self.character.name,
                "description": (
                    self.character.description
                ),
                "personality": (
                    self.character.personality
                ),
                "learning_style": (
                    self.character.learning_style
                ),
            },

            "personality": (
                self.character.personality
            ),

            "learning_style": (
                self.character.learning_style
            ),

            "message": message,

            "intelligence_context": (
                conversation_request
                .conversation_context
            ),

            "conversation_request": (
                conversation_request
            ),
        }

        if provider_result is not None:
            output["provider_id"] = (
                provider_result.provider_id
            )

        return AgentResult(
            success=True,
            agent_id=self.manifest.agent_id,
            output=output,
        )

===== backend/conversation/conversation_request.py =====
"""
Provider-independent conversation request.

Represents the information required to generate a
character-aware conversational response.

This contract belongs to the application/runtime boundary.

It deliberately does not know about:

- Gemini
- OpenAI
- Ollama
- Parallel
- voice
- HTTP
- persistence
- provider SDKs

Architecture:

CharacterProfile
        +
ExecutionContext
        +
Optional IntelligenceContext
        |
        v
ConversationRequest
        |
        v
Future / existing EasyChatBot LLM service

The production EasyChatBot conversation and voice
architecture remains outside this hackathon runtime.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class ConversationRequest:
    """
    Immutable provider-independent conversation request.

    Attributes
    ----------
    system_instruction:
        Stable instruction defining how the character
        should behave.

    user_prompt:
        User-facing request to be answered.

    conversation_context:
        Optional contextual information, such as live
        intelligence retrieved by the Research Agent.

    character_id:
        Identifier of the character producing the response.

    character_name:
        Display name of the character.
    """

    system_instruction: str

    user_prompt: str

    conversation_context: str | None = None

    character_id: str = ""

    character_name: str = ""

    def __post_init__(self) -> None:
        """
        Validate request invariants.
        """

        if not isinstance(
            self.system_instruction,
            str,
        ):
            raise ValueError(
                "system_instruction must be a string."
            )

        if not isinstance(
            self.user_prompt,
            str,
        ):
            raise ValueError(
                "user_prompt must be a string."
            )

        system_instruction = (
            self.system_instruction.strip()
        )

        user_prompt = (
            self.user_prompt.strip()
        )

        if not system_instruction:
            raise ValueError(
                "system_instruction cannot be empty."
            )

        if not user_prompt:
            raise ValueError(
                "user_prompt cannot be empty."
            )

        object.__setattr__(
            self,
            "system_instruction",
            system_instruction,
        )

        object.__setattr__(
            self,
            "user_prompt",
            user_prompt,
        )

        if self.conversation_context is not None:
            context = (
                self.conversation_context.strip()
            )

            if not context:
                raise ValueError(
                    "conversation_context cannot be blank."
                )

            object.__setattr__(
                self,
                "conversation_context",
                context,
            )

        if not isinstance(
            self.character_id,
            str,
        ):
            raise ValueError(
                "character_id must be a string."
            )

        if not isinstance(
            self.character_name,
            str,
        ):
            raise ValueError(
                "character_name must be a string."
            )

        object.__setattr__(
            self,
            "character_id",
            self.character_id.strip(),
        )

        object.__setattr__(
            self,
            "character_name",
            self.character_name.strip(),
        )

    @property
    def has_conversation_context(self) -> bool:
        """
        Return whether contextual information exists.
        """

        return bool(
            self.conversation_context
        )

    @property
    def context_length(self) -> int:
        """
        Return the length of the conversation context.

        Useful for diagnostics and future token estimation.
        """

        if not self.conversation_context:
            return 0

        return len(
            self.conversation_context
        )

    def summary(self) -> dict[str, str]:
        """
        Return a safe diagnostic summary.

        Prompt and context content are deliberately
        excluded from the summary.
        """

        return {
            "character_id": self.character_id,
            "character_name": self.character_name,
            "has_context": str(
                self.has_conversation_context
            ),
            "system_instruction_length": str(
                len(self.system_instruction)
            ),
            "user_prompt_length": str(
                len(self.user_prompt)
            ),
            "context_length": str(
                self.context_length
            ),
        }
