"""
Voice Call Orchestrator.

Coordinates the preparation of a voice call.

Responsibilities:
- Build interaction context.
- Create character voice session.
- Create domain voice call session.
- Apply voice call runtime lifecycle.

Does NOT:
- Execute AI providers.
- Manage LiveKit.
- Handle HTTP.
- Persist data.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.context.interaction_context_builder import (
    InteractionContextBuilder,
)


from app.application.uai.voice.voice_character_session_service import (
    VoiceCharacterSessionService,
)


from app.ai_runtime.voice.voice_call_runtime import (
    VoiceCallRuntime,
)


from app.domain.uai.context.learner_context import (
    LearnerContext,
)


from app.domain.uai.context.voice_context import (
    VoiceContext,
)


from app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallOrchestrator:
    """
    Voice call workflow coordinator.
    """

    context_builder: InteractionContextBuilder

    session_service: VoiceCharacterSessionService

    voice_runtime: VoiceCallRuntime = (
        VoiceCallRuntime()
    )



    def prepare_call(
        self,
        *,
        request: VoiceCallRequest,
    ) -> VoiceCallSession:
        """
        Prepare voice call session.

        Flow:

        Request
          |
          â†“
        Interaction Context
          |
          â†“
        Character Voice Session
          |
          â†“
        Voice Call Session
          |
          â†“
        Runtime Preparation
          |
          â†“
        READY state
        """


        learner = LearnerContext(
            learner_id=request.learner_id,
            learning_goals=[
                "conversation",
            ],
        )


        voice = VoiceContext(
            language_code=request.language_code,
            voice_provider="openai",
            voice_profile="verse",
        )


        context = (
            self.context_builder
            .build(
                learner=learner,
                world_id=request.world_id,
                character_id=request.character_id,
                voice=voice,
            )
        )


        voice_session = (
            self.session_service
            .create_session(
                context=context,
            )
        )


        call_session = VoiceCallSession(
            session_id=(
                voice_session.session_id
            ),
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
        )


        return (
            self.voice_runtime
            .prepare(
                session=call_session,
            )
        )
"""
Voice Character Session Service.

Application service responsible for
creating character voice sessions.

Responsibilities:
- Coordinate character runtime.
- Coordinate speech runtime.
- Create domain voice session.
- Request LiveKit room configuration.

Does NOT:
- Call speech providers directly.
- Manage LiveKit directly.
- Store configuration.
"""

from __future__ import annotations


from dataclasses import dataclass
from uuid import uuid4


from app.ai_runtime.voice.voice_character_runtime import (
    VoiceCharacterRuntime,
)


from app.ai_runtime.voice.voice_speech_runtime import (
    VoiceSpeechRuntime,
)


from app.domain.uai.context.interaction_context import (
    InteractionContext,
)


from app.domain.uai.voice.voice_session import (
    VoiceSession,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCharacterSessionService:
    """
    Creates voice sessions for AI characters.
    """

    livekit_adapter: object

    speech_runtime: VoiceSpeechRuntime



    def create_session(
        self,
        *,
        context: InteractionContext,
    ) -> VoiceSession:
        """
        Create a character voice session.
        """


        character_runtime = (
            VoiceCharacterRuntime(
                context=context,
                speech_runtime=(
                    self.speech_runtime
                ),
            )
        )


        character_id = (
            character_runtime
            .character_id()
        )


        session = VoiceSession(
            session_id=str(
                uuid4()
            ),
            room_name=(
                f"{character_id}-voice-room"
            ),
            character_id=character_id,
            language_code=(
                context.voice.language_code
            ),
        )


        self.livekit_adapter.create_room_configuration(
            session=session,
        )


        return session
"""
Voice Call Application Service.

Application boundary for voice calls.

Responsibilities:
- Receive voice call requests.
- Delegate preparation workflow.
- Return prepared call information.

Does NOT:
- Execute speech.
- Manage providers.
- Manage HTTP.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.voice.voice_call_orchestrator import (
    VoiceCallOrchestrator,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallService:
    """
    Voice call application service.
    """

    orchestrator: VoiceCallOrchestrator



    def prepare_call(
        self,
        *,
        request: VoiceCallRequest,
    ) -> dict:
        """
        Prepare voice call.
        """


        session = (
            self.orchestrator
            .prepare_call(
                request=request
            )
        )


        return {
            "room_name": (
                session.session_id
            ),
            "character_id": (
                session.character_id
            ),
            "language_code": (
                session.language_code
            ),
            "provider": (
                "livekit"
            ),
        }
"""
Voice Calls API Router.

HTTP boundary for preparing
character voice calls.

Responsibilities:
- Receive API requests.
- Validate request payload.
- Delegate to application services.
- Return HTTP responses.

Does NOT:
- Execute AI agents.
- Connect to LiveKit directly.
- Manage conversations.
- Select providers.
- Create runtime dependencies.
- Build domain execution contexts.
"""

from __future__ import annotations


from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
)


from pydantic import BaseModel


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.voice.voice_call_lifecycle_service import (
    VoiceCallLifecycleService,
)


from app.application.uai.voice.voice_call_context_factory import (
    VoiceCallContextFactory,
)


from app.application.composition.voice_lifecycle_composition import (
    build_voice_call_lifecycle_service,
)



router = APIRouter(
    prefix="/voice",
    tags=[
        "voice",
    ],
)



class VoiceCallRequestModel(
    BaseModel
):
    """
    HTTP request model.
    """

    learner_id: str

    world_id: str

    character_id: str

    language_code: str



class VoiceCallResponseModel(
    BaseModel
):
    """
    HTTP response model.
    """

    room_name: str

    character_id: str

    language_code: str

    provider: str

    state: str

    token: str

    expires_at: str



def get_voice_call_lifecycle_service(
) -> VoiceCallLifecycleService:
    """
    Dependency provider.

    Composition decisions remain
    outside the HTTP adapter.
    """

    return (
        build_voice_call_lifecycle_service()
    )



def get_voice_call_context_factory(
) -> VoiceCallContextFactory:
    """
    Dependency provider for context creation.

    Domain creation remains inside
    application layer.
    """

    return VoiceCallContextFactory()



@router.post(
    "/call",
    response_model=VoiceCallResponseModel,
)
def prepare_voice_call(
    request: VoiceCallRequestModel,
    service: VoiceCallLifecycleService = Depends(
        get_voice_call_lifecycle_service
    ),
    context_factory: VoiceCallContextFactory = Depends(
        get_voice_call_context_factory
    ),
) -> VoiceCallResponseModel:
    """
    Prepare and start a character voice call.
    """


    try:

        call_request = VoiceCallRequest(
            learner_id=request.learner_id,
            world_id=request.world_id,
            character_id=request.character_id,
            language_code=request.language_code,
        )


        context = (
            context_factory.create(
                request=call_request,
            )
        )


        result = (
            service.start_call(
                context=context,
            )
        )


        room = result["room"]

        session = result["session"]

        connection = result["connection"]


        return VoiceCallResponseModel(
            room_name=room["room_name"],
            character_id=session.character_id,
            language_code=session.language_code,
            provider=room["provider"],
            state=session.state.value,
            token=connection.token,
            expires_at=(
                connection.expires_at
                .isoformat()
            ),
        )


    except ValueError as exc:

        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

"""
LiveKit Voice Worker.

Infrastructure entry point for realtime voice sessions.

Responsibilities:
- Receive a LiveKit job context.
- Read application-approved job metadata.
- Convert metadata into a voice call request.
- Build the existing EasyChatBot voice execution context.
- Build the existing UAI InteractionContext.
- Create the LiveKit realtime character agent.
- Start the LiveKit AgentSession.

Does NOT:
- Own EasyChatBot business logic.
- Resolve worlds or characters directly.
- Decide learner authorization.
- Persist conversations.
- Manage call lifecycle state.
- Bypass the EasyChatBot Runtime.
"""

from __future__ import annotations


import json
import logging


from dataclasses import dataclass


from livekit.agents import (
    AgentServer,
    JobContext,
    WorkerOptions,
    cli,
)


from app.application.uai.voice.voice_call_context_factory import (
    VoiceCallContextFactory,
)


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.voice.voice_interaction_context_service import (
    VoiceInteractionContextService,
)


from app.application.uai.voice.voice_dependencies import (
    build_voice_interaction_context_service,
)


from app.infrastructure.voice.livekit_realtime_agent import (
    LiveKitRealtimeAgent,
)


logger = logging.getLogger(__name__)


@dataclass(
    frozen=True,
    slots=True,
)
class LiveKitVoiceWorker:
    """
    Infrastructure worker for EasyChatBot realtime voice.

    Application services are injected so the worker does not
    construct domain/application dependencies itself.
    """

    context_factory: VoiceCallContextFactory

    interaction_context_service: (
        VoiceInteractionContextService
    )

    async def run(
        self,
        *,
        ctx: JobContext,
    ) -> None:
        """
        Connect to LiveKit and start the EasyChatBot
        realtime character session.
        """

        metadata = self._read_job_metadata(
            ctx,
        )


        request = self._create_voice_call_request(
            metadata,
        )


        execution_context = (
            self.context_factory.create(
                request=request,
            )
        )


        interaction_context = (
            self.interaction_context_service.build(
                context=execution_context,
            )
        )


        character = (
            interaction_context.character
        )


        instructions = (
            "You are "
            f"{character.name}, "
            f"a French learning character. "
            f"Your personality reference is "
            f"{character.personality_reference}. "
            f"Your teaching style is "
            f"{character.teaching_style}. "
            "Speak naturally, conversationally, "
            "and help the learner practice the "
            "target language."
        )


        realtime_agent = (
            LiveKitRealtimeAgent(
                context=interaction_context,
                instructions=instructions,
            )
        )


        ctx.log_context_fields = {
            "room_name": ctx.room.name,
            "learner_id": (
                request.learner_id
            ),
            "world_id": (
                request.world_id
            ),
            "character_id": (
                request.character_id
            ),
        }


        logger.info(
            "Starting EasyChatBot LiveKit voice agent",
            extra={
                "room_name": ctx.room.name,
                "learner_id": request.learner_id,
                "world_id": request.world_id,
                "character_id": request.character_id,
            },
        )


        await ctx.connect()


        agent = (
            realtime_agent.create_agent()
        )


        session = (
            realtime_agent.create_session()
        )


        await session.start(
            agent=agent,
            room=ctx.room,
        )


    @staticmethod
    def _read_job_metadata(
        ctx: JobContext,
    ) -> dict[str, object]:
        """
        Read and validate LiveKit job metadata.

        The metadata is supplied by the application when
        the LiveKit job is dispatched.
        """

        raw_metadata = (
            ctx.job.metadata
        )


        if not raw_metadata:
            raise ValueError(
                "LiveKit job metadata is required."
            )


        try:
            metadata = json.loads(
                raw_metadata,
            )

        except json.JSONDecodeError as exc:
            raise ValueError(
                "LiveKit job metadata must be valid JSON."
            ) from exc


        if not isinstance(
            metadata,
            dict,
        ):
            raise ValueError(
                "LiveKit job metadata must be a JSON object."
            )


        return metadata


    @staticmethod
    def _create_voice_call_request(
        metadata: dict[str, object],
    ) -> VoiceCallRequest:
        """
        Convert validated LiveKit job metadata
        into the existing application request contract.
        """

        required_fields = (
            "learner_id",
            "world_id",
            "character_id",
            "language_code",
        )


        values: dict[str, str] = {}


        for field in required_fields:

            value = metadata.get(
                field,
            )


            if not isinstance(
                value,
                str,
            ):
                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' must be a string."
                )


            if not value.strip():
                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' cannot be empty."
                )


            values[field] = value.strip()


        return VoiceCallRequest(
            learner_id=values["learner_id"],
            world_id=values["world_id"],
            character_id=values["character_id"],
            language_code=values["language_code"],
        )


server = AgentServer()


_worker = LiveKitVoiceWorker(
    context_factory=VoiceCallContextFactory(),
    interaction_context_service=(
        build_voice_interaction_context_service()
    ),
)


@server.rtc_session(
    agent_name="easychatbot-voice",
)
async def entrypoint(
    ctx: JobContext,
) -> None:
    """
    LiveKit Agents entrypoint.
    """

    await _worker.run(
        ctx=ctx,
    )


def run() -> None:
    """
    Start the LiveKit Agents worker.
    """

    cli.run_app(
        WorkerOptions(
            agent_name="easychatbot-voice",
        )
    )


__all__ = [
    "LiveKitVoiceWorker",
    "entrypoint",
    "run",
]
