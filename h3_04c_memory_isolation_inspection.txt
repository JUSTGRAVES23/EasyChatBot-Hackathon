"""
Memory Service.

Application-layer memory service.

This service adapts the legacy backend MemoryRecord contract
to the hackathon AI Runtime MemoryEntry / MemoryStore
persistence contract.

Architecture:

Application
    |
    v
MemoryService
    |
    v
MemoryEntry / MemoryType
    |
    v
MemoryStore
    |
    v
SQLite

Responsibilities:
- save application memories
- retrieve application memories
- translate application memory categories
- isolate persistence details from callers

Does not:
- perform semantic retrieval
- call an LLM
- execute agents
- make autonomous memory decisions
"""

from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from backend.memory.memory_record import (
    MemoryRecord,
)

from backend.memory.memory_store import (
    MemoryStore,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


class MemoryService:
    """
    Controls application memory operations.

    The service keeps the existing MemoryRecord interface
    for the backend application while delegating persistence
    to the current AI Runtime MemoryStore.
    """

    DEFAULT_DATABASE_PATH = (
        Path("data") / "memory.db"
    )

    def __init__(
        self,
        store: MemoryStore | None = None,
        database_path: str | Path | None = None,
    ) -> None:
        """
        Initialise the memory service.

        Args:
            store:
                Optional injected MemoryStore.

                Primarily useful for tests and controlled
                application configuration.

            database_path:
                Optional SQLite database path.

                Ignored when an explicit store is supplied.
        """

        if store is not None:
            self.store = store
            return

        path = (
            Path(database_path)
            if database_path is not None
            else self.DEFAULT_DATABASE_PATH
        )

        path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        self.store = MemoryStore(
            database_path=path
        )

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> MemoryRecord:
        """
        Save an application memory.

        The legacy category is preserved in metadata while
        the persistence layer receives a supported MemoryType.
        """

        if not user_id.strip():
            raise ValueError(
                "Memory user_id cannot be empty."
            )

        if not category.strip():
            raise ValueError(
                "Memory category cannot be empty."
            )

        if not content.strip():
            raise ValueError(
                "Memory content cannot be empty."
            )

        created_at = datetime.now(
            timezone.utc
        )

        record = MemoryRecord(
            user_id=user_id,
            category=category,
            content=content,
            created_at=created_at,
        )

        memory_entry = MemoryEntry(
            user_id=user_id,
            memory_type=self._to_memory_type(
                category
            ),
            content=content,
            created_at=created_at,
            updated_at=created_at,
            metadata={
                "category": category,
            },
        )

        self.store.save(
            memory_entry
        )

        return record

    def recall(
        self,
        user_id: str,
    ) -> list[MemoryRecord]:
        """
        Retrieve memories for a user.

        Persistence-specific MemoryEntry objects are
        translated back into the application-facing
        MemoryRecord contract.
        """

        entries = self.store.list_for_user(
            user_id
        )

        return [
            self._to_memory_record(entry)
            for entry in entries
        ]

    @staticmethod
    def _to_memory_type(
        category: str,
    ) -> MemoryType:
        """
        Translate application categories into the
        supported AI Runtime memory types.

        The original category remains available in
        MemoryEntry.metadata.
        """

        normalized = category.strip().lower()

        if normalized == "saved_item":
            return MemoryType.SAVED_ITEM

        if normalized == "discovery_context":
            return MemoryType.DISCOVERY_CONTEXT

        return MemoryType.PREFERENCE

    @staticmethod
    def _to_memory_record(
        entry: MemoryEntry,
    ) -> MemoryRecord:
        """
        Translate a persistent MemoryEntry into the
        application-facing MemoryRecord.
        """

        category = entry.metadata.get(
            "category",
            entry.memory_type.value,
        )

        return MemoryRecord(
            user_id=entry.user_id,
            category=category,
            content=entry.content,
            created_at=entry.created_at,
        )

def build_flash_deal():
    """
    Build the same application-level FlashDealCard used
    by the hackathon discovery flow.
    """

    deal = build_demo_deal()
    memory = build_demo_memory()

    request = DiscoveryRequest(
        user_id=memory.user_id,
        deal=deal,
        character=CharacterContext(
            character_id="yeti",
            character_name="Yeti",
            message=(
                "Look what I found! "
                "This could be perfect for you."
            ),
        ),
        product_image_url=(
            "https://example.com/demo/iphone17-pro.jpg"
        ),
        tagline="Pro. Beautiful. Powerful.",
        highlights=(
            "Pro camera system",
            "High-performance chip",
            "All-day battery life",
            "Premium display",
        ),
    )

    card = DiscoveryService().create_flash_deal(
        request,
        memories=[memory],
    )

    return memory, card

from src.ai_runtime.commerce.flash_deal_commerce import (

from src.ai_runtime.commerce.models import (

    CommerceActionType,

def test_complete_hackathon_vertical_slice():

    # ---------------------------------------------------------------
    # MEMORY
    # ---------------------------------------------------------------

    memory = build_demo_memory()

    assert memory.user_id == "demo-user"

    assert memory.content

    # ---------------------------------------------------------------
    # DISCOVERY INPUT
    # ---------------------------------------------------------------

    deal = build_demo_deal()

    request = DiscoveryRequest(
        user_id=memory.user_id,
        deal=deal,
        character=CharacterContext(
            character_id="yeti",
            character_name="Yeti",
            message=(
                "Look what I found! "
                "This could be perfect for you."
            ),
        ),
        product_image_url=(
            "https://example.com/demo/iphone17-pro.jpg"
        ),
        tagline="Pro. Beautiful. Powerful.",
        highlights=(
            "Pro camera system",
            "High-performance chip",
            "All-day battery life",
            "Premium display",
        ),
    )

    # ---------------------------------------------------------------
    # DISCOVERY SERVICE
    # ---------------------------------------------------------------

    card = DiscoveryService().create_flash_deal(
        request,
        memories=[memory],
    )

    assert card.deal_id == deal.deal_id

    assert card.character_id == "yeti"

    assert card.product_name == deal.product_name

    assert card.vendor_name == deal.vendor

    assert card.relevance_reason

    # ---------------------------------------------------------------
    # COMMERCE ACTION
    # ---------------------------------------------------------------

    action = FlashDealCommerce().create_action(
        user_id=memory.user_id,
        card=card,
    )

    assert action.user_id == memory.user_id

    assert action.deal_id == card.deal_id

    assert (
        action.action_type
        == CommerceActionType.INITIATE_PURCHASE
    )

    assert (
        action.destination_url
        == card.get_deal_url
    )

    # ---------------------------------------------------------------
    # COMMERCE SERVICE
    # ---------------------------------------------------------------

    result = CommerceService().initiate(
        action
    )

    assert (
        result.status
        == CommerceActionStatus.REDIRECT
    )

    assert result.deal_id == card.deal_id

    assert result.action_id == action.action_id

    assert (
        result.next_url
        == card.get_deal_url

@dataclass
class FakeMemoryService:
    """
    Deterministic MemoryService test double.
    """

    memories: list[MemoryRecord]

    remembered: list[tuple[str, str, str]] | None = None

    def __post_init__(self) -> None:
        if self.remembered is None:
            self.remembered = []

    def recall(
        self,
        user_id: str,
    ) -> list[MemoryRecord]:
        return [
            memory
            for memory in self.memories
            if memory.user_id == user_id
        ]

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> None:
        assert self.remembered is not None

        self.remembered.append(
            (
                user_id,
                category,
                content,
            )
        )


class FakeResearchAgent:
    """
    Deterministic ResearchAgent test double.

    The agent produces the same provider-independent
    IntelligenceContext contract used by the real system.

    def __init__(self) -> None:
        self.received_contexts: list[
            ExecutionContext
        ] = []

    async def execute(

        self.received_contexts.append(context)

@dataclass
class FakeMemory:
    """
    Deterministic memory record for tests.
    """

    user_id: str
    category: str
    content: str


class FakeMemoryService:
    """
    Deterministic MemoryService test double.
    """

    def __init__(self) -> None:
        self.memories = [
            FakeMemory(
                user_id="demo-user",
                category="conversation",
                content="Previous conversation",
            ),
        ]

        self.recall_calls = []
        self.remember_calls = []

    def recall(
        self,
        user_id: str,
    ):
        self.recall_calls.append(
            user_id
        )

        return [
            memory
            for memory in self.memories
            if memory.user_id == user_id
        ]

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> None:
        self.remember_calls.append(
            {
                "user_id": user_id,
                "category": category,
                "content": content,
            }
        )

@pytest.mark.asyncio
async def test_companion_service_orchestrates_conversation():
    """
    CompanionService should execute the ConversationAgent
    and QualityAgent for a normal interaction.

    Research must not execute when use_research=False.
    """

    conversation_agent = FakeConversationAgent(
        AgentResult(
            success=True,
            agent_id="conversation_agent",
            output={
                "message": "Fake conversation response",
            },
        )
    )

    research_agent = FakeResearchAgent(
        AgentResult(
            success=True,
            agent_id="research_agent",
            output="Should not execute",
        )
    )

    quality_agent = FakeQualityAgent(
        AgentResult(
            success=True,
            agent_id="quality_agent",
            output={
                "quality_score": 1.0,
                "passed": True,
            },
        )
    )

    memory_service = FakeMemoryService()

    service = CompanionService(
        character=create_character(),
        conversation_agent=conversation_agent,
        research_agent=research_agent,
        quality_agent=quality_agent,
        memory_service=memory_service,
    )

    context = create_context()

    result = await service.interact(
        context,
        use_research=False,
    )

    assert result["character"] == "Yeti"

    assert result["conversation"] == {
        "message": "Fake conversation response",
    }

    assert result["research"] is None

    assert result["quality"] == {
        "quality_score": 1.0,
        "passed": True,
    }

    assert result["memory_count"] == 1

    assert (
        len(
            conversation_agent.received_contexts
        )
        == 1
    )

    assert (
        conversation_agent.received_contexts[0]
        is context
    )

    assert (
        len(
            research_agent.received_contexts
        )
        == 0
    )

    assert (
        len(
            quality_agent.received_contexts
        )
        == 1
    )

    assert (
        quality_agent.received_contexts[0]
        is context
    )

    assert memory_service.recall_calls == [
        "demo-user",
    ]

    assert memory_service.remember_calls == [
        {
            "user_id": "demo-user",
            "category": "conversation",
            "content": (
                "Explain artificial intelligence"
            ),
        }
    ]


@pytest.mark.asyncio
async def test_companion_service_executes_research_when_requested():
    """
    When research is requested, CompanionService should:

    1. execute ResearchAgent
    2. attach its IntelligenceContext to a new
       ExecutionContext
    3. execute ConversationAgent with the enriched context
    4. execute QualityAgent with the enriched context
    """

    intelligence_context = object()

    research_agent = FakeResearchAgent(
        AgentResult(
            success=True,
            agent_id="research_agent",
            output=intelligence_context,
        )
    )

    conversation_agent = FakeConversationAgent(
        AgentResult(
            success=True,
            agent_id="conversation_agent",
            output={
                "message": "Research-aware response",
            },
        )
    )

    quality_agent = FakeQualityAgent(
        AgentResult(
            success=True,
            agent_id="quality_agent",
            output={
                "quality_score": 0.95,
                "passed": True,
            },
        )
    )

    memory_service = FakeMemoryService()

    service = CompanionService(
        character=create_character(),
        conversation_agent=conversation_agent,
        research_agent=research_agent,
        quality_agent=quality_agent,
        memory_service=memory_service,
    )

    context = create_context(
        "What are the latest AI developments?"
    )

    result = await service.interact(
        context,
        use_research=True,
    )

    assert result["character"] == "Yeti"

    assert result["conversation"] == {
        "message": "Research-aware response",
    }

    assert (
        result["research"]
        is intelligence_context
    )

    assert result["quality"] == {
        "quality_score": 0.95,
        "passed": True,
    }

    assert (
        len(
            research_agent.received_contexts
        )
        == 1
    )

    assert (
        research_agent.received_contexts[0]
        is context
    )

    assert (
        len(
            conversation_agent.received_contexts
        )
        == 1
    )

    enriched_context = (
        conversation_agent.received_contexts[0]
    )

    assert enriched_context is not context

    assert (
        enriched_context.request
        == context.request
    )

    assert (
        enriched_context.user_id
        == context.user_id
    )

    assert (
        enriched_context.intelligence_context
        is intelligence_context
    )

    assert (
        quality_agent.received_contexts[0]
        is enriched_context
    )

@pytest.mark.asyncio
async def test_companion_service_handles_research_failure():
    """
    If research fails, ConversationAgent should still
    execute using the original context.

    The service should not attach failed research output
    to the conversation context.
    """

    research_agent = FakeResearchAgent(
        AgentResult(
            success=False,
            agent_id="research_agent",
            output=None,
        )
    )

    conversation_agent = FakeConversationAgent(
        AgentResult(
            success=True,
            agent_id="conversation_agent",
            output={
                "message": "Fallback response",
            },
        )
    )

    quality_agent = FakeQualityAgent(
        AgentResult(
            success=True,
            agent_id="quality_agent",
            output={
                "quality_score": 0.9,
                "passed": True,
            },
        )
    )

    memory_service = FakeMemoryService()

    service = CompanionService(
        character=create_character(),
        conversation_agent=conversation_agent,
        research_agent=research_agent,
        quality_agent=quality_agent,
        memory_service=memory_service,
    )

    context = create_context()

    result = await service.interact(
        context,
        use_research=True,
    )

    assert result["research"] is None

    assert result["conversation"] == {
        "message": "Fallback response",
    }

    assert (
        conversation_agent.received_contexts[0]
        is context
    )

    assert (
        conversation_agent.received_contexts[0]
        .intelligence_context
        is None
    )
