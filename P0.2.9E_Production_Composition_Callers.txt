6. External Runtime Container
============================================================

"""
External Runtime Container.

Stores the application-configured ExternalRuntimeService.

The container:

- stores the already-composed service;
- exposes the configured service;
- supports controlled test reset.

The container does NOT:

- compose Runtime dependencies;
- create providers;
- create gateways;
- execute capabilities;
- implement governance.

Composition remains the responsibility of application bootstrap.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from backend.services.external_runtime_service import (
        ExternalRuntimeService,
    )


_runtime_service: ExternalRuntimeService | None = None


def configure_external_runtime_service(
    service: ExternalRuntimeService,
) -> None:
    """
    Register the application-configured Runtime service.

    The service must already be fully composed.
    """

    global _runtime_service

    if service is None:
        raise ValueError(
            "ExternalRuntimeService cannot be None"
        )

    _runtime_service = service


def get_external_runtime_service() -> ExternalRuntimeService:
    """
    Return the configured Runtime service.

    Raises:
        RuntimeError:
            If the application has not been bootstrapped.
    """

    if _runtime_service is None:
        raise RuntimeError(
            "ExternalRuntimeService has not been configured."
        )

    return _runtime_service


def reset_runtime_container() -> None:
    """
    Reset the configured service.

    Intended for controlled test lifecycle management.
    """

    global _runtime_service

    _runtime_service = None

_bootstrapped = False


def bootstrap_application() -> None:
    """
    Bootstrap application dependencies.

    Creates the ExternalRuntimeService and registers it
    inside the application runtime container.
    """

    global _bootstrapped

    if _bootstrapped:
        return

    factory = ApplicationFactory()

    gateway = create_external_runtime_gateway(
        live_intelligence_agent=(
            create_live_intelligence_agent()
        ),
        podcast_discovery_agent=(
            create_podcast_discovery_agent()
        ),
        podcast_providers=(
            create_podcast_providers()
        ),
        governance=(
            create_governance()
        ),
        internal_documentation_agent=(
            create_internal_documentation_agent_from_repository()
        ),
    )

    runtime_service = (
        factory.create_external_runtime_service(
            gateway=gateway,
        )
    )

    configure_external_runtime_service(
        runtime_service,
    )

    _bootstrapped = True


def get_external_runtime_service() -> ExternalRuntimeService:
    """
    Backward-compatible Runtime accessor.
    """

    return _get_external_runtime_service()


def reset_bootstrap() -> None:
    """
    Test-only reset.
    """

    global _bootstrapped

    _bootstrapped = False

# Governance

class BootstrapGovernanceDecision:
    """
    Runtime governance decision.
    """

    def __init__(
        self,
        allowed: bool,
    ) -> None:

        self.allowed = allowed


class BootstrapGovernance:
    """
    Temporary hackathon governance implementation.

    Sensitive internal-data capabilities fail closed.

    The Runtime receives a provider-independent
    RuntimeActorContext. Application/API identity remains
    outside the Runtime boundary.
    """

    def evaluate(
        self,
        request,
    ) -> BootstrapGovernanceDecision:

        capability = getattr(
            request,
            "capability",
            None,
        )

        capability_value = getattr(
            capability,
            "value",
            capability,
        )

        if capability_value == "internal_data_read":
            actor = getattr(
                request,
                "actor",
                None,
            )

            if not isinstance(
                actor,
                RuntimeActorContext,
            ):
                return BootstrapGovernanceDecision(
                    allowed=False,
                )

            allowed = has_permission(
                roles=actor.roles,
                permission=Permission.INTERNAL_DATA_READ,
            )

            return BootstrapGovernanceDecision(
                allowed=allowed,
            )

        return BootstrapGovernanceDecision(
            allowed=True,
        )


def create_governance():
    """
    Create Runtime governance dependency.
    """

    return BootstrapGovernance()

8. Application Factory
============================================================

"""
Application Factory.

Creates application-layer services around the authoritative
external Runtime composition.

The factory delegates Runtime capability composition to
``src.ai_runtime.composition`` and wraps the resulting gateway
with application-layer services.

It does not create a second Runtime, registry, executor,
provider abstraction, or execution engine.
"""

from __future__ import annotations

from typing import Any, Sequence

from src.ai_runtime.capabilities.gateway import CapabilityGateway
from src.ai_runtime.composition import (
    create_external_runtime_gateway,
)

from backend.services.external_runtime_service import (
    ExternalRuntimeService,
)
from backend.services.runtime.runtime_application_service import (
    RuntimeApplicationService,
)


class ApplicationFactory:
    """Application composition factory."""

    @staticmethod
    def create_external_runtime(
        *,
        live_intelligence_agent: Any,
        podcast_discovery_agent: Any,
        podcast_providers: Sequence[Any],
        governance: Any,
        internal_documentation_agent: Any | None = None,
    ) -> ExternalRuntimeService:
        """
        Compose the authoritative external Runtime gateway and
        expose it through the application-facing service.

        Runtime internals remain owned by
        ``src.ai_runtime.composition``.

        This method does not create a parallel Runtime architecture.
        """

        gateway = create_external_runtime_gateway(
            live_intelligence_agent=live_intelligence_agent,
            podcast_discovery_agent=podcast_discovery_agent,
            podcast_providers=podcast_providers,
            governance=governance,
            internal_documentation_agent=(
                internal_documentation_agent
            ),
        )

        return ApplicationFactory.create_external_runtime_service(
            gateway=gateway,
        )

    @staticmethod
    def create_external_runtime_service(
        *,
        gateway: CapabilityGateway,
    ) -> ExternalRuntimeService:
        """
        Create the application-facing ExternalRuntimeService
        around an already-composed CapabilityGateway.
        """

        if gateway is None:
            raise ValueError(
                "CapabilityGateway cannot be None"
            )

        return ExternalRuntimeService(
            gateway=gateway,
        )

    @staticmethod
    def create_runtime_application_service(
        *,
        runtime_service: ExternalRuntimeService,
    ) -> RuntimeApplicationService:
        """
        Create the application RuntimeApplicationService.

        The RuntimeApplicationService receives an already-composed
        ExternalRuntimeService.

        This factory does not:

        - create providers;
        - create agents;
        - create gateways;
        - compose Runtime internals;
        - execute capabilities.
        """

        if runtime_service is None:
            raise ValueError(
                "ExternalRuntimeService cannot be None"
            )

        return RuntimeApplicationService(
            runtime_service=runtime_service,
        )
