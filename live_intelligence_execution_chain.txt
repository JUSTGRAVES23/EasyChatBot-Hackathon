class DemoSearchProvider:
    """
    Controlled search provider for the unified demo.

    The real Parallel provider is already validated separately
    by H1 integration tests.

    This keeps the unified demo deterministic and suitable for
    repeatable hackathon demonstrations.
    """

    async def search(
        self,
        query: str,
        max_results: int = 5,
    ) -> list[SearchResult]:

        return [
            SearchResult(
                title="European AI Act — Demo Source",
                url="https://example.com/ai-act",
                content=(
                    "The European Union AI Act establishes "
                    "a risk-based framework for artificial "
                    "intelligence."
                ),
                source="demo-provider",
            )
        ]


class DemoIntelligenceProcessor:
    """
    Controlled intelligence processor.

    The real provider-backed intelligence flow is validated
    separately in H1.
    """

    async def summarise(
        self,
        results: list[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, list[str]]:

        return (
            "The European Union AI Act establishes "
            "a risk-based framework for artificial "
            "intelligence.",
            [
                "AI governance",
                "risk management",
                "compliance",
            ],
        )


class DemoChallengeContentGenerator:
    """
    Controlled educational content generator.
    """

    async def generate_vocabulary(
        self,
        summary: str,
        language: str,
        learner_level: str,
    ) -> list[VocabularyItem]:

        return [
            VocabularyItem(
                term="gouvernance",
                definition=(
                    "La manière dont une organisation "
                    "est dirigée et contrôlée."
                ),
                example=(
                    "La gouvernance de l'IA devient "
                    "essentielle."
                ),
            ),
            VocabularyItem(
                term="réglementation",
                definition=(
                    "Ensemble des règles qui encadrent "
                    "une activité."
                ),
                example=(
                    "La réglementation européenne évolue."
                ),
            ),
        ]

    async def generate_questions(
        self,
        summary: str,
        language: str,
        learner_level: str,
    ) -> list[DiscussionQuestion]:

        return [
            DiscussionQuestion(
                question=(
                    "Pourquoi la réglementation de "
                    "l'IA est-elle importante ?"
                ),
                difficulty=learner_level,
            ),
            DiscussionQuestion(
                question=(
                    "Quels sont les avantages d'une "
                    "réglementation européenne de l'IA ?"
                ),
                difficulty=learner_level,
            ),
        ]


def create_governance_policy() -> GovernancePolicy:
    """
    Governance policy for the live intelligence demo.

    Live research may access external information but
    must never create external side effects.
    """

    return GovernancePolicy(
        capability="live_intelligence_research",
        risk_level=RiskLevel.MEDIUM,
        requires_human_review=False,
        allows_external_calls=True,
        allows_side_effects=False,
    )


async def create_learning_challenge() -> LearningChallenge:
    """
    Execute the H1 learning challenge flow.
    """

    intelligence_agent = LiveIntelligenceAgent(
        search_provider=DemoSearchProvider(),
        processor=DemoIntelligenceProcessor(),
    )

    intelligence = await intelligence_agent.execute(
        LiveIntelligenceRequest(
            topic="AI Act",
            language="French",
            learner_level="B2",
        )
    )

    challenge_generator = ChallengeGenerator(
        content_generator=DemoChallengeContentGenerator()
    )

    return await challenge_generator.generate(
        intelligence=intelligence,
        language="French",
        learner_level="B2",
    )


async def run_demo() -> None:
    """
    Execute the complete hackathon demonstration.
    """

    print()
    print("=" * 72)
    print("EasyChatBot — HACKATHON DEMONSTRATION")
    print("=" * 72)
    print()

    # -----------------------------------------------------------------------
    # STEP 1 — Governance
    # -----------------------------------------------------------------------

    print("STEP 1 — GOVERNANCE")
    print("-" * 72)

    governance_engine = GovernancePolicyEngine()

    governance_result = governance_engine.evaluate(
        create_governance_policy()
    )

    print(
        f"Capability: {governance_result.capability}"
    )

    print(
        f"Risk level: "
        f"{governance_result.risk_level.value}"
    )

    print(
        f"Decision: "
        f"{governance_result.decision.value.upper()}"
    )

    if governance_result.decision != GovernanceDecision.ALLOW:
        raise RuntimeError(
            "Hackathon demo stopped because governance "
            "did not allow the capability."
        )

    print("Governance check: PASSED")
    print()

    # -----------------------------------------------------------------------
    # STEP 2 — Live Intelligence + Challenge
    # -----------------------------------------------------------------------

    print("STEP 2 — LIVE INTELLIGENCE")
    print("-" * 72)

    challenge = await create_learning_challenge()

    print(f"Topic: {challenge.topic}")
    print(f"Language: {challenge.language}")
    print(f"Learner level: {challenge.learner_level}")
    print()

    print("Summary:")
    print(challenge.summary)
    print()

    print("Vocabulary:")

    for item in challenge.vocabulary:
        print(f"  • {item.term}: {item.definition}")

    print()

    print("Discussion questions:")

    for index, question in enumerate(
        challenge.discussion_questions,
        start=1,
    ):
        print(
            f"  {index}. {question.question}"
        )

    print()

    # -----------------------------------------------------------------------
    # STEP 3 — Quality Evaluation
    # -----------------------------------------------------------------------

    print("STEP 3 — AI QUALITY EVALUATION")
    print("-" * 72)

    evaluator = DeterministicEvaluator()

    criterion_scores = evaluator.evaluate(
        challenge
    )

    evaluation_engine = EvaluationEngine(
        criteria=[
            COMPLETENESS,
            LANGUAGE_LEVEL,
            EDUCATIONAL_VALUE,
            SOURCE_COVERAGE,
        ],
        passing_score=0.70,
    )

    evaluation_result = evaluation_engine.evaluate(
        criterion_scores
    )

    report = EvaluationReport().generate(
        evaluation_result
    )

    print(report)

    # -----------------------------------------------------------------------
    # STEP 4 — PERSONALISED FLASH DEAL
    # -----------------------------------------------------------------------

    print("STEP 4 — PERSONALISED DISCOVERY")
    print("-" * 72)

    # Build the same domain inputs that a real discovery
    # application flow would receive.
    demo_deal = build_demo_deal()
    demo_memory = build_demo_memory()

    discovery_request = DiscoveryRequest(
        user_id=demo_memory.user_id,
        deal=demo_deal,
        character=CharacterContext(
            character_id="yeti",
            character_name="Yeti",
            message=(
                "Look what I found! "
                "This could be perfect for you."
            ),
        ),
        product_image_url=(
            "https://example.com/demo/iphone17-pro.jpg"
        ),
        tagline="Pro. Beautiful. Powerful.",
        highlights=(
            "Pro camera system",
            "High-performance chip",
            "All-day battery life",
            "Premium display",
        ),
    )

    flash_deal_card = DiscoveryService().create_flash_deal(
        discovery_request,
        memories=[demo_memory],
    )

    print(
        render_demo_flash_deal_card(
            flash_deal_card
        )
    )

    assert flash_deal_card.character_id == "yeti"
    assert flash_deal_card.deal_id == demo_deal.deal_id
    assert flash_deal_card.product_name
    assert demo_memory.user_id == "demo-user"
    assert flash_deal_card.relevance_reason
    assert flash_deal_card.get_deal_url
    assert flash_deal_card.more_information_url

    # -----------------------------------------------------------------------
    # STEP 5 — COMMERCE ACTION
    # -----------------------------------------------------------------------

    print("STEP 5 — COMMERCE ACTION")
    print("-" * 72)

    commerce_action = FlashDealCommerce().create_action(
        user_id="demo-user",
        card=flash_deal_card,
    )

    print(
        f"Product: {flash_deal_card.product_name}"
    )
    print(
        f"Vendor: {flash_deal_card.vendor_name}"
    )
    print(
        f"Commerce action: "
        f"{commerce_action.action_type.value.upper()}"
    )

    commerce_result = CommerceService().initiate(
        commerce_action
    )

    print(
        f"Status: "
        f"{commerce_result.status.value.upper()}"
    )

    print()
    print("GET DEAL")
    print("  -> Approved commerce boundary")
    print("  -> Vendor handoff")
    print()
    print(
        "Payment is not executed in the hackathon demo."
    )

    assert (
        commerce_result.status
        == CommerceActionStatus.REDIRECT
    )
    assert (
        commerce_result.deal_id
        == flash_deal_card.deal_id
    )
    assert (
        commerce_result.next_url
        == flash_deal_card.get_deal_url
    )

    # -----------------------------------------------------------------------
    # STEP 6 — Final result
    # -----------------------------------------------------------------------

    print()
    print("=" * 72)

    if evaluation_result.passed:
        print("HACKATHON DEMO: SUCCESS")
        print()
        # ASCII-safe output keeps the executable reliable when
        # stdout is captured by Windows pytest/subprocess.
        # Preserve the stable hackathon readiness marker.
        print(
            "Governance -> Intelligence -> Learning -> Evaluation"
        )
        print(
            "Governance -> Intelligence -> Learning -> "
            "Evaluation -> Personalised Discovery"
        )
        print(
            "All demonstrated capabilities completed successfully."
        )
    else:
        print("HACKATHON DEMO: QUALITY CHECK FAILED")

    print("=" * 72)
    print()

    assert governance_result.decision == GovernanceDecision.ALLOW
    assert challenge.topic == "AI Act"
    assert evaluation_result.passed is True


def main() -> None:
    """
    Application entry point.
    """

    asyncio.run(run_demo())


if __name__ == "__main__":
    main()

"""
Live Intelligence Agent Handler.

Responsible for executing the Live Intelligence Agent workflow.

Architecture:

AI Runtime
    |
    v
Live Intelligence Agent Handler
    |
    +--> Search Provider Interface
    |
    +--> Knowledge Processing
    |
    v
Structured Intelligence Output

Responsibilities:
- Validate agent input
- Call approved provider interfaces
- Normalise research output
- Return explicit structured results

Non-responsibilities:
- Provider SDK management
- Direct external API handling
- Autonomous actions
- Configuration changes
"""

from dataclasses import dataclass
from typing import List, Protocol

from src.providers.search_provider.interface import (
    SearchProvider,
    SearchResult,
)


@dataclass(frozen=True)
class LiveIntelligenceRequest:
    """
    Input contract for Live Intelligence Agent.
    """

    topic: str
    language: str = "French"
    learner_level: str = "B2"


@dataclass(frozen=True)
class LiveIntelligenceResponse:
    """
    Output contract from Live Intelligence Agent.
    """

    topic: str
    summary: str
    sources: List[SearchResult]
    key_information: List[str]


class IntelligenceProcessor(Protocol):
    """
    Contract for future LLM-based processing.

    Keeps the agent independent from
    any specific model/provider.
    """

    async def summarise(
        self,
        results: List[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, List[str]]:
        ...


class LiveIntelligenceAgent:
    """
    Live Intelligence Agent implementation.
    """

    def __init__(
        self,
        search_provider: SearchProvider,
        processor: IntelligenceProcessor,
    ) -> None:
        self._search_provider = search_provider
        self._processor = processor

    async def execute(
        self,
        request: LiveIntelligenceRequest,
    ) -> LiveIntelligenceResponse:
        """
        Execute research workflow.

        Flow:

        Request
          |
          v
        Search Provider
          |
          v
        Intelligence Processor
          |
          v
        Structured Response
        """

        self._validate_request(request)

        search_results = await self._search_provider.search(
            query=request.topic,
            max_results=5,
        )

        summary, key_information = await self._processor.summarise(
            results=search_results,
            language=request.language,
            learner_level=request.learner_level,
        )

        return LiveIntelligenceResponse(
            topic=request.topic,
            summary=summary,
            sources=search_results,
            key_information=key_information,
        )

    @staticmethod
    def _validate_request(
        request: LiveIntelligenceRequest,
    ) -> None:
        """
        Validate required input.
        """

        if not request.topic.strip():
            raise ValueError(
                "Topic cannot be empty"
            )

"""
AI Runtime capability execution contract.

This module defines the interface between the AI Runtime
and concrete capability executors.

Architecture:

AI Runtime
    |
    v
Capability Executor Interface
    |
    +---- Live Intelligence Executor
    +---- Learning Executor
    +---- Evaluation Executor
    +---- Future Executors

The Runtime depends on this contract.

It does not depend directly on:
- agents
- providers
- SDKs
- external APIs

Concrete executors are responsible for translating the
provider-independent CapabilityRequest into the appropriate
capability implementation.
"""

from __future__ import annotations

from abc import ABC, abstractmethod

from src.ai_runtime.capabilities.models import (
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)


class CapabilityExecutor(ABC):
    """
    Abstract executor for one Runtime capability.

    Each concrete executor must declare which capability
    it handles and implement asynchronous execution.
    """

    @property
    @abstractmethod
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """
        raise NotImplementedError

    @abstractmethod
    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a capability request.

        Implementations must return a provider-independent
        CapabilityResult.

        They must not bypass the Runtime governance boundary.
        """
        raise NotImplementedError

"""
AI Runtime Capability Gateway.

Controlled execution boundary between applications and
concrete AI capabilities.

Flow:

Application
    ↓
CapabilityRequest

CapabilityGateway

"""

from __future__ import annotations

from typing import Protocol

from src.ai_runtime.capabilities.errors import (
    CapabilityExecutionError,
    CapabilityGovernanceError,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
)

from src.ai_runtime.capabilities.registry import (
    CapabilityRegistry,
)


class GovernanceDecisionResult(Protocol):
    """
    Minimal governance result required by the Runtime.
    """

    @property
    def allowed(self) -> bool:
        """
        Whether execution is permitted.
        """
        ...


class CapabilityGovernance(Protocol):
    """
    Governance boundary required by the Runtime gateway.
    """

    def evaluate(
        self,
        request: CapabilityRequest,
    ) -> GovernanceDecisionResult:
        """
        Evaluate whether a capability may execute.
        """
        ...


class CapabilityGateway:
    """
    Controlled Runtime entry point for capability execution.
    """

    def __init__(
        self,
        registry: CapabilityRegistry,
        governance: CapabilityGovernance,
    ) -> None:
        self._registry = registry
        self._governance = governance

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute an approved capability.

        Governance is evaluated before resolving and executing
        the capability.
        """

        try:
            governance_result = (
                self._governance.evaluate(request)
            )
        except Exception as exc:
            raise CapabilityGovernanceError(
                "Governance evaluation failed."
            ) from exc

        if not governance_result.allowed:
            return CapabilityResult(
                request_id=request.request_id,
                capability=request.capability,

"""
AI Runtime executor for Live Intelligence.

This adapter bridges the provider-independent AI Runtime
capability contract with the existing LiveIntelligenceAgent.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
LiveIntelligenceExecutor
    |
    v
LiveIntelligenceAgent
    |
    v
LiveIntelligenceResponse

The adapter:

- validates the Runtime capability
- translates Runtime payloads into LiveIntelligenceRequest
- invokes the existing Live Intelligence Agent
- translates the agent response into CapabilityResult

It does not:

- implement intelligence logic
- call Parallel directly
- modify the LiveIntelligenceAgent
- perform governance decisions
"""

from __future__ import annotations

from src.ai_agents.live_intelligence_agent.handler import (
    LiveIntelligenceAgent,
    LiveIntelligenceRequest,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)


class LiveIntelligenceExecutor(
    CapabilityExecutor,
):
    """
    Runtime adapter for the existing LiveIntelligenceAgent.
    """

    def __init__(
        self,
        agent: LiveIntelligenceAgent,
    ) -> None:
        self._agent = agent

    @property
    def capability(self) -> CapabilityType:
        """
        Return the capability handled by this executor.
        """

        return CapabilityType.LIVE_INTELLIGENCE

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Translate and execute a Live Intelligence request.
        """

        if (
            request.capability
            != CapabilityType.LIVE_INTELLIGENCE
        ):
            raise ValueError(
                "LiveIntelligenceExecutor received "
                "an unsupported capability: "
                f"{request.capability.value}"
            )

        agent_request = (
            self._build_agent_request(request)
        )

        response = await self._agent.execute(
            agent_request
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=response,
            metadata={
                "executor": (
                    "live_intelligence_executor"
                ),
            },
        )

    @staticmethod
    def _build_agent_request(
        request: CapabilityRequest,
    ) -> LiveIntelligenceRequest:
        """
        Translate Runtime payload into the existing
        LiveIntelligenceRequest contract.
        """

        payload = request.payload

        topic = payload.get("topic")

        if not isinstance(topic, str):
            raise ValueError(
                "Live Intelligence capability requires "
                "a string 'topic'."
            )

        language = payload.get(
            "language",
            "French",
        )

        learner_level = payload.get(
            "learner_level",
            "B2",
        )

        if not isinstance(language, str):
            raise ValueError(
                "Live Intelligence 'language' "
                "must be a string."
            )

        if not isinstance(
            learner_level,
            str,
        ):
            raise ValueError(
                "Live Intelligence 'learner_level' "
                "must be a string."
            )

        return LiveIntelligenceRequest(
            topic=topic,
            language=language,
            learner_level=learner_level,
        )

"""
AI Runtime capability registry.

The registry maps logical Runtime capabilities to their
concrete executors.

The registry:
- resolves executors by capability
- prevents duplicate registrations
- provides explicit dependency injection
- does not execute capabilities
- does not perform governance decisions
- does not know about providers
"""

from __future__ import annotations

from src.ai_runtime.capabilities.errors import (
    CapabilityAlreadyRegisteredError,
    CapabilityNotRegisteredError,
)

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityType,
)


class CapabilityRegistry:
    """
    Dependency-injection registry for Runtime capabilities.
    """

    def __init__(
        self,
        executors: list[CapabilityExecutor] | None = None,
    ) -> None:
        self._executors: dict[
            CapabilityType,
            CapabilityExecutor,
        ] = {}

        for executor in executors or []:
            self.register(executor)

    def register(
        self,
        executor: CapabilityExecutor,
    ) -> None:
        """
        Register an executor for its declared capability.
        """

        capability = executor.capability

        if capability in self._executors:
            raise CapabilityAlreadyRegisteredError(
                "Capability executor already registered: "
                f"{capability.value}"
            )

        self._executors[capability] = executor

    def resolve(
        self,
        capability: CapabilityType,
    ) -> CapabilityExecutor:
        """
        Resolve the executor responsible for a capability.
        """

        try:
            return self._executors[capability]
        except KeyError as exc:
            raise CapabilityNotRegisteredError(
                "No executor registered for capability: "
                f"{capability.value}"
            ) from exc

    def contains(
        self,
        capability: CapabilityType,
    ) -> bool:
        """
        Return whether an executor is registered.
        """

        return capability in self._executors

    def capabilities(
        self,
    ) -> tuple[CapabilityType, ...]:
        """
        Return all registered capabilities.
        """

        return tuple(self._executors.keys())

"""
Integration test for Live Challenge vertical slice.

Flow tested:

Search Provider
        ↓
Live Intelligence Agent
        ↓
Challenge Generator
        ↓
Learning Challenge

This test validates the Sprint H1 integration boundaries.

Important:
- Uses test doubles only.
- Does not call the real Parallel API.
- Does not modify or depend on the Conversation Agent.
"""

import pytest

from src.ai_agents.live_intelligence_agent.handler import (
    LiveIntelligenceAgent,
    LiveIntelligenceRequest,
)

from src.challenge_engine.generator import (
    ChallengeGenerator,
)

from src.challenge_engine.models import (
    DiscussionQuestion,
    VocabularyItem,
)

from src.providers.search_provider.interface import (
    SearchResult,
)


class FakeSearchProvider:
    """
    Controlled search provider for integration testing.
    """

    async def search(
        self,
        query: str,
        max_results: int = 5,
    ) -> list[SearchResult]:
        return [
            SearchResult(
                title="AI Act Update",
                url="https://example.com/ai-act",
                content="European AI regulation update.",
                source="test-provider",
            )
        ]


class FakeIntelligenceProcessor:
    """
    Simulates intelligence synthesis without
    requiring an LLM provider.
    """

    async def summarise(
        self,
        results: list[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, list[str]]:
        return (
            "The AI Act introduces new rules for "
            "artificial intelligence.",
            [
                "AI governance",
                "risk management",
            ],
        )


class FakeChallengeContentGenerator:
    """
    Simulates educational content generation.
    """

    async def generate_vocabulary(
        self,
        summary: str,
        language: str,
        learner_level: str,
    ) -> list[VocabularyItem]:
        return [
            VocabularyItem(
                term="gouvernance",
                definition=(
                    "Organisation and control of activities."
                ),
            )
        ]

    async def generate_questions(
        self,
        summary: str,
        language: str,
        learner_level: str,
    ) -> list[DiscussionQuestion]:
        return [
            DiscussionQuestion(
                question=(
                    "Pourquoi la gouvernance de l'IA "
                    "est-elle importante ?"
                ),
            )
        ]


@pytest.mark.asyncio
async def test_live_challenge_vertical_slice():
    """
    Validate the complete Sprint H1 backend flow:

    Search Provider
        ↓
    Live Intelligence Agent
        ↓
    Challenge Generator
        ↓
    Learning Challenge
    """

    # ---------------------------------------------------------
    # Step 1 — Create Live Intelligence Agent
    # ---------------------------------------------------------

    intelligence_agent = LiveIntelligenceAgent(
        search_provider=FakeSearchProvider(),
        processor=FakeIntelligenceProcessor(),
    )

    # ---------------------------------------------------------
    # Step 2 — Execute Live Intelligence research
    # ---------------------------------------------------------

    intelligence = await intelligence_agent.execute(
        request=LiveIntelligenceRequest(
            topic="AI Act",
        )
    )

    # ---------------------------------------------------------
    # Step 3 — Validate intelligence output
    # ---------------------------------------------------------

    assert intelligence.topic == "AI Act"

    assert intelligence.summary == (
        "The AI Act introduces new rules for "
        "artificial intelligence."
    )

    assert intelligence.key_information == [
        "AI governance",
        "risk management",
    ]

    assert len(intelligence.sources) == 1
    assert intelligence.sources[0].title == "AI Act Update"
    assert intelligence.sources[0].source == "test-provider"

    # ---------------------------------------------------------
    # Step 4 — Create Challenge Generator
    # ---------------------------------------------------------

    challenge_generator = ChallengeGenerator(
        content_generator=FakeChallengeContentGenerator()
    )

    # ---------------------------------------------------------
    # Step 5 — Generate learning challenge
    # ---------------------------------------------------------

    challenge = await challenge_generator.generate(
        intelligence=intelligence,
        language="French",
        learner_level="B2",
    )

    # ---------------------------------------------------------
    # Step 6 — Validate final learning output
    # ---------------------------------------------------------

    assert challenge.title == (
        "AI Act Learning Challenge"
    )

    assert challenge.topic == "AI Act"
    assert challenge.language == "French"
    assert challenge.learner_level == "B2"

    assert challenge.summary == (
        "The AI Act introduces new rules for "
        "artificial intelligence."
    )

    assert len(challenge.vocabulary) == 1

    assert challenge.vocabulary[0].term == (
        "gouvernance"
    )

    assert (
        challenge.vocabulary[0].definition
        == "Organisation and control of activities."
    )

    assert len(challenge.discussion_questions) == 1

    assert (
        challenge.discussion_questions[0].question
        == "Pourquoi la gouvernance de l'IA est-elle importante ?"
    )

    assert (
        challenge.discussion_questions[0].difficulty
        == "B2"
    )

    assert (
        challenge.created_by_agent
        == "live_intelligence_agent"
    )
