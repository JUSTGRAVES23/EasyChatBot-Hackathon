import logging

from runtime.agents.framework.base_agent import BaseAgent

class PlatformIntelligenceAgent(BaseAgent):
    """
    Runtime adapter for the Platform Intelligence capability.

    BaseAgent owns the Runtime lifecycle.

    Platform Intelligence owns:
    - investigation reasoning
    - evidence-aware assessment
    - structured investigation output

    The agent does not:
    - call providers directly
    - access databases directly
    - access repositories directly
    - execute external actions
    - orchestrate other agents
    """

    CAPABILITY_ID = "platform_intelligence"

    def __init__(
        self,
        *,
        metadata: AgentMetadata,
        execution_service: AgentExecutionService,
    ) -> None:
        if execution_service is None:
            raise ValueError(
                "AgentExecutionService cannot be None."
            )

        super().__init__(
            metadata=metadata,
        )

        self._execution_service = execution_service

    async def run(
        self,
        context: RuntimeContext,
    ) -> ExecutionResult:
        """
        Execute the Platform Intelligence capability.

        BaseAgent.execute() owns the surrounding lifecycle.
        """

        if context is None:
            raise ValueError(
                "RuntimeContext cannot be None."
            )

        logger.info(
            (
                "Executing Platform Intelligence. "
                "execution_id=%s"
            ),
            context.execution_id,
        )

        request = self._build_investigation_request(
            context,
        )

        plan = self._build_initial_plan(
            request,
        )

        agent_request = AgentExecutionRequest(
            agent_id=self.agent_id,
            capability=self.CAPABILITY_ID,
            instruction=request.question,
            context={
                "investigation_id": request.investigation_id,
                "scope": request.scope,
                "depth": request.depth,
                "constraints": request.constraints,
                "investigation_plan": {
                    "steps": plan.steps,
                    "required_evidence": plan.required_evidence,
                    "max_skill_calls": plan.max_skill_calls,
                    "max_sources": plan.max_sources,
                    "max_execution_seconds": (
                        plan.max_execution_seconds
                    ),
                },
            },
            metadata={
                "capability_id": self.CAPABILITY_ID,
                "investigation_id": request.investigation_id,
                "execution_mode": "read_only",
            },
        )

        ai_response = await self._execution_service.execute(
            request=agent_request,
            context=context,
        )

        if ai_response is None:
            raise RuntimeError(
                "AgentExecutionService returned no response."
            )

        assessment = OutputAssessment(
            output_id="investigation_assessment",
            value=ai_response.content,
            confidence_score=0.80,
            confidence_level="HIGH",
            evidence=(),
            rationale=(
                "Initial investigation reasoning has been generated. "
                "Evidence-producing skills have not yet been attached."
            ),
            quality_status="UNASSESSED",
        )

        result = InvestigationResult(
            investigation_id=request.investigation_id,
            request=request,
            plan=plan,
            evidence=(),
            output_assessments=(assessment,),
            executive_summary=ai_response.content,
            metadata={
                "agent": self.agent_id,
                "capability": self.CAPABILITY_ID,
                "execution_mode": "read_only",
                "evidence_status": "not_yet_collected",
            },
        )

        result.validate()

        logger.info(
            (
                "Platform Intelligence execution completed. "
                "execution_id=%s investigation_id=%s"
            ),
            context.execution_id,
            request.investigation_id,
        )

        return ExecutionResult.success_result(
            execution_id=context.execution_id,
            agent_id=self.agent_id,
            output=result,
            metadata={
                "agent": self.agent_id,
                "capability": self.CAPABILITY_ID,
                "investigation_id": request.investigation_id,
            },
        )

    def _build_investigation_request(
        self,
        context: RuntimeContext,
    ) -> InvestigationRequest:
        """
        Build an investigation request from Runtime context.
        """

        question = ""

        if context.work_item is not None:
            question = context.work_item.description

        if not question.strip():
            question = context.metadata.get(
                "user_request",
                "",
            )

        if not isinstance(question, str) or not question.strip():
            raise ValueError(
                "Platform Intelligence requires an investigation question."
            )

        request = InvestigationRequest(
            investigation_id=str(context.execution_id),
            question=question,
            scope=context.metadata.get(
                "investigation_scope",
                "platform",
            ),
            depth=context.metadata.get(
                "investigation_depth",
                "standard",
            ),
            constraints=tuple(
                context.metadata.get(
                    "investigation_constraints",
                    (),
                )
            ),
            metadata={
                "agent_id": self.agent_id,
                "execution_id": str(context.execution_id),
            },
        )

        request.validate()

        return request

    def _build_initial_plan(
        self,
        request: InvestigationRequest,
    ) -> InvestigationPlan:
        """
        Build the initial governed investigation plan.

        Evidence-producing skills will be connected in a later
        vertical slice. The agent does not perform uncontrolled
        retrieval.
        """

        steps = (
            "understand_question",
            "identify_required_evidence",
            "analyse_available_evidence",
            "assess_findings",
            "identify_evidence_gaps",
            "produce_investigation_result",
        )

        required_evidence = (
            "source",
            "source_authority",
            "source_provenance",
        )

        if request.depth == "deep":
            steps = (
                "understand_question",
                "identify_required_evidence",
                "broad_discovery",
                "cross_source_analysis",
                "analyse_available_evidence",
                "identify_contradictions",
                "assess_findings",
                "identify_evidence_gaps",
                "quality_review",
                "produce_investigation_result",
            )

        plan = InvestigationPlan(
            investigation_id=request.investigation_id,
            steps=steps,
            required_evidence=required_evidence,
            max_skill_calls=10 if request.depth == "standard" else 20,
            max_sources=20 if request.depth == "standard" else 50,
            max_execution_seconds=300 if request.depth == "standard" else 600,
            rationale=(
                "Initial governed investigation plan. "
                "Evidence retrieval remains Runtime/Skill controlled."
            ),
        )

        plan.validate()

        return plan

class AgentExecutionService:
    """
    Runtime service responsible for agent reasoning execution.

    Coordinates:

    - Prompt generation.
    - Provider execution.
    - Response normalization.


    Does not own:

    - Agent lifecycle.
    - Memory.
    - Agent discovery.
    - Provider implementations.
    """



    def __init__(
        self,
        *,
        prompt_builder: RuntimePromptBuilder,
        provider_runtime: ProviderRuntime,
        builder_id: str,
    ) -> None:
        """
        Initialise AgentExecutionService.

        Parameters
        ----------
        prompt_builder:
            Runtime prompt generation service.

        provider_runtime:
            Runtime provider boundary.

        builder_id:
            Registered prompt builder identifier.
        """

        if prompt_builder is None:

            raise ValueError(
                "RuntimePromptBuilder cannot be None."
            )


        if provider_runtime is None:

            raise ValueError(
                "ProviderRuntime cannot be None."
            )


        if not isinstance(
            builder_id,
            str,
        ):

            raise TypeError(
                "builder_id must be string."
            )


        if not builder_id.strip():

            raise ValueError(
                "builder_id cannot be empty."
            )


        self._prompt_builder = prompt_builder

        self._provider_runtime = provider_runtime

        self._builder_id = builder_id


        logger.info(
            (
                "AgentExecutionService initialised. "
                "builder_id=%s"
            ),
            builder_id,
        )



    async def execute(
        self,
        *,
        request: AgentExecutionRequest,
        context: RuntimeContext,
    ) -> AgentExecutionResponse:
        """
        Execute agent reasoning request.
        """

        if request is None:

            raise ValueError(
                "AgentExecutionRequest cannot be None."
            )


        if context is None:

            raise ValueError(
                "RuntimeContext cannot be None."
            )


        request.validate_request()



        logger.info(
            (
                "Agent reasoning started. "
                "agent=%s capability=%s execution_id=%s"
            ),
            request.agent_id,
            request.capability,
            context.execution_id,
        )



        prompt = self._prompt_builder.build(

            builder_id=self._builder_id,

            context=context,

        )



        if not prompt.strip():

            raise ValueError(
                "Generated prompt cannot be empty."
            )



        context.rendered_prompt = prompt



        result = await self._provider_runtime.execute(
            context=context,
        )



        response = AgentExecutionResponse(

            execution_id=context.execution_id,

            agent_id=request.agent_id,

            capability=request.capability,

            content=result.content,

            metadata={

                "provider_id": result.provider_id.value,

                "model": (
                    result.display_model_name
                ),

            },

        )



        response.validate_response()



        logger.info(
            (
                "Agent reasoning completed. "
                "agent=%s execution_id=%s"
            ),
            request.agent_id,
            context.execution_id,
        )


        return response

from __future__ import annotations

import logging

from runtime.agents.services.agent_execution_service import (
    AgentExecutionService,
)
from runtime.capabilities.registries.skill_registry import (
    SkillRegistry,
)
from runtime.contracts.agent_execution_request import (
    AgentExecutionRequest,
)
from runtime.contracts.agent_execution_response import (
    AgentExecutionResponse,
)
from runtime.core.runtime_context import RuntimeContext


logger = logging.getLogger(__name__)


class SkillExecutionService:
    """
    Runtime-controlled execution boundary for Skills.

    Authorization is reusable independently from direct Skill execution.

    Skill manifests may use governed agent aliases for compatibility.
    Canonical Runtime agent IDs remain authoritative internally.
    """

    AGENT_COMPATIBILITY_ALIASES: dict[str, tuple[str, ...]] = {
        "project_manager": ("aurelius",),
        "platform_intelligence_agent": ("romanoide",),
    }

    def __init__(
        self,
        *,
        skill_registry: SkillRegistry,
        agent_execution_service: AgentExecutionService,
    ) -> None:
        if skill_registry is None:
            raise ValueError("SkillRegistry cannot be None.")

        if agent_execution_service is None:
            raise ValueError(
                "AgentExecutionService cannot be None."
            )

        self._skill_registry = skill_registry
        self._agent_execution_service = agent_execution_service

        logger.info("SkillExecutionService initialised.")

    @classmethod
    def _agent_is_compatible(
        cls,
        *,
        agent_id: str,
        compatible_agents: tuple[str, ...] | list[str],
    ) -> bool:
        """
        Determine whether a canonical Runtime agent is authorised by
        the Skill's declared compatibility identifiers.

        The Skill manifest remains authoritative for the aliases it
        declares. Canonical Runtime IDs are never replaced.
        """

        if agent_id in compatible_agents:
            return True

        aliases = cls.AGENT_COMPATIBILITY_ALIASES.get(
            agent_id,
            (),
        )

        return any(
            alias in compatible_agents
            for alias in aliases
        )

    def authorize(
        self,
        *,
        skill_id: str,
        agent_id: str,
        context: RuntimeContext,
    ):
        """
        Resolve and authorize a Skill for an Agent.

        This method performs governance and Runtime-context enrichment
        only. It does not execute the Agent or call a provider.
        """

        if not isinstance(skill_id, str) or not skill_id.strip():
            raise ValueError("skill_id cannot be empty.")

        if not isinstance(agent_id, str) or not agent_id.strip():
            raise ValueError("agent_id cannot be empty.")

        if context is None:
            raise ValueError("RuntimeContext cannot be None.")

        manifest = self._skill_registry.get(skill_id)

        if manifest is None:
            raise KeyError(
                f"Skill '{skill_id}' is not registered."
            )

        manifest.validate()

        if manifest.status.lower() != "active":
            raise PermissionError(
                f"Skill '{skill_id}' is not active "
                "and cannot be executed."
            )

        if manifest.compatible_agents:
            if not self._agent_is_compatible(
                agent_id=agent_id,
                compatible_agents=manifest.compatible_agents,
            ):
                raise PermissionError(
                    f"Agent '{agent_id}' is not authorised "
                    f"to execute Skill '{skill_id}'."
                )

        context.add_metadata("skill_id", manifest.id)
        context.add_metadata("skill_name", manifest.name)
        context.add_metadata(
            "skill_description",
            manifest.description,
        )

        context.add_metadata(
            "skill_prompt",
            self._skill_registry.load_prompt(manifest.id),
        )

        context.add_metadata("skill_version", manifest.version)
        context.add_metadata("skill_status", manifest.status)
        context.add_metadata("skill_owner", manifest.owner)
        context.add_metadata(
            "skill_execution_mode",
            "read_only",
        )

        context.add_event(
            "Runtime Skill authorization validated.",
            metadata={
                "skill_id": manifest.id,
                "agent_id": agent_id,
            },
        )

        logger.info(
            (
                "Runtime Skill authorization validated. "
                "skill=%s agent=%s execution_id=%s"
            ),
            manifest.id,
            agent_id,
            context.execution_id,
        )

        return manifest

    async def execute(
        self,
        *,
        skill_id: str,
        request: AgentExecutionRequest,
        context: RuntimeContext,
    ) -> AgentExecutionResponse:
        """
        Execute a Runtime-authorised Skill.

        Authorization is performed first; Agent execution remains owned
        by AgentExecutionService.
        """

        if request is None:
            raise ValueError(
                "AgentExecutionRequest cannot be None."
            )

        if context is None:
            raise ValueError(
                "RuntimeContext cannot be None."
            )

        request.validate_request()

        manifest = self.authorize(
            skill_id=skill_id,
            agent_id=request.agent_id,
            context=context,
        )

        logger.info(
            (
                "Executing governed Skill. "
                "skill=%s agent=%s execution_id=%s"
            ),
            manifest.id,
            request.agent_id,
            context.execution_id,
        )

        response = await self._agent_execution_service.execute(
            request=request,
            context=context,
        )

        if response is None:
            raise RuntimeError(
                "AgentExecutionService returned no response."
            )

        context.add_event(
            "Runtime Skill execution completed.",
            metadata={
                "skill_id": manifest.id,
                "agent_id": request.agent_id,
            },
        )

        return response

class AgentResolver:
    """
    Resolves AI Agent metadata and executable AI Agents.

    The resolver is the Runtime boundary responsible for transforming
    an agent identifier into either:

    - authoritative AgentMetadata, or
    - an executable Agent instance.

    It does not make authorization decisions.
    """

    def __init__(
        self,
        *,
        registry: AgentRegistry,
        factory: AgentFactory,
    ) -> None:
        """
        Initialise AgentResolver.

        Parameters
        ----------
        registry:
            Authoritative Agent metadata registry.

        factory:
            Agent construction factory.

        Raises
        ------
        ValueError:
            If a required dependency is missing.
        """

        if registry is None:
            raise ValueError(
                "AgentRegistry cannot be None."
            )

        if factory is None:
            raise ValueError(
                "AgentFactory cannot be None."
            )

        self._registry = registry
        self._factory = factory

    @staticmethod
    def _validate_agent_id(
        agent_id: str,
    ) -> None:
        """
        Validate an agent identifier.
        """

        if not agent_id:
            raise ValueError(
                "Agent id cannot be empty."
            )

        if not isinstance(agent_id, str):
            raise ValueError(
                "Agent id must be a string."
            )

        if not agent_id.strip():
            raise ValueError(
                "Agent id cannot contain only whitespace."
            )

    def resolve_metadata(
        self,
        agent_id: str,
    ) -> Any:
        """
        Resolve authoritative AgentMetadata without constructing
        an executable agent.

        This method exists so Runtime governance stages can inspect
        the authoritative agent definition before execution.

        Parameters
        ----------
        agent_id:
            Unique agent identifier.

        Returns
        -------
        AgentMetadata
            Authoritative metadata registered for the agent.
        """

        self._validate_agent_id(agent_id)

        logger.info(
            "Resolving AI Agent metadata. agent_id=%s",
            agent_id,
        )

        metadata = self._registry.resolve(
            agent_id,
        )

        if metadata is None:
            raise ValueError(
                f"Agent metadata could not be resolved: {agent_id}"
            )

        return metadata

    def resolve(
        self,
        agent_id: str,
    ) -> Any:
        """
        Resolve an executable AI Agent.

        Parameters
        ----------
        agent_id:
            Unique agent identifier.

        Returns
        -------
        object
            Executable AI Agent instance.
        """

        self._validate_agent_id(agent_id)

        logger.info(
            "Resolving executable AI Agent. agent_id=%s",
            agent_id,
        )

        metadata = self.resolve_metadata(
            agent_id,
        )

        agent = self._factory.create(
            metadata,
        )

        if agent is None:
            raise TypeError(
                f"AgentFactory returned None for agent: {agent_id}"
            )

        logger.info(
            "AI Agent resolved successfully. agent_id=%s",
            agent_id,
        )

        return agent

class AgentComposer:
    """
    Agent subsystem composition root.

    Responsible only for constructing Runtime Agent infrastructure.
    """

    @classmethod
    def create(
        cls,
        *,
        execution_service: AgentExecutionService,
        execution_services: dict[
            str,
            AgentExecutionService,
        ] | None = None,
    ) -> AgentComposition:
        """
        Create complete Agent subsystem.

        execution_service remains the default execution boundary.

        execution_services provides explicit per-agent execution
        boundaries for agents requiring specialized Runtime configuration.
        """

        if execution_service is None:
            raise ValueError(
                "AgentExecutionService cannot be None."
            )

        if execution_services is None:
            execution_services = {
                "project_manager": execution_service,
                "platform_intelligence_agent": execution_service,
            }
        else:
            execution_services = dict(execution_services)

            if "project_manager" not in execution_services:
                execution_services["project_manager"] = execution_service

        for agent_id, service in execution_services.items():
            if not isinstance(agent_id, str) or not agent_id.strip():
                raise ValueError(
                    "execution_services contains an invalid agent id."
                )

            if service is None:
                raise ValueError(
                    f"Execution service for '{agent_id}' cannot be None."
                )

            if not isinstance(service, AgentExecutionService):
                raise TypeError(
                    (
                        f"Execution service for '{agent_id}' "
                        "must be AgentExecutionService."
                    )
                )

        logger.info(
            "Creating Agent subsystem."
        )

        registry, factory = AgentBootstrap.create_services(
            execution_service=execution_service,
            execution_services=execution_services,
        )

        resolver = AgentResolver(
            registry=registry,
            factory=factory,
        )

        logger.info(
            (
                "Agent subsystem created. "
                "agents=%s execution_services=%s"
            ),
            registry.count(),
            tuple(execution_services.keys()),
        )

        return AgentComposition(
            registry=registry,
            factory=factory,
            resolver=resolver,
        )

class RuntimeComposer:
    """
    Top-level Runtime subsystem composer.

    Responsible only for assembling
    Runtime dependencies.
    """

    @classmethod
    def create(
        cls,
        *,
        provider_override=None,
    ) -> RuntimeComposition:
        """
        Compose the complete AI Runtime.

        Parameters
        ----------
        provider_override:
            Optional provider dependency override.

            Used by tests to inject fake providers.

            Production runtime leaves this as None.

        Returns
        -------
        RuntimeComposition
            Fully composed Runtime dependency graph.
        """

        logger.info(
            "Creating complete Runtime composition."
        )

        # ======================================================================
        # Provider subsystem
        # ======================================================================

        provider = ProviderComposer.create(
            provider_override=provider_override,
        )

        # ======================================================================
        # Memory subsystem
        # ======================================================================

        memory = MemoryComposer.create()

        # ======================================================================
        # Prompt subsystem
        # ======================================================================

        prompt = PromptComposer.create()

        # ======================================================================
        # Execution subsystem
        # ======================================================================

        execution = ExecutionComposer.create(
            prompt_builder=prompt.builder,
            prompt_builder_id=prompt.default_builder_id,
            provider_execution_service=provider.execution_service,
            provider_runtime=provider.runtime,
        )

        # ======================================================================
        # Agent subsystem
        # ======================================================================

        agent = AgentComposer.create(
            execution_service=execution.agent_execution_service,
        )

        # ======================================================================
        # Capability catalogue subsystem
        # ======================================================================

        capability = CapabilityComposer.create()

        # ======================================================================
        # Capability execution subsystem
        # ======================================================================

        capability_execution = CapabilityExecutionComposer.create(
            registry=capability.registry,
            agent_resolver=agent.resolver,
            agent_execution_service=(
                execution.agent_execution_service
            ),
        )

        # ======================================================================
        # Skill subsystem
        # ======================================================================

        # Skills are governed Runtime resources loaded from the
        # repository-level skills directory.
        #
        # The Skill subsystem remains independent from the Capability
        # subsystem. A Skill is not embedded into CapabilityDefinition.

        skills_directory = Path("skills")

        skill_loader = SkillLoader(
            skills_directory=skills_directory,
        )

        skill_registry = SkillRegistry(
            loader=skill_loader,
        )

        skill_execution_service = SkillExecutionService(
            skill_registry=skill_registry,
            agent_execution_service=(
                execution.agent_execution_service
            ),
        )

        # ======================================================================
        # Pipeline subsystem
        # ======================================================================

        pipeline = PipelineComposer.create(
            execution_engine=execution.engine,
            agent_resolver=agent.resolver,
            skill_execution_service=(
                skill_execution_service
            ),
        )

        pipeline_runner = PipelineRunner(
            pipeline=pipeline.pipeline,
        )

        # ======================================================================
        # Runtime Kernel
        # ======================================================================

        kernel = RuntimeKernel(
            pipeline_runner=pipeline_runner,
        )

        logger.info(
            (
                "Runtime composition completed. "
                "agents=%s capabilities=%s skills=%s "
                "provider_override=%s"
            ),
            agent.registry.count(),
            capability.registry.count(),
            skill_registry.count,
            provider_override is not None,
        )

        return RuntimeComposition(
            kernel=kernel,
            agent_registry=agent.registry,
            agent_factory=agent.factory,
            capability_registry=capability.registry,
            capability_executor=(
                capability_execution.executor
            ),
            execution_engine=execution.engine,
            pipeline_runner=pipeline_runner,
            provider_execution_service=(
                provider.execution_service
            ),
            memory_service=memory.service,
        )
