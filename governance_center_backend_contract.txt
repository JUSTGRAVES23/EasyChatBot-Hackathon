                + " is paused. "

"""
Governance Dashboard Service.

Read-only operations center projection.

Does NOT:
- modify controls;
- pause providers;
- resume delivery;
- execute workflows.
"""

from __future__ import annotations


class GovernanceDashboardService:
    """
    Builds a read model for the Operations Center.
    """

    def __init__(
        self,
        audit_service=None,
        provider_control=None,
        cybersecurity_control=None,
    ) -> None:

        self.audit_service = audit_service
        self.provider_control = provider_control
        self.cybersecurity_control = cybersecurity_control


    def get_status(self) -> dict:
        """
        Return governance dashboard state.
        """

        controls = []

        if self.provider_control:

            controls.append(
                {
                    "component": "provider_management",
                    "status": (
                        "paused"
                        if self.provider_control.is_paused()
                        else "active"
                    ),
                    "details": (
                        self.provider_control.status()
                    ),
                }
            )


        if self.cybersecurity_control:

            controls.append(
                {
                    "component": "cybersecurity_delivery",
                    "status": (
                        "paused"
                        if self.cybersecurity_control.is_paused()
                        else "active"
                    ),
                    "details": (
                        self.cybersecurity_control.status()
                    ),
                }
            )


        events = []

        if self.audit_service:

            events = [
                {
                    "event": item.event,
                    "actor": item.actor,
                    "reason": item.reason,
                    "component": item.component,
                    "timestamp": (
                        item.timestamp.isoformat()
                    ),
                }
                for item in self.audit_service.list_events()
            ]


        return {
            "controls": controls,
            "recent_events": events,
        }

from __future__ import annotations


class GovernanceEvidenceService:
    '''
    Read-only governance evidence projection.
    '''

    def __init__(
        self,
        audit_service=None,
    ):
        self.audit_service = audit_service


    def timeline(self) -> list[dict]:

        if not self.audit_service:
            return []

        return [
            {
                "event": event.event,
                "actor": event.actor,
                "reason": event.reason,
                "component": event.component,
                "timestamp": event.timestamp.isoformat(),
            }
            for event in self.audit_service.list_events()
        ]

class GovernanceTimelineDashboardService:
    """
    Read-only Operations Center aggregation boundary.
    """

    def build_dashboard(self, events):

        return {
            "timeline": events,
            "total_events": len(events),
            "evidence_available": sum(
                1
                for event in events
                if event.evidence_available
            ),
        }

It explains operational state and audit evidence.
It has no authority to change controls.
"""

from dataclasses import dataclass

    product_name: str = "EasyChatBot Operations Center"
    mascot: str = "YETI"
    mascot_role: str = "Governance Assistant"
    font_family: str = "Nunito"
    character_font: str = "Fredoka"
    accent_font: str = "Caveat"


@dataclass(frozen=True)

    Human-readable governance explanation.
