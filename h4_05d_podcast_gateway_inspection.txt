"""
EasyChatBot Hackathon — Podcast Discovery Agent.

The Agent converts explicit learner context and preference memory into
provider-neutral search intent.

It does not call Apple/Spotify, write memory, select a provider, or invent
podcast metadata.
"""

from __future__ import annotations

import re

from src.ai_runtime.memory.models import MemoryEntry, MemoryType
from src.ai_runtime.podcast.models import (
    LearningContext,
    PodcastSearchIntent,
)


class PodcastDiscoveryAgent:
    """Build provider-neutral podcast discovery intent."""

    _STOP_WORDS = {
        "a", "an", "and", "are", "be", "for", "from", "i", "in", "is",
        "like", "of", "on", "podcasts", "prefers", "prefer", "the", "to",
        "with", "you", "your",
    }

    def build_search_intent(
        self,
        *,
        user_id: str,
        learning_context: LearningContext,
        memories: list[MemoryEntry],
        max_results: int = 10,
    ) -> PodcastSearchIntent:
        """
        Convert explicit preferences + learning-world context into intent.

        Only PREFERENCE memories are used, matching the existing
        EasyChatBot discovery contract.
        """

        if not user_id.strip():
            raise ValueError("Podcast Agent user_id cannot be empty.")

        preferences = [
            memory
            for memory in memories
            if memory.user_id == user_id
            and memory.memory_type == MemoryType.PREFERENCE
        ]

        topics: list[str] = []
        keywords: list[str] = []

        for memory in preferences:
            topics.extend(
                self._metadata_terms(
                    memory.metadata,
                    "topics",
                    "interests",
                    "preferred_topics",
                )
            )
            keywords.extend(
                self._metadata_terms(
                    memory.metadata,
                    "keywords",
                )
            )

            if not topics and not keywords:
                keywords.extend(
                    self._content_keywords(memory.content)
                )

        # The learning world is always part of the search context.
        topics.append(learning_context.world_id)

        # Target language is deliberately a hard eligibility constraint,
        # not just another search keyword.
        return PodcastSearchIntent(
            user_id=user_id,
            learning_context=learning_context,
            topics=tuple(self._dedupe(topics)),
            keywords=tuple(self._dedupe(keywords)),
            language_required=True,
            max_results=max_results,
        )

    @staticmethod
    def _metadata_terms(
        metadata: dict[str, str],
        *keys: str,
    ) -> list[str]:
        terms: list[str] = []
        for key in keys:
            raw = metadata.get(key, "")
            if raw:
                terms.extend(
                    item.strip()
                    for item in re.split(r"[,;|]", raw)
                    if item.strip()
                )
        return terms

    @classmethod
    def _content_keywords(cls, content: str) -> list[str]:
        words = re.findall(
            r"[A-Za-zÀ-ÿ0-9][A-Za-zÀ-ÿ0-9'-]{2,}",
            content.lower(),
        )
        return [
            word for word in words
            if word not in cls._STOP_WORDS
        ][:8]

    @staticmethod
    def _dedupe(values: list[str]) -> list[str]:
        seen: set[str] = set()
        result: list[str] = []
        for value in values:
            key = value.strip().lower()
            if key and key not in seen:
                seen.add(key)
                result.append(value.strip())
        return result

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum
from typing import Final
from uuid import uuid4


class PodcastProviderName(str, Enum):
    APPLE = "apple_podcasts"
    SPOTIFY = "spotify"


@dataclass(frozen=True)
class LearningContext:
    """Explicit learning-world context for podcast discovery."""

    world_id: str
    target_language: str
    learner_level: str
    learning_goal: str = ""
    preferred_languages: tuple[str, ...] = ()
    country_code: str = "LU"

    def __post_init__(self) -> None:
        if not self.world_id.strip():
            raise ValueError("LearningContext world_id cannot be empty.")
        if not self.target_language.strip():
            raise ValueError(
                "LearningContext target_language cannot be empty."
            )
        if not self.learner_level.strip():
            raise ValueError(
                "LearningContext learner_level cannot be empty."
            )
        if len(self.country_code.strip()) != 2:
            raise ValueError(
                "LearningContext country_code must be a two-letter code."
            )


@dataclass(frozen=True)
class PodcastSearchIntent:
    """
    Provider-neutral search intent produced by the Podcast Agent.

    The Agent does not know whether Apple or Spotify will execute it.
    """

    user_id: str
    learning_context: LearningContext
    topics: tuple[str, ...]
    keywords: tuple[str, ...] = ()
    language_required: bool = True
    max_results: int = 10
    intent_id: str = field(default_factory=lambda: str(uuid4()))

    def __post_init__(self) -> None:
        if not self.user_id.strip():
            raise ValueError("PodcastSearchIntent user_id cannot be empty.")
        if not self.topics and not self.keywords:
            raise ValueError(
                "PodcastSearchIntent requires at least one topic or keyword."
            )
        if self.max_results < 1:
            raise ValueError(
                "PodcastSearchIntent max_results must be at least 1."
            )


@dataclass(frozen=True)
class PodcastSearchResult:
    """Normalised podcast/episode candidate from any provider."""

    provider: PodcastProviderName
    external_id: str
    title: str
    description: str
    publisher: str
    languages: tuple[str, ...]
    url: str
    image_url: str | None = None
    published_at: str | None = None
    duration_seconds: int | None = None
    explicit: bool = False
    metadata: dict[str, str] = field(default_factory=dict)

    def __post_init__(self) -> None:
        for name, value in {
            "external_id": self.external_id,
            "title": self.title,
            "publisher": self.publisher,
            "url": self.url,
        }.items():
            if not value.strip():
                raise ValueError(
                    f"PodcastSearchResult {name} cannot be empty."
                )

        if not (
            self.url.startswith("https://")
            or self.url.startswith("http://")
        ):
            raise ValueError(
                "PodcastSearchResult url must use HTTP or HTTPS."
            )

        if not self.languages:
            raise ValueError(
                "PodcastSearchResult requires at least one language."
            )


@dataclass(frozen=True)
class PodcastRecommendation:
    """Character-facing recommendation selected from verified candidates."""

    recommendation_id: str
    user_id: str
    character_id: str
    character_name: str
    character_message: str
    result: PodcastSearchResult
    relevance_reason: str
    match_score: float

    def __post_init__(self) -> None:
        for name, value in {
            "recommendation_id": self.recommendation_id,
            "user_id": self.user_id,
            "character_id": self.character_id,
            "character_name": self.character_name,
            "character_message": self.character_message,
            "relevance_reason": self.relevance_reason,
        }.items():
            if not value.strip():
                raise ValueError(
                    f"PodcastRecommendation {name} cannot be empty."
                )

        if not 0 <= self.match_score <= 100:
            raise ValueError(
                "PodcastRecommendation match_score must be between 0 and 100."
            )


LANGUAGE_ALIASES: Final[dict[str, str]] = {

    "en": "en", "eng": "en", "english": "en",
    "es": "es", "spa": "es", "spanish": "es", "español": "es",
    "de": "de", "deu": "de", "ger": "de", "german": "de",

def normalise_language(value: str) -> str:
    """Return a stable ISO-639-1-style code when recognised."""
    return LANGUAGE_ALIASES.get(
        value.strip().lower(),
        value.strip().lower(),
    )

"""
EasyChatBot Hackathon — personalised podcast recommendation service.

This is the application-level selection boundary.

Language is a HARD filter because EasyChatBot is a language-learning
platform. Preference/topic relevance is evaluated only after a candidate
is language-eligible.
"""

from __future__ import annotations

import re
from uuid import uuid4

from src.ai_runtime.podcast.models import (
    PodcastRecommendation,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)
from src.ai_runtime.podcast.providers import PodcastProvider


class PodcastRecommendationService:
    """Search providers and select a personalised candidate."""

    def __init__(self, providers: list[PodcastProvider]) -> None:
        if not providers:
            raise ValueError(
                "PodcastRecommendationService requires at least one provider."
            )
        self._providers = list(providers)

    def recommend(
        self,
        *,
        intent: PodcastSearchIntent,
        character_id: str,
        character_name: str,
        character_message: str,
    ) -> PodcastRecommendation:
        candidates: list[PodcastSearchResult] = []

        for provider in self._providers:
            candidates.extend(provider.search(intent))

        ranked = self.rank_candidates(
            candidates=candidates,
            intent=intent,
        )

        if not ranked:
            raise LookupError(
                "No podcast matched the learner's language and preferences."
            )

        result, score, reason = ranked[0]

        return PodcastRecommendation(
            recommendation_id=str(uuid4()),
            user_id=intent.user_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            result=result,
            relevance_reason=reason,
            match_score=score,
        )

    @classmethod
    def rank_candidates(
        cls,
        *,
        candidates: list[PodcastSearchResult],
        intent: PodcastSearchIntent,
    ) -> list[tuple[PodcastSearchResult, float, str]]:
        ranked: list[tuple[PodcastSearchResult, float, str]] = []

        target_language = normalise_language(
            intent.learning_context.target_language
        )

        preference_terms = [
            term.lower()
            for term in list(intent.topics) + list(intent.keywords)
            if term.strip()
        ]

        for candidate in candidates:
            candidate_languages = {
                normalise_language(language)
                for language in candidate.languages
            }

            # Critical learning-platform rule:
            # unknown/wrong-language content is not eligible.
            if (
                intent.language_required
                and target_language not in candidate_languages
            ):
                continue

            haystack = " ".join(
                [
                    candidate.title,
                    candidate.description,
                    candidate.publisher,
                ]
            ).lower()

            matched_terms = [
                term
                for term in preference_terms
                if cls._term_matches(term, haystack)
            ]

            score = 50.0

            if matched_terms:
                score += min(30.0, len(matched_terms) * 10.0)

            if intent.learning_context.world_id.lower() in haystack:
                score += 10.0

            level = intent.learning_context.learner_level.lower()
            if level and level in haystack:
                score += 10.0

            reason = cls._build_reason(
                language=intent.learning_context.target_language,
                matched_terms=matched_terms,
                world_id=intent.learning_context.world_id,
            )

            ranked.append(
                (candidate, min(score, 100.0), reason)
            )

        ranked.sort(
            key=lambda item: (
                item[1],
                item[0].published_at or "",
            ),
            reverse=True,
        )

        return ranked[: intent.max_results]

    @staticmethod
    def _term_matches(term: str, haystack: str) -> bool:
        term = term.strip().lower()
        if not term:
            return False
        if term in haystack:
            return True

        words = [
            word
            for word in re.findall(r"[a-z0-9À-ÿ'-]+", term)
            if len(word) > 2
        ]
        return bool(words) and any(
            word in haystack
            for word in words
        )

    @staticmethod
    def _build_reason(
        *,
        language: str,
        matched_terms: list[str],
        world_id: str,
    ) -> str:
        reasons = [f"It is available in {language}."]

        if matched_terms:
            reasons.append(
                "It matches your interests in "
                + ", ".join(matched_terms[:3])
                + "."
            )

        reasons.append(
            f"It fits your {world_id} learning world."
        )

        return " ".join(reasons)

"""
AI Runtime executor for Podcast Discovery.

This adapter bridges the provider-independent AI Runtime capability
contract with the existing Podcast Discovery Agent and
Podcast Recommendation Service.

Architecture:

AI Runtime
    |
    v
CapabilityRequest
    |
    v
PodcastDiscoveryExecutor
    |
    +--> PodcastDiscoveryAgent
    |
    +--> PodcastRecommendationService
    |
    v
PodcastRecommendation
    |
    v
CapabilityResult

The executor:

- validates the Runtime capability;
- validates explicit capability inputs;
- builds the existing LearningContext;
- passes explicit learner memories to the PodcastDiscoveryAgent;
- invokes the existing PodcastRecommendationService;
- returns a provider-independent CapabilityResult.

It does not:

- call Apple directly;
- call Spotify directly;
- implement provider logic;
- retrieve memory itself;
- modify MemoryService;
- perform purchases;
- bypass the AI Runtime;
- make governance decisions.
"""

from __future__ import annotations

from collections.abc import Mapping, Sequence
from typing import Any

from src.ai_runtime.capabilities.executor import (
    CapabilityExecutor,
)

from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
)

from src.ai_runtime.podcast.agent import (
    PodcastDiscoveryAgent,
)

from src.ai_runtime.podcast.models import (
    LearningContext,
)

from src.ai_runtime.podcast.providers import (
    PodcastProvider,
)

from src.ai_runtime.podcast.service import (
    PodcastRecommendationService,
)


class PodcastDiscoveryExecutor(CapabilityExecutor):
    """
    Runtime adapter for the existing Podcast Discovery flow.
    """

    def __init__(
        self,
        agent: PodcastDiscoveryAgent,
        providers: Sequence[PodcastProvider],
    ) -> None:
        if not providers:
            raise ValueError(
                "PodcastDiscoveryExecutor requires at least one provider."
            )

        self._agent = agent
        self._recommendation_service = PodcastRecommendationService(
            providers=list(providers),
        )

    @property
    def capability(self) -> CapabilityType:
        """Return the capability handled by this executor."""

        return CapabilityType.PODCAST_DISCOVERY

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        """
        Execute a provider-independent podcast discovery request.
        """

        if request.capability != CapabilityType.PODCAST_DISCOVERY:
            raise ValueError(
                "PodcastDiscoveryExecutor received an unsupported "
                "capability: "
                f"{request.capability.value}"
            )

        intent = self._build_search_intent(request)

        recommendation = self._recommendation_service.recommend(
            intent=intent,
            character_id=self._required_string(
                request.payload,
                "character_id",
            ),
            character_name=self._required_string(
                request.payload,
                "character_name",
            ),
            character_message=self._required_string(
                request.payload,
                "character_message",
            ),
        )

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output=recommendation,
            metadata={
                "executor": "podcast_discovery_executor",
            },
        )

    def _build_search_intent(
        self,
        request: CapabilityRequest,
    ):
        """
        Translate the Runtime payload into the existing
        PodcastSearchIntent through PodcastDiscoveryAgent.
        """

        payload = request.payload

        user_id = self._required_string(
            payload,
            "user_id",
        )

        learning_context = self._build_learning_context(
            payload
        )

        memories = self._build_memories(
            payload
        )

        max_results = payload.get(
            "max_results",
            10,
        )

        if not isinstance(max_results, int):
            raise ValueError(
                "Podcast Discovery 'max_results' must be an integer."
            )

        if max_results < 1:
            raise ValueError(
                "Podcast Discovery 'max_results' must be at least 1."
            )

        return self._agent.build_search_intent(
            user_id=user_id,
            learning_context=learning_context,
            memories=memories,
            max_results=max_results,
        )

    @staticmethod
    def _build_learning_context(
        payload: Mapping[str, Any],
    ) -> LearningContext:
        """
        Build the existing provider-independent LearningContext.
        """

        world_id = payload.get(
            "world_id",
            "French World",
        )

        target_language = payload.get(
            "target_language",
            "fr",
        )

        learner_level = payload.get(
            "learner_level",
            "B2",
        )

        learning_goal = payload.get(
            "learning_goal",
            "",
        )

        country_code = payload.get(
            "country_code",
            "LU",
        )

        preferred_languages = payload.get(
            "preferred_languages",
            (),
        )

        values = {
            "world_id": world_id,
            "target_language": target_language,
            "learner_level": learner_level,
            "learning_goal": learning_goal,
            "country_code": country_code,
        }

        for name, value in values.items():
            if not isinstance(value, str):
                raise ValueError(
                    f"Podcast Discovery '{name}' must be a string."
                )

        if not isinstance(
            preferred_languages,
            (list, tuple),
        ):
            raise ValueError(
                "Podcast Discovery 'preferred_languages' "
                "must be a list or tuple."
            )

        if not all(
            isinstance(language, str)
            for language in preferred_languages
        ):
            raise ValueError(
                "Podcast Discovery 'preferred_languages' "
                "must contain only strings."
            )

        return LearningContext(
            world_id=world_id,
            target_language=target_language,
            learner_level=learner_level,
            learning_goal=learning_goal,
            preferred_languages=tuple(
                preferred_languages
            ),
            country_code=country_code,
        )

    @staticmethod
    def _build_memories(
        payload: Mapping[str, Any],
    ) -> list[MemoryEntry]:
        """
        Validate explicit learner memories.

        Memory retrieval remains outside this executor. The caller
        supplies the memories that the Runtime has approved for context.
        """

        memories = payload.get(
            "memories",
            (),
        )

        if not isinstance(
            memories,
            (list, tuple),
        ):
            raise ValueError(
                "Podcast Discovery 'memories' must be a list or tuple."
            )

        for memory in memories:
            if not isinstance(
                memory,
                MemoryEntry,
            ):
                raise ValueError(
                    "Podcast Discovery 'memories' must contain only "
                    "MemoryEntry objects."
                )

        return list(memories)

    @staticmethod
    def _required_string(
        payload: Mapping[str, Any],
        field_name: str,
    ) -> str:
        """Return a required non-empty string payload value."""

        value = payload.get(field_name)

        if not isinstance(value, str):
            raise ValueError(
                f"Podcast Discovery '{field_name}' must be a string."
            )

        if not value.strip():
            raise ValueError(
                f"Podcast Discovery '{field_name}' cannot be empty."
            )

        return value

def candidate(title: str, language: str, description: str) -> PodcastSearchResult:
    return PodcastSearchResult(
        provider=PodcastProviderName.APPLE,

        title=title,
        description=description,

def test_wrong_language_is_rejected() -> None:
    ranked = PodcastRecommendationService.rank_candidates(
        candidates=[
            candidate(
                "AI Europe English",
                "en",
                "artificial intelligence European regulation",
            ),
            candidate(
                "IA Europe Français",
                "fr",
                "artificial intelligence European regulation",
            ),
        ],
        intent=intent(),
    )

    assert len(ranked) == 1
    assert ranked[0][0].title == "IA Europe Français"

                "fr",
                "artificial intelligence and European regulation",

        ],
        intent=intent(),
    )

    assert ranked[0][0].title == "French AI Europe"
