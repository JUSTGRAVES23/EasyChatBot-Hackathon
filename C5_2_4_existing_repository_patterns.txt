        self._items[announcement.id] = announcement
        return announcement

    async def list(self) -> tuple[Announcement, ...]:
        return tuple(self._items.values())

        self._events.append(event)

from __future__ import annotations

from app.application.beta_access.repositories import (
    TelegramBindingRepository,
)

    ConversationMemoryRepository,
)
from app.application.privacy.privacy_identifier_service import (

from .models import LearnerContext

class CrossChannelLearnerContextRepository(LearnerContextRepository):
    """Concrete C4.2/C4.3 repository composed from existing application ports.

    No Telegram, memory or conversation tables are introduced here. The
    repository translates the existing Telegram-scoped stores into the
    canonical learner/world read model created by C4.1.

    World compatibility
    -------------------
    Existing Beta/Telegram data may contain legacy World identifiers such as
    ``world-fr`` while the runtime World configuration uses the canonical
    identifier ``french_world``.

    The requested World identifier is always attempted first. A legacy alias
    is used only when the requested identifier cannot be resolved.

    The original requested identifier remains the storage scope for Telegram
    memories and conversation events. Runtime World resolution is therefore
    compatible with the current YAML World configuration without modifying
    historical production data.
    """

    CHARACTER = "Alix"
    LEVEL = "B1"

    # Existing Beta/Telegram compatibility identifiers.
    #
    # These mappings are intentionally local to this compatibility boundary.
    # They do not alter persistence identifiers.
    LEGACY_WORLD_ALIASES = {
        "world-fr": "french_world",
    }

    def __init__(
        self,
        *,
        telegram_bindings: TelegramBindingRepository,
        memory_repository: ConversationMemoryRepository,
        conversation_event_repository: ConversationEventRepository,
        privacy_identifier_service: PrivacyIdentifierService,
        world_resolver: WorldResolver,
    ) -> None:
        if telegram_bindings is None:
            raise ValueError("telegram_bindings is required.")

        if memory_repository is None:
            raise ValueError("memory_repository is required.")

        if conversation_event_repository is None:
            raise ValueError(
                "conversation_event_repository is required."
            )

        if privacy_identifier_service is None:
            raise ValueError(
                "privacy_identifier_service is required."
            )

        if world_resolver is None:
            raise ValueError("world_resolver is required.")

        self._world_resolver = world_resolver

        self._telegram_projector = TelegramLearningContextProjector(
            telegram_bindings=telegram_bindings,
            memory_repository=memory_repository,
            conversation_event_repository=conversation_event_repository,
            privacy_identifier_service=privacy_identifier_service,
        )

    async def get(
        self,
        *,
        learner_id: str,
        world_id: str,
        recent_limit: int,
    ) -> LearnerContext | None:
        """Assemble context for one authenticated learner and World.

        The requested World identifier is preserved as the cross-channel
        storage scope.

        Runtime resolution first attempts the requested identifier. If that
        identifier is a known legacy alias and cannot be resolved, the
        canonical runtime identifier is attempted.

        This allows historical Telegram data using ``world-fr`` to remain
        readable while the active runtime World is ``french_world``.
        """

        world = self._resolve_world(world_id)

        projection = await self._telegram_projector.project(
            learner_id=learner_id,
            world_id=world_id,
            recent_limit=recent_limit,
        )

        return LearnerContext(
            learner_id=learner_id,
            world_id=world_id,
            language_code=world.language_code or "",
            character=self.CHARACTER,
            level=self.LEVEL,
            telegram_connected=projection.telegram_connected,
            learning_memories=projection.learning_memories,
            recent_learning_context=projection.recent_learning_context,
        )

    def _resolve_world(self, world_id: str):
        """Resolve a requested World with legacy compatibility fallback.

        The requested identifier is authoritative when it exists.

        For a known legacy identifier, the canonical runtime identifier is
        attempted only after the original identifier fails to resolve.

        Unknown identifiers fail through the existing WorldResolver contract.
        """

        try:
            world = self._world_resolver.resolve(world_id)
        except ValueError as exc:
            canonical_world_id = self.LEGACY_WORLD_ALIASES.get(
                world_id
            )

            if canonical_world_id is None:
                raise

            world = self._world_resolver.resolve(
                canonical_world_id
            )

        if world.world_id not in {
            world_id,
            self.LEGACY_WORLD_ALIASES.get(world_id),
        }:
            raise ValueError(
                "Resolved World mismatch."
            )

        return world

"""
AI Service Provider Manager escalation repository.

Defines the persistence boundary for provider
operational escalations.

This module does not:

- send emails;
- calculate escalation rules;
- execute providers;
- modify Runtime configuration.

Responsibilities:

- store escalations;
- retrieve escalation history;
- provide a persistence abstraction.

Architecture:

EscalationService
        |
        v
EscalationRepository
        |
        v
Storage Implementation

Current implementation:

- in-memory repository.

Future implementations:

- SQLite;
- PostgreSQL;
- enterprise incident platform.
"""

from __future__ import annotations

from abc import ABC, abstractmethod

from app.application.provider_management.models import (
    ProviderEscalation,
)


class EscalationRepository(ABC):
    """
    Persistence contract for provider escalations.

    Application services depend on this abstraction,
    not on a concrete storage implementation.
    """

    @abstractmethod
    def save(
        self,
        escalation: ProviderEscalation,
    ) -> None:
        """
        Store one provider escalation.
        """

        raise NotImplementedError(
            "Escalation save must be implemented."
        )

    @abstractmethod
    def get_all(
        self,
    ) -> list[ProviderEscalation]:
        """
        Retrieve all stored escalations.
        """

        raise NotImplementedError(
            "Escalation get_all must be implemented."
        )

    @abstractmethod
    def get_by_id(
        self,
        escalation_id: str,
    ) -> ProviderEscalation | None:
        """
        Retrieve an escalation by its identifier.
        """

        raise NotImplementedError(
            "Escalation get_by_id must be implemented."
        )

    @abstractmethod
    def clear(
        self,
    ) -> None:
        """
        Clear stored escalations.

        Mainly used by tests.
        """

        raise NotImplementedError(
            "Escalation clear must be implemented."
        )


class InMemoryEscalationRepository(
    EscalationRepository,
):
    """
    In-memory escalation repository.

    Used for:

    - development;
    - testing;
    - hackathon validation.

    It can later be replaced by a persistent
    implementation without changing the service
    contract.
    """

    def __init__(
        self,
    ) -> None:
        self._escalations: list[
            ProviderEscalation
        ] = []

    def save(
        self,
        escalation: ProviderEscalation,
    ) -> None:
        """
        Store one escalation.
        """

        if escalation is None:
            raise ValueError(
                "Escalation cannot be None."
            )

        self._escalations.append(
            escalation
        )

    def get_all(
        self,
    ) -> list[ProviderEscalation]:
        """
        Return all stored escalations.

        A copy is returned to prevent callers
        from mutating repository state directly.
        """

        return list(
            self._escalations
        )

    def get_by_id(
        self,
        escalation_id: str,
    ) -> ProviderEscalation | None:
        """
        Retrieve one escalation by ID.
        """

        if not escalation_id.strip():
            raise ValueError(
                "Escalation id cannot be empty."
            )

        for escalation in self._escalations:
            if (
                escalation.escalation_id
                == escalation_id
            ):
                return escalation

        return None

    def clear(
        self,
    ) -> None:
        """
        Remove all stored escalations.
        """

class ProviderHealthHistoryRepository:
    """In-memory provider-health observation repository for Beta."""

    def __init__(self) -> None:
        self._observations: dict[
            str, list[ProviderHealthObservation]
        ] = defaultdict(list)

    def save(self, observation: ProviderHealthObservation) -> None:
        if observation is None:
            raise ValueError(
                "Provider health observation cannot be None."
            )
        self._observations[observation.provider].append(observation)

    def get_all(self) -> tuple[ProviderHealthObservation, ...]:
        result: list[ProviderHealthObservation] = []
        for observations in self._observations.values():
            result.extend(observations)
        return tuple(result)

    def get_by_provider(
        self,
        provider: str,
    ) -> tuple[ProviderHealthObservation, ...]:
        if not provider.strip():
            raise ValueError("Provider cannot be empty.")
        return tuple(self._observations.get(provider, []))

    def latest(
        self,
        provider: str,
    ) -> ProviderHealthObservation | None:
        observations = self.get_by_provider(provider)
        if not observations:
            return None
        return observations[-1]

    def count(self) -> int:
        return sum(
            len(items)
            for items in self._observations.values()
        )

"""
Filesystem repository for character theme YAML files.

This repository is intentionally read-only and database-independent.
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

import yaml


class ThemeRepositoryError(Exception):
    """Base error for theme repository operations."""


class ThemeNotFoundError(ThemeRepositoryError):
    """Raised when a requested theme does not exist."""


class ThemeSecurityError(ThemeRepositoryError):
    """Raised when a requested path escapes the theme directory."""


class ThemeFormatError(ThemeRepositoryError):
    """Raised when a theme file is invalid."""


class FileThemeRepository:
    """Read-only repository for YAML character themes."""

    def __init__(self, *, themes_directory: Path) -> None:
        if themes_directory is None:
            raise ValueError("themes_directory cannot be None.")

        if not isinstance(themes_directory, Path):
            raise TypeError("themes_directory must be Path.")

        self._themes_directory = themes_directory.resolve()

    def get(self, character_id: str) -> dict[str, Any]:
        identifier = self._normalise_id(character_id)
        file_path = self._themes_directory / f"{identifier}.yaml"
        self._ensure_inside_directory(file_path)

        if not file_path.is_file():
            raise ThemeNotFoundError(
                f"Character theme not found: {file_path}"
            )

        try:
            with file_path.open("r", encoding="utf-8") as handle:
                data = yaml.safe_load(handle)
        except yaml.YAMLError as exc:
            raise ThemeFormatError(
                f"Invalid YAML character theme: {file_path}"
            ) from exc
        except OSError as exc:
            raise ThemeRepositoryError(
                f"Unable to read character theme: {file_path}"
            ) from exc

        if not isinstance(data, dict) or not data:
            raise ThemeFormatError(
                f"Character theme must be a non-empty YAML mapping: {file_path}"
            )

        return data

    def exists(self, character_id: str) -> bool:
        try:
            identifier = self._normalise_id(character_id)
        except ValueError:
            return False

        return (self._themes_directory / f"{identifier}.yaml").is_file()

    def identifiers(self) -> tuple[str, ...]:
        if not self._themes_directory.is_dir():
            return ()

        return tuple(
            sorted(
                path.stem
                for path in self._themes_directory.glob("*.yaml")
            )
        )

    @property
    def themes_directory(self) -> Path:
        return self._themes_directory

    @staticmethod
    def _normalise_id(character_id: str) -> str:
        if not character_id:
            raise ValueError("character_id cannot be empty.")

        value = character_id.strip().lower()

        if not value.replace("_", "").isalnum():
            raise ValueError(
                "character_id contains invalid characters."
            )

        return value

    def _ensure_inside_directory(self, file_path: Path) -> None:
        try:
            file_path.resolve().relative_to(self._themes_directory)
        except ValueError as exc:
            raise ThemeSecurityError(
                "Character theme path escapes theme directory."
            ) from exc

    async def claim(
        self,
        *,
        user_id: UUID,

        now: datetime,

                        1,
                        :now,
                        :now

                        status = 'SENDING',
                        attempt_count =
                            academy_notification_deliveries.attempt_count + 1,
                        updated_at = EXCLUDED.updated_at
                    WHERE academy_notification_deliveries.status = 'FAILED'

                        user_id,

                        status,
                        attempt_count,
                        provider_message_id,

                    "now": now,

            if row is None:
                await session.rollback()

                user_id=row.user_id,
                notification_type=row.notification_type,
                status=AcademyDeliveryStatus(row.status),
                attempt_count=row.attempt_count,

        self._reject_sync_boundary()
        return None

class SqlAlchemyPublicationExecutionRepository(
    PublicationExecutionRepository
):
    """Durable repository using the existing injected async session factory."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    async def get_by_idempotency_key(
        self,
        idempotency_key: str,
    ) -> PublicationExecutionRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )
            model = result.scalar_one_or_none()

            return self._to_domain(model) if model else None

    async def create_started(
        self,
        *,
        publication_id,
        idempotency_key,
        correlation_id,
    ) -> PublicationExecutionRecord:
        """Create the first durable execution attempt."""

        now = datetime.now(timezone.utc)

        model = PublicationExecutionRecordModel(
            publication_id=str(publication_id),
            idempotency_key=idempotency_key,
            correlation_id=correlation_id,
            attempt=1,
            status=DurableExecutionStatus.STARTED.value,
            created_at=now,
            updated_at=now,
        )

        async with self._session_factory() as session:
            session.add(model)
            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def mark_succeeded(
        self,
        *,
        idempotency_key,
        provider_message_id,
    ) -> PublicationExecutionRecord:
        """Persist successful completion without changing the attempt count."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.SUCCEEDED.value,
            provider_message_id=provider_message_id,
            error_code=None,
        )

    async def mark_failed(
        self,
        *,
        idempotency_key,
        error_code,
    ) -> PublicationExecutionRecord:
        """Persist failure while retaining the current attempt number."""

        return await self._update(
            idempotency_key,
            status=DurableExecutionStatus.FAILED.value,
            error_code=error_code,
        )

    async def start_retry(
        self,
        *,
        idempotency_key: str,
    ) -> PublicationExecutionRecord:
        """Increment the durable attempt counter for a retry.

        A retry is valid only for an existing FAILED execution. The same
        idempotency record is retained; no second database row is created.
        """

        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            if model.status != DurableExecutionStatus.FAILED.value:
                raise ValueError(
                    "Only failed executions can be retried."
                )

            model.attempt += 1
            model.status = DurableExecutionStatus.STARTED.value
            model.error_code = None
            model.provider_message_id = None
            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    async def _update(
        self,
        idempotency_key,
        **values,
    ) -> PublicationExecutionRecord:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationExecutionRecordModel).where(
                    PublicationExecutionRecordModel.idempotency_key
                    == idempotency_key
                )
            )

            model = result.scalar_one()

            for key, value in values.items():
                setattr(model, key, value)

            model.updated_at = datetime.now(timezone.utc)

            await session.commit()
            await session.refresh(model)

            return self._to_domain(model)

    @staticmethod
    def _to_domain(
        model: PublicationExecutionRecordModel,
    ) -> PublicationExecutionRecord:
        return PublicationExecutionRecord(
            publication_id=UUID(model.publication_id),
            idempotency_key=model.idempotency_key,
            correlation_id=model.correlation_id,
            attempt=model.attempt,
            status=DurableExecutionStatus(model.status),
            provider_message_id=model.provider_message_id,
            error_code=model.error_code,
            created_at=model.created_at,
            updated_at=model.updated_at,
        )

    async def get_password_reset_context(
        self,

        The raw token is never persisted or returned. Only a valid, unused,

        self,

        password_hash: str,

    ) -> bool:
        """Atomically consume a reset token, update the password,
        and revoke existing sessions.
        """
        token_hash = hashlib.sha256(
            token.encode("utf-8")

        async with self._session_factory() as session:

                        """

                        FROM auth_password_reset_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now

                        """
                    ),
                    {
                        "token_hash": token_hash,
                        "now": now,
                    },
                )

                row = result.mappings().first()

                if row is None:
                    return False

                user_id = row["user_id"]

                credential_result = await session.execute(
                    text(
                        """
                        SELECT id
                        FROM user_credentials
                        WHERE user_id = :user_id
                          AND credential_type = 'PASSWORD'
                        LIMIT 1
                        """
                    ),
                    {
                        "user_id": user_id,
                    },
                )

                credential = credential_result.mappings().first()

                if credential is None:
                    return False

                update_result = await session.execute(
                    text(
                        """
                        UPDATE user_credentials
                        SET password_hash = :password_hash,
                            updated_at = :updated_at
                        WHERE id = :credential_id
                          AND (
                              CAST(:expected_password_hash AS TEXT) IS NULL
                              OR password_hash = CAST(:expected_password_hash AS TEXT)
                          )
                        """
                    ),
                    {
                        "password_hash": password_hash,
                        "updated_at": now,
                        "credential_id": credential["id"],
                        "expected_password_hash": expected_password_hash,
                    },
                )

                if update_result.rowcount != 1:
                    return False

                await session.execute(
                    text(
                        """
                        UPDATE auth_password_reset_tokens
                        SET used_at = :used_at
                        WHERE id = :id
                        """
                    ),
                    {
                        "used_at": now,
                        "id": row["id"],
                    },
                )

                await session.execute(
                    text(
                        """
                        UPDATE user_sessions
                        SET is_revoked = TRUE,
                            status = 'revoked'
                        WHERE user_id = :user_id
                          AND is_revoked = FALSE
                        """
                    ),

                    },
                )

from pathlib import Path
from typing import Any

import yaml


class YamlCharacterRepository:
    """
    Loads Character Profiles from YAML files.
    """

    def __init__(
        self,
        *,
        profiles_path: Path,
    ) -> None:
        """
        Initialise repository.

        Parameters
        ----------
        profiles_path:
            Location containing character YAML profiles.
        """

        self._profiles_path = profiles_path


    def get_by_id(
        self,
        character_id: str,
    ) -> dict[str, Any] | None:
        """
        Load a character profile.

        Parameters
        ----------
        character_id:
            Character identifier.

        Returns
        -------
        dict | None
            Character configuration if found.

            self._profiles_path
            / f"{character_id}.yaml"
        )

        if not file_path.exists():
            return None


        with file_path.open(
            "r",
            encoding="utf-8",
        ) as file:

            data = yaml.safe_load(file)


        return data

class PostgreSQLConversationEventRepository(
    ConversationEventRepository
):
    """
    PostgreSQL implementation of conversation event storage.

    Privacy isolation:

        telegram_user_hash
                +
          conversation_id
                +
             world_id

    Governance:

        privacy_hash_version
                +
        retention_policy_id

    Responsibility:

    - persist conversation events
    - retrieve conversation history

    Does NOT:

    - validate business rules
    - create memories
    - analyse learning behaviour
    - create RAG knowledge
    """

    def __init__(
        self,
        session_factory,
    ) -> None:
        if session_factory is None:
            raise ValueError(
                "session_factory cannot be None."
            )

        self._session_factory = session_factory

    async def save(
        self,
        event: ConversationEvent,
    ) -> ConversationEvent:
        """
        Persist an anonymised conversation event.

        Privacy and retention governance values are taken
        from the validated event metadata and persisted into
        their dedicated database columns.
        """

        metadata = event.metadata or {}

        privacy_hash_version = metadata.get(
            "privacy_hash_version",
        )

        retention_policy_id = metadata.get(
            "retention_policy_id",
        )

        if not privacy_hash_version:
            raise ValueError(
                "Conversation event is missing "
                "privacy_hash_version."
            )

        if not retention_policy_id:
            raise ValueError(
                "Conversation event is missing "
                "retention_policy_id."
            )

        with self._session_factory() as session:
            record = ConversationEventRecord(
                id=event.event_id,
                conversation_id=event.conversation_id,
                telegram_user_hash=(
                    event.telegram_user_hash
                ),
                world_id=event.world_id,
                channel=event.channel,
                role=event.role,
                content=event.content,
                metadata_json=metadata,
                privacy_hash_version=(
                    privacy_hash_version
                ),
                retention_policy_id=(
                    retention_policy_id
                ),
                created_at=event.created_at,
            )

            session.add(record)
            session.commit()

        return event

    async def get_recent(
        self,
        *,
        telegram_user_hash: str,
        conversation_id: str,
        world_id: str,
        limit: int = 20,
    ) -> list[ConversationEvent]:
        """
        Retrieve recent conversation events for one
        pseudonymised Telegram identity, conversation,
        and learning world.

        Isolation key:

            telegram_user_hash
                    +
            conversation_id
                    +
                world_id

        Results are returned in chronological order so
        callers can construct conversation context safely.
        """

        if limit <= 0:
            return []

        if not telegram_user_hash:
            raise ValueError(
                "telegram_user_hash cannot be empty."
            )

        if not conversation_id:
            raise ValueError(
                "conversation_id cannot be empty."
            )

        if not world_id:
            raise ValueError(
                "world_id cannot be empty."
            )

        with self._session_factory() as session:
            rows = session.scalars(
                select(
                    ConversationEventRecord
                )
                .where(
                    ConversationEventRecord.telegram_user_hash
                    == telegram_user_hash
                )
                .where(
                    ConversationEventRecord.conversation_id
                    == conversation_id
                )
                .where(
                    ConversationEventRecord.world_id
                    == world_id
                )
                .order_by(
                    ConversationEventRecord.created_at.desc()
                )
                .limit(limit)
            ).all()

        return [
            _event_from_record(row)
            for row in reversed(rows)
        ]


def _event_from_record(
    row: ConversationEventRecord,
) -> ConversationEvent:
    """
    Convert the persistence record back into the
    application-domain conversation event.

    Governance fields remain available through metadata
    so the domain event retains its original contract.
    """

    metadata = dict(
        row.metadata_json or {}
    )

    metadata.setdefault(
        "privacy_hash_version",
        row.privacy_hash_version,
    )

    metadata.setdefault(
        "retention_policy_id",
        row.retention_policy_id,
    )

    return ConversationEvent(
        event_id=row.id,
        conversation_id=row.conversation_id,
        telegram_user_hash=(
            row.telegram_user_hash
        ),
        world_id=row.world_id,
        channel=row.channel,
        role=row.role,
        content=row.content,
        metadata=metadata,
        created_at=row.created_at,
    )

    ) -> PrivateChatActiveConversation | None:
        """
        Return the server-selected active conversation and its recent
        persisted message history for the supplied user and World.
        """
        if limit < 1:
            raise ValueError("limit must be greater than zero")

        active_record = self._session.scalar(
            select(PrivateChatActiveConversationRecord).where(
                PrivateChatActiveConversationRecord.user_id == user_id,
                PrivateChatActiveConversationRecord.world_id == world_id,
            )
        )

        if active_record is None:
            return None

        conversation_record = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == active_record.conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
                PrivateChatConversationRecord.world_id == world_id,
            )
        )

        if conversation_record is None:
            return None

        messages = self.get_recent_messages(
            conversation_id=conversation_record.conversation_id,
            user_id=user_id,
            world_id=world_id,
            limit=limit,
        )

        return PrivateChatActiveConversation(
            conversation=self._to_conversation(conversation_record),
            messages=messages,
        )

    def create_conversation(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> PrivateChatConversation:
        now = datetime.now(timezone.utc)

        record = PrivateChatConversationRecord(
            user_id=user_id,
            world_id=world_id,
            created_at=now,
            updated_at=now,
        )

        self._session.add(record)
        self._session.flush()

        return self._to_conversation(record)

    def set_active_conversation(
        self,
        *,
        user_id: str,
        world_id: str,
        conversation_id: UUID,
    ) -> None:
        """
        Point one user/World pair at the supplied conversation.

        This method deliberately does not delete or modify historical
        conversations. It only changes the active pointer.
        """
        conversation = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
                PrivateChatConversationRecord.world_id == world_id,
            )
        )

        if conversation is None:
            raise ValueError(
                "Private chat conversation not found"
            )

        active_record = self._session.scalar(
            select(PrivateChatActiveConversationRecord).where(
                PrivateChatActiveConversationRecord.user_id == user_id,
                PrivateChatActiveConversationRecord.world_id == world_id,
            )
        )

        if active_record is None:
            active_record = PrivateChatActiveConversationRecord(
                user_id=user_id,
                world_id=world_id,
                conversation_id=conversation_id,
            )
            self._session.add(active_record)
        else:
            active_record.conversation_id = conversation_id

        self._session.flush()

    def save_message(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        message: PrivateChatMessage,
    ) -> PrivateChatMessage:
        record = PrivateChatMessageRecord(
            conversation_id=conversation_id,
            user_id=user_id,
            world_id=world_id,
            role=message.role,
            content=message.content,
            created_at=message.created_at,
        )

        self._session.add(record)
        self._session.flush()

        return self._to_message(record)

    def get_recent_messages(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
        limit: int = 20,
    ) -> list[PrivateChatMessage]:
        if limit < 1:
            raise ValueError("limit must be greater than zero")

        records = self._session.scalars(
            select(PrivateChatMessageRecord)
            .where(
                PrivateChatMessageRecord.conversation_id
                == conversation_id,
                PrivateChatMessageRecord.user_id == user_id,
                PrivateChatMessageRecord.world_id == world_id,
            )
            .order_by(
                PrivateChatMessageRecord.created_at.desc()
            )
            .limit(limit)
        ).all()

        return [
            self._to_message(record)
            for record in reversed(records)
        ]

    def touch_conversation(
        self,
        *,
        conversation_id: UUID,
        user_id: str,
        world_id: str,
    ) -> None:
        record = self._session.scalar(
            select(PrivateChatConversationRecord).where(
                PrivateChatConversationRecord.conversation_id
                == conversation_id,
                PrivateChatConversationRecord.user_id == user_id,
                PrivateChatConversationRecord.world_id == world_id,
            )
        )

        if record is None:
            raise ValueError(
                "Private chat conversation not found"
            )

        record.updated_at = datetime.now(timezone.utc)
        self._session.flush()

    def commit(self) -> None:
        self._session.commit()

    def rollback(self) -> None:
        self._session.rollback()

    def close(self) -> None:
        self._session.close()

    @staticmethod
    def _to_conversation(
        record: PrivateChatConversationRecord,
    ) -> PrivateChatConversation:
        return PrivateChatConversation(
            conversation_id=record.conversation_id,
            user_id=record.user_id,
            world_id=record.world_id,
            created_at=record.created_at,
            updated_at=record.updated_at,
        )

    @staticmethod
    def _to_message(
        record: PrivateChatMessageRecord,
    ) -> PrivateChatMessage:
        return PrivateChatMessage(
            conversation_id=record.conversation_id,
            role=record.role,
            content=record.content,
            created_at=record.created_at,
        )

    @staticmethod
    def _to_active_conversation(
        record: PrivateChatActiveConversationRecord,
        conversation: PrivateChatConversationRecord,
        messages: list[PrivateChatMessage],
    ) -> PrivateChatActiveConversation:
        return PrivateChatActiveConversation(
            conversation=PostgreSQLPrivateChatRepository._to_conversation(
                conversation
            ),
            messages=messages,
        )

                    quality_score=event.quality_score,
                    success=event.success,
                    provider_request_id=event.provider_request_id,
                    created_at=event.created_at,
                )
            )
            session.commit()

    def get_all(self) -> list[ProviderUsageEvent]:
        with self._session_factory() as session:
            rows = (
                session.query(ProviderUsageRecord)
                .order_by(
                    ProviderUsageRecord.created_at.asc(),
                    ProviderUsageRecord.id.asc(),
                )
                .all()
            )

        return [_event_from_record(row) for row in rows]

    def clear(self) -> None:
        with self._session_factory() as session:
            session.query(ProviderUsageRecord).delete()
            session.commit()

"""
YAML World Repository.

Infrastructure implementation for loading World configurations
from YAML files.
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

import yaml

from app.domain.uai.context.language_policy import (
    LanguagePolicy,
)


class WorldConfiguration:
    """
    Infrastructure representation of a World.

    Validates configuration shape before the configuration can
    enter the application/domain runtime.
    """

    def __init__(
        self,
        data: dict[str, Any],
    ) -> None:

        if not isinstance(data, dict):
            raise ValueError(
                "World configuration must be a mapping."
            )

        self.id = self._required_string(
            data,
            "id",
            "World id cannot be empty.",
        )

        self.name = self._required_string(
            data,
            "name",
            "World name cannot be empty.",
        )

        self.world_type = self._required_string(
            data,
            "world_type",
            "World type cannot be empty.",
            default="language",
        )

        self.learning_domain = self._required_string(
            data,
            "learning_domain",
            "Learning domain cannot be empty.",
            default="language_learning",
        )

        language_code = data.get("language_code")

        if language_code is not None:
            if not isinstance(language_code, str):
                raise ValueError(
                    "Language code must be a string when provided."
                )

            if not language_code.strip():
                raise ValueError(
                    "Language code cannot be empty when provided."
                )

            language_code = language_code.strip()

        if (
            self.world_type == "language"
            and not language_code
        ):
            raise ValueError(
                "Language Worlds require language_code."
            )

        self.language_code = language_code

        self.language_policy = self._load_language_policy(
            data.get("language_policy"),
            language_code=language_code,
            world_type=self.world_type,
        )

        available_characters = data.get(
            "available_characters",
            [],
        )

        if not isinstance(
            available_characters,
            list,
        ):
            raise ValueError(
                "available_characters must be a list."
            )

        validated_characters: list[str] = []

        for character_id in available_characters:
            if not isinstance(
                character_id,
                str,
            ):
                raise ValueError(
                    "Character id must be a string."
                )

            if not character_id.strip():
                raise ValueError(
                    "Character id cannot be empty."
                )

            validated_characters.append(
                character_id.strip()
            )

        self.available_characters = validated_characters

    @staticmethod
    def _required_string(
        data: dict[str, Any],
        field_name: str,
        error_message: str,
        *,
        default: str | None = None,
    ) -> str:

        value = data.get(
            field_name,
            default,
        )

        if not isinstance(value, str):
            raise ValueError(error_message)

        if not value.strip():
            raise ValueError(error_message)

        return value.strip()

    @staticmethod
    def _load_language_policy(
        value: Any,
        *,
        language_code: str | None,
        world_type: str,
    ) -> LanguagePolicy:

        # Existing language World configurations remain compatible.
        if value is None:
            if world_type == "language":
                if language_code is None:
                    raise ValueError(
                        "Language Worlds require language_code."
                    )

                return LanguagePolicy(
                    mode=LanguagePolicy.TARGET_LANGUAGE,
                    target_language=language_code,
                )

            raise ValueError(
                "Domain Worlds require language_policy."
            )

        if not isinstance(value, dict):
            raise ValueError(
                "language_policy must be a mapping."
            )

        mode = value.get("mode")

        if not isinstance(mode, str) or not mode.strip():
            raise ValueError(
                "language_policy mode is required."
            )

        target_language = value.get(
            "target_language"
        )

        if target_language is not None:
            if not isinstance(target_language, str):
                raise ValueError(
                    "target_language must be a string."
                )

            target_language = target_language.strip()

        supported_languages = value.get(
            "supported_languages",
            [],
        )

        if not isinstance(
            supported_languages,
            list,
        ):
            raise ValueError(
                "supported_languages must be a list."
            )

        validated_languages: list[str] = []

        for language in supported_languages:
            if not isinstance(language, str):
                raise ValueError(
                    "supported_languages must contain strings."
                )

            if not language.strip():
                raise ValueError(
                    "supported_languages cannot contain empty values."
                )

            validated_languages.append(
                language.strip()
            )

        return LanguagePolicy(
            mode=mode.strip(),
            target_language=target_language,
            supported_languages=tuple(
                validated_languages
            ),
        )


class YamlWorldRepository:
    """
    Loads Worlds from YAML files.
    """

    def __init__(
        self,
        *,
        worlds_path: Path,
    ) -> None:
        self._worlds_path = worlds_path

    def get_by_id(
        self,
        world_id: str,
    ) -> WorldConfiguration | None:
        """
        Load a World configuration by identifier.
        """

        file_path = (
            self._worlds_path.resolve()
            / f"{world_id}.yaml"
        )

        if not file_path.exists():
            return None

        with file_path.open(
            "r",
            encoding="utf-8",
        ) as file:
            data = yaml.safe_load(file)

        return WorldConfiguration(data)
