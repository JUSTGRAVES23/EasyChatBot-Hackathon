def create_telegram_runtime(
    *,
    functionality_control_service: FunctionalityOperationalControlService,
) -> TelegramBotRuntime:
    """
    Create the production Beta Telegram runtime.

    The functionality-control service is supplied explicitly by the
    application composition. Telegram bootstrap does not create a
    second control repository or control service.
    """

    global _telegram_production

    if functionality_control_service is None:
        raise ValueError(
            "functionality_control_service is required."
        )

    if _telegram_production is None:
        _telegram_production = (
            create_production_beta_telegram_composition(
                database_engine_factory=(
                    create_database_engine
                ),
                session_factory_factory=(
                    create_session_factory
                ),
                conversation_runtime_factory=(
                    create_conversation_runtime
                ),
                functionality_control_service=(
                    functionality_control_service
                ),
            )
        )

        logger.info(
            "Production Beta Telegram composition created."
        )

    return _telegram_production.beta_telegram.runtime


async def start_telegram_runtime(
    *,
    functionality_control_service: FunctionalityOperationalControlService,
) -> None:
    """
    Start the production Telegram runtime.
    """

    runtime = create_telegram_runtime(
        functionality_control_service=(
            functionality_control_service
        ),
    )

    await runtime.start()

    logger.info(
        "Production Beta Telegram runtime started."
    )


async def stop_telegram_runtime() -> None:
    """
    Stop the production Telegram runtime.
    """

    if _telegram_production is None:
        return

    await _telegram_production.beta_telegram.runtime.stop()

    logger.info(
        "Production Beta Telegram runtime stopped."
    )


def reset_telegram_runtime() -> None:
    """
    Reset the Telegram production composition.

    Intended primarily for tests and controlled lifecycle
    management.
    """

    global _telegram_production

    _telegram_production = None


def get_telegram_production_composition() -> (
    BetaTelegramProductionComposition
):
    """
    Return the active production Beta Telegram composition.
    """

    if _telegram_production is None:
        raise RuntimeError(
            "Telegram production composition has not been created."
        )

    return _telegram_production

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = (
        application.state.scheduler_runtime
    )

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )
    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )

def create_production_beta_telegram_composition(
    *,
    expiration_minutes: int = 1440,
    database_engine: Any | None = None,
    session_factory: Any | None = None,
    database_engine_factory: Callable[[], Any] | None = None,
    session_factory_factory: Callable[[Any], Any] | None = None,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: Callable[[], ConversationRuntime] | None = None,
    functionality_control_service: (
        FunctionalityOperationalControlService | None
    ) = None,
) -> BetaTelegramProductionComposition:
    """Create the production Beta Telegram composition.

    Production Telegram processing requires the operational
    functionality-control service. The service is injected from
    the application composition; this factory never creates a
    second control service or control repository.
    """

    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    if functionality_control_service is None:
        raise ValueError(
            "functionality_control_service is required "
            "for production Telegram composition."
        )

    if (
        database_engine is not None
        and database_engine_factory is not None
    ):
        raise ValueError(
            "Provide either database_engine or "
            "database_engine_factory, not both."
        )

    if (
        session_factory is not None
        and session_factory_factory is not None
    ):
        raise ValueError(
            "Provide either session_factory or "
            "session_factory_factory, not both."
        )

    if (
        conversation_runtime is not None
        and conversation_runtime_factory is not None
    ):
        raise ValueError(
            "Provide either conversation_runtime or "
            "conversation_runtime_factory, not both."
        )

    if database_engine is None:
        database_engine = (
            database_engine_factory
            or create_database_engine
        )()

    if session_factory is None:
        session_factory = (
            session_factory_factory
            or create_session_factory
        )(database_engine)

    beta_user_repository = PostgreSQLBetaUserRepository(
        session_factory,
    )
    beta_token_repository = PostgreSQLBetaTokenRepository(
        session_factory,
    )
    telegram_binding_repository = (
        PostgreSQLTelegramBindingRepository(
            session_factory,
        )
    )
    conversation_event_repository = (
        PostgreSQLConversationEventRepository(
            session_factory,
        )
    )

    # Durable provider usage ledger.
    # Accounting remains outside the Operations Center.
    provider_usage_repository = (
        PostgreSQLProviderUsageRepository(
            session_factory,
        )
    )

    privacy_identifier_service = PrivacyIdentifierService()
    conversation_event_validator = (
        ConversationEventValidator()
    )

    beta_telegram = create_beta_telegram_composition(
        user_repository=beta_user_repository,
        token_repository=beta_token_repository,
        binding_repository=telegram_binding_repository,
        conversation_event_repository=(
            conversation_event_repository
        ),
        privacy_identifier_service=(
            privacy_identifier_service
        ),
        conversation_event_validator=(
            conversation_event_validator
        ),
        expiration_minutes=expiration_minutes,
        conversation_runtime=conversation_runtime,
        conversation_runtime_factory=(
            conversation_runtime_factory
        ),
        provider_usage_repository=(
            provider_usage_repository
        ),
        functionality_control_service=(
            functionality_control_service
        ),
    )

    logger.info(
        "Production Beta Telegram composition created "
        "with durable usage ledger and functionality control."
    )

    return BetaTelegramProductionComposition(
        beta_telegram=beta_telegram,
        database_engine=database_engine,
        session_factory=session_factory,
    )

from __future__ import annotations

import os

from sqlalchemy import create_engine
from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker


class Base(DeclarativeBase):
    """Canonical SQLAlchemy metadata registry for the platform."""

    pass


def database_url() -> str:
    """Return the canonical PostgreSQL connection URL."""

    value = os.getenv("DATABASE_URL")
    if not value:
        raise RuntimeError(
            "DATABASE_URL is required for PostgreSQL persistence."
        )

    return value


def create_database_engine():
    """Create the existing synchronous PostgreSQL engine.

    This API is preserved for the existing Beta 1 persistence layer.
    """

    return create_engine(
        database_url(),
        pool_pre_ping=True,
        future=True,
    )


def create_session_factory(
    engine,
) -> sessionmaker[Session]:
    """Create the existing synchronous session factory.

    Preserved for existing Beta 1 repositories.
    """

    return sessionmaker(
        bind=engine,
        class_=Session,
        expire_on_commit=False,
    )


def async_database_url() -> str:
    """Return DATABASE_URL adapted for SQLAlchemy async PostgreSQL.

    The environment remains the single source of truth. We only adapt the
    driver at the infrastructure boundary when the configured URL uses the
    standard PostgreSQL SQLAlchemy scheme.
    """

    value = database_url()

    if value.startswith("postgresql+asyncpg://"):
        return value

    if value.startswith("postgresql://"):
        return value.replace(
            "postgresql://",
            "postgresql+asyncpg://",
            1,
        )

    raise RuntimeError(
        "DATABASE_URL must use a PostgreSQL URL compatible with "
        "the synchronous or async PostgreSQL persistence boundary."
    )


def create_async_database_engine() -> AsyncEngine:
    """Create the canonical asynchronous PostgreSQL engine.

    This is used by durable WS-22 persistence while sharing the same
    DATABASE_URL and canonical Base as the existing persistence layer.
    """

    return create_async_engine(
        async_database_url(),
        pool_pre_ping=True,
    )


def create_async_session_factory(
    engine: AsyncEngine,
) -> async_sessionmaker[AsyncSession]:
    """Create the canonical asynchronous PostgreSQL session factory."""

    return async_sessionmaker(
        bind=engine,
        class_=AsyncSession,
        expire_on_commit=False,
    )

from __future__ import annotations

from typing import Any

from backend.app.application.internal_data.beta_user_read_repository import (
    BetaUserReadRepository,
)


class BetaUserReadService:
    """Application service exposing the approved Beta-user read boundary.

    Security properties:
    - read-only;
    - no SQL;
    - no SQLAlchemy dependency;
    - no ORM objects returned;
    - no repository write methods exposed.
    """

    MAX_USER_ID_LENGTH = 128
    MAX_OUTPUT_STRING_LENGTH = 512

    def __init__(
        self,
        repository: BetaUserReadRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")

        self._repository = repository

    async def get_beta_user(
        self,
        user_id: str,
    ) -> dict[str, Any] | None:
        self._validate_user_id(user_id)

        user = await self._repository.get(user_id)

        if user is None:
            return None

        return self._to_safe_dto(user)

    async def get_beta_user_by_telegram_id(
        self,
        telegram_id: int,
    ) -> dict[str, Any] | None:
        self._validate_telegram_id(telegram_id)

        user = await self._repository.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            return None

        return self._to_safe_dto(user)

    @classmethod
    def _validate_user_id(
        cls,
        user_id: str,
    ) -> None:
        if not isinstance(user_id, str):
            raise ValueError("user_id must be a string.")

        user_id = user_id.strip()

        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if len(user_id) > cls.MAX_USER_ID_LENGTH:
            raise ValueError(
                "user_id exceeds the maximum allowed length."
            )

    @staticmethod
    def _validate_telegram_id(
        telegram_id: int,
    ) -> None:
        if isinstance(telegram_id, bool):
            raise ValueError(
                "telegram_id must be an integer."
            )

        if not isinstance(telegram_id, int):
            raise ValueError(
                "telegram_id must be an integer."
            )

        if telegram_id <= 0:
            raise ValueError(
                "telegram_id must be greater than zero."
            )

    @classmethod
    def _to_safe_dto(
        cls,
        user: Any,
    ) -> dict[str, Any]:
        """Convert the domain object to a controlled DTO.

        Do not return the domain/ORM object itself.
        """

        return {
            "user_id": cls._safe_string(
                getattr(user, "user_id", None)
            ),
            "email": cls._safe_string(
                getattr(user, "email", None)
            ),
            "email_verified": bool(
                getattr(user, "email_verified", False)
            ),
            "beta_approved": bool(
                getattr(user, "beta_approved", False)
            ),
            "telegram_id": getattr(
                user,
                "telegram_id",
                None,
            ),
            "active_world_id": cls._safe_string(
                getattr(user, "active_world_id", None)
            ),
        }

    @classmethod
    def _safe_string(
        cls,
        value: Any,
    ) -> str | None:
        if value is None:
            return None

        value = str(value)

        if len(value) > cls.MAX_OUTPUT_STRING_LENGTH:
            return value[
                : cls.MAX_OUTPUT_STRING_LENGTH
            ]

        return value

from __future__ import annotations

from src.ai_runtime.capabilities.executor import CapabilityExecutor
from src.ai_runtime.capabilities.models import (
    CapabilityExecutionStatus,
    CapabilityRequest,
    CapabilityResult,
    CapabilityType,
)


class InternalDataReadExecutor(CapabilityExecutor):
    """Runtime adapter for the governed internal-data read capability.

    This executor deliberately contains:
    - no SQL;
    - no SQLAlchemy;
    - no database session;
    - no provider SDK;
    - no write operation.

    It delegates to an application-level read service.
    """

    GET_BETA_USER = "get_beta_user"
    GET_BETA_USER_BY_TELEGRAM_ID = (
        "get_beta_user_by_telegram_id"
    )

    _ALLOWED_OPERATIONS = frozenset(
        {
            GET_BETA_USER,
            GET_BETA_USER_BY_TELEGRAM_ID,
        }
    )

    @property
    def capability(self) -> CapabilityType:
        return CapabilityType.INTERNAL_DATA_READ

    def __init__(
        self,
        beta_user_read_service,
    ) -> None:
        if beta_user_read_service is None:
            raise ValueError(
                "beta_user_read_service cannot be None."
            )

        self._service = beta_user_read_service

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:
        if request is None:
            raise ValueError(
                "CapabilityRequest cannot be None."
            )

        if request.capability != self.capability:
            raise ValueError(
                "CapabilityRequest capability does not match "
                "InternalDataReadExecutor."
            )

        payload = request.payload

        operation = payload.get("operation")

        if not isinstance(operation, str):
            raise ValueError(
                "operation must be a string."
            )

        operation = operation.strip()

        if operation not in self._ALLOWED_OPERATIONS:
            raise ValueError(
                f"Unsupported internal data operation: {operation!r}"
            )

        user_id_present = "user_id" in payload
        telegram_id_present = "telegram_id" in payload

        if user_id_present == telegram_id_present:
            raise ValueError(
                "Exactly one of user_id or telegram_id "
                "must be provided."
            )

        if operation == self.GET_BETA_USER:
            if not user_id_present:
                raise ValueError(
                    "get_beta_user requires user_id."
                )

            if telegram_id_present:
                raise ValueError(
                    "get_beta_user does not accept telegram_id."
                )

            output = await (
                self._service.get_beta_user(
                    payload["user_id"]
                )
            )

        else:
            if not telegram_id_present:
                raise ValueError(
                    "get_beta_user_by_telegram_id "
                    "requires telegram_id."
                )

            if user_id_present:
                raise ValueError(
                    "get_beta_user_by_telegram_id "
                    "does not accept user_id."
                )

            output = await (
                self._service.get_beta_user_by_telegram_id(
                    payload["telegram_id"]
                )
            )

        return CapabilityResult(
            request_id=request.request_id,
            capability=self.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output={
                "found": output is not None,
                "user": output,
            },
        )
