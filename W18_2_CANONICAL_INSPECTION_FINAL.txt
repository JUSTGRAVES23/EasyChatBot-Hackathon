"""Application service for learner-to-learning-world authorization."""

from __future__ import annotations

import asyncio
from typing import Protocol
from uuid import UUID

from app.domain.learning.learner_world_access import (
    BETA,
    LearnerWorldAccess,
)
from app.domain.learning.learner_world_access_repository import (
    LearnerWorldAccessRepository,
)


class CanonicalUserIdentityReader(Protocol):
    """Minimal provider-neutral read boundary over canonical user identity."""

    async def get_user_for_identity(self, user_id: UUID):
        """Return the canonical user record or None."""
        ...


class WorldAccessError(ValueError):
    """Expected World access validation failure."""


class UserNotFoundError(WorldAccessError):
    """The requested canonical user does not exist."""


class LearningWorldNotFoundError(WorldAccessError):
    """The requested Learning World does not exist."""


class LearningWorldDisabledError(WorldAccessError):
    """The requested Learning World cannot receive new access."""


class WorldAccessService:
    """
    Authorize learners for one or more Learning Worlds.

    Identity comes from the canonical authentication boundary.
    Beta is represented only by ``access_source=BETA`` and is never used
    as the permanent identity/authorization repository.
    """

    def __init__(
        self,
        *,
        access_repository: LearnerWorldAccessRepository,
        user_identity_reader: CanonicalUserIdentityReader,
        learning_world_repository,
    ) -> None:
        if access_repository is None:
            raise ValueError("access_repository is required.")
        if user_identity_reader is None:
            raise ValueError("user_identity_reader is required.")
        if learning_world_repository is None:
            raise ValueError("learning_world_repository is required.")

        self._access_repository = access_repository
        self._user_identity_reader = user_identity_reader
        self._learning_world_repository = learning_world_repository

    async def grant_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
        access_source: str = BETA,
    ) -> LearnerWorldAccess:
        self._validate_user_id(user_id)
        identifier = self._normalize_world_identifier(world_identifier)

        user = await self._user_identity_reader.get_user_for_identity(user_id)
        if user is None:
            raise UserNotFoundError("Canonical user does not exist.")

        world = await asyncio.to_thread(
            self._learning_world_repository.get_by_identifier,
            identifier,
        )

        if world is None:
            raise LearningWorldNotFoundError(
                f"Learning World does not exist: {identifier}"
            )

        if not world.enabled:
            raise LearningWorldDisabledError(
                f"Learning World is disabled: {identifier}"
            )

        world_id = self._world_id(world)

        existing = await asyncio.to_thread(
            self._access_repository.get_by_user_and_world,
            user_id=user_id,
            world_id=world_id,
        )

        if existing is not None:
            if existing.is_active():
                return existing

            revived = LearnerWorldAccess.grant(
                user_id=user_id,
                world_id=world_id,
                access_source=access_source,
                access_id=existing.access_id,
            )
            return await asyncio.to_thread(
                self._access_repository.save,
                revived,
            )

        access = LearnerWorldAccess.grant(
            user_id=user_id,
            world_id=world_id,
            access_source=access_source,
        )

        return await asyncio.to_thread(
            self._access_repository.save,
            access,
        )

    async def has_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> bool:
        access = await self.get_access(
            user_id=user_id,
            world_identifier=world_identifier,
        )
        return access is not None and access.is_active()

    async def require_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> LearnerWorldAccess:
        access = await self.get_access(
            user_id=user_id,
            world_identifier=world_identifier,
        )

        if access is None or not access.is_active():
            raise PermissionError(
                "Learner does not have active access to this Learning World."
            )

        return access

    async def get_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
    ) -> LearnerWorldAccess | None:
        self._validate_user_id(user_id)
        identifier = self._normalize_world_identifier(world_identifier)

        world = await asyncio.to_thread(
            self._learning_world_repository.get_by_identifier,
            identifier,
        )

        if world is None:
            return None

        world_id = self._world_id(world)

        return await asyncio.to_thread(
            self._access_repository.get_by_user_and_world,
            user_id=user_id,
            world_id=world_id,
        )

    async def list_world_access(
        self,
        *,
        user_id: UUID,
    ) -> tuple[LearnerWorldAccess, ...]:
        self._validate_user_id(user_id)

        return await asyncio.to_thread(
            self._access_repository.list_by_user,
            user_id=user_id,
        )


    @staticmethod
    def _world_id(world) -> UUID:
        world_id = getattr(world, "id", None)
        if not isinstance(world_id, UUID):
            raise WorldAccessError(
                "Learning World catalogue entry must expose the authoritative World UUID as 'id'."
            )
        return world_id

    @staticmethod
    def _validate_user_id(user_id: UUID) -> None:
        if not isinstance(user_id, UUID):
            raise ValueError("user_id must be a UUID.")

    @staticmethod
    def _normalize_world_identifier(world_identifier: str) -> str:
        if not isinstance(world_identifier, str):
            raise ValueError("world_identifier must be a string.")

        identifier = world_identifier.strip()

        if not identifier:
            raise ValueError("world_identifier is required.")

        if len(identifier) > 100:
            raise ValueError("world_identifier is too long.")

        return identifier

"""Domain model for durable learner-to-learning-world access."""

from __future__ import annotations

from dataclasses import dataclass, replace
from datetime import datetime, timezone
from uuid import UUID, uuid4


ACTIVE = "active"
REVOKED = "revoked"

BETA = "beta"
SUBSCRIPTION = "subscription"
ADMIN = "admin"
TRIAL = "trial"

_VALID_STATUSES = frozenset({ACTIVE, REVOKED})
_VALID_SOURCES = frozenset({BETA, SUBSCRIPTION, ADMIN, TRIAL})


@dataclass(frozen=True, slots=True)
class LearnerWorldAccess:
    """Durable authorization record for one learner and one World."""

    access_id: UUID
    user_id: UUID
    world_id: UUID
    status: str
    access_source: str
    created_at: datetime
    updated_at: datetime

    @classmethod
    def grant(
        cls,
        *,
        user_id: UUID,
        world_id: UUID,
        access_source: str,
        access_id: UUID | None = None,
        now: datetime | None = None,
    ) -> "LearnerWorldAccess":
        if not isinstance(user_id, UUID):
            raise ValueError("user_id must be a UUID.")
        if not isinstance(world_id, UUID):
            raise ValueError("world_id must be a UUID.")
        if access_source not in _VALID_SOURCES:
            raise ValueError(f"Unsupported access source: {access_source!r}.")

        timestamp = now or datetime.now(timezone.utc)

        return cls(
            access_id=access_id or uuid4(),
            user_id=user_id,
            world_id=world_id,
            status=ACTIVE,
            access_source=access_source,
            created_at=timestamp,
            updated_at=timestamp,
        )

    def is_active(self) -> bool:
        return self.status == ACTIVE

    def revoke(self, *, now: datetime | None = None) -> "LearnerWorldAccess":
        if self.status == REVOKED:
            return self

        if self.status not in _VALID_STATUSES:
            raise ValueError(f"Unsupported access status: {self.status!r}.")

        timestamp = now or datetime.now(timezone.utc)
        return replace(self, status=REVOKED, updated_at=timestamp)

@dataclass(frozen=True)
class BetaConnectionTokenResult:
    token: str
    connection: BetaConnectionToken

    def __iter__(self) -> Iterator[object]:
        yield self.token
        yield self.connection


@dataclass(frozen=True)
class BetaConversationContext:
    user_id: str
    telegram_user_id: int
    email: str
    world_id: str

    character: str

class BetaAccessService:
    """
    Application boundary for Beta eligibility,
    Telegram binding and authorization.

    Telegram identity invariant:

        One Telegram identity
        -> one Beta account

    A Telegram identity that is already connected cannot
    be rebound to another Beta account through a new
    connection token.

    World assignment invariant:

        A world must be resolved through the authoritative
        WorldResolver before its ID can be persisted as the
        user's active world.

    World conversation invariant:

        A conversation context must be derived from the
        user's persisted active World through WorldResolver.

        Missing or invalid World configuration fails closed.
        The service never silently substitutes a default World.

    World compatibility:

        Legacy Beta/Telegram world identifier:
            world-fr

        Canonical World identifier:
            french_world

    Legacy identifiers are accepted only at this application
    compatibility boundary. New persisted assignments always
    use the canonical identifier returned by WorldResolver.
    """

    LEGACY_WORLD_ALIASES = {
        "world-fr": "french_world",
    }

    def __init__(
        self,
        *,
        user_repository: BetaUserRepository,
        token_repository: BetaTokenRepository,
        binding_repository: TelegramBindingRepository,
        world_resolver: WorldResolver,
        expiration_minutes: int = 1440,
    ) -> None:
        if expiration_minutes <= 0:
            raise ValueError(
                "expiration_minutes must be positive."
            )

        self._users = user_repository
        self._tokens = token_repository
        self._bindings = binding_repository
        self._world_resolver = world_resolver
        self._expiration_minutes = expiration_minutes

    async def approve_beta(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before Beta approval."
            )

        return await self._users.save(
            replace(
                user,
                beta_approved=True,
            )
        )

    async def assign_world(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> BetaUserState:
        if not world_id:
            raise ValueError(
                "world_id is required."
            )

        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before World assignment."
            )

        if not user.beta_approved:
            raise ValueError(
                "User is not Beta approved."
            )

        # Resolve legacy application identifiers to the canonical
        # World identifier before crossing the authoritative
        # WorldResolver boundary.
        canonical_world_id = self.LEGACY_WORLD_ALIASES.get(
            world_id,
            world_id,
        )

        # Resolve the requested World before persistence.
        #
        # This is the authoritative World catalogue boundary.
        # If the World does not exist or is invalid, WorldResolver
        # raises and active_world_id is not modified.
        world = self._world_resolver.resolve(
            canonical_world_id,
        )

        return await self._users.save(
            replace(
                user,
                active_world_id=world.world_id,
            )
        )

    async def create_connection_token(
        self,
        user_id: str,
    ) -> BetaConnectionTokenResult:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email is not verified."
            )

        if not user.beta_approved:
            raise ValueError(
                "User is not Beta approved."
            )

        if not user.active_world_id:
            raise ValueError(
                "User must have a Learning World before Telegram access."
            )

        raw_token = secrets.token_urlsafe(32)

        token = BetaConnectionToken(
            token_id=secrets.token_hex(16),
            user_id=user.user_id,
            token_hash=self.hash_token(
                raw_token
            ),
            expires_at=(
                datetime.now(timezone.utc)
                + timedelta(
                    minutes=self._expiration_minutes
                )
            ),
        )

        await self._tokens.save(token)

        return BetaConnectionTokenResult(
            token=raw_token,
            connection=token,
        )

    async def validate_and_bind(
        self,
        *,
        token: str,
        telegram_id: int,
        chat_id: int,
    ) -> BetaUserState:
        record = await self._tokens.get_by_hash(
            self.hash_token(token)
        )

        if record is None:
            raise ValueError(
                "Invalid connection token."
            )

        if record.used:
            raise ValueError(
                "Connection token already used."
            )

        expires_at = record.expires_at

        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(
                tzinfo=timezone.utc
            )

        if expires_at <= datetime.now(timezone.utc):
            raise ValueError(
                "Connection token expired."
            )

        user = await self._require_user(
            record.user_id
        )

        if not user.eligible:
            raise ValueError(
                "User is not eligible."
            )

        # -------------------------------------------------
        # Telegram identity ownership invariant
        #
        # A Telegram identity may belong to only one
        # Beta account. Never silently transfer ownership.
        # -------------------------------------------------

        existing_binding = (
            await self._bindings.get_by_telegram_user_id(
                str(telegram_id)
            )
        )

        if existing_binding is not None:
            raise ValueError(
                "Telegram identity is already connected."
            )

        await self._bindings.save(
            TelegramBinding(
                binding_id=secrets.token_hex(16),
                user_id=user.user_id,
                telegram_user_id=str(
                    telegram_id
                ),
                chat_id=str(
                    chat_id
                ),
                connected_at=datetime.now(
                    timezone.utc
                ),
            )
        )

        await self._tokens.mark_used(
            replace(
                record,
                used_at=datetime.now(
                    timezone.utc
                ),
            )
        )

        return await self._users.save(
            replace(
                user,
                telegram_id=telegram_id,
            )
        )

    async def authorize_private_chat(
        self,
        telegram_id: int,
    ) -> BetaUserState:
        user = await self._users.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            raise PermissionError(
                "Telegram account is not connected."
            )

        if not user.active:
            raise PermissionError(
                "User is not active."
            )

        return user

    async def get_conversation_context(
        self,
        telegram_user_id: int,
    ) -> BetaConversationContext:
        """
        Resolve the authoritative conversation context for
        an authorized Telegram learner.

        The persisted active_world_id is treated only as the
        World identifier. The authoritative World configuration,
        including its language code, is always obtained through
        WorldResolver.

        This method deliberately fails closed when no active World
        is assigned or when the assigned World cannot be resolved.
        """

        user = await self.authorize_private_chat(
            telegram_user_id
        )

        if not user.active_world_id:
            raise ValueError(
                "Active learning World is required."
            )

        canonical_world_id = self._canonical_world_id(
            user.active_world_id
        )

        # WorldResolver is the authoritative runtime World boundary.
        # No language or World defaults are inferred here.
        world = self._world_resolver.resolve(
            canonical_world_id
        )

        return BetaConversationContext(
            user_id=user.user_id,
            telegram_user_id=telegram_user_id,
            email=user.email,
            world_id=world.world_id,
            language_code=world.language_code,
            character="Alix",
            level="B1",
        )

    @classmethod
    def _canonical_world_id(
        cls,
        world_id: str | None,
    ) -> str:
        """
        Convert an accepted legacy application identifier to the
        canonical World identifier.

        A missing identifier is intentionally not replaced with a
        default World. Callers must fail closed instead.
        """

        if not world_id:
            raise ValueError(
                "Active learning World is required."
            )

        return cls.LEGACY_WORLD_ALIASES.get(
            world_id,
            world_id,
        )

    @staticmethod
    def hash_token(
        token: str,
    ) -> str:
        return hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

    async def _require_user(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._users.get(
            user_id
        )

        if user is None:
            raise ValueError(
                "Academy user does not exist."
            )

        return user

@dataclass(frozen=True, slots=True)
class BetaUserState:
    """
    Domain model representing a Beta user.

    Identity flow:

        beta_users
             |
             v
        telegram_bindings
             |
             v
        Telegram private chat
    """

    user_id: str

    email: str

    email_verified: bool

    beta_approved: bool

    telegram_id: int | None = None

    active_world_id: str | None = None


    @property
    def eligible(self) -> bool:
        """
        User can access Beta if approved and verified.
        """

        return (
            self.email_verified
            and self.beta_approved
        )


    @property
    def active(self) -> bool:
        """
        User can enter private Telegram classroom.
        """

        return (
            self.eligible
            and self.telegram_id is not None
            and self.active_world_id is not None
        )

from __future__ import annotations

from typing import Any

from app.application.internal_data.beta_user_read_repository import (
    BetaUserReadRepository,
)


class BetaUserReadService:
    """Application service exposing the approved Beta-user read boundary.

    Security properties:
    - read-only;
    - no SQL;
    - no SQLAlchemy dependency;
    - no ORM objects returned;
    - no repository write methods exposed.
    """

    MAX_USER_ID_LENGTH = 128
    MAX_OUTPUT_STRING_LENGTH = 512

    def __init__(
        self,
        repository: BetaUserReadRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")

        self._repository = repository

    async def get_beta_user(
        self,
        user_id: str,
    ) -> dict[str, Any] | None:
        self._validate_user_id(user_id)

        user = await self._repository.get(user_id)

        if user is None:
            return None

        return self._to_safe_dto(user)

    async def get_beta_user_by_telegram_id(
        self,
        telegram_id: int,
    ) -> dict[str, Any] | None:
        self._validate_telegram_id(telegram_id)

        user = await self._repository.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            return None

        return self._to_safe_dto(user)

    @classmethod
    def _validate_user_id(
        cls,
        user_id: str,
    ) -> None:
        if not isinstance(user_id, str):
            raise ValueError("user_id must be a string.")

        user_id = user_id.strip()

        if not user_id:
            raise ValueError("user_id cannot be empty.")

        if len(user_id) > cls.MAX_USER_ID_LENGTH:
            raise ValueError(
                "user_id exceeds the maximum allowed length."
            )

    @staticmethod
    def _validate_telegram_id(
        telegram_id: int,
    ) -> None:
        if isinstance(telegram_id, bool):
            raise ValueError(
                "telegram_id must be an integer."
            )

        if not isinstance(telegram_id, int):
            raise ValueError(
                "telegram_id must be an integer."
            )

        if telegram_id <= 0:
            raise ValueError(
                "telegram_id must be greater than zero."
            )

    @classmethod
    def _to_safe_dto(
        cls,
        user: Any,
    ) -> dict[str, Any]:
        """Convert the domain object to a controlled DTO.

        Do not return the domain/ORM object itself.
        """

        return {
            "user_id": cls._safe_string(
                getattr(user, "user_id", None)
            ),
            "email": cls._safe_string(
                getattr(user, "email", None)
            ),
            "email_verified": bool(
                getattr(user, "email_verified", False)
            ),
            "beta_approved": bool(
                getattr(user, "beta_approved", False)
            ),
            "telegram_id": getattr(
                user,
                "telegram_id",
                None,
            ),
            "active_world_id": cls._safe_string(
                getattr(user, "active_world_id", None)
            ),
        }

    @classmethod
    def _safe_string(
        cls,
        value: Any,
    ) -> str | None:
        if value is None:
            return None

        value = str(value)

        if len(value) > cls.MAX_OUTPUT_STRING_LENGTH:
            return value[
                : cls.MAX_OUTPUT_STRING_LENGTH
            ]

        return value

from typing import Protocol

# IMPORTANT:

"""Application service for catalogue-to-runtime World resolution."""
from __future__ import annotations
from app.application.uai.world.world_resolver import WorldResolver
from .repositories import LearningWorldCatalogueRepository

class LearningWorldCatalogueService:
    """Apply catalogue policy and delegate runtime resolution to WorldResolver."""
    def __init__(self, *, repository: LearningWorldCatalogueRepository, world_resolver: WorldResolver) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")
        if world_resolver is None:
            raise ValueError("world_resolver cannot be None.")
        self._repository = repository
        self._world_resolver = world_resolver

    def get_by_identifier(self, identifier: str):
        if not identifier or not identifier.strip():
            raise ValueError("catalogue identifier is required.")
        return self._repository.get_by_identifier(identifier.strip())

    def resolve_runtime_world(self, catalogue_identifier: str):
        entry = self.get_by_identifier(catalogue_identifier)
        if entry is None:
            raise ValueError(f"Learning World not found: {catalogue_identifier.strip()}")
        if not entry.enabled:
            raise ValueError(f"Learning World is disabled: {entry.identifier}")
        if not entry.runtime_world_id:
            raise ValueError(f"Learning World has no runtime configuration: {entry.identifier}")
        return self._world_resolver.resolve(entry.runtime_world_id)

class PostgreSQLLearningWorldCatalogueRepository(
    LearningWorldCatalogueRepository

    """PostgreSQL implementation of the Learning World catalogue."""

    def __init__(self, session_factory: sessionmaker[Session]) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")

        self._session_factory = session_factory

    def get_by_identifier(
        self,
        identifier: str,
    ) -> LearningWorldCatalogueEntry | None:
        if not identifier or not identifier.strip():
            raise ValueError("identifier is required.")

        normalized_identifier = identifier.strip()

        with self._session_factory() as session:
            row = session.scalar(
                select(LearningWorldModel).where(
                    LearningWorldModel.identifier == normalized_identifier
                )
            )

            return None if row is None else self._to_domain(row)

    @staticmethod
    def _to_domain(
        row: LearningWorldModel,
    ) -> LearningWorldCatalogueEntry:
        return LearningWorldCatalogueEntry(
            id=row.id,
            identifier=row.identifier,
            name=row.name,
            description=row.description,
            language=row.language,
            version=row.version,
            status=row.status,

"""PostgreSQL adapter for learner-to-World access."""

from __future__ import annotations

from sqlalchemy import select
from sqlalchemy.orm import Session, sessionmaker

from app.domain.learning.learner_world_access import LearnerWorldAccess
from app.domain.learning.learner_world_access_repository import (
    LearnerWorldAccessRepository,
)
from app.infrastructure.persistence.learner_world_access_model import (
    LearnerWorldAccessRecord,
)


class PostgreSQLLearnerWorldAccessRepository(LearnerWorldAccessRepository):
    """Synchronous repository; application async callers execute it off-loop."""

    def __init__(self, session_factory: sessionmaker[Session]) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")
        self._session_factory = session_factory

    def save(self, access: LearnerWorldAccess) -> LearnerWorldAccess:
        with self._session_factory() as session:
            record = session.get(LearnerWorldAccessRecord, access.access_id)

            if record is None:
                record = LearnerWorldAccessRecord(
                    id=access.access_id,
                    user_id=access.user_id,
                    world_id=access.world_id,
                    status=access.status,
                    access_source=access.access_source,
                    created_at=access.created_at,
                    updated_at=access.updated_at,
                )
                session.add(record)
            else:
                record.status = access.status
                record.access_source = access.access_source
                record.updated_at = access.updated_at

            session.commit()
            session.refresh(record)

            return self._to_domain(record)

    def get_by_id(self, access_id):
        with self._session_factory() as session:
            record = session.get(LearnerWorldAccessRecord, access_id)
            return self._to_domain(record) if record is not None else None

    def get_by_user_and_world(self, *, user_id, world_id):
        with self._session_factory() as session:
            record = session.scalar(
                select(LearnerWorldAccessRecord).where(
                    LearnerWorldAccessRecord.user_id == user_id,
                    LearnerWorldAccessRecord.world_id == world_id,
                )
            )
            return self._to_domain(record) if record is not None else None

    def list_by_user(self, *, user_id):
        with self._session_factory() as session:
            records = session.scalars(
                select(LearnerWorldAccessRecord)
                .where(LearnerWorldAccessRecord.user_id == user_id)
                .order_by(LearnerWorldAccessRecord.created_at)
            ).all()

            return tuple(self._to_domain(record) for record in records)

    @staticmethod
    def _to_domain(record: LearnerWorldAccessRecord) -> LearnerWorldAccess:
        return LearnerWorldAccess(
            access_id=record.id,
            user_id=record.user_id,
            world_id=record.world_id,
            status=record.status,
            access_source=record.access_source,
            created_at=record.created_at,
            updated_at=record.updated_at,
        )

def create_production_beta_telegram_composition(
    *,
    expiration_minutes: int = 1440,
    database_engine: Any | None = None,
    session_factory: Any | None = None,
    database_engine_factory: Callable[[], Any] | None = None,
    session_factory_factory: Callable[[Any], Any] | None = None,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: Callable[[], ConversationRuntime] | None = None,
    functionality_control_service: (
        FunctionalityOperationalControlService | None
    ) = None,
    language_detector: LanguageDetector | None = None,
) -> BetaTelegramProductionComposition:
    """Create the production Beta Telegram composition.

    Production Telegram processing requires the operational
    functionality-control service. The service is injected from
    the application composition; this factory never creates a
    second control service or control repository.

    The production composition also creates the learner Study
    Material service. Study Materials reuse the authoritative
    Learning World catalogue service and its shared WorldResolver.
    """

    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    if functionality_control_service is None:
        raise ValueError(
            "functionality_control_service is required "
            "for production Telegram composition."
        )

    if language_detector is None:
        raise ValueError(
            "language_detector is required "
            "for production Telegram composition."
        )

    if (
        database_engine is not None
        and database_engine_factory is not None
    ):
        raise ValueError(
            "Provide either database_engine or "
            "database_engine_factory, not both."
        )

    if (
        session_factory is not None
        and session_factory_factory is not None
    ):
        raise ValueError(
            "Provide either session_factory or "
            "session_factory_factory, not both."
        )

    if (
        conversation_runtime is not None
        and conversation_runtime_factory is not None
    ):
        raise ValueError(
            "Provide either conversation_runtime or "
            "conversation_runtime_factory, not both."
        )

    if database_engine is None:
        database_engine = (
            database_engine_factory
            or create_database_engine
        )()

    if session_factory is None:
        session_factory = (
            session_factory_factory
            or create_session_factory
        )(database_engine)

    beta_user_repository = PostgreSQLBetaUserRepository(
        session_factory,
    )
    beta_token_repository = PostgreSQLBetaTokenRepository(
        session_factory,
    )
    telegram_binding_repository = (
        PostgreSQLTelegramBindingRepository(
            session_factory,
        )
    )
    conversation_event_repository = (
        PostgreSQLConversationEventRepository(
            session_factory,
        )
    )

    # Durable provider usage ledger.
    # Accounting remains outside the Operations Center.
    provider_usage_repository = (
        PostgreSQLProviderUsageRepository(
            session_factory,
        )
    )

    privacy_identifier_service = PrivacyIdentifierService()
    conversation_event_validator = (
        ConversationEventValidator()
    )

    # One authoritative WorldResolver is shared by Beta access
    # and Learning World catalogue resolution.
    app_root = Path(__file__).resolve().parents[2]
    world_repository = YamlWorldRepository(
        worlds_path=app_root / "configuration" / "worlds"
    )
    world_resolver = WorldResolver(
        world_repository=world_repository,
    )

    # Authoritative Learning World catalogue.
    #
    # This service is shared with Study Material access. Do not
    # create another WorldResolver or another catalogue service.
    catalogue_repository = (
        PostgreSQLLearningWorldCatalogueRepository(
            session_factory
        )
    )
    catalogue_service = LearningWorldCatalogueService(
        repository=catalogue_repository,
        world_resolver=world_resolver,
    )

    # M2-C5 Study Materials.
    #
    # The repository owns PostgreSQL persistence while the
    # application service owns learner-facing catalogue access.
    #
    # Study Material World validation deliberately reuses the
    # authoritative LearningWorldCatalogueService above.
    study_material_repository = (
        PostgreSQLStudyMaterialRepository(
            session_factory,
        )
    )
    study_material_service = StudyMaterialService(
        repository=study_material_repository,
        world_catalogue_service=catalogue_service,
    )

    beta_telegram = create_beta_telegram_composition(
        user_repository=beta_user_repository,
        token_repository=beta_token_repository,
        binding_repository=telegram_binding_repository,
        conversation_event_repository=(
            conversation_event_repository
        ),
        privacy_identifier_service=(
            privacy_identifier_service
        ),
        conversation_event_validator=(
            conversation_event_validator
        ),
        expiration_minutes=expiration_minutes,
        conversation_runtime=conversation_runtime,
        conversation_runtime_factory=(
            conversation_runtime_factory
        ),
        provider_usage_repository=(
            provider_usage_repository
        ),
        functionality_control_service=(
            functionality_control_service
        ),
        world_resolver=world_resolver,
        language_response_policy=LanguageResponsePolicy(),
        language_detector=language_detector,
    )

    logger.info(
        "Production Beta Telegram composition created "
        "with durable usage ledger, functionality control, "
        "and Study Material service."
    )

    return BetaTelegramProductionComposition(
        beta_telegram=beta_telegram,
        database_engine=database_engine,
        session_factory=session_factory,
        catalogue_service=catalogue_service,
        study_material_service=study_material_service,
    )

def _required_environment(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for authentication email composition."
        )

    return value

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )

    # Developer Administration
    #
    # Production path:
    #
    #   DeveloperAdminService
    #          |
    #          +--> DeveloperAdminUnitOfWorkFactory
    #                  |
    #                  +--> fresh PostgreSQLDeveloperAdminUnitOfWork
    #                          |
    #                          +--> fresh AsyncSession
    #                          |
    #                          +--> fresh transaction
    #                          |
    #                          +--> auth_user_roles
    #                          |
    #                          +--> developer_admin_audit_events
    #          |
    #          +--> DeveloperAdminAuditService
    #
    # Each Developer Admin mutation receives a fresh Unit-of-Work
    # instance. The role mutation and its security audit evidence
    # therefore remain within the same database transaction while
    # mutable UoW/session state is never shared between operations.
    #
    # The standalone audit repository owns transactions only when its
    # standalone ``record`` operation is used. The production Developer
    # Admin mutation uses ``record_in_session`` through the UoW.
    # ------------------------------------------------------------------

    developer_admin_unit_of_work_factory = (
        PostgreSQLDeveloperAdminUnitOfWorkFactory(
            auth_session_factory,

        DeveloperAdminAuditService()

    # W15.2.1 Transaction Authorization
    #
    # Transaction authorization is composed once at the application root.
    # The registry and consumer are application-lifetime components and

    # Authentication assurance remains a separate trusted-evidence

class FakeUser:
    def __init__(self, user_id: UUID) -> None:
        self.id = user_id


class FakeUserIdentityReader:
    """Deterministic fake for the canonical user-identity boundary."""

    def __init__(self, users: set[UUID] | None = None) -> None:
        self.users = users or set()

    async def get_user_for_identity(
        self,
        user_id: UUID,
    ) -> FakeUser | None:
        if user_id not in self.users:
            return None

        return FakeUser(user_id)


class FakeWorld:
    def __init__(
        self,
        *,
        world_id: UUID,
        identifier: str = "french_world",
        enabled: bool = True,
    ) -> None:
        self.id = world_id
        self.identifier = identifier
        self.enabled = enabled


class FakeWorldRepository:
    def __init__(self, world: FakeWorld | None) -> None:
        self.world = world

    def get_by_identifier(
        self,
        identifier: str,
    ) -> FakeWorld | None:
        if self.world is None:
            return None

        if self.world.identifier != identifier:
            return None

        return self.world


class FakeAccessRepository:
    def __init__(self) -> None:
        self.records: dict[
            tuple[UUID, UUID],
            LearnerWorldAccess,
        ] = {}

    def save(
        self,
        access: LearnerWorldAccess,
    ) -> LearnerWorldAccess:
        self.records[
            (access.user_id, access.world_id)
        ] = access

        return access

    def get_by_id(
        self,
        access_id: UUID,
    ) -> LearnerWorldAccess | None:
        for access in self.records.values():
            if access.access_id == access_id:
                return access

        return None

    def get_by_user_and_world(
        self,
        *,
        user_id: UUID,
        world_id: UUID,
    ) -> LearnerWorldAccess | None:
        return self.records.get(
            (user_id, world_id)
        )

    def list_by_user(
        self,
        *,
        user_id: UUID,
    ) -> tuple[LearnerWorldAccess, ...]:
        return tuple(
            access
            for access in self.records.values()
            if access.user_id == user_id
        )


def run(awaitable):
    return asyncio.run(awaitable)


def create_service(

def test_grant_access_creates_active_beta_access() -> None:
    user_id = uuid4()
    world_id = uuid4()

    service, access_repository = create_service(
        user_id=user_id,
        world=FakeWorld(

    )

    access = run(

    )

    assert access.user_id == user_id
    assert access.world_id == world_id
    assert access.status == ACTIVE
    assert access.access_source == BETA
    assert access.is_active() is True

    stored = access_repository.get_by_user_and_world(
        user_id=user_id,
        world_id=world_id,
    )

    assert stored is not None
    assert stored.access_id == access.access_id


def test_grant_access_is_idempotent_for_existing_active_access() -> None:

    first = run(
        service.grant_access(
            user_id=user_id,
