"""

Validates execution lifecycle
for prepared AI character voice calls
using execution context.
"""

from app.ai_runtime.voice.voice_call_execution_runtime import (
    VoiceCallExecutionRuntime,
)


from app.domain.uai.voice.voice_call_execution_context import (
    VoiceCallExecutionContext,
)


from app.domain.uai.voice.voice_call_room import (
    VoiceCallRoom,
)


from app.domain.uai.voice.voice_call_participant import (
    VoiceCallParticipant,
    VoiceParticipantRole,
)


from app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)


from app.domain.uai.voice.voice_call_state import (
    VoiceCallState,
)



def create_session(
    *,
    state: VoiceCallState,
) -> VoiceCallSession:
    """
    Create test voice call session.
    """

    return VoiceCallSession(
        session_id="call-001",
        learner_id="learner-001",
        character_id="yeti",
        world_id="french_world",
        language_code="fr",
        state=state,
    )



def create_room() -> VoiceCallRoom:
    """
    Create valid voice room.
    """

    return VoiceCallRoom(
        room_id="room-001",
        participants=(
            VoiceCallParticipant(
                participant_id="learner-001",
                role=VoiceParticipantRole.LEARNER,
                display_name="Learner",
            ),
            VoiceCallParticipant(
                participant_id="yeti",
                role=VoiceParticipantRole.CHARACTER,
                display_name="Yeti",
            ),
        ),
    )



def create_context(
    *,
    state: VoiceCallState,
) -> VoiceCallExecutionContext:
    """
    Create execution context.
    """

    return VoiceCallExecutionContext(
        session=create_session(
            state=state,
        ),
        room=create_room(),
    )



def test_start_ready_voice_call():

    runtime = VoiceCallExecutionRuntime()


    session = runtime.start(
        context=create_context(
            state=VoiceCallState.READY,
        )
    )


    assert session.state == (
        VoiceCallState.ACTIVE
    )


    assert session.character_id == (
        "yeti"
    )



def test_stop_active_voice_call():

    runtime = VoiceCallExecutionRuntime()


    session = runtime.stop(
        context=create_context(
            state=VoiceCallState.ACTIVE,
        )
    )


    assert session.state == (
        VoiceCallState.ENDED
    )



def test_fail_voice_call_execution():

    runtime = VoiceCallExecutionRuntime()


    session = runtime.fail(
        context=create_context(
            state=VoiceCallState.ACTIVE,
        )
    )


    assert session.state == (
        VoiceCallState.FAILED
    )



def test_start_without_ready_state_is_rejected():

    runtime = VoiceCallExecutionRuntime()


    try:

        runtime.start(
            context=create_context(
                state=VoiceCallState.CREATED,
            )
        )


        assert False

    except ValueError:

        assert True



def test_stop_without_active_state_is_rejected():

    runtime = VoiceCallExecutionRuntime()


    try:

        runtime.stop(

        )

    except ValueError:

        assert True

def decode_jwt_payload(
    token: str,
) -> dict:
    """
    Decode the JWT payload without verifying
    the signature.

    Signature verification belongs to LiveKit.
    These tests only validate the claims generated
    by our infrastructure adapter.
    """

    parts = token.split(".")

    assert len(parts) == 3

    payload = parts[1]

    padding = (
        "="
        * (
            4
            - len(payload) % 4
        )
    )

    decoded = base64.urlsafe_b64decode(
        payload + padding
    )

    return json.loads(
        decoded.decode("utf-8")
    )



def test_create_livekit_token():

    service = (
        LiveKitTokenService()
    )


    token = (
        service.create_token(
            room_name="yeti-room-001",
        )
    )


    assert isinstance(
        token,
        VoiceConnectionToken,
    )


    assert token.room_name == (
        "yeti-room-001"
    )


    assert token.provider == (
        "livekit"
    )


    # A real LiveKit JWT is generated.
    assert token.token.count(".") == 2


    assert token.token.startswith(
        "eyJ"
    )


    payload = (
        decode_jwt_payload(
            token.token
        )
    )


    assert payload["sub"].startswith(
        "learner-"
    )


    assert payload["video"]["roomJoin"] is True


    assert payload["video"]["room"] == (
        "yeti-room-001"
    )



def test_custom_expiration_time():

    service = (
        LiveKitTokenService()
    )


    token = (
        service.create_token(
            room_name="room-001",
            expires_minutes=60,
        )
    )


    now = (
        datetime.now(
            timezone.utc
        )
    )


    assert token.expires_at > (
        now
    )


    remaining_seconds = (
        token.expires_at - now
    ).total_seconds()


    # Allow a small execution-time tolerance.
    assert 3500 < (
        remaining_seconds
    ) <= 3600


    payload = (
        decode_jwt_payload(
            token.token
        )
    )


    assert payload["exp"] > (
        payload["nbf"]
    )
