    """Describes the operational data flow for one processing activity."""

    map_id: str
    processing_id: str
    source: str
    input_data: tuple[str, ...]
    runtime_processing: tuple[str, ...]
    pseudonymisation: str
    persistence: tuple[str, ...]

    transfer_safeguard: str | None = None

    def __init__(self, entries: list[DataProcessingMapEntry] | None = None) -> None:
        self._entries = {entry.processing_id: entry for entry in entries or []}

    def add(self, entry: DataProcessingMapEntry) -> None:
        if entry.processing_id in self._entries:
            raise ValueError(
                f"Data map already exists for: {entry.processing_id}"
            )
        self._entries[entry.processing_id] = entry

    def get(self, processing_id: str) -> DataProcessingMapEntry:
        return self._entries[processing_id]

    def all(self) -> tuple[DataProcessingMapEntry, ...]:
        return tuple(self._entries.values())

from __future__ import annotations

from .privacy_change import PrivacyChange, PrivacyChangeStatus


class PrivacyChangeRegistry:
    """In-memory application registry for controlled privacy changes.

    Beta 1 intentionally keeps this registry simple. Persistence can be
    introduced behind the same contract once the workflow is proven.
    """

    def __init__(self) -> None:
        self._changes: dict[str, PrivacyChange] = {}

    def register(self, change: PrivacyChange) -> PrivacyChange:
        if change is None:
            raise ValueError("change cannot be None.")
        if change.change_id in self._changes:
            raise ValueError(f"Privacy change already registered: {change.change_id}")
        self._changes[change.change_id] = change
        return change

    def get(self, change_id: str) -> PrivacyChange | None:
        return self._changes.get(change_id)

    def list_all(self) -> list[PrivacyChange]:
        return list(self._changes.values())

    def update(self, change: PrivacyChange) -> PrivacyChange:
        if change is None:
            raise ValueError("change cannot be None.")
        if change.change_id not in self._changes:
            raise KeyError(f"Unknown privacy change: {change.change_id}")
        self._changes[change.change_id] = change
        return change

    def mark_blocked(self, change_id: str) -> PrivacyChange:
        change = self._require(change_id)
        updated = _replace(change, status=PrivacyChangeStatus.BLOCKED)
        self._changes[change_id] = updated
        return updated

    def _require(self, change_id: str) -> PrivacyChange:
        change = self.get(change_id)
        if change is None:
            raise KeyError(f"Unknown privacy change: {change_id}")
        return change


def _replace(change: PrivacyChange, **changes) -> PrivacyChange:
    from dataclasses import replace
    return replace(change, **changes)

from __future__ import annotations

from dataclasses import replace
from typing import Any

from .privacy_approval_email_service import (
    PrivacyApprovalEmailService,
)
from .privacy_change import (
    PrivacyChange,
    PrivacyChangeStatus,
)
from .privacy_change_registry import (
    PrivacyChangeRegistry,
)
from .privacy_validation_service import (
    PrivacyValidationResult,
    PrivacyValidationService,
)


class PrivacyGovernanceGate:
    """Privacy governance decision gate.

    Responsibilities:
    - consume deterministic privacy validation results;
    - block changes with validation failures;
    - route privacy-impacting changes to DPO/GRC;
    - trigger an approval escalation only when validation passes;
    - never interpret email delivery as approval.

    Does NOT:
    - make legal decisions;
    - grant DPO/GRC approval;
    - execute production changes;
    - schedule recurring approval emails.
    """

    def __init__(
        self,
        *,
        registry: PrivacyChangeRegistry,
        validation_service: PrivacyValidationService | None = None,
        approval_email_service: PrivacyApprovalEmailService | None = None,
    ) -> None:
        if registry is None:
            raise ValueError(
                "registry cannot be None."
            )

        self._registry = registry
        self._validation = validation_service
        self._approval_email = approval_email_service

    def route(
        self,
        change: PrivacyChange,
        *,
        automated_validation_passed: bool | None = None,
        dpo_approved: bool = False,
        grc_approved: bool = False,
    ) -> PrivacyChange:
        """Route a change through the governance decision.

        This synchronous method preserves the existing Beta contract.

        If a validation service is configured, it is the authoritative
        source for automated privacy validation.
        """

        if change is None:
            raise ValueError(
                "change cannot be None."
            )

        validation_passed = self._resolve_validation(
            automated_validation_passed
        )

        if not validation_passed:
            updated = replace(
                change,
                status=PrivacyChangeStatus.BLOCKED,
            )

        elif change.dpo_required or change.grc_required:
            if not (
                dpo_approved
                and grc_approved
            ):
                updated = replace(
                    change,
                    status=(
                        PrivacyChangeStatus.PENDING_DPO_GRC
                    ),
                )
            else:
                updated = replace(
                    change,
                    status=(
                        PrivacyChangeStatus.APPROVED
                    ),
                )

        else:
            updated = replace(
                change,
                status=(
                    PrivacyChangeStatus.APPROVED
                ),
            )

        self._save(updated)

        return updated

    async def route_and_escalate(
        self,
        change: PrivacyChange,
    ) -> PrivacyChange:
        """Run automated validation and escalate when governance requires it.

        Email is sent ONLY when:

        1. PrivacyValidationService returns PASS;
        2. the change requires DPO/GRC review;
        3. the change is not already approved.

        Email delivery never changes the governance status to APPROVED.
        """

        if change is None:
            raise ValueError(
                "change cannot be None."
            )

        if self._validation is None:
            raise RuntimeError(
                "PrivacyValidationService is required "
                "for automatic governance routing."
            )

        if self._approval_email is None:
            raise RuntimeError(
                "PrivacyApprovalEmailService is required "
                "for automatic governance escalation."
            )

        validation = self._validation.validate()

        if validation.status != "PASS":
            updated = replace(
                change,
                status=PrivacyChangeStatus.BLOCKED,
            )

            self._save(updated)
            return updated

        if not (
            change.dpo_required
            or change.grc_required
        ):
            updated = replace(
                change,
                status=PrivacyChangeStatus.APPROVED,
            )

            self._save(updated)
            return updated

        updated = replace(
            change,
            status=(
                PrivacyChangeStatus.PENDING_DPO_GRC
            ),
        )

        self._save(updated)

        self._approval_email.escalate(
    change=updated,
    validation_passed=True,
)
        return updated

    def can_proceed(
        self,
        change_id: str,
    ) -> bool:
        change = self._registry.get(
            change_id
        )

        return bool(
            change
            and change.status
            == PrivacyChangeStatus.APPROVED
        )

    def _resolve_validation(
        self,
        explicit_result: bool | None,
    ) -> bool:
        if self._validation is not None:
            result = self._validation.validate()

            return (
                result.status == "PASS"
            )

        if explicit_result is None:
            raise ValueError(
                "automated_validation_passed is required "
                "when no PrivacyValidationService is configured."
            )

        return explicit_result

    def _save(
        self,
        change: PrivacyChange,
    ) -> None:
        existing = self._registry.get(
            change.change_id
        )

        if existing is None:
            self._registry.register(
                change
            )
        else:
            self._registry.update(
                change
            )

from __future__ import annotations

import os
from typing import Any

import resend

from backend.app.application.privacy.privacy_approval_email_service import (
    ApprovalEmailRequest,
)


class ResendApprovalEmailSender:
    """Resend infrastructure adapter for privacy approval emails.

    Responsibilities:
    - translate ApprovalEmailRequest into a Resend API request;
    - obtain the API key from configuration;
    - delegate delivery to Resend.

    Does NOT:
    - approve or reject privacy changes;
    - modify governance state;
    - perform privacy validation;
    - make DPO/GRC decisions.

    No secrets are stored in source code.
    """

    def __init__(
        self,
        *,
        api_key: str | None = None,
        sender_email: str | None = None,
    ) -> None:
        self._api_key = (
            api_key
            if api_key is not None
            else os.getenv("RESEND_API_KEY", "")
        )

        self._sender_email = (
            sender_email
            if sender_email is not None
            else os.getenv(
                "PRIVACY_APPROVAL_EMAIL_FROM",
                "",
            )
        )

    def send(
        self,
        request: ApprovalEmailRequest,
    ) -> None:
        """Send a privacy approval email through Resend."""

        if request is None:
            raise ValueError(
                "request cannot be None."
            )

        if not self._api_key.strip():
            raise RuntimeError(
                "RESEND_API_KEY is not configured."
            )

        if not self._sender_email.strip():
            raise RuntimeError(
                "PRIVACY_APPROVAL_EMAIL_FROM is not configured."
            )

        resend.api_key = self._api_key

        try:
            resend.Emails.send(
                {
                    "from": self._sender_email,
                    "to": [request.recipient],
                    "subject": request.subject,
                    "text": request.body,
                }
            )
        except Exception as exc:
            raise RuntimeError(
                "Resend privacy approval email delivery failed."
            ) from exc
