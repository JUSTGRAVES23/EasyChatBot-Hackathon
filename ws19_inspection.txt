from __future__ import annotations

import pytest

    BetaTelegramConversationBoundary,
)


@dataclass(frozen=True)
class FakeConversationContext:
    world_id: str = "world-fr"
    character: str = "Alix"
    level: str = "B1"


class FakeBetaAccessService:
    """Deterministic Beta authorization boundary."""

    def __init__(
        self,
        *,
        approved_users: set[int] | None = None,
    ) -> None:
        self.approved_users = (
            approved_users
            if approved_users is not None
            else {111, 222}
        )

        self.calls: list[int] = []

    async def get_conversation_context(
        self,
        telegram_user_id: int,
    ) -> FakeConversationContext:
        self.calls.append(
            telegram_user_id,
        )

        if telegram_user_id not in self.approved_users:
            raise PermissionError(
                "Telegram user is not approved."
            )

        return FakeConversationContext()


class FakeConversationService:
    """
    Security-focused conversation-service double.

    State is explicitly scoped by Telegram identity so
    accidental cross-user mixing becomes observable.
    """

    def __init__(self) -> None:
        self.calls: list[dict] = []
        self.sessions: dict[int, list[str]] = {}

    async def handle_telegram_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        self.calls.append(
            {
                "telegram_user_id": telegram_user_id,
                "message": message,
            }
        )

        history = self.sessions.setdefault(
            telegram_user_id,
            [],
        )

        history.append(message)

        return (
            f"response:{telegram_user_id}:"
            f"{len(history)}"
        )

    async def get_recent_history(
        self,
        *,
        telegram_user_id: int,
        limit: int = 20,
    ) -> list[str]:
        history = self.sessions.get(
            telegram_user_id,
            [],
        )

        return history[-limit:]


def make_boundary(
    *,
    approved_users: set[int] | None = None,
):
    access = FakeBetaAccessService(
        approved_users=approved_users,
    )

    conversation_service = FakeConversationService()

    boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation_service,
        beta_access_service=access,
    )

    return (
        boundary,
        access,
        conversation_service,
    )


@pytest.mark.asyncio
async def test_ws09_unauthorized_identity_cannot_enter_conversation():
    (
        boundary,

        conversation_service,
    ) = make_boundary(

    )

        telegram_user_id=999,

    )

    assert access.calls == [

from dataclasses import dataclass

    level: str = "B1"

class FakeConversationService:

        self.calls: list[dict] = []

    async def handle_telegram_message(
        self,

        return "Bonjour !"


class FakeRateLimiter:
    """
    Deterministic WS-11 rate-limit policy double.

    The limiter is intentionally placed at the application boundary
    in these integration tests. Production wiring may provide a
    different implementation.
    """

    def __init__(
        self,
        *,
        limit: int = 3,
    ) -> None:
        self.limit = limit
        self.calls: list[int] = []
        self._counts: dict[int, int] = {}

    async def allow(
        self,
        *,
        telegram_user_id: int,
    ) -> bool:
        self.calls.append(telegram_user_id)

        count = self._counts.get(
            telegram_user_id,
            0,
        )

        if count >= self.limit:
            return False

        self._counts[telegram_user_id] = count + 1
        return True


class RateLimitedBoundary:
    """
    Minimal WS-11 integration composition.

    This test composition intentionally keeps rate limiting separate
    from the conversation service and from Telegram transport.
    """

    RATE_LIMIT_RESPONSE = (
        "You have reached the classroom message limit. "
        "Please try again later."
    )

    def __init__(
        self,
        *,
        conversation_service: FakeConversationService,
        beta_access_service: FakeBetaAccessService,
        rate_limiter: FakeRateLimiter,
    ) -> None:
        self._conversation_service = conversation_service
        self._beta_access_service = beta_access_service
        self._rate_limiter = rate_limiter

        self._conversation_boundary = BetaTelegramConversationBoundary(
            conversation_service=conversation_service,
            beta_access_service=beta_access_service,
        )

    async def handle_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        # Input validation remains owned by the existing boundary.
        # Rate limiting is applied before downstream conversation
        # processing.
        if not await self._rate_limiter.allow(
            telegram_user_id=telegram_user_id,
        ):
            return self.RATE_LIMIT_RESPONSE

        return await self._conversation_boundary.handle_message(
            telegram_user_id=telegram_user_id,
            message=message,
        )


def make_boundary(
    *,
    approved: bool = True,
    limit: int = 3,
):
    access = FakeBetaAccessService(
        approved=approved,
    )
    conversation_service = FakeConversationService()
    rate_limiter = FakeRateLimiter(
        limit=limit,
    )

    boundary = RateLimitedBoundary(
        conversation_service=conversation_service,
        beta_access_service=access,
        rate_limiter=rate_limiter,
    )

    return (
        boundary,
        access,
        conversation_service,
        rate_limiter,
    )

class FakeAbuseProtector:
    """
    Deterministic WS-12 abuse-protection policy double.

    The policy counts accepted requests per Telegram identity and
    blocks requests after the configured burst threshold.
    """

    def __init__(
        self,
        *,
        max_messages: int = 3,
    ) -> None:
        self.max_messages = max_messages
        self.calls: list[int] = []
        self._counts: dict[int, int] = {}

    async def allow(
        self,
        *,
        telegram_user_id: int,
    ) -> bool:
        self.calls.append(telegram_user_id)

        count = self._counts.get(
            telegram_user_id,
            0,
        )

        if count >= self.max_messages:
            return False

        self._counts[telegram_user_id] = count + 1
        return True


class AbuseProtectedBoundary:
    """
    Minimal WS-12 integration composition.

    Abuse protection remains a separate application policy and does
    not belong inside the Telegram transport adapter or conversation
    service.
    """

    ABUSE_LIMIT_RESPONSE = (
        "Sorry, you are sending messages too quickly. "
        "Please wait a moment and try again."
    )

    def __init__(
        self,
        *,
        conversation_service: FakeConversationService,
        beta_access_service: FakeBetaAccessService,
        abuse_protector: FakeAbuseProtector,
    ) -> None:
        self._conversation_service = conversation_service
        self._beta_access_service = beta_access_service
        self._abuse_protector = abuse_protector

        self._conversation_boundary = BetaTelegramConversationBoundary(
            conversation_service=conversation_service,
            beta_access_service=beta_access_service,
        )

    async def handle_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        # Input validation remains owned by the established
        # conversation boundary.
        self._conversation_boundary._validate_telegram_user_id(
            telegram_user_id,
        )
        self._conversation_boundary._validate_message(
            message,
        )

        if not await self._abuse_protector.allow(
            telegram_user_id=telegram_user_id,
        ):
            return self.ABUSE_LIMIT_RESPONSE

        return await self._conversation_boundary.handle_message(
            telegram_user_id=telegram_user_id,
            message=message,
        )

    )

class FakeObservability:
    """
    Deterministic WS-13 observability/audit event sink.

    Events intentionally contain only operational metadata. Raw
    learner message content is not accepted by the event contract.
    """

    def __init__(
        self,
        *,
        error: Exception | None = None,
    ) -> None:
        self.error = error
        self.events: list[dict] = []

    async def record(
        self,
        *,
        event_type: str,
        telegram_user_id: int,
        outcome: str,
    ) -> None:
        if self.error is not None:
            raise self.error

        self.events.append(
            {
                "event_type": event_type,
                "telegram_user_id": telegram_user_id,
                "outcome": outcome,
            }
        )


class ObservableBoundary:
    """
    Minimal WS-13 integration composition.

    Observability is deliberately separated from the conversation
    service. An observability failure must not become a classroom
    application failure.
    """

    SAFE_ERROR_RESPONSE = (
        "Sorry, something went wrong while processing your message. "
        "Please try again later."
    )

    UNAUTHORIZED_RESPONSE = (
        "Sorry, this Telegram account is not authorized "
        "for Beta access."
    )

    def __init__(
        self,
        *,
        conversation_service: FakeConversationService,
        beta_access_service: FakeBetaAccessService,
        observability: FakeObservability,
    ) -> None:
        self._conversation_service = conversation_service
        self._beta_access_service = beta_access_service
        self._observability = observability

    async def _record(
        self,
        *,
        event_type: str,
        telegram_user_id: int,
        outcome: str,
    ) -> None:
        try:
            await self._observability.record(
                event_type=event_type,
                telegram_user_id=telegram_user_id,
                outcome=outcome,
            )
        except Exception:
            # Observability is non-blocking infrastructure. Failure
            # must never break the classroom flow.
            return

    async def handle_message(
        self,
        *,
        telegram_user_id: int,
        message: str,
    ) -> str:
        if isinstance(telegram_user_id, bool):
            raise ValueError(
                "Telegram user ID must be a positive integer."
            )

        if not isinstance(telegram_user_id, int):
            raise TypeError(
                "Telegram user ID must be a positive integer."
            )

        if telegram_user_id <= 0:
            raise ValueError(
                "Telegram user ID must be a positive integer."
            )

        if message is None:
            raise ValueError(
                "Telegram message cannot be empty."
            )

        if not isinstance(message, str):
            raise TypeError(
                "Telegram message must be a string."
            )

        if not message.strip():
            raise ValueError(
                "Telegram message cannot be empty."
            )

        clean_message = message.strip()

        if len(clean_message) > 4096:
            raise ValueError(
                "Telegram message is too long. "
                "Maximum length is 4096 characters."
            )

        try:
            context = await (
                self._beta_access_service
                .get_conversation_context(
                    telegram_user_id
                )
            )
        except PermissionError:
            await self._record(
                event_type="telegram_classroom_authorization",
                telegram_user_id=telegram_user_id,
                outcome="denied",
            )

            return self.UNAUTHORIZED_RESPONSE

        # Context is operational metadata and is never injected into
        # the learner message or observability event content.
        _ = context

        try:
            response = await (
                self._conversation_service
                .handle_telegram_message(
                    telegram_user_id=telegram_user_id,
                    message=clean_message,
                )
            )
        except Exception:
            await self._record(
                event_type="telegram_classroom_conversation",
                telegram_user_id=telegram_user_id,
                outcome="error",
            )

            return self.SAFE_ERROR_RESPONSE

        await self._record(
            event_type="telegram_classroom_conversation",
            telegram_user_id=telegram_user_id,
            outcome="success",
        )

        return response


def make_boundary(
    *,
    approved: bool = True,
    conversation_error: Exception | None = None,
    observability_error: Exception | None = None,
):
    access = FakeBetaAccessService(

    )

    return (
        boundary,
        access,
        conversation_service,
        observability,
    )
