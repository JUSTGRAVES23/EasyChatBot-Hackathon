from __future__ import annotations


from dataclasses import dataclass
from datetime import datetime
from typing import Any


@dataclass(frozen=True, slots=True)
class MemoryCandidate:
    """
    Temporary educational memory proposal.

    Compatibility boundary between:

    Beta memory workflow
        +
    Future domain MemoryItem model.


    Not persisted.
    """

    candidate_id: str

    learner_id: str | None = None

    memory_type: str | None = None

    content: str | None = None

    confidence_score: float | None = None

    source_event_id: str | None = None

    metadata: dict[str, Any] | None = None

    created_at: datetime | None = None


    # -------------------------------------------------
    # Backward compatibility fields
    # -------------------------------------------------

    learner_hash: str | None = None

    category: str | None = None

    observation: str | None = None

    confidence: float | None = None

    evidence_event_ids: list[str] | None = None



    def normalized_memory_type(self) -> str:
        """
        Returns unified memory type.
        """

        return (
            self.memory_type
            or self.category
            or ""
        )



    def normalized_content(self) -> str:
        """
        Returns unified content.
        """

        return (
            self.content
            or self.observation
            or ""
        )



    def normalized_confidence(self) -> float:
        """
        Returns unified confidence.
        """

        return (
            self.confidence_score
            if self.confidence_score is not None
            else self.confidence
            or 0.0
        )



    def normalized_learner_id(self) -> str:
        """
        Returns learner identifier.

        Future:
        this becomes learner_id only.
        """

        return (
            self.learner_id
            or self.learner_hash
            or ""
        )



    def normalized_source_events(self) -> list[str]:
        """
        Returns evidence events.
        """

        if self.evidence_event_ids:
            return self.evidence_event_ids

        if self.source_event_id:
            return [self.source_event_id]

        return []

from __future__ import annotations


from typing import Final


class MemoryPolicy:
    """
    Governance policy for learner memory.

    Application boundary.

    Responsibilities:

    - educational relevance validation
    - privacy minimisation
    - prevent unnecessary personal data retention

    Does NOT:

    - persist memory
    - approve memory
    - create learner profiles
    """



    ALLOWED_TYPES: Final = {
        "learning_progress",
        "grammar_mistake",
        "grammar_errors",
        "vocabulary",
        "vocabulary_gap",
        "pronunciation",
        "pronunciation_issue",
        "confidence",
        "confidence_level",
        "learning_preference",
        "communication_strength",
    }



    FORBIDDEN_TERMS: Final = {
        "email",
        "phone",
        "password",
        "passport",
        "address",
        "medical",
        "religion",
        "political",
        "financial",
    }



    def validate(
        self,
        *,
        memory_type: str | None = None,
        content: str | None = None,
        category: str | None = None,
        observation: str | None = None,
    ) -> bool:
        """
        Validate memory candidate.

        Supports:

        Current approval flow:

            memory_type
            content


        Legacy:

            category
            observation
        """



        normalized_type = (
            memory_type
            or category
            or ""
        )


        normalized_content = (
            content
            or observation
            or ""
        )



        if not normalized_type:

            raise ValueError(
                "Memory type required."
            )



        if normalized_type not in self.ALLOWED_TYPES:

            raise ValueError(
                f"Memory type not allowed: {normalized_type}"
            )



        lowered = normalized_content.lower()



        for term in self.FORBIDDEN_TERMS:

            if term in lowered:

                raise ValueError(
                    f"Forbidden personal information detected: {term}"
                )



        return True

from __future__ import annotations

import logging

from typing import Any

from backend.app.application.memory.memory_candidate import (
    MemoryCandidate,
)

from backend.app.application.memory.memory_policy import (
    MemoryPolicy,
)


logger = logging.getLogger(__name__)


class MemoryApprovalService:
    """
    Memory governance approval service.

    Responsibilities:

    - validate memory candidates
    - approve educational memories
    - reject unsuitable memories
    - create governance decisions

    Does NOT:

    - persist memory
    - call AI providers
    - modify learner profiles
    - bypass privacy controls

    Beta 1.5.4 boundary:

        MemoryCandidate
              |
              v
        MemoryApprovalService
              |
              v
        MemoryPolicy
              |
              v
        Governance Decision

    Future:

        Approved Candidate
              |
              v
        Domain MemoryItem
    """

    def __init__(
        self,
        *,
        memory_policy: MemoryPolicy,
    ) -> None:

        if memory_policy is None:
            raise ValueError(
                "memory_policy cannot be None."
            )

        self._policy = memory_policy

    def approve(
        self,
        candidate: MemoryCandidate,
    ) -> dict[str, Any]:
        """
        Approve a memory candidate.

        Current Beta behaviour:

        - validates the candidate
        - returns a governance decision
        - does NOT persist memory

        The response intentionally contains both:

        Current Beta fields:
            category
            observation

        Future domain-aligned fields:
            memory_type
            content
        """

        if candidate is None:
            raise ValueError(
                "candidate cannot be None."
            )

        # -------------------------------------------------
        # Normalize candidate fields
        # -------------------------------------------------

        memory_type = (
            candidate.memory_type
            or candidate.category
        )

        content = (
            candidate.content
            or candidate.observation
        )

        learner_id = (
            candidate.learner_id
            or candidate.learner_hash
        )

        confidence_score = (
            candidate.confidence_score
            if candidate.confidence_score is not None
            else candidate.confidence
        )

        source_event_id = (
            candidate.source_event_id
        )

        if (
            source_event_id is None
            and candidate.evidence_event_ids
        ):
            source_event_id = (
                candidate.evidence_event_ids[0]
            )

        # -------------------------------------------------
        # Policy validation
        # -------------------------------------------------

        self._policy.validate(
            memory_type=memory_type,
            content=content,
        )

        # -------------------------------------------------
        # Governance decision
        # -------------------------------------------------

        decision = {
            "status": "approved",

            "candidate_id": (
                candidate.candidate_id
            ),

            # Current Beta contract
            "category": (
                candidate.category
                or memory_type
            ),

            "observation": (
                candidate.observation
                or content
            ),

            # Future domain-aligned contract
            "memory_type": memory_type,

            "content": content,

            "learner_id": learner_id,

            "confidence_score": confidence_score,

            "source_event_id": source_event_id,

            "evidence_event_ids": (
                candidate.evidence_event_ids
                or (
                    [source_event_id]
                    if source_event_id
                    else []
                )
            ),

            "metadata": {
                **(
                    candidate.metadata
                    or {}
                ),
                "approval_status": "approved",
            },
        }

        logger.info(
            "Memory candidate approved id=%s",
            candidate.candidate_id,
        )

        return decision

    def reject(
        self,
        candidate: MemoryCandidate,
        *,
        reason: str,
    ) -> dict[str, Any]:
        """
        Reject a memory candidate.

        Rejection does not:

        - persist memory
        - modify learner data
        - create a MemoryItem

        A governance decision is returned instead.
        """

        if candidate is None:
            raise ValueError(
                "candidate cannot be None."
            )

        if not reason or not reason.strip():
            raise ValueError(
                "Rejection reason required."
            )

        memory_type = (
            candidate.memory_type
            or candidate.category
        )

        learner_id = (
            candidate.learner_id
            or candidate.learner_hash
        )

        source_event_id = (
            candidate.source_event_id
        )

        if (
            source_event_id is None
            and candidate.evidence_event_ids
        ):
            source_event_id = (
                candidate.evidence_event_ids[0]
            )

        decision = {
            "status": "rejected",

            "candidate_id": (
                candidate.candidate_id
            ),

            "learner_id": learner_id,

            # Current Beta contract
            "category": (
                candidate.category
                or memory_type
            ),

            # Future domain-aligned contract
            "memory_type": memory_type,

            "reason": reason.strip(),

            "source_event_id": source_event_id,

            "evidence_event_ids": (
                candidate.evidence_event_ids
                or (
                    [source_event_id]
                    if source_event_id
                    else []
                )
            ),

            "metadata": {
                **(
                    candidate.metadata
                    or {}
                ),
                "approval_status": "rejected",
            },
        }

        logger.info(
            "Memory candidate rejected id=%s",
            candidate.candidate_id,
        )

        return decision
