"""
LLM Provider Configuration.

Defines approved provider configuration
for the EasyChatBot hackathon runtime.

Configuration is environment-driven so the
application does not hard-code deployment-specific
provider settings.

Local development supports a .env file located
at the EasyChatBot project root.

For the Hackathon runtime, the project .env file
is authoritative. This prevents stale local shell
environment variables from silently overriding the
configured development credentials.

Future production versions should use an external
secret/configuration management system.
"""

from __future__ import annotations

import os
from pathlib import Path

from dotenv import load_dotenv


# Resolve the EasyChatBot project root.
#
# config.py:
# backend/
#   providers/
#     llm/
#       config.py
#
# parents[3] -> EasyChatBot-Hackathon/
PROJECT_ROOT = Path(
    __file__
).resolve().parents[3]

ENV_FILE = PROJECT_ROOT / ".env"


# Load the Hackathon project's local configuration.
#
# override=True is intentional for the local Hackathon
# runtime because .env is the authoritative project
# configuration source.
#
# Production deployments must not rely on this mechanism.
load_dotenv(
    dotenv_path=ENV_FILE,
    override=True,
)


DEFAULT_PROVIDER = "openai"

DEFAULT_MODEL = "gpt-5-mini"

DEFAULT_GEMINI_MODEL = "gemini-3.6-flash"


def get_llm_provider_name() -> str:
    """
    Return the configured LLM provider name.

    Environment variable:

        EASYCHATBOT_LLM_PROVIDER

    Default:

        openai

    Returns
    -------
    str
        Normalised provider identifier.

    Raises
    ------
    ValueError
        If the configured provider name is empty.
    """

    provider = os.getenv(
        "EASYCHATBOT_LLM_PROVIDER",
        DEFAULT_PROVIDER,
    )

    provider = provider.strip().lower()

    if not provider:
        raise ValueError(
            "LLM provider name cannot be empty."
        )

    return provider


def get_llm_model() -> str:
    """
    Return the configured model for the
    default/current LLM path.

    Environment variable:

        EASYCHATBOT_LLM_MODEL

    Default:

        gpt-5-mini

    Returns
    -------
    str
        Configured model identifier.

    Raises
    ------
    ValueError
        If the configured model is empty.
    """

    model = os.getenv(
        "EASYCHATBOT_LLM_MODEL",
        DEFAULT_MODEL,
    )

    model = model.strip()

    if not model:
        raise ValueError(
            "LLM model cannot be empty."
        )

    return model


def get_gemini_model() -> str:

"""
Gemini LLM Provider.

Concrete Gemini implementation of the provider-independent
LLMProvider contract used by the EasyChatBot hackathon runtime.

Architecture:

    ConversationAgent
            |
            v
    ConversationRequest
            |
            v
       LLMProvider
            |
            v
      GeminiProvider
            |
            v
       Gemini API
            |
            v
      ProviderResult

Responsibilities:
- execute ConversationRequest through Gemini
- use the configured Gemini model
- translate Gemini responses into ProviderResult
- isolate the Gemini SDK from the application/runtime layers

Does not:
- build application workflows
- perform research
- manage memory
- select providers
- manage conversations
- modify application state

The provider uses Google's current `google-genai` SDK.
The SDK client is injectable so unit tests do not need a real
Gemini API call.
"""

from __future__ import annotations

import logging
import os
from typing import Any

from backend.conversation.conversation_request import (
    ConversationRequest,
)
from backend.providers.llm.llm_provider import (
    LLMProvider,
)
from backend.providers.provider_result import (
    ProviderResult,
)


logger = logging.getLogger(__name__)


class GeminiProvider(LLMProvider):
    """
    Gemini implementation of the LLMProvider contract.

    The Google Gemini SDK is deliberately isolated inside this
    infrastructure adapter.

    Args:
        model:
            Gemini model identifier.

        client:
            Optional injected `google.genai.Client`.

            Dependency injection allows unit tests to provide a
            fake client without making a real API call.
    """

    def __init__(
        self,
        model: str = "gemini-3.6-flash",
        client: Any | None = None,
    ) -> None:
        super().__init__(
            provider_id="gemini",
        )

        if not isinstance(model, str) or not model.strip():
            raise ValueError(
                "Gemini model cannot be empty."
            )

        self.model = model.strip()
        self._client = client

    def _get_client(self) -> Any:
        """
        Lazily create the Google GenAI client.

        The API key is read from GEMINI_API_KEY and is never
        hard-coded into the provider.
        """

        if self._client is not None:
            return self._client

        api_key = os.getenv("GEMINI_API_KEY")

        if not api_key:
            raise RuntimeError(
                "GEMINI_API_KEY environment variable is required."
            )

        try:
            from google import genai
        except ImportError as exc:
            raise RuntimeError(
                "The google-genai package is required for "
                "GeminiProvider. Install it with: "
                "pip install google-genai"
            ) from exc

        self._client = genai.Client(
            api_key=api_key,
        )

        return self._client

    @staticmethod
    def _build_contents(
        request: ConversationRequest,
    ) -> str:
        """
        Convert the provider-independent request into the
        user-facing content sent to Gemini.

        System behaviour remains separate and is supplied through
        Gemini's system_instruction configuration.

        Live intelligence, when present, is deliberately carried
        through the existing ConversationRequest context rather
        than retrieved by Gemini itself.
        """

        sections: list[str] = []

        if request.conversation_context:
            sections.extend(
                (
                    "CONTEXT:",
                    request.conversation_context.strip(),
                    "",
                )
            )

        sections.extend(
            (
                "USER REQUEST:",
                request.user_prompt.strip(),
            )
        )

        return "\n".join(sections)

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate a response using Gemini.

        Returns a provider-independent ProviderResult and never
        exposes Gemini SDK response objects to the application.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        try:
            client = self._get_client()

            from google.genai import types

            contents = self._build_contents(
                request,
            )

            config = types.GenerateContentConfig(
                system_instruction=(
                    request.system_instruction
                ),
            )

            response = await client.aio.models.generate_content(
                model=self.model,
                contents=contents,
                config=config,
            )

        except Exception as exc:
            logger.exception(
                "Gemini provider execution failed."
            )

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error=str(exc),
            )

        output_text = getattr(
            response,
            "text",
            None,
        )

        if not output_text or not output_text.strip():
            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error="Gemini returned an empty response.",
            )

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output=output_text.strip(),
        )

"""
LLM Provider Contract.

Defines the provider-independent contract for language
model providers used by the EasyChatBot hackathon runtime.

The contract sits between the Conversation Agent and
concrete LLM provider implementations.

Architecture:

ConversationAgent
        |
        v
ConversationRequest
        |
        v
LLMProvider
        |
        v
Concrete LLM Provider
        |
        v
ProviderResult

Responsibilities:

- define the LLM generation boundary
- keep ConversationAgent independent from provider SDKs
- provide a stable contract for concrete LLM providers

Does not:

- call an external API
- contain provider SDK logic
- select models
- implement routing
- manage memory
- manage conversations
"""

from __future__ import annotations

from abc import ABC
from abc import abstractmethod

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.provider_result import (
    ProviderResult,
)


class LLMProvider(ABC):
    """
    Abstract provider contract for language model generation.

    Concrete implementations may use OpenAI, Ollama,
    Gemini, or another approved LLM provider.

    The application and agent layers depend only on
    this contract.
    """

    provider_id: str

    def __init__(
        self,
        provider_id: str,
    ) -> None:
        """
        Initialise the LLM provider.

        Args:
            provider_id:
                Stable identifier for the concrete provider.
        """

        if not provider_id.strip():
            raise ValueError(
                "LLM provider_id cannot be empty."
            )

        self.provider_id = provider_id

    @abstractmethod
    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate an LLM response.

        Args:
            request:
                Provider-independent conversation request.

        Returns:
            ProviderResult containing the generated
            response or an error.

        Implementations must not expose provider SDK
        objects through the result.
        """

        raise NotImplementedError

"""
LLM Provider Factory.

Creates approved LLM provider implementations.

The factory is the composition boundary between
application/runtime code and infrastructure providers.

Architecture:

Application/API
|
v
LLMProviderFactory
|
v
LLMProvider
|
+--> OpenAIProvider
+--> Future OllamaProvider
+--> GeminiProvider

Responsibilities:

- create configured LLM providers
- isolate provider construction
- provide provider independence
- obtain approved provider configuration

Does not:

- execute LLM requests
- build prompts
- manage conversations
- manage memory
- apply routing decisions
- call provider SDKs directly
"""

from __future__ import annotations

from backend.providers.llm.config import (
    get_gemini_model,
    get_llm_model,
    get_llm_provider_name,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.providers.llm.openai_provider import (
    OpenAIProvider,
)

from backend.providers.llm.gemini_provider import (
    GeminiProvider,
)


class LLMProviderFactory:
    """
    Factory responsible for creating LLM providers.

    Provider selection and configuration are centralised
    here so application/runtime code does not depend
    directly on concrete infrastructure implementations.
    """

    @staticmethod
    def create(
        provider: str | None = None,
    ) -> LLMProvider:
        """
        Create a configured LLM provider.

        Parameters
        ----------
        provider:
            Optional provider identifier.

            When omitted, the configured provider from
            LLM provider configuration is used.

        Returns
        -------
        LLMProvider
            Configured provider implementation.

        Raises
        ------
        ValueError
            If provider is invalid or unsupported.
        """

        if provider is None:
            provider_name = (
                get_llm_provider_name()
            )
        else:
            if not isinstance(
                provider,
                str,
            ):
                raise ValueError(
                    "Provider name must be a string."
                )

            provider_name = (
                provider.strip().lower()
            )

            if not provider_name:
                raise ValueError(
                    "Provider name cannot be empty."
                )

        if provider_name == "openai":
            return OpenAIProvider(
                model=get_llm_model(),
            )

        if provider_name == "gemini":
            return GeminiProvider(
                model=get_gemini_model(),
            )

        raise ValueError(
            (
                "Unsupported LLM provider: "
                f"{provider_name}"
            )
        )

"""
OpenAI LLM Provider.

Concrete LLM provider implementation for the EasyChatBot
hackathon runtime.

Architecture:

ConversationAgent
|

LLMService
|
v
LLMProvider
|

OpenAIProvider
|

OpenAI Responses API
|
v
ProviderResult

- execute ConversationRequest through OpenAI
- use the configured OpenAI model
- translate the OpenAI response into ProviderResult
- isolate the OpenAI SDK from the application layer

Does not:

- build prompts
- manage characters
- manage memory
- perform research
- perform routing
- modify application state
"""

from __future__ import annotations

import logging
import os

from openai import AsyncOpenAI

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.providers.provider_result import (
    ProviderResult,
)


logger = logging.getLogger(__name__)


class OpenAIProvider(LLMProvider):
    """
    OpenAI implementation of the LLMProvider contract.

    The OpenAI SDK is deliberately isolated inside this
    infrastructure adapter.
    """

    def __init__(
        self,
        model: str = "gpt-5-mini",
        client: AsyncOpenAI | None = None,
    ) -> None:
        """
        Initialise the OpenAI provider.

        Parameters
        ----------
        model:
            OpenAI model identifier.

        client:
            Optional injected AsyncOpenAI client.

            Dependency injection allows unit tests to
            provide a fake client without making a real
            API call.
        """

        super().__init__(
            provider_id="openai",
        )

        if not isinstance(
            model,
            str,
        ):
            raise ValueError(
                "OpenAI model must be a string."
            )

        model = model.strip()

        if not model:
            raise ValueError(
                "OpenAI model cannot be empty."
            )

        self.model = model

        if client is not None:
            self._client = client
            return

        api_key = os.getenv(
            "OPENAI_API_KEY"
        )

        if not api_key:
            raise ValueError(
                "OPENAI_API_KEY is not configured."
            )

        self._client = AsyncOpenAI(
            api_key=api_key,
        )

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate a response using OpenAI.

        The provider translates the provider-independent
        ConversationRequest into the OpenAI Responses API
        request.

        Parameters
        ----------
        request:
            Provider-independent conversation request.

        Returns
        -------
        ProviderResult
            Provider-independent execution result.

        Raises
        ------
        ValueError
            If request is None.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        messages = [
            {
                "role": "system",
                "content": request.system_instruction,
            },
        ]

        if request.conversation_context:
            messages.append(
                {
                    "role": "user",
                    "content": (
                        "Additional context:\n"
                        f"{request.conversation_context}"
                    ),
                }
            )

        messages.append(
            {
                "role": "user",
                "content": request.user_prompt,
            }
        )

        try:
            response = (
                await self._client.responses.create(
                    model=self.model,
                    input=messages,
                )
            )

        except Exception as exc:
            logger.exception(
                "OpenAI provider execution failed."
            )

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=str(exc),
            )

        output_text = getattr(
            response,
            "output_text",
            None,
        )

        if not output_text:
            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=(
                    "OpenAI returned an empty response."
                ),
            )

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output=output_text,
        )
