"""
Base AI Agent Contract.

All EasyChatBot AI Agents inherit this contract.

The BaseAgent defines:
- agent identity
- execution interface
- runtime interaction

Agents do not:
- call providers directly
- manage memory
- modify external systems

The AI Runtime controls execution.
"""

from abc import ABC, abstractmethod

from backend.runtime.context.execution_context import ExecutionContext
from backend.agents.agent_manifest import AgentManifest
from backend.agents.agent_result import AgentResult


class BaseAgent(ABC):
    """
    Abstract base class for EasyChatBot agents.
    """

    def __init__(
        self,
        manifest: AgentManifest,
    ):
        """
        Initialise an agent.

        Args:
            manifest:
                Agent identity definition.
        """

        self.manifest = manifest

    @abstractmethod
    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute the agent.

        Args:
            context:
                Runtime execution context.

        Returns:
            AgentResult.
        """

        pass

"""
Conversation Agent.

Provides character-aware conversational responses
for the EasyChatBot hackathon runtime.

The agent remains provider-independent.

Architecture:

ExecutionContext
|
v
ConversationAgent
|
+--> CharacterProfile
|
+--> Optional IntelligenceContext
|
v
ConversationRequest
|
v
LLMService
|
v

|
v

|
v

|
v
AgentResult

Responsibilities:

- apply character identity
- construct ConversationRequest
- include optional live intelligence
- execute an injected LLMService
- preserve deterministic demo behaviour when no
  LLM service is configured
- return a standard AgentResult

Does not:

- depend directly on OpenAI
- depend directly on Ollama
- call Parallel directly
- manage memory
- perform research
- manage voice
- execute external workflows
- select providers
- execute provider SDKs directly

Architecture boundary:

ConversationAgent
-> LLMService
-> LLMProvider
-> Concrete Provider
"""

from __future__ import annotations

from backend.agents.agent_manifest import (
    AgentManifest,
)

from backend.agents.agent_result import (
    AgentResult,
)

from backend.agents.base_agent import (
    BaseAgent,
)

from backend.characters.character_profile import (
    CharacterProfile,
)

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)

from backend.services.llm_service import (
    LLMService,
)


class ConversationAgent(BaseAgent):
    """
    Character-aware conversation agent.

    The agent depends primarily on the application-level
    LLMService boundary.

    The LLMService is responsible for delegating
    execution to the configured LLMProvider.

    A legacy ``llm_provider`` argument remains accepted
    as a compatibility seam for existing tests and
    callers. When supplied, it is wrapped by LLMService.

    New composition code should inject ``llm_service``.
    """

    def __init__(
        self,
        character: CharacterProfile,
        llm_service: LLMService | None = None,
        llm_provider: LLMProvider | None = None,
    ) -> None:
        """
        Initialise the Conversation Agent.

        Parameters
        ----------
        character:
            Approved EasyChatBot character profile.

        llm_service:
            Application-level LLM execution service.

        llm_provider:
            Compatibility injection seam.

            When supplied and ``llm_service`` is not supplied,
            the provider is wrapped inside LLMService.

        Raises
        ------
        ValueError
            If both ``llm_service`` and ``llm_provider``
            are supplied simultaneously.
        """

        super().__init__(
            AgentManifest(
                agent_id="conversation_agent",
                name="Conversation Agent",
                description=(
                    "Provides character-aware "
                    "conversation responses."
                ),
            )
        )

        if (
            llm_service is not None
            and llm_provider is not None
        ):
            raise ValueError(
                "Provide either llm_service or "
                "llm_provider, not both."
            )

        self.character = character

        if llm_service is not None:
            self.llm_service = llm_service

        elif llm_provider is not None:
            self.llm_service = LLMService(
                llm_provider
            )

        else:
            self.llm_service = None

    def build_request(
        self,
        context: ExecutionContext,
    ) -> ConversationRequest:
        """
        Build a provider-independent conversation request.

        Runtime state and character identity are translated
        into the ConversationRequest contract.

        No provider-specific implementation is introduced.
        """

        system_instruction = (
            f"You are {self.character.name}, "
            f"a character in EasyChatBot. "
            f"Your personality is: "
            f"{self.character.personality} "
            f"Your learning style is: "
            f"{self.character.learning_style} "
            f"Communicate in a way that is consistent "
            f"with this character."
        )

        conversation_context = None

        if context.has_intelligence_context:
            conversation_context = (
                context.intelligence_context.to_context_text()
            )

        return ConversationRequest(
            system_instruction=system_instruction,
            user_prompt=context.request,
            conversation_context=conversation_context,
            character_id=self.character.character_id,
            character_name=self.character.name,
        )

    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute the conversation agent.

        Flow:

        ExecutionContext
            |
            v
        ConversationRequest
            |
            v
        LLMService
            |
            v
        LLMProvider
            |
            v
        ProviderResult
            |
            v
        AgentResult

        When no LLMService is configured, the existing
        deterministic hackathon behaviour is preserved.

        When an LLMService is configured, execution
        passes through the application service boundary.
        """

        conversation_request = self.build_request(
            context
        )

        provider_result = None

        if self.llm_service is not None:
            provider_result = (
                await self.llm_service.generate(
                    conversation_request
                )
            )

            if not provider_result.success:
                return AgentResult(
                    success=False,
                    agent_id=self.manifest.agent_id,
                    output={
                        "character": self.character.name,

                        "character_profile": {
                            "character_id": (
                                self.character.character_id
                            ),
                            "name": self.character.name,
                            "description": (
                                self.character.description
                            ),
                            "personality": (
                                self.character.personality
                            ),
                            "learning_style": (
                                self.character.learning_style
                            ),
                        },

                        "personality": (
                            self.character.personality
                        ),

                        "learning_style": (
                            self.character.learning_style
                        ),

                        "message": (
                            "The conversation provider "
                            "could not generate a response."
                        ),

                        "intelligence_context": (
                            conversation_request
                            .conversation_context
                        ),

                        "conversation_request": (
                            conversation_request
                        ),

                        "provider_id": (
                            provider_result.provider_id
                        ),

                        "provider_error": (
                            provider_result.error
                        ),
                    },
                )

        if provider_result is not None:
            message = provider_result.output

        else:
            message = (
                f"{self.character.name}: "
                f"Let's explore this together. "
                f"You asked: {context.request}"
            )

        output = {
            "character": self.character.name,

            "character_profile": {
                "character_id": (
                    self.character.character_id
                ),
                "name": self.character.name,
                "description": (
                    self.character.description
                ),
                "personality": (
                    self.character.personality
                ),
                "learning_style": (
                    self.character.learning_style
                ),
            },

            "personality": (
                self.character.personality
            ),

            "learning_style": (
                self.character.learning_style
            ),

            "message": message,

            "intelligence_context": (
                conversation_request
                .conversation_context
            ),

            "conversation_request": (
                conversation_request
            ),
        }

        if provider_result is not None:
            output["provider_id"] = (
                provider_result.provider_id
            )

        return AgentResult(
            success=True,
            agent_id=self.manifest.agent_id,
            output=output,
        )

"""
Research Agent.

Retrieves external information through approved
knowledge providers and converts the result into
provider-independent IntelligenceContext data.

Architecture:

Research Agent
        |
        v
ParallelSearchProvider
        |
        v
ParallelClientWrapper
        |
        v
IntelligenceContext
"""

from __future__ import annotations

from backend.agents.agent_manifest import AgentManifest
from backend.agents.agent_result import AgentResult
from backend.agents.base_agent import BaseAgent

from backend.intelligence.intelligence_context import (
    IntelligenceContext,
    IntelligenceSource,
)

from backend.providers.parallel.parallel_provider import (
    ParallelSearchProvider,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class ResearchAgent(BaseAgent):
    """
    Agent responsible for world information retrieval.
    """

    def __init__(
        self,
        provider=None,
    ):
        super().__init__(
            AgentManifest(
                agent_id="research_agent",
                name="Research Agent",
                description=(
                    "Retrieves live world information "
                    "for AI learning experiences."
                ),
            )
        )

        self.provider = (
            provider
            or ParallelSearchProvider()
        )

    async def execute(
        self,
        context: ExecutionContext,
    ) -> AgentResult:
        """
        Execute research and produce a normalized
        IntelligenceContext.
        """

        result = await self.provider.execute(
            context
        )

        if not result.success:
            return AgentResult(
                success=False,
                agent_id=self.manifest.agent_id,
                output=None,
                error=result.error,
            )

        intelligence = (
            self._build_intelligence_context(
                query=context.request,
                provider_output=result.output,
            )
        )

        return AgentResult(
            success=True,
            agent_id=self.manifest.agent_id,
            output=intelligence,
        )

    @staticmethod
    def _build_intelligence_context(
        query: str,
        provider_output,
    ) -> IntelligenceContext:
        """
        Convert provider output into the
        provider-independent IntelligenceContext.
        """

        if not isinstance(
            provider_output,
            dict,
        ):
            raise TypeError(
                "Research provider output must be a dictionary."
            )

        raw_results = provider_output.get(
            "results",
            {},
        )

        if isinstance(
            raw_results,
            dict,
        ):
            raw_results = raw_results.get(
                "results",
                [],
            )

        if not isinstance(
            raw_results,
            list,
        ):
            raw_results = []

        sources: list[IntelligenceSource] = []

        for raw_result in raw_results:

            if not isinstance(
                raw_result,
                dict,
            ):
                continue

            title = str(
                raw_result.get(
                    "title",
                    "",
                )
            ).strip()

            url = str(
                raw_result.get(
                    "url",
                    "",
                )
            ).strip()

            if not title or not url:
                continue

            content = ResearchAgent._extract_content(
                raw_result
            )

            sources.append(
                IntelligenceSource(
                    title=title,
                    url=url,
                    source="parallel",
                    content=content,
                )
            )

        return IntelligenceContext(
            query=query,
            sources=tuple(sources),
        )

    @staticmethod
    def _extract_content(
        raw_result: dict,
    ) -> str:
        """
        Extract available content from a normalized
        Parallel result.
        """

        excerpts = raw_result.get(
            "excerpts",
        )

        if excerpts is None:
            excerpts = raw_result.get(
                "snippet",
                raw_result.get(
                    "content",
                    "",
                ),
            )

        if isinstance(
            excerpts,
            list,
        ):
            return "\n\n".join(
                str(item).strip()
                for item in excerpts
                if str(item).strip()
            )

        return str(
            excerpts or ""
        ).strip()

router = APIRouter()

character_registry = CharacterRegistry()


def create_llm_service() -> LLMService:
    """
    Create the application LLM service.

    Provider construction is isolated through the
    LLMProviderFactory.

    The LLMService remains provider-independent.
    """

    llm_provider = (
        LLMProviderFactory.create()
    )

    return LLMService(
        llm_provider
    )


def create_companion_service(
    character,
) -> CompanionService:
    """
    Create the application-level CompanionService.

    This function is the API composition boundary.

    Infrastructure dependencies are created here and
    injected into application services and agents.

    The application service remains unaware of:

    - OpenAI
    - Ollama
    - provider SDKs
    - API credentials
    """

    llm_service = create_llm_service()

    conversation_agent = ConversationAgent(
        character=character,
        llm_service=llm_service,
    )

    return CompanionService(
        character=character,
        conversation_agent=conversation_agent,
    )


@router.get("/characters")
def get_characters():
    """
    Return available AI companions.
    """

    characters = (
        character_registry.list_characters()
    )

    return {
        "characters": [
            {
                "id": character.character_id,
                "name": character.name,
                "description": character.description,
                "personality": character.personality,
                "learning_style": character.learning_style,
            }
            for character in characters
        ]
    }


@router.post("/chat")
async def chat(
    payload: dict,
):
    """
    Execute an EasyChatBot companion interaction.

    Expected payload:

    {
        "user_id": "demo-user",
        "character": "yeti",
        "message": "Explain AI",
        "research": false
    }
    """

    message = payload.get(
        "message"
    )

    if not message:
        raise HTTPException(
            status_code=400,
            detail="message is required",
        )

    character_id = payload.get(
        "character",
        "yeti",
    )

    try:
        character = (
            character_registry.get(
                character_id
            )
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail=str(exc),
        ) from exc

    companion_service = (
        create_companion_service(
            character
        )
    )

    context = ExecutionContext(
        user_id=payload.get(
            "user_id",
            "demo-user",
        ),
        request=message,
    )

    result = await companion_service.interact(
        context,
        use_research=payload.get(
            "research",
            False,
        ),
    )

    return result

"""
Provider-independent conversation request.

Represents the information required to generate a
character-aware conversational response.

This contract belongs to the application/runtime boundary.

It deliberately does not know about:

- Gemini
- OpenAI
- Ollama
- Parallel
- voice
- HTTP
- persistence
- provider SDKs

Architecture:

CharacterProfile
        +
ExecutionContext
        +
Optional IntelligenceContext
        |
        v
ConversationRequest
        |
        v
Future / existing EasyChatBot LLM service

The production EasyChatBot conversation and voice
architecture remains outside this hackathon runtime.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class ConversationRequest:
    """
    Immutable provider-independent conversation request.

    Attributes
    ----------
    system_instruction:
        Stable instruction defining how the character
        should behave.

    user_prompt:
        User-facing request to be answered.

    conversation_context:
        Optional contextual information, such as live
        intelligence retrieved by the Research Agent.

    character_id:
        Identifier of the character producing the response.

    character_name:
        Display name of the character.
    """

    system_instruction: str

    user_prompt: str

    conversation_context: str | None = None

    character_id: str = ""

    character_name: str = ""

    def __post_init__(self) -> None:
        """
        Validate request invariants.
        """

        if not isinstance(
            self.system_instruction,
            str,
        ):
            raise ValueError(
                "system_instruction must be a string."
            )

        if not isinstance(
            self.user_prompt,
            str,
        ):
            raise ValueError(
                "user_prompt must be a string."
            )

        system_instruction = (
            self.system_instruction.strip()
        )

        user_prompt = (
            self.user_prompt.strip()
        )

        if not system_instruction:
            raise ValueError(
                "system_instruction cannot be empty."
            )

        if not user_prompt:
            raise ValueError(
                "user_prompt cannot be empty."
            )

        object.__setattr__(
            self,
            "system_instruction",
            system_instruction,
        )

        object.__setattr__(
            self,
            "user_prompt",
            user_prompt,
        )

        if self.conversation_context is not None:
            context = (
                self.conversation_context.strip()
            )

            if not context:
                raise ValueError(
                    "conversation_context cannot be blank."
                )

            object.__setattr__(
                self,
                "conversation_context",
                context,
            )

        if not isinstance(
            self.character_id,
            str,
        ):
            raise ValueError(
                "character_id must be a string."
            )

        if not isinstance(
            self.character_name,
            str,
        ):
            raise ValueError(
                "character_name must be a string."
            )

        object.__setattr__(
            self,
            "character_id",
            self.character_id.strip(),
        )

        object.__setattr__(
            self,
            "character_name",
            self.character_name.strip(),
        )

    @property
    def has_conversation_context(self) -> bool:
        """
        Return whether contextual information exists.
        """

        return bool(
            self.conversation_context
        )

    @property
    def context_length(self) -> int:
        """
        Return the length of the conversation context.

        Useful for diagnostics and future token estimation.
        """

        if not self.conversation_context:
            return 0

        return len(
            self.conversation_context
        )

    def summary(self) -> dict[str, str]:
        """
        Return a safe diagnostic summary.

        Prompt and context content are deliberately
        excluded from the summary.
        """

        return {
            "character_id": self.character_id,
            "character_name": self.character_name,
            "has_context": str(
                self.has_conversation_context
            ),
            "system_instruction_length": str(
                len(self.system_instruction)
            ),
            "user_prompt_length": str(
                len(self.user_prompt)
            ),
            "context_length": str(
                self.context_length
            ),
        }

"""
Intelligence Context.

Provider-independent representation of external
information prepared for AI execution.

Architecture:

Research Agent
        |
        v
IntelligenceContext
        |
        v
AI Runtime integration
        |
        v
LLMRequest.context

The context must not expose provider-specific
implementation details to the application layer.
"""

from __future__ import annotations

from dataclasses import dataclass, field


@dataclass(frozen=True, slots=True)
class IntelligenceSource:
    """
    Normalised external information source.
    """

    title: str
    url: str
    source: str
    content: str = ""

    def __post_init__(self) -> None:
        if not self.title.strip():
            raise ValueError(
                "Intelligence source title cannot be empty."
            )

        if not self.url.strip():
            raise ValueError(
                "Intelligence source URL cannot be empty."
            )

        if not self.source.strip():
            raise ValueError(

            success=True,

"""
EasyChatBot — Memory Context Builder.

H3-01 — Memory-Aware Live Intelligence.

The builder operates on the APPLICATION memory contract:

    MemoryService
        ↓
    MemoryRecord
        ↓
    MemoryContextBuilder
        ↓
    MemoryContext

It deliberately does NOT consume MemoryEntry directly.

MemoryEntry belongs to the persistence/runtime storage boundary.
MemoryRecord is the application-facing memory contract.

Responsibilities:
- validate already-retrieved memory
- enforce user isolation
- determine deterministic relevance
- bound memory count
- bound context size
- produce provider-independent context

Does not:
- retrieve memory
- persist memory
- call an LLM
- call a provider
- execute an agent
- mutate ExecutionContext
- perform semantic/vector search
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
import re

from backend.memory.memory_record import (
    MemoryRecord,
)


DEFAULT_MAX_MEMORY_ITEMS = 5
DEFAULT_MAX_CONTEXT_CHARACTERS = 4000

_WORD_PATTERN = re.compile(
    r"[\wÀ-ÿ'-]+",
    re.UNICODE,
)


# Application-level memory categories.
#
# These are deliberately strings because MemoryService exposes
# MemoryRecord.category rather than MemoryEntry.memory_type.
_MEMORY_CATEGORY_PRIORITY: dict[str, int] = {
    "preference": 30,
    "learning_goal": 30,
    "discovery_context": 20,
    "saved_item": 10,
    "conversation": 10,
}


@dataclass(frozen=True)
class MemoryContext:
    """
    Bounded, provider-independent learner context.
    """

    entries: tuple[MemoryRecord, ...]
    text: str

    @property
    def is_empty(self) -> bool:
        """Return True when no memory was selected."""
        return not self.entries

    @property
    def memory_count(self) -> int:
        """Return the number of selected memories."""
        return len(self.entries)


class MemoryContextBuilder:
    """
    Build bounded learner context from MemoryRecord objects.

    The builder is stateless and deterministic.
    """

    def __init__(
        self,
        *,
        max_memory_items: int = DEFAULT_MAX_MEMORY_ITEMS,
        max_context_characters: int = (
            DEFAULT_MAX_CONTEXT_CHARACTERS
        ),
    ) -> None:
        if max_memory_items < 1:
            raise ValueError(
                "max_memory_items must be greater than zero."
            )

        if max_context_characters < 1:
            raise ValueError(
                "max_context_characters must be greater than zero."
            )

        self._max_memory_items = max_memory_items
        self._max_context_characters = (
            max_context_characters
        )

    def build(
        self,
        *,
        user_id: str,
        request: str,
        memories: list[MemoryRecord]
        | tuple[MemoryRecord, ...],
    ) -> MemoryContext:
        """
        Build bounded learner context.

        Memory is supplied by the caller.

        The builder does not retrieve or persist memory.
        """

        if not user_id.strip():
            raise ValueError(
                "user_id cannot be empty."
            )

        if not request.strip():
            raise ValueError(
                "request cannot be empty."
            )

        if not memories:
            return MemoryContext(
                entries=(),
                text="",
            )

        request_tokens = self._tokens(
            request
        )

        candidates: list[
            tuple[
                int,
                datetime,
                str,
                MemoryRecord,
            ]
        ] = []

        for memory in memories:

            if not isinstance(
                memory,
                MemoryRecord,
            ):
                continue

            # Defence in depth.
            if memory.user_id != user_id:
                continue

            if not memory.category.strip():
                continue

            if not memory.content.strip():
                continue

            score = self._score(
                memory=memory,
                request_tokens=request_tokens,
            )

            # A memory category alone does NOT make the memory
            # relevant to the current request.
            if score <= 0:
                continue

            candidates.append(
                (
                    score,
                    memory.created_at,
                    memory.content,
                    memory,
                )
            )

        # Deterministic ordering:
        #
        # 1. relevance descending
        # 2. newest memory first
        # 3. content as stable tie-breaker
        candidates.sort(
            key=lambda item: (
                -item[0],
                -item[1].timestamp(),
                item[2].casefold(),
            )
        )

        selected: list[MemoryRecord] = []
        rendered_items: list[str] = []

        current_length = len(
            "Learner context:\n"
        )

        for (
            _score,
            _created_at,
            _content,
            memory,
        ) in candidates:

            if (
                len(selected)
                >= self._max_memory_items
            ):
                break

            line = self._render_memory(
                memory
            )

            separator_length = (
                1
                if rendered_items
                else 0
            )

            projected_length = (
                current_length
                + separator_length
                + len(line)
            )

            if (
                projected_length
                > self._max_context_characters
            ):
                # Do not partially truncate an individual
                # memory because that could alter meaning.
                continue

            selected.append(memory)
            rendered_items.append(line)

            current_length = projected_length

        if not selected:
            return MemoryContext(
                entries=(),
                text="",
            )

        text = (
            "Learner context:\n"
            + "\n".join(
                f"- {item}"
                for item in rendered_items
            )
        )

        if (
            len(text)
            > self._max_context_characters
        ):
            raise AssertionError(
                "MemoryContextBuilder produced "
                "an oversized context."
            )

        return MemoryContext(
            entries=tuple(selected),
            text=text,
        )

    @staticmethod
    def _tokens(
        value: str,
    ) -> set[str]:
        """
        Return normalised tokens for deterministic matching.
        """

        return {
            token.casefold()
            for token in _WORD_PATTERN.findall(
                value
            )
            if len(token) > 1
        }

    @classmethod
    def _score(
        cls,
        *,
        memory: MemoryRecord,
        request_tokens: set[str],
    ) -> int:
        """
        Calculate deterministic relevance.

        Relevance requires actual token overlap.

        This prevents an unrelated preference such as
        "Prefers hiking" from being injected into a request
        about "European AI regulation" simply because it is
        a preference.
        """

        content_tokens = cls._tokens(
            memory.content
        )

        category_tokens = cls._tokens(
            memory.category
        )

        overlap = request_tokens & (
            content_tokens
            | category_tokens
        )

        if not overlap:
            return 0

        category = (
            memory.category.strip().casefold()
        )

        category_priority = (
            _MEMORY_CATEGORY_PRIORITY.get(
                category,
                10,
            )
        )

        return (
            category_priority
            + len(overlap)
        )

    @staticmethod
    def _render_memory(
        memory: MemoryRecord,
    ) -> str:
        """
        Render memory explicitly as learner context.

        Memory content is treated as data, not instructions.
        """

        category = (
            memory.category.strip()
        )

        content = " ".join(
            memory.content.split()
        )

        return (

@dataclass
class FakeMemory:
    """
    Deterministic memory record for tests.
    """

    user_id: str
    category: str
    content: str


class FakeMemoryService:
    """
    Deterministic MemoryService test double.
    """

    def __init__(self) -> None:
        self.memories = [
            FakeMemory(
                user_id="demo-user",
                category="conversation",
                content="Previous conversation",
            ),
        ]

        self.recall_calls = []
        self.remember_calls = []

    def recall(
        self,
        user_id: str,
    ):
        self.recall_calls.append(
            user_id
        )

        return [
            memory
            for memory in self.memories
            if memory.user_id == user_id
        ]

    def remember(
        self,
        user_id: str,
        category: str,
        content: str,
    ) -> None:
        self.remember_calls.append(
            {
                "user_id": user_id,
                "category": category,
                "content": content,
            }
        )

@pytest.mark.asyncio
async def test_companion_service_executes_research_when_requested():
    """
    When research is requested, CompanionService should:

    1. execute ResearchAgent
    2. attach its IntelligenceContext to a new
       ExecutionContext
    3. execute ConversationAgent with the enriched context
    4. execute QualityAgent with the enriched context
    """

    intelligence_context = object()

    research_agent = FakeResearchAgent(
        AgentResult(
            success=True,
            agent_id="research_agent",
            output=intelligence_context,
        )
    )

    conversation_agent = FakeConversationAgent(
        AgentResult(
            success=True,
            agent_id="conversation_agent",
            output={
                "message": "Research-aware response",
            },
        )
    )

    quality_agent = FakeQualityAgent(
        AgentResult(
            success=True,
            agent_id="quality_agent",
            output={
                "quality_score": 0.95,
                "passed": True,
            },
        )
    )

    memory_service = FakeMemoryService()

    service = CompanionService(
        character=create_character(),
        conversation_agent=conversation_agent,
        research_agent=research_agent,
        quality_agent=quality_agent,
        memory_service=memory_service,
    )

    context = create_context(
        "What are the latest AI developments?"
    )

    result = await service.interact(
        context,
        use_research=True,
    )

    assert result["character"] == "Yeti"

    assert result["conversation"] == {
        "message": "Research-aware response",
    }

    assert (
        result["research"]
        is intelligence_context
    )

    assert result["quality"] == {
        "quality_score": 0.95,
        "passed": True,
    }

    assert (
        len(
            research_agent.received_contexts
        )
        == 1
    )

    assert (
        research_agent.received_contexts[0]
        is context
    )

    assert (
        len(
            conversation_agent.received_contexts
        )
        == 1
    )

    enriched_context = (
        conversation_agent.received_contexts[0]
    )

    assert enriched_context is not context

    assert (
        enriched_context.request
        == context.request
    )

    assert (
        enriched_context.user_id
        == context.user_id
    )

    assert (
        enriched_context.intelligence_context
        is intelligence_context
    )

    assert (
        quality_agent.received_contexts[0]
        is enriched_context
    )

    Deterministic MemoryService test double.

            FakeMemory(
