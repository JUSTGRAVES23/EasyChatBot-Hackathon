"""Beta policy for provisioning access to the complete World catalogue."""

from __future__ import annotations

from typing import Any, Protocol
from uuid import UUID

from app.domain.learning.learner_world_access import BETA, LearnerWorldAccess


class BetaUserReadBoundary(Protocol):
    async def get_beta_user(
        self,
        user_id: str,
    ) -> dict[str, Any] | None:
        """Return the controlled Beta-user DTO or None."""
        ...


class CanonicalUserIdentityReader(Protocol):
    async def get_user_for_identity(
        self,
        user_id: UUID,
    ):
        """Return the canonical user record or None."""
        ...


class WorldCatalogueBoundary(Protocol):
    def list_all(self):
        """Return all configured catalogue Worlds."""
        ...


class WorldAccessBoundary(Protocol):
    async def grant_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
        access_source: str = BETA,
        allow_disabled_world: bool = False,
    ) -> LearnerWorldAccess:
        ...


class BetaWorldAccessError(ValueError):
    """Expected Beta World-access provisioning failure."""


class BetaUserNotFoundError(BetaWorldAccessError):
    """The requested Beta user does not exist."""


class BetaUserNotEligibleError(BetaWorldAccessError):
    """The Beta user is not verified and approved."""


class CanonicalUserNotFoundError(BetaWorldAccessError):
    """The Beta user cannot be resolved through canonical identity."""


class BetaWorldAccessService:
    """
    Provision durable Beta access for every World in the authoritative catalogue.

    Policy:
        verified + Beta-approved learner
            -> all catalogue Worlds

    ``enabled`` is intentionally NOT filtered here. The service uses the
    explicit WorldAccessService provisioning flag so disabled Worlds can
    receive durable portfolio access. Runtime/Study Mode entry must still
    enforce current World availability.
    """

    def __init__(
        self,
        *,
        beta_user_reader: BetaUserReadBoundary,
        user_identity_reader: CanonicalUserIdentityReader,
        world_catalogue: WorldCatalogueBoundary,
        world_access: WorldAccessBoundary,
    ) -> None:
        if beta_user_reader is None:
            raise ValueError("beta_user_reader is required.")
        if user_identity_reader is None:
            raise ValueError("user_identity_reader is required.")
        if world_catalogue is None:
            raise ValueError("world_catalogue is required.")
        if world_access is None:
            raise ValueError("world_access is required.")

        self._beta_user_reader = beta_user_reader
        self._user_identity_reader = user_identity_reader
        self._world_catalogue = world_catalogue
        self._world_access = world_access

    async def provision_all_worlds(
        self,
        *,
        beta_user_id: str,
    ) -> tuple[LearnerWorldAccess, ...]:
        if not isinstance(beta_user_id, str) or not beta_user_id.strip():
            raise ValueError("beta_user_id is required.")

        beta_user = await self._beta_user_reader.get_beta_user(
            beta_user_id.strip()
        )

        if beta_user is None:
            raise BetaUserNotFoundError(
                "Beta user does not exist."
            )

        if not bool(beta_user.get("email_verified")):
            raise BetaUserNotEligibleError(
                "Email must be verified before World access provisioning."
            )

        if not bool(beta_user.get("beta_approved")):
            raise BetaUserNotEligibleError(
                "User is not Beta approved."
            )

        canonical_user_id = self._canonical_uuid(
            beta_user.get("user_id")
        )

        canonical_user = await self._user_identity_reader.get_user_for_identity(
            canonical_user_id
        )

        if canonical_user is None:
            raise CanonicalUserNotFoundError(
                "Canonical user does not exist."
            )

        worlds = tuple(self._world_catalogue.list_all())

        # The authoritative catalogue is the only source of World scope.
        # There is deliberately no language/World allow-list here.
        granted: list[LearnerWorldAccess] = []

        for world in worlds:
            identifier = getattr(world, "identifier", None)

            if not isinstance(identifier, str) or not identifier.strip():
                raise BetaWorldAccessError(
                    "Authoritative World catalogue contains an invalid identifier."
                )

            access = await self._world_access.grant_access(
                user_id=canonical_user_id,
                world_identifier=identifier.strip(),
                access_source=BETA,
                allow_disabled_world=True,
            )
            granted.append(access)

        return tuple(granted)

    @staticmethod
    def _canonical_uuid(value: Any) -> UUID:
        if isinstance(value, UUID):
            return value

        if not isinstance(value, str) or not value.strip():
            raise BetaWorldAccessError(
                "Beta user must expose a canonical UUID user_id."
            )

        try:
            return UUID(value.strip())
        except ValueError as exc:
            raise BetaWorldAccessError(
                "Beta user user_id is not a valid canonical UUID."
            ) from exc

    "BetaUserNotFoundError",
    "BetaWorldAccessError",
    "BetaWorldAccessService",
    "CanonicalUserNotFoundError",

class FakeWorldAccess:
    def __init__(self) -> None:
        self.grants: list[tuple[UUID, str, str]] = []
        self.records: dict[str, LearnerWorldAccess] = {}

    async def grant_access(
        self,
        *,
        user_id: UUID,
        world_identifier: str,
        access_source: str = BETA,
        allow_disabled_world: bool = False,
    ) -> LearnerWorldAccess:
        self.grants.append(
            (user_id, world_identifier, access_source)
        )

        existing = self.records.get(world_identifier)
        if existing is not None:
            return existing

        world_id = uuid4()
        access = LearnerWorldAccess.grant(
            user_id=user_id,
            world_id=world_id,
            access_source=access_source,
        )
        self.records[world_identifier] = access
        return access

def _env_positive_int(name: str, default: int) -> int:
    raw = os.getenv(name)

    if raw is None:
        return default

    try:
        value = int(raw)
    except ValueError as exc:
        raise ValueError(
            f"{name} must be a positive integer."
        ) from exc

    if value <= 0:
        raise ValueError(
            f"{name} must be a positive integer."
        )

    return value


def _env_positive_float(name: str, default: float) -> float:
    raw = os.getenv(name)

    if raw is None:
        return default

    try:
        value = float(raw)
    except ValueError as exc:
        raise ValueError(
            f"{name} must be a positive number."
        ) from exc

    if value <= 0:
        raise ValueError(
            f"{name} must be a positive number."
        )

    return value


def create_beta_telegram_composition(
    *,
    user_repository: Any,
    token_repository: Any,
    binding_repository: Any,
    conversation_event_repository: Any,
    privacy_identifier_service: Any,
    conversation_event_validator: Any,
    expiration_minutes: int = 1440,
    conversation_runtime: ConversationRuntime | None = None,
    conversation_runtime_factory: Callable[[], ConversationRuntime] | None = None,
    rate_limiter: Any | None = None,
    abuse_protector: Any | None = None,
    observability: Any | None = None,
    provider_usage_capture: ProviderUsageCapture | None = None,
    provider_usage_repository: ProviderUsageRepository | None = None,
    functionality_control_service: (
        FunctionalityOperationalControlService | None
    ) = None,
    world_resolver: WorldResolver | None = None,
    language_response_policy: LanguageResponsePolicy | None = None,
    language_detector: LanguageDetector | None = None,
) -> BetaTelegramComposition:
    """Create the Beta Telegram application composition.

    ``functionality_control_service`` is optional at this lower-level
    composition boundary so isolated application/unit compositions can
    continue to construct Telegram without requiring infrastructure.

    Production composition is responsible for requiring and supplying
    the authoritative functionality-control service.
    """

    if expiration_minutes <= 0:
        raise ValueError(
            "expiration_minutes must be greater than zero."
        )

    if (
        conversation_runtime is not None
        and conversation_runtime_factory is not None
    ):
        raise ValueError(
            "Provide either conversation_runtime or "
            "conversation_runtime_factory, not both."
        )

    # Application configuration root:
    #
    # backend/app
    #
    app_root = (
        Path(__file__)
        .resolve()
        .parents[2]
    )

    world_repository = YamlWorldRepository(
        worlds_path=(
            app_root
            / "configuration"
            / "worlds"
        )
    )

    if world_resolver is None:
        world_resolver = WorldResolver(
            world_repository=world_repository,
        )

    beta_access_service = BetaAccessService(
        user_repository=user_repository,
        token_repository=token_repository,
        binding_repository=binding_repository,
        world_resolver=world_resolver,
        expiration_minutes=expiration_minutes,
    )

    resolved_usage_repository = provider_usage_repository

    if conversation_runtime is not None:
        resolved_conversation_runtime = conversation_runtime

    elif conversation_runtime_factory is not None:
        resolved_conversation_runtime = conversation_runtime_factory()

        # P1.6.1: when production supplies a durable usage repository,
        # attach usage capture to the factory-created ConversationRuntime.
        # Keep the existing factory API and explicit runtime injection seam.
        if (
            provider_usage_repository is not None
            and provider_usage_capture is None
            and isinstance(
                resolved_conversation_runtime,
                ConversationRuntime,
            )
            and resolved_conversation_runtime.usage_capture is None
        ):
            provider_usage_service = ProviderUsageService(
                provider_usage_repository
            )

            provider_usage_capture = ProviderUsageCapture(
                provider_usage_service
            )

            resolved_conversation_runtime = replace(
                resolved_conversation_runtime,
                usage_capture=provider_usage_capture,
                usage_context=ProviderUsageContext(
                    capability="conversation",
                    functionality="chat",
                    agent="conversation_agent",
                ),
            )

    else:
        if provider_usage_capture is None:
            if resolved_usage_repository is None:
                resolved_usage_repository = (
                    InMemoryProviderUsageRepository()
                )

            provider_usage_service = ProviderUsageService(
                resolved_usage_repository
            )

            provider_usage_capture = ProviderUsageCapture(
                provider_usage_service
            )

        resolved_conversation_runtime = ConversationRuntime(
            llm_provider=LLMProviderFactory.create(),
            usage_capture=provider_usage_capture,
            usage_context=ProviderUsageContext(
                capability="conversation",
                functionality="chat",
                agent="conversation_agent",
            ),
        )

    resolved_language_response_policy = (
        language_response_policy or LanguageResponsePolicy()
    )

    if language_detector is None:
        raise ValueError(
            "language_detector is required for Beta Telegram composition."
        )

    conversation_service = TelegramConversationService(
        conversation_runtime=resolved_conversation_runtime,
        conversation_event_repository=conversation_event_repository,
        privacy_identifier_service=privacy_identifier_service,
        conversation_event_validator=conversation_event_validator,
        language_response_policy=resolved_language_response_policy,
        language_detector=language_detector,
        world_resolver=world_resolver,
    )

    resolved_rate_limiter = (
        rate_limiter
        or InMemoryTelegramRateLimiter(
            max_messages=_env_positive_int(
                "TELEGRAM_RATE_LIMIT_MAX_MESSAGES",
                100,
            ),
            window_seconds=_env_positive_float(
                "TELEGRAM_RATE_LIMIT_WINDOW_SECONDS",
                86400.0,
            ),
        )
    )

    resolved_abuse_protector = (
        abuse_protector
        or InMemoryTelegramAbuseProtector(
            max_messages=_env_positive_int(
                "TELEGRAM_ABUSE_MAX_MESSAGES",
                10,
            ),
            window_seconds=_env_positive_float(
                "TELEGRAM_ABUSE_WINDOW_SECONDS",
                60.0,
            ),
        )
    )

    resolved_observability = (
        observability
        or LoggingTelegramObservability(
            privacy_identifier_service=privacy_identifier_service,
        )
    )

    conversation_boundary = BetaTelegramConversationBoundary(
        conversation_service=conversation_service,
        beta_access_service=beta_access_service,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,
        observability=resolved_observability,
        functionality_control_service=functionality_control_service,
    )

    private_chat_gate = BetaPrivateChatGate(
        beta_access_service=beta_access_service
    )

    start_handler = BetaTelegramStartHandler(
        beta_access_service=beta_access_service
    )

    adapter = BetaTelegramAdapter(
        start_handler=start_handler,
        private_chat_gate=private_chat_gate,
        conversation_boundary=conversation_boundary,
    )

    runtime = TelegramBotRuntime(
        adapter=adapter,
        conversation_boundary=conversation_boundary,
    )

    return BetaTelegramComposition(
        beta_access_service=beta_access_service,
        conversation_runtime=resolved_conversation_runtime,
        conversation_service=conversation_service,
        conversation_boundary=conversation_boundary,
        private_chat_gate=private_chat_gate,
        start_handler=start_handler,
        adapter=adapter,
        runtime=runtime,
        rate_limiter=resolved_rate_limiter,
        abuse_protector=resolved_abuse_protector,
        observability=resolved_observability,
        provider_usage_repository=resolved_usage_repository,
    )

class BetaAccessService:
    """
    Application boundary for Beta eligibility,
    Telegram binding and authorization.

    Telegram identity invariant:

        One Telegram identity
        -> one Beta account

    A Telegram identity that is already connected cannot
    be rebound to another Beta account through a new
    connection token.

    World assignment invariant:

        A world must be resolved through the authoritative
        WorldResolver before its ID can be persisted as the
        user's active world.

    World conversation invariant:

        A conversation context must be derived from the
        user's persisted active World through WorldResolver.

        Missing or invalid World configuration fails closed.
        The service never silently substitutes a default World.

    World compatibility:

        Legacy Beta/Telegram world identifier:
            world-fr

        Canonical World identifier:
            french_world

    Legacy identifiers are accepted only at this application
    compatibility boundary. New persisted assignments always
    use the canonical identifier returned by WorldResolver.
    """

    LEGACY_WORLD_ALIASES = {
        "world-fr": "french_world",
    }

    def __init__(
        self,
        *,
        user_repository: BetaUserRepository,
        token_repository: BetaTokenRepository,
        binding_repository: TelegramBindingRepository,
        world_resolver: WorldResolver,
        expiration_minutes: int = 1440,
    ) -> None:
        if expiration_minutes <= 0:
            raise ValueError(
                "expiration_minutes must be positive."
            )

        self._users = user_repository
        self._tokens = token_repository
        self._bindings = binding_repository
        self._world_resolver = world_resolver
        self._expiration_minutes = expiration_minutes

    async def approve_beta(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before Beta approval."
            )

        return await self._users.save(
            replace(
                user,
                beta_approved=True,
            )
        )

    async def assign_world(
        self,
        *,
        user_id: str,
        world_id: str,
    ) -> BetaUserState:
        if not world_id:
            raise ValueError(
                "world_id is required."
            )

        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email must be verified before World assignment."
            )

        if not user.beta_approved:
            raise ValueError(
                "User is not Beta approved."
            )

        # Resolve legacy application identifiers to the canonical
        # World identifier before crossing the authoritative
        # WorldResolver boundary.
        canonical_world_id = self.LEGACY_WORLD_ALIASES.get(
            world_id,
            world_id,
        )

        # Resolve the requested World before persistence.
        #
        # This is the authoritative World catalogue boundary.
        # If the World does not exist or is invalid, WorldResolver
        # raises and active_world_id is not modified.
        world = self._world_resolver.resolve(
            canonical_world_id,
        )

        return await self._users.save(
            replace(
                user,
                active_world_id=world.world_id,
            )
        )

    async def create_connection_token(
        self,
        user_id: str,
    ) -> BetaConnectionTokenResult:
        user = await self._require_user(user_id)

        if not user.email_verified:
            raise ValueError(
                "Email is not verified."
            )

        if not user.beta_approved:
            raise ValueError(
                "User is not Beta approved."
            )

        if not user.active_world_id:
            raise ValueError(
                "User must have a Learning World before Telegram access."
            )

        raw_token = secrets.token_urlsafe(32)

        token = BetaConnectionToken(
            token_id=secrets.token_hex(16),
            user_id=user.user_id,
            token_hash=self.hash_token(
                raw_token
            ),
            expires_at=(
                datetime.now(timezone.utc)
                + timedelta(
                    minutes=self._expiration_minutes
                )
            ),
        )

        await self._tokens.save(token)

        return BetaConnectionTokenResult(
            token=raw_token,
            connection=token,
        )

    async def validate_and_bind(
        self,
        *,
        token: str,
        telegram_id: int,
        chat_id: int,
    ) -> BetaUserState:
        record = await self._tokens.get_by_hash(
            self.hash_token(token)
        )

        if record is None:
            raise ValueError(
                "Invalid connection token."
            )

        if record.used:
            raise ValueError(
                "Connection token already used."
            )

        expires_at = record.expires_at

        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(
                tzinfo=timezone.utc
            )

        if expires_at <= datetime.now(timezone.utc):
            raise ValueError(
                "Connection token expired."
            )

        user = await self._require_user(
            record.user_id
        )

        if not user.eligible:
            raise ValueError(
                "User is not eligible."
            )

        # -------------------------------------------------
        # Telegram identity ownership invariant
        #
        # A Telegram identity may belong to only one
        # Beta account. Never silently transfer ownership.
        # -------------------------------------------------

        existing_binding = (
            await self._bindings.get_by_telegram_user_id(
                str(telegram_id)
            )
        )

        if existing_binding is not None:
            raise ValueError(
                "Telegram identity is already connected."
            )

        await self._bindings.save(
            TelegramBinding(
                binding_id=secrets.token_hex(16),
                user_id=user.user_id,
                telegram_user_id=str(
                    telegram_id
                ),
                chat_id=str(
                    chat_id
                ),
                connected_at=datetime.now(
                    timezone.utc
                ),
            )
        )

        await self._tokens.mark_used(
            replace(
                record,
                used_at=datetime.now(
                    timezone.utc
                ),
            )
        )

        return await self._users.save(
            replace(
                user,
                telegram_id=telegram_id,
            )
        )

    async def authorize_private_chat(
        self,
        telegram_id: int,
    ) -> BetaUserState:
        user = await self._users.get_by_telegram_id(
            telegram_id
        )

        if user is None:
            raise PermissionError(
                "Telegram account is not connected."
            )

        if not user.active:
            raise PermissionError(
                "User is not active."
            )

        return user

    async def get_conversation_context(
        self,
        telegram_user_id: int,
    ) -> BetaConversationContext:
        """
        Resolve the authoritative conversation context for
        an authorized Telegram learner.

        The persisted active_world_id is treated only as the
        World identifier. The authoritative World configuration,
        including its language code, is always obtained through
        WorldResolver.

        This method deliberately fails closed when no active World
        is assigned or when the assigned World cannot be resolved.
        """

        user = await self.authorize_private_chat(
            telegram_user_id
        )

        if not user.active_world_id:
            raise ValueError(
                "Active learning World is required."
            )

        canonical_world_id = self._canonical_world_id(
            user.active_world_id
        )

        # WorldResolver is the authoritative runtime World boundary.
        # No language or World defaults are inferred here.
        world = self._world_resolver.resolve(
            canonical_world_id
        )

        return BetaConversationContext(
            user_id=user.user_id,
            telegram_user_id=telegram_user_id,
            email=user.email,
            world_id=world.world_id,
            language_code=world.language_code,
            character="Alix",
            level="B1",
        )

    @classmethod
    def _canonical_world_id(
        cls,
        world_id: str | None,
    ) -> str:
        """
        Convert an accepted legacy application identifier to the
        canonical World identifier.

        A missing identifier is intentionally not replaced with a
        default World. Callers must fail closed instead.
        """

        if not world_id:
            raise ValueError(
                "Active learning World is required."
            )

        return cls.LEGACY_WORLD_ALIASES.get(
            world_id,
            world_id,
        )

    @staticmethod
    def hash_token(
        token: str,
    ) -> str:
        return hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

    async def _require_user(
        self,
        user_id: str,
    ) -> BetaUserState:
        user = await self._users.get(
            user_id
        )

        if user is None:
            raise ValueError(
                "Academy user does not exist."
            )

        return user
