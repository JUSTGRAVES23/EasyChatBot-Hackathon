"""Application port for the canonical Platform Intelligence review projection."""
from __future__ import annotations

from abc import ABC, abstractmethod
from collections.abc import Sequence
from typing import Any


class PlatformIntelligenceReviewSource(ABC):
    """Read-only application port for PI human-review projections.

    The backend deliberately depends on this port rather than importing or
    constructing Platform Intelligence runtime services. A production adapter
    can bind this port to the canonical ``ai_agents`` runtime integration.
    """

    @abstractmethod
    def list_reviews(self) -> Sequence[Any]:
        """Return current governed review projections."""
        raise NotImplementedError

    @abstractmethod
    def get_review(self, review_item_id: str) -> Any | None:
        """Return one governed review projection, if available."""
        raise NotImplementedError

    @abstractmethod
    def list_escalations(self) -> Sequence[Any]:
        """Return governed review projections with escalation recommendations."""
        raise NotImplementedError


class UnconfiguredPlatformIntelligenceReviewSource(PlatformIntelligenceReviewSource):
    """Safe inert source object when the canonical PI integration is absent.

    The application API intentionally uses ``source=None`` to distinguish
    an unconfigured integration from a valid empty result set. This class is
    retained as an explicit port implementation for composition/tests and is
    not used as the default API source.
    """

    def list_reviews(self) -> Sequence[Any]:
        return ()

    def get_review(self, review_item_id: str) -> Any | None:
        return None

    def list_escalations(self) -> Sequence[Any]:
        return ()

"""
Cybersecurity Decision Support Agent.

Reads cybersecurity intelligence and produces
human decision-support information.
"""

from __future__ import annotations


from cybersecurity.decision_support.priority_engine import (
    CybersecurityPriorityEngine,
)

from cybersecurity.decision_support.agent.models import (
    CybersecurityDecisionBrief,
    DecisionBriefItem,
)



class CybersecurityDecisionSupportAgent:


    agent_id = (
        "cybersecurity.decision_support"
    )


    def __init__(self):

        self.priority_engine = (
            CybersecurityPriorityEngine()
        )


    async def analyse(
        self,
        intelligence: dict,
    ) -> CybersecurityDecisionBrief:


        signals = intelligence.get(
            "signals",
            []
        )


        decisions = []


        for signal in signals:

            priority_result = (
                self.priority_engine
                .calculate_priority(
                    signal
                )
            )


            decisions.append(

                DecisionBriefItem(

                    title=
                    priority_result.title,

                    priority=
                    priority_result.priority.level,

                    severity=
                    priority_result.severity,

                    impact=(

                        f"Potential impact "
                        f"on "
                        f"{priority_result.affected_area}."

                    ),

                    affected_area=
                    priority_result.affected_area,

                    decision_required=
                    priority_result.decision_required,

                )

            )


        critical = [

            item for item in decisions

            if item.priority ==
            "CRITICAL REVIEW"

        ]


        high = [

            item for item in decisions

            if item.priority ==
            "HIGH REVIEW"

        ]



        if critical:

            overall = "CRITICAL REVIEW REQUIRED"

        elif high:

            overall = "HIGH REVIEW REQUIRED"

        else:

            overall = "MONITOR"



        return CybersecurityDecisionBrief(

            executive_summary=(

                f"Cybersecurity intelligence "
                f"review completed. "

                f"{len(decisions)} "
                f"signals analysed."

            ),

            overall_priority=overall,

            critical_items=critical,

            high_priority_items=high,

            governance_observations=[

                "Human review required for "
                "security and AI governance matters."

            ],

            trend_analysis=(

                "Trend analysis requires "
                "historical intelligence data."

            ),

            recommended_decisions=[

                "Review highest priority "
                "security matters.",

                "Confirm ownership and "
                "next review actions."

            ],

        )
