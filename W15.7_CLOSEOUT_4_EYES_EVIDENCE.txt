@dataclass(frozen=True, slots=True)
class CanonicalTransaction:
    transaction_id: str
    principal_id: str
    amount: Decimal | None
    currency: str | None
    merchant_or_service: str | None
    operation: str
    timestamp: datetime
    nonce: str
    risk_context: tuple[tuple[str, str], ...] = ()

    def __post_init__(self) -> None:
        if not self.transaction_id:
            raise ValueError("transaction_id must not be empty")

        if not self.principal_id:
            raise ValueError("principal_id must not be empty")

        if not self.operation:
            raise ValueError("operation must not be empty")

        if not self.nonce:
            raise ValueError("nonce must not be empty")

        if self.amount is not None:
            normalized = Decimal(str(self.amount))

            if not normalized.is_finite():
                raise ValueError("amount must be finite")

            object.__setattr__(
                self,
                "amount",
                normalized.quantize(Decimal("0.01")),
            )

        if self.currency is not None:
            normalized_currency = self.currency.strip().upper()

            if not normalized_currency:
                raise ValueError("currency must not be empty")

            object.__setattr__(
                self,
                "currency",
                normalized_currency,
            )

        if self.merchant_or_service is not None:
            normalized_merchant = self.merchant_or_service.strip()

            if not normalized_merchant:
                raise ValueError(
                    "merchant_or_service must not be empty"
                )

            object.__setattr__(
                self,
                "merchant_or_service",
                normalized_merchant,
            )

        normalized_operation = self.operation.strip()

        if not normalized_operation:
            raise ValueError("operation must not be empty")

        object.__setattr__(
            self,
            "operation",
            normalized_operation,
        )

        object.__setattr__(
            self,
            "timestamp",
            _normalize_datetime(self.timestamp),
        )

        object.__setattr__(
            self,
            "risk_context",
            tuple(
                sorted(
                    (str(k), str(v))
                    for k, v in self.risk_context
                )
            ),
        )

    @classmethod
    def from_mapping(
        cls,
        *,
        transaction_id: str,
        principal_id: str,
        amount: Decimal | str | int | float | None,
        currency: str | None,
        merchant_or_service: str | None,
        operation: str,
        timestamp: datetime,
        nonce: str,
        risk_context: Mapping[str, str] | None = None,
    ) -> "CanonicalTransaction":
        return cls(
            transaction_id=transaction_id,
            principal_id=principal_id,
            amount=(
                None
                if amount is None
                else Decimal(str(amount))
            ),
            currency=currency,
            merchant_or_service=merchant_or_service,
            operation=operation,
            timestamp=timestamp,
            nonce=nonce,
            risk_context=tuple(
                sorted(
                    (str(k), str(v))
                    for k, v in (risk_context or {}).items()
                )
            ),
        )

    def canonical_representation(self) -> str:
        payload = {
            "amount": (
                None
                if self.amount is None
                else format(self.amount, "f")
            ),
            "currency": self.currency,
            "merchant_or_service": self.merchant_or_service,
            "nonce": self.nonce,
            "operation": self.operation,
            "principal_id": self.principal_id,
            "risk_context": [
                [key, value]
                for key, value in self.risk_context
            ],
            "timestamp": self.timestamp.isoformat(),
            "transaction_id": self.transaction_id,
        }

        return json.dumps(
            payload,
            sort_keys=True,
            separators=(",", ":"),
            ensure_ascii=False,
        )


@dataclass(frozen=True, slots=True)
class TrustedApprovalEvidence:
    """
    Approval evidence must already have crossed a trusted application
    boundary. W15.7 never derives authentication assurance itself.
    """

    authorization_id: str
    transaction: CanonicalTransaction
    approved_at: datetime
    evidence_id: str
    assurance_context: AuthenticationAssuranceContext

    def __post_init__(self) -> None:
        if not self.authorization_id:
            raise ValueError(
                "authorization_id must not be empty"
            )

        if not self.evidence_id:
            raise ValueError(
                "evidence_id must not be empty"
            )

        object.__setattr__(
            self,
            "approved_at",
            _normalize_datetime(self.approved_at),
        )

        if not isinstance(
            self.assurance_context,
            AuthenticationAssuranceContext,
        ):
            raise TypeError(
                "assurance_context must be "
                "AuthenticationAssuranceContext"
            )

        if not isinstance(
            self.assurance_context.assurance_level,
            AuthenticationAssuranceLevel,
        ):
            raise TypeError(
                "assurance_context.assurance_level must be "
                "AuthenticationAssuranceLevel"
            )

        if (
            self.assurance_context.principal_id
            != self.transaction.principal_id
        ):
            raise ValueError(
                "approval assurance principal binding mismatch"
            )

        authentication_time = _normalize_datetime(
            self.assurance_context.authentication_time
        )

        assurance_time = _normalize_datetime(
            self.assurance_context.assurance_time
        )

        if authentication_time > assurance_time:
            raise ValueError(
                "authentication time cannot be after assurance time"
            )

    @property
    def assurance_level(
        self,
    ) -> AuthenticationAssuranceLevel:
        return self.assurance_context.assurance_level


@dataclass(frozen=True, slots=True)
class TransactionAuthorization:
    authorization_id: str
    transaction: CanonicalTransaction
    created_at: datetime
    expires_at: datetime
    state: TransactionAuthorizationState = (
        TransactionAuthorizationState.CREATED
    )
    approval: TrustedApprovalEvidence | None = None

    def __post_init__(self) -> None:
        if not self.authorization_id:
            raise ValueError(
                "authorization_id must not be empty"
            )

        created = _normalize_datetime(self.created_at)
        expires = _normalize_datetime(self.expires_at)

        if expires <= created:
            raise ValueError(
                "expires_at must be after created_at"
            )

        object.__setattr__(
            self,
            "created_at",
            created,
        )

        object.__setattr__(
            self,
            "expires_at",
            expires,
        )

        if self.approval is not None:
            if (
                self.approval.authorization_id
                != self.authorization_id
            ):
                raise ValueError(
                    "approval authorization binding mismatch"
                )

            if (
                self.approval.assurance_context.principal_id
                != self.transaction.principal_id
            ):
                raise ValueError(
                    "approval assurance principal binding mismatch"
                )

            if (
                self.approval.transaction.canonical_representation()
                != self.transaction.canonical_representation()
            ):
                raise ValueError(
                    "approval transaction binding mismatch"
                )

    TransactionAuthorizationState.CREATED: {
        TransactionAuthorizationState.PENDING_USER_APPROVAL,
    },
    TransactionAuthorizationState.PENDING_USER_APPROVAL: {
        TransactionAuthorizationState.APPROVED,

    TransactionAuthorizationState.EXPIRED: set(),

class NonceRegistry:
    """
    Trusted process-local uniqueness boundary for authorization nonces.
    """

    def __init__(self) -> None:
        self._lock = Lock()
        self._nonces: set[str] = set()

    def register(self, nonce: str) -> None:
        if not nonce:
            raise ValueError("nonce must not be empty")

        with self._lock:
            if nonce in self._nonces:
                raise ValueError(
                    TransactionAuthorizationFailure.NONCE_REPLAY.value
                )

            self._nonces.add(nonce)


class AuthorizationIdRegistry:

    def __init__(self) -> None:

        self._authorization_ids: set[str] = set()

    def register(self, authorization_id: str) -> None:

class TransactionAuthorizationRegistry:
    """
    Trusted registration and atomic state-update boundary.
    """

    def __init__(self) -> None:
        self._lock = Lock()
        self._authorizations: dict[
            str,
            TransactionAuthorization,
        ] = {}
        self._nonces: set[str] = set()

    def register(
        self,
        authorization: TransactionAuthorization,
    ) -> None:
        authorization_id = authorization.authorization_id
        nonce = authorization.transaction.nonce

        with self._lock:
            if authorization_id in self._authorizations:
                raise ValueError(
                    TransactionAuthorizationFailure
                    .DUPLICATE_AUTHORIZATION_ID.value
                )

            if nonce in self._nonces:
                raise ValueError(
                    TransactionAuthorizationFailure.NONCE_REPLAY.value
                )

            self._authorizations[authorization_id] = authorization
            self._nonces.add(nonce)

    def get(
        self,
        authorization_id: str,
    ) -> TransactionAuthorization | None:
        with self._lock:
            return self._authorizations.get(authorization_id)

    def replace(
        self,
        authorization_id: str,
        expected: TransactionAuthorization,
        replacement: TransactionAuthorization,
    ) -> bool:
        with self._lock:
            current = self._authorizations.get(
                authorization_id
            )

            if current is None or current != expected:
                return False

            self._authorizations[authorization_id] = replacement
            return True


class TransactionAuthorizationConsumer:
    """
    Atomic single-use consumption boundary.

    The in-memory lock provides concurrency semantics for this domain
    implementation. Durable multi-process persistence must later provide
    equivalent atomic conditional-update/row-lock semantics.
    """

    def __init__(
        self,
        registry: TransactionAuthorizationRegistry,
    ) -> None:
        self._registry = registry
        self._lock = Lock()

    def consume_once(
        self,
        authorization_id: str,
        *,
        expected_transaction: CanonicalTransaction,
        clock: TrustedClock,
        required_assurance_level: AuthenticationAssuranceLevel = (
            AuthenticationAssuranceLevel.A0
        ),
    ) -> TransactionAuthorizationDecisionResult:
        if not isinstance(
            required_assurance_level,
            AuthenticationAssuranceLevel,
        ):
            return TransactionAuthorizationDecisionResult(
                TransactionAuthorizationDecision.DENY,
                TransactionAuthorizationFailure
                .INSUFFICIENT_AUTHENTICATION_ASSURANCE,
            )

        now = _trusted_now(clock)

        with self._lock:
            authorization = self._registry.get(
                authorization_id
            )

            if authorization is None:
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.INVALID_TRANSACTION,
                )

            if (
                authorization.transaction.canonical_representation()
                != expected_transaction.canonical_representation()
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure
                    .INVALID_TRANSACTION_BINDING,
                )

            if (
                authorization.transaction.principal_id
                != expected_transaction.principal_id
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure
                    .INVALID_PRINCIPAL_BINDING,
                )

            if (
                authorization.state
                is TransactionAuthorizationState.CONSUMED
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.ALREADY_CONSUMED,
                )

            if (
                authorization.state
                is TransactionAuthorizationState.REVOKED
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.REVOKED,
                )

            if (
                authorization.state
                is TransactionAuthorizationState.EXPIRED
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.EXPIRED,
                )

            if now >= authorization.expires_at:
                expired = transition(
                    authorization,
                    TransactionAuthorizationState.EXPIRED,
                )

                self._registry.replace(
                    authorization_id,
                    authorization,
                    expired,
                )

                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.EXPIRED,
                )

            if (
                authorization.state
                is not TransactionAuthorizationState.APPROVED
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.INVALID_APPROVAL,
                )

            approval = authorization.approval

            if approval is None:
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.INVALID_APPROVAL,
                )

            if (
                approval.assurance_context.principal_id
                != authorization.transaction.principal_id
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure
                    .INVALID_ASSURANCE_PROVENANCE,
                )

            if (
                approval.assurance_context.assurance_time
                > now
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure
                    .INVALID_ASSURANCE_PROVENANCE,
                )

            if (
                approval.assurance_context.authentication_time
                > now
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure
                    .INVALID_ASSURANCE_PROVENANCE,
                )

            if (
                approval.transaction.canonical_representation()
                != authorization.transaction.canonical_representation()
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure
                    .INVALID_TRANSACTION_BINDING,
                )

            if (
                approval.assurance_context.assurance_level.value
                < required_assurance_level.value
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure
                    .INSUFFICIENT_AUTHENTICATION_ASSURANCE,
                )

            consumed = transition(
                authorization,
                TransactionAuthorizationState.CONSUMED,
            )

            if not self._registry.replace(
                authorization_id,
                authorization,
                consumed,
            ):
                return TransactionAuthorizationDecisionResult(
                    TransactionAuthorizationDecision.DENY,
                    TransactionAuthorizationFailure.ALREADY_CONSUMED,
                )

            return TransactionAuthorizationDecisionResult(
                TransactionAuthorizationDecision.ALLOW
            )


def transition(
    authorization: TransactionAuthorization,
    new_state: TransactionAuthorizationState,
) -> TransactionAuthorization:
    allowed = _ALLOWED_TRANSITIONS[
        authorization.state
    ]

    if new_state not in allowed:
        raise ValueError(
            "invalid transaction authorization transition: "
            f"{authorization.state.value} -> "
            f"{new_state.value}"
        )

    return TransactionAuthorization(
        authorization_id=authorization.authorization_id,
        transaction=authorization.transaction,
        created_at=authorization.created_at,
        expires_at=authorization.expires_at,
        state=new_state,
        approval=authorization.approval,
    )


def attach_approval(
    authorization: TransactionAuthorization,
    approval: TrustedApprovalEvidence,
    *,
    clock: TrustedClock,
) -> TransactionAuthorization:
    if (
        authorization.state
        is not TransactionAuthorizationState.PENDING_USER_APPROVAL
    ):
        raise ValueError(
            "approval can only be attached while pending"
        )

    if (
        approval.authorization_id
        != authorization.authorization_id
    ):
        raise ValueError(
            "approval authorization binding mismatch"
        )

    if (
        approval.assurance_context.principal_id
        != authorization.transaction.principal_id
    ):
        raise ValueError(
            "approval assurance principal binding mismatch"
        )

    if (
        approval.transaction.canonical_representation()
        != authorization.transaction.canonical_representation()
    ):
        raise ValueError(
            "approval is not bound to the exact transaction"
        )

    now = _trusted_now(clock)

    if (
        approval.assurance_context.assurance_time
        > now
    ):
        raise ValueError(
            "future-dated authentication assurance"
        )

    if (
        approval.assurance_context.authentication_time
        > now
    ):
        raise ValueError(
            "future-dated authentication evidence"
        )

    if approval.approved_at > now:
        raise ValueError(
            "future-dated approval is invalid"
        )

    return transition(
        TransactionAuthorization(
            authorization_id=authorization.authorization_id,
            transaction=authorization.transaction,
            created_at=authorization.created_at,
            expires_at=authorization.expires_at,
            state=TransactionAuthorizationState.PENDING_USER_APPROVAL,
            approval=approval,
        ),
        TransactionAuthorizationState.APPROVED,
    )


def expire_if_due(
    authorization: TransactionAuthorization,
    *,
    clock: TrustedClock,
) -> TransactionAuthorization:
    now = _trusted_now(clock)

    if (
        authorization.state
        is TransactionAuthorizationState.APPROVED
        and now >= authorization.expires_at
    ):
        return transition(
            authorization,
            TransactionAuthorizationState.EXPIRED,
        )

    return authorization


def consume(
    authorization: TransactionAuthorization,
    *,
    clock: TrustedClock,
    required_assurance_level: AuthenticationAssuranceLevel = (
        AuthenticationAssuranceLevel.A0
    ),
) -> TransactionAuthorizationDecisionResult:
    if not isinstance(
        required_assurance_level,
        AuthenticationAssuranceLevel,
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure
            .INSUFFICIENT_AUTHENTICATION_ASSURANCE,
        )

    now = _trusted_now(clock)

    if (
        authorization.state
        is TransactionAuthorizationState.CONSUMED
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure.ALREADY_CONSUMED,
        )

    if (
        authorization.state
        is TransactionAuthorizationState.EXPIRED
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure.EXPIRED,
        )

    if (
        authorization.state
        is TransactionAuthorizationState.REVOKED
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure.REVOKED,
        )

    if (
        authorization.state
        is not TransactionAuthorizationState.APPROVED
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure.INVALID_APPROVAL,
        )

    if now >= authorization.expires_at:
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure.EXPIRED,
        )

    approval = authorization.approval

    if approval is None:
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure.INVALID_APPROVAL,
        )

    if (
        approval.assurance_context.principal_id
        != authorization.transaction.principal_id
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure
            .INVALID_ASSURANCE_PROVENANCE,
        )

    if (
        approval.assurance_context.assurance_time
        > now
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure
            .INVALID_ASSURANCE_PROVENANCE,
        )

    if (
        approval.assurance_context.authentication_time
        > now
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure
            .INVALID_ASSURANCE_PROVENANCE,
        )

    if (
        approval.transaction.canonical_representation()
        != authorization.transaction.canonical_representation()
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure
            .INVALID_TRANSACTION_BINDING,
        )

    if (
        approval.assurance_context.assurance_level.value
        < required_assurance_level.value
    ):
        return TransactionAuthorizationDecisionResult(
            TransactionAuthorizationDecision.DENY,
            TransactionAuthorizationFailure
            .INSUFFICIENT_AUTHENTICATION_ASSURANCE,
        )

    return TransactionAuthorizationDecisionResult(
        TransactionAuthorizationDecision.ALLOW
    )


def _normalize_datetime(value: datetime) -> datetime:
    if value.tzinfo is None:
        raise ValueError(
            "datetime must be timezone-aware"
        )

    return value.astimezone(timezone.utc)


def _trusted_now(clock: TrustedClock) -> datetime:
    return _normalize_datetime(clock.now())

def assurance_context(
    *,
    principal_id: str = "user-001",
    level: AuthenticationAssuranceLevel = AuthenticationAssuranceLevel.A2,
    assurance_time: datetime | None = None,

    authorization: TransactionAuthorization,
    clock: FixedClock,
) -> None:
    approved = approved_authorization(
        authorization,
        clock,
    )

    registry = TransactionAuthorizationRegistry()
    registry.register(approved)

    consumer = TransactionAuthorizationConsumer(registry)

    def attempt():
        return consumer.consume_once(
            authorization.authorization_id,
            expected_transaction=authorization.transaction,
            clock=clock,
            required_assurance_level=AuthenticationAssuranceLevel.A2,
        )

    with ThreadPoolExecutor(max_workers=16) as executor:
        results = list(
            executor.map(
                lambda _: attempt(),
                range(16),
            )
        )

    allows = [
        result
        for result in results
        if result.decision is TransactionAuthorizationDecision.ALLOW
    ]

    assert len(allows) == 1

class TransactionAuthorizationApplicationError(RuntimeError):
    """Application-layer transaction-authorization failure."""


@dataclass(frozen=True, slots=True)
class TransactionAuthorizationApplicationService:
    """Application boundary for transaction authorization.

    Coordinates authenticated-principal validation, domain transaction
    authorization, assurance-context consumption, and atomic consumption.

    This service does not implement authentication, security-factor
    management, external identity integration, authentication freshness
    policy, business authorization policy, payment execution, money movement,
    or cryptographic transaction signing.

    W15.7 consumes the W15.6 AuthenticationAssuranceContext.
    """

    authorization_service: AuthorizationService
    registry: TransactionAuthorizationRegistry
    consumer: TransactionAuthorizationConsumer

    def require_authorized_principal(
        self,
        *,
        principal: Principal,
    ) -> Principal:
        """Require an authenticated and active principal."""
        return self.authorization_service.require_authenticated(principal)

    def create_pending_authorization(
        self,
        *,
        principal: Principal,
        transaction: CanonicalTransaction,
        authorization_id: str,
        created_at: datetime,
        expires_at: datetime,
    ) -> TransactionAuthorization:
        """Create and register a pending transaction authorization."""
        authenticated = self.require_authorized_principal(
            principal=principal,
        )

        if transaction.principal_id != authenticated.user_id:
            raise TransactionAuthorizationApplicationError(
                "Transaction principal does not match authenticated principal."
            )

        authorization = TransactionAuthorization(
            authorization_id=authorization_id,
            transaction=transaction,
            created_at=created_at,
            expires_at=expires_at,
        )

        pending = transition(
            authorization,
            TransactionAuthorizationState.PENDING_USER_APPROVAL,
        )

        self.registry.register(pending)

        return pending

    def attach_user_approval(
        self,
        *,
        principal: Principal,
        authorization: TransactionAuthorization,
        approval: TrustedApprovalEvidence,
        clock: TrustedClock,
    ) -> TransactionAuthorization:
        """Attach trusted approval to the exact pending transaction.

        Assurance is consumed from the W15.6 AuthenticationAssuranceContext.
        W15.7 does not derive or recreate authentication assurance.
        """
        authenticated = self.require_authorized_principal(
            principal=principal,
        )

        if authorization.transaction.principal_id != authenticated.user_id:
            raise TransactionAuthorizationApplicationError(
                "Transaction principal does not match authenticated principal."
            )

        if approval.transaction.principal_id != authenticated.user_id:
            raise TransactionAuthorizationApplicationError(
                "Approval principal does not match authenticated principal."
            )

        if not isinstance(
            approval.assurance_context,
            AuthenticationAssuranceContext,
        ):
            raise TransactionAuthorizationApplicationError(
                "W15.6 AuthenticationAssuranceContext is required."
            )

        if approval.assurance_context.principal_id != authenticated.user_id:
            raise TransactionAuthorizationApplicationError(
                "Authentication assurance principal does not match "
                "authenticated principal."
            )

        if not isinstance(
            approval.assurance_context.assurance_level,
            AuthenticationAssuranceLevel,
        ):
            raise TransactionAuthorizationApplicationError(
                "Authentication assurance level is invalid."
            )

        approved = attach_approval(
            authorization,
            approval,
            clock=clock,
        )

        # TransactionAuthorization is immutable. Replace the pending
        # snapshot in the trusted registry with the approved snapshot.
        # Compare-and-swap prevents overwriting a concurrent state change.
        replaced = self.registry.replace(
            authorization.authorization_id,
            authorization,
            approved,
        )

        if not replaced:
            raise TransactionAuthorizationApplicationError(
                "Transaction authorization state changed concurrently."
            )

        return approved

    def consume_authorization(
        self,
        *,
        principal: Principal,
        authorization_id: str,
        expected_transaction: CanonicalTransaction,
        clock: TrustedClock,
        required_assurance_level: AuthenticationAssuranceLevel,
    ) -> TransactionAuthorizationDecisionResult:
        """Atomically consume one exact transaction authorization."""
        authenticated = self.require_authorized_principal(
            principal=principal,
        )

        if expected_transaction.principal_id != authenticated.user_id:
            raise TransactionAuthorizationApplicationError(
                "Transaction principal does not match authenticated principal."
            )

        if not isinstance(
            required_assurance_level,
            AuthenticationAssuranceLevel,
        ):
            raise TransactionAuthorizationApplicationError(
                "Required authentication assurance level is invalid."
            )

        return self.consumer.consume_once(
            authorization_id,
            expected_transaction=expected_transaction,
            clock=clock,
            required_assurance_level=required_assurance_level,
        )

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from enum import Enum, IntEnum
from typing import Protocol


class AuthenticationAssuranceLevel(IntEnum):
    """
    Monotonic authentication-assurance levels.

    A higher level represents stronger authentication assurance.
    Higher assurance does NOT imply authorization.
    """

    A0 = 0
    A1 = 1
    A2 = 2
    A3 = 3


class AuthenticationAssuranceDecision(str, Enum):
    """Deterministic result of an assurance evaluation."""

    ALLOW = "ALLOW"
    DENY = "DENY"


class AuthenticationAssuranceFailure(str, Enum):
    """Deterministic fail-closed assurance failures."""

    UNAUTHENTICATED = "UNAUTHENTICATED"
    INVALID_AUTHENTICATION_EVIDENCE = "INVALID_AUTHENTICATION_EVIDENCE"
    INVALID_PRINCIPAL_BINDING = "INVALID_PRINCIPAL_BINDING"
    INVALID_AUDIENCE = "INVALID_AUDIENCE"
    STALE_AUTHENTICATION_ASSURANCE = "STALE_AUTHENTICATION_ASSURANCE"
    INSUFFICIENT_AUTHENTICATION_ASSURANCE = (
        "INSUFFICIENT_AUTHENTICATION_ASSURANCE"
    )


class TrustedClock(Protocol):
    """
    Trusted application/server clock.

    Freshness decisions MUST use this clock rather than timestamps
    supplied by the caller.
    """

    def now(self) -> datetime:
        ...


@dataclass(frozen=True, slots=True)
class ValidatedAuthenticationEvidence:
    """
    Immutable authentication evidence produced after Identity Adapter
    validation.

    This object deliberately contains NO caller-supplied assurance level.
    Assurance is derived by the central AuthenticationAssuranceService
    from validated evidence and centrally supplied policy.

    Raw credentials, access tokens, refresh tokens, OTP secrets, and
    WebAuthn private material MUST NOT be stored here.
    """

    principal_id: str
    issuer: str
    subject: str
    audience: str
    authentication_time: datetime
    authentication_method: str
    authentication_strength: str
    user_verified: bool
    phishing_resistant: bool
    evidence_id: str
    policy_version: str


@dataclass(frozen=True, slots=True)
class AuthenticationAssuranceContext:
    """
    Immutable, validated assurance context returned after successful
    assurance evaluation.

    The context describes authentication assurance only. It does not
    grant business authorization and does not represent transaction
    authorization.
    """

    principal_id: str
    assurance_level: AuthenticationAssuranceLevel
    authentication_time: datetime
    assurance_time: datetime
    authentication_method: str
    authentication_strength: str
    user_verified: bool
    phishing_resistant: bool
    issuer: str
    subject: str
    audience: str
    evidence_id: str
    policy_version: str


@dataclass(frozen=True, slots=True)
class AuthenticationAssuranceDecisionResult:
    """
    Deterministic assurance decision.

    ALLOW contains a validated context.
    DENY contains a deterministic failure reason.
    """

    decision: AuthenticationAssuranceDecision
    context: AuthenticationAssuranceContext | None = None
    failure: AuthenticationAssuranceFailure | None = None
