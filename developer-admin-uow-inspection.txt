from dataclasses import dataclass
from uuid import UUID

)

from app.domain.identity.principal import Principal

class DeveloperAdminError(Exception):
    """Base error for Developer Administration operations."""


class DeveloperAdminUserNotFoundError(DeveloperAdminError):
    """Requested target user does not exist."""


class DeveloperAdminUserInactiveError(DeveloperAdminError):
    """Requested target user cannot receive Developer Admin privileges."""


class DeveloperAdminLastAdminError(DeveloperAdminError):
    """The final Developer Administrator cannot be removed."""


@dataclass(frozen=True, slots=True)
class DeveloperAdminRecord:
    user_id: UUID
    email: str
    username: str
    status: str
    email_verified: bool


class DeveloperAdminService:
    """Application boundary for controlled Developer Administrator management.

    Production composition injects a DeveloperAdminUnitOfWork so role
    mutations and their security audit evidence share one database
    transaction.

    The repository-only constructor path is retained for compatibility with
    existing isolated service tests. That compatibility path is deliberately
    not used by the production application composition.
    """

    ROLE = Role.DEVELOPER_ADMIN.value

    def __init__(
        self,
        repository: PostgreSQLAuthenticationRepository,
        *,
        unit_of_work: DeveloperAdminUnitOfWork | None = None,
        audit_service: DeveloperAdminAuditService | None = None,
    ) -> None:
        if repository is None:
            raise ValueError("repository is required.")

        if (unit_of_work is None) != (audit_service is None):
            raise ValueError(
                "unit_of_work and audit_service must be supplied together."
            )

        self._repository = repository
        self._unit_of_work = unit_of_work
        self._audit_service = audit_service

    async def list_developer_administrators(
        self,
    ) -> tuple[DeveloperAdminRecord, ...]:
        users = await self._repository.get_users_with_role(
            role=self.ROLE,
        )

        return tuple(
            self._to_record(user)
            for user in users
        )

    async def grant_developer_admin(
        self,
        *,
        user_id: UUID,
        actor: Principal | None = None,
        correlation_id: str | None = None,
    ) -> DeveloperAdminRecord:
        # Production path: role mutation + audit share one transaction.
        if self._unit_of_work is not None:
            if actor is None:
                raise ValueError(
                    "actor is required when using the atomic Developer Admin path."
                )

            return await self._grant_atomically(
                actor=actor,
                user_id=user_id,
                correlation_id=correlation_id,
            )

        # Compatibility path for existing isolated application tests.
        # Production composition never uses this path.
        return await self._grant_legacy(
            user_id=user_id,
        )

    async def revoke_developer_admin(
        self,
        *,
        user_id: UUID,
        actor: Principal | None = None,
        correlation_id: str | None = None,
    ) -> None:
        # Production path: role mutation + audit share one transaction.
        if self._unit_of_work is not None:
            if actor is None:
                raise ValueError(
                    "actor is required when using the atomic Developer Admin path."
                )

            await self._revoke_atomically(
                actor=actor,
                user_id=user_id,
                correlation_id=correlation_id,
            )
            return

        # Compatibility path for existing isolated application tests.
        # Production composition never uses this path.
        await self._revoke_legacy(
            user_id=user_id,
        )

    async def _grant_atomically(
        self,
        *,
        actor: Principal,
        user_id: UUID,
        correlation_id: str | None,
    ) -> DeveloperAdminRecord:
        assert self._unit_of_work is not None
        assert self._audit_service is not None

        async with self._unit_of_work as uow:
            target = await uow.get_user(user_id)

            if target is None:
                raise DeveloperAdminUserNotFoundError(
                    "Target user does not exist."
                )

            if target.status != "active" or not target.email_verified:
                raise DeveloperAdminUserInactiveError(
                    "Developer Administrator privileges require an active, "
                    "verified user."
                )

            created = await uow.add_user_role(
                user_id=user_id,
                role=self.ROLE,
            )

            if created:
                event = await self._audit_service.build_event(
                    event_type=DeveloperAdminAuditService.GRANTED,
                    actor_user_id=UUID(actor.user_id),
                    target_user_id=target.id,
                    actor_email=actor.email,
                    target_email=target.email,
                    correlation_id=correlation_id,
                    metadata={"operation": "grant"},
                )

                await uow.record_audit(event)

            return self._to_record(target)

    async def _revoke_atomically(
        self,
        *,
        actor: Principal,
        user_id: UUID,
        correlation_id: str | None,
    ) -> None:
        assert self._unit_of_work is not None
        assert self._audit_service is not None

        async with self._unit_of_work as uow:
            target = await uow.get_user(user_id)

            if target is None:
                raise DeveloperAdminUserNotFoundError(
                    "Target user does not exist."
                )

            roles = await uow.get_user_roles(user_id)

            if self.ROLE not in roles:
                return

            current_count = await uow.count_users_with_role(
                role=self.ROLE,
            )

            if current_count <= 1:
                raise DeveloperAdminLastAdminError(
                    "The last Developer Administrator cannot be removed."
                )

            removed = await uow.remove_user_role(
                user_id=user_id,
                role=self.ROLE,
            )

            if removed:
                event = await self._audit_service.build_event(
                    event_type=DeveloperAdminAuditService.REVOKED,
                    actor_user_id=UUID(actor.user_id),
                    target_user_id=target.id,
                    actor_email=actor.email,
                    target_email=target.email,
                    correlation_id=correlation_id,
                    metadata={"operation": "revoke"},
                )

                await uow.record_audit(event)

    async def _grant_legacy(
        self,
        *,
        user_id: UUID,
    ) -> DeveloperAdminRecord:
        user = await self._repository.get_user_for_identity(user_id)

        if user is None:
            raise DeveloperAdminUserNotFoundError(
                "Target user does not exist."
            )

        if user.status != "active" or not user.email_verified:
            raise DeveloperAdminUserInactiveError(
                "Developer Administrator privileges require an active, "
                "verified user."
            )

        await self._repository.add_user_role(
            user_id=user_id,
            role=self.ROLE,
        )

        return self._to_record(user)

    async def _revoke_legacy(
        self,
        *,
        user_id: UUID,
    ) -> None:
        user = await self._repository.get_user_for_identity(user_id)

        if user is None:
            raise DeveloperAdminUserNotFoundError(
                "Target user does not exist."
            )

        current_count = await self._repository.count_users_with_role(
            role=self.ROLE,
        )

        roles = await self._repository.get_user_roles(user_id)

        if self.ROLE not in roles:
            return

        if current_count <= 1:
            raise DeveloperAdminLastAdminError(
                "The last Developer Administrator cannot be removed."
            )

        await self._repository.remove_user_role(
            user_id=user_id,
            role=self.ROLE,
        )

    @staticmethod
    def _to_record(
        user: AuthUserRecord,
    ) -> DeveloperAdminRecord:
        return DeveloperAdminRecord(
            user_id=user.id,
            email=user.email,
            username=user.username,
            status=user.status,
            email_verified=user.email_verified,
        )

from __future__ import annotations

import json

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker

from app.application.developer_admin.developer_admin_audit import (
    DeveloperAdminAuditEvent,
    DeveloperAdminAuditPort,
)


class PostgreSQLDeveloperAdminAuditRepository(DeveloperAdminAuditPort):
    """Durable PostgreSQL repository for Developer Admin audit events.

    ``record`` is the standalone repository operation and owns its
    transaction.

    ``record_in_session`` is the transaction-aware operation used by the
    Developer Admin Unit of Work. It deliberately does not commit or
    rollback; the Unit of Work owns that transaction boundary.
    """

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")

        self._session_factory = session_factory

    async def record(
        self,
        event: DeveloperAdminAuditEvent,
    ) -> DeveloperAdminAuditEvent:
        """Persist one audit event using an independent transaction.

        This standalone operation owns its transaction and therefore commits
        automatically when the ``session.begin()`` context exits normally.

        The SQL is intentionally present here rather than delegated to
        ``record_in_session`` so this public repository operation remains
        independently inspectable and explicit.
        """
        metadata = dict(event.metadata)

        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO developer_admin_audit_events (
                            id,
                            event_type,
                            actor_user_id,
                            target_user_id,
                            actor_email,
                            target_email,
                            occurred_at,
                            correlation_id,
                            outcome,
                            metadata
                        )
                        VALUES (
                            :id,
                            :event_type,
                            :actor_user_id,
                            :target_user_id,
                            :actor_email,
                            :target_email,
                            :occurred_at,
                            :correlation_id,
                            :outcome,
                            CAST(:metadata AS jsonb)
                        )
                        """
                    ),
                    {
                        "id": event.event_id,
                        "event_type": event.event_type,
                        "actor_user_id": event.actor_user_id,
                        "target_user_id": event.target_user_id,
                        "actor_email": event.actor_email,
                        "target_email": event.target_email,
                        "occurred_at": event.occurred_at,
                        "correlation_id": event.correlation_id,
                        "outcome": event.outcome,
                        "metadata": json.dumps(metadata),
                    },
                )

        return event

    async def record_in_session(
        self,
        session: AsyncSession,
        event: DeveloperAdminAuditEvent,
    ) -> DeveloperAdminAuditEvent:
        """Persist an audit event inside an existing transaction.

        The caller owns the transaction. This method MUST NOT commit or
        rollback.
        """
        if session is None:
            raise ValueError("session is required.")

        metadata = dict(event.metadata)

        await session.execute(
            text(
                """
                INSERT INTO developer_admin_audit_events (
                    id,
                    event_type,
                    actor_user_id,
                    target_user_id,
                    actor_email,
                    target_email,
                    occurred_at,
                    correlation_id,
                    outcome,
                    metadata
                )
                VALUES (
                    :id,
                    :event_type,
                    :actor_user_id,
                    :target_user_id,
                    :actor_email,
                    :target_email,
                    :occurred_at,
                    :correlation_id,
                    :outcome,
                    CAST(:metadata AS jsonb)
                )
                """
            ),
            {
                "id": event.event_id,
                "event_type": event.event_type,
                "actor_user_id": event.actor_user_id,
                "target_user_id": event.target_user_id,
                "actor_email": event.actor_email,
                "target_email": event.target_email,
                "occurred_at": event.occurred_at,
                "correlation_id": event.correlation_id,
                "outcome": event.outcome,
                "metadata": json.dumps(metadata),
            },
        )

        return event

class DeveloperAdminGrantRequest(BaseModel):
    user_id: UUID


class DeveloperAdminResponse(BaseModel):
    user_id: UUID
    email: str
    username: str
    status: str
    email_verified: bool


class DeveloperAdminListResponse(BaseModel):
    administrators: list[DeveloperAdminResponse]


def _get_service(request: Request) -> DeveloperAdminService:
    service = getattr(request.app.state, "developer_admin_service", None)
    if service is None:
        raise RuntimeError("developer_admin_service is not configured.")
    return service


router = APIRouter(
    prefix="/api/operations-center/developer-administrators",
    tags=["operations-center-developer-administrators"],
)


@router.get("", response_model=DeveloperAdminListResponse)
async def list_developer_administrators(
    request: Request,
    _principal: Principal = Depends(
        require_permission(Permission.DEVELOPER_ADMIN_MANAGE)
    ),
) -> DeveloperAdminListResponse:
    administrators = await _get_service(request).list_developer_administrators()
    return DeveloperAdminListResponse(
        administrators=[
            DeveloperAdminResponse(
                user_id=item.user_id,
                email=item.email,
                username=item.username,
                status=item.status,
                email_verified=item.email_verified,
            )
            for item in administrators
        ]
    )


@router.post(
    "",
    response_model=DeveloperAdminResponse,
    status_code=status.HTTP_201_CREATED,
)
async def grant_developer_admin(
    payload: DeveloperAdminGrantRequest,
    request: Request,
    principal: Principal = Depends(
        require_permission(Permission.DEVELOPER_ADMIN_MANAGE)
    ),
) -> DeveloperAdminResponse:
    try:
        administrator = await _get_service(request).grant_developer_admin(
            actor=principal,
            user_id=payload.user_id,
        )
    except DeveloperAdminUserNotFoundError as exc:
        raise HTTPException(status_code=404, detail=str(exc)) from None
    except DeveloperAdminUserInactiveError as exc:
        raise HTTPException(status_code=409, detail=str(exc)) from None

    return DeveloperAdminResponse(
        user_id=administrator.user_id,
        email=administrator.email,
        username=administrator.username,
        status=administrator.status,
        email_verified=administrator.email_verified,
    )


@router.delete(
    "/{user_id}",
    status_code=status.HTTP_204_NO_CONTENT,
)
async def revoke_developer_admin(
    user_id: UUID,
    request: Request,
    principal: Principal = Depends(
        require_permission(Permission.DEVELOPER_ADMIN_MANAGE)
    ),
) -> None:
    try:
        await _get_service(request).revoke_developer_admin(
            actor=principal,
            user_id=user_id,
        )
    except DeveloperAdminUserNotFoundError as exc:
        raise HTTPException(status_code=404, detail=str(exc)) from None
    except DeveloperAdminLastAdminError as exc:
        raise HTTPException(status_code=409, detail=str(exc)) from None

def _required_environment(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for authentication email composition."

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )


def create_application() -> FastAPI:
    academy_config = (
        AcademyActivationConfig.from_environment()
    )

    auth_engine = create_async_database_engine()

    auth_session_factory = (
        create_async_session_factory(
            auth_engine,
        )
    )

    auth_repository = (
        PostgreSQLAuthenticationRepository(
            auth_session_factory,
        )
    )

    email_provider_manager = EmailProviderManager()

    authentication_email_sender = AuthenticationEmailSenderAdapter(
        provider_manager=email_provider_manager,
    )

    authentication_email_service = AuthenticationEmailService(
        authentication_email_sender,
        frontend_base_url=_required_environment(
            "EASYCHATBOT_PORTAL_URL",
        ),
    )

    authentication_service = AuthenticationService(
        auth_repository,
        email_service=authentication_email_service,
    )

    # ------------------------------------------------------------------
    # Developer Administration
    #
    # Production path:
    #
    #   DeveloperAdminService
    #          |
    #          +--> PostgreSQLDeveloperAdminUnitOfWork
    #          |       |
    #          |       +--> auth_user_roles
    #          |       |
    #          |       +--> developer_admin_audit_events
    #          |
    #          +--> DeveloperAdminAuditService
    #
    # Role mutation and its security audit evidence are therefore handled
    # through the same Unit-of-Work transaction.
    #
    # The standalone audit repository owns transactions only when its
    # standalone ``record`` operation is used. The production Developer
    # Admin mutation uses ``record_in_session`` through the UoW.
    # ------------------------------------------------------------------

    developer_admin_audit_repository = (
        PostgreSQLDeveloperAdminAuditRepository(
            auth_session_factory,
        )
    )

    developer_admin_unit_of_work = (
        PostgreSQLDeveloperAdminUnitOfWork(
            auth_session_factory,
            audit_repository=developer_admin_audit_repository,
        )

    ) -> tuple[AuthUserRecord, ...]:

        )

        role: str,
    ) -> bool:
        user_roles = self.roles.setdefault(
            user_id,

        )

        if role in user_roles:

        user_roles.add(role)
        return True

    async def remove_user_role(

        role: str,
    ) -> bool:
        user_roles = self.roles.setdefault(

        return True

class FakeUnitOfWork:
    def __init__(self, users=(), roles=None, *, fail_role=False, fail_audit=False):
        self.users = {u.id: u for u in users}
        self.roles = {uid: set(rs) for uid, rs in (roles or {}).items()}
        self.fail_role = fail_role
        self.fail_audit = fail_audit
        self.events = []
        self.committed = False
        self.rolled_back = False
        self._snapshot = None

    async def __aenter__(self):
        self._snapshot = {k: set(v) for k, v in self.roles.items()}
        return self

    async def __aexit__(self, exc_type, exc, traceback):
        if exc_type is None:
            self.committed = True
        else:
            self.roles = {k: set(v) for k, v in self._snapshot.items()}
            self.rolled_back = True

    async def get_user(self, user_id):
        return self.users.get(user_id)

    async def get_user_roles(self, user_id):
        return tuple(self.roles.get(user_id, set()))

    async def count_users_with_role(self, role):
        return sum(role in rs for rs in self.roles.values())

    async def add_user_role(self, *, user_id, role):
        if self.fail_role:
            raise RuntimeError("role mutation failed")
        rs = self.roles.setdefault(user_id, set())
        if role in rs:
            return False
        rs.add(role)
        return True

    async def remove_user_role(self, *, user_id, role):
        if self.fail_role:
            raise RuntimeError("role mutation failed")
        rs = self.roles.setdefault(user_id, set())
        if role not in rs:
            return False
        rs.remove(role)
        return True

    async def record_audit(self, event: DeveloperAdminAuditEvent):
        if self.fail_audit:
            raise RuntimeError("audit failed")
        self.events.append(event)
        return event

    await service.revoke_developer_admin(actor=principal(), user_id=target.id)

    assert Role.DEVELOPER_ADMIN.value not in uow.roles[target.id]
    assert len(uow.events) == 1
    assert uow.events[0].event_type == DeveloperAdminAuditService.REVOKED
    assert uow.committed is True
