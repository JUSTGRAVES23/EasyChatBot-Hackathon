export default function ForgotPasswordPage() {

  const [submitted, setSubmitted] = useState(false);
  const [error, setError] = useState<string | null>(null);

    event: FormEvent<HTMLFormElement>,

    const normalizedEmail = email.trim();

      setLoading(false);

      aria-label={

          <form
            onSubmit={handleSubmit}
            noValidate
            style={styles.form}
          >
            <label
              htmlFor="password-reset-email"
              style={styles.visuallyHidden}
            >
              Email

            <input
              id="password-reset-email"
              name="email"
              type="email"
              autoComplete="email"
              inputMode="email"
              value={email}
              onChange={(event) => {
                setEmail(event.target.value);

                if (error) {

                }
              }}
              placeholder="you@example.com"
              required
              aria-invalid={Boolean(error)}
              aria-describedby={
                error
                  ? "password-reset-error"
                  : undefined

              style={styles.emailInput}
            />

            {error && (
              <p
                id="password-reset-error"
                role="alert"
                style={styles.error}
              >
                {error}
              </p>
            )}

            <button
              type="submit"
              disabled={loading}
              style={{
                ...styles.submitButton,
                ...(loading
                  ? styles.submitButtonDisabled
                  : {}),
              }}
            >
              {loading
                ? "Sending..."
                : "Send me a reset link â†’"}
            </button>
          </form>
        )}

import {
  FormEvent,
  useEffect,
  useRef,
  useState,
} from "react";
import {
  Link,
  useSearchParams,
} from "react-router-dom";

import {
  AuthenticationApiError,
  confirmPasswordReset,
} from "../../src/auth/authApi";

type PageState =
  | "ready"
  | "submitting"
  | "success"
  | "missing"
  | "failed";

export default function ResetPasswordPage() {
  const [searchParams] = useSearchParams();

  const [token, setToken] = useState("");
  const [state, setState] =
    useState<PageState>("ready");

  const [newPassword, setNewPassword] =
    useState("");

  const [confirmPassword, setConfirmPassword] =
    useState("");

  const [error, setError] =
    useState<string | null>(null);

  const initialized = useRef(false);

  useEffect(() => {
    if (initialized.current) {
      return;
    }

    initialized.current = true;

    const rawToken =
      searchParams.get("token")?.trim() ?? "";

    if (!rawToken) {
      setState("missing");
      return;
    }

    /*
     * Capture the token once and immediately remove it
     * from the visible browser URL.
     */
    setToken(rawToken);

    window.history.replaceState(
      {},
      document.title,
      window.location.pathname,
    );
  }, [searchParams]);

  async function handleSubmit(
    event: FormEvent<HTMLFormElement>,
  ) {
    event.preventDefault();

    setError(null);

    if (!newPassword) {
      setError("Please enter a new password.");
      return;
    }

    if (newPassword !== confirmPassword) {
      setError("Passwords do not match.");
      return;
    }

    setState("submitting");

    try {
     await confirmPasswordReset({
  token,
  new_password: newPassword,
});

      /*
       * Clear credentials/token from component state after
       * successful submission.
       */
      setToken("");
      setNewPassword("");
      setConfirmPassword("");

      setState("success");
    } catch (cause) {
      setState("failed");

      setError(
        cause instanceof AuthenticationApiError
          ? cause.message
          : "This password reset link is invalid or has expired.",
      );
    }
  }

  if (state === "missing") {
    return (
      <main
        aria-label="Password reset"
        style={styles.page}
      >
        <section style={styles.card}>
          <div style={styles.eyebrow}>
            EASYCHATBOT ACADEMY
          </div>

          <h1 style={styles.title}>
            Reset link missing
          </h1>

          <p style={styles.text}>
            This password reset link does not contain
            the information we need.
          </p>

          <Link
            to="/forgot-password"
            style={styles.primaryLink}
          >
            Request a new link
          </Link>
        </section>
      </main>
    );
  }

  if (state === "success") {
    return (
      <main
        aria-label="Password updated"
        style={styles.page}
      >
        <section style={styles.card}>
          <div style={styles.eyebrow}>
            EASYCHATBOT ACADEMY
          </div>

          <h1 style={styles.title}>
            Password updated! ðŸŽ‰
          </h1>

          <p style={styles.text}>
            You're all set.
          </p>

          <p style={styles.text}>
            Your password has been changed and your
            previous sessions have been signed out
            for security.
          </p>

          <Link
            to="/login"
            style={styles.primaryLink}
          >
            Continue to sign in
          </Link>
        </section>
      </main>
    );
  }

  return (
    <main
      aria-label="Create new password"
      style={styles.page}
    >
      <section style={styles.card}>
        <div style={styles.eyebrow}>
          EASYCHATBOT ACADEMY
        </div>

        <h1 style={styles.title}>
          Create a new password
        </h1>

        <p style={styles.text}>
          Let's get you back into the Academy.
        </p>

        <form onSubmit={handleSubmit}>
          <label
            htmlFor="new-password"
            style={styles.label}
          >
            New password
          </label>

          <input
            id="new-password"
            type="password"
            autoComplete="new-password"
            minLength={12}
            maxLength={256}
            value={newPassword}
            onChange={(event) =>
              setNewPassword(event.target.value)
            }
            required
            style={styles.input}
          />

          <label
            htmlFor="confirm-password"
            style={styles.label}
          >
            Confirm new password
          </label>

          <input
            id="confirm-password"
            type="password"
            autoComplete="new-password"
            minLength={12}
            maxLength={256}
            value={confirmPassword}
            onChange={(event) =>
              setConfirmPassword(event.target.value)
            }
            required
            style={styles.input}
          />

          {error && (
            <p
              role="alert"
              style={styles.error}
            >
              {error}
            </p>
          )}

          <p style={styles.requirement}>
            Your password must contain at least
            12 characters.
          </p>

          <button
            type="submit"
            disabled={
              state === "submitting" ||
              !token
            }
            style={styles.button}
          >
            {state === "submitting"
              ? "Updating..."
              : "Set my new password â†’"}

        <Link
          to="/login"
          style={styles.secondaryLink}

          Back to sign in

}

===== frontend/app/login/page.tsx =====
import {
  useState,
  type FormEvent,
} from "react";

import {
  useNavigate,
  useSearchParams,
} from "react-router-dom";

import { useAuth } from "../../src/auth/useAuth";
import { AuthenticationApiError } from "../../src/auth/authApi";

export default function LoginPage() {
  const navigate = useNavigate();
  const [searchParams] = useSearchParams();

  const { login } = useAuth();

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const [error, setError] =
    useState<string | null>(null);

  const [loading, setLoading] =
    useState(false);

  async function handleSubmit(
    event: FormEvent<HTMLFormElement>,
  ) {
    event.preventDefault();

    setLoading(true);
    setError(null);

    try {
      await login({
        email: email.trim(),
        password,
      });

      const requestedRedirect =
        searchParams.get("redirect");

      let redirect = "/project-control";

      if (requestedRedirect) {
        try {
          const decoded =
            decodeURIComponent(requestedRedirect);

          /*
           * Only permit internal application paths.
           *
           * This prevents the login redirect from becoming
           * an open redirect.
           */
          if (
            decoded.startsWith("/") &&
            !decoded.startsWith("//")
          ) {
            redirect = decoded;
          }
        } catch {
          /*
           * Keep the safe default.
           */
        }
      }

      navigate(
        redirect,
        {
          replace: true,
        },
      );
    } catch (err) {
      setError(
        err instanceof AuthenticationApiError
          ? err.message
          : "Unable to sign in.",
      );
    } finally {
      setLoading(false);
    }
  }

  return (
    <main
      style={{
        minHeight: "100vh",
        display: "grid",
        placeItems: "center",
        padding: "2rem",
      }}
    >
      <section
        aria-labelledby="login-title"
        style={{
          width: "100%",
          maxWidth: "420px",
        }}
      >
        <h1 id="login-title">
          EasyChatBot
        </h1>

        <p>
          Authorized access required.
        </p>

        <form onSubmit={handleSubmit}>
          <label>
            Email

            <input
              type="email"
              autoComplete="username"
              value={email}
              onChange={(event) =>
                setEmail(event.target.value)
              }
              required
            />
          </label>

          <label>
            Password

            <input
              type="password"
              autoComplete="current-password"
              value={password}
              onChange={(event) =>
                setPassword(event.target.value)
              }
              required
            />
          </label>

          <p>
            <a href="/forgot-password">
              Forgot your password?
            </a>
          </p>

          {error && (
            <p role="alert">
              {error}
            </p>
          )}

          <button
            type="submit"
            disabled={loading}
          >
            {loading
              ? "Signing in…"
              : "Sign in"}
          </button>
        </form>
      </section>
    </main>
  );
}

  detail: string;

export class AuthenticationApiError extends Error {
  readonly status: number;

  constructor(
    message: string,
    status: number,
  ) {
    super(message);
    this.name = "AuthenticationApiError";
    this.status = status;
  }
}

async function parseApiError(
  response: Response,
  fallback: string,
): Promise<AuthenticationApiError> {
  let detail = fallback;

  try {
    const body: unknown = await response.json();

    if (
      body &&
      typeof body === "object" &&
      "detail" in body &&
      typeof body.detail === "string" &&
      body.detail.trim()
    ) {
      detail = body.detail;
    }
  } catch {
    // Preserve safe generic API error.
  }

  return new AuthenticationApiError(
    detail,
    response.status,
  );
}

export async function getSession(): Promise<SessionResponse> {
  const response = await fetch("/auth/session", {
    method: "GET",
    credentials: "include",
    headers: {
      Accept: "application/json",
    },
  });

  if (!response.ok) {
    throw await parseApiError(
      response,
      `Authentication session check failed (${response.status}).`,
    );
  }

  return response.json() as Promise<SessionResponse>;
}

export async function register(
  credentials: RegisterRequest,
): Promise<RegisterResponse> {
  const response = await fetch("/auth/register", {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
      Accept: "application/json",
    },
    body: JSON.stringify(credentials),
  });

  if (!response.ok) {
    throw await parseApiError(
      response,
      "Unable to create the account. Please try again later.",
    );
  }

  return response.json() as Promise<RegisterResponse>;
}

export async function verifyEmail(
  token: string,
): Promise<VerifyEmailResponse> {
  const response = await fetch(
    `/auth/verify-email?token=${encodeURIComponent(token)}`,
    {
      method: "POST",
      credentials: "include",
      headers: {
        Accept: "application/json",
      },
    },
  );

  if (!response.ok) {
    throw await parseApiError(
      response,
      "This verification link is invalid or has expired.",
    );
  }

  return response.json() as Promise<VerifyEmailResponse>;
}

export async function requestPasswordReset(
  request: PasswordResetRequest,
): Promise<PasswordResetRequestResponse> {
  const response = await fetch(
    "/auth/password-reset/request",
    {
      method: "POST",
      credentials: "include",
      headers: {
        "Content-Type": "application/json",
        Accept: "application/json",
      },
      body: JSON.stringify(request),
    },
  );

  if (!response.ok) {
    throw await parseApiError(
      response,
      "Unable to request a password reset. Please try again later.",
    );
  }

  return response.json() as Promise<PasswordResetRequestResponse>;
}

export async function confirmPasswordReset(
  request: PasswordResetConfirmRequest,
): Promise<PasswordResetConfirmResponse> {
  const response = await fetch(
    "/auth/password-reset/confirm",
    {
      method: "POST",
      credentials: "include",
      headers: {
        "Content-Type": "application/json",
        Accept: "application/json",
      },
      body: JSON.stringify(request),
    },
  );

  if (!response.ok) {
    throw await parseApiError(
      response,
      "This password reset link is invalid or has expired.",
    );
  }

  return response.json() as Promise<PasswordResetConfirmResponse>;
}

export async function login(
  credentials: LoginRequest,
): Promise<LoginResponse> {
  const response = await fetch("/auth/login", {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
      Accept: "application/json",
    },
    body: JSON.stringify(credentials),
  });

  if (!response.ok) {
    throw await parseApiError(
      response,
      `Authentication failed (${response.status}).`,
    );
  }

  return response.json() as Promise<LoginResponse>;
}

export async function logout(): Promise<void> {
  const response = await fetch("/auth/logout", {
    method: "POST",
    credentials: "include",
  });

  if (!response.ok && response.status !== 204) {
    throw new AuthenticationApiError(
      `Logout failed (${response.status}).`,
      response.status,
    );
  }
}
