async def get_session(
    principal: Principal = Depends(get_current_principal),
) -> SessionResponse:
    """
    Return the current authentication state.

    Authentication is resolved exclusively through the existing
    get_current_principal() dependency and SessionIdentityProvider.

    It does not expose the session identifier, credentials, or other

        authenticated=principal.is_active,

@router.post(
    "/register",
    response_model=RegisterResponse,
    status_code=status.HTTP_201_CREATED,
)
async def register(
    request: RegisterRequest,
    http_request: Request,
) -> RegisterResponse:
    """
    Register a new account through the composed authentication service.

    Registration creates the account and verification token and triggers
    verification-email delivery through the configured application/provider
    boundary. The route itself does not select or call an email provider.
    """
    service = http_request.app.state.authentication_service

    try:
        user_id = await service.register(
            email=str(request.email),
            username=request.username,
            password=request.password,
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(exc),
        ) from None

    return RegisterResponse(
        user_id=user_id,
        detail=(
            "Registration successful. "
            "Email verification is required."
        ),
    )


@router.post(
    "/login",
    response_model=LoginResponse,
)
async def login(
    request: LoginRequest,
    http_request: Request,
    response: Response,
) -> LoginResponse:
    """
    Authenticate through the composed authentication service.

    The service enforces account status and email verification before a
    server-side session is created.
    """
    service = http_request.app.state.authentication_service

    try:
        result = await service.authenticate(
            email=str(request.email),
            password=request.password,
            device_name=http_request.headers.get("user-agent"),
            ip_address=(
                http_request.client.host
                if http_request.client
                else None
            ),
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=str(exc),
            headers={
                "WWW-Authenticate": "Session",
            },
        ) from None

    expires_in_seconds = max(
        0,
        int(
            (
                result.expires_at
                - datetime.now(timezone.utc)
            ).total_seconds()
        ),
    )

    response.set_cookie(
        key="easychatbot_session",
        value=str(result.session_id),
        httponly=True,
        secure=True,
        samesite="lax",
        max_age=expires_in_seconds,
        path="/",
    )

    return LoginResponse(
        detail="Login successful.",
        expires_at=result.expires_at.isoformat(),
    )

@router.post(
    "/logout",
    status_code=status.HTTP_204_NO_CONTENT,
)
async def logout(
    http_request: Request,
    response: Response,
) -> None:
    """
    Revoke the current server-side authentication session.

    No authentication service is constructed here; the application's
    composed authentication service is reused.
    """
    session_id = http_request.cookies.get(
        "easychatbot_session"
    )

    if session_id:
        service = http_request.app.state.authentication_service

        try:
            parsed_session_id = UUID(session_id)
        except ValueError:
            parsed_session_id = None

        if parsed_session_id is not None:
            await service.revoke_session(
                parsed_session_id
            )

    response.delete_cookie(
        key="easychatbot_session",
        path="/",
    )

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta, timezone

import hashlib
import hmac

class AuthenticationError(ValueError):
    """Expected authentication failure."""


@dataclass(frozen=True, slots=True)
class SessionResult:
    session_id: UUID
    user_id: UUID
    expires_at: datetime


class PasswordHasher:
    """PBKDF2 password hashing using the Python standard library."""

    algorithm = "pbkdf2_sha256"
    iterations = 600_000
    salt_bytes = 32

    def hash(self, password: str) -> str:
        if len(password) < 12:
            raise AuthenticationError(
                "Password must contain at least 12 characters."
            )

        salt = secrets.token_bytes(self.salt_bytes)
        digest = hashlib.pbkdf2_hmac(
            "sha256",
            password.encode("utf-8"),
            salt,
            self.iterations,
        )
        return (
            f"{self.algorithm}${self.iterations}$"
            f"{salt.hex()}${digest.hex()}"
        )

    def verify(self, password: str, encoded: str) -> bool:
        try:
            algorithm, iterations, salt_hex, digest_hex = encoded.split("$")

            if algorithm != self.algorithm:
                return False

            candidate = hashlib.pbkdf2_hmac(
                "sha256",
                password.encode("utf-8"),
                bytes.fromhex(salt_hex),
                int(iterations),
            )

            return hmac.compare_digest(
                candidate.hex(),
                digest_hex,
            )
        except (ValueError, TypeError):
            return False


class AuthenticationService:
    """Application service for registration/login/session lifecycle.

    Persistence is supplied through a repository protocol so the service does
    not depend on a particular ORM implementation.

    Authentication email delivery is supplied through the application email
    service boundary. Provider selection and provider credentials remain
    outside this service.
    """

    def __init__(
        self,
        repository,
        *,
        email_service=None,
        password_hasher: PasswordHasher | None = None,
        session_ttl: timedelta = timedelta(hours=12),
    ) -> None:
        self._repository = repository
        self._email_service = email_service
        self._password_hasher = password_hasher or PasswordHasher()
        self._session_ttl = session_ttl

    async def register(
        self,
        *,
        email: str,
        username: str,
        password: str,
    ):
        normalized_email = email.strip().lower()
        normalized_username = username.strip()

        if not normalized_email or "@" not in normalized_email:
            raise AuthenticationError("A valid email is required.")

        if not normalized_username:
            raise AuthenticationError("Username is required.")

        if await self._repository.email_exists(normalized_email):
            raise AuthenticationError(
                "An account with this email already exists."
            )

        if await self._repository.username_exists(normalized_username):
            raise AuthenticationError(
                "That username is already in use."
            )

        user_id = uuid4()
        password_hash = self._password_hasher.hash(password)

        await self._repository.create_user(
            user_id=user_id,
            email=normalized_email,
            username=normalized_username,
        )

        await self._repository.create_password_credential(
            user_id=user_id,
            password_hash=password_hash,
        )

        verification_token = (
            await self._repository.create_email_verification(
                user_id=user_id,
                email=normalized_email,
            )
        )

        if self._email_service is None:
            raise AuthenticationError(
                "Registration email delivery is not configured."
            )

        try:
            await self._email_service.send_verification_email(
                email=normalized_email,
                username=normalized_username,
                verification_token=verification_token,
            )
        except Exception:
            raise AuthenticationError(
                "Registration email could not be sent. "
                "Please try again later."
            ) from None

        return user_id

    async def authenticate(
        self,
        *,
        email: str,
        password: str,
        device_name: str | None = None,
        ip_address: str | None = None,
    ) -> SessionResult:
        user = await self._repository.get_user_by_email(
            email.strip().lower()
        )

        if user is None:
            raise AuthenticationError(
                "Invalid email or password."
            )

        if user.status != "active":
            raise AuthenticationError(
                "Account is not active."
            )

        if not user.email_verified:
            raise AuthenticationError(
                "Email verification is required."
            )

        credential = await self._repository.get_password_credential(
            user.id
        )

        if credential is None or not self._password_hasher.verify(
            password,
            credential.password_hash,
        ):
            raise AuthenticationError(
                "Invalid email or password."
            )

        now = datetime.now(timezone.utc)
        expires_at = now + self._session_ttl
        session_id = uuid4()

        await self._repository.create_session(
            session_id=session_id,
            user_id=user.id,
            created_at=now,
            expires_at=expires_at,
            device_name=device_name,
            ip_address=ip_address,
        )

        return SessionResult(
            session_id=session_id,
            user_id=user.id,
            expires_at=expires_at,
        )

    async def verify_email(
        self,
        token: str,
    ) -> None:
        """Verify an email address using a repository-managed token.

        Token generation, hashing, expiration, single-use enforcement,
        and persistence remain the responsibility of the repository.
        The application service exposes the business operation without
        depending on the persistence implementation.
        """

        normalized_token = token.strip()

        if not normalized_token:
            raise AuthenticationError(
                "Email verification token is required."
            )

        verified = await self._repository.verify_email(
            normalized_token,
        )

        if not verified:
            raise AuthenticationError(
                "Invalid or expired email verification token."
            )

    async def revoke_session(
        self,
        session_id: UUID,
    ) -> None:
        await self._repository.revoke_session(session_id)

def _registration_email_sender() -> str:
    """Return the dedicated EasyChatBot registration sender."""
    repository_root = Path(__file__).resolve().parents[4]
    env_file = repository_root / ".env"

    configuration = dotenv_values(env_file)

    sender = configuration.get(
        "EASYCHATBOT_REGISTRATION_EMAIL"
    )

    if not isinstance(sender, str) or not sender.strip():
        raise RuntimeError(
            "EASYCHATBOT_REGISTRATION_EMAIL is not configured."
        )

    return sender.strip()

    async def create_email_verification(self, *, user_id: UUID, email: str) -> str:
        """Create a one-time verification token.

        The raw token is returned to the application boundary. Only its SHA-256
        digest is persisted. Delivery is deliberately outside this repository.
        """
        import hashlib
        import secrets

        raw_token = secrets.token_urlsafe(32)
        token_hash = hashlib.sha256(raw_token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)
        verification_id = __import__("uuid").uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_email_verification_tokens
                            (id, user_id, email, token_hash, expires_at, created_at)
                        VALUES
                            (:id, :user_id, :email, :token_hash,
                             :expires_at, :created_at)
                        """
                    ),
                    {
                        "id": verification_id,
                        "user_id": user_id,
                        "email": email,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=24),
                        "created_at": now,
                    },
                )
        return raw_token

    async def verify_email(self, token: str) -> bool:
        import hashlib

        token_hash = hashlib.sha256(token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    text(
                        """
                        SELECT id, user_id
                        FROM auth_email_verification_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now
                        FOR UPDATE
                        """
                    ),
                    {"token_hash": token_hash, "now": now},
                )
                row = result.mappings().first()
                if row is None:
                    return False

                await session.execute(
                    text(
                        "UPDATE users SET email_verified = TRUE WHERE id = :user_id"
                    ),
                    {"user_id": row["user_id"]},
                )
                await session.execute(
                    text(
                        "UPDATE auth_email_verification_tokens SET used_at = :now WHERE id = :id"
                    ),
                    {"now": now, "id": row["id"]},
                )
                return True

class AuthenticationEmailSenderAdapter:
    """Authentication email adapter over the shared provider boundary.

    sender contract. Provider selection and credentials remain owned by the

    - make authentication or verification decisions.

            provider_manager

            else EmailProviderManager()
        )

        self,

    ) -> dict[str, Any]:

        if not isinstance(subject, str) or not subject.strip():
            raise ValueError("Email subject cannot be empty.")

        if not isinstance(content, dict):

            content,

            raise TypeError("Email provider returned an invalid result.")

"""
Email Provider Manager.

Controls provider selection, configuration, availability,
and controlled email delivery.

The manager is deliberately provider-independent.
Concrete providers such as Resend are injected into this
boundary rather than called directly by application services.

Does NOT:
- generate cybersecurity reports;
- schedule emails;
- analyse security;
- execute remediation;
- expose provider credentials.
"""

from __future__ import annotations

from typing import Any

from cybersecurity.email.resend_email_provider import (
    ResendEmailProvider,
)


class EmailProviderManager:
    """
    Provider abstraction for controlled email delivery.
    """

    def __init__(
        self,
        provider: Any | None = None,
    ) -> None:

        self.provider = (
            provider
            if provider is not None
            else self._build_default_provider()
        )

    @staticmethod
    def _build_default_provider() -> Any | None:
        """
        Build the configured default email provider.

        Resend is currently the approved hackathon provider.

        If Resend is not configured, return None so the
        application can report provider_not_configured
        rather than failing during application import.
        """

        try:
            return ResendEmailProvider()

        except ValueError:
            return None

    def status(self) -> dict[str, Any]:
        """
        Return non-sensitive provider status.

        This method never exposes:
        - API keys;
        - secrets;
        - credentials.

        Status is determined from the provider already
        configured in the manager. No external provider
        request is made.
        """

        if self.provider is None:

            return {
                "provider": None,
                "configured": False,
                "available": False,
                "status": "not_configured",
            }

        provider_id = getattr(
            self.provider,
            "provider_id",
            self.provider.__class__.__name__,
        )

        return {
            "provider": provider_id,
            "configured": True,
            "available": True,
            "status": "ready",
        }

    async def send(
        self,
        subject: str,
        content: dict,
    ) -> dict:

        if self.provider is None:

            return {
                "status": "provider_not_configured",
            }

        return await self.provider.send(
            subject,
            content,
        )

==================== RESEND PROVIDER ========================

"""
Resend email provider.

Infrastructure adapter responsible only for email delivery.

The provider supports multiple controlled email contracts:

- Cybersecurity notifications:
  rendered by the existing cybersecurity HTML renderer.

- Privacy Governance reminders:
  the application service supplies the authoritative HTML and
  plain-text content. The provider delivers that content without
  replacing it with the cybersecurity template.

- Privacy Governance email threading:
  optional In-Reply-To and References headers may be supplied
  by the application layer when a notification is replying to
  an existing governance email thread.

The provider does not:
- make governance decisions;
- modify governance state;
- execute remediation;
- generate application decisions;
- access provider credentials outside configuration;
- decide which governance emails belong to a thread.
"""

from __future__ import annotations

import base64
from pathlib import Path
from typing import Any

import resend

from cybersecurity.email.email_configuration import (
    EMAIL_FROM,
    RESEND_API_KEY,
)


class ResendEmailProvider:
    """
    Provider adapter for Resend email delivery.

    Routing rule:

        privacy_governance_reminder
            -> supplied html_body

        all other email types
            -> existing cybersecurity renderer

    Email threading rule:

        in_reply_to
            -> In-Reply-To header

        references
            -> References header

    Threading metadata is supplied by the application workflow.
    This provider only translates that metadata into the provider
    request.
    """

    provider_id = "resend"

    def __init__(
        self,
        *,
        api_key: str | None = None,
        sender: str | None = None,
    ) -> None:
        self.api_key = api_key or RESEND_API_KEY
        self.sender = sender or EMAIL_FROM

        if not self.api_key:
            raise ValueError(
                "RESEND_API_KEY is not configured."
            )

        if not self.sender:
            raise ValueError(
                "Email sender is not configured."
            )

        resend.api_key = self.api_key

    async def send(
        self,
        subject: str,
        content: dict[str, Any],
    ) -> dict[str, Any]:
        """Send an email through Resend."""

        if not isinstance(subject, str) or not subject.strip():
            raise ValueError(
                "Email subject cannot be empty."
            )

        if not isinstance(content, dict):
            raise TypeError(
                "Email content must be a dictionary."
            )

        recipient = content.get("recipient")

        if not isinstance(recipient, str) or not recipient.strip():
            raise ValueError(
                "Email recipient is required."
            )

        recipient = recipient.strip()

        email_type = content.get(
            "email_type",
            "",
        )

        # --------------------------------------------------------
        # Presentation
        # --------------------------------------------------------

        if email_type in {
            "authentication_verification",
            "academy_notification",
        }:
            supplied_html = content.get("html")

            if (
                not isinstance(supplied_html, str)
                or not supplied_html.strip()
            ):
                raise ValueError(
                    f"{email_type} requires "
                    "html to be a non-empty string."
                )

            html = supplied_html

        elif email_type == "privacy_governance_reminder":
            html = self._get_supplied_html_body(content)

        else:
            # Preserve existing Cybersecurity behaviour.
            html = self._build_html(content)

        text_body = content.get("body")

        if text_body is not None:
            if not isinstance(text_body, str):
                raise TypeError(
                    "body must be a string when provided."
                )

            if not text_body.strip():
                raise ValueError(
                    "body cannot be empty when provided."
                )

        # --------------------------------------------------------
        # Base Resend request
        # --------------------------------------------------------

        sender = self.sender

        if email_type in {
            "authentication_verification",
            "academy_notification",
        }:
            supplied_sender = content.get("from")

            if (
                not isinstance(supplied_sender, str)
                or not supplied_sender.strip()
            ):
                raise ValueError(
                    f"{email_type} sender is required."
                )

            sender = supplied_sender.strip()

        params: dict[str, Any] = {
            "from": sender,
            "to": [recipient],
            "subject": subject.strip(),
            "html": html,
        }

        if isinstance(text_body, str) and text_body.strip():
            params["text"] = text_body

        # --------------------------------------------------------
        # Optional email threading
        # --------------------------------------------------------

        in_reply_to = self._normalize_header_value(
            content.get("in_reply_to"),
            field_name="in_reply_to",
        )

        references = self._normalize_header_value(
            content.get("references"),
            field_name="references",
        )

        headers: dict[str, str] = {}

        if in_reply_to:
            headers["In-Reply-To"] = in_reply_to

        if references:
            headers["References"] = references

        if headers:
            params["headers"] = headers

        # --------------------------------------------------------
        # Attachments
        # --------------------------------------------------------

        attachments = content.get("attachments")

        if attachments is not None:
            params["attachments"] = self._build_attachments(
                attachments
            )

        # --------------------------------------------------------
        # Provider delivery
        # --------------------------------------------------------

        response = await resend.Emails.send_async(
            params
        )

        if isinstance(response, dict):
            message_id = response.get("id")
        else:
            message_id = getattr(
                response,
                "id",
                None,
            )

        return {
            "provider": self.provider_id,
            "status": "accepted",
            "message_id": message_id,
        }

    @staticmethod
    def _normalize_header_value(
        value: Any,
        *,
        field_name: str,
    ) -> str | None:
        """
        Validate and normalize optional email-header metadata.
        """

        if value is None:
            return None

        if not isinstance(value, str):
            raise TypeError(
                f"{field_name} must be a string when provided."
            )

        normalized = value.strip()

        if not normalized:
            return None

        return normalized

    @staticmethod
    def _get_supplied_html_body(
        content: dict[str, Any],
    ) -> str:
        """
        Return application-owned Privacy Governance HTML.
        """

        html_body = content.get("html_body")

        if not isinstance(html_body, str):
            raise TypeError(
                "Privacy Governance email requires "
                "html_body to be a string."
            )

        if not html_body.strip():
            raise ValueError(
                "Privacy Governance email html_body cannot be empty."
            )

        return html_body

    @staticmethod
    def _build_attachments(
        attachments: Any,
    ) -> list[dict[str, Any]]:
        """
        Convert local application attachment metadata into the
        Resend attachment representation.
        """

        if not isinstance(attachments, list):
            raise TypeError(
                "attachments must be a list when provided."
            )

        normalized: list[dict[str, Any]] = []

        for attachment in attachments:
            if not isinstance(attachment, dict):
                raise TypeError(
                    "Each email attachment must be a dictionary."
                )

            path_value = attachment.get("path")

            if not path_value:
                raise ValueError(
                    "Email attachment path is required."
                )

            path = Path(str(path_value))

            if not path.exists():
                raise FileNotFoundError(
                    f"Email attachment not found: {path}"
                )

            if not path.is_file():
                raise ValueError(
                    f"Email attachment is not a file: {path}"
                )

            filename = attachment.get("filename")

            resolved_filename = (
                str(filename).strip()
                if filename
                else path.name
            )

            if not resolved_filename:
                raise ValueError(
                    "Email attachment filename is required."
                )

            content_id = attachment.get("content_id")

            inline = bool(
                attachment.get(
                    "inline",
                    False,
                )
            )

            if inline and not content_id:
                raise ValueError(
                    "Inline email attachments require "
                    "content_id."
                )

            encoded_content = base64.b64encode(
                path.read_bytes()
            ).decode("ascii")

            resend_attachment: dict[str, Any] = {
                "filename": resolved_filename,
                "content": encoded_content,
            }

            if content_id:
                resend_attachment["content_id"] = str(
                    content_id
                )

            normalized.append(
                resend_attachment
            )

        return normalized

    @staticmethod
    def _build_html(
        content: dict[str, Any],
    ) -> str:
        """
        Existing Cybersecurity HTML renderer.

        This remains the Cybersecurity fallback and is intentionally
        not used for Privacy Governance reminders.
        """

        posture = str(
            content.get(
                "posture",
                "Security notification",
            )
        )

        critical_count = content.get(
            "critical_count",
            0,
        )

        high_count = content.get(
            "high_count",
            0,
        )

        priority_signals = content.get(
            "priority_signals",
            [],
        )

        signal_rows = ""

        if isinstance(
            priority_signals,
            list,
        ):
            for signal in priority_signals:
                if not isinstance(
                    signal,
                    dict,
                ):
                    continue

                title = str(
                    signal.get(
                        "title",
                        "Security signal",
                    )
                )

                severity = str(
                    signal.get(
                        "severity",
                        "",
                    )
                ).upper()

                signal_rows += (
                    "<li>"
                    f"<strong>{title}</strong>"
                    f" — {severity}"
                    "</li>"
                )

        return f"""
<!DOCTYPE html>
<html>
<head>
    <meta charset="utf-8">
    <title>EasyChatBot Cybersecurity</title>
</head>
<body>
    <h2>EasyChatBot Cybersecurity Notification</h2>

    <p>
        <strong>Security posture:</strong>
        {posture}
    </p>

    <p>
        <strong>Critical findings:</strong>
        {critical_count}
    </p>

    <p>
        <strong>High findings:</strong>
        {high_count}
    </p>

    <h3>Priority Security Signals</h3>

    <ul>
        {signal_rows}
    </ul>

    <hr>

    <p>
        This notification is decision support only.
        No remediation action was executed by EasyChatBot.
    </p>

    <p>
        <strong>Environment:</strong> DEMO
    </p>
</body>
</html>
"""


__all__ = [
    "ResendEmailProvider",

def _required_environment(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for authentication email composition."
        )

    return value


def _create_announcement_components():
    telegram_client = TelegramAnnouncementClient()

    announcement_publisher = TelegramAnnouncementPublisher(
        telegram_client=telegram_client,
    )

    return (
        telegram_client,
        announcement_publisher,
    )


@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )
    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )


def create_application() -> FastAPI:
    academy_config = (
        AcademyActivationConfig.from_environment()
    )

    auth_engine = create_async_database_engine()

    auth_session_factory = (
        create_async_session_factory(
            auth_engine,
        )
    )

    auth_repository = (
        PostgreSQLAuthenticationRepository(
            auth_session_factory,
        )
    )

    email_provider_manager = EmailProviderManager()

    authentication_email_sender = AuthenticationEmailSenderAdapter(
        provider_manager=email_provider_manager,
    )

    authentication_email_service = AuthenticationEmailService(
        authentication_email_sender,
        frontend_base_url=_required_environment(
            "EASYCHATBOT_PORTAL_URL"
        ),
    )

    authentication_service = AuthenticationService(
        auth_repository,
        email_service=authentication_email_service,
    )

    # Provider-usage repository is synchronous by
    # contract. It therefore receives a synchronous
    # engine/session factory.
    #
    # It must not receive the async authentication
    # session factory.
    provider_usage_engine = (
        create_database_engine()
    )

    persistence_diagnostics_service = PersistenceDiagnosticsService(
        metadata=Base.metadata,
        project_root=Path(__file__).resolve().parents[1],
        engine=provider_usage_engine,
    )

    provider_usage_session_factory = (
        create_session_factory(
            provider_usage_engine,
        )
    )

    provider_usage_repository = (
        PostgreSQLProviderUsageRepository(
            provider_usage_session_factory,
        )
    )

    provider_usage_dashboard_service = (
        ProviderUsageDashboardService(
            repository=provider_usage_repository,
        )
    )

    functionality_control_repository = (
        PostgreSQLFunctionalityControlRepository(
            provider_usage_session_factory,
        )
    )

    functionality_control_authorizer = (
        AuthorizationFunctionalityControlAuthorizer()
    )

    functionality_control_service = (
        FunctionalityOperationalControlService(
            authorizer=functionality_control_authorizer,
            repository=functionality_control_repository,
        )
    )

    # ------------------------------------------------------------------
    # Project Control / operational health composition
    #
    # Project Control is read-only. It consumes application-owned health
    # and operational services and owns only its own PCD registry and
    # snapshot repository.
    # ------------------------------------------------------------------

    production_health_configuration = (
        ProductionHealthConfiguration(
            database_engine=provider_usage_engine,
            telegram_bot_factory=None,
            llm_provider=None,
            llm_request_get=None,
        )
    )

    production_health_service = (
        ProductionHealthService(
            configuration=production_health_configuration,
        )
    )

    production_readiness_service = (
        production_health_service._readiness_service
    )

    operational_health_snapshot_service = (
        OperationalHealthSnapshotService(
            readiness_service=production_readiness_service,
        )
    )

    operational_event_service = (
        OperationalEventService()
    )

    operational_dashboard_service = (
        OperationalDashboardService(
            health_snapshot_service=operational_health_snapshot_service,
            event_service=operational_event_service,
        )
    )

    privacy_governance_composition = (
        get_privacy_governance_composition()
    )

    project_registry = (
        InMemoryProjectRegistry()
    )

    project_snapshot_repository = (
        InMemorySnapshotRepository()
    )

    project_control_service = (
        create_project_control_service(
            registry=project_registry,
            snapshots=project_snapshot_repository,
            health_service=production_health_service,
            dashboard_service=operational_dashboard_service,
            case_registry=(
                privacy_governance_composition.case_registry
            ),
        )
    )

    academy_one_week_workflow = None
    announcement_telegram_client = None

    if academy_config.activation_allowed:
        academy_session_factory = (
            auth_session_factory
        )

        (
            announcement_telegram_client,
            _announcement_publisher,
        ) = _create_announcement_components()

        academy_one_week_workflow = (
            create_academy_one_week_workflow(
                session_factory=academy_session_factory,
                telegram_bot=announcement_telegram_client,
            )
        )

    scheduler_runtime = (
        create_cybersecurity_scheduler_runtime(
            academy_one_week_workflow=academy_one_week_workflow,
        )
    )

    application = FastAPI(
        title="EasyChatBot API",
        description="AI character learning platform API",
        version="0.1.0",
        lifespan=lifespan,
    )

    application.state.scheduler_runtime = (
        scheduler_runtime
    )

    application.state.auth_repository = (
        auth_repository
    )

    application.state.authentication_service = (
        authentication_service
    )

    application.state.auth_engine = (
        auth_engine
    )

    application.state.provider_usage_engine = (
        provider_usage_engine
    )

    application.state.provider_usage_dashboard_service = (
        provider_usage_dashboard_service
    )

    application.state.functionality_control_service = (
        functionality_control_service
    )

    application.state.project_control_service = (
        project_control_service
    )

    application.state.production_health_service = (
        production_health_service
    )

    application.state.operational_dashboard_service = (
        operational_dashboard_service
    )

    application.state.announcement_telegram_client = (
        announcement_telegram_client
    )

    application.add_middleware(
        CORSMiddleware,
        allow_origins=[
            "http://localhost:3000",
            "http://127.0.0.1:3000",
            "http://localhost:5173",
            "http://127.0.0.1:5173",
        ],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    application.include_router(
        authentication_router,
    )

    application.include_router(
        voice_calls_router,
    )

    application.include_router(
        voice_reconnect_router,
    )

    application.include_router(
        voice_call_trigger_router,
    )

    application.include_router(
        voice_notification_acceptance_router,
    )

    operations_center_composition = (
        create_operations_center_composition(
            provider_usage_dashboard_service=(
                provider_usage_dashboard_service
            ),
            functionality_control_service=(
                functionality_control_service
            ),
        )
    )

    application.include_router(
        create_operations_center_router(
            composition=operations_center_composition,
        ),
    )

    application.include_router(
        create_persistence_operations_router(
            persistence_diagnostics_service,
            operator_dependency=require_permission(
                Permission.INTERNAL_DATA_READ,
            ),
        )
    )

    application.include_router(
        create_project_control_router(
            project_control=project_control_service,
        ),
    )

    application.include_router(
        create_email_workflows_router(),
    )

    application.include_router(
        privacy_governance_router,
    )

    processing_registry = (
        create_beta1_processing_registry()
    )

    screening_service = ScreeningService()

    screening_adapter = (
        ProcessingActivityScreeningAdapter(
            screening_service,
        )
    )

    application.include_router(
        create_privacy_screening_router(
            processing_registry=processing_registry,
            screening_adapter=screening_adapter,
        ),
    )

    application.include_router(
        provider_management_controls_router,
    )

    application.include_router(
        governance_audit_router,
    )

    application.include_router(
        governance_operations_router,
    )

    application.include_router(
        governance_evidence_router,
    )

    application.include_router(
        governance_change_lifecycle_router,
    )

    application.include_router(
        cybersecurity_intelligence_router,
    )

    application.include_router(
        cybersecurity_decision_support_router,
    )

    application.include_router(
        cybersecurity_reports_router,
    )

    application.include_router(
        cybersecurity_live_reports_router,
    )

    # Academy World ? Static Questionnaires
    #
    # This exposes:
    # GET /api/academy/worlds/{world_id}/questionnaires
    #
    # Questionnaire content remains controlled/static.
    # Recommendation metadata is handled separately.
    application.include_router(
        academy_questionnaires_router,
    )

    return application


app = create_application()
