class InMemoryExecutionRepository:
    """Minimal durable repository double for the integration test."""

    def __init__(self) -> None:
        self.records = {}

    async def get_by_idempotency_key(self, idempotency_key: str):
        return self.records.get(idempotency_key)

    async def create_started(
        self,
        *,
        publication_id,
        idempotency_key,
        correlation_id,
    ):
        from datetime import datetime, timezone

        from app.application.announcements.execution.durable_contracts import (
            DurableExecutionStatus,
            PublicationExecutionRecord,
        )

        existing = self.records.get(idempotency_key)

        if existing is not None:
            return existing

        now = datetime.now(timezone.utc)

        record = PublicationExecutionRecord(
            publication_id=publication_id,
            idempotency_key=idempotency_key,
            correlation_id=correlation_id,
            attempt=1,
            status=DurableExecutionStatus.STARTED,
            provider_message_id=None,
            error_code=None,
            created_at=now,
            updated_at=now,
        )

        self.records[idempotency_key] = record
        return record

    async def mark_succeeded(
        self,
        *,
        idempotency_key,
        provider_message_id,
    ):
        from dataclasses import replace
        from datetime import datetime, timezone

        from app.application.announcements.execution.durable_contracts import (
            DurableExecutionStatus,
        )

        record = self.records[idempotency_key]

        updated = replace(
            record,
            status=DurableExecutionStatus.SUCCEEDED,
            provider_message_id=provider_message_id,
            error_code=None,
            updated_at=datetime.now(timezone.utc),
        )

        self.records[idempotency_key] = updated
        return updated

    async def mark_failed(
        self,
        *,
        idempotency_key,
        error_code,
    ):
        from dataclasses import replace
        from datetime import datetime, timezone

        from app.application.announcements.execution.durable_contracts import (
            DurableExecutionStatus,

        record = self.records[idempotency_key]

        updated = replace(

        )

        self.records[idempotency_key] = updated
        return updated

        self,
        *,
        conversation_boundary,
    ) -> None:
        self._conversation_boundary = conversation_boundary

    async def handle_start(
        self,
        update,
        context,
    ):
        return None

    async def handle_message(
        self,
        update,
        context,
    ):
        user = update.effective_user
        message = update.effective_message

        if user is None:
            return None

        if message is None:
            return None

        if not message.text:
            return None

        response = await (
            self._conversation_boundary.handle_message(
                telegram_user_id=user.id,
                message=message.text,
            )
        )

        # The real BetaTelegramAdapter owns Telegram response
        # delivery. The fake must model the same contract.
        if (
            isinstance(response, str)
            and response.strip()
        ):
            await message.reply_text(response)

        return response
