===== DOMAIN: VOICE CONNECTION TOKEN =====
"""
Voice Connection Token Domain Model.

Represents provider-independent credentials
required to connect to a voice room.

Responsibilities:
- Preserve connection token identity.
- Preserve room and provider information.
- Validate token expiration.
- Support both legacy naive UTC datetimes
  and timezone-aware UTC datetimes.

Does NOT:
- Generate provider tokens.
- Connect to LiveKit.
- Handle HTTP.
- Persist credentials.
"""


from __future__ import annotations


from dataclasses import dataclass


from datetime import (
    datetime,
    timezone,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceConnectionToken:
    """
    Provider-independent voice connection token.
    """

    token: str

    room_name: str

    provider: str

    expires_at: datetime



    def __post_init__(
        self,
    ) -> None:
        """
        Validate token invariants.

        The existing application supports legacy
        naive UTC datetimes.

        New infrastructure integrations such as
        LiveKit may provide timezone-aware UTC
        datetimes.

        The domain therefore accepts both forms
        while performing expiration validation
        using UTC-aware values internally.
        """



        if not self.token.strip():

            raise ValueError(
                "token cannot be empty."
            )



        if not self.room_name.strip():

            raise ValueError(
                "room_name cannot be empty."
            )



        if not self.provider.strip():

            raise ValueError(
                "provider cannot be empty."
            )



        expiration_utc = (
            self._to_utc(
                self.expires_at
            )
        )



        now_utc = (
            datetime.now(
                timezone.utc
            )
        )



        if expiration_utc <= now_utc:

            raise ValueError(
                "expires_at must be in the future."
            )



    @staticmethod
    def _to_utc(
        value: datetime,
    ) -> datetime:
        """
        Convert a datetime to timezone-aware UTC
        for internal validation.

        Naive datetimes are treated as legacy UTC.

        The original value is intentionally preserved
        on the immutable domain object so existing
        callers and contracts remain compatible.
        """



        if value.tzinfo is None:

            return value.replace(
                tzinfo=timezone.utc
            )



        return value.astimezone(
            timezone.utc
        )

===== INFRASTRUCTURE: LIVEKIT TOKEN SERVICE =====
"""
LiveKit Token Service.

Infrastructure service responsible for generating
real LiveKit participant access tokens.

Responsibilities:
- Generate signed LiveKit JWT access tokens.
- Grant participant access to a specific room.
- Keep LiveKit credentials inside infrastructure.
- Return provider-independent VoiceConnectionToken.

Does NOT:
- Handle HTTP.
- Manage application workflows.
- Decide business authorization.
- Persist tokens.
- Execute AI agents.
"""

from __future__ import annotations


from datetime import (
    datetime,
    timedelta,
    timezone,
)


import os
import uuid


from dotenv import load_dotenv


from livekit import api


from app.domain.uai.voice.voice_connection_token import (
    VoiceConnectionToken,
)


load_dotenv()



class LiveKitTokenService:
    """
    Real LiveKit access-token infrastructure adapter.
    """



    def create_token(
        self,
        *,
        room_name: str,
        identity: str | None = None,
        expires_minutes: int = 30,
    ) -> VoiceConnectionToken:
        """
        Create a signed LiveKit participant token.

        Parameters
        ----------
        room_name:
            LiveKit room the participant may join.

        identity:
            Unique LiveKit participant identity.

        expires_minutes:
            Token validity period.
        """



        if not room_name.strip():

            raise ValueError(
                "room_name cannot be empty."
            )



        if expires_minutes <= 0:

            raise ValueError(
                "expires_minutes must be positive."
            )



        participant_identity = (
            identity.strip()
            if identity is not None
            else f"learner-{uuid.uuid4()}"
        )



        if not participant_identity:

            raise ValueError(
                "identity cannot be empty."
            )



        api_key = os.getenv(
            "LIVEKIT_API_KEY"
        )



        api_secret = os.getenv(
            "LIVEKIT_API_SECRET"
        )



        if not api_key:

            raise RuntimeError(
                "LIVEKIT_API_KEY is not configured."
            )



        if not api_secret:

            raise RuntimeError(
                "LIVEKIT_API_SECRET is not configured."
            )



        expires_at = (
            datetime.now(
                timezone.utc
            )
            + timedelta(
                minutes=expires_minutes,
            )
        )



        access_token = (
            api.AccessToken(
                api_key=api_key,
                api_secret=api_secret,
            )
            .with_identity(
                participant_identity,
            )
            .with_name(
                participant_identity,
            )
            .with_ttl(
                timedelta(
                    minutes=expires_minutes,
                )
            )
            .with_grants(
                api.VideoGrants(
                    room_join=True,
                    room=room_name,
                    can_publish=True,
                    can_subscribe=True,
                    can_publish_data=True,
                )
            )
        )



        token = (
            access_token.to_jwt()
        )



        return VoiceConnectionToken(
            token=token,
            room_name=room_name,
            provider="livekit",
            expires_at=expires_at,
        )

===== INFRASTRUCTURE: LIVEKIT ADAPTER =====
"""
LiveKit Infrastructure Adapter.

Infrastructure boundary for LiveKit room configuration.

Responsibilities:
- Translate voice session information into
  LiveKit provider configuration.
- Keep LiveKit-specific details inside infrastructure.
- Prepare provider connection metadata.

Does NOT:
- Decide when a call starts.
- Manage learner business logic.
- Execute AI agents.
- Manage application lifecycle.
- Persist data.
- Generate participant tokens.
"""

from __future__ import annotations


import logging


from app.domain.uai.voice.voice_session import (
    VoiceSession,
)


logger = logging.getLogger(
    __name__
)



class LiveKitAdapter:
    """
    Infrastructure adapter for LiveKit.

    Converts the provider-independent voice session
    into LiveKit room configuration.

    Token generation is deliberately handled by
    LiveKitTokenService.
    """



    provider = "livekit"



    def create_room_configuration(
        self,
        *,
        session: VoiceSession,
    ) -> dict[str, str]:
        """
        Prepare LiveKit room configuration.

        Parameters
        ----------
        session:
            Provider-independent voice session.

        Returns
        -------
        dict[str, str]
            LiveKit room metadata.
        """



        if not session.room_name.strip():

            raise ValueError(
                "Voice room name cannot be empty."
            )



        if not session.character_id.strip():

            raise ValueError(
                "Voice session character id "
                "cannot be empty."
            )



        logger.info(
            "Preparing LiveKit room configuration",
            extra={
                "session_id": (
                    session.session_id
                ),
                "room_name": (
                    session.room_name
                ),
                "character_id": (
                    session.character_id
                ),
                "language_code": (
                    session.language_code
                ),
            },
        )



        return {
            "room_name": (
                session.room_name
            ),
            "character_id": (
                session.character_id
            ),
            "language_code": (
                session.language_code
            ),
            "provider": self.provider,
        }

@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallLifecycleService:
    """
    Application lifecycle facade.
    """


    execution_coordinator: (
        VoiceCallExecutionCoordinator
    )


    connection_service: (
        VoiceConnectionService | None
    ) = None


    voice_call_management_service: (
        VoiceCallManagementService | None
    ) = None


    voice_call_session_service: (
        VoiceCallSessionService | None
    ) = None


    voice_call_event_service: (
        VoiceCallEventService | None
    ) = None



    def start_call(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> dict:
        """
        Prepare room, execute call,
        create connection and record event.
        """


        result = (
            self.execution_coordinator
            .start(
                context=context,
            )
        )


        session = result["session"]


        if self.voice_call_event_service is not None:

            self.voice_call_event_service.record_event(
                voice_call_id=(
                    session.session_id
                ),
                learner_id=(
                    session.learner_id
                ),
                character_id=(
                    session.character_id
                ),
                event_type=(
                    VoiceCallEventType.STARTED
                ),
            )


        if self.connection_service is not None:

            connection = (
                self.connection_service
                .create_connection(
                    room_name=(
                        result["room"]["room_name"]
                    ),
                )
            )


            result["connection"] = (
                connection
            )


        return result



    def create_voice_call(
        self,
        *,
        learner_id: str,
        character_id: str,
        world_id: str,
        language_code: str,
    ) -> VoiceCall:
        """
        Create persistent voice identity.
        """


        if (
            self.voice_call_management_service
            is None
        ):
            raise RuntimeError(
                "Voice call management service "
                "not configured."
            )


        return (
            self.voice_call_management_service
            .create_call(
                learner_id=learner_id,
                character_id=character_id,
                world_id=world_id,
                language_code=language_code,
            )
        )



    def create_session(
        self,
        *,
        voice_call: VoiceCall,
    ) -> VoiceCallSession:
        """
        Create execution session.
        """


        if (
            self.voice_call_session_service
            is None
        ):
            raise RuntimeError(
                "Voice call session service "
                "not configured."
            )


        return (
            self.voice_call_session_service
            .create_session(
                voice_call=voice_call,
            )
        )



    def reconnect_call(
        self,
        *,
        voice_call: VoiceCall,
    ) -> VoiceCallSession:
        """
        Reconnect existing voice call.
        """


        if (
            self.voice_call_session_service
            is None
        ):
            raise RuntimeError(
                "Voice call session service "
                "not configured."
            )


        session = (
            self.voice_call_session_service
            .reconnect_session(
                voice_call=voice_call,
            )
        )


        if self.voice_call_event_service is not None:

            self.voice_call_event_service.record_event(
                voice_call_id=(
                    voice_call.id
                ),
                learner_id=(
                    voice_call.learner_id
                ),
                character_id=(
                    voice_call.character_id
                ),
                event_type=(
                    VoiceCallEventType.RECONNECTED
                ),
            )


        return session

def decode_jwt_payload(
    token: str,
) -> dict:
    """
    Decode the JWT payload without verifying
    the signature.

    Signature verification belongs to LiveKit.
    These tests only validate the claims generated
    by our infrastructure adapter.
    """

    parts = token.split(".")

    assert len(parts) == 3

    payload = parts[1]

    padding = (
        "="
        * (
            4
            - len(payload) % 4
        )
    )

    decoded = base64.urlsafe_b64decode(
        payload + padding
    )

    return json.loads(
        decoded.decode("utf-8")
    )



def test_create_livekit_token():

    service = (
        LiveKitTokenService()
    )


    token = (
        service.create_token(
            room_name="yeti-room-001",
        )
    )


    assert isinstance(
        token,
        VoiceConnectionToken,
    )


    assert token.room_name == (
        "yeti-room-001"
    )


    assert token.provider == (
        "livekit"
    )


    # A real LiveKit JWT is generated.
    assert token.token.count(".") == 2


    assert token.token.startswith(
        "eyJ"
    )


    payload = (
        decode_jwt_payload(
            token.token
        )
    )


    assert payload["sub"].startswith(
        "learner-"
    )


    assert payload["video"]["roomJoin"] is True


    assert payload["video"]["room"] == (
        "yeti-room-001"
    )



def test_custom_expiration_time():

    service = (
        LiveKitTokenService()
    )


    token = (
        service.create_token(
            room_name="room-001",
            expires_minutes=60,
        )
    )


    now = (
        datetime.now(
            timezone.utc
        )
    )


    assert token.expires_at > (
        now
    )


    remaining_seconds = (
        token.expires_at - now
    ).total_seconds()


    # Allow a small execution-time tolerance.
    assert 3500 < (
        remaining_seconds
    ) <= 3600


    payload = (
        decode_jwt_payload(
            token.token
        )
    )


    assert payload["exp"] > (
        payload["nbf"]
    )
