class Candidate:
    def __init__(self, key: str, attempt: int = 1) -> None:
        self.notification_key = key
        self.recommendation_id = f"rec-{key}"
        self.attempt = attempt


class Discovery:
    def __init__(self, candidates) -> None:
        self.candidates = tuple(candidates)
        self.calls = 0
        self.limit = None

    async def discover(self, *, limit: int):
        self.calls += 1
        self.limit = limit
        return PlatformIntelligenceEscalationRetryCandidateBatch(
            candidates=self.candidates,
            limit=limit,
        )


class Orchestrator:
    def __init__(self, results=None, errors=None) -> None:
        self.results = results or {}
        self.errors = errors or {}
        self.calls = []

    async def run_candidate(self, *, candidate, reference_at):
        self.calls.append((candidate.notification_key, reference_at))
        if candidate.notification_key in self.errors:
            raise self.errors[candidate.notification_key]
        return self.results[candidate.notification_key]


def run_result(key, status, reason, attempt=1, provider=None):
    return PlatformIntelligenceRetryRunResult(
        notification_key=key,
        status=status,
        reason=reason,
        attempt=attempt,
        provider_message_id=provider,
    )

def _required_environment(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for authentication email composition."
        )

    return value

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )

"""
Cybersecurity APScheduler Adapter.

Infrastructure boundary for scheduled workflows.

Responsibilities:
- register existing cybersecurity jobs;
- optionally register privacy governance reminders;
- optionally register Academy notification workflows;
- start and stop the single APScheduler instance.

Does NOT:
- own Academy eligibility;
- own Academy delivery state;
- send emails directly;
- send Telegram messages directly;
- make governance decisions;
- create a second scheduler.
"""

from __future__ import annotations

from cybersecurity.scheduler.scheduler_configuration import (
    DAILY_REPORT_HOUR,
    DAILY_REPORT_MINUTE,
    WEEKLY_REPORT_DAY,
    WEEKLY_REPORT_HOUR,
    WEEKLY_REPORT_MINUTE,
)


class CybersecurityAPSchedulerAdapter:
    """APScheduler implementation boundary.

    All scheduled application workflows share this single scheduler.

    Optional workflows remain dependency-injected so existing
    cybersecurity-only tests and deployments remain backwards compatible.
    """

    def __init__(
        self,
        scheduler,
        workflow_service,
        governance_reminder_workflow=None,
        governance_reminder_interval_minutes: int = 60,
        academy_one_week_workflow=None,
        academy_one_week_hour: int = 8,
        academy_one_week_minute: int = 15,
    ) -> None:
        if scheduler is None:
            raise ValueError("scheduler cannot be None.")

        if workflow_service is None:
            raise ValueError("workflow_service cannot be None.")

        if governance_reminder_interval_minutes <= 0:
            raise ValueError(
                "governance_reminder_interval_minutes must be > 0."
            )

        if not 0 <= academy_one_week_hour <= 23:
            raise ValueError(
                "academy_one_week_hour must be between 0 and 23."
            )

        if not 0 <= academy_one_week_minute <= 59:
            raise ValueError(
                "academy_one_week_minute must be between 0 and 59."
            )

        self.scheduler = scheduler
        self.workflow_service = workflow_service

        self.governance_reminder_workflow = (
            governance_reminder_workflow
        )
        self.governance_reminder_interval_minutes = (
            governance_reminder_interval_minutes
        )

        self.academy_one_week_workflow = (
            academy_one_week_workflow
        )
        self.academy_one_week_hour = academy_one_week_hour
        self.academy_one_week_minute = academy_one_week_minute

    def register_jobs(self) -> None:
        """Register all enabled workflows on the single scheduler."""

        self.scheduler.add_job(
            self.workflow_service.create_daily_report,
            trigger="cron",
            hour=DAILY_REPORT_HOUR,
            minute=DAILY_REPORT_MINUTE,
            id="cybersecurity_daily_report",
            replace_existing=True,
        )

        self.scheduler.add_job(
            self.workflow_service.create_weekly_report,
            trigger="cron",
            day_of_week=WEEKLY_REPORT_DAY,
            hour=WEEKLY_REPORT_HOUR,
            minute=WEEKLY_REPORT_MINUTE,
            id="cybersecurity_weekly_report",

"""
Cybersecurity Scheduler Runtime.

Lifecycle manager for cybersecurity scheduling.

Responsibilities:
- start scheduler;
- register jobs;
- stop scheduler;
- expose runtime state.

Does NOT:
- execute reports;
- send emails;
- make security decisions;
- perform remediation.
"""

from __future__ import annotations


from cybersecurity.scheduler.cybersecurity_job_registry import (
    CybersecurityJobRegistry,
)



class CybersecuritySchedulerRuntime:
    """
    Scheduler lifecycle boundary.

    Supports:

    - registry based execution/testing;
    - production scheduler adapter.

    Architecture:

        FastAPI lifespan
              |
              v
    CybersecuritySchedulerRuntime
              |
              +----------------+
              |                |
              v                v

        Job Registry     Scheduler Adapter

          (tests)          (production)

    """



    def __init__(
        self,
        scheduler_adapter=None,
        job_registry: CybersecurityJobRegistry | None = None,
    ) -> None:
        """
        Initialize scheduler runtime.
        """


        self.scheduler_adapter = (
            scheduler_adapter
        )


        self.job_registry = (
            job_registry
            or CybersecurityJobRegistry()
        )


        self._started = False



    async def start(
        self,
    ) -> None:
        """
        Start scheduler runtime.

        Registers scheduled cybersecurity workflows.
        """


        if self.scheduler_adapter is not None:

            #
            # Production scheduler boundary.
            #
            # Example:
            # APScheduler adapter
            #

            self.scheduler_adapter.register_jobs()

            self.scheduler_adapter.start()


        else:

            #
            # Registry mode.
            #
            # Used for testing and controlled execution.
            #

            self.job_registry.register(

                name=(
                    "cybersecurity_daily_report"
                ),

                schedule=(
                    "daily_08:00"
                ),

            )


            self.job_registry.register(

                name=(
                    "cybersecurity_weekly_report"
                ),

                schedule=(
                    "weekly_monday_08:00"
                ),

            )


            self.job_registry.register(

                name=(
                    "cybersecurity_critical_alert"
                ),

                schedule=(
                    "event_immediate"
                ),

            )



        self._started = True



    async def stop(
        self,
    ) -> None:
        """
        Stop scheduler runtime.
        """


        if self.scheduler_adapter is not None:

            self.scheduler_adapter.shutdown()



        self._started = False



    def status(
        self,
    ) -> dict:
        """
        Return scheduler runtime status.

        Provides backward-compatible fields:

        - jobs
        - registered_jobs
        """


        jobs = (
            self.job_registry.list_jobs()
        )



        return {

            "started":
                self._started,


            "adapter_enabled":
                self.scheduler_adapter is not None,


            #
            # Existing test contract
            #
            "jobs":
                jobs,


            #
            # Extended API contract
            #
            "registered_jobs":
                jobs,

        }
