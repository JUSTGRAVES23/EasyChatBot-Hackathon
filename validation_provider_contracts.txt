================================================================
=== BASE PROVIDER ===
================================================================
"""
Base Provider Contract.

Defines the interface for external AI and knowledge providers.

Examples:
- Gemini
- Parallel Search
- Future LLM providers

The Runtime communicates with providers through this contract.

Providers are replaceable implementations.
The Agent layer must remain independent from provider details.
"""

from abc import ABC, abstractmethod

from backend.providers.provider_result import ProviderResult
from backend.runtime.context.execution_context import ExecutionContext


class BaseProvider(ABC):
    """
    Abstract base class for EasyChatBot providers.

    Providers are responsible for communicating
    with external services.

    They are NOT responsible for:
    - agent decisions
    - workflow orchestration
    - memory management
    - user interaction
    """

    provider_id: str

    def __init__(
        self,
        provider_id: str,
    ):
        """
        Initialise provider.

        Args:
            provider_id:
                Unique provider identifier.
        """

        self.provider_id = provider_id

    @abstractmethod
    async def execute(
        self,
        context: ExecutionContext,
    ) -> ProviderResult:
        """
        Execute provider operation.

        Args:
            context:
                Runtime execution context.

        Returns:
            ProviderResult containing provider output.
        """

        pass

================================================================
=== PROVIDER RESULT ===
================================================================
"""
Provider Result Contract.

Defines the standard response returned
by external providers.

All providers return this structure so
the Runtime can process results consistently.
"""

from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class ProviderResult:
    """
    Immutable provider execution result.

    Attributes:

        success:
            Indicates whether provider execution succeeded.

        provider_id:
            Identifier of the provider.

        output:
            Provider response payload.

        error:
            Error details if execution failed.
    """

    success: bool
    provider_id: str
    output: Any = None
    error: str | None = None

================================================================
=== LLM PROVIDER ===
================================================================
"""
LLM Provider Contract.

Defines the provider-independent contract for language
model providers used by the EasyChatBot hackathon runtime.

The contract sits between the Conversation Agent and
concrete LLM provider implementations.

Architecture:

ConversationAgent
        |
        v
ConversationRequest
        |
        v
LLMProvider
        |
        v
Concrete LLM Provider
        |
        v
ProviderResult

Responsibilities:

- define the LLM generation boundary
- keep ConversationAgent independent from provider SDKs
- provide a stable contract for concrete LLM providers

Does not:

- call an external API
- contain provider SDK logic
- select models
- implement routing
- manage memory
- manage conversations
"""

from __future__ import annotations

from abc import ABC
from abc import abstractmethod

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.provider_result import (
    ProviderResult,
)


class LLMProvider(ABC):
    """
    Abstract provider contract for language model generation.

    Concrete implementations may use OpenAI, Ollama,
    Gemini, or another approved LLM provider.

    The application and agent layers depend only on
    this contract.
    """

    provider_id: str

    def __init__(
        self,
        provider_id: str,
    ) -> None:
        """
        Initialise the LLM provider.

        Args:
            provider_id:
                Stable identifier for the concrete provider.
        """

        if not provider_id.strip():
            raise ValueError(
                "LLM provider_id cannot be empty."
            )

        self.provider_id = provider_id

    @abstractmethod
    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate an LLM response.

        Args:
            request:
                Provider-independent conversation request.

        Returns:
            ProviderResult containing the generated
            response or an error.

        Implementations must not expose provider SDK
        objects through the result.
        """

        raise NotImplementedError

================================================================
=== LLM PROVIDER FACTORY ===
================================================================
"""
LLM Provider Factory.

Creates approved LLM provider implementations.

The factory is the composition boundary between
application/runtime code and infrastructure providers.

Architecture:

Application/API
|
v
LLMProviderFactory
|
v
LLMProvider
|
+--> OpenAIProvider
+--> Future OllamaProvider
+--> GeminiProvider

Responsibilities:

- create configured LLM providers
- isolate provider construction
- provide provider independence
- obtain approved provider configuration

Does not:

- execute LLM requests
- build prompts
- manage conversations
- manage memory
- apply routing decisions
- call provider SDKs directly
"""

from __future__ import annotations

from backend.providers.llm.config import (
    get_gemini_model,
    get_llm_model,
    get_llm_provider_name,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.providers.llm.openai_provider import (
    OpenAIProvider,
)

from backend.providers.llm.gemini_provider import (
    GeminiProvider,
)


class LLMProviderFactory:
    """
    Factory responsible for creating LLM providers.

    Provider selection and configuration are centralised
    here so application/runtime code does not depend
    directly on concrete infrastructure implementations.
    """

    @staticmethod
    def create(
        provider: str | None = None,
    ) -> LLMProvider:
        """
        Create a configured LLM provider.

        Parameters
        ----------
        provider:
            Optional provider identifier.

            When omitted, the configured provider from
            LLM provider configuration is used.

        Returns
        -------
        LLMProvider
            Configured provider implementation.

        Raises
        ------
        ValueError
            If provider is invalid or unsupported.
        """

        if provider is None:
            provider_name = (
                get_llm_provider_name()
            )
        else:
            if not isinstance(
                provider,
                str,
            ):
                raise ValueError(
                    "Provider name must be a string."
                )

            provider_name = (
                provider.strip().lower()
            )

            if not provider_name:
                raise ValueError(
                    "Provider name cannot be empty."
                )

        if provider_name == "openai":
            return OpenAIProvider(
                model=get_llm_model(),
            )

        if provider_name == "gemini":
            return GeminiProvider(
                model=get_gemini_model(),
            )

        raise ValueError(
            (
                "Unsupported LLM provider: "
                f"{provider_name}"
            )
        )

================================================================
=== OPENAI PROVIDER ===
================================================================
"""
OpenAI LLM Provider.

Concrete LLM provider implementation for the EasyChatBot
hackathon runtime.

Architecture:

ConversationAgent
|
v
LLMService
|
v
LLMProvider
|
v
OpenAIProvider
|
v
OpenAI Responses API
|
v
ProviderResult

Responsibilities:

- execute ConversationRequest through OpenAI
- use the configured OpenAI model
- translate the OpenAI response into ProviderResult
- isolate the OpenAI SDK from the application layer

Does not:

- build prompts
- manage characters
- manage memory
- perform research
- perform routing
- modify application state
"""

from __future__ import annotations

import logging
import os

from openai import AsyncOpenAI

from backend.conversation.conversation_request import (
    ConversationRequest,
)

from backend.providers.llm.llm_provider import (
    LLMProvider,
)

from backend.providers.provider_result import (
    ProviderResult,
)


logger = logging.getLogger(__name__)


class OpenAIProvider(LLMProvider):
    """
    OpenAI implementation of the LLMProvider contract.

    The OpenAI SDK is deliberately isolated inside this
    infrastructure adapter.
    """

    def __init__(
        self,
        model: str = "gpt-5-mini",
        client: AsyncOpenAI | None = None,
    ) -> None:
        """
        Initialise the OpenAI provider.

        Parameters
        ----------
        model:
            OpenAI model identifier.

        client:
            Optional injected AsyncOpenAI client.

            Dependency injection allows unit tests to
            provide a fake client without making a real
            API call.
        """

        super().__init__(
            provider_id="openai",
        )

        if not isinstance(
            model,
            str,
        ):
            raise ValueError(
                "OpenAI model must be a string."
            )

        model = model.strip()

        if not model:
            raise ValueError(
                "OpenAI model cannot be empty."
            )

        self.model = model

        if client is not None:
            self._client = client
            return

        api_key = os.getenv(
            "OPENAI_API_KEY"
        )

        if not api_key:
            raise ValueError(
                "OPENAI_API_KEY is not configured."
            )

        self._client = AsyncOpenAI(
            api_key=api_key,
        )

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate a response using OpenAI.

        The provider translates the provider-independent
        ConversationRequest into the OpenAI Responses API
        request.

        Parameters
        ----------
        request:
            Provider-independent conversation request.

        Returns
        -------
        ProviderResult
            Provider-independent execution result.

        Raises
        ------
        ValueError
            If request is None.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        messages = [
            {
                "role": "system",
                "content": request.system_instruction,
            },
        ]

        if request.conversation_context:
            messages.append(
                {
                    "role": "user",
                    "content": (
                        "Additional context:\n"
                        f"{request.conversation_context}"
                    ),
                }
            )

        messages.append(
            {
                "role": "user",
                "content": request.user_prompt,
            }
        )

        try:
            response = (
                await self._client.responses.create(
                    model=self.model,
                    input=messages,
                )
            )

        except Exception as exc:
            logger.exception(
                "OpenAI provider execution failed."
            )

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=str(exc),
            )

        output_text = getattr(
            response,
            "output_text",
            None,
        )

        if not output_text:
            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                error=(
                    "OpenAI returned an empty response."
                ),
            )

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output=output_text,
        )

================================================================
=== GEMINI PROVIDER ===
================================================================
"""
Gemini LLM Provider.

Concrete Gemini implementation of the provider-independent
LLMProvider contract used by the EasyChatBot hackathon runtime.

Architecture:

    ConversationAgent
            |
            v
    ConversationRequest
            |
            v
       LLMProvider
            |
            v
      GeminiProvider
            |
            v
       Gemini API
            |
            v
      ProviderResult

Responsibilities:
- execute ConversationRequest through Gemini
- use the configured Gemini model
- translate Gemini responses into ProviderResult
- isolate the Gemini SDK from the application/runtime layers

Does not:
- build application workflows
- perform research
- manage memory
- select providers
- manage conversations
- modify application state

The provider uses Google's current `google-genai` SDK.
The SDK client is injectable so unit tests do not need a real
Gemini API call.
"""

from __future__ import annotations

import logging
import os
from typing import Any

from backend.conversation.conversation_request import (
    ConversationRequest,
)
from backend.providers.llm.llm_provider import (
    LLMProvider,
)
from backend.providers.provider_result import (
    ProviderResult,
)


logger = logging.getLogger(__name__)


class GeminiProvider(LLMProvider):
    """
    Gemini implementation of the LLMProvider contract.

    The Google Gemini SDK is deliberately isolated inside this
    infrastructure adapter.

    Args:
        model:
            Gemini model identifier.

        client:
            Optional injected `google.genai.Client`.

            Dependency injection allows unit tests to provide a
            fake client without making a real API call.
    """

    def __init__(
        self,
        model: str = "gemini-3.6-flash",
        client: Any | None = None,
    ) -> None:
        super().__init__(
            provider_id="gemini",
        )

        if not isinstance(model, str) or not model.strip():
            raise ValueError(
                "Gemini model cannot be empty."
            )

        self.model = model.strip()
        self._client = client

    def _get_client(self) -> Any:
        """
        Lazily create the Google GenAI client.

        The API key is read from GEMINI_API_KEY and is never
        hard-coded into the provider.
        """

        if self._client is not None:
            return self._client

        api_key = os.getenv("GEMINI_API_KEY")

        if not api_key:
            raise RuntimeError(
                "GEMINI_API_KEY environment variable is required."
            )

        try:
            from google import genai
        except ImportError as exc:
            raise RuntimeError(
                "The google-genai package is required for "
                "GeminiProvider. Install it with: "
                "pip install google-genai"
            ) from exc

        self._client = genai.Client(
            api_key=api_key,
        )

        return self._client

    @staticmethod
    def _build_contents(
        request: ConversationRequest,
    ) -> str:
        """
        Convert the provider-independent request into the
        user-facing content sent to Gemini.

        System behaviour remains separate and is supplied through
        Gemini's system_instruction configuration.

        Live intelligence, when present, is deliberately carried
        through the existing ConversationRequest context rather
        than retrieved by Gemini itself.
        """

        sections: list[str] = []

        if request.conversation_context:
            sections.extend(
                (
                    "CONTEXT:",
                    request.conversation_context.strip(),
                    "",
                )
            )

        sections.extend(
            (
                "USER REQUEST:",
                request.user_prompt.strip(),
            )
        )

        return "\n".join(sections)

    async def generate(
        self,
        request: ConversationRequest,
    ) -> ProviderResult:
        """
        Generate a response using Gemini.

        Returns a provider-independent ProviderResult and never
        exposes Gemini SDK response objects to the application.
        """

        if request is None:
            raise ValueError(
                "Conversation request cannot be None."
            )

        try:
            client = self._get_client()

            from google.genai import types

            contents = self._build_contents(
                request,
            )

            config = types.GenerateContentConfig(
                system_instruction=(
                    request.system_instruction
                ),
            )

            response = await client.aio.models.generate_content(
                model=self.model,
                contents=contents,
                config=config,
            )

        except Exception as exc:
            logger.exception(
                "Gemini provider execution failed."
            )

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error=str(exc),
            )

        output_text = getattr(
            response,
            "text",
            None,
        )

        if not output_text or not output_text.strip():
            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error="Gemini returned an empty response.",
            )

        return ProviderResult(
            success=True,
            provider_id=self.provider_id,
            output=output_text.strip(),
        )

=== PARALLEL PROVIDER ===

"""
Parallel Search Provider.

Provides live world information
through the Parallel Search API.
"""

from backend.providers.base_provider import (
    BaseProvider,
)

from backend.providers.provider_result import (
    ProviderResult,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


class ParallelSearchProvider(BaseProvider):
    """
    Provider implementation using Parallel Search.

    The SDK client is created only when required.
    This allows tests to run without API credentials.
    """

    def __init__(
        self,
        client=None,
    ):
        super().__init__(
            provider_id="parallel_search"
        )

        self.client = client


    def _get_client(self):
        """
        Lazy-load the real Parallel client.

        Avoids requiring API keys during unit tests.
        """

        if self.client is None:

            from backend.providers.parallel.parallel_client import (
                ParallelClientWrapper,
            )

            self.client = ParallelClientWrapper()

        return self.client


    async def execute(
        self,
        context: ExecutionContext,
    ) -> ProviderResult:
        """
        Execute a live Parallel search.
        """

        try:

            client = self._get_client()

            result = await client.search(
                query=context.request
            )

            return ProviderResult(
                success=True,
                provider_id=self.provider_id,
                output=result,
            )

        except Exception as exc:

            return ProviderResult(
                success=False,
                provider_id=self.provider_id,
                output=None,
                error=str(exc),
            )

================================================================
=== PARALLEL CLIENT ===
================================================================
"""
Parallel Search API Client Wrapper.

This module isolates the official Parallel SDK.

The rest of EasyChatBot does not directly depend on
the Parallel SDK implementation.

Architecture:

Research Agent
        |
        v
Parallel Provider
        |
        v
Parallel Client Wrapper
        |
        v
Parallel API
"""

import asyncio
import os
from typing import Any

from dotenv import load_dotenv


load_dotenv()


class ParallelClientWrapper:
    """
    Wrapper around the official Parallel Search SDK.

    Responsibilities:
    - authenticate with Parallel
    - execute searches
    - normalize SDK responses

    Does not:
    - create learning content
    - manage agents
    - decide workflows
    """

    def __init__(
        self,
        api_key: str | None = None,
    ):
        """
        Initialise the Parallel SDK client.
        """

        self.api_key = (
            api_key
            or os.getenv("PARALLEL_API_KEY")
        )

        if not self.api_key:
            raise ValueError(
                "PARALLEL_API_KEY environment variable is required"
            )

        self._client = self._create_client()


    def _create_client(self):
        """
        Create the official Parallel SDK client.

        SDK:
            parallel-web 1.1.0
        """

        from parallel import Parallel

        return Parallel(
            api_key=self.api_key
        )


    @staticmethod
    def _normalize_response(
        response: Any,
    ) -> Any:
        """
        Convert the Parallel SDK response into
        application-safe data.

        Pydantic SDK models are converted to dictionaries.
        Other JSON-compatible values are returned unchanged.
        """

        if hasattr(response, "model_dump"):
            return response.model_dump()

        if hasattr(response, "dict"):
            return response.dict()

        return response


    async def search(
        self,
        query: str,
    ) -> dict[str, Any]:
        """
        Execute a Parallel web search.

        Args:
            query:
                Natural-language research objective.

        Returns:
            Normalized application-level search result.
        """

        search_query = self._build_search_query(
            query
        )

        response = await asyncio.to_thread(
            self._client.search,
            search_queries=[
                search_query
            ],
            mode="advanced",
            objective=query,
        )

        return {
            "query": query,
            "search_queries": [
                search_query
            ],
            "results": self._normalize_response(
                response
            ),
        }


    @staticmethod
    def _build_search_query(
        query: str,
    ) -> str:
        """
        Build a concise search query from the
        user's research request.

        The full request remains the Parallel
        objective while the search query is kept
        concise.
        """

        words = query.strip().split()

        if len(words) <= 6:
            return query.strip()

        return " ".join(words[:6])
