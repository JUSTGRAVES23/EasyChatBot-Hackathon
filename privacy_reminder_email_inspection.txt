        self,
        *,
        case: Any,
        recipient_role: str = "DPO/GRC",
        reason: str = "pending_case_reminder",
    ) -> dict[str, object]:
        """Build and send one controlled governance reminder."""

        del recipient_role
        del reason

        if self.sender is None:
            raise ValueError(
                "sender cannot be None."
            )

        if not isinstance(self.recipient, str):
            raise ValueError(
                "recipient is required."
            )

        if not self.recipient.strip():
            raise ValueError(
                "recipient is required."
            )

            raise ValueError(
                "case_url_builder cannot be None."
            )

            raise ValueError(
                "logo_path cannot be None."
            )

            raise FileNotFoundError(
                "Governance email logo not found: "
                f"{self.logo_path}"

            case,
            "case_id",

        if not isinstance(case_id, str) or not case_id.strip():
            raise ValueError(

            "[EasyChatBot Privacy Governance] "
            f"Reminder â€” Decision Required for {case_id}"
        )

            case=case,
            case_url=case_url,
        )

        text_body = self._render_text(
            case=case,
            case_url=case_url,
        )

        response = await self.sender.send(
            recipient=self.recipient.strip(),
            subject=subject,
            body=text_body,
            html_body=html_body,
            attachments=[
                {
                    "path": str(self.logo_path),
                    "filename": "easychatbot_logo.png",
                    "content_id": "easychatbot-governance-logo",
                    "inline": True,
                }
            ],
        )

        status = "sent"
        reference: str | None = None

        if isinstance(response, dict):
            status = str(
                response.get(
                    "status",
                    "sent",
                )
            )

            value = (
                response.get("reference")
                or response.get("message_id")
            )

            if value is not None:
                reference = str(value)

        return {
            "status": status,
            "reference": reference,
            "case_url": case_url,
        }

from __future__ import annotations

from typing import Any

from backend.app.application.privacy.email.privacy_governance_reminder_email_service import (
    PrivacyGovernanceEmailSender,
)
from backend.cybersecurity.email.email_provider_manager import (
    EmailProviderManager,
)


class ResendGovernanceReminderEmailSender:
    """Infrastructure adapter for privacy governance reminder emails.

    Delivery path:

        PrivacyGovernanceReminderEmailService
            |
            v
        ResendGovernanceReminderEmailSender
            |
            v
        EmailProviderManager
            |
            v
        ResendEmailProvider
            |
            v
        Resend

    The adapter translates the richer HTML/text reminder contract into
    the existing provider-independent email infrastructure.

    Does NOT:
    - approve or reject governance cases;
    - modify governance state;
    - bypass the reminder scheduler;
    - access provider credentials directly;
    - make autonomous governance decisions.
    """

    def __init__(
        self,
        *,
        provider_manager: EmailProviderManager | None = None,
    ) -> None:
        self._provider_manager = (
            provider_manager
            if provider_manager is not None
            else EmailProviderManager()
        )

    async def send(
        self,
        *,
        recipient: str,
        subject: str,
        body: str,
        html_body: str,
        attachments: list[dict[str, Any]] | None = None,
    ) -> Any:
        """Deliver a governance reminder through the existing email boundary."""

        if not recipient or not recipient.strip():
            raise ValueError("Email recipient is required.")

        if not subject or not subject.strip():
            raise ValueError("Email subject cannot be empty.")

        if not body or not body.strip():
            raise ValueError("Email body cannot be empty.")

        if not html_body or not html_body.strip():
            raise ValueError("HTML email body cannot be empty.")

        content: dict[str, Any] = {
            "recipient": recipient.strip(),
            "body": body,
            "html_body": html_body,
            "email_type": "privacy_governance_reminder",
        }

        if attachments:
            content["attachments"] = attachments

        return await self._provider_manager.send(
            subject,
            content,
        )

"""
Resend email provider.

Infrastructure adapter responsible only for email delivery.

The provider supports multiple controlled email contracts:

- Cybersecurity notifications:
  rendered by the existing cybersecurity HTML renderer.

- Privacy Governance reminders:
  the application service supplies the authoritative HTML and
  plain-text content. The provider must deliver that content
  without replacing it with the cybersecurity template.

The provider does not:
- make governance decisions;
- modify governance state;
- execute remediation;
- generate application decisions;
- access provider credentials outside configuration.
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

import resend

from backend.cybersecurity.email.email_configuration import (
    EMAIL_FROM,
    RESEND_API_KEY,
)


class ResendEmailProvider:
    """
    Provider adapter for Resend email delivery.

    Routing rule:

        privacy_governance_reminder
            -> supplied html_body

        all other email types
            -> existing cybersecurity renderer

    This preserves the existing Cybersecurity email while allowing
    application-owned Privacy Governance templates to pass through
    unchanged.
    """

    provider_id = "resend"

    def __init__(
        self,
        *,
        api_key: str | None = None,
        sender: str | None = None,
    ) -> None:
        self.api_key = api_key or RESEND_API_KEY
        self.sender = sender or EMAIL_FROM

        if not self.api_key:
            raise ValueError(
                "RESEND_API_KEY is not configured."
            )

        if not self.sender:
            raise ValueError(
                "Email sender is not configured."
            )

        resend.api_key = self.api_key

    async def send(
        self,
        subject: str,
        content: dict[str, Any],
    ) -> dict[str, Any]:
        """
        Send an email through Resend.

        Privacy Governance emails provide their own HTML template.
        Cybersecurity emails continue to use _build_html().
        """

        if not isinstance(subject, str) or not subject.strip():
            raise ValueError(
                "Email subject cannot be empty."
            )

        if not isinstance(content, dict):
            raise TypeError(
                "Email content must be a dictionary."
            )

        recipient = content.get("recipient")

        if not isinstance(recipient, str) or not recipient.strip():
            raise ValueError(
                "Email recipient is required."
            )

        recipient = recipient.strip()

        email_type = content.get(
            "email_type",
            "",
        )

        # ========================================================
        # HTML routing
        # ========================================================
        #
        # Privacy Governance owns its complete presentation.
        #
        # The reminder service already generated:
        #
        # - EasyChatBot Privacy Governance branding
        # - CASE-E2E-001
        # - PCR-E2E-001
        # - risk
        # - validation
        # - DPO/GRC reviewer information
        # - REVIEW THIS CASE button
        # - case URL
        # - governance disclaimer
        # - EasyChatBot logo CID
        #
        # Do NOT replace that HTML with the Cybersecurity renderer.
        #
        if email_type == "privacy_governance_reminder":
            html = self._get_supplied_html_body(content)
        else:
            # Existing Cybersecurity behaviour remains unchanged.
            html = self._build_html(content)

        # ========================================================
        # Plain-text body
        # ========================================================

        text_body = content.get("body")

        if text_body is not None:
            if not isinstance(text_body, str):
                raise TypeError(
                    "body must be a string when provided."
                )

            if not text_body.strip():
                raise ValueError(
                    "body cannot be empty when provided."
                )

        # ========================================================
        # Resend payload
        # ========================================================

        params: dict[str, Any] = {
            "from": self.sender,
            "to": [recipient],
            "subject": subject,
            "html": html,
        }

        if isinstance(text_body, str) and text_body.strip():
            params["text"] = text_body

        attachments = content.get("attachments")

        if attachments is not None:
            params["attachments"] = self._build_attachments(
                attachments
            )

        response = await resend.Emails.send_async(
            params
        )

        if isinstance(response, dict):
            message_id = response.get("id")
        else:
            message_id = getattr(
                response,
                "id",
                None,
            )

        return {
            "provider": self.provider_id,
            "status": "accepted",
            "message_id": message_id,
        }

    @staticmethod
    def _get_supplied_html_body(
        content: dict[str, Any],
    ) -> str:
        """
        Return the application-owned Privacy Governance HTML.

        The provider deliberately does not generate or alter this
        HTML because the application service owns the presentation
        and governance message.
        """

        html_body = content.get("html_body")

        if not isinstance(html_body, str):
            raise TypeError(
                "Privacy Governance email requires "
                "html_body to be a string."
            )

        if not html_body.strip():
            raise ValueError(
                "Privacy Governance email html_body cannot be empty."
            )

        return html_body

    @staticmethod
    def _build_attachments(
        attachments: Any,
    ) -> list[dict[str, Any]]:
        """
        Convert application attachment metadata into Resend
        attachment payloads.

        Supports the Privacy Governance inline logo:

            cid:easychatbot-governance-logo
        """

        if not isinstance(attachments, list):
            raise TypeError(
                "attachments must be a list when provided."
            )

        normalized: list[dict[str, Any]] = []

        for attachment in attachments:
            if not isinstance(attachment, dict):
                raise TypeError(
                    "Each email attachment must be a dictionary."
                )

            path_value = attachment.get("path")

            if not path_value:
                raise ValueError(
                    "Email attachment path is required."
                )

            path = Path(str(path_value))

            if not path.exists():
                raise FileNotFoundError(
                    f"Email attachment not found: {path}"
                )

            if not path.is_file():
                raise ValueError(
                    f"Email attachment is not a file: {path}"
                )

            filename = attachment.get("filename")

            resolved_filename = (
                str(filename).strip()
                if filename
                else path.name
            )

            if not resolved_filename:
                raise ValueError(
                    "Email attachment filename is required."
                )

            content_id = attachment.get(
                "content_id"
            )

            inline = bool(
                attachment.get(
                    "inline",
                    False,
                )
            )

            if inline and not content_id:
                raise ValueError(
                    "Inline email attachments require "
                    "content_id."
                )

            resend_attachment: dict[str, Any] = {
                "path": str(path),
                "filename": resolved_filename,
            }

            if content_id:
                resend_attachment["content_id"] = str(
                    content_id
                )

            normalized.append(
                resend_attachment
            )

        return normalized

    @staticmethod
    def _build_html(
        content: dict[str, Any],
    ) -> str:
        """
        Existing Cybersecurity HTML renderer.

        This remains the Cybersecurity fallback and is intentionally
        not used for Privacy Governance reminders.
        """

        posture = str(
            content.get(
                "posture",
                "Security notification",
            )
        )

        critical_count = content.get(
            "critical_count",
            0,
        )

        high_count = content.get(
            "high_count",
            0,
        )

        priority_signals = content.get(
            "priority_signals",
            [],
        )

        signal_rows = ""

        if isinstance(
            priority_signals,
            list,
        ):
            for signal in priority_signals:
                if not isinstance(
                    signal,
                    dict,
                ):
                    continue

                title = str(
                    signal.get(
                        "title",
                        "Security signal",
                    )
                )

                severity = str(
                    signal.get(
                        "severity",
                        "",
                    )
                ).upper()

                signal_rows += (
                    "<li>"
                    f"<strong>{title}</strong>"
                    f" — {severity}"
                    "</li>"
                )

        return f"""
<!DOCTYPE html>
<html>
<head>
    <meta charset="utf-8">
    <title>EasyChatBot Cybersecurity</title>
</head>
<body>
    <h2>EasyChatBot Cybersecurity Notification</h2>

    <p>
        <strong>Security posture:</strong>
        {posture}
    </p>

    <p>
        <strong>Critical findings:</strong>
        {critical_count}
    </p>

    <p>
        <strong>High findings:</strong>
        {high_count}
    </p>

    <h3>Priority Security Signals</h3>

    <ul>
        {signal_rows}
    </ul>

    <hr>

    <p>
        This notification is decision support only.
        No remediation action was executed by EasyChatBot.
    </p>

    <p>
        <strong>Environment:</strong> DEMO
    </p>
</body>
</html>
"""
