"""
Intelligence Context.

Provider-independent representation of external
information prepared for AI execution.

Architecture:

Research Agent
        |
        v
IntelligenceContext
        |
        v
AI Runtime integration
        |
        v
LLMRequest.context

The context must not expose provider-specific
implementation details to the application layer.
"""

from __future__ import annotations

from dataclasses import dataclass, field


@dataclass(frozen=True, slots=True)
class IntelligenceSource:
    """
    Normalised external information source.
    """

    title: str
    url: str
    source: str
    content: str = ""

    def __post_init__(self) -> None:
        if not self.title.strip():
            raise ValueError(
                "Intelligence source title cannot be empty."
            )

        if not self.url.strip():
            raise ValueError(
                "Intelligence source URL cannot be empty."
            )

        if not self.source.strip():
            raise ValueError(
                "Intelligence source name cannot be empty."
            )


@dataclass(frozen=True, slots=True)
class IntelligenceContext:
    """
    Provider-independent intelligence context.

    Represents external information that may be supplied
    to an existing AI conversation.

    This object deliberately does not contain:
    - provider SDK objects
    - API credentials
    - model information
    - runtime state
    - persistence objects
    """

    query: str

    sources: tuple[IntelligenceSource, ...] = field(
        default_factory=tuple,
    )

    summary: str = ""

    def __post_init__(self) -> None:
        if not self.query.strip():
            raise ValueError(
                "Intelligence query cannot be empty."
            )

        if not isinstance(
            self.sources,
            tuple,
        ):
            raise ValueError(
                "Intelligence sources must be a tuple."
            )

        if not isinstance(
            self.summary,
            str,
        ):
            raise ValueError(
                "Intelligence summary must be a string."
            )

    @property
    def has_sources(self) -> bool:
        """
        Return whether external sources are available.
        """

        return bool(self.sources)

    @property
    def source_count(self) -> int:
        """
        Return the number of external sources.
        """

        return len(self.sources)

    @property
    def has_summary(self) -> bool:
        """
        Return whether a summary is available.
        """

        return bool(
            self.summary.strip()
        )

    def to_context_text(self) -> str:
        """
        Convert intelligence into a safe textual context.

        The result is suitable for eventual inclusion
        in an existing LLMRequest.context field.
        """

        sections: list[str] = [
            "LIVE INTELLIGENCE",
            f"Research query: {self.query.strip()}",
        ]

        if self.summary.strip():
            sections.extend(
                (
                    "",
                    "Summary:",
                    self.summary.strip(),
                )
            )

        if self.sources:
            sections.extend(
                (
                    "",
                    "Sources:",
                )
            )

            for index, source in enumerate(
                self.sources,
                start=1,
            ):
                sections.append(
                    f"{index}. {source.title}"
                )
                sections.append(
                    f"   URL: {source.url}"
                )
                sections.append(
                    f"   Source: {source.source}"
                )

                if source.content.strip():
                    sections.append(
                        f"   Content: {source.content.strip()}"
                    )

        return "\n".join(sections)

"""
Live Intelligence Agent Handler.

Responsible for executing the Live Intelligence Agent workflow.

Architecture:

AI Runtime
    |
    v
Live Intelligence Agent Handler
    |
    +--> Search Provider Interface
    |
    +--> Knowledge Processing
    |
    v
Structured Intelligence Output

Responsibilities:
- Validate agent input
- Call approved provider interfaces
- Normalise research output
- Return explicit structured results

Non-responsibilities:
- Provider SDK management
- Direct external API handling
- Autonomous actions
- Configuration changes
"""

from dataclasses import dataclass
from typing import List, Protocol

from src.providers.search_provider.interface import (
    SearchProvider,
    SearchResult,
)


@dataclass(frozen=True)
class LiveIntelligenceRequest:
    """
    Input contract for Live Intelligence Agent.
    """

    topic: str
    language: str = "French"
    learner_level: str = "B2"


@dataclass(frozen=True)
class LiveIntelligenceResponse:
    """
    Output contract from Live Intelligence Agent.
    """

    topic: str
    summary: str
    sources: List[SearchResult]
    key_information: List[str]


class IntelligenceProcessor(Protocol):
    """
    Contract for future LLM-based processing.

    Keeps the agent independent from
    any specific model/provider.
    """

    async def summarise(
        self,
        results: List[SearchResult],
        language: str,
        learner_level: str,
    ) -> tuple[str, List[str]]:
        ...


class LiveIntelligenceAgent:
    """
    Live Intelligence Agent implementation.
    """

    def __init__(
        self,
        search_provider: SearchProvider,
        processor: IntelligenceProcessor,
    ) -> None:
        self._search_provider = search_provider
        self._processor = processor

    async def execute(
        self,
        request: LiveIntelligenceRequest,
    ) -> LiveIntelligenceResponse:
        """
        Execute research workflow.

        Flow:

        Request
          |
          v
        Search Provider
          |
          v
        Intelligence Processor
          |
          v
        Structured Response
        """

        self._validate_request(request)

        search_results = await self._search_provider.search(
            query=request.topic,
            max_results=5,
        )

        summary, key_information = await self._processor.summarise(
            results=search_results,
            language=request.language,
            learner_level=request.learner_level,
        )

        return LiveIntelligenceResponse(
            topic=request.topic,
            summary=summary,
            sources=search_results,
            key_information=key_information,
        )

    @staticmethod
    def _validate_request(
        request: LiveIntelligenceRequest,
    ) -> None:
        """
        Validate required input.
        """

        if not request.topic.strip():
            raise ValueError(
                "Topic cannot be empty"
            )

"""
AI Runtime capability contracts.

This module defines the provider-independent contract used by
the AI Runtime to request execution of an approved capability.

Architecture:

Application
    |
    v
AI Runtime
    |
    v
Capability Request
    |
    v
Governance
    |
    v
Capability Execution

This module contains contracts only.

It does not:
- execute agents
- call providers
- evaluate governance policies
- perform external side effects
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import StrEnum
from typing import Any, Mapping


class CapabilityType(StrEnum):
    """
    Supported runtime capability categories.

    The list is intentionally small for the initial framework
    implementation. New capabilities should be added explicitly.
    """

    LIVE_INTELLIGENCE = "live_intelligence"
    LEARNING_GENERATION = "learning_generation"
    EVALUATION = "evaluation"


class CapabilityExecutionStatus(StrEnum):
    """
    Runtime-level execution status.
    """

    ACCEPTED = "accepted"
    COMPLETED = "completed"
    REJECTED = "rejected"
    FAILED = "failed"


@dataclass(frozen=True, slots=True)
class CapabilityRequest:
    """
    Provider-independent request submitted to the AI Runtime.

    Attributes:
        capability:
            Logical capability requested by the application.

        request_id:
            Correlation identifier for the execution.

        payload:
            Explicit capability input.

        metadata:
            Non-business execution metadata.

        requested_by:
            Component or application requesting execution.
    """

    capability: CapabilityType
    request_id: str
    payload: Mapping[str, Any]
    metadata: Mapping[str, Any] = field(
        default_factory=dict
    )
    requested_by: str = "application"

    def __post_init__(self) -> None:
        if not self.request_id.strip():
            raise ValueError(
                "Capability request_id cannot be empty."
            )

        if not self.requested_by.strip():
            raise ValueError(
                "Capability requested_by cannot be empty."
            )


@dataclass(frozen=True, slots=True)
class CapabilityResult:
    """
    Provider-independent result returned by the AI Runtime.

    The result contains execution information but does not expose
    provider-specific response objects.
    """

    request_id: str
    capability: CapabilityType
    status: CapabilityExecutionStatus
    output: Any = None
    error: str | None = None
    metadata: Mapping[str, Any] = field(
        default_factory=dict
    )

    @property
    def succeeded(self) -> bool:
        """
        Return True when execution completed successfully.
        """

        return (
            self.status
            == CapabilityExecutionStatus.COMPLETED
        )

    @property
    def failed(self) -> bool:
        """
        Return True when execution failed.
        """

        return (
            self.status
            == CapabilityExecutionStatus.FAILED
        )

    @property
    def rejected(self) -> bool:
        """
        Return True when execution was rejected before execution.
        """

        return (
            self.status
            == CapabilityExecutionStatus.REJECTED
        )

import pytest

class FakeLiveIntelligenceExecutor(
    CapabilityExecutor,
):
    """
    Fake executor used to test dependency injection.
    """

    @property
    def capability(self) -> CapabilityType:
        return CapabilityType.LIVE_INTELLIGENCE

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output={"ok": True},
        )


class FakeEvaluationExecutor(
    CapabilityExecutor,
):
    """
    Second fake executor used to verify multiple
    capability registrations.
    """

    @property
    def capability(self) -> CapabilityType:
        return CapabilityType.EVALUATION

    async def execute(
        self,
        request: CapabilityRequest,
    ) -> CapabilityResult:

        return CapabilityResult(
            request_id=request.request_id,
            capability=request.capability,
            status=CapabilityExecutionStatus.COMPLETED,
            output={"evaluated": True},
        )


def test_registry_registers_executor():

    executor = FakeLiveIntelligenceExecutor()

    registry = CapabilityRegistry()

    registry.register(executor)

    assert registry.contains(
        CapabilityType.LIVE_INTELLIGENCE
    )


def test_registry_resolves_registered_executor():

    executor = FakeLiveIntelligenceExecutor()

    registry = CapabilityRegistry(
        executors=[executor]
    )

    resolved = registry.resolve(
        CapabilityType.LIVE_INTELLIGENCE
    )

    assert resolved is executor


def test_registry_supports_multiple_capabilities():

    live_executor = FakeLiveIntelligenceExecutor()
    evaluation_executor = FakeEvaluationExecutor()

    registry = CapabilityRegistry(
        executors=[
            live_executor,
            evaluation_executor,
        ]
    )

    assert registry.resolve(
        CapabilityType.LIVE_INTELLIGENCE
    ) is live_executor

    assert registry.resolve(
        CapabilityType.EVALUATION
    ) is evaluation_executor


def test_duplicate_capability_registration_is_rejected():

    registry = CapabilityRegistry()

    registry.register(
        FakeLiveIntelligenceExecutor()
    )

    with pytest.raises(
        CapabilityAlreadyRegisteredError,
        match="already registered",
    ):
        registry.register(
            FakeLiveIntelligenceExecutor()
        )


def test_unknown_capability_resolution_is_rejected():

    registry = CapabilityRegistry()

    with pytest.raises(
        CapabilityNotRegisteredError,
        match="No executor registered",
    ):
        registry.resolve(
            CapabilityType.LEARNING_GENERATION
        )


def test_contains_returns_false_for_unknown_capability():

    registry = CapabilityRegistry()

    assert (
        registry.contains(
            CapabilityType.EVALUATION
        )
        is False
    )


def test_capabilities_returns_registered_capabilities():

    registry = CapabilityRegistry(
        executors=[
            FakeLiveIntelligenceExecutor(),
            FakeEvaluationExecutor(),
        ]
    )

    capabilities = registry.capabilities()

    assert (
        CapabilityType.LIVE_INTELLIGENCE
        in capabilities
    )

    assert (
        CapabilityType.EVALUATION
        in capabilities
    )

    assert (
        CapabilityType.LEARNING_GENERATION
        not in capabilities
    )
