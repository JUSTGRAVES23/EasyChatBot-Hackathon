async def get_current_principal(request: Request) -> Principal:
    """Resolve identity from the HttpOnly application session cookie."""

    session_id = request.cookies.get("easychatbot_session")

    repository = request.app.state.auth_repository

    provider = SessionIdentityProvider(
        session_id=session_id,
        repository=repository,
    )

    principal = await provider.get_current_principal()

    if principal is None or not principal.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authentication required.",
        )

    return principal

from __future__ import annotations

from app.domain.authorization.permissions import Permission, has_permission
from app.domain.identity.principal import Principal


class AuthenticationRequiredError(PermissionError):
    """No valid authenticated principal exists."""


class PermissionDeniedError(PermissionError):
    """Authenticated principal lacks a permission."""


class AuthorizationService:
    """Provider-neutral authorization boundary."""

    def require_authenticated(
        self,
        principal: Principal | None,
    ) -> Principal:
        if principal is None or not principal.is_active:
            raise AuthenticationRequiredError(
                "Authenticated application identity is required."
            )
        return principal

    def require_permission(
        self,
        *,
        principal: Principal | None,
        permission: Permission,
    ) -> Principal:
        authenticated = self.require_authenticated(principal)

        if not has_permission(
            roles=authenticated.roles,
            permission=permission,
        ):
            raise PermissionDeniedError(
                f"Permission required: {permission.value}."
            )

        return authenticated

from enum import StrEnum


class Permission(StrEnum):
    ANNOUNCEMENT_VIEW = "announcement.view"
    ANNOUNCEMENT_PUBLISH = "announcement.publish"
    FUNCTIONALITY_CONTROL = "functionality.control"
    INTERNAL_DATA_READ = "internal_data.read"


class Role(StrEnum):
    BETA_USER = "beta_user"
    OPERATOR = "operator"
    ADMIN = "admin"


ROLE_PERMISSIONS: dict[Role, frozenset[Permission]] = {
    Role.BETA_USER: frozenset(),
    Role.OPERATOR: frozenset(
        {
            Permission.ANNOUNCEMENT_VIEW,
            Permission.ANNOUNCEMENT_PUBLISH,
            Permission.FUNCTIONALITY_CONTROL,
            Permission.INTERNAL_DATA_READ,
        }
    ),
    Role.ADMIN: frozenset(
        {
            Permission.ANNOUNCEMENT_VIEW,
            Permission.ANNOUNCEMENT_PUBLISH,
            Permission.FUNCTIONALITY_CONTROL,
            Permission.INTERNAL_DATA_READ,
        }
    ),
}


def has_permission(
    *,
    roles: frozenset[str],
    permission: Permission,
) -> bool:
    return any(
        permission in ROLE_PERMISSIONS.get(
            Role(raw_role),
            frozenset(),
        )
        for raw_role in roles
        if raw_role in {role.value for role in Role}
    )

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone

from .template_models import PublicationStatus, PublicationType


_ALLOWED_TRANSITIONS = {
    PublicationStatus.DRAFT: {
        PublicationStatus.PENDING_APPROVAL,
        PublicationStatus.CANCELLED,
    },
    PublicationStatus.PENDING_APPROVAL: {
        PublicationStatus.APPROVED,
        PublicationStatus.CANCELLED,
    },
    PublicationStatus.APPROVED: {
        PublicationStatus.PUBLISHING,
        PublicationStatus.CANCELLED,
    },
    PublicationStatus.PUBLISHING: {
        PublicationStatus.PUBLISHED,
        PublicationStatus.FAILED,
    },
    PublicationStatus.FAILED: {
        PublicationStatus.PUBLISHING,
        PublicationStatus.CANCELLED,
    },
    PublicationStatus.PUBLISHED: set(),
    PublicationStatus.CANCELLED: set(),
}


@dataclass
class Publication:
    publication_id: str
    template_id: str
    template_version: int
    publication_type: PublicationType
    title: str
    content: str
    status: PublicationStatus = PublicationStatus.DRAFT
    created_at: datetime | None = None
    approved_at: datetime | None = None
    published_at: datetime | None = None
    failure_reason: str | None = None

    def __post_init__(self) -> None:
        if not self.publication_id.strip():
            raise ValueError("publication_id cannot be empty.")
        if not self.template_id.strip():
            raise ValueError("template_id cannot be empty.")
        if self.template_version < 1:
            raise ValueError("template_version must be >= 1.")
        if not self.title.strip():
            raise ValueError("title cannot be empty.")
        if not self.content.strip():
            raise ValueError("content cannot be empty.")
        if self.created_at is None:
            self.created_at = datetime.now(timezone.utc)

    def transition(self, target: PublicationStatus) -> None:
        if target not in _ALLOWED_TRANSITIONS[self.status]:
            raise ValueError(
                f"Invalid publication transition: "
                f"{self.status.value} -> {target.value}"
            )
        self.status = target
        now = datetime.now(timezone.utc)
        if target is PublicationStatus.APPROVED:
            self.approved_at = now
        elif target is PublicationStatus.PUBLISHED:
            self.published_at = now
            self.failure_reason = None

    def request_approval(self) -> None:
        self.transition(PublicationStatus.PENDING_APPROVAL)

    def approve(self) -> None:
        self.transition(PublicationStatus.APPROVED)

    def begin_publish(self) -> None:
        self.transition(PublicationStatus.PUBLISHING)

    def mark_failure(self, reason: str) -> None:
        if not reason.strip():
            raise ValueError("failure reason cannot be empty.")
        if self.status is not PublicationStatus.PUBLISHING:
            raise ValueError("A publication can only fail while publishing.")
        self.failure_reason = reason.strip()
        self.transition(PublicationStatus.FAILED)

    def publish_success(self) -> None:
        self.transition(PublicationStatus.PUBLISHED)

class SqlAlchemyPublicationRepository(PublicationRepository):
    """Async PostgreSQL adapter for the existing repository contract."""

    def __init__(
        self,
        session_factory: async_sessionmaker[AsyncSession],
    ) -> None:
        if session_factory is None:
            raise ValueError("session_factory cannot be None.")
        self._session_factory = session_factory

    def _to_domain(self, model: PublicationModel) -> Publication:
        return Publication(
            publication_id=model.publication_id,
            template_id=model.template_id,
            template_version=model.template_version,
            publication_type=PublicationType(model.publication_type),
            title=model.title,
            content=model.content,
            status=PublicationStatus(model.status),
            created_at=model.created_at,
            approved_at=model.approved_at,
            published_at=model.published_at,
            failure_reason=model.failure_reason,
        )

    def _from_domain(self, publication: Publication) -> PublicationModel:
        return PublicationModel(
            publication_id=publication.publication_id,
            template_id=publication.template_id,
            template_version=publication.template_version,
            publication_type=publication.publication_type.value,
            title=publication.title,
            content=publication.content,
            status=publication.status.value,
            created_at=publication.created_at or datetime.now(timezone.utc),
            approved_at=publication.approved_at,
            published_at=publication.published_at,
            failure_reason=publication.failure_reason,
        )

    async def save_async(self, publication: Publication) -> None:
        async with self._session_factory() as session:
            model = await session.get(
                PublicationModel,
                publication.publication_id,
            )
            if model is None:
                session.add(self._from_domain(publication))
            else:
                model.template_id = publication.template_id
                model.template_version = publication.template_version
                model.publication_type = publication.publication_type.value
                model.title = publication.title
                model.content = publication.content
                model.status = publication.status.value
                model.created_at = publication.created_at or model.created_at
                model.approved_at = publication.approved_at
                model.published_at = publication.published_at
                model.failure_reason = publication.failure_reason
            await session.commit()

    async def get_async(self, publication_id: str) -> Publication | None:
        async with self._session_factory() as session:
            model = await session.get(PublicationModel, publication_id)
            return self._to_domain(model) if model else None

    async def list_all_async(self) -> tuple[Publication, ...]:
        async with self._session_factory() as session:
            result = await session.execute(
                select(PublicationModel).order_by(PublicationModel.created_at)
            )
            return tuple(self._to_domain(model) for model in result.scalars().all())

    # The existing repository contract is synchronous. These methods make
    # accidental async DB use explicit rather than silently blocking.
    def save(self, publication: Publication) -> None:
        raise RuntimeError(
            "SqlAlchemyPublicationRepository requires its async boundary; "
            "use save_async()."
        )

    def get(self, publication_id: str) -> Publication | None:
        raise RuntimeError(
            "SqlAlchemyPublicationRepository requires its async boundary; "
            "use get_async()."
        )

    def list_all(self) -> tuple[Publication, ...]:
        raise RuntimeError(
            "SqlAlchemyPublicationRepository requires its async boundary; "
            "use list_all_async()."
        )

def create_template_editor_router(
    service: AnnouncementTemplateEditorService,
) -> APIRouter:
    """Create the HTTP adapter for the template-driven editor.

    Composition owns the service. This router never creates repositories,
    providers, Telegram clients, or publication executors.
    """

    if service is None:
        raise ValueError("service cannot be None.")

    router = APIRouter(
        prefix="/api/announcement-operations/template-editor",
        tags=["announcement-operations"],
    )

    @router.get("/templates")
    def list_templates() -> list[dict[str, Any]]:
        return [asdict(item) for item in service.list_approved_templates()]

    @router.post("/preview")
    def preview(request: PreviewRequest) -> dict[str, Any]:
        try:
            result = service.preview(
                template_id=request.template_id,
                values=request.values,
                version=request.template_version,
            )
        except (LookupError, ValueError) as exc:
            raise HTTPException(status_code=400, detail=str(exc)) from exc

        return asdict(result)

    @router.post("/drafts")
    def create_draft(request: CreateDraftRequest) -> dict[str, Any]:
        try:
            publication = service.create_draft(
                publication_id=request.publication_id or str(uuid4()),
                template_id=request.template_id,
                title=request.title,
                values=request.values,
                version=request.template_version,
            )
        except (LookupError, ValueError) as exc:
            raise HTTPException(status_code=400, detail=str(exc)) from exc

        return {
            "id": publication.publication_id,
            "title": publication.title,
            "content": publication.content,
            "template_id": publication.template_id,
            "template_version": publication.template_version,
            "publication_type": publication.publication_type.value,
            "status": publication.status.value,
            "created_at": publication.created_at.isoformat()
            if publication.created_at
            else None,
        }

    return router

class AsyncAnnouncementPublicationApprovalService:
    """Async PostgreSQL-backed approval application boundary.

    The domain model remains the authority for lifecycle transitions.
    This service never invokes WS-22 execution or Telegram.
    """

    def __init__(
        self,
        *,
        publications: AsyncPublicationStore,
        audit: PublicationApprovalAudit,
    ) -> None:
        if publications is None or audit is None:
            raise ValueError("publications and audit are required.")
        self._publications = publications
        self._audit = audit

    async def approve(
        self,
        *,

        actor: str,

            raise ValueError("actor cannot be empty.")

        publication = await self._publications.get_async(publication_id)
        if publication is None:
            raise LookupError(f"Unknown publication '{publication_id}'.")

        if publication.status is not PublicationStatus.PENDING_APPROVAL:
            # Use the domain transition so the canonical lifecycle error remains
            # authoritative and consistent with the existing domain model.
            publication.approve()

        else:
            publication.approve()

        approved_at = publication.approved_at or datetime.now(timezone.utc)

        await self._publications.save_async(publication)
        await self._audit.record_approval(
            publication=publication,
            actor=actor.strip(),
            occurred_at=approved_at,
        )

        return AsyncPublicationApprovalResult(
            publication=publication,
            actor=actor.strip(),
            approved_at=approved_at,
        )

async def _default_admin_authorizer() -> bool:
    return False


def create_announcement_operations_approval_router(
    *,
    approval_service: AsyncAnnouncementPublicationApprovalService,
    admin_authorizer: AdminAuthorizer = _default_admin_authorizer,
) -> APIRouter:
    if approval_service is None:
        raise ValueError("approval_service cannot be None.")
    if admin_authorizer is None:
        raise ValueError("admin_authorizer cannot be None.")

    router = APIRouter(
        prefix="/api/announcement-operations",
        tags=["announcement-operations"],
    )

    async def require_admin() -> None:
        try:
            authorized = await admin_authorizer()
        except Exception:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Administrator authorization required.",
            ) from None

        if not authorized:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Administrator authorization required.",
            )

    @router.post(
        "/publications/{publication_id}/approve",
        response_model=PublicationApprovalResponse,
        status_code=status.HTTP_200_OK,
        dependencies=[Depends(require_admin)],
    )
    async def approve_publication(
        publication_id: str,
        request: PublicationApprovalRequest,
    ) -> PublicationApprovalResponse:
        try:
            result = await approval_service.approve(
                publication_id=publication_id,
                actor=request.actor,
            )
        except LookupError as exc:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=str(exc),
            ) from exc
        except ValueError as exc:
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail=str(exc),
            ) from exc

        publication = result.publication
        return PublicationApprovalResponse(
            publication_id=publication.publication_id,
            status=publication.status.value,
            approved_at=(
                publication.approved_at.isoformat()
                if publication.approved_at is not None
                else None
            ),
        )

    return router

from __future__ import annotations

    PublicationExecutionErrorCode,
    PublicationExecutionResult,
)


class PublicationExecutionService:
    """Execute an approved publication through an injected publishing port.

    This service contains orchestration only. It does not know Telegram,
    HTTP, frontend concerns, credentials, or provider-specific APIs.
    """

    def __init__(
        self,
        *,
        repository: PublicationExecutionRepository,
        publisher: PublicationPublisher,
        idempotency_store: PublicationIdempotencyStore,
        audit_sink: PublicationAuditSink,
    ) -> None:
        if any(
            dependency is None
            for dependency in (
                repository,
                publisher,
                idempotency_store,
                audit_sink,
            )
        ):
            raise ValueError("All publication execution dependencies are required.")

        self._repository = repository
        self._publisher = publisher
        self._idempotency_store = idempotency_store
        self._audit_sink = audit_sink

    async def execute(
        self,
        command: PublishAnnouncementCommand,
    ) -> PublicationExecutionResult:
        """Execute one explicit publication request."""

        existing = await self._idempotency_store.get_result(
            command.idempotency_key
        )
        if existing is not None:
            return existing

        publication = await self._repository.get_for_execution(
            command.publication_id
        )

        if publication is None:
            result = PublicationExecutionResult.failed(
                publication_id=command.publication_id,
                idempotency_key=command.idempotency_key,
                error_code=PublicationExecutionErrorCode.INVALID_STATE,
                error_message="Publication is not available for execution.",
            )
            await self._record_result(command, result)
            return result

        if publication.status.lower() not in {"approved", "ready_to_publish"}:
            result = PublicationExecutionResult.failed(
                publication_id=command.publication_id,
                idempotency_key=command.idempotency_key,
                error_code=PublicationExecutionErrorCode.INVALID_STATE,
                error_message="Publication is not in an executable state.",
            )
            await self._record_result(command, result)
            return result

        try:
            provider_reference = await self._publisher.publish(
                publication=publication,
                idempotency_key=command.idempotency_key,
            )
        except Exception as exc:  # noqa: BLE001
            result = PublicationExecutionResult.failed(
                publication_id=command.publication_id,
                idempotency_key=command.idempotency_key,
                error_code=PublicationExecutionErrorCode.PROVIDER_ERROR,
                error_message=str(exc) or "Publication provider failed.",
            )
            await self._record_result(command, result)
            return result

        result = PublicationExecutionResult.succeeded(
            publication_id=command.publication_id,
            idempotency_key=command.idempotency_key,
            provider_reference=provider_reference,
        )
        await self._record_result(command, result)
        return result

    async def _record_result(
        self,
        command: PublishAnnouncementCommand,
        result: PublicationExecutionResult,
    ) -> None:
        await self._idempotency_store.store_result(result)
        await self._repository.record_execution(result)

        await self._audit_sink.record(
            PublicationAuditEvent(
                event_name="announcement_publication_execution",
                publication_id=result.publication_id,
                requested_by=command.requested_by,
                status=result.status,
                occurred_at=result.executed_at,
                idempotency_key=result.idempotency_key,
                provider_reference=result.provider_reference,
                error_code=result.error_code,
                safe_metadata={"execution_status": result.status.value},
            )
        )

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone
from typing import Protocol


class AnnouncementPublisher(Protocol):
    async def publish(
        self,
        *,
        channel_id: str,
        text: str,
    ) -> str:
        """Publish an announcement and return its external message ID."""


class PublicationDisabledError(RuntimeError):
    """Raised when controlled announcement publishing is disabled."""


@dataclass(frozen=True)
class PublicationRecord:
    channel_id: str
    message_id: str
    announcement_type: str
    published_at: datetime


class ControlledAnnouncementPublisher:
    """Application boundary enforcing the explicit Beta 1 publication switch."""

    def __init__(
        self,
        *,
        publisher: AnnouncementPublisher,
        publishing_enabled: bool,
    ) -> None:
        if publisher is None:
            raise ValueError("publisher cannot be None.")

        self._publisher = publisher
        self._publishing_enabled = publishing_enabled

    async def publish(
        self,
        *,
        channel_id: str,
        text: str,
        announcement_type: str,
    ) -> PublicationRecord:
        if not self._publishing_enabled:
            raise PublicationDisabledError(
                "Telegram announcement publishing is disabled."
            )

        message_id = await self._publisher.publish(
            channel_id=channel_id,
            text=text,
        )

        return PublicationRecord(
            channel_id=channel_id,
            message_id=message_id,
            announcement_type=announcement_type,
            published_at=datetime.now(timezone.utc),
        )

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )
    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )

"use client";

import { useEffect, useMemo, useState } from "react";
import {
  createTemplateDraft,
  getEditorTemplates,
  previewTemplate,
  type TemplateEditorOption,
} from "@/app/lib/api/announcementTemplateEditor";

export default function TemplateDrivenPublicationEditor() {
  const [templates, setTemplates] = useState<TemplateEditorOption[]>([]);
  const [templateId, setTemplateId] = useState("");
  const [values, setValues] = useState<Record<string, string>>({});
  const [title, setTitle] = useState("");
  const [preview, setPreview] = useState("");
  const [loadingTemplates, setLoadingTemplates] = useState(true);
  const [previewing, setPreviewing] = useState(false);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [message, setMessage] = useState<string | null>(null);

  const selectedTemplate = useMemo(
    () => templates.find((item) => item.template_id === templateId) ?? null,
    [templateId, templates],
  );

  useEffect(() => {
    getEditorTemplates()
      .then((items) => {
        setTemplates(items);
        if (items.length > 0) {
          setTemplateId(items[0].template_id);
        }
      })
      .catch((reason: unknown) => {
        setError(
          reason instanceof Error
            ? reason.message
            : "Announcement types could not be loaded.",
        );
      })
      .finally(() => setLoadingTemplates(false));
  }, []);

  useEffect(() => {
    if (!selectedTemplate) {
      setValues({});
      setPreview("");
      return;
    }

    const nextValues: Record<string, string> = {};
    for (const field of selectedTemplate.fields) {
      nextValues[field.name] = "";
    }

    setValues(nextValues);
    setPreview("");
    setError(null);
  }, [selectedTemplate]);

  const requiredFieldsMissing = selectedTemplate
    ? selectedTemplate.fields.some(
        (field) => field.required && !values[field.name]?.trim(),
      )
    : true;

  const canPreview =
    Boolean(selectedTemplate) &&
    Boolean(title.trim()) &&
    !requiredFieldsMissing;

  function updateValue(name: string, value: string) {
    setValues((current) => ({ ...current, [name]: value }));
    setMessage(null);
  }

  async function handlePreview() {
    if (!selectedTemplate || !canPreview) {
      return;
    }

    setPreviewing(true);
    setError(null);
    setMessage(null);

    try {
      const result = await previewTemplate({
        templateId: selectedTemplate.template_id,
        templateVersion: selectedTemplate.version,
        values,
      });
      setPreview(result.content);
    } catch (reason: unknown) {
      setError(
        reason instanceof Error
          ? reason.message
          : "Preview could not be generated.",
      );
    } finally {
      setPreviewing(false);
    }
  }

  async function handleSaveDraft() {
    if (!selectedTemplate || !canPreview) {
      return;
    }

    setSaving(true);
    setError(null);
    setMessage(null);

    try {
      const result = await createTemplateDraft({
        templateId: selectedTemplate.template_id,
        templateVersion: selectedTemplate.version,
        title: title.trim(),
        values,
      });

      setMessage(
        `Draft ${result.id} saved from ${result.template_id} v${result.template_version}.`,
      );
    } catch (reason: unknown) {
      setError(
        reason instanceof Error
          ? reason.message
          : "The draft could not be saved.",
      );
    } finally {
      setSaving(false);
    }
  }

  return (
    <main className="min-h-full p-6 lg:p-10">
      <div className="mb-8">
        <p className="text-xs font-bold uppercase tracking-[0.2em] text-fuchsia-300">
          Announcement Operations
        </p>
        <h1 className="mt-2 text-3xl font-bold text-white">
          Create publication
        </h1>
        <p className="mt-2 text-sm text-slate-400">
          Select an approved announcement type, complete its controlled fields,
          preview it, and save a draft.
        </p>
      </div>

      {error && (
        <div
          role="alert"
          className="mb-6 rounded-2xl border border-rose-500/30 bg-rose-500/10 p-4 text-sm text-rose-200"
        >
          {error}
        </div>
      )}

      {message && (
        <div className="mb-6 rounded-2xl border border-emerald-500/30 bg-emerald-500/10 p-4 text-sm text-emerald-200">
          {message}
        </div>
      )}

      <div className="grid gap-6 xl:grid-cols-2">
        <section className="rounded-2xl border border-slate-800 bg-slate-950/70 p-6">
          <h2 className="font-semibold text-white">Publication content</h2>

          <label
            htmlFor="announcement-type"
            className="mt-5 block text-xs font-semibold uppercase tracking-wider text-slate-500"
          >
            Announcement type
          </label>

          <select
            id="announcement-type"
            aria-label="Announcement type"
            value={templateId}
            disabled={loadingTemplates || templates.length === 0}
            onChange={(event) => setTemplateId(event.target.value)}
            className="mt-2 w-full rounded-xl border border-slate-700 bg-slate-900 px-4 py-3 text-sm text-white outline-none focus:border-fuchsia-500 disabled:opacity-50"
          >
            {loadingTemplates && (
              <option value="">Loading announcement typesâ€¦</option>
            )}
            {!loadingTemplates && templates.length === 0 && (
              <option value="">No announcement types available</option>
            )}
            {templates.map((template) => (
              <option key={template.template_id} value={template.template_id}>
                {template.name}
              </option>
            ))}
          </select>

          {selectedTemplate && (
            <div className="mt-3 rounded-xl border border-slate-800 bg-slate-900/60 p-4">
              <div className="flex flex-wrap items-center gap-2">
                <span className="rounded-full bg-fuchsia-500/10 px-2.5 py-1 text-xs font-semibold text-fuchsia-300">
                  {selectedTemplate.category}
                </span>
                <span className="text-xs text-slate-500">
                  Approved version {selectedTemplate.version}
                </span>
              </div>
              <p className="mt-2 text-sm text-slate-400">
                {selectedTemplate.description}
              </p>
            </div>
          )}

          <label
            htmlFor="announcement-title"
            className="mt-5 block text-xs font-semibold uppercase tracking-wider text-slate-500"
          >
            Publication title
          </label>
          <input
            id="announcement-title"
            value={title}
            onChange={(event) => setTitle(event.target.value)}
            className="mt-2 w-full rounded-xl border border-slate-700 bg-slate-900 px-4 py-3 text-sm text-white outline-none focus:border-fuchsia-500"
            placeholder="Announcement title"
          />

          {selectedTemplate?.fields.map((field) => (
            <div key={field.name}>
              <label
                htmlFor={`template-field-${field.name}`}
                className="mt-5 block text-xs font-semibold uppercase tracking-wider text-slate-500"
              >
                {field.label}
                {field.required && (
                  <span className="ml-1 text-fuchsia-300">*</span>
                )}
              </label>

              {field.multiline ? (
                <textarea
                  id={`template-field-${field.name}`}
                  value={values[field.name] ?? ""}
                  onChange={(event) =>
                    updateValue(field.name, event.target.value)
                  }
                  rows={5}
                  className="mt-2 w-full resize-y rounded-xl border border-slate-700 bg-slate-900 px-4 py-3 text-sm text-white outline-none focus:border-fuchsia-500"
                />
              ) : (
                <input
                  id={`template-field-${field.name}`}
                  value={values[field.name] ?? ""}
                  onChange={(event) =>
                    updateValue(field.name, event.target.value)
                  }
                  className="mt-2 w-full rounded-xl border border-slate-700 bg-slate-900 px-4 py-3 text-sm text-white outline-none focus:border-fuchsia-500"
                />
              )}
            </div>
          ))}

          {selectedTemplate && requiredFieldsMissing && (
            <p className="mt-4 text-xs text-amber-300">
              Complete all required fields before previewing or saving.
            </p>
          )}

          <div className="mt-5 flex flex-wrap gap-3">
            <button
              type="button"
              disabled={!canPreview || previewing}
              onClick={handlePreview}
              className="rounded-xl border border-fuchsia-500/50 px-5 py-3 text-sm font-semibold text-fuchsia-200 disabled:cursor-not-allowed disabled:opacity-40"
            >
              {previewing ? "Previewingâ€¦" : "Preview"}
            </button>

            <button
              type="button"
              disabled={!canPreview || saving}
              onClick={handleSaveDraft}
              className="rounded-xl bg-fuchsia-500 px-5 py-3 text-sm font-semibold text-white disabled:cursor-not-allowed disabled:opacity-40"
            >
              {saving ? "Savingâ€¦" : "Save draft"}
            </button>
          </div>
        </section>

        <section className="rounded-2xl border border-slate-800 bg-slate-950/70 p-6">
          <p className="text-xs font-bold uppercase tracking-wider text-fuchsia-300">
            Preview
          </p>

          <div className="mt-4 rounded-2xl border border-slate-800 bg-[#050816] p-6">
            <p className="text-xs uppercase tracking-wider text-slate-500">
              EasyChatBot Academy
            </p>
            <h2 className="mt-3 text-2xl font-bold text-white">
              {title || "Announcement title"}
            </h2>
            <p className="mt-4 whitespace-pre-wrap text-sm leading-7 text-slate-300">
              {preview ||
                "Complete the selected announcement type and choose Preview to see the rendered announcement."}
            </p>
          </div>
        </section>
      </div>
    </main>
  );
}

export interface TemplateEditorField {
  name: string;
  required: boolean;
  label: string;
  multiline: boolean;
}

export interface TemplateEditorOption {
  template_id: string;
  name: string;
  category: string;
  description: string;
  version: number;
  publication_type: string;
  fields: TemplateEditorField[];
}

export interface TemplateEditorPreview {
  template_id: string;
  template_version: number;
  publication_type: string;
  content: string;
}

export interface CreatedDraftPublication {
  id: string;
  title: string;
  content: string;
  template_id: string;
  template_version: number;
  publication_type: string;
  status: string;
  created_at: string | null;
}

const API_BASE =
  process.env.NEXT_PUBLIC_API_BASE_URL ?? "http://localhost:8000";

async function request<T>(
  path: string,
  init?: RequestInit,
): Promise<T> {
  const response = await fetch(`${API_BASE}${path}`, {
    ...init,
    headers: {
      Accept: "application/json",
      "Content-Type": "application/json",
      ...(init?.headers ?? {}),
    },
    cache: "no-store",
  });

  if (!response.ok) {
    let detail = `Announcement template API returned ${response.status}.`;

    try {
      const body = (await response.json()) as { detail?: string };
      if (body.detail) {
        detail = body.detail;
      }
    } catch {
      // Preserve the stable HTTP error for non-JSON responses.
    }

    throw new Error(detail);
  }

  return response.json() as Promise<T>;
}

export function getEditorTemplates(): Promise<TemplateEditorOption[]> {
  return request<TemplateEditorOption[]>(
    "/api/announcement-operations/template-editor/templates",
  );
}

export function previewTemplate(input: {
  templateId: string;
  templateVersion: number;
  values: Record<string, string>;
}): Promise<TemplateEditorPreview> {
  return request<TemplateEditorPreview>(
    "/api/announcement-operations/template-editor/preview",
    {
      method: "POST",
      body: JSON.stringify({
        template_id: input.templateId,
        template_version: input.templateVersion,
        values: input.values,
      }),
    },
  );
}

export function createTemplateDraft(input: {
  templateId: string;
  templateVersion: number;
  title: string;
  values: Record<string, string>;
}): Promise<CreatedDraftPublication> {
  return request<CreatedDraftPublication>(
    "/api/announcement-operations/template-editor/drafts",
    {
      method: "POST",
      body: JSON.stringify({
        template_id: input.templateId,
        template_version: input.templateVersion,
        title: input.title,
        values: input.values,
      }),
    },
  );
}
