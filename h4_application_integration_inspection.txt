"""
EasyChatBot Hackathon — Podcast Discovery Agent.

The Agent converts explicit learner context and preference memory into
provider-neutral search intent.

It does not call Apple/Spotify, write memory, select a provider, or invent
podcast metadata.
"""

from __future__ import annotations

import re

from src.ai_runtime.memory.models import MemoryEntry, MemoryType
from src.ai_runtime.podcast.models import (
    LearningContext,
    PodcastSearchIntent,
)


class PodcastDiscoveryAgent:
    """Build provider-neutral podcast discovery intent."""

    _STOP_WORDS = {
        "a", "an", "and", "are", "be", "for", "from", "i", "in", "is",
        "like", "of", "on", "podcasts", "prefers", "prefer", "the", "to",
        "with", "you", "your",
    }

    def build_search_intent(
        self,
        *,
        user_id: str,
        learning_context: LearningContext,
        memories: list[MemoryEntry],
        max_results: int = 10,
    ) -> PodcastSearchIntent:
        """
        Convert explicit preferences + learning-world context into intent.

        Only PREFERENCE memories are used, matching the existing
        EasyChatBot discovery contract.
        """

        if not user_id.strip():
            raise ValueError("Podcast Agent user_id cannot be empty.")

        preferences = [
            memory
            for memory in memories
            if memory.user_id == user_id
            and memory.memory_type == MemoryType.PREFERENCE
        ]

        topics: list[str] = []
        keywords: list[str] = []

        for memory in preferences:
            topics.extend(
                self._metadata_terms(
                    memory.metadata,
                    "topics",
                    "interests",
                    "preferred_topics",
                )
            )
            keywords.extend(
                self._metadata_terms(
                    memory.metadata,
                    "keywords",
                )
            )

            if not topics and not keywords:
                keywords.extend(
                    self._content_keywords(memory.content)
                )

        # The learning world is always part of the search context.
        topics.append(learning_context.world_id)

        # Target language is deliberately a hard eligibility constraint,
        # not just another search keyword.
        return PodcastSearchIntent(
            user_id=user_id,
            learning_context=learning_context,
            topics=tuple(self._dedupe(topics)),
            keywords=tuple(self._dedupe(keywords)),
            language_required=True,
            max_results=max_results,
        )

    @staticmethod
    def _metadata_terms(
        metadata: dict[str, str],
        *keys: str,
    ) -> list[str]:
        terms: list[str] = []
        for key in keys:
            raw = metadata.get(key, "")
            if raw:
                terms.extend(
                    item.strip()
                    for item in re.split(r"[,;|]", raw)
                    if item.strip()
                )
        return terms

    @classmethod
    def _content_keywords(cls, content: str) -> list[str]:
        words = re.findall(
            r"[A-Za-zÀ-ÿ0-9][A-Za-zÀ-ÿ0-9'-]{2,}",
            content.lower(),
        )
        return [
            word for word in words
            if word not in cls._STOP_WORDS
        ][:8]

    @staticmethod
    def _dedupe(values: list[str]) -> list[str]:
        seen: set[str] = set()
        result: list[str] = []
        for value in values:
            key = value.strip().lower()
            if key and key not in seen:
                seen.add(key)
                result.append(value.strip())
        return result

"""
EasyChatBot Hackathon — personalised podcast recommendation service.

This is the application-level selection boundary.

Language is a HARD filter because EasyChatBot is a language-learning
platform. Preference/topic relevance is evaluated only after a candidate
is language-eligible.
"""

from __future__ import annotations

import re
from uuid import uuid4

from src.ai_runtime.podcast.models import (
    PodcastRecommendation,
    PodcastSearchIntent,
    PodcastSearchResult,
    normalise_language,
)
from src.ai_runtime.podcast.providers import PodcastProvider


class PodcastRecommendationService:
    """Search providers and select a personalised candidate."""

    def __init__(self, providers: list[PodcastProvider]) -> None:
        if not providers:
            raise ValueError(
                "PodcastRecommendationService requires at least one provider."
            )
        self._providers = list(providers)

    def recommend(
        self,
        *,
        intent: PodcastSearchIntent,
        character_id: str,
        character_name: str,
        character_message: str,
    ) -> PodcastRecommendation:
        candidates: list[PodcastSearchResult] = []

        for provider in self._providers:
            candidates.extend(provider.search(intent))

        ranked = self.rank_candidates(
            candidates=candidates,
            intent=intent,
        )

        if not ranked:
            raise LookupError(
                "No podcast matched the learner's language and preferences."
            )

        result, score, reason = ranked[0]

        return PodcastRecommendation(
            recommendation_id=str(uuid4()),
            user_id=intent.user_id,
            character_id=character_id,
            character_name=character_name,
            character_message=character_message,
            result=result,
            relevance_reason=reason,
            match_score=score,
        )

    @classmethod
    def rank_candidates(
        cls,
        *,
        candidates: list[PodcastSearchResult],
        intent: PodcastSearchIntent,
    ) -> list[tuple[PodcastSearchResult, float, str]]:
        ranked: list[tuple[PodcastSearchResult, float, str]] = []

        target_language = normalise_language(
            intent.learning_context.target_language
        )

        preference_terms = [
            term.lower()
            for term in list(intent.topics) + list(intent.keywords)
            if term.strip()
        ]

        for candidate in candidates:
            candidate_languages = {
                normalise_language(language)
                for language in candidate.languages
            }

            # Critical learning-platform rule:
            # unknown/wrong-language content is not eligible.
            if (
                intent.language_required
                and target_language not in candidate_languages
            ):
                continue

            haystack = " ".join(
                [
                    candidate.title,
                    candidate.description,
                    candidate.publisher,
                ]
            ).lower()

            matched_terms = [
                term
                for term in preference_terms
                if cls._term_matches(term, haystack)
            ]

            score = 50.0

            if matched_terms:
                score += min(30.0, len(matched_terms) * 10.0)

            if intent.learning_context.world_id.lower() in haystack:
                score += 10.0

            level = intent.learning_context.learner_level.lower()
            if level and level in haystack:
                score += 10.0

            reason = cls._build_reason(
                language=intent.learning_context.target_language,
                matched_terms=matched_terms,
                world_id=intent.learning_context.world_id,
            )

            ranked.append(
                (candidate, min(score, 100.0), reason)
            )

        ranked.sort(
            key=lambda item: (
                item[1],
                item[0].published_at or "",
            ),
            reverse=True,
        )

        return ranked[: intent.max_results]

    @staticmethod
    def _term_matches(term: str, haystack: str) -> bool:
        term = term.strip().lower()
        if not term:
            return False
        if term in haystack:
            return True

        words = [
            word
            for word in re.findall(r"[a-z0-9À-ÿ'-]+", term)
            if len(word) > 2
        ]
        return bool(words) and any(
            word in haystack
            for word in words
        )

    @staticmethod
    def _build_reason(
        *,
        language: str,
        matched_terms: list[str],
        world_id: str,
    ) -> str:
        reasons = [f"It is available in {language}."]

        if matched_terms:
            reasons.append(
                "It matches your interests in "
                + ", ".join(matched_terms[:3])
                + "."
            )

        reasons.append(
            f"It fits your {world_id} learning world."
        )

        return " ".join(reasons)

"""
Deterministic integration demo:
recommendation -> character -> podcast learning challenge -> result.
"""

from __future__ import annotations

from src.ai_runtime.memory.models import MemoryEntry, MemoryType
from src.ai_runtime.podcast.agent import PodcastDiscoveryAgent
from src.ai_runtime.podcast.challenge_presenter import (
    PodcastLearningChallengePresenter,
)
from src.ai_runtime.podcast.challenge_service import (
    PodcastLearningChallengeService,
)
from src.ai_runtime.podcast.models import (
    LearningContext,
    PodcastProviderName,
    PodcastSearchIntent,
    PodcastSearchResult,
)
from src.ai_runtime.podcast.service import PodcastRecommendationService


class DemoProvider:
    def search(
        self,
        intent: PodcastSearchIntent,
    ) -> list[PodcastSearchResult]:
        return [
            PodcastSearchResult(
                provider=PodcastProviderName.APPLE,
                external_id="apple-fr-001",
                title="L'IA en Europe",
                description=(
                    "Artificial intelligence and European regulation "
                    "in French."
                ),
                publisher="Demo Publisher",
                languages=("fr",),
                url="https://podcasts.apple.com/demo/ai-europe",
                metadata={
                    "learner_level": "B2",
                },
            )
        ]


def main() -> None:
    memory = MemoryEntry(
        user_id="demo-user",
        memory_type=MemoryType.PREFERENCE,
        content="Prefers technology and current affairs.",
        metadata={
            "topics": (
                "artificial intelligence, European regulation"
            ),
        },
    )

    learning_context = LearningContext(
        world_id="French World",
        target_language="fr",
        learner_level="B2",
        learning_goal="Learn French through current events.",
        country_code="LU",
    )

    intent = PodcastDiscoveryAgent().build_search_intent(
        user_id="demo-user",
        learning_context=learning_context,
        memories=[memory],
    )

    recommendation = PodcastRecommendationService(
        [DemoProvider()]
    ).recommend(
        intent=intent,
        character_id="yeti",
        character_name="Yeti",
        character_message=(
            "I found a podcast I think you'll enjoy. "
            "Listen to it, then let's turn it into a French challenge!"
        ),
    )

    challenge = PodcastLearningChallengeService().create_challenge(
        recommendation=recommendation,
    )

    result = PodcastLearningChallengeService().evaluate(
        challenge=challenge,
        answers=(0, 0),
    )

    presenter = PodcastLearningChallengePresenter()

    print()
    print("=" * 72)
    print("EASYCHATBOT — PODCAST → LEARNING CHALLENGE")
    print("=" * 72)
    print()
    print("PERSONALISED RECOMMENDATION")
    print("-" * 72)
    print(f"{recommendation.character_name}:")
    print(f'"{recommendation.character_message}"')
    print()
    print(
        f"[PASS] Podcast selected: "
        f"{recommendation.result.title}"
    )
    print(
        f"[PASS] Language: "
        f"{recommendation.result.languages[0]}"
    )
    print(
        f"[PASS] Preference relevance: "
        f"{recommendation.relevance_reason}"
    )

    print(
        presenter.render(challenge)
    )

    print(
        presenter.render_result(result)
    )

    print()
    print("=" * 72)
    print("PODCAST LEARNING CHALLENGE: SUCCESS")
    print("=" * 72)
    print()


if __name__ == "__main__":
    main()

"""
Hackathon Discovery Service.

This module provides the application-level orchestration boundary
for personalised product discovery.

Architecture:

Discovery inputs
    |
    +--> DealCard
    |
    +--> Memory context
    |
    +--> Character context
    |
    v
DiscoveryService
    |
    v
FlashDealCard

Responsibilities:
- orchestrate existing discovery components
- apply explicit memory context
- apply explicit character context
- create the user-facing FlashDealCard
- preserve domain data and identifiers

Non-responsibilities:
- provider SDK calls
- web searching
- memory persistence
- payment processing
- vendor APIs
- autonomous purchasing
- character business logic

The service deliberately receives already-discovered DealCard data.
Provider integration remains in the provider/agent layers.
"""

from __future__ import annotations

from dataclasses import dataclass

from src.ai_runtime.discovery.flash_deal_card import (
    FlashDealCard,
)

from src.ai_runtime.discovery.models import (
    DealCard,
)

from src.ai_runtime.memory.models import (
    MemoryEntry,
    MemoryType,
)


@dataclass(frozen=True)
class CharacterContext:
    """
    Explicit character presentation context.

    The character is a presentation/personality concern.
    It does not perform discovery or commerce operations.
    """

    character_id: str
    character_name: str
    message: str

    def __post_init__(self) -> None:
        if not self.character_id.strip():
            raise ValueError(
                "Character context character_id cannot be empty."
            )

        if not self.character_name.strip():
            raise ValueError(
                "Character context character_name cannot be empty."
            )

        if not self.message.strip():
            raise ValueError(
                "Character context message cannot be empty."
            )


@dataclass(frozen=True)
class DiscoveryRequest:
    """
    Application-level request for personalised discovery.

    The request contains already-discovered product information
    and explicit user context.

    No external search is performed by DiscoveryService.
    """

    user_id: str
    deal: DealCard
    character: CharacterContext
    product_image_url: str
    tagline: str
    highlights: tuple[str, ...]

    def __post_init__(self) -> None:
        if not self.user_id.strip():
            raise ValueError(
                "Discovery request user_id cannot be empty."
            )

        if not self.product_image_url.strip():
            raise ValueError(
                "Discovery request product_image_url "
                "cannot be empty."
            )

        if not self.tagline.strip():
            raise ValueError(
                "Discovery request tagline cannot be empty."
            )

        if not self.highlights:
            raise ValueError(
                "Discovery request must contain at least "
                "one highlight."
            )


class DiscoveryService:
    """
    Orchestrates personalised discovery.

    This service intentionally contains no provider logic and no
    persistence logic. It composes the existing DealCard,
    MemoryEntry and FlashDealCard contracts.
    """

    def create_flash_deal(
        self,
        request: DiscoveryRequest,
        memories: list[MemoryEntry] | None = None,
    ) -> FlashDealCard:
        """
        Create a personalised FlashDealCard.

        Memory is read-only input to this operation.

        Preference memory is used to generate the personal
        relevance explanation. Other memory types are not
        automatically treated as preferences.
        """

        memory_context = memories or []

        relevance_reason = self._build_relevance_reason(
            memory_context
        )

        return FlashDealCard.from_deal(
            deal=request.deal,
            character_id=request.character.character_id,
            character_name=request.character.character_name,
            character_message=request.character.message,
            relevance_reason=relevance_reason,
            product_image_url=request.product_image_url,
            tagline=request.tagline,
            highlights=request.highlights,
            metadata={
                "user_id": request.user_id,
                "discovery_service": "hackathon",
            },
        )

    @staticmethod
    def _build_relevance_reason(
        memories: list[MemoryEntry],
    ) -> str:
        """
        Build a deterministic relevance explanation.

        Only explicit preference memories are used.

        This is deliberately deterministic for the hackathon;
        an LLM-based personalisation layer can be introduced later
        behind an explicit interface.
        """

        preferences = [
            memory
            for memory in memories
            if memory.memory_type
            == MemoryType.PREFERENCE
        ]

        if not preferences:
            return (
                "This discovery may be relevant to your "
                "interests."
            )

        preference = preferences[0].content.strip()

        return (
            "This may be relevant because you "
            f"{preference.rstrip('.') }."
        )

"""
Hackathon Discovery Notification.

Defines the lightweight user notification produced from a
personalised FlashDealCard.

Architecture:

DiscoveryService
      |
      v
FlashDealCard
      |
      v
DiscoveryNotification
      |
      +--> Telegram
      +--> Web UI
      +--> Future mobile UI

The notification layer is presentation/delivery oriented.

It does not:
- search for products
- access providers
- decide recommendations
- access memory directly
- process payments
- execute purchases
- build a shopping dashboard

The FlashDealCard remains the source of truth for the
discovered offer.
"""

from __future__ import annotations

from dataclasses import dataclass

from src.ai_runtime.discovery.flash_deal_card import (
    FlashDealCard,
)


@dataclass(frozen=True)
class DiscoveryNotification:
    """
    Lightweight character-driven notification.

    Designed for a single discovery event rather than a
    persistent shopping dashboard.
    """

    notification_id: str
    user_id: str
    character_id: str
    character_name: str
    title: str
    message: str
    deal: FlashDealCard

    @classmethod
    def from_flash_deal(
        cls,
        *,
        notification_id: str,
        user_id: str,
        deal: FlashDealCard,
    ) -> "DiscoveryNotification":
        """
        Create a notification from an existing FlashDealCard.

        No discovery or commerce logic occurs here.
        """

        if not notification_id.strip():
            raise ValueError(
                "Notification notification_id cannot be empty."
            )

        if not user_id.strip():
            raise ValueError(
                "Notification user_id cannot be empty."
            )

        return cls(
            notification_id=notification_id,
            user_id=user_id,
            character_id=deal.character_id,
            character_name=deal.character_name,
            title=(
                f"{deal.character_name} found something for you"
            ),
            message=deal.character_message,
            deal=deal,
        )

    @property
    def product_name(self) -> str:
        """Return the discovered product name."""

        return self.deal.product_name

    @property
    def relevance_reason(self) -> str:
        """Return the personalisation explanation."""

        return self.deal.relevance_reason

    @property
    def get_deal_url(self) -> str:
        """Return the controlled commerce destination."""

        return self.deal.get_deal_url

    @property
    def more_information_url(self) -> str:
        """Return the vendor information destination."""

        return self.deal.more_information_url

    def render(self) -> str:
        """
        Render a compact notification suitable for a chat UI.

        The full FlashDealCard remains available for the richer
        product presentation.
        """

        discount = (
            self.deal.display_discount
            or "NEW FIND"
        )

        return "\n".join(
            [
                f"{self.character_name}:",
                self.message,
                "",
                f"{discount} — {self.product_name}",
                f"{self.deal.currency} "
                f"{self.deal.current_price:.2f}",
                "",
                "Why this may interest you:",
                self.relevance_reason,
            ]
        )
