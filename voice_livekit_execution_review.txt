"""
LiveKit Agent Dispatcher.

Infrastructure adapter responsible for dispatching
an approved EasyChatBot voice call to the LiveKit
realtime agent worker.

Responsibilities:
- Create a LiveKit agent dispatch.
- Target the EasyChatBot voice worker.
- Pass approved voice-call metadata to the worker.
- Keep LiveKit API details inside infrastructure.

Does NOT:
- Decide whether a call is authorized.
- Create voice-call business state.
- Manage application lifecycle.
- Resolve worlds or characters.
- Execute AI agents.
- Persist conversations.
- Generate participant tokens.
"""

from __future__ import annotations


import json
import logging
import os


from dataclasses import dataclass


from livekit import api


logger = logging.getLogger(
    __name__
)


@dataclass(
    frozen=True,
    slots=True,
)
class LiveKitAgentDispatcher:
    """
    Infrastructure adapter for LiveKit Agent dispatch.

    The application layer supplies an already-approved
    voice-call context. This adapter translates that
    context into a LiveKit AgentDispatch request.

    LiveKit credentials are read from the environment
    and remain inside the infrastructure boundary.
    """

    agent_name: str = "easychatbot-voice"


    def dispatch(
        self,
        *,
        room_name: str,
        learner_id: str,
        world_id: str,
        character_id: str,
        language_code: str,
    ) -> str:
        """
        Dispatch the EasyChatBot voice agent to a room.

        Parameters
        ----------
        room_name:
            LiveKit room that the agent must join.

        learner_id:
            Approved learner identifier.

        world_id:
            Active learning world.

        character_id:
            Approved AI character.

        language_code:
            Active interaction language.

        Returns
        -------
        str
            LiveKit dispatch identifier.

        Raises
        ------
        ValueError
            If required values are empty or credentials
            are not configured.
        """

        self._validate_required_value(
            room_name,
            "room_name",
        )

        self._validate_required_value(
            learner_id,
            "learner_id",
        )

        self._validate_required_value(
            world_id,
            "world_id",
        )

        self._validate_required_value(
            character_id,
            "character_id",
        )

        self._validate_required_value(
            language_code,
            "language_code",
        )


        api_key = os.getenv(
            "LIVEKIT_API_KEY",
        )

        api_secret = os.getenv(
            "LIVEKIT_API_SECRET",
        )

        livekit_url = os.getenv(
            "LIVEKIT_URL",
        )


        if not api_key:
            raise ValueError(
                "LIVEKIT_API_KEY is not configured."
            )


        if not api_secret:
            raise ValueError(
                "LIVEKIT_API_SECRET is not configured."
            )


        if not livekit_url:
            raise ValueError(
                "LIVEKIT_URL is not configured."
            )


        metadata = {
            "learner_id": learner_id.strip(),
            "world_id": world_id.strip(),
            "character_id": character_id.strip(),
            "language_code": language_code.strip(),
        }


        metadata_json = json.dumps(
            metadata,
            separators=(
                ",",
                ":",
            ),
        )


        logger.info(
            "Dispatching EasyChatBot LiveKit agent",
            extra={
                "room_name": room_name,
                "agent_name": self.agent_name,
                "learner_id": learner_id,
                "world_id": world_id,
                "character_id": character_id,
                "language_code": language_code,
            },
        )


        livekit_api = api.LiveKitAPI(
            url=livekit_url,
            api_key=api_key,
            api_secret=api_secret,
        )


        try:
            dispatch = (
                livekit_api
                .agent_dispatch
                .create_dispatch(
                    api.CreateAgentDispatchRequest(
                        agent_name=(
                            self.agent_name
                        ),
                        room=room_name.strip(),
                        metadata=metadata_json,
                    )
                )
            )


            logger.info(
                "LiveKit agent dispatch created",
                extra={
                    "dispatch_id": (
                        dispatch.id
                    ),
                    "room_name": room_name,
                    "agent_name": (
                        self.agent_name
                    ),
                },
            )


            return dispatch.id

        finally:
            livekit_api.aclose()


    @staticmethod
    def _validate_required_value(
        value: str,
        field_name: str,
    ) -> None:
        """
        Validate a required string field.
        """

        if not isinstance(
            value,
            str,
        ):
            raise ValueError(
                f"{field_name} must be a string."
            )


        if not value.strip():
            raise ValueError(
                f"{field_name} cannot be empty."
            )


__all__ = [
    "LiveKitAgentDispatcher",
]

"""
LiveKit Voice Worker.

Infrastructure entry point for realtime voice sessions.

Responsibilities:
- Receive a LiveKit job context.
- Read application-approved job metadata.
- Convert metadata into a voice call request.
- Build the existing EasyChatBot voice execution context.
- Build the existing UAI InteractionContext.
- Create the LiveKit realtime character agent.
- Start the LiveKit AgentSession.

Does NOT:
- Own EasyChatBot business logic.
- Resolve worlds or characters directly.
- Decide learner authorization.
- Persist conversations.
- Manage call lifecycle state.
- Bypass the EasyChatBot Runtime.
"""

from __future__ import annotations


import json
import logging


from dataclasses import dataclass


from livekit.agents import (
    AgentServer,
    JobContext,
    WorkerOptions,
    cli,
)


from app.application.uai.voice.voice_call_context_factory import (
    VoiceCallContextFactory,
)


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.voice.voice_interaction_context_service import (
    VoiceInteractionContextService,
)


from app.application.uai.voice.voice_dependencies import (
    build_voice_interaction_context_service,
)


from app.infrastructure.voice.livekit_realtime_agent import (
    LiveKitRealtimeAgent,
)


logger = logging.getLogger(__name__)


@dataclass(
    frozen=True,
    slots=True,
)
class LiveKitVoiceWorker:
    """
    Infrastructure worker for EasyChatBot realtime voice.

    Application services are injected so the worker does not
    construct domain/application dependencies itself.
    """

    context_factory: VoiceCallContextFactory

    interaction_context_service: (
        VoiceInteractionContextService
    )

    async def run(
        self,
        *,
        ctx: JobContext,
    ) -> None:
        """
        Connect to LiveKit and start the EasyChatBot
        realtime character session.
        """

        metadata = self._read_job_metadata(
            ctx,
        )


        request = self._create_voice_call_request(
            metadata,
        )


        execution_context = (
            self.context_factory.create(
                request=request,
            )
        )


        interaction_context = (
            self.interaction_context_service.build(
                context=execution_context,
            )
        )


        character = (
            interaction_context.character
        )


        instructions = (
            "You are "
            f"{character.name}, "
            f"a French learning character. "
            f"Your personality reference is "
            f"{character.personality_reference}. "
            f"Your teaching style is "
            f"{character.teaching_style}. "
            "Speak naturally, conversationally, "
            "and help the learner practice the "
            "target language."
        )


        realtime_agent = (
            LiveKitRealtimeAgent(
                context=interaction_context,
                instructions=instructions,
            )
        )


        ctx.log_context_fields = {
            "room_name": ctx.room.name,
            "learner_id": (
                request.learner_id
            ),
            "world_id": (
                request.world_id
            ),
            "character_id": (
                request.character_id
            ),
        }


        logger.info(
            "Starting EasyChatBot LiveKit voice agent",
            extra={
                "room_name": ctx.room.name,
                "learner_id": request.learner_id,
                "world_id": request.world_id,
                "character_id": request.character_id,
            },
        )


        await ctx.connect()


        agent = (
            realtime_agent.create_agent()
        )


        session = (
            realtime_agent.create_session()
        )


        await session.start(
            agent=agent,
            room=ctx.room,
        )


    @staticmethod
    def _read_job_metadata(
        ctx: JobContext,
    ) -> dict[str, object]:
        """
        Read and validate LiveKit job metadata.

        The metadata is supplied by the application when
        the LiveKit job is dispatched.
        """

        raw_metadata = (
            ctx.job.metadata
        )


        if not raw_metadata:
            raise ValueError(
                "LiveKit job metadata is required."
            )


        try:
            metadata = json.loads(
                raw_metadata,
            )

        except json.JSONDecodeError as exc:
            raise ValueError(
                "LiveKit job metadata must be valid JSON."
            ) from exc


        if not isinstance(
            metadata,
            dict,
        ):
            raise ValueError(
                "LiveKit job metadata must be a JSON object."
            )


        return metadata


    @staticmethod
    def _create_voice_call_request(
        metadata: dict[str, object],
    ) -> VoiceCallRequest:
        """
        Convert validated LiveKit job metadata
        into the existing application request contract.
        """

        required_fields = (
            "learner_id",
            "world_id",
            "character_id",
            "language_code",
        )


        values: dict[str, str] = {}


        for field in required_fields:

            value = metadata.get(
                field,
            )


            if not isinstance(
                value,
                str,
            ):
                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' must be a string."
                )


            if not value.strip():
                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' cannot be empty."
                )


            values[field] = value.strip()


        return VoiceCallRequest(
            learner_id=values["learner_id"],
            world_id=values["world_id"],
            character_id=values["character_id"],
            language_code=values["language_code"],
        )


server = AgentServer()


_worker = LiveKitVoiceWorker(
    context_factory=VoiceCallContextFactory(),
    interaction_context_service=(
        build_voice_interaction_context_service()
    ),
)


@server.rtc_session(
    agent_name="easychatbot-voice",
)
async def entrypoint(
    ctx: JobContext,
) -> None:
    """
    LiveKit Agents entrypoint.
    """

    await _worker.run(
        ctx=ctx,
    )


def run() -> None:
    """
    Start the LiveKit Agents worker.
    """

    cli.run_app(
        WorkerOptions(
            agent_name="easychatbot-voice",
        )
    )


__all__ = [
    "LiveKitVoiceWorker",
    "entrypoint",
    "run",
]

"""
LiveKit Realtime Agent Infrastructure.

Infrastructure boundary between LiveKit Agents and EasyChatBot.

Responsibilities:
- Define the LiveKit Agents character wrapper.
- Create an AgentSession.
- Keep LiveKit Agents implementation details inside infrastructure.
- Expose a small provider-independent application-facing boundary.

Does NOT:
- Own EasyChatBot business logic.
- Manage call lifecycle state.
- Persist conversations.
- Modify production data.
- Bypass the EasyChatBot Runtime.
- Directly manage application authorization.
"""

from __future__ import annotations


from dataclasses import dataclass


from livekit.agents import (
    Agent,
    AgentSession,
)


from app.domain.uai.context.interaction_context import (
    InteractionContext,
)


@dataclass(
    frozen=True,
    slots=True,
)
class LiveKitRealtimeAgent:
    """
    Infrastructure adapter for a LiveKit realtime character.

    The adapter creates the LiveKit Agents objects required for
    realtime voice execution while keeping EasyChatBot character
    context outside the LiveKit-specific implementation.

    LiveKit Agents remains an infrastructure/realtime boundary.
    EasyChatBot remains responsible for application and AI runtime
    orchestration.
    """

    context: InteractionContext

    instructions: str


    def create_agent(
        self,
    ) -> Agent:
        """
        Create the LiveKit Agent representation of the character.

        Character-specific instructions are supplied explicitly by
        the EasyChatBot application/runtime layer.
        """

        return Agent(
            instructions=self.instructions,
        )


    def create_session(
        self,
    ) -> AgentSession:
        """
        Create a LiveKit AgentSession.

        Provider-specific STT, LLM and TTS configuration is deliberately
        not hard-coded here.

        Composition/root configuration should inject those capabilities
        when the realtime voice worker is assembled.
        """

        return AgentSession()


    def character_id(
        self,
    ) -> str:
        """
        Return the active EasyChatBot character identifier.
        """

        return (
            self.context
            .character
            .character_id
        )


    def language_code(
        self,
    ) -> str:
        """
        Return the active interaction language.
        """

        return (
            self.context
            .world
            .language_code
        )


__all__ = [
    "LiveKitRealtimeAgent",
]

"""
LiveKit Token Service.

Infrastructure service responsible for generating
real LiveKit participant access tokens.

Responsibilities:
- Generate signed LiveKit JWT access tokens.
- Grant participant access to a specific room.
- Keep LiveKit credentials inside infrastructure.
- Return provider-independent VoiceConnectionToken.

Does NOT:
- Handle HTTP.
- Manage application workflows.
- Decide business authorization.
- Persist tokens.
- Execute AI agents.
"""

from __future__ import annotations


from datetime import (
    datetime,
    timedelta,
    timezone,
)


import os
import uuid


from dotenv import load_dotenv


from livekit import api


from app.domain.uai.voice.voice_connection_token import (
    VoiceConnectionToken,
)


load_dotenv()



class LiveKitTokenService:
    """
    Real LiveKit access-token infrastructure adapter.
    """



    def create_token(
        self,
        *,
        room_name: str,
        identity: str | None = None,
        expires_minutes: int = 30,
    ) -> VoiceConnectionToken:
        """
        Create a signed LiveKit participant token.

        Parameters
        ----------
        room_name:
            LiveKit room the participant may join.

        identity:
            Unique LiveKit participant identity.

        expires_minutes:
            Token validity period.
        """



        if not room_name.strip():

            raise ValueError(
                "room_name cannot be empty."
            )



        if expires_minutes <= 0:

            raise ValueError(
                "expires_minutes must be positive."
            )



        participant_identity = (
            identity.strip()
            if identity is not None
            else f"learner-{uuid.uuid4()}"
        )



        if not participant_identity:

            raise ValueError(
                "identity cannot be empty."
            )



        api_key = os.getenv(
            "LIVEKIT_API_KEY"
        )



        api_secret = os.getenv(
            "LIVEKIT_API_SECRET"
        )



        if not api_key:

            raise RuntimeError(
                "LIVEKIT_API_KEY is not configured."
            )



        if not api_secret:

            raise RuntimeError(
                "LIVEKIT_API_SECRET is not configured."
            )



        expires_at = (
            datetime.now(
                timezone.utc
            )
            + timedelta(
                minutes=expires_minutes,
            )
        )



        access_token = (
            api.AccessToken(
                api_key=api_key,
                api_secret=api_secret,
            )
            .with_identity(
                participant_identity,
            )
            .with_name(
                participant_identity,
            )
            .with_ttl(
                timedelta(
                    minutes=expires_minutes,
                )
            )
            .with_grants(
                api.VideoGrants(
                    room_join=True,
                    room=room_name,
                    can_publish=True,
                    can_subscribe=True,
                    can_publish_data=True,
                )
            )
        )



        token = (
            access_token.to_jwt()
        )



        return VoiceConnectionToken(
            token=token,
            room_name=room_name,
            provider="livekit",
            expires_at=expires_at,
        )



__all__ = [
    "LiveKitTokenService",
]

"""
LiveKit Infrastructure Adapter.

Infrastructure boundary for LiveKit room configuration.

  LiveKit provider configuration.

- Prepare provider connection metadata.

from app.domain.uai.voice.voice_session import (

    __name__

class LiveKitAdapter:
    """
    Infrastructure adapter for LiveKit.

    Converts the provider-independent voice session
    into LiveKit room configuration.

    Token generation is deliberately handled by
    LiveKitTokenService.
    """



    provider = "livekit"



    def create_room_configuration(
        self,
        *,

            raise ValueError(
                "Voice session character id "
                "cannot be empty."
            )



        logger.info(
            "Preparing LiveKit room configuration",
            extra={
                "session_id": (
                    session.session_id
                ),
                "room_name": (
                    session.room_name
                ),
                "character_id": (
                    session.character_id
                ),
                "language_code": (
                    session.language_code
                ),
            },
        )



        return {
            "room_name": (
                session.room_name
            ),
            "character_id": (
                session.character_id
            ),
            "language_code": (
                session.language_code
            ),
            "provider": self.provider,
        }

"""
Voice Call Orchestrator.

Coordinates the preparation of a voice call.

Responsibilities:
- Build interaction context.
- Create character voice session.
- Create domain voice call session.
- Apply voice call runtime lifecycle.

Does NOT:
- Execute AI providers.
- Manage LiveKit.
- Handle HTTP.
- Persist data.
"""

from __future__ import annotations


from dataclasses import dataclass


from app.application.uai.voice.voice_call_request import (
    VoiceCallRequest,
)


from app.application.uai.context.interaction_context_builder import (
    InteractionContextBuilder,
)


from app.application.uai.voice.voice_character_session_service import (
    VoiceCharacterSessionService,
)


from app.ai_runtime.voice.voice_call_runtime import (
    VoiceCallRuntime,
)


from app.domain.uai.context.learner_context import (
    LearnerContext,
)


from app.domain.uai.context.voice_context import (
    VoiceContext,
)


from app.domain.uai.voice.voice_call_session import (
    VoiceCallSession,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallOrchestrator:
    """
    Voice call workflow coordinator.
    """

    context_builder: InteractionContextBuilder

    session_service: VoiceCharacterSessionService

    voice_runtime: VoiceCallRuntime = (
        VoiceCallRuntime()
    )



    def prepare_call(
        self,
        *,
        request: VoiceCallRequest,
    ) -> VoiceCallSession:
        """
        Prepare voice call session.

        Flow:

        Request
          |
          ↓
        Interaction Context
          |
          ↓
        Character Voice Session
          |
          ↓
        Voice Call Session
          |
          ↓
        Runtime Preparation
          |
          ↓
        READY state
        """


        learner = LearnerContext(
            learner_id=request.learner_id,
            learning_goals=[
                "conversation",
            ],
        )


        voice = VoiceContext(
            language_code=request.language_code,
            voice_provider="openai",
            voice_profile="verse",
        )


        context = (
            self.context_builder
            .build(
                learner=learner,
                world_id=request.world_id,
                character_id=request.character_id,
                voice=voice,
            )
        )


        voice_session = (
            self.session_service
            .create_session(
                context=context,
            )
        )


        call_session = VoiceCallSession(
            session_id=(
                voice_session.session_id
            ),
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
        )


        return (
            self.voice_runtime
            .prepare(
                session=call_session,
            )
        )

"""
Voice Character Session Service.

Application service responsible for
creating character voice sessions.

Responsibilities:
- Coordinate character runtime.
- Coordinate speech runtime.
- Create domain voice session.
- Request LiveKit room configuration.

Does NOT:
- Call speech providers directly.
- Manage LiveKit directly.
- Store configuration.
"""

from __future__ import annotations


from dataclasses import dataclass
from uuid import uuid4


from app.ai_runtime.voice.voice_character_runtime import (
    VoiceCharacterRuntime,
)


from app.ai_runtime.voice.voice_speech_runtime import (
    VoiceSpeechRuntime,
)


from app.domain.uai.context.interaction_context import (
    InteractionContext,
)


from app.domain.uai.voice.voice_session import (
    VoiceSession,
)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCharacterSessionService:
    """
    Creates voice sessions for AI characters.
    """

    livekit_adapter: object

    speech_runtime: VoiceSpeechRuntime



    def create_session(
        self,
        *,
        context: InteractionContext,
    ) -> VoiceSession:
        """
        Create a character voice session.
        """


        character_runtime = (
            VoiceCharacterRuntime(
                context=context,
                speech_runtime=(
                    self.speech_runtime
                ),
            )
        )


        character_id = (
            character_runtime
            .character_id()
        )


        session = VoiceSession(
            session_id=str(
                uuid4()
            ),
            room_name=(
                f"{character_id}-voice-room"
            ),
            character_id=character_id,
            language_code=(
                context.voice.language_code
            ),
        )


        self.livekit_adapter.create_room_configuration(
            session=session,
        )


        return session

from dataclasses import dataclass

    VoiceCallParticipant,
    VoiceParticipantRole,

@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallRoom:
    """
    Immutable voice call room.

    Supports:

    Beta:
        Learner + Character

    Future:
        Learner + Learner + Character
    """

    room_id: str

    participants: tuple[
        VoiceCallParticipant,
        ...

    ]


    def __post_init__(self) -> None:
        """
        Validate room composition.
        """

        if not self.room_id.strip():
            raise ValueError(
                "room_id cannot be empty."
            )


        if not self.participants:
            raise ValueError(
                "Voice call room requires "
                "at least one participant."
            )


        character_count = sum(
            1
            for participant in self.participants
            if participant.role
            ==
            VoiceParticipantRole.CHARACTER
        )


        if character_count > 1:
            raise ValueError(
                "Voice call room supports "
                "only one character participant."
            )


    def has_character(self) -> bool:
        """
        Check whether a character exists
        in the room.
        """

        return any(
            participant.role
            ==
            VoiceParticipantRole.CHARACTER
            for participant in self.participants
        )


    def learner_count(self) -> int:
        """
        Return number of learners.
        """

        return sum(
            1
            for participant in self.participants
            if participant.role
            ==
            VoiceParticipantRole.LEARNER
        )

)



@dataclass(
    frozen=True,
    slots=True,
)
class VoiceCallRuntime:
    """
    Runtime coordinator for voice calls.
    """



    def prepare(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        Prepare a voice call for activation.

        Transition:

        CREATED
            |
            ↓
        PREPARING
            |
            ↓
        READY
        """


        if session.state not in {
            VoiceCallState.CREATED,
            VoiceCallState.PREPARING,
        }:
            raise ValueError(
                "Voice call cannot be prepared "
                f"from state {session.state.value}."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.READY,
        )



    def activate(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        Activate a prepared voice call.

        Transition:

        READY
          |
          ↓
        ACTIVE
        """


        if session.state != (
            VoiceCallState.READY
        ):
            raise ValueError(
                "Voice call must be ready "
                "before activation."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.ACTIVE,
        )



    def end(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        End an active voice call.

        Transition:

        ACTIVE
          |
          ↓
        ENDED
        """


        if session.state != (
            VoiceCallState.ACTIVE
        ):
            raise ValueError(
                "Only active calls can be ended."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.ENDED,
        )



    def fail(
        self,
        *,
        session: VoiceCallSession,
    ) -> VoiceCallSession:
        """
        Mark voice call as failed.
        """


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.FAILED,
        )

            state=VoiceCallState.ACTIVE,
        )



    def stop(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> VoiceCallSession:
        """
        Stop voice call execution.

        Expected lifecycle:

        ACTIVE
          |
          ↓
        ENDED
        """


        session = (
            context.session
        )


        if session.state != (
            VoiceCallState.ACTIVE
        ):
            raise ValueError(
                "Only ACTIVE voice calls "
                "can be stopped."
            )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.ENDED,
        )



    def fail(
        self,
        *,
        context: VoiceCallExecutionContext,
    ) -> VoiceCallSession:
        """
        Mark execution as failed.
        """


        session = (
            context.session
        )


        return VoiceCallSession(
            session_id=session.session_id,
            learner_id=session.learner_id,
            character_id=session.character_id,
            world_id=session.world_id,
            language_code=session.language_code,
            state=VoiceCallState.FAILED,

"""
OpenAI Speech Provider.

Concrete speech provider implementation for OpenAI.

Responsibilities:
- Transcribe audio using OpenAI speech-to-text.
- Synthesize speech using OpenAI text-to-speech.
- Hide OpenAI SDK details behind SpeechProvider.

Does NOT:
- Handle HTTP.
- Manage application workflows.
- Manage LiveKit.
- Persist conversations.
- Execute AI agents.
"""

from __future__ import annotations


import io


import os


from typing import Any


from openai import OpenAI


from app.providers.speech.speech_provider import (
    SpeechProvider,
)



class OpenAISpeechProvider(
    SpeechProvider,
):
    """
    OpenAI implementation of the speech provider contract.

    The OpenAI client is created lazily so that importing
    the provider and running unit tests does not require
    OPENAI_API_KEY to be present.

    A client can also be injected for testing.
    """


    DEFAULT_TRANSCRIPTION_MODEL = (
        "gpt-4o-mini-transcribe"
    )


    DEFAULT_TTS_MODEL = (
        "gpt-4o-mini-tts"
    )


    DEFAULT_VOICE = (
        "alloy"
    )


    def __init__(
        self,
        *,
        client: Any | None = None,
        transcription_model: str | None = None,
        tts_model: str | None = None,
    ) -> None:
        """
        Initialize the OpenAI speech provider.

        Parameters
        ----------
        client:
            Optional OpenAI client.

            Primarily intended for tests and dependency
            injection.

        transcription_model:
            Optional OpenAI transcription model.

        tts_model:
            Optional OpenAI text-to-speech model.
        """

        self._client = client

        self._transcription_model = (
            transcription_model
            or os.getenv(
                "OPENAI_TRANSCRIPTION_MODEL",
                self.DEFAULT_TRANSCRIPTION_MODEL,
            )
        )

        self._tts_model = (
            tts_model
            or os.getenv(
                "OPENAI_TTS_MODEL",
                self.DEFAULT_TTS_MODEL,
            )
        )


    @property
    def client(self) -> OpenAI:
        """
        Return the configured OpenAI client.

        The client is initialized lazily.
        """

        if self._client is None:

            api_key = os.getenv(
                "OPENAI_API_KEY"
            )

            if not api_key:

                raise RuntimeError(
                    "OPENAI_API_KEY is not configured."
                )

            self._client = OpenAI(
                api_key=api_key,
            )

        return self._client


    def transcribe(
        self,
        *,
        audio: bytes,
        language_code: str,
    ) -> str:
        """
        Transcribe audio into text.

        Parameters
        ----------
        audio:
            Raw audio bytes.

        language_code:
            ISO language code, for example ``fr``.

        Returns
        -------
        str
            Transcribed text.
        """

        if not audio:

            raise ValueError(
                "audio cannot be empty."
            )


        if not language_code.strip():

            raise ValueError(
                "language_code cannot be empty."
            )


        audio_file = io.BytesIO(
            audio
        )


        audio_file.name = (
            "voice_input.wav"
        )


        response = (
            self.client
            .audio
            .transcriptions
            .create(
                model=(
                    self._transcription_model
                ),
                file=audio_file,
                language=(
                    language_code
                ),
            )
        )


        text = getattr(
            response,
            "text",
            None,
        )


        if text is None:

            raise RuntimeError(
                "OpenAI transcription response "
                "did not contain text."
            )


        return str(text)



    def synthesize(
        self,
        *,
        text: str,
        voice_profile: str,
        language_code: str,
    ) -> bytes:
        """
        Generate speech audio.

        Parameters
        ----------
        text:
            Text to synthesize.

        voice_profile:
            Character voice profile.

        language_code:
            Language requested by the caller.

        Returns
        -------
        bytes
            Generated audio bytes.
        """

        if not text.strip():

            raise ValueError(
                "text cannot be empty."
            )


        if not voice_profile.strip():

            raise ValueError(
                "voice_profile cannot be empty."
            )


        if not language_code.strip():

            raise ValueError(
                "language_code cannot be empty."
            )


        # Character voice profiles are currently mapped
        # to provider-supported voices at the infrastructure
        # boundary. This prevents provider-specific voice
        # identifiers from leaking into the domain/runtime.
        voice = self._resolve_voice(
            voice_profile
        )


        response = (
            self.client
            .audio
            .speech
            .create(
                model=(
                    self._tts_model
                ),
                voice=voice,
                input=text,
            )
        )


        audio_content = getattr(
            response,
            "content",
            None,
        )


        if audio_content is not None:

            return bytes(
                audio_content
            )


        if hasattr(
            response,
            "read",
        ):

            return bytes(
                response.read()
            )


        raise RuntimeError(
            "OpenAI speech response "
            "did not contain audio content."
        )



    @classmethod
    def _resolve_voice(
        cls,
        voice_profile: str,
    ) -> str:
        """
        Map an application character voice profile
        to an OpenAI-supported voice.

        Character identifiers such as ``yeti`` must not
        be sent directly to the provider because they are
        application concepts rather than provider voice IDs.
        """

        normalized = (
            voice_profile
            .strip()
            .lower()
        )


        voice_mapping = {
            "yeti": "onyx",
        }


        return voice_mapping.get(
            normalized,
            normalized
            if normalized in {
                "alloy",
                "ash",
                "ballad",
                "coral",
                "echo",
                "fable",
                "nova",
                "onyx",
                "sage",
                "shimmer",
                "verse",
                "marin",
                "cedar",
            }
            else cls.DEFAULT_VOICE,
        )



__all__ = [
    "OpenAISpeechProvider",
]

        get_voice_call_lifecycle_service
    ),
    context_factory: VoiceCallContextFactory = Depends(
        get_voice_call_context_factory

) -> VoiceCallResponseModel:

            language_code=session.language_code,
            provider=room["provider"],
            state=session.state.value,

def trigger_voice_call(
    request: VoiceCallTriggerRequest,
    service: VoiceCallTriggerService = Depends(
        get_voice_call_trigger_service
    ),
) -> VoiceCallTriggerResponse:
    """
    Trigger a voice call workflow.
    """


    result = (
        service.trigger_voice_call(
            learner_id=request.learner_id,
            character_id=request.character_id,
            world_id=request.world_id,
            language_code=request.language_code,
            notification_channel=request.channel,
        )
    )


    return VoiceCallTriggerResponse(
        voice_call_id=(
            result["voice_call"].id
        ),
        invitation_id=(
            result["invitation"].id
        ),
        notification_id=(
            result["notification"].id
        ),
    )



__all__ = [
    "router",
]
