==================== CHARACTER PROFILE ====================

"""
Character Profile Model.

Defines AI companion identity.

Characters are separate from agents.

Character:
- personality
- communication style
- learning approach

Agent:
- execution capability
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class CharacterProfile:
    """
    Defines an EasyChatBot AI companion.
    """

    character_id: str

    name: str

    description: str

    personality: str

    learning_style: str


==================== CHARACTER REGISTRY ====================

"""
EasyChatBot Character Registry.

The registry provides the approved characters available
to the application.

Characters define identity and behaviour.

Agents remain responsible for capabilities and execution.
"""

from backend.characters.character_profile import (
    CharacterProfile,
)


class CharacterRegistry:
    """
    Registry of approved EasyChatBot AI companions.
    """

    def __init__(self) -> None:
        self._characters = {
            "yeti": CharacterProfile(
                character_id="yeti",
                name="Yeti",
                description=(
                    "A friendly AI companion who helps "
                    "you discover knowledge and practice languages."
                ),
                personality=(
                    "Curious, encouraging and supportive."
                ),
                learning_style=(
                    "Conversation based learning."
                ),
            ),
            "clara": CharacterProfile(
                character_id="clara",
                name="Clara",
                description=(
                    "A professional learning companion "
                    "focused on technology and career topics."
                ),
                personality=(
                    "Analytical, structured and supportive."
                ),
                learning_style=(
                    "Structured and practical learning."
                ),
            ),
            "luminate": CharacterProfile(
                character_id="luminate",
                name="Luminate",
                description=(
                    "A creative companion for culture, "
                    "ideas and exploration."
                ),
                personality=(
                    "Creative, inspiring and curious."
                ),
                learning_style=(
                    "Exploration and discussion."
                ),
            ),
        }

    def get(
        self,
        character_id: str,
    ) -> CharacterProfile:
        """
        Return an approved character.

        Raises:
            ValueError:
                If the character does not exist.
        """

        normalized_id = character_id.strip().lower()

        try:
            return self._characters[normalized_id]
        except KeyError as exc:
            raise ValueError(
                f"Unknown character: {character_id}"
            ) from exc

    def list_characters(
        self,
    ) -> list[CharacterProfile]:
        """
        Return all approved characters.
        """

        return list(self._characters.values())

 import pytest

from backend.agents.conversation_agent import (
    ConversationAgent,
)

from backend.characters.character_registry import (
    CharacterRegistry,
)

from backend.runtime.context.execution_context import (
    ExecutionContext,
)


@pytest.mark.asyncio
async def test_clara_uses_clara_profile():

    registry = CharacterRegistry()

    clara = registry.get("clara")

    agent = ConversationAgent(
        clara
    )

    context = ExecutionContext(
        user_id="demo-user",
        request="Explain artificial intelligence",
    )

    result = await agent.execute(
        context
    )

    assert result.success is True

    assert result.output["character"] == (
        "Clara"
    )

    assert result.output["personality"] == (
        "Analytical, structured and supportive."
    )

    assert result.output["learning_style"] == (
        "Structured and practical learning."
    )

def test_character_registry_returns_yeti():

    registry = CharacterRegistry()

    character = registry.get("yeti")

    assert character.character_id == "yeti"

    assert character.name == "Yeti"


def test_character_registry_returns_clara():

    registry = CharacterRegistry()

    character = registry.get("clara")

    assert character.character_id == "clara"

    assert character.name == "Clara"


def test_character_registry_is_case_insensitive():

    registry = CharacterRegistry()

    character = registry.get("CLARA")

    assert character.character_id == "clara"


def test_character_registry_lists_characters():

    registry = CharacterRegistry()

    characters = registry.list_characters()

    assert len(characters) == 3
