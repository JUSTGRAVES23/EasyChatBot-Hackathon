/*
 * =========================================================
 * EasyChatBot Main Dashboard Application
 * =========================================================
 *
 * The Home dashboard does NOT allow the learner to select
 * an AI companion.
 *
 * The learner already has a companion.
 *
 * The flow is:
 *
 *     Authenticated Learner
 *             +
 *       Existing Companion
 *             +
 *            World
 *             +
 *         Languages
 *             ↓
 *       Dashboard Context
 *             ↓
 *       Character Theme API
 *             ↓
 *       Resolved Dashboard Theme
 *
 *
 * Character identity is therefore NOT controlled by the
 * Home page.
 *
 * The current development learner context is supplied by:
 *
 *     window.EasyChatBotLearner
 *
 * from index.html.
 *
 * Later this object will be replaced by the authenticated
 * learner/session API.
 *
 *
 * Responsibilities of this file:
 *
 * - initialise the Home dashboard;
 * - read the learner's existing companion;
 * - load companion information;
 * - request the resolved character theme;
 * - populate companion presentation;
 * - handle dashboard actions;
 * - send chat messages;
 * - preserve separation between:
 *
 *       Character identity
 *       Theme resolution
 *       Dashboard presentation
 *       Chat execution
 *
 * Character themes are NOT defined here.
 *
 * Theme resolution remains a backend responsibility.
 *
 * =========================================================
 */


/*
 * =========================================================
 * Global Dashboard State
 * =========================================================
 */


/*
 * Technical character identifier.
 *
 * Example:
 *
 *     "luminate"
 *     "yeti"
 *     "clara"
 *     "layla"
 *
 * This is NOT selected by the Home UI.
 */

let characterId = null;


/*
 * Character display name.
 *
 * Example:
 *
 *     "Luminate"
 */

let character = null;


/*
 * Resolved character profile from characters.js.
 */

let activeCharacter = null;


/*
 * Current learner context.
 */

let learnerContext = null;


/*
 * Current resolved dashboard theme.
 */

let activeDashboardTheme = null;


/*
 * =========================================================
 * Development / Session Context
 * =========================================================
 *
 * index.html currently provides:
 *
 *     window.EasyChatBotLearner
 *
 * Example:
 *
 * {
 *     characterId: "luminate",
 *     worldId: "france",
 *     primaryLanguage: "fr",
 *     secondaryLanguage: "es"
 * }
 *
 *
 * This is deliberately a context object rather than a
 * character selector.
 *
 * =========================================================
 */


function getLearnerContext() {

    const context =
        window.EasyChatBotLearner;


    if (!context) {

        console.error(
            "EasyChatBot learner context was not found."
        );

        return null;

    }


    return {

        characterId:
            context.characterId || null,

        worldId:
            context.worldId || "france",

        primaryLanguage:
            context.primaryLanguage || "fr",

        secondaryLanguage:
            context.secondaryLanguage || null

    };

}


/*
 * =========================================================
 * Character Lookup
 * =========================================================
 *
 * characters.js remains the frontend character metadata
 * source for presentation copy.
 *
 * It is NOT used to select a character.
 *
 * The learner's character ID comes from the learner context.
 *
 * =========================================================
 */


function getCharacter(
    requestedCharacterId
) {

    if (
        typeof characters ===
        "undefined"
    ) {

        console.error(
            "characters.js was not loaded."
        );

        return null;

    }


    if (
        !Array.isArray(characters)
    ) {

        console.error(
            "The characters collection is invalid."
        );

        return null;

    }


    return characters.find(
        function (item) {

            return (
                item.id ===
                requestedCharacterId
            );

        }
    ) || null;

}


/*
 * =========================================================
 * Dashboard Initialisation
 * =========================================================
 */


async function initialiseDashboard() {

    console.log(
        "EasyChatBot Home dashboard initialising..."
    );


    /*
     * -----------------------------------------------------
     * Read learner context
     * -----------------------------------------------------
     */

    learnerContext =
        getLearnerContext();


    if (!learnerContext) {

        showDashboardError(
            "Learner context is unavailable."
        );

        return;

    }


    /*
     * -----------------------------------------------------
     * Validate existing companion
     * -----------------------------------------------------
     */

    characterId =
        learnerContext.characterId;


    if (!characterId) {

        showDashboardError(
            "No companion is assigned to this learner."
        );

        return;

    }


    /*
     * -----------------------------------------------------
     * Load character metadata
     * -----------------------------------------------------
     */

    activeCharacter =
        getCharacter(
            characterId
        );


    if (!activeCharacter) {

        console.error(
            "Character metadata not found:",
            characterId
        );


        showDashboardError(
            "Your companion could not be loaded."
        );

        return;

    }


    character =
        activeCharacter.name;


    console.log(
        "Existing learner companion:",
        character
    );


    console.log(
        "Character ID:",
        characterId
    );


    /*
     * -----------------------------------------------------
     * Populate dashboard companion presentation
     * -----------------------------------------------------
     */

    updateCompanionPresentation(
        activeCharacter
    );


    /*
     * -----------------------------------------------------
     * Load character/world theme
     * -----------------------------------------------------
     */

    await loadCharacterTheme();


    /*
     * -----------------------------------------------------
     * Initialise dashboard interactions
     * -----------------------------------------------------
     */

    initialiseNavigation();

    initialiseDashboardActions();


    console.log(
        "EasyChatBot Home dashboard ready."
    );

}


/*
 * =========================================================
 * Companion Presentation
 * =========================================================
 *
 * This function updates the Home dashboard with the
 * companion that already belongs to the learner.
 *
 * It does NOT create a character-selection UI.
 *
 * =========================================================
 */


function updateCompanionPresentation(
    selected
) {

    if (!selected) {

        return;

    }


    /*
     * -----------------------------------------------------
     * Companion name
     * -----------------------------------------------------
     */

    setText(
        "companion-name-pick",
        selected.name
    );


    /*
     * -----------------------------------------------------
     * Hero title
     * -----------------------------------------------------
     */

    const heroTitle =
        document.getElementById(
            "companion-title"
        );


    if (heroTitle) {

        heroTitle.textContent =
            selected.name +
            " is ready for you!";

    }


    /*
     * -----------------------------------------------------
     * Hero message
     * -----------------------------------------------------
     */

    const heroMessage =
        document.getElementById(
            "companion-message"
        );


    if (heroMessage) {

        heroMessage.textContent =
            getCompanionMessage(
                selected
            );

    }


    /*
     * -----------------------------------------------------
     * Companion quote
     * -----------------------------------------------------
     */

    const quote =
        document.getElementById(
            "companion-quote"
        );


    if (quote) {

        quote.textContent =
            getCompanionQuote(
                selected
            );

    }


    /*
     * -----------------------------------------------------
     * Quote author
     * -----------------------------------------------------
     */

    setText(
        "companion-quote-author",
        "— " + selected.name
    );


    /*
     * -----------------------------------------------------
     * Companion pick message
     * -----------------------------------------------------
     */

    const pickMessage =
        document.getElementById(
            "pick-message"
        );


    if (pickMessage) {

        pickMessage.textContent =
            getCompanionPickMessage(
                selected
            );

    }


    /*
     * -----------------------------------------------------
     * Character visual
     * -----------------------------------------------------
     */

    updateCompanionVisual(
        selected
    );


    /*
     * -----------------------------------------------------
     * HTML metadata
     * -----------------------------------------------------
     */

    document.documentElement.dataset.character =
        selected.id;


}

/*
 * =========================================================
 * Companion Visual
 * =========================================================
 *
 * The character theme system may eventually provide an
 * avatar/image URL.
 *
 * We therefore support:
 *
 *     theme.avatar
 *     theme.character_image
 *     theme.characterImage
 *     selected.avatar
 *     selected.image
 *
 * without requiring them today.
 *
 * If none exists, the dashboard keeps a visual placeholder.
 *
 * =========================================================
 */


function updateCompanionVisual(
    selected
) {

    const visual =
        document.getElementById(
            "companion-visual"
        );


    const placeholder =
        document.getElementById(
            "companion-placeholder"
        );


    if (!visual) {

        return;

    }


    /*
     * Try known character metadata fields.
     */

    const imageSource =
        selected.avatar ||
        selected.image ||
        selected.avatarUrl ||
        null;


    if (imageSource) {

        visual.style.backgroundImage =
            "url('" +
            imageSource +
            "')";

        visual.classList.add(
            "has-character-image"
        );


        if (placeholder) {

            placeholder.style.display =
                "none";

        }


        return;

    }


    /*
     * No image is currently available.
     *
     * Keep the themed placeholder.
     */

    if (placeholder) {

        placeholder.textContent =
            getCharacterSymbol(
                selected.id
            );

    }

}

/*
 * =========================================================
 * Character Theme Integration
 * =========================================================
 *
 * Backend endpoint:
 *
 *     GET /character-theme
 *
 * Inputs:
 *
 *     character_id
 *     world_id
 *     primary_language
 *     secondary_language
 *
 * The backend remains responsible for resolving the theme.
 *
 * =========================================================
 */


async function loadCharacterTheme() {

    if (!learnerContext) {

        console.error(
            "Cannot load theme without learner context."
        );

        return;

    }


    if (!characterId) {

        console.error(
            "Cannot load theme without character ID."
        );

        return;

    }


    const worldId =
        learnerContext.worldId;


    const primaryLanguage =
        learnerContext.primaryLanguage;


    const secondaryLanguage =
        learnerContext.secondaryLanguage;


    try {

        /*
         * -------------------------------------------------
         * Build query
         * -------------------------------------------------
         */

        const queryParameters = {

            character_id:
                characterId,

            world_id:
                worldId,

            primary_language:
                primaryLanguage

        };


        /*
         * Only send secondary language when
         * one actually exists.
         */

        if (secondaryLanguage) {

            queryParameters.secondary_language =
                secondaryLanguage;

        }


        const query =
            new URLSearchParams(
                queryParameters
            );


        /*
         * -------------------------------------------------
         * Request resolved theme
         * -------------------------------------------------
         */

        const response =
            await fetch(

                "http://127.0.0.1:8000/character-theme?" +
                query.toString()

            );


        if (!response.ok) {

            throw new Error(

                "Theme API request failed: " +
                response.status

            );

        }


        const themeResponse =
            await response.json();


        activeDashboardTheme =
            themeResponse;


        /*
         * -------------------------------------------------
         * Delegate presentation to theme.js
         * -------------------------------------------------
         */

        if (
            typeof applyCharacterTheme !==
            "function"
        ) {

            console.error(

                "applyCharacterTheme() was not found. " +
                "Make sure theme.js is loaded before app.js."

            );

            return;

        }


        applyCharacterTheme(
            themeResponse
        );


        /*
         * -------------------------------------------------
         * Apply optional theme-specific presentation
         * -------------------------------------------------
         */

        applyThemePresentation(
            themeResponse
        );


        console.log(
            "Character theme loaded:",
            characterId
        );


        console.log(
            "World:",
            worldId
        );


        console.log(
            "Primary language:",
            primaryLanguage
        );


        console.log(
            "Secondary language:",
            secondaryLanguage
        );


    } catch (error) {

        /*
         * Theme failure must NOT break the dashboard.
         */

        console.error(
            "Unable to load character theme:",
            error
        );


        /*
         * Keep the EasyChatBot core UI functional.
         */

    }

}


/*
 * =========================================================
 * Optional Theme Presentation
 * =========================================================
 *
 * theme.js owns CSS variable application.
 *
 * This function handles optional presentation metadata
 * without making the dashboard dependent on it.
 *
 * =========================================================
 */


function applyThemePresentation(
    themeResponse
) {

    if (!themeResponse) {

        return;

    }


    const theme =
        themeResponse.theme || {};


    const root =
        document.documentElement;


    /*
     * Character metadata
     */

    if (
        themeResponse.character_id
    ) {

        root.dataset.character =
            themeResponse.character_id;

    }


    /*
     * World metadata
     */

    if (
        themeResponse.world_id
    ) {

        root.dataset.world =
            themeResponse.world_id;

    }


    /*
     * Theme name
     */

    if (
        theme.name
    ) {

        root.dataset.theme =
            theme.name;

    }


    /*
     * Atmosphere
     */

    if (
        theme.atmosphere
    ) {

        if (
            typeof theme.atmosphere ===
            "string"
        ) {

            root.dataset.atmosphere =
                theme.atmosphere;

        }

        else if (
            typeof theme.atmosphere ===
            "object"
        ) {

            if (
                theme.atmosphere.id
            ) {

                root.dataset.atmosphere =
                    theme.atmosphere.id;

            }


            if (
                theme.atmosphere.type
            ) {

                root.dataset.atmosphereType =
                    theme.atmosphere.type;

            }

        }

    }

}

/*
 * =========================================================
 * Chat
 * =========================================================
 *
 * Kept for compatibility with the existing backend chat
 * endpoint and future Chat UI.
 *
 * The important difference is that the character is now
 * derived from the learner's existing companion.
 *
 * =========================================================
 */


async function sendMessage() {

    const messageElement =
        document.getElementById(
            "message"
        );


    const researchElement =
        document.getElementById(
            "research"
        );


    const responseElement =
        document.getElementById(
            "response"
        );


    if (!messageElement) {

        console.error(
            "Chat message field not found."
        );

        return;

    }


    const message =
        messageElement.value.trim();


    const research =
        researchElement
            ? researchElement.checked
            : false;


    if (!message) {

        if (responseElement) {

            responseElement.textContent =
                "Please enter a message.";

        }

        return;

    }


    if (!character) {

        if (responseElement) {

            responseElement.textContent =
                "Your companion is not available.";

        }

        return;

    }


    if (responseElement) {

        responseElement.textContent =
            "Your companion is thinking...";

    }


    try {

        const response =
            await fetch(

                "http://127.0.0.1:8000/chat",

                {

                    method:
                        "POST",


                    headers: {

                        "Content-Type":
                            "application/json"

                    },


                    body:
                        JSON.stringify({

                            user_id:
                                "demo-user",


                            /*
                             * Preserve the existing backend
                             * contract.
                             *
                             * The backend currently receives
                             * the character name.
                             */

                            character:
                                character,


                            message:
                                message,


                            research:
                                research

                        })

                }

            );


        if (!response.ok) {

            throw new Error(

                "API request failed: " +
                response.status

            );

        }


        const data =
            await response.json();


        if (responseElement) {

            responseElement.textContent =
                JSON.stringify(

                    data,

                    null,

                    2

                );

        }


    } catch (error) {

        console.error(

            "EasyChatBot request failed:",

            error

        );


        if (responseElement) {

            responseElement.textContent =
                "Unable to connect to EasyChatBot. " +
                "Please make sure the backend is running.\n\n" +
                error.message;

        }

    }

}

/*
 * =========================================================
 * EasyChatBot Character Theme Adapter
 * =========================================================
 *
 * The frontend does not define character themes.
 *
 * The backend resolves:
 *
 *     Character
 *          +
 *       World
 *          +
 *      Languages
 *          ↓
 *     Theme Response
 *
 * This adapter applies the resolved presentation
 * to the existing EasyChatBot UI.
 *
 * Core UI structure remains unchanged.
 *
 * Character-specific presentation may change:
 *
 * - colours
 * - atmosphere
 * - background
 * - surfaces
 * - accents
 * - decorative elements
 * - character identity
 * - world identity
 *
 * IMPORTANT:
 *
 * app.js owns the dashboard's active theme state.
 *
 * This file therefore uses its own adapter state:
 *
 *     themeAdapterState
 *
 * This prevents a global JavaScript variable collision
 * with app.js.
 */


/*
 * =========================================================
 * Theme Adapter State
 * =========================================================
 *
 * This is the theme.js adapter's local global state.
 *
 * app.js separately owns:
 *
 *     activeDashboardTheme
 *
 * Do NOT rename this back to activeDashboardTheme.
 */

let themeAdapterState = null;


/*
 * =========================================================
 * Main Theme Application
 * =========================================================
 */


function applyCharacterTheme(
    themeResponse
) {

    if (!themeResponse) {

        console.error(
            "Character theme response is empty."
        );

        return;

    }


    /*
     * Store the theme inside the adapter state.
     *
     * This is intentionally separate from the
     * activeDashboardTheme variable owned by app.js.
     */

    themeAdapterState =
        themeResponse;


    const theme =
        themeResponse.theme;


    if (!theme) {

        console.error(
            "Character theme payload is missing."
        );

        return;

    }


    const root =
        document.documentElement;


    /*
     * -------------------------------------------------------
     * Palette
     * -------------------------------------------------------
     */


    const palette =
        theme.palette || {};


    applyPalette(
        root,
        palette
    );


    /*
     * -------------------------------------------------------
     * Character / World identity
     * -------------------------------------------------------
     */


    if (
        themeResponse.character_id
    ) {

        root.dataset.character =
            themeResponse.character_id;

    }


    if (
        themeResponse.world_id
    ) {

        root.dataset.world =
            themeResponse.world_id;

    }


    /*
     * -------------------------------------------------------
     * Language context
     * -------------------------------------------------------
     */


    const languages =
        themeResponse.languages || {};


    if (
        languages.primary
    ) {

        root.dataset.primaryLanguage =
            languages.primary;

    }


    if (
        languages.secondary
    ) {

        root.dataset.secondaryLanguage =
            languages.secondary;

    }


    /*
     * -------------------------------------------------------
     * Theme metadata
     * -------------------------------------------------------
     */


    if (
        theme.name
    ) {

        root.dataset.theme =
            theme.name;

    }


    if (
        theme.version
    ) {

        root.dataset.themeVersion =
            theme.version;

    }


    /*
     * -------------------------------------------------------
     * Atmosphere
     * -------------------------------------------------------
     */


    applyAtmosphere(
        root,
        theme
    );


    /*
     * -------------------------------------------------------
     * Log resolved theme
     * -------------------------------------------------------
     */


    console.log(
        "EasyChatBot theme applied:",
        themeResponse.character_id,
        themeResponse.world_id
    );

}

    );


    /*

     */

    setThemeVariable(

    );


    /*

    setThemeVariable(

    );

/*
 * =========================================================
 * Safe CSS Variable Setter
 * =========================================================
 */


function setThemeVariable(
    root,
    variableName,
    value
) {

    if (
        value === undefined ||
        value === null ||
        value === ""
    ) {

        return;

    }


    root.style.setProperty(
        variableName,
        value
    );

}


/*
 * =========================================================
 * Atmosphere
 * =========================================================
 *
 * The theme may define an atmosphere/background
 * representation.
 *
 * The core dashboard layout is not changed here.
 */


function applyAtmosphere(
    root,
    theme
) {

    const atmosphere =
        theme.atmosphere;


    if (!atmosphere) {

        return;

    }


    if (
        typeof atmosphere ===
        "string"
    ) {

        root.dataset.atmosphere =
            atmosphere;

        return;

    }


    if (
        typeof atmosphere ===
        "object"
    ) {

        if (
            atmosphere.id
        ) {

            root.dataset.atmosphere =
                atmosphere.id;

        }


        if (
            atmosphere.type
        ) {

            root.dataset.atmosphereType =
                atmosphere.type;

        }


        if (
            atmosphere.background
        ) {

            root.style.setProperty(
                "--ecb-character-atmosphere",
                atmosphere.background
            );

        }

    }

}

/*
 * =========================================================
 * Theme Reset
 * =========================================================
 *
 * Useful when the learner changes character or when the
 * dashboard needs to return to the EasyChatBot core state.
 *
 * Home itself does NOT provide character selection.
 *
 * This function simply resets the visual theme state.
 */


function resetCharacterTheme() {

    const root =
        document.documentElement;


    const themeVariables = [

        "--ecb-character-primary",

        "--ecb-character-secondary",

        "--ecb-character-accent",

        "--ecb-character-background",

        "--ecb-character-surface",

        "--ecb-character-surface-alt",

        "--ecb-character-text-primary",

        "--ecb-character-text-secondary",

        "--ecb-character-highlight",

        "--ecb-character-atmosphere"

    ];


    themeVariables.forEach(
        function (variableName) {

            root.style.removeProperty(
                variableName
            );

        }
    );


    delete root.dataset.character;

    delete root.dataset.world;

    delete root.dataset.primaryLanguage;

    delete root.dataset.secondaryLanguage;

    delete root.dataset.theme;

    delete root.dataset.themeVersion;

    delete root.dataset.atmosphere;

    delete root.dataset.atmosphereType;


    /*
     * Reset ONLY theme.js adapter state.
     *
     * app.js owns its own activeDashboardTheme.
     */

    themeAdapterState =
        null;


    console.log(
        "EasyChatBot character theme reset."
    );

}
