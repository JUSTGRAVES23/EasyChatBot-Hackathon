        items = [
            self._serialize_projection(item)
            for item in source.list_reviews()

            "items": items,
        }

            raise ValueError("review_item_id must not be blank.")

        source = self._require_source()
        item = source.get_review(review_item_id)

        if item is None:

        return self._serialize_projection(item)

class PlatformIntelligenceEscalationDispatchService:
    """Revalidate, deduplicate and dispatch one governed escalation."""

    def __init__(
        self,
        *,
        delivery_repository: Any,
        email_workflow: Any,
    ) -> None:
        if delivery_repository is None:
            raise ValueError(
                "delivery_repository cannot be None."
            )

        if email_workflow is None:
            raise ValueError(
                "email_workflow cannot be None."
            )

        self._repository = delivery_repository
        self._email_workflow = email_workflow

    @staticmethod
    def build_notification_key(
        recommendation_id: str,
    ) -> str:
        """Return the canonical recommendation-scoped notification key."""
        return _build_notification_key(recommendation_id)

    async def dispatch(
        self,
        *,
        request: PlatformIntelligenceEscalationDispatchRequest,
        email_request: Any,
    ) -> PlatformIntelligenceEscalationDispatchResult:
        self._revalidate(request)

        existing = await self._repository.get_by_notification_key(
            request.notification_key
        )

        if (
            existing is not None
            and existing.status
            is PlatformIntelligenceEscalationDeliveryStatus.SUCCEEDED
        ):
            return PlatformIntelligenceEscalationDispatchResult(
                status=(
                    PlatformIntelligenceEscalationDispatchStatus
                    .IDEMPOTENT_REPLAY
                ),
                notification_key=existing.notification_key,
                provider_message_id=(
                    existing.provider_message_id
                ),
                attempt=existing.attempt,
            )

        if existing is None:
            try:
                await self._repository.create_started(
                    notification_key=request.notification_key,
                    recommendation_id=request.recommendation_id,
                    risk_decision_id=request.risk_decision_id,
                    review_item_id=request.review_item_id,
                    correlation_id=request.correlation_id,
                )
            except Exception:
                # The durable repository owns the unique-key race.
                # If another worker created the record first, honour
                # the persisted state.
                existing = (
                    await self._repository
                    .get_by_notification_key(
                        request.notification_key
                    )
                )

                if existing is None:
                    raise

        elif (
            existing.status
            is PlatformIntelligenceEscalationDeliveryStatus.FAILED
        ):
            await self._repository.start_retry(
                notification_key=request.notification_key
            )

        elif (
            existing.status
            is PlatformIntelligenceEscalationDeliveryStatus.STARTED
        ):
            # An interrupted attempt is retryable, following the
            # established durable execution convention.
            pass

        try:
            workflow_result = self._email_workflow.send(
                email_request
            )

            if inspect.isawaitable(workflow_result):
                workflow_result = await workflow_result

            # The current PI email workflow may return its request
            # containing an awaitable `delivery` result.
            #
            # Resolve it before marking the durable delivery as
            # successful so that invocation of the workflow itself
            # cannot be mistaken for successful provider delivery.
            delivery = getattr(
                workflow_result,
                "delivery",
                None,
            )

            if inspect.isawaitable(delivery):
                delivery = await delivery

            provider_message_id = (
                self._extract_provider_message_id(
                    delivery
                    if delivery is not None
                    else workflow_result
                )
            )

        except Exception as exc:
            error_code = self._error_code(exc)

            await self._repository.mark_failed(
                notification_key=request.notification_key,
                error_code=error_code,
            )

            return PlatformIntelligenceEscalationDispatchResult(
                status=(
                    PlatformIntelligenceEscalationDispatchStatus
                    .FAILED
                ),
                notification_key=request.notification_key,
                error_code=error_code,
            )

        await self._repository.mark_succeeded(
            notification_key=request.notification_key,
            provider_message_id=provider_message_id,
        )

        record = (
            await self._repository.get_by_notification_key(
                request.notification_key
            )
        )

        return PlatformIntelligenceEscalationDispatchResult(
            status=(
                PlatformIntelligenceEscalationDispatchStatus
                .SENT
            ),
            notification_key=request.notification_key,
            provider_message_id=provider_message_id,
            attempt=(
                record.attempt
                if record is not None
                else None
            ),
        )

    @staticmethod
    def _revalidate(
        request: PlatformIntelligenceEscalationDispatchRequest,
    ) -> None:
        if request.quality_gate_status != "PASSED":
            raise ValueError(
                "Dispatch blocked: quality gate is not PASSED."
            )

        if request.review_status != "IN_REVIEW":
            raise ValueError(
                "Dispatch blocked: review is not IN_REVIEW."
            )

        if request.decision_status != "PENDING_HUMAN_REVIEW":
            raise ValueError(
                "Dispatch blocked: decision is not "
                "PENDING_HUMAN_REVIEW."
            )

        expected_key = _build_notification_key(
            request.recommendation_id
        )

        if request.notification_key != expected_key:
            raise ValueError(
                "notification_key does not match "
                "recommendation_id."
            )

    @staticmethod
    def _extract_provider_message_id(
        result: object,
    ) -> str | None:
        if isinstance(result, dict):
            value = (
                result.get("provider_message_id")
                or result.get("id")
                or result.get("message_id")
            )

            return (
                str(value)
                if value is not None
                else None
            )

        return None

    @staticmethod
    def _error_code(
        exc: Exception,
    ) -> str:
        explicit = getattr(
            exc,
            "error_code",
            None,
        )

        if (
            isinstance(explicit, str)
            and explicit.strip()
        ):
            return explicit.strip()[:100]

        return type(exc).__name__.upper()[:100]

class PlatformIntelligenceEscalationRetryOrchestrator:
    """Coordinates policy, durable claim, state revalidation and retry.

    This service deliberately does not discover candidates, implement retry
    policy, send email directly, or mutate human-review/risk state.

    The canonical retry execution service remains the owner of retry-policy
    enforcement and existing dispatch integration. The orchestrator only
    supplies the authoritative dispatch context after the durable claim and
    final state revalidation.
    """

    def __init__(
        self,
        *,
        state_reader: PlatformIntelligenceRetryStateReader,
        policy_evaluator: PlatformIntelligenceRetryPolicyEvaluator,
        claim_service: PlatformIntelligenceEscalationRetryClaimService,
        retry_execution: PlatformIntelligenceRetryExecutor,
    ) -> None:
        self._state_reader = state_reader
        self._policy_evaluator = policy_evaluator
        self._claim_service = claim_service
        self._retry_execution = retry_execution

    async def run_candidate(
        self,
        *,
        candidate: PlatformIntelligenceEscalationRetryCandidate,
        reference_at: datetime,
    ) -> PlatformIntelligenceRetryRunResult:
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("reference_at must be timezone-aware.")

        eligible = await self._policy_evaluator.evaluate(
            candidate=candidate,
            reference_at=reference_at,
        )
        if not eligible:
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason="RETRY_POLICY_NOT_ELIGIBLE",
                attempt=candidate.attempt,
            )

        claim = await self._claim_service.claim(
            notification_key=candidate.notification_key,
            claimed_at=reference_at,
        )
        if claim is None:
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason="RETRY_CANDIDATE_ALREADY_CLAIMED",
                attempt=candidate.attempt,
            )

        state = await self._state_reader.get_retry_state(
            notification_key=candidate.notification_key,
            recommendation_id=candidate.recommendation_id,
        )
        if state is None:
            return self._skipped(candidate, "AUTHORITATIVE_STATE_NOT_FOUND")
        if state.notification_key != candidate.notification_key:
            return self._skipped(candidate, "NOTIFICATION_KEY_MISMATCH")
        if state.recommendation_id != candidate.recommendation_id:
            return self._skipped(candidate, "RECOMMENDATION_ID_MISMATCH")
        if state.recommendation_status != "RECOMMENDED":
            return self._skipped(
                candidate, "RECOMMENDATION_NO_LONGER_RECOMMENDED"
            )
        if state.review_status != "IN_REVIEW":
            return self._skipped(candidate, "REVIEW_NOT_IN_REVIEW")
        if state.decision_status != "PENDING_HUMAN_REVIEW":
            return self._skipped(
                candidate, "DECISION_NO_LONGER_PENDING_HUMAN_REVIEW"
            )
        if state.quality_gate_status != "PASSED":
            return self._skipped(
                candidate, "QUALITY_GATE_NO_LONGER_PASSED"
            )

        execution = await self._retry_execution.retry(
            request=state.dispatch_request,
            email_request=state.email_request,
            reference_at=reference_at,
        )

        if (
            execution.status
            is PlatformIntelligenceEscalationRetryExecutionStatus.RETRIED
        ):
            dispatch_result = execution.dispatch_result
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SUCCEEDED,
                reason=execution.reason,
                attempt=execution.attempt,
                provider_message_id=(
                    dispatch_result.provider_message_id
                    if dispatch_result is not None
                    else None
                ),
            )

        if execution.status in (
            PlatformIntelligenceEscalationRetryExecutionStatus.NOT_ELIGIBLE,
            PlatformIntelligenceEscalationRetryExecutionStatus.NOT_DUE,
        ):
            return PlatformIntelligenceRetryRunResult(
                notification_key=candidate.notification_key,
                status=PlatformIntelligenceRetryRunStatus.SKIPPED,
                reason=execution.reason,
                attempt=execution.attempt,
            )

        return PlatformIntelligenceRetryRunResult(
            notification_key=candidate.notification_key,
            status=PlatformIntelligenceRetryRunStatus.FAILED,
            reason=execution.reason,
            attempt=execution.attempt,
        )

    @staticmethod
    def _skipped(
        candidate: PlatformIntelligenceEscalationRetryCandidate,
        reason: str,
    ) -> PlatformIntelligenceRetryRunResult:
        return PlatformIntelligenceRetryRunResult(
            notification_key=candidate.notification_key,
            status=PlatformIntelligenceRetryRunStatus.SKIPPED,
            reason=reason,
            attempt=candidate.attempt,
        )

class PlatformIntelligenceEscalationRetryExecutionService:
    """Execute one policy-authorized retry without owning scheduling.

    This service:
    - retrieves the authoritative durable delivery record;
    - evaluates retry policy;
    - enforces the policy-derived due time;
    - increments FAILED attempts only through the repository;
    - preserves the attempt number for STARTED recovery;
    - delegates actual delivery to the existing dispatch service.

    It does not:
    - schedule future work;
    - sleep;
    - call an email provider directly;
    - mutate delivery state directly.
    """

    def __init__(
        self,
        *,
        delivery_repository: object,
        retry_policy: PlatformIntelligenceEscalationRetryPolicyService,
        dispatch_service: PlatformIntelligenceEscalationDispatchService,
    ) -> None:
        if delivery_repository is None:
            raise ValueError(
                "delivery_repository cannot be None."
            )

        if retry_policy is None:
            raise ValueError(
                "retry_policy cannot be None."
            )

        if dispatch_service is None:
            raise ValueError(
                "dispatch_service cannot be None."
            )

        self._repository = delivery_repository
        self._retry_policy = retry_policy
        self._dispatch_service = dispatch_service

    async def retry(
        self,
        *,
        request: PlatformIntelligenceEscalationDispatchRequest,
        email_request: object,
        reference_at: datetime,
    ) -> PlatformIntelligenceEscalationRetryExecutionResult:
        record = await self._repository.get_by_notification_key(
            request.notification_key
        )

        if record is None:
            raise ValueError(
                "Retry requires an existing durable delivery record."
            )

        if record.notification_key != request.notification_key:
            raise ValueError(
                "Delivery record notification key does not match request."
            )

        policy_result = self._retry_policy.evaluate(
            record=record,
            reference_at=reference_at,
        )

        if (
            policy_result.decision
            is PlatformIntelligenceEscalationRetryDecision.NOT_ELIGIBLE
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_ELIGIBLE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=policy_result.reason,
            )

        if (
            policy_result.next_eligible_at is not None
            and reference_at < policy_result.next_eligible_at
        ):
            return PlatformIntelligenceEscalationRetryExecutionResult(
                status=(
                    PlatformIntelligenceEscalationRetryExecutionStatus
                    .NOT_DUE
                ),
                notification_key=record.notification_key,
                attempt=record.attempt,
                reason=(
                    "Retry is eligible but the policy-derived "
                    "backoff window has not elapsed."
                ),
            )

        if (
            record.status
            is PlatformIntelligenceEscalationDeliveryStatus.FAILED
        ):
            await self._repository.start_retry(
                notification_key=record.notification_key,
            )

        # STARTED recovery deliberately does not call start_retry().
        # The durable attempt remains the same because this is recovery
        # of the interrupted attempt rather than a new FAILED retry.
        dispatch_result = await self._dispatch_service.dispatch(
            request=request,
            email_request=email_request,
        )

        return PlatformIntelligenceEscalationRetryExecutionResult(
            status=(
                PlatformIntelligenceEscalationRetryExecutionStatus
                .RETRIED
            ),
            notification_key=record.notification_key,
            attempt=(
                dispatch_result.attempt
                if dispatch_result.attempt is not None
                else record.attempt
            ),
            reason=policy_result.reason,
            dispatch_result=dispatch_result,
        )

from app.application.operations_center.platform_intelligence_escalation_delivery import (

class PlatformIntelligenceEscalationRetryCandidateDiscoveryService:
    """Discover durable retry work without executing it.

    This boundary deliberately does not:
    - call the retry policy;
    - mutate delivery state;
    - send email;
    - schedule work;
    - claim/lock candidates.

    Policy eligibility and execution remain separate steps.
    """

    def __init__(
        self,
        *,
        repository: PlatformIntelligenceEscalationRetryCandidateRepository,
    ) -> None:
        if repository is None:
            raise ValueError("repository cannot be None.")

        self._repository = repository

    async def discover(
        self,
        *,
        limit: int = 100,
    ) -> PlatformIntelligenceEscalationRetryCandidateBatch:
        if limit < 1:
            raise ValueError("limit must be at least 1.")

        records = await self._repository.list_retry_candidates(
            limit=limit,
        )

        candidates: list[PlatformIntelligenceEscalationRetryCandidate] = []

        for record in records:
            if record.status not in (
                PlatformIntelligenceEscalationDeliveryStatus.FAILED,
                PlatformIntelligenceEscalationDeliveryStatus.STARTED,
            ):
                continue

            candidates.append(
                PlatformIntelligenceEscalationRetryCandidate(
                    notification_key=record.notification_key,
                    recommendation_id=record.recommendation_id,
                    risk_decision_id=record.risk_decision_id,

from __future__ import annotations

class PlatformIntelligenceEscalationRetryClaimRepository(Protocol):
    async def claim_retry_candidate(
        self,
        *,
        notification_key: str,

class PlatformIntelligenceEscalationRetryClaimService:
    """Claims a candidate for one bounded execution attempt.

    Claiming is coordination only. It does not evaluate retry policy, send
    email, mutate the delivery lifecycle beyond the lease, or authorize the
    escalation itself.
    """

    def __init__(
        self,
        repository: PlatformIntelligenceEscalationRetryClaimRepository,
        *,
        lease_seconds: int = 300,
    ) -> None:
        if lease_seconds < 1:
            raise ValueError("lease_seconds must be at least 1.")
        self._repository = repository
        self._lease_seconds = lease_seconds

    async def claim(
        self,
        *,
        notification_key: str,
        claimed_at: datetime | None = None,
    ) -> PlatformIntelligenceEscalationRetryClaim | None:
        if not isinstance(notification_key, str) or not notification_key.strip():
            raise ValueError("notification_key must be nonblank.")

        reference_at = claimed_at or datetime.now(timezone.utc)
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("claimed_at must be timezone-aware.")

        claim_token = str(uuid4())
        lease_expires_at = reference_at + timedelta(seconds=self._lease_seconds)

        claimed = await self._repository.claim_retry_candidate(
            notification_key=notification_key,
            claim_token=claim_token,
            claimed_at=reference_at,
            lease_expires_at=lease_expires_at,
        )

        if not claimed:
            return None

        return PlatformIntelligenceEscalationRetryClaim(
            notification_key=notification_key,
            claim_token=claim_token,
            claimed_at=reference_at,
            lease_expires_at=lease_expires_at,
            status=RetryCandidateClaimStatus.CLAIMED,
        )

"""Bounded, scheduler-free orchestration boundary for PI retry work."""

from __future__ import annotations

from datetime import datetime

from app.application.operations_center.platform_intelligence_escalation_retry_candidate_discovery import (
    PlatformIntelligenceEscalationRetryCandidateDiscoveryService,
)
from app.application.operations_center.platform_intelligence_escalation_retry_orchestration import (
    PlatformIntelligenceRetryRunStatus,
)
from app.application.operations_center.platform_intelligence_escalation_retry_orchestrator import (
    PlatformIntelligenceEscalationRetryOrchestrator,
)
from app.application.operations_center.platform_intelligence_retry_worker import (
    PlatformIntelligenceRetryWorkerItemResult,
    PlatformIntelligenceRetryWorkerItemStatus,
    PlatformIntelligenceRetryWorkerRequest,
    PlatformIntelligenceRetryWorkerResult,
)


class PlatformIntelligenceRetryWorker:
    """Process one bounded candidate batch through the governed retry chain.

    This class deliberately does not:
    - schedule itself;
    - sleep;
    - create background tasks;
    - access providers;
    - implement retry policy;
    - claim candidates directly;
    - mutate risk or human-review state.

    The caller owns invocation timing and supplies the reference timestamp.
    """

    def __init__(
        self,
        *,
        discovery_service: PlatformIntelligenceEscalationRetryCandidateDiscoveryService,
        orchestrator: PlatformIntelligenceEscalationRetryOrchestrator,
    ) -> None:
        if discovery_service is None:
            raise ValueError("discovery_service cannot be None.")

        if orchestrator is None:
            raise ValueError("orchestrator cannot be None.")

        self._discovery_service = discovery_service
        self._orchestrator = orchestrator

    async def run(
        self,
        *,
        request: PlatformIntelligenceRetryWorkerRequest,
        reference_at: datetime,
    ) -> PlatformIntelligenceRetryWorkerResult:
        if reference_at.tzinfo is None or reference_at.utcoffset() is None:
            raise ValueError("reference_at must be timezone-aware.")

        batch = await self._discovery_service.discover(
            limit=request.limit,
        )

        results: list[PlatformIntelligenceRetryWorkerItemResult] = []

        for candidate in batch.candidates:
            try:
                execution = await self._orchestrator.run_candidate(
                    candidate=candidate,
                    reference_at=reference_at,
                )

            except Exception as exc:
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=candidate.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .FAILED
                        ),
                        reason="CANDIDATE_EXECUTION_ERROR",
                        attempt=candidate.attempt,
                        error_type=type(exc).__name__,
                    )
                )
                continue

            if (
                execution.status
                is PlatformIntelligenceRetryRunStatus.SUCCEEDED
            ):
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .SUCCEEDED
                        ),
                        reason=execution.reason,
                        attempt=execution.attempt,
                        provider_message_id=(
                            execution.provider_message_id
                        ),
                    )
                )

            elif (
                execution.status
                is PlatformIntelligenceRetryRunStatus.SKIPPED
            ):
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .SKIPPED
                        ),
                        reason=execution.reason,
                        attempt=execution.attempt,
                    )
                )

            else:
                results.append(
                    PlatformIntelligenceRetryWorkerItemResult(
                        notification_key=execution.notification_key,
                        status=(
                            PlatformIntelligenceRetryWorkerItemStatus
                            .FAILED
                        ),
                        reason=execution.reason,
                        attempt=execution.attempt,
                    )
                )

        succeeded = sum(
            item.status
            is PlatformIntelligenceRetryWorkerItemStatus.SUCCEEDED
            for item in results
        )

        skipped = sum(
            item.status
            is PlatformIntelligenceRetryWorkerItemStatus.SKIPPED
            for item in results
        )

        failed = sum(
            item.status
            is PlatformIntelligenceRetryWorkerItemStatus.FAILED
            for item in results
        )

        return PlatformIntelligenceRetryWorkerResult(
            requested_limit=request.limit,
            discovered_count=len(batch.candidates),
            processed_count=len(results),
            succeeded_count=succeeded,
            skipped_count=skipped,
            failed_count=failed,
            items=tuple(results),
        )

class CybersecurityAPSchedulerAdapter:
    """APScheduler implementation boundary.

    All scheduled application workflows share this single scheduler.

    Optional workflows remain dependency-injected so existing
    cybersecurity-only tests and deployments remain backwards compatible.
    """

    def __init__(
        self,
        scheduler,
        workflow_service,
        governance_reminder_workflow=None,
        governance_reminder_interval_minutes: int = 60,
        academy_one_week_workflow=None,
        academy_one_week_hour: int = 8,
        academy_one_week_minute: int = 15,
    ) -> None:
        if scheduler is None:
            raise ValueError("scheduler cannot be None.")

        if workflow_service is None:
            raise ValueError("workflow_service cannot be None.")

        if governance_reminder_interval_minutes <= 0:
            raise ValueError(
                "governance_reminder_interval_minutes must be > 0."
            )

        if not 0 <= academy_one_week_hour <= 23:
            raise ValueError(
                "academy_one_week_hour must be between 0 and 23."
            )

        if not 0 <= academy_one_week_minute <= 59:
            raise ValueError(
                "academy_one_week_minute must be between 0 and 59."
            )

        self.scheduler = scheduler
        self.workflow_service = workflow_service

        self.governance_reminder_workflow = (
            governance_reminder_workflow
        )
        self.governance_reminder_interval_minutes = (
            governance_reminder_interval_minutes
        )

        self.academy_one_week_workflow = (
            academy_one_week_workflow
        )
        self.academy_one_week_hour = academy_one_week_hour
        self.academy_one_week_minute = academy_one_week_minute

    def register_jobs(self) -> None:
        """Register all enabled workflows on the single scheduler."""

        self.scheduler.add_job(
            self.workflow_service.create_daily_report,
            trigger="cron",
            hour=DAILY_REPORT_HOUR,
            minute=DAILY_REPORT_MINUTE,
            id="cybersecurity_daily_report",
            replace_existing=True,
        )

        self.scheduler.add_job(
            self.workflow_service.create_weekly_report,
            trigger="cron",
            day_of_week=WEEKLY_REPORT_DAY,
            hour=WEEKLY_REPORT_HOUR,
            minute=WEEKLY_REPORT_MINUTE,
            id="cybersecurity_weekly_report",
            replace_existing=True,
        )

        if self.governance_reminder_workflow is not None:
            self.scheduler.add_job(
                self.governance_reminder_workflow.run,
                trigger="interval",
                minutes=self.governance_reminder_interval_minutes,
                id="privacy_governance_reminder",
                replace_existing=True,
            )

        if self.academy_one_week_workflow is not None:
            self.scheduler.add_job(
                self.academy_one_week_workflow.run,
                trigger="cron",
                hour=self.academy_one_week_hour,
                minute=self.academy_one_week_minute,
                id="academy_one_week_milestone",
                replace_existing=True,
            )

    def register_critical_handler(self, handler) -> None:
        self.critical_handler = handler

    async def trigger_critical_event(self, event: dict):
        if not hasattr(self, "critical_handler"):
            return None

        return await self.critical_handler(event)

    def start(self) -> None:
        if not self.scheduler.running:
            self.scheduler.start()

    def shutdown(self) -> None:
        if self.scheduler.running:
            self.scheduler.shutdown()

) -> CybersecuritySchedulerRuntime:

    if workflow_service is None:
        email_service = CybersecurityEmailService(
            recipient="davdbarbero1999@gmail.com",
        )

        workflow_service = CybersecuritySchedulerWorkflow(
            email_service=email_service,
        )

    scheduler = AsyncIOScheduler(
        timezone=SCHEDULER_TIMEZONE,

        academy_one_week_workflow=academy_one_week_workflow,
    )

    return CybersecuritySchedulerRuntime(
        scheduler_adapter=adapter,

"""
Cybersecurity Scheduler Runtime.

Lifecycle manager for cybersecurity scheduling.

Responsibilities:
- start scheduler;
- register jobs;
- stop scheduler;
- expose runtime state.

Does NOT:
- execute reports;
- send emails;
- make security decisions;
- perform remediation.
"""

from __future__ import annotations


from cybersecurity.scheduler.cybersecurity_job_registry import (
    CybersecurityJobRegistry,
)



class CybersecuritySchedulerRuntime:
    """
    Scheduler lifecycle boundary.

    Supports:

    - registry based execution/testing;
    - production scheduler adapter.

    Architecture:

        FastAPI lifespan
              |
              v
    CybersecuritySchedulerRuntime
              |
              +----------------+
              |                |
              v                v

        Job Registry     Scheduler Adapter

          (tests)          (production)

    """



    def __init__(
        self,
        scheduler_adapter=None,
        job_registry: CybersecurityJobRegistry | None = None,
    ) -> None:
        """
        Initialize scheduler runtime.
        """


        self.scheduler_adapter = (
            scheduler_adapter
        )


        self.job_registry = (
            job_registry
            or CybersecurityJobRegistry()
        )


        self._started = False



    async def start(
        self,
    ) -> None:
        """
        Start scheduler runtime.

        Registers scheduled cybersecurity workflows.
        """


        if self.scheduler_adapter is not None:

            #
            # Production scheduler boundary.
            #
            # Example:
            # APScheduler adapter
            #

            self.scheduler_adapter.register_jobs()

            self.scheduler_adapter.start()


        else:

            #
            # Registry mode.
            #
            # Used for testing and controlled execution.
            #

            self.job_registry.register(

                name=(
                    "cybersecurity_daily_report"
                ),

                schedule=(
                    "daily_08:00"
                ),

            )


            self.job_registry.register(

                name=(
                    "cybersecurity_weekly_report"
                ),

                schedule=(
                    "weekly_monday_08:00"
                ),

            )


            self.job_registry.register(

                name=(
                    "cybersecurity_critical_alert"
                ),

                schedule=(
                    "event_immediate"
                ),

            )



        self._started = True



    async def stop(
        self,
    ) -> None:
        """
        Stop scheduler runtime.
        """


        if self.scheduler_adapter is not None:

            self.scheduler_adapter.shutdown()



        self._started = False



    def status(
        self,
    ) -> dict:
        """
        Return scheduler runtime status.

        Provides backward-compatible fields:

        - jobs
        - registered_jobs
        """


        jobs = (
            self.job_registry.list_jobs()
        )



        return {

            "started":
                self._started,


            "adapter_enabled":
                self.scheduler_adapter is not None,


            #
            # Existing test contract
            #
            "jobs":
                jobs,


            #
            # Extended API contract
            #
            "registered_jobs":
                jobs,

        }

def _required_environment(name: str) -> str:
    value = os.getenv(name, "").strip()

    if not value:
        raise RuntimeError(
            f"{name} is required for authentication email composition."
        )

    return value

@asynccontextmanager
async def lifespan(
    application: FastAPI,
):
    scheduler_runtime = application.state.scheduler_runtime

    try:
        await scheduler_runtime.start()
    except Exception as exc:
        print(
            "Cybersecurity scheduler startup failed:",
            exc,
        )

    yield

    try:
        await scheduler_runtime.stop()
    except Exception as exc:
        print(
            "Cybersecurity scheduler shutdown failed:",
            exc,
        )

    finally:
        telegram_client = getattr(
            application.state,
            "announcement_telegram_client",
            None,
        )

        if telegram_client is not None:
            try:
                await telegram_client.close()
            except Exception as exc:
                print(
                    "Telegram announcement client shutdown failed:",
                    exc,
                )

        auth_engine = getattr(
            application.state,
            "auth_engine",
            None,
        )

        if auth_engine is not None:
            try:
                await auth_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL authentication engine shutdown failed:",
                    exc,
                )

        provider_usage_engine = getattr(
            application.state,
            "provider_usage_engine",
            None,
        )

        if provider_usage_engine is not None:
            try:
                provider_usage_engine.dispose()
            except Exception as exc:
                print(
                    "PostgreSQL provider-usage engine shutdown failed:",
                    exc,
                )


def create_application() -> FastAPI:
    academy_config = (
        AcademyActivationConfig.from_environment()
    )

    auth_engine = create_async_database_engine()

    auth_session_factory = (
        create_async_session_factory(
            auth_engine,
        )
    )

    auth_repository = (
        PostgreSQLAuthenticationRepository(
            auth_session_factory,
        )
    )

    email_provider_manager = EmailProviderManager()

    authentication_email_sender = AuthenticationEmailSenderAdapter(
        provider_manager=email_provider_manager,
    )

    authentication_email_service = AuthenticationEmailService(
        authentication_email_sender,
        frontend_base_url=_required_environment(
            "EASYCHATBOT_PORTAL_URL",
        ),
    )

    authentication_service = AuthenticationService(
        auth_repository,
        email_service=authentication_email_service,
    )

    # ------------------------------------------------------------------
    # Developer Administration
    #
    # Production path:
    #
    #   DeveloperAdminService
    #          |
    #          +--> DeveloperAdminUnitOfWorkFactory
    #                  |
    #                  +--> fresh PostgreSQLDeveloperAdminUnitOfWork
    #                          |
    #                          +--> fresh AsyncSession
    #                          |
    #                          +--> fresh transaction
    #                          |
    #                          +--> auth_user_roles
    #                          |
    #                          +--> developer_admin_audit_events
    #          |
    #          +--> DeveloperAdminAuditService
    #
    # Each Developer Admin mutation receives a fresh Unit-of-Work
    # instance. The role mutation and its security audit evidence
    # therefore remain within the same database transaction while
    # mutable UoW/session state is never shared between operations.
    #
    # The standalone audit repository owns transactions only when its
    # standalone ``record`` operation is used. The production Developer
    # Admin mutation uses ``record_in_session`` through the UoW.
    # ------------------------------------------------------------------

    developer_admin_unit_of_work_factory = (
        PostgreSQLDeveloperAdminUnitOfWorkFactory(
            auth_session_factory,
        )
    )

    developer_admin_audit_service = (
        DeveloperAdminAuditService()
    )

    developer_admin_service = DeveloperAdminService(
        repository=auth_repository,
        unit_of_work_factory=developer_admin_unit_of_work_factory,
        audit_service=developer_admin_audit_service,
    )

    # W15.2.1 Transaction Authorization
    #
    # Transaction authorization is composed once at the application root.
    # The registry and consumer are application-lifetime components and
    # provide their own synchronization boundaries.
    #
    # Authentication assurance remains a separate trusted-evidence
    # boundary. This composition does not manufacture MFA/A2 evidence.
    # ------------------------------------------------------------------

    authorization_service = AuthorizationService()

    transaction_authorization_registry = (
        TransactionAuthorizationRegistry()
    )

    transaction_authorization_consumer = (
        TransactionAuthorizationConsumer(
            transaction_authorization_registry,
        )
    )

    transaction_authorization_service = (
        TransactionAuthorizationApplicationService(
            authorization_service=authorization_service,
            registry=transaction_authorization_registry,
            consumer=transaction_authorization_consumer,
        )
    )

    trusted_clock = SystemTrustedClock()

    developer_admin_transaction_service = (

            developer_admin_service=developer_admin_service,
            transaction_authorization_service=(
                transaction_authorization_service

            clock=trusted_clock,

    )

    # Provider usage / synchronous application persistence
    #
    # The provider-usage repository is synchronous by contract.
    # It therefore receives a synchronous engine/session factory.
    #
    # It must not receive the async authentication session factory.
    # ------------------------------------------------------------------

    provider_usage_engine = (

    # ------------------------------------------------------------------
