from __future__ import annotations

from dataclasses import dataclass
from enum import StrEnum
from datetime import datetime


class BetaAccessStatus(StrEnum):
    PENDING_VERIFICATION = "PENDING_VERIFICATION"
    PENDING_APPROVAL = "PENDING_APPROVAL"
    ELIGIBLE = "ELIGIBLE"
    TELEGRAM_CONNECTED = "TELEGRAM_CONNECTED"
    ACTIVE = "ACTIVE"
    DENIED = "DENIED"


@dataclass(frozen=True, slots=True)
class BetaUserState:
    user_id: str
    email: str
    email_verified: bool
    beta_approved: bool
    telegram_id: int | None = None
    active_world_id: str | None = None

    @property
    def eligible(self) -> bool:
        return self.email_verified and self.beta_approved

    @property
    def active(self) -> bool:
        return (
            self.eligible
            and self.telegram_id is not None
            and self.active_world_id is not None
        )


@dataclass(frozen=True, slots=True)
class BetaConnectionToken:
    user_id: str
    token: str
    expires_at: datetime
    used: bool = False


@dataclass(frozen=True, slots=True)
class TelegramBinding:
    user_id: str
    telegram_id: int
    chat_id: int
    connected_at: datetime

from __future__ import annotations

from dataclasses import replace
from datetime import datetime, timedelta, timezone
import logging
import secrets

from .models import BetaConnectionToken, BetaUserState, TelegramBinding
from .repositories import BetaTokenRepository, BetaUserRepository, TelegramBindingRepository

logger = logging.getLogger(__name__)


class BetaAccessService:
    def __init__(
        self,
        *,
        user_repository: BetaUserRepository,
        token_repository: BetaTokenRepository,
        binding_repository: TelegramBindingRepository,
        expiration_minutes: int = 15,
    ) -> None:
        if expiration_minutes <= 0:
            raise ValueError("expiration_minutes must be positive.")
        self._users = user_repository
        self._tokens = token_repository
        self._bindings = binding_repository
        self._expiration_minutes = expiration_minutes

    async def approve_beta(self, user_id: str) -> BetaUserState:
        user = await self._require_user(user_id)
        if not user.email_verified:
            raise ValueError("Email must be verified before Beta approval.")
        return await self._users.save(replace(user, beta_approved=True))

    async def create_connection_token(self, user_id: str) -> BetaConnectionToken:
        user = await self._require_user(user_id)
        if not user.email_verified:
            raise ValueError("Email is not verified.")
        if not user.beta_approved:
            raise ValueError("User is not Beta approved.")

        token = BetaConnectionToken(
            user_id=user.user_id,
            token=secrets.token_urlsafe(32),
            expires_at=(
                datetime.now(timezone.utc)
                + timedelta(minutes=self._expiration_minutes)
            ),
        )
        await self._tokens.save(token)
        logger.info("Beta Telegram token created user_id=%s", user.user_id)
        return token

    async def validate_and_bind(
        self,
        *,
        token: str,
        telegram_id: int,
        chat_id: int,
    ) -> BetaUserState:
        if not token:
            raise ValueError("Connection token is required.")
        if telegram_id <= 0:
            raise ValueError("Telegram ID must be positive.")
        if chat_id == 0:
            raise ValueError("Chat ID is required.")

        record = await self._tokens.get_by_token(token)
        if record is None:
            raise ValueError("Invalid connection token.")
        if record.used:
            raise ValueError("Connection token already used.")

        expires_at = record.expires_at
        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(tzinfo=timezone.utc)
        if expires_at <= datetime.now(timezone.utc):
            raise ValueError("Connection token expired.")

        user = await self._require_user(record.user_id)
        if not user.eligible:
            raise ValueError("User is not eligible for Beta Telegram access.")

        existing = await self._users.get_by_telegram_id(telegram_id)
        if existing is not None and existing.user_id != user.user_id:
            raise ValueError("Telegram account is already connected to another user.")

        await self._bindings.save(
            TelegramBinding(
                user_id=user.user_id,
                telegram_id=telegram_id,
                chat_id=chat_id,
                connected_at=datetime.now(timezone.utc),
            )
        )
        await self._tokens.mark_used(token)
        return await self._users.save(replace(user, telegram_id=telegram_id))

    async def authorize_private_chat(self, telegram_id: int) -> BetaUserState:
        if telegram_id <= 0:
            raise ValueError("Telegram ID must be positive.")
        user = await self._users.get_by_telegram_id(telegram_id)
        if user is None:
            raise PermissionError("Telegram account is not connected.")
        if not user.active:
            raise PermissionError(
                "User is connected but is not ACTIVE for a Learning World."
            )
        return user

    async def _require_user(self, user_id: str) -> BetaUserState:
        user = await self._users.get(user_id)
        if user is None:
            raise ValueError("Academy user does not exist.")
        return user

from __future__ import annotations

from dataclasses import dataclass
from typing import Protocol

from .models import BetaUserState


@dataclass(frozen=True, slots=True)
class PreparedEmail:
    recipient: str
    subject: str
    body: str
    classroom_url: str


class EmailSender(Protocol):
    async def send(self, email: PreparedEmail) -> None: ...


class BetaTelegramInvitationService:
    def __init__(self, *, token_service, deep_link_service, email_sender: EmailSender):
        self._token_service = token_service
        self._deep_link_service = deep_link_service
        self._email_sender = email_sender

    async def send_invitation(self, user: BetaUserState) -> PreparedEmail:
        if not user.email_verified:
            raise ValueError("Email must be verified before Beta access.")
        if not user.beta_approved:
            raise ValueError("Beta approval is required before Telegram access.")

        connection_token = await self._token_service.execute(user.user_id)
        deep_link = self._deep_link_service.execute(connection_token.token)

        email = PreparedEmail(
            recipient=user.email,
            subject="Your EasyChatBot Academy classroom is ready",
            body=(
                "Welcome to EasyChatBot Academy.\n\n"
                "Your Beta access has been approved.\n\n"
                "Enter your private AI classroom in Telegram:\n"
                f"{deep_link.url}\n\n"
                "This link is single-use and expires shortly."
            ),
            classroom_url=deep_link.url,
        )
        await self._email_sender.send(email)
        return email

from __future__ import annotations

from dataclasses import dataclass
from typing import Protocol

from app.application.beta_access.beta_access_service import BetaAccessService


class TelegramReply(Protocol):
    async def __call__(self, *, chat_id: int, text: str) -> None: ...


@dataclass(frozen=True, slots=True)
class TelegramStartResult:
    accepted: bool
    message: str
    user_id: str | None = None


class BetaTelegramStartHandler:
    def __init__(self, *, beta_access_service: BetaAccessService, reply: TelegramReply):
        self._access = beta_access_service
        self._reply = reply

    async def handle(
        self,
        *,
        token: str,
        telegram_id: int,
        chat_id: int,
    ) -> TelegramStartResult:
        try:
            user = await self._access.validate_and_bind(
                token=token,
                telegram_id=telegram_id,
                chat_id=chat_id,
            )
        except (ValueError, PermissionError):
            message = (
                "This classroom link is invalid, expired, already used, "
                "or your Beta access is not ready. "
                "Please return to the Academy and request a new classroom link."
            )
            await self._reply(chat_id=chat_id, text=message)
            return TelegramStartResult(False, message)

        message = (
            "Welcome to your private EasyChatBot Academy classroom. "
            "Your Telegram account is now connected."
        )
        await self._reply(chat_id=chat_id, text=message)
        return TelegramStartResult(True, message, user.user_id)

from __future__ import annotations

from app.application.beta_access.models import BetaConnectionToken, BetaUserState, TelegramBinding


class InMemoryBetaUserRepository:
    def __init__(self):
        self._users = {}

    async def get(self, user_id):
        return self._users.get(user_id)

    async def save(self, user):
        self._users[user.user_id] = user
        return user

    async def get_by_telegram_id(self, telegram_id):
        return next(
            (u for u in self._users.values() if u.telegram_id == telegram_id),
            None,
        )


class InMemoryBetaTokenRepository:
    def __init__(self):
        self._tokens = {}

    async def save(self, token):
        self._tokens[token.token] = token

    async def get_by_token(self, token):
        return self._tokens.get(token)

    async def mark_used(self, token):
        record = self._tokens[token]
        self._tokens[token] = BetaConnectionToken(
            user_id=record.user_id,
            token=record.token,
            expires_at=record.expires_at,
            used=True,
        )


class InMemoryTelegramBindingRepository:
    def __init__(self):
        self._bindings = {}

    async def save(self, binding):
        self._bindings[binding.telegram_id] = binding
        return binding

    async def get_by_telegram_id(self, telegram_id):
        return self._bindings.get(telegram_id)

import pytest

from app.application.beta_access.beta_access_service import BetaAccessService

    InMemoryBetaTokenRepository,

@pytest.mark.asyncio
async def test_telegram_identity_cannot_be_owned_by_two_users():
    users = InMemoryBetaUserRepository()
    tokens = InMemoryBetaTokenRepository()
    bindings = InMemoryTelegramBindingRepository()

    service = BetaAccessService(
        user_repository=users,
        token_repository=tokens,
        binding_repository=bindings,
    )

    for user_id in ("a", "b"):
        await users.save(
            BetaUserState(
                user_id=user_id,
                email=f"{user_id}@example.com",
                email_verified=True,
                beta_approved=True,
                active_world_id="world-fr",
            )
        )

    first = await service.create_connection_token("a")
    await service.validate_and_bind(
        token=first.token,
        telegram_id=111,
        chat_id=111,
    )

    second = await service.create_connection_token("b")

    with pytest.raises(ValueError, match="already connected"):
        await service.validate_and_bind(
            token=second.token,
            telegram_id=111,
            chat_id=111,
        )

import pytest

from app.application.beta_access.beta_access_service import BetaAccessService

    InMemoryBetaTokenRepository,

)

class FakeTokenService:
    def __init__(self, service):
        self.service = service

    async def execute(self, user_id):
        return await self.service.create_connection_token(user_id)


class FakeDeepLinkService:
    def execute(self, token):
        class Link:
            url = f"https://t.me/EasyChatBotAcademyBot?start={token}"
        return Link()


@pytest.mark.asyncio
async def test_registration_to_active_private_telegram_flow():
    users = InMemoryBetaUserRepository()
    tokens = InMemoryBetaTokenRepository()
    bindings = InMemoryTelegramBindingRepository()
    access = BetaAccessService(
        user_repository=users,
        token_repository=tokens,
        binding_repository=bindings,
    )

    await users.save(
        BetaUserState(
            user_id="user-001",
            email="learner@example.com",
            email_verified=True,
            beta_approved=False,
            active_world_id="world-fr",
        )
    )

    approved = await access.approve_beta("user-001")
    email_sender = InMemoryBetaEmailSender()
    invitation = BetaTelegramInvitationService(
        token_service=FakeTokenService(access),
        deep_link_service=FakeDeepLinkService(),
        email_sender=email_sender,
    )
    email = await invitation.send_invitation(approved)

    raw_token = email.classroom_url.split("start=", 1)[1]
    replies = []

    async def reply(*, chat_id, text):
        replies.append((chat_id, text))

    handler = BetaTelegramStartHandler(
        beta_access_service=access,
        reply=reply,
    )

    result = await handler.handle(
        token=raw_token,
        telegram_id=123456789,
        chat_id=123456789,
    )

    assert result.accepted is True
    active = await access.authorize_private_chat(123456789)
    assert active.user_id == "user-001"
    assert active.active is True

    second = await handler.handle(
        token=raw_token,
        telegram_id=123456789,
        chat_id=123456789,
    )
    assert second.accepted is False
