          screen.getByText(
            "ON TRACK",
          ),

            method: "GET",
            credentials: "include",
          }),
        );

        fetchMock.mockRestore();
      },
    );

    it(
      "shows an API failure and provides retry",
      async () => {
        const fetchMock =
          vi
            .spyOn(globalThis, "fetch")
            .mockRejectedValue(
              new Error(
                "Network failure",
              ),
            );

        render(
          <ProjectControlPage />,
        );

        await waitFor(() => {
          expect(
            screen.getByRole(
              "alert",
            ),
          ).toBeInTheDocument();
        });

        expect(
          screen.getByText(
            "Unable to load Project Control data.",
          ),
        ).toBeInTheDocument();

        const retryButton =
          screen.getByRole(
            "button",
            {
              name: "Retry",
            },
          );

        fireEvent.click(
          retryButton,
        );

        await waitFor(() => {
          expect(
            fetchMock,
          ).toHaveBeenCalledTimes(2);

          expect(
            screen.getByRole(
              "alert",
            ),
          ).toHaveTextContent(
            "Unable to load Project Control data.",
          );
        });

        fetchMock.mockRestore();
      },
    );

    it(
      "shows unavailable source information",
      async () => {
        const degradedSnapshot = {
          ...snapshot,
          sources: [
            {
              ...snapshot.sources[0],
              status: "unavailable",
              error:
                "Source unavailable",
