  detail: string;

export class AuthenticationApiError extends Error {
  readonly status: number;

  constructor(
    message: string,
    status: number,
  ) {
    super(message);
    this.name = "AuthenticationApiError";
    this.status = status;
  }
}

async function parseApiError(
  response: Response,
  fallback: string,
): Promise<AuthenticationApiError> {
  let detail = fallback;

  try {
    const body: unknown = await response.json();

    if (
      body &&
      typeof body === "object" &&
      "detail" in body &&
      typeof body.detail === "string" &&
      body.detail.trim()
    ) {
      detail = body.detail;
    }
  } catch {
    // Preserve safe generic API error.
  }

  return new AuthenticationApiError(detail, response.status);
}

export async function getSession(): Promise<SessionResponse> {
  const response = await fetch("/auth/session", {
    method: "GET",
    credentials: "include",
    headers: {
      Accept: "application/json",
    },
  });

  if (!response.ok) {
    throw await parseApiError(
      response,
      `Authentication session check failed (${response.status}).`,
    );
  }

  return response.json() as Promise<SessionResponse>;
}

export async function register(
  credentials: RegisterRequest,
): Promise<RegisterResponse> {
  const response = await fetch("/auth/register", {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
      Accept: "application/json",
    },
    body: JSON.stringify(credentials),
  });

  if (!response.ok) {
    throw await parseApiError(
      response,
      "Unable to create the account. Please try again later.",
    );
  }

  return response.json() as Promise<RegisterResponse>;
}

export async function verifyEmail(
  token: string,
): Promise<VerifyEmailResponse> {
  const response = await fetch(
    `/auth/verify-email?token=${encodeURIComponent(token)}`,
    {
      method: "POST",
      credentials: "include",
      headers: {
        Accept: "application/json",
      },
    },
  );

  if (!response.ok) {
    throw await parseApiError(
      response,
      "This verification link is invalid or has expired.",
    );
  }

  return response.json() as Promise<VerifyEmailResponse>;
}

export async function login(
  credentials: LoginRequest,
): Promise<LoginResponse> {
  const response = await fetch("/auth/login", {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
      Accept: "application/json",
    },
    body: JSON.stringify(credentials),
  });

  if (!response.ok) {
    throw await parseApiError(
      response,
      `Authentication failed (${response.status}).`,
    );
  }

  return response.json() as Promise<LoginResponse>;
}

export async function logout(): Promise<void> {
  const response = await fetch("/auth/logout", {
    method: "POST",
    credentials: "include",
  });

  if (!response.ok && response.status !== 204) {
    throw new AuthenticationApiError(
      `Logout failed (${response.status}).`,
      response.status,
    );
  }
}

import {
  FormEvent,
  useEffect,
  useRef,
  useState,
} from "react";
import {
  Link,
  useSearchParams,
} from "react-router-dom";

import {
  AuthenticationApiError,
  confirmPasswordReset,
} from "../../src/auth/authApi";

type PageState =
  | "ready"
  | "submitting"
  | "success"
  | "missing"
  | "failed";

export default function ResetPasswordPage() {
  const [searchParams] = useSearchParams();

  const [token, setToken] = useState("");
  const [state, setState] =
    useState<PageState>("ready");

  const [newPassword, setNewPassword] =
    useState("");

  const [confirmPassword, setConfirmPassword] =
    useState("");

  const [error, setError] =
    useState<string | null>(null);

  const initialized = useRef(false);

  useEffect(() => {
    if (initialized.current) {
      return;
    }

    initialized.current = true;

    const rawToken =
      searchParams.get("token")?.trim() ?? "";

    if (!rawToken) {
      setState("missing");
      return;
    }

    /*
     * Capture the token once and immediately remove it
     * from the visible browser URL.
     */
    setToken(rawToken);

    window.history.replaceState(
      {},
      document.title,
      window.location.pathname,
    );
  }, [searchParams]);

  async function handleSubmit(
    event: FormEvent<HTMLFormElement>,
  ) {
    event.preventDefault();

    setError(null);

    if (!newPassword) {
      setError("Please enter a new password.");
      return;
    }

    if (newPassword !== confirmPassword) {
      setError("Passwords do not match.");
      return;
    }

    setState("submitting");

    try {
      await confirmPasswordReset({
        token,
        new_password: newPassword,
        confirm_password: confirmPassword,
      });

      /*
       * Clear credentials/token from component state after
       * successful submission.
       */
      setToken("");
      setNewPassword("");
      setConfirmPassword("");

      setState("success");
    } catch (cause) {
      setState("failed");

      setError(
        cause instanceof AuthenticationApiError
          ? cause.message
          : "This password reset link is invalid or has expired.",
      );
    }
  }

  if (state === "missing") {
    return (
      <main
        aria-label="Password reset"
        style={styles.page}
      >
        <section style={styles.card}>
          <div style={styles.eyebrow}>
            EASYCHATBOT ACADEMY
          </div>

          <h1 style={styles.title}>
            Reset link missing
          </h1>

          <p style={styles.text}>
            This password reset link does not contain
            the information we need.
          </p>

          <Link
            to="/forgot-password"
            style={styles.primaryLink}
          >
            Request a new link
          </Link>
        </section>
      </main>
    );
  }

  if (state === "success") {
    return (
      <main
        aria-label="Password updated"
        style={styles.page}
      >
        <section style={styles.card}>
          <div style={styles.eyebrow}>
            EASYCHATBOT ACADEMY
          </div>

          <h1 style={styles.title}>
            Password updated! ðŸŽ‰

            You're all set.

            Your password has been changed and your

            for security.
          </p>

          <Link
            to="/login"
            style={styles.primaryLink}
          >
            Continue to sign in
          </Link>
        </section>
      </main>
    );
  }

  return (
    <main
      aria-label="Create new password"
      style={styles.page}
    >
      <section style={styles.card}>
        <div style={styles.eyebrow}>
          EASYCHATBOT ACADEMY
        </div>

        <h1 style={styles.title}>
          Create a new password
        </h1>

        <p style={styles.text}>
          Let's get you back into the Academy.
        </p>

        <form onSubmit={handleSubmit}>
          <label
            htmlFor="new-password"
            style={styles.label}
          >
            New password
          </label>

          <input
            id="new-password"
            type="password"
            autoComplete="new-password"
            minLength={12}
            maxLength={256}
            value={newPassword}
            onChange={(event) =>
              setNewPassword(event.target.value)
            }
            required
            style={styles.input}
          />

          <label
            htmlFor="confirm-password"
            style={styles.label}
          >
            Confirm new password
          </label>

          <input
            id="confirm-password"
            type="password"
            autoComplete="new-password"
            minLength={12}
            maxLength={256}
            value={confirmPassword}
            onChange={(event) =>
              setConfirmPassword(event.target.value)
            }
            required
            style={styles.input}
          />

          {error && (
            <p
              role="alert"
              style={styles.error}
            >
              {error}
            </p>
          )}

          <p style={styles.requirement}>
            Your password must contain at least
            12 characters.
          </p>

          <button
            type="submit"
            disabled={
              state === "submitting" ||
