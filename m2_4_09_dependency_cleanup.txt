def build_voice_call_service() -> VoiceCallService:
    """
    Create configured voice call service.
    """

    #
    # Application configuration root:
    #
    # backend/app
    #

    app_root = (
        Path(__file__)
        .resolve()
        .parents[3]
    )


    world_repository = YamlWorldRepository(
        worlds_path=(
            app_root
            / "configuration"
            / "worlds"
        )
    )


    character_repository = YamlCharacterRepository(
        profiles_path=(
            Path(
                r"C:\AI Folder\FrenchTeacherAI"
            )
            / "docs"
            / "specifications"
            / "characters"
            / "profiles"
        )
    )


    context_builder = InteractionContextBuilder(
        world_resolver=WorldResolver(
            world_repository=world_repository,
        ),
        character_resolver=CharacterResolver(
            character_repository=character_repository,
        ),
    )


    #
    # Voice interaction-context bridge.
    #
    # Converts the existing voice call execution
    # context into the established UAI InteractionContext.
    #

    interaction_context_service = (
        VoiceInteractionContextService(
            context_builder=context_builder,
        )
    )


    #
    # LiveKit infrastructure boundary.
    #

    livekit_adapter = LiveKitAdapter()


    #
    # Speech provider configuration.
    #

    speech_config = SpeechProviderConfig(
        provider_name="openai",
        model_name="tts-1",
        default_language="fr",
    )


    speech_provider = (
        SpeechProviderFactory.create(
            config=speech_config,
        )
    )


    speech_runtime = VoiceSpeechRuntime(
        speech_provider=speech_provider,
    )


    session_service = VoiceCharacterSessionService(
        livekit_adapter=livekit_adapter,
        speech_runtime=speech_runtime,
    )


    orchestrator = VoiceCallOrchestrator(
        context_builder=context_builder,
        session_service=session_service,
    )


    #
    # Keep the interaction-context bridge available
    # through the composition root for realtime voice
    # infrastructure.
    #
    # The existing VoiceCallService contract remains
    # unchanged.
    #

    _ = interaction_context_service


    return VoiceCallService(
        orchestrator=orchestrator,
    )



def build_voice_interaction_context_service(
) -> VoiceInteractionContextService:
    """
    Build the application service that converts
    a voice execution context into InteractionContext.

    This is the composition entry point for
    realtime voice infrastructure.
    """

    app_root = (
        Path(__file__)
        .resolve()
        .parents[3]
    )


    world_repository = YamlWorldRepository(
        worlds_path=(
            app_root
            / "configuration"
            / "worlds"
        )
    )


    character_repository = YamlCharacterRepository(
        profiles_path=(
            Path(
                r"C:\AI Folder\FrenchTeacherAI"
            )
            / "docs"
            / "specifications"
            / "characters"
            / "profiles"
        )
    )


    context_builder = InteractionContextBuilder(
        world_resolver=WorldResolver(
            world_repository=world_repository,
        ),
        character_resolver=CharacterResolver(
            character_repository=character_repository,
        ),

    return VoiceInteractionContextService(
        context_builder=context_builder,

Infrastructure entry point for realtime voice sessions.
"""

def _validate_livekit_configuration() -> None:
    """Validate required LiveKit configuration."""

    required = (
        "LIVEKIT_URL",
        "LIVEKIT_API_KEY",
        "LIVEKIT_API_SECRET",
    )

    missing = [
        item
        for item in required
        if not os.getenv(item)
    ]

    if missing:
        raise RuntimeError(
            "Missing LiveKit configuration: "
            + ", ".join(missing)
        )


_validate_livekit_configuration()


logger = logging.getLogger(__name__)


@dataclass(frozen=True, slots=True)
class LiveKitVoiceWorker:
    """Infrastructure worker for EasyChatBot realtime voice."""

    context_factory: VoiceCallContextFactory
    conversation_runtime: ConversationRuntime
    speech_runtime: VoiceSpeechRuntime

    async def run(
        self,
        *,
        ctx: JobContext,
    ) -> None:
        """Connect to LiveKit and start a realtime session."""

        metadata = self._read_job_metadata(ctx)

        request = self._create_voice_call_request(metadata)

        execution_context = self.context_factory.create(
            request=request,
        )

        interaction_context = self.interaction_context_service.build(
            context=execution_context,
        )

        character = interaction_context.character

        instructions = (
            f"You are {character.name}. "
            "You are a French learning character. "
            "Your personality reference is "
            f"{character.personality_reference}. "
            "Your teaching style is "
            f"{character.teaching_style}. "
            "Speak naturally and help the learner "
            "practice the target language."
        )

        realtime_agent = LiveKitRealtimeAgent(
            context=interaction_context,
            instructions=instructions,
            conversation_runtime=self.conversation_runtime,
            speech_runtime=self.speech_runtime,
        )

        ctx.log_context_fields = {
            "room_name": ctx.room.name,
            "learner_id": request.learner_id,
            "world_id": request.world_id,
            "character_id": request.character_id,
            "language_code": request.language_code,
        }

        logger.info(
            "Starting EasyChatBot LiveKit voice agent",
            extra={
                "room_name": ctx.room.name,
                "learner_id": request.learner_id,
                "world_id": request.world_id,
                "character_id": request.character_id,
                "language_code": request.language_code,
            },
        )

        await ctx.connect()

        agent = realtime_agent.create_agent()
        session = realtime_agent.create_session()

        await session.start(
            agent=agent,
            room=ctx.room,
        )

    @staticmethod
    def _read_job_metadata(
        ctx: JobContext,
    ) -> dict[str, object]:
        """Read and validate LiveKit job metadata."""

        raw_metadata = ctx.job.metadata

        if not raw_metadata:
            raise ValueError(
                "LiveKit job metadata is required."
            )

        try:
            metadata = json.loads(raw_metadata)
        except json.JSONDecodeError as exc:
            raise ValueError(
                "LiveKit job metadata must be valid JSON."
            ) from exc

        if not isinstance(metadata, dict):
            raise ValueError(
                "LiveKit job metadata must be a JSON object."
            )

        return metadata

    @staticmethod
    def _create_voice_call_request(
        metadata: dict[str, object],
    ) -> VoiceCallRequest:
        """Convert LiveKit metadata into application request."""

        required_fields = (
            "learner_id",
            "world_id",
            "character_id",
            "language_code",
        )

        values: dict[str, str] = {}

        for field in required_fields:
            value = metadata.get(field)

            if not isinstance(value, str):
                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' must be a string."
                )

            if not value.strip():
                raise ValueError(
                    f"LiveKit metadata field "
                    f"'{field}' cannot be empty."
                )

            values[field] = value.strip()

        return VoiceCallRequest(
            learner_id=values["learner_id"],
            world_id=values["world_id"],
            character_id=values["character_id"],
            language_code=values["language_code"],
        )


server = AgentServer()

_functionality_control_service, _control_engine = (
    create_standalone_functionality_control_service()
)

_worker = LiveKitVoiceWorker(
    context_factory=VoiceCallContextFactory(),
    conversation_runtime=build_conversation_runtime(),
    speech_runtime=build_voice_speech_runtime(
        functionality_control_service=(
            _functionality_control_service
        ),
    ),
)


@server.rtc_session(
    agent_name="easychatbot-voice",
)
async def entrypoint(
    ctx: JobContext,
) -> None:
    """LiveKit Agents entrypoint."""

    await _worker.run(ctx=ctx)


def run() -> None:
    """Start LiveKit Agents worker."""

    try:
        cli.run_app(server)
    finally:
        _control_engine.dispose()


__all__ = [
    "LiveKitVoiceWorker",
    "entrypoint",
    "run",
]


if __name__ == "__main__":
    run()
