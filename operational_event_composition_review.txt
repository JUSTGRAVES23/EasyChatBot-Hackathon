"""
EasyChatBot AI Operations Center operational event repository.

Provides the persistence boundary for immutable operational
events.

Architecture:

OperationalEvent
        |
        v
OperationalEventRepository
        |
        v
In-Memory / Future Persistent Store


Responsibilities:

- define the repository contract;
- store operational events;
- retrieve events deterministically;
- keep persistence concerns outside the event model.

Does NOT:

- interpret event meaning;
- execute agents;
- execute providers;
- change Runtime routing;
- create autonomous actions;
- send notifications.
"""

from __future__ import annotations

from abc import ABC, abstractmethod

from app.application.provider_management.operational_event import (
    OperationalEvent,
)


class OperationalEventRepository(ABC):
    """
    Persistence boundary for operational events.
    """

    @abstractmethod
    def save(
        self,
        event: OperationalEvent,
    ) -> None:
        """
        Persist one operational event.
        """

        raise NotImplementedError

    @abstractmethod
    def get_by_id(
        self,
        event_id,
    ) -> OperationalEvent | None:
        """
        Return an event by its unique identifier.
        """

        raise NotImplementedError

    @abstractmethod
    def get_all(
        self,
    ) -> tuple[
        OperationalEvent,
        ...
    ]:
        """
        Return all stored events in deterministic order.
        """

        raise NotImplementedError

    @abstractmethod
    def clear(self) -> None:
        """
        Clear repository state.

        Intended primarily for isolated tests and demonstrations.
        """

        raise NotImplementedError


class InMemoryOperationalEventRepository(
    OperationalEventRepository
):
    """
    In-memory operational event repository.

    This implementation is intentionally small and deterministic.
    It can later be replaced by a persistent implementation
    without changing the application service contract.
    """

    def __init__(self) -> None:
        self._events: dict[
            object,
            OperationalEvent,
        ] = {}

    def save(
        self,
        event: OperationalEvent,
    ) -> None:
        """
        Store an operational event.
        """

        if event is None:
            raise ValueError(
                "Operational event cannot be None."
            )

        self._events[
            event.event_id
        ] = event

    def get_by_id(
        self,
        event_id,
    ) -> OperationalEvent | None:
        """
        Retrieve an event by ID.
        """

        if event_id is None:
            raise ValueError(
                "Operational event ID cannot be None."
            )

        return self._events.get(
            event_id
        )

    def get_all(
        self,
    ) -> tuple[
        OperationalEvent,
        ...
    ]:
        """
        Return events ordered by occurrence time and event ID.
        """

        return tuple(
            sorted(
                self._events.values(),
                key=lambda event: (
                    event.occurred_at,
                    str(event.event_id),
                ),
            )
        )

    def clear(self) -> None:
        """
        Remove all stored events.
        """

        self._events.clear()

"""
EasyChatBot AI Operations Center operational event service.

Provides the application boundary for recording operational
facts emitted by Runtime components, provider services, and
future agent frameworks.

Architecture:

Agent / Runtime / Provider Service
                |
                v
       OperationalEventService
                |
                v
       OperationalEventRepository
                |
                v
        Operations Center


Responsibilities:

- accept validated operational events;
- persist events through the repository abstraction;
- retrieve operational events;
- provide a stable application boundary for future
  internal and external agent integrations.

Does NOT:

- execute agents;
- execute providers;
- change Runtime routing;
- approve escalations;
- perform autonomous remediation;
- send notifications;
- access persistence directly.

Operational events remain immutable facts.
"""

from __future__ import annotations

from uuid import UUID

from app.application.provider_management.operational_event import (
    OperationalEvent,
)

from app.application.provider_management.operational_event_repository import (
    OperationalEventRepository,
)


class OperationalEventService:
    """
    Application service for operational event management.

    The service deliberately delegates persistence to the
    repository abstraction and contains no infrastructure logic.
    """

    def __init__(
        self,
        *,
        repository: OperationalEventRepository,
    ) -> None:
        """
        Initialise the operational event service.
        """

        if repository is None:
            raise ValueError(
                "Operational event repository "
                "cannot be None."
            )

        self._repository = repository

    def record(
        self,
        event: OperationalEvent,
    ) -> OperationalEvent:
        """
        Record one operational event.

        The event itself is immutable. The service therefore
        returns the same event after successful persistence.
        """

        if event is None:
            raise ValueError(
                "Operational event cannot be None."
            )

        self._repository.save(event)

        return event

    def get_event(
        self,
        event_id: UUID,
    ) -> OperationalEvent | None:
        """
        Retrieve one operational event by ID.
        """

        if event_id is None:
            raise ValueError(
                "Operational event ID cannot be None."
            )

        return self._repository.get_by_id(
            event_id
        )

    def list_events(
        self,
    ) -> tuple[
        OperationalEvent,
        ...
    ]:
        """
        Return all operational events in repository order.
        """

        return self._repository.get_all()

    def clear(self) -> None:
        """
        Clear operational event state.

        Intended for isolated tests and demonstrations.
        """

        self._repository.clear()

"""
EasyChatBot AI Operations Center provider operational event service.

Provides the provider-management adapter between provider-specific
operational lifecycle facts and the generic OperationalEventService.

Architecture:

Provider Incident Service
        |
        v
ProviderOperationalEventService
        |
        v
OperationalEventService
        |
        v
OperationalEventRepository
"""

from __future__ import annotations

from app.application.provider_management.operational_event import (
    OperationalEvent,
    OperationalEventSeverity,
    OperationalEventType,
)
from app.application.provider_management.operational_event_service import (
    OperationalEventService,
)
from app.application.provider_management.provider_health_models import (
    ProviderIncident,
)


class ProviderOperationalEventService:
    """
    Translate provider-management lifecycle facts into the
    generic operational-event contract.
    """

    def __init__(
        self,
        *,
        operational_event_service: OperationalEventService,
    ) -> None:
        if operational_event_service is None:
            raise ValueError(
                "Operational event service cannot be None."
            )

        self._operational_event_service = (
            operational_event_service
        )

    def record_incident_created(
        self,
        *,
        incident: ProviderIncident,
    ) -> OperationalEvent:
        """
        Record a provider incident-created operational fact.
        """

        if incident is None:
            raise ValueError(
                "Provider incident cannot be None."
            )

        event = OperationalEvent.create(
            event_type=(
                OperationalEventType.PROVIDER_INCIDENT_CREATED
            ),
            severity=OperationalEventSeverity.CRITICAL,
            source="provider_management",
            component="ProviderIncidentService",
            occurred_at=incident.started_at,
            message=(
                f"Provider incident created "
                f"for {incident.provider}."
            ),
            correlation_id=incident.incident_id,
            metadata={
                "provider": incident.provider,
                "incident_id": incident.incident_id,
                "status": incident.status.value,
                "affected_capabilities": list(
                    incident.affected_capabilities
                ),
                "consecutive_failures": (
                    incident.consecutive_failures
                ),
            },
        )

        return self._operational_event_service.record(
            event
        )

    def record_provider_recovered(
        self,
        *,
        incident: ProviderIncident,
    ) -> OperationalEvent:
        """
        Record a provider-recovered operational fact.
        """

        if incident is None:
            raise ValueError(
                "Provider incident cannot be None."
            )

        if incident.recovered_at is None:
            raise ValueError(
                "Recovered provider incident must "
                "have a recovery timestamp."
            )

        event = OperationalEvent.create(
            event_type=(
                OperationalEventType.PROVIDER_RECOVERED
            ),
            severity=OperationalEventSeverity.INFO,
            source="provider_management",
            component="ProviderIncidentService",
            occurred_at=incident.recovered_at,
            message=(
                f"Provider {incident.provider} "
                f"recovered."
            ),
            correlation_id=incident.incident_id,
            metadata={
                "provider": incident.provider,
                "incident_id": incident.incident_id,
                "status": incident.status.value,
                "affected_capabilities": list(
                    incident.affected_capabilities
                ),
                "downtime_seconds": (
                    incident.downtime_seconds
                ),
            },
        )

        return self._operational_event_service.record(
            event
        )
