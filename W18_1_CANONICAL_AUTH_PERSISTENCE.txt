from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from uuid import UUID, uuid4
import hashlib
import secrets

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker


@dataclass(frozen=True, slots=True)
class AuthUserRecord:
    id: UUID
    email: str
    username: str
    status: str
    email_verified: bool


@dataclass(frozen=True, slots=True)
class AuthCredentialRecord:
    user_id: UUID
    password_hash: str
    credential_type: str


@dataclass(frozen=True, slots=True)
class PasswordResetContextRecord:
    user_id: UUID
    email: str
    password_hash: str


@dataclass(frozen=True, slots=True)
class AuthSessionRecord:
    session_id: UUID
    user_id: UUID
    status: str
    created_at: datetime
    expires_at: datetime
    last_activity_at: datetime
    device_name: str | None
    ip_address: str | None
    is_revoked: bool
    metadata_json: dict


class PostgreSQLAuthenticationRepository:
    """PostgreSQL adapter for the existing EasyChatBot auth tables.

    The adapter intentionally uses SQLAlchemy Core statements rather than
    importing speculative ORM models. This binds the application contract to
    the inspected database schema while keeping persistence isolated.
    """

    def __init__(self, session_factory: async_sessionmaker[AsyncSession]) -> None:
        if session_factory is None:
            raise ValueError("session_factory is required.")
        self._session_factory = session_factory

    async def email_exists(self, email: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text("SELECT 1 FROM users WHERE email = :email LIMIT 1"),
                {"email": email},
            )
            return result.first() is not None

    async def username_exists(self, username: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                text("SELECT 1 FROM users WHERE username = :username LIMIT 1"),
                {"username": username},
            )
            return result.first() is not None

    async def create_user(self, *, user_id: UUID, email: str, username: str) -> None:
        now = datetime.now(timezone.utc)
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO users
                            (id, email, username, status, trial_status, created_at,
                             email_verified)
                        VALUES
                            (:id, :email, :username, 'active', 'not_started', :created_at,
                             FALSE)
                        """
                    ),
                    {
                        "id": user_id,
                        "email": email,
                        "username": username,
                        "created_at": now,
                    },
                )
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_user_roles (user_id, role)
                        VALUES (:user_id, 'beta_user')
                        ON CONFLICT DO NOTHING
                        """
                    ),
                    {"user_id": user_id},
                )

    async def create_password_credential(self, *, user_id: UUID, password_hash: str) -> None:
        now = datetime.now(timezone.utc)
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO user_credentials
                            (id, user_id, password_hash, credential_type,
                             created_at, updated_at)
                        VALUES
                            (:id, :user_id, :password_hash, 'PASSWORD',
                             :created_at, :updated_at)
                        """
                    ),
                    {
                        "id": uuid4(),
                        "user_id": user_id,
                        "password_hash": password_hash,
                        "created_at": now,
                        "updated_at": now,
                    },
                )

    async def create_email_verification(self, *, user_id: UUID, email: str) -> str:
        """Create a one-time verification token.

        The raw token is returned to the application boundary. Only its SHA-256
        digest is persisted. Delivery is deliberately outside this repository.
        """
        raw_token = secrets.token_urlsafe(32)
        token_hash = hashlib.sha256(raw_token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)
        verification_id = uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO auth_email_verification_tokens
                            (id, user_id, email, token_hash, expires_at, created_at)
                        VALUES
                            (:id, :user_id, :email, :token_hash,
                             :expires_at, :created_at)
                        """
                    ),
                    {
                        "id": verification_id,
                        "user_id": user_id,
                        "email": email,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=24),
                        "created_at": now,
                    },
                )
        return raw_token


    async def create_password_reset_token(
        self,
        *,
        user_id: UUID,
    ) -> str:
        """Create a secure one-time password-reset token.

        Only the SHA-256 digest is persisted. The raw token is returned
        exclusively to the application layer for email delivery.
        """
        raw_token = secrets.token_urlsafe(32)

        token_hash = hashlib.sha256(
            raw_token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)
        reset_id = uuid4()

        async with self._session_factory() as session:
            async with session.begin():
                # A newly issued reset link supersedes older unused links.
                await session.execute(
                    text(
                        """
                        UPDATE auth_password_reset_tokens
                        SET used_at = :now
                        WHERE user_id = :user_id
                          AND used_at IS NULL
                        """
                    ),
                    {
                        "now": now,
                        "user_id": user_id,
                    },
                )

                await session.execute(
                    text(
                        """
                        INSERT INTO auth_password_reset_tokens
                            (
                                id,
                                user_id,
                                token_hash,
                                expires_at,
                                created_at
                            )
                        VALUES
                            (
                                :id,
                                :user_id,
                                :token_hash,
                                :expires_at,
                                :created_at
                            )
                        """
                    ),
                    {
                        "id": reset_id,
                        "user_id": user_id,
                        "token_hash": token_hash,
                        "expires_at": now + timedelta(hours=1),
                        "created_at": now,
                    },
                )

        return raw_token

    async def count_recent_password_reset_tokens(
        self,
        *,
        user_id: UUID,
        since: datetime,
    ) -> int:
        """Count password-reset requests in the supplied time window."""
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT COUNT(*)
                    FROM auth_password_reset_tokens
                    WHERE user_id = :user_id
                      AND created_at >= :since
                    """
                ),
                {
                    "user_id": user_id,
                    "since": since,
                },
            )

            return int(result.scalar_one())

    async def get_password_reset_context(
        self,
        *,
        token: str,
    ) -> PasswordResetContextRecord | None:
        """Resolve a valid reset token to non-sensitive account context.

        The raw token is never persisted or returned. Only a valid, unused,
        unexpired token can resolve its account email and current credential
        hash. The credential hash is consumed only by the application service
        for the previous-password check and is never exposed by the HTTP API.
        """
        token_hash = hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()
        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT t.user_id, u.email, c.password_hash
                    FROM auth_password_reset_tokens AS t
                    INNER JOIN users AS u
                        ON u.id = t.user_id
                    INNER JOIN user_credentials AS c
                        ON c.user_id = t.user_id
                       AND c.credential_type = 'PASSWORD'
                    WHERE t.token_hash = :token_hash
                      AND t.used_at IS NULL
                      AND t.expires_at > :now
                    LIMIT 1
                    """
                ),
                {
                    "token_hash": token_hash,
                    "now": now,
                },
            )
            row = result.mappings().first()

            if row is None:
                return None

            return PasswordResetContextRecord(
                user_id=row["user_id"],
                email=row["email"],
                password_hash=row["password_hash"],
            )

    async def get_password_reset_user(
        self,
        token: str,
    ) -> PasswordResetContextRecord | None:
        """Backward-compatible repository name for password-reset context.

        The reset flow historically exposed this operation as
        ``get_password_reset_user``. Keep that contract as a thin delegate to
        the canonical context method so callers/tests do not need a second
        implementation or a duplicate SQL query.
        """
        return await self.get_password_reset_context(token=token)

    async def reset_password(
        self,
        *,
        token: str,
        password_hash: str,
        expected_password_hash: str | None = None,
    ) -> bool:
        """Atomically consume a reset token, update the password,
        and revoke existing sessions.
        """
        token_hash = hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    text(
                        """
                        SELECT id, user_id
                        FROM auth_password_reset_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now
                        FOR UPDATE
                        """
                    ),
                    {
                        "token_hash": token_hash,
                        "now": now,
                    },
                )

                row = result.mappings().first()

                if row is None:
                    return False

                user_id = row["user_id"]

                credential_result = await session.execute(
                    text(
                        """
                        SELECT id
                        FROM user_credentials
                        WHERE user_id = :user_id
                          AND credential_type = 'PASSWORD'
                        LIMIT 1
                        """
                    ),
                    {
                        "user_id": user_id,
                    },
                )

                credential = credential_result.mappings().first()

                if credential is None:
                    return False

                # Keep the optimistic-concurrency predicate out of a nullable
                # PostgreSQL bind expression.  asyncpg/PostgreSQL can otherwise
                # fail to infer the parameter type for ``:expected_password_hash
                # IS NULL`` on some database/driver combinations.  Branching
                # here also makes the two repository contracts explicit: a
                # normal reset and a reset guarded by the previously observed
                # password hash.
                if expected_password_hash is None:
                    update_result = await session.execute(
                        text(
                            """
                            UPDATE user_credentials
                            SET password_hash = :password_hash,
                                updated_at = :updated_at
                            WHERE id = :credential_id
                            """
                        ),
                        {
                            "password_hash": password_hash,
                            "updated_at": now,
                            "credential_id": credential["id"],
                        },
                    )
                else:
                    update_result = await session.execute(
                        text(
                            """
                            UPDATE user_credentials
                            SET password_hash = :password_hash,
                                updated_at = :updated_at
                            WHERE id = :credential_id
                              AND password_hash = :expected_password_hash
                            """
                        ),
                        {
                            "password_hash": password_hash,
                            "updated_at": now,
                            "credential_id": credential["id"],
                            "expected_password_hash": expected_password_hash,
                        },
                    )

                if update_result.rowcount != 1:
                    return False

                await session.execute(
                    text(
                        """
                        UPDATE auth_password_reset_tokens
                        SET used_at = :used_at
                        WHERE id = :id
                        """
                    ),
                    {
                        "used_at": now,
                        "id": row["id"],
                    },
                )

                await session.execute(
                    text(
                        """
                        UPDATE user_sessions
                        SET is_revoked = TRUE,
                            status = 'revoked'
                        WHERE user_id = :user_id
                          AND is_revoked = FALSE
                        """
                    ),
                    {
                        "user_id": user_id,
                    },
                )

                return True
    async def verify_email(self, token: str) -> bool:
        token_hash = hashlib.sha256(token.encode("utf-8")).hexdigest()
        now = datetime.now(timezone.utc)

        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    text(
                        """
                        SELECT id, user_id
                        FROM auth_email_verification_tokens
                        WHERE token_hash = :token_hash
                          AND used_at IS NULL
                          AND expires_at > :now
                        FOR UPDATE
                        """
                    ),
                    {"token_hash": token_hash, "now": now},
                )
                row = result.mappings().first()
                if row is None:
                    return False

                await session.execute(
                    text(
                        "UPDATE users SET email_verified = TRUE WHERE id = :user_id"
                    ),
                    {"user_id": row["user_id"]},
                )
                await session.execute(
                    text(
                        "UPDATE auth_email_verification_tokens SET used_at = :now WHERE id = :id"
                    ),
                    {"now": now, "id": row["id"]},
                )
                return True

    async def get_user_by_email(self, email: str) -> AuthUserRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT id, email, username, status, email_verified
                    FROM users
                    WHERE email = :email
                    LIMIT 1
                    """
                ),
                {"email": email},
            )
            row = result.mappings().first()
            return self._user(row) if row else None

    async def get_password_credential(self, user_id: UUID) -> AuthCredentialRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT user_id, password_hash, credential_type
                    FROM user_credentials
                    WHERE user_id = :user_id AND credential_type = 'PASSWORD'
                    LIMIT 1
                    """
                ),
                {"user_id": user_id},
            )
            row = result.mappings().first()
            if row is None:
                return None
            return AuthCredentialRecord(
                user_id=row["user_id"],
                password_hash=row["password_hash"],
                credential_type=row["credential_type"],
            )

    async def create_session(
        self,
        *,
        session_id: UUID,
        user_id: UUID,
        created_at: datetime,
        expires_at: datetime,
        device_name: str | None,
        ip_address: str | None,
    ) -> None:
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        INSERT INTO user_sessions
                            (session_id, user_id, status, created_at, expires_at,
                             last_activity_at, device_name, ip_address,
                             is_revoked, metadata_json)
                        VALUES
                            (:session_id, :user_id, 'active', :created_at, :expires_at,
                             :last_activity_at, :device_name, :ip_address,
                             FALSE, CAST(:metadata_json AS JSON))
                        """
                    ),
                    {
                        "session_id": session_id,
                        "user_id": user_id,
                        "created_at": created_at,
                        "expires_at": expires_at,
                        "last_activity_at": created_at,
                        "device_name": device_name,
                        "ip_address": ip_address,
                        "metadata_json": "{}",
                    },
                )

    async def get_session(self, session_id: str | UUID) -> AuthSessionRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT session_id, user_id, status, created_at, expires_at,
                           last_activity_at, device_name, ip_address,
                           is_revoked, metadata_json
                    FROM user_sessions
                    WHERE session_id = :session_id
                    LIMIT 1
                    """
                ),
                {"session_id": session_id},
            )
            row = result.mappings().first()
            if row is None:
                return None
            return AuthSessionRecord(
                session_id=row["session_id"],
                user_id=row["user_id"],
                status=row["status"],
                created_at=row["created_at"],
                expires_at=row["expires_at"],
                last_activity_at=row["last_activity_at"],
                device_name=row["device_name"],
                ip_address=row["ip_address"],
                is_revoked=bool(row["is_revoked"]),
                metadata_json=row["metadata_json"] or {},
            )

    async def revoke_session(self, session_id: UUID) -> None:
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    text(
                        """
                        UPDATE user_sessions
                        SET is_revoked = TRUE, status = 'revoked'
                        WHERE session_id = :session_id
                        """
                    ),
                    {"session_id": session_id},
                )

    async def get_user_for_identity(self, user_id: UUID) -> AuthUserRecord | None:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    """
                    SELECT id, email, username, status, email_verified
                    FROM users
                    WHERE id = :user_id
                    LIMIT 1
                    """
                ),
                {"user_id": user_id},
            )
            row = result.mappings().first()
            return self._user(row) if row else None

    async def get_user_roles(self, user_id: UUID) -> tuple[str, ...]:
        async with self._session_factory() as session:
            result = await session.execute(
                text(
                    "SELECT role FROM auth_user_roles WHERE user_id = :user_id ORDER BY role"
                ),
                {"user_id": user_id},
            )
            return tuple(row[0] for row in result.all())

    @staticmethod
    def _user(row) -> AuthUserRecord:
        return AuthUserRecord(
            id=row["id"],
            email=row["email"],
            username=row["username"],
            status=row["status"],
            email_verified=bool(row["email_verified"]),
        )

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from uuid import UUID, uuid4
import hashlib
import hmac
import secrets
import logging


logger = logging.getLogger(__name__)


class AuthenticationError(ValueError):
    """Expected authentication failure."""


@dataclass(frozen=True, slots=True)
class SessionResult:
    session_id: UUID
    user_id: UUID
    expires_at: datetime


@dataclass(frozen=True, slots=True)
class PasswordResetContext:
    """Safe reset-link context exposed to the frontend."""

    email: str


class PasswordHasher:
    """PBKDF2 password hashing using the Python standard library."""

    algorithm = "pbkdf2_sha256"
    iterations = 600_000
    salt_bytes = 32

    def hash(self, password: str) -> str:
        if len(password) < 12:
            raise AuthenticationError(
                "Password must contain at least 12 characters."
            )

        salt = secrets.token_bytes(self.salt_bytes)
        digest = hashlib.pbkdf2_hmac(
            "sha256",
            password.encode("utf-8"),
            salt,
            self.iterations,
        )
        return (
            f"{self.algorithm}${self.iterations}$"
            f"{salt.hex()}${digest.hex()}"
        )

    def verify(self, password: str, encoded: str) -> bool:
        try:
            algorithm, iterations, salt_hex, digest_hex = encoded.split("$")

            if algorithm != self.algorithm:
                return False

            candidate = hashlib.pbkdf2_hmac(
                "sha256",
                password.encode("utf-8"),
                bytes.fromhex(salt_hex),
                int(iterations),
            )

            return hmac.compare_digest(
                candidate.hex(),
                digest_hex,
            )
        except (ValueError, TypeError):
            return False


class AuthenticationService:
    """Application service for registration/login/session lifecycle.

    Persistence is supplied through a repository protocol so the service does
    not depend on a particular ORM implementation.

    Authentication email delivery is supplied through the application email
    service boundary. Provider selection and provider credentials remain
    outside this service.
    """

    def __init__(
        self,
        repository,
        *,
        email_service=None,
        password_hasher: PasswordHasher | None = None,
        session_ttl: timedelta = timedelta(hours=12),
    ) -> None:
        self._repository = repository
        self._email_service = email_service
        self._password_hasher = password_hasher or PasswordHasher()
        self._session_ttl = session_ttl

    async def register(
        self,
        *,
        email: str,
        username: str,
        password: str,
    ):
        normalized_email = email.strip().lower()
        normalized_username = username.strip()

        if not normalized_email or "@" not in normalized_email:
            raise AuthenticationError("A valid email is required.")

        if not normalized_username:
            raise AuthenticationError("Username is required.")

        if await self._repository.email_exists(normalized_email):
            raise AuthenticationError(
                "An account with this email already exists."
            )

        if await self._repository.username_exists(normalized_username):
            raise AuthenticationError(
                "That username is already in use."
            )

        user_id = uuid4()
        password_hash = self._password_hasher.hash(password)

        await self._repository.create_user(
            user_id=user_id,
            email=normalized_email,
            username=normalized_username,
        )

        await self._repository.create_password_credential(
            user_id=user_id,
            password_hash=password_hash,
        )

        verification_token = (
            await self._repository.create_email_verification(
                user_id=user_id,
                email=normalized_email,
            )
        )

        if self._email_service is None:
            raise AuthenticationError(
                "Registration email delivery is not configured."
            )

        try:
            await self._email_service.send_verification_email(
                email=normalized_email,
                username=normalized_username,
                verification_token=verification_token,
            )
        except Exception:
            raise AuthenticationError(
                "Registration email could not be sent. "
                "Please try again later."
            ) from None

        return user_id

    async def authenticate(
        self,
        *,
        email: str,
        password: str,
        device_name: str | None = None,
        ip_address: str | None = None,
    ) -> SessionResult:
        user = await self._repository.get_user_by_email(
            email.strip().lower()
        )

        if user is None:
            raise AuthenticationError(
                "Invalid email or password."
            )

        if user.status != "active":
            raise AuthenticationError(
                "Account is not active."
            )

        if not user.email_verified:
            raise AuthenticationError(
                "Email verification is required."
            )

        credential = await self._repository.get_password_credential(
            user.id
        )

        if credential is None or not self._password_hasher.verify(
            password,
            credential.password_hash,
        ):
            raise AuthenticationError(
                "Invalid email or password."
            )

        now = datetime.now(timezone.utc)
        expires_at = now + self._session_ttl
        session_id = uuid4()

        await self._repository.create_session(
            session_id=session_id,
            user_id=user.id,
            created_at=now,
            expires_at=expires_at,
            device_name=device_name,
            ip_address=ip_address,
        )

        return SessionResult(
            session_id=session_id,
            user_id=user.id,
            expires_at=expires_at,
        )

    async def request_password_reset(
        self,
        *,
        email: str,
    ) -> None:
        """Request a password-reset email without revealing account existence.

        Unknown accounts intentionally produce the same successful outcome as
        known accounts. Reset-token generation and persistence remain inside
        the repository, while email delivery remains inside the email service.
        """
        normalized_email = email.strip().lower()

        if not normalized_email or "@" not in normalized_email:
            raise AuthenticationError("A valid email is required.")

        user = await self._repository.get_user_by_email(
            normalized_email,
        )

        # Do not reveal whether the account exists.
        if user is None:
            return

        now = datetime.now(timezone.utc)
        rate_limit_since = now - timedelta(hours=1)

        recent_requests = (
            await self._repository.count_recent_password_reset_tokens(
                user_id=user.id,
                since=rate_limit_since,
            )
        )

        # Maximum of three reset requests per account per hour.
        if recent_requests >= 3:
            return

        reset_token = await self._repository.create_password_reset_token(
            user_id=user.id,
        )

        if self._email_service is None:
            raise AuthenticationError(
                "Password reset email delivery is not configured."
            )

        await self._email_service.send_password_reset_email(
            email=user.email,
            username=user.username,
            reset_token=reset_token,
        )

    async def get_password_reset_context(
        self,
        *,
        token: str,
    ) -> PasswordResetContext:
        """Return safe account context for a valid reset token.

        The raw token is used only for transient lookup. The repository
        validates its digest, expiry and single-use state. No credential
        material is exposed to the caller.
        """
        normalized_token = token.strip()

        if not normalized_token:
            raise AuthenticationError(
                "Password reset token is required."
            )

        user = await self._repository.get_password_reset_user(
            normalized_token,
        )

        if user is None:
            raise AuthenticationError(
                "Invalid or expired password reset token."
            )

        return PasswordResetContext(
            email=user.email,
        )

    @staticmethod
    def _validate_password_reset_password(
        password: str,
    ) -> None:
        """Enforce the reset-specific password policy."""
        if len(password) < 12:
            raise AuthenticationError(
                "Password must contain at least 12 characters."
            )

        if not any(character.isupper() for character in password):
            raise AuthenticationError(
                "Password must contain at least one uppercase letter."
            )

        if not any(character.islower() for character in password):
            raise AuthenticationError(
                "Password must contain at least one lowercase letter."
            )

        if not any(character.isdigit() for character in password):
            raise AuthenticationError(
                "Password must contain at least one number."
            )

        if not any(
            not character.isalnum()
            for character in password
        ):
            raise AuthenticationError(
                "Password must contain at least one special character."
            )

    async def confirm_password_reset(
        self,
        *,
        token: str,
        new_password: str,
    ) -> None:
        """Set a new password using a valid one-time reset token."""
        normalized_token = token.strip()

        if not normalized_token:
            raise AuthenticationError(
                "Password reset token is required."
            )

        user = await self._repository.get_password_reset_user(
            normalized_token,
        )

        if user is None:
            raise AuthenticationError(
                "Invalid or expired password reset token."
            )

        credential = await self._repository.get_password_credential(
            user.id,
        )

        if credential is None:
            raise AuthenticationError(
                "Password reset is not available for this account."
            )

        self._validate_password_reset_password(new_password)

        if self._password_hasher.verify(
            new_password,
            credential.password_hash,
        ):
            raise AuthenticationError(
                "New password must be different from your previous password."
            )

        password_hash = self._password_hasher.hash(new_password)

        reset = await self._repository.reset_password(
            token=normalized_token,
            password_hash=password_hash,
            expected_password_hash=credential.password_hash,
        )

        if not reset:
            raise AuthenticationError(
                "Invalid or expired password reset token."
            )

        # The password transaction has succeeded. Confirmation email delivery
        # is deliberately best-effort: a provider outage must not turn a
        # successful password change into a misleading API failure. The
        # timestamp below is the application-level confirmation time and is
        # rendered by the email service in UTC.
        reset_at = datetime.now(timezone.utc)
        if self._email_service is not None:
            try:
                await self._email_service.send_password_reset_confirmation_email(
                    email=user.email,
                    username=user.username,
                    reset_at=reset_at,
                )
            except Exception:
                logger.exception(
                    "Password reset confirmation email delivery failed."
                )

    async def verify_email(
        self,
        token: str,
    ) -> None:
        """Verify an email address using a repository-managed token.

        Token generation, hashing, expiration, single-use enforcement,
        and persistence remain the responsibility of the repository.
        The application service exposes the business operation without
        depending on the persistence implementation.
        """

        normalized_token = token.strip()

        if not normalized_token:
            raise AuthenticationError(
                "Email verification token is required."
            )

        verified = await self._repository.verify_email(
            normalized_token,
        )

        if not verified:
            raise AuthenticationError(
                "Invalid or expired email verification token."
            )

    async def revoke_session(
        self,
        session_id: UUID,
    ) -> None:
        await self._repository.revoke_session(session_id)

from __future__ import annotations

from datetime import datetime, timezone

from app.application.identity.identity_provider import IdentityProvider
from app.domain.identity.principal import AccountStatus, Principal


class SessionIdentityProvider(IdentityProvider):
    """Resolve the authenticated Principal from a server-side session."""

    def __init__(
        self,
        *,
        session_id: str | None,
        repository,
    ) -> None:
        self._session_id = session_id
        self._repository = repository

    async def get_current_principal(self) -> Principal | None:
        if not self._session_id:
            return None

        session = await self._repository.get_session(self._session_id)

        if session is None:
            return None

        now = datetime.now(timezone.utc)

        if session.is_revoked or session.expires_at <= now:
            return None

        user = await self._repository.get_user_for_identity(
            session.user_id
        )

        if user is None:
            return None

        roles = await self._repository.get_user_roles(user.id)

        try:
            status = AccountStatus(user.status)
        except ValueError:
            return None

        return Principal(
            user_id=str(user.id),
            email=user.email,
            email_verified=bool(user.email_verified),
            account_status=status,
            roles=frozenset(roles),
        )

from __future__ import annotations

from app.domain.authorization.permissions import Permission, has_permission
from app.domain.identity.principal import Principal


class AuthenticationRequiredError(PermissionError):
    """No valid authenticated principal exists."""


class PermissionDeniedError(PermissionError):
    """Authenticated principal lacks a permission."""


class AuthorizationService:
    """Provider-neutral authorization boundary."""

    def require_authenticated(
        self,
        principal: Principal | None,
    ) -> Principal:
        if principal is None or not principal.is_active:
            raise AuthenticationRequiredError(
                "Authenticated application identity is required."
            )
        return principal

    def require_permission(
        self,
        *,
        principal: Principal | None,
        permission: Permission,
    ) -> Principal:
        authenticated = self.require_authenticated(principal)

        if not has_permission(
            roles=authenticated.roles,
            permission=permission,
        ):
            raise PermissionDeniedError(
                f"Permission required: {permission.value}."
            )

        return authenticated

from __future__ import annotations

from fastapi import HTTPException, Request, status

from app.domain.identity.principal import Principal
from app.infrastructure.auth.session_identity_provider import (
    SessionIdentityProvider,
)


async def get_current_principal(request: Request) -> Principal:
    """Resolve identity from the HttpOnly application session cookie."""

    session_id = request.cookies.get("easychatbot_session")

    repository = request.app.state.auth_repository

    provider = SessionIdentityProvider(
        session_id=session_id,
        repository=repository,
    )

    principal = await provider.get_current_principal()

    if principal is None or not principal.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authentication required.",
        )

    return principal

from __future__ import annotations

# NOTE:

@router.get(
    "/session",
    response_model=SessionResponse,
    summary="Check the current authentication session",
    description=(
        "Return whether the current request has a valid authenticated "
        "application session. The session identifier and other sensitive "
        "authentication data are never returned."
    ),
)
async def get_session(
    principal: Principal = Depends(get_current_principal),
) -> SessionResponse:
    """
    Return the current authentication state.

    Authentication is resolved exclusively through the existing
    get_current_principal() dependency and SessionIdentityProvider.

    This endpoint intentionally exposes only authentication state.
    It does not expose the session identifier, credentials, or other
    sensitive authentication material.
    """
    return SessionResponse(
        authenticated=principal.is_active,
    )


@router.post(
    "/register",
    response_model=RegisterResponse,
    status_code=status.HTTP_201_CREATED,
)
async def register(
    request: RegisterRequest,
    http_request: Request,
) -> RegisterResponse:
    """
    Register a new account through the composed authentication service.

    Registration creates the account and verification token and triggers
    verification-email delivery through the configured application/provider
    boundary. The route itself does not select or call an email provider.
    """
    service = http_request.app.state.authentication_service

    try:
        user_id = await service.register(
            email=str(request.email),
            username=request.username,
            password=request.password,
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(exc),
        ) from None

    return RegisterResponse(
        user_id=user_id,
        detail=(
            "Registration successful. "
            "Email verification is required."
        ),
    )


@router.post(
    "/login",
    response_model=LoginResponse,
)
async def login(
    request: LoginRequest,
    http_request: Request,
    response: Response,
) -> LoginResponse:
    """
    Authenticate through the composed authentication service.

    The service enforces account status and email verification before a
    server-side session is created.
    """
    service = http_request.app.state.authentication_service

    try:
        result = await service.authenticate(
            email=str(request.email),
            password=request.password,
            device_name=http_request.headers.get("user-agent"),
            ip_address=(
                http_request.client.host
                if http_request.client
                else None
            ),
        )
    except AuthenticationError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=str(exc),
            headers={
                "WWW-Authenticate": "Session",
            },
        ) from None

    expires_in_seconds = max(
        0,
        int(
            (
                result.expires_at
                - datetime.now(timezone.utc)
            ).total_seconds()
        ),
    )

    response.set_cookie(
        key="easychatbot_session",
        value=str(result.session_id),
        httponly=True,
        secure=True,
        samesite="lax",
        max_age=expires_in_seconds,
        path="/",
    )

    return LoginResponse(
        detail="Login successful.",
        expires_at=result.expires_at.isoformat(),
    )

@router.post(
    "/logout",
    status_code=status.HTTP_204_NO_CONTENT,
)
async def logout(
    http_request: Request,
    response: Response,
) -> None:
    """
    Revoke the current server-side authentication session.

    No authentication service is constructed here; the application's
    composed authentication service is reused.
    """
    session_id = http_request.cookies.get(
        "easychatbot_session"
    )

    if session_id:
        service = http_request.app.state.authentication_service

        try:
            parsed_session_id = UUID(session_id)
        except ValueError:
            parsed_session_id = None

        if parsed_session_id is not None:
            await service.revoke_session(
                parsed_session_id
            )

    response.delete_cookie(
        key="easychatbot_session",
        path="/",
    )
