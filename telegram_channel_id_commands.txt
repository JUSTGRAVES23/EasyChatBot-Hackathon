cd "C:\AI Folder\EasyChatBot-Hackathon"

# 1. Verify TELEGRAM_BOT_TOKEN exists without displaying it
Get-Content .env |
    Where-Object {
        $_ -match '^\s*TELEGRAM_BOT_TOKEN\s*='
    } |
    ForEach-Object {
        $value = $_ -replace '^\s*TELEGRAM_BOT_TOKEN\s*=\s*', ''
        if ($value) {
            "TELEGRAM_BOT_TOKEN: PRESENT"
        } else {
            "TELEGRAM_BOT_TOKEN: EMPTY"
        }
    }

# 2. Find how EasyChatBot loads configuration
Get-ChildItem -Recurse -File -Include *.py |
    Select-String -Pattern 'TELEGRAM_BOT_TOKEN|load_dotenv|BaseSettings|Settings|dotenv' |
    Select-Object Path, LineNumber, Line

# 3. Find running Telegram/Python processes
Get-CimInstance Win32_Process |
    Where-Object {
        $_.CommandLine -match "telegram|run_telegram_runtime|python"
    } |
    Select-Object ProcessId, Name, CommandLine

# 4. Check specifically for Telegram runtime
Get-CimInstance Win32_Process |
    Where-Object {
        $_.CommandLine -match "run_telegram_runtime"
    } |
    Select-Object ProcessId, Name, CommandLine

# 5. Stop ONLY the identified Telegram runtime
# Replace 12345 with the actual ProcessId.
# Stop-Process -Id 12345

# 6. Verify Telegram runtime has stopped
Get-CimInstance Win32_Process |
    Where-Object {
        $_.CommandLine -match "run_telegram_runtime"
    } |
    Select-Object ProcessId, Name, CommandLine

# 7. Check whether the current PowerShell process has the token
if ($env:TELEGRAM_BOT_TOKEN) {
    "TELEGRAM_BOT_TOKEN environment variable: PRESENT"
} else {
    "TELEGRAM_BOT_TOKEN environment variable: MISSING"
}

# 8. Verify .env variable name without displaying its value
Get-Content .env |
    Where-Object {
        $_ -match '^\s*TELEGRAM_BOT_TOKEN\s*='
    } |
    ForEach-Object {
        "TELEGRAM_BOT_TOKEN variable in .env: PRESENT"
    }

# 9. Find all Telegram configuration references
Get-ChildItem -Recurse -File -Include *.py |
    Select-String -Pattern 'TELEGRAM_BOT_TOKEN|BETA1_ANNOUNCEMENT_CHANNEL_ID|load_dotenv|BaseSettings|Settings|dotenv' |
    Select-Object Path, LineNumber, Line
