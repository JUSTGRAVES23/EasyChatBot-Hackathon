import {
  privacyGovernanceEndpoints,
} from "./config";
import type {
  PrivacyGovernanceCase,
  PrivacyGovernanceStatusItem,
  PrivacyGovernanceSummary,
} from "./types";

async function requestJson<T>(url: string): Promise<T> {
  const response = await fetch(url, {
    headers: {
      Accept: "application/json",
    },
  });

  if (!response.ok) {
    throw new Error(
      `Privacy governance API request failed (${response.status}).`,
    );
  }

  return response.json() as Promise<T>;
}

export function getPrivacyGovernanceSummary(): Promise<PrivacyGovernanceSummary> {
  return requestJson<PrivacyGovernanceSummary>(
    privacyGovernanceEndpoints.summary,
  );
}

export function getPrivacyGovernanceStatusChart(): Promise<
  PrivacyGovernanceStatusItem[]
> {
  return requestJson<PrivacyGovernanceStatusItem[]>(
    privacyGovernanceEndpoints.statusChart,
  );
}

export function getPrivacyGovernanceCases(): Promise<PrivacyGovernanceCase[]> {
  return requestJson<PrivacyGovernanceCase[]>(
    privacyGovernanceEndpoints.cases,
  );
}

export function getPrivacyGovernanceCase(
  caseId: string,
): Promise<PrivacyGovernanceCase> {
  return requestJson<PrivacyGovernanceCase>(
    privacyGovernanceEndpoints.caseDetail(caseId),
  );
}

export async function loadPrivacyGovernanceDashboard() {
  const [summary, statusChart, cases] = await Promise.all([
    getPrivacyGovernanceSummary(),
    getPrivacyGovernanceStatusChart(),
    getPrivacyGovernanceCases(),
  ]);

  return { summary, statusChart, cases };
}

import type {
  PrivacyGovernanceCaseDetail,
} from "./types";

export type GovernanceDecision = "approve" | "reject" | "request-changes";
export type GovernanceRole = "DPO" | "GRC";

export interface GovernanceDecisionRequest {
  change_id: string;
  approval_id: string;
  reviewer: string;
  role: GovernanceRole;
  comment: string;
}

export interface GovernanceDecisionResponse {
  case: PrivacyGovernanceCaseDetail;
  approval_record: Record<string, unknown>;
}

const API_BASE = "/api/privacy/governance";

export async function submitGovernanceDecision(
  caseId: string,
  decision: GovernanceDecision,
  request: GovernanceDecisionRequest,
  signal?: AbortSignal,
): Promise<GovernanceDecisionResponse> {
  const response = await fetch(
    `${API_BASE}/cases/${encodeURIComponent(caseId)}/${decision}`,
    {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify(request),
      signal,
    },
  );

  if (!response.ok) {
    let message = `Governance decision failed (${response.status}).`;

    try {
      const payload = (await response.json()) as { detail?: string };
      if (payload.detail) {
        message = payload.detail;
      }
    } catch {
      // Keep the stable HTTP error message when the body is not JSON.
    }

    throw new Error(message);
  }

  return (await response.json()) as GovernanceDecisionResponse;
}
