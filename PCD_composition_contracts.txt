class ProductionHealthService:

    Application service exposing production health status.

    Responsibilities:

    - Compose approved production health checks.
    - Execute readiness validation.
    - Return sanitized structured results.

    Does NOT:

    - Render dashboards.
    - Store monitoring data.
    - Send notifications.
    - Access Telegram conversations.
    """

    def __init__(
        self,
        *,
        configuration: ProductionHealthConfiguration,
    ) -> None:

        if configuration is None:
            raise ValueError(
                "configuration cannot be None."
            )

        self._readiness_service = (
            ProductionReadinessService(
                checks=(
                    configuration.build_checks()
                )
            )
        )

    async def check_production_health(
        self,
    ) -> ProductionReadinessResult:
        """
        Execute production dependency checks.
        """

        return await (
            self._readiness_service
            .check()
        )

from __future__ import annotations

from datetime import datetime, timezone

from .health_result import HealthStatus

class OperationalDashboardService:
    """
    Application service producing the read-only operational dashboard.

    Dependency health is probed fresh for every snapshot.

    Incident/event state comes from the application-owned
    OperationalEventService.

    The event service is exposed through a read-only property so dependent
    Operations Center application APIs can explicitly reuse the same
    authoritative operational state.
    """

    def __init__(
        self,
        *,
        health_snapshot_service: OperationalHealthSnapshotService,
        event_service: OperationalEventService,
    ) -> None:
        if health_snapshot_service is None:
            raise ValueError(
                "health_snapshot_service cannot be None."
            )

        if event_service is None:
            raise ValueError(
                "event_service cannot be None."
            )

        self._health_snapshot_service = (
            health_snapshot_service
        )

        self._event_service = event_service

    @property
    def event_service(self) -> OperationalEventService:
        """
        Return the authoritative application-owned event service.

        This does not create a new service or event store.

        It allows dependent read-only Operations Center APIs to use the
        exact same operational state as this dashboard service.
        """
        return self._event_service

    async def create_dashboard_snapshot(
        self,
    ) -> OperationalDashboardSnapshot:
        """
        Create a fresh operational dashboard snapshot.

        Dependency health and current incident state are evaluated for
        every request.
        """

        health = (
            await self._health_snapshot_service.create_snapshot()
        )

        incidents = self._event_service.list_incidents()

        operational_events = tuple(
            event
            for incident in incidents
            for event in incident.events
        )

        dashboard_incidents = tuple(
            DashboardIncident.from_incident(
                incident
            )
            for incident in incidents
        )

        private_incidents = tuple(
            incident
            for incident in incidents
            if incident.surface
            is OperationalSurface.PRIVATE_CLASSROOM
        )

        announcement_incidents = tuple(
            incident
            for incident in incidents
            if incident.surface
            is OperationalSurface.ANNOUNCEMENT_GROUP
        )

        private_events = tuple(
            event
            for event in operational_events
            if event.surface
            is OperationalSurface.PRIVATE_CLASSROOM
        )

        announcement_events = tuple(
            event
            for event in operational_events
            if event.surface
            is OperationalSurface.ANNOUNCEMENT_GROUP
        )

        return OperationalDashboardSnapshot(
            generated_at=datetime.now(timezone.utc),
            overall_status=health.overall_status,
            health_score=self._calculate_health_score(
                health.overall_status,
                incidents,
            ),
            private_classroom=self._surface_status(
                health=health,
                surface=OperationalSurface.PRIVATE_CLASSROOM,
                incidents=private_incidents,
                events=private_events,
            ),
            announcement_group=self._surface_status(
                health=health,
                surface=OperationalSurface.ANNOUNCEMENT_GROUP,
                incidents=announcement_incidents,
                events=announcement_events,
            ),
            incidents=dashboard_incidents,
        )

    def list_incidents(
        self,
        *,
        surface: OperationalSurface | None = None,
    ):
        """
        Read current incidents through the existing operational-event
        abstraction.

        No second incident store is created.
        """

        return self._event_service.list_incidents(
            surface=surface
        )

    def _surface_status(
        self,
        *,
        health,
        surface: OperationalSurface,
        incidents,
        events,
    ) -> DashboardSurfaceStatus:
        return DashboardSurfaceStatus(
            surface=surface,
            status=self._surface_health_status(
                health=health,
                surface=surface,
                incidents=incidents,
            ),
            dependencies=self._dependency_statuses(
                health
            ),
            incidents=self._incident_summary(
                incidents
            ),
            metrics=self._operational_metrics(
                events
            ),
        )

    @staticmethod
    def _dependency_statuses(health):
        result = []

        for name, value in health.dependency_health.items():
            try:
                status = HealthStatus(
                    value.get(
                        "status",
                        HealthStatus.UNHEALTHY.value,
                    )
                )
            except ValueError:
                status = HealthStatus.UNHEALTHY

            result.append(
                DashboardDependencyStatus(
                    name=name,
                    status=status,
                    details=dict(
                        value.get(
                            "details",
                            {}
                        )
                    ),
                )
            )

        return tuple(result)

    @staticmethod
    def _incident_summary(incidents):
        return DashboardIncidentSummary(
            total=len(incidents),
            open=sum(
                incident.is_open
                for incident in incidents
            ),
            critical=sum(
                incident.severity.value == "critical"
                for incident in incidents
            ),
            high=sum(
                incident.severity.value == "high"
                for incident in incidents
            ),
            medium=sum(
                incident.severity.value == "medium"
                for incident in incidents
            ),
            low=sum(
                incident.severity.value == "low"
                for incident in incidents
            ),
            info=sum(
                incident.severity.value == "info"
                for incident in incidents
            ),
        )

    @staticmethod
    def _operational_metrics(events):
        return DashboardOperationalMetrics(
            message_failures=sum(
                event.event_type
                is OperationalEventType.MESSAGE_FAILURE
                for event in events
            ),
            rate_limit_events=sum(
                event.event_type
                is OperationalEventType.RATE_LIMIT
                for event in events
            ),
            abuse_events=sum(
                event.event_type
                is OperationalEventType.ABUSE
                for event in events
            ),
        )

    @staticmethod
    def _surface_health_status(
        *,
        health,
        surface,
        incidents,
    ):
        if any(
            incident.is_open
            and incident.severity.value
            in {"critical", "high"}
            for incident in incidents
        ):
            return HealthStatus.UNHEALTHY

        if surface is OperationalSurface.PRIVATE_CLASSROOM:
            statuses = (
                health.telegram_status,
                health.postgresql_status,
                health.llm_status,
            )
        else:
            statuses = (
                health.telegram_status,
            )

        if all(
            status is HealthStatus.HEALTHY
            for status in statuses
        ):
            return HealthStatus.HEALTHY

        return HealthStatus.UNHEALTHY

    @staticmethod
    def _calculate_health_score(
        overall_status,
        incidents,
    ) -> int:
        score = (
            100
            if overall_status is HealthStatus.HEALTHY
            else 60
        )

        critical_open = sum(
            incident.is_open
            and incident.severity.value == "critical"
            for incident in incidents
        )

        high_open = sum(
            incident.is_open
            and incident.severity.value == "high"
            for incident in incidents
        )

        score -= critical_open * 20
        score -= high_open * 10

        return max(
            0,
            min(100, score),
        )

from __future__ import annotations

from .privacy_approval_record import PrivacyApprovalRecord


class PrivacyApprovalRegistry:
    """Registry for explicit human approval/rejection evidence."""

    def __init__(
        self,
        records: list[PrivacyApprovalRecord] | None = None,
    ) -> None:
        self._records: dict[str, PrivacyApprovalRecord] = {}
        for record in records or []:
            self.register(record)

    def register(self, record: PrivacyApprovalRecord) -> None:
        if record.approval_id in self._records:
            raise ValueError(
                f"Approval record already registered: {record.approval_id}"
            )
        self._records[record.approval_id] = record

    def get(self, approval_id: str) -> PrivacyApprovalRecord:
        try:
            return self._records[approval_id]
        except KeyError as exc:
            raise KeyError(
                f"Approval record not registered: {approval_id}"
            ) from exc

    def for_case(self, case_id: str) -> tuple[PrivacyApprovalRecord, ...]:
        return tuple(
            record
            for record in self._records.values()
            if record.case_id == case_id
        )

class PrivacyChangeRegistry:
    """In-memory application registry for controlled privacy changes.

    Beta 1 intentionally keeps this registry simple. Persistence can be
    introduced behind the same contract once the workflow is proven.

    def __init__(self) -> None:
        self._changes: dict[str, PrivacyChange] = {}

    def register(self, change: PrivacyChange) -> PrivacyChange:
        if change is None:
            raise ValueError("change cannot be None.")
        if change.change_id in self._changes:
            raise ValueError(f"Privacy change already registered: {change.change_id}")
        self._changes[change.change_id] = change
        return change

    def get(self, change_id: str) -> PrivacyChange | None:
        return self._changes.get(change_id)

    def list_all(self) -> list[PrivacyChange]:
        return list(self._changes.values())

    def update(self, change: PrivacyChange) -> PrivacyChange:
        if change is None:
            raise ValueError("change cannot be None.")
        if change.change_id not in self._changes:
            raise KeyError(f"Unknown privacy change: {change.change_id}")
        self._changes[change.change_id] = change
        return change

    def mark_blocked(self, change_id: str) -> PrivacyChange:
        change = self._require(change_id)
        updated = _replace(change, status=PrivacyChangeStatus.BLOCKED)
        self._changes[change_id] = updated
        return updated

    def _require(self, change_id: str) -> PrivacyChange:
        change = self.get(change_id)
        if change is None:
            raise KeyError(f"Unknown privacy change: {change_id}")
        return change

@dataclass(slots=True)
class PrivacyGovernanceComposition:
    """Single composition root for the privacy governance capability.

    Beta 1 uses in-memory registries.

    All application services receive the same registry instances so that:

    - case creation;
    - governance decisions;
    - approval history;
    - dashboard reads;
    - API reads

    operate on one coherent runtime state.

    Email infrastructure is optional at composition-construction time.
    This is intentional:

    - unit tests must not require external email configuration;
    - application startup must not fail merely because email is disabled;
    - actual escalation requires an explicitly configured email service.

    Repository-backed implementations can replace the registries later
    without changing the application service contracts.
    """

    change_registry: PrivacyChangeRegistry
    case_registry: PrivacyGovernanceCaseRegistry
    approval_registry: PrivacyApprovalRegistry

    decision_service: PrivacyGovernanceDecisionService
    governance_gate: PrivacyGovernanceGate
    case_service: PrivacyGovernanceCaseService
    flow_service: PrivacyGovernanceFlowService
    dashboard_service: PrivacyGovernanceDashboardService


def _build_runtime_approval_email_service(
    *,
    approval_email_service: PrivacyApprovalEmailService | None,
) -> PrivacyApprovalEmailService | None:
    """Resolve the optional runtime approval-email service.

    Explicit dependency injection always wins.

    When no service is injected, the service is constructed only when
    the required runtime configuration exists.

    Missing email configuration is NOT a composition-construction error.

    This is important because the privacy governance application layer
    must remain usable without an external email provider, especially in:

    - unit tests;
    - local development;
    - provider-independent deployments;
    - environments where email escalation is intentionally disabled.

    Actual governance escalation remains protected by
    PrivacyGovernanceGate.route_and_escalate(), which requires an
    approval email service before attempting delivery.
    """

    if approval_email_service is not None:
        return approval_email_service

    recipient = os.getenv(
        "PRIVACY_APPROVAL_EMAIL_TO",
        "",
    ).strip()

    if not recipient:
        return None

    try:
        from backend.app.infrastructure.email.resend_approval_email_sender import (
            ResendApprovalEmailSender,
        )
    except ImportError:
        # Keep the application composition provider-independent.
        #
        # Infrastructure may not be installed in lightweight unit-test
        # environments.
        return None

    try:
        sender: ApprovalEmailSender = ResendApprovalEmailSender()
    except (RuntimeError, ValueError):
        # Configuration/provider availability must not prevent creation
        # of the governance application composition.
        return None

    return PrivacyApprovalEmailService(
        sender=sender,
        approval_recipient=recipient,
    )


def create_privacy_governance_composition(
    *,
    validation_service: PrivacyValidationService | None = None,
    approval_email_service: PrivacyApprovalEmailService | None = None,
) -> PrivacyGovernanceComposition:
    """Create an isolated privacy governance composition.

    This function is the primary dependency-injection boundary.

    Important design rule:

        Composition creation != external side effect.

    Creating this composition:

    - creates registries;
    - wires application services;
    - optionally wires configured infrastructure;

    but does NOT:

    - send email;
    - call Resend;
    - make governance decisions;
    - modify production systems.
    """

    # ------------------------------------------------------------------
    # Authoritative in-memory state
    # ------------------------------------------------------------------

    change_registry = PrivacyChangeRegistry()
    case_registry = PrivacyGovernanceCaseRegistry()
    approval_registry = PrivacyApprovalRegistry()

    # ------------------------------------------------------------------
    # Optional infrastructure
    # ------------------------------------------------------------------

    resolved_approval_email_service = (
        _build_runtime_approval_email_service(
            approval_email_service=approval_email_service,
        )
    )

    # ------------------------------------------------------------------
    # Decision service
    # ------------------------------------------------------------------

    decision_service = PrivacyGovernanceDecisionService(
        approval_registry=approval_registry,
        change_registry=change_registry,
        case_registry=case_registry,
    )

    # ------------------------------------------------------------------
    # Governance gate
    # ------------------------------------------------------------------

    governance_gate = PrivacyGovernanceGate(
        registry=change_registry,
        validation_service=validation_service,
        approval_email_service=resolved_approval_email_service,
    )

    # ------------------------------------------------------------------
    # Case application service
    # ------------------------------------------------------------------

    case_service = PrivacyGovernanceCaseService(
        case_registry=case_registry,
        change_registry=change_registry,
        approval_registry=approval_registry,
        decision_service=decision_service,
    )

    # ------------------------------------------------------------------
    # End-to-end governance flow
    # ------------------------------------------------------------------

    flow_service = PrivacyGovernanceFlowService(
        change_registry=change_registry,
        governance_gate=governance_gate,
        case_service=case_service,
    )

    # ------------------------------------------------------------------
    # Dashboard/read model
    # ------------------------------------------------------------------

    dashboard_service = PrivacyGovernanceDashboardService(
        case_registry=case_registry,
        approval_registry=approval_registry,
    )

    return PrivacyGovernanceComposition(
        change_registry=change_registry,
        case_registry=case_registry,
        approval_registry=approval_registry,
        decision_service=decision_service,
        governance_gate=governance_gate,
        case_service=case_service,
        flow_service=flow_service,
        dashboard_service=dashboard_service,
    )

# Runtime composition
# ----------------------------------------------------------------------

_runtime_composition: PrivacyGovernanceComposition | None = None


def get_privacy_governance_composition() -> PrivacyGovernanceComposition:
    """Return the process-wide Beta privacy governance composition.

    The same composition instance is reused during the application
