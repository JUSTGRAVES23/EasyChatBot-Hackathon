class FakeUser:

class FakeAuthenticationRepository:
    """Minimal repository contract required by authentication tests."""

    def __init__(self) -> None:
        self.created_user = None
        self.created_password = None
        self.created_verification = None
        self.email_verified = False

        self.user_by_email: FakeUser | None = None

        self.recent_password_reset_tokens = 0
        self.created_password_reset_token = None

        self.reset_password_call = None
        self.reset_password_result = True

    async def email_exists(self, email: str) -> bool:
        return False

    async def username_exists(self, username: str) -> bool:
        return False

    async def create_user(
        self,
        *,
        user_id: UUID,
        email: str,
        username: str,
    ) -> UUID:
        self.created_user = {
            "user_id": user_id,
            "email": email,
            "username": username,
        }
        return user_id

    async def create_password_credential(
        self,
        *,
        user_id: UUID,
        password_hash: str,
    ) -> None:
        self.created_password = {
            "user_id": user_id,
            "password_hash": password_hash,
        }

    async def create_email_verification(
        self,
        *,
        user_id: UUID,
        email: str,
    ) -> str:
        self.created_verification = {
            "user_id": user_id,
            "email": email,
        }
        return "raw-verification-token"

    async def get_user_by_email(
        self,
        email: str,
    ) -> FakeUser | None:
        if self.user_by_email is None:
            return None

        if self.user_by_email.email != email:
            return None

        return self.user_by_email

    async def count_recent_password_reset_tokens(
        self,
        *,
        user_id: UUID,
        since: datetime,
    ) -> int:
        return self.recent_password_reset_tokens

    async def create_password_reset_token(
        self,
        *,
        user_id: UUID,
    ) -> str:
        self.created_password_reset_token = {
            "user_id": user_id,
        }
        return "raw-password-reset-token"

    async def reset_password(
        self,
        *,
        token: str,
        password_hash: str,
    ) -> bool:
        self.reset_password_call = {
            "token": token,
            "password_hash": password_hash,
        }
        return self.reset_password_result

from __future__ import annotations

from pathlib import Path

import pytest

from app.application.identity.authentication_email_service import (
    AuthenticationEmailService,
)


class StubAuthenticationEmailSender:
    def __init__(self) -> None:
        self.calls: list[tuple[str, dict]] = []

    async def send(
        self,
        subject: str,
        content: dict,
    ) -> dict:
        self.calls.append((subject, content))
        return {
            "provider": "stub",
            "status": "accepted",
        }


def build_service(
    monkeypatch: pytest.MonkeyPatch,
    tmp_path: Path,
    *,
    privacy_url: str | None = None,
) -> tuple[AuthenticationEmailService, StubAuthenticationEmailSender]:
    monkeypatch.setattr(
        "app.application.identity.authentication_email_service."
        "_registration_email_sender",
        lambda: "EasyChatBot <noreply@example.com>",
    )

    logo_path = tmp_path / "easychatbot-logo.png"
    logo_path.write_bytes(b"fake-logo")

    sender = StubAuthenticationEmailSender()

    service = AuthenticationEmailService(
        sender,
        frontend_base_url="https://academy.example.com/",
        privacy_url=privacy_url,
        logo_path=logo_path,
    )

    return service, sender

) -> None:

from __future__ import annotations

from app.application.identity.authentication_service import (
    AuthenticationError,
    PasswordHasher,
)


def test_password_hash_round_trip() -> None:
    hasher = PasswordHasher()

    encoded = hasher.hash("A sufficiently strong password!")

    assert encoded != "A sufficiently strong password!"
    assert hasher.verify(
        "A sufficiently strong password!",
        encoded,
    )
    assert not hasher.verify(
        "wrong password",
        encoded,
    )


def test_short_password_rejected() -> None:
    hasher = PasswordHasher()

    try:
        hasher.hash("short")
    except AuthenticationError:
        pass
    else:
        raise AssertionError("Short password was accepted.")
